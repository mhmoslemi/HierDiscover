def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using a real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Given a probability vector (sum‑1) returns the exact value of
    ``evaluate_sequence``: 2·n·max(a*a)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families (all return a vector with sum = 1)
# ----------------------------------------------------------------------


def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    """w_i ∝ ((i+0.5)*(n-i-0.5))^{-p}  (p=0.5 gives arcsine)."""
    idx = np.arange(n, dtype=np.float64)
    vals = ((idx + 0.5) * (n - idx - 0.5)) ** (-p)
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Discrete symmetric Beta(p,p) on the centred grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """Discrete asymmetric Beta(p,q)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_edge_dirichlet(n: int, beta_exp: float) -> np.ndarray:
    """Edge‑biased Dirichlet draw: concentration ∝ ((i+0.5)*(n-i-0.5))^{-beta_exp}
    plus a tiny uniform offset."""
    idx = np.arange(n, dtype=np.float64)
    conc = ((idx + 0.5) * (n - idx - 0.5)) ** (-beta_exp) + 0.1
    vec = np.random.dirichlet(conc)
    return vec


def _mix_with_uniform(base: np.ndarray, eps: float) -> np.ndarray:
    """Return (1‑eps)·base + eps·uniform, normalised to sum‑1."""
    uni = _distribution_uniform(base.shape[0])
    mix = (1.0 - eps) * base + eps * uni
    mix = np.maximum(mix, 1e-15)
    mix /= mix.sum()
    return mix

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) minimising
    ``evaluate_sequence``.  The routine respects a total time budget
    (default 1000 s) and returns the best vector it has found.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0       # safety margin
    eps = 1e-15

    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1️⃣  Broad deterministic + random exploration
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 5000]          # moderate sizes – fast enough
    power_exps = [0.30, 0.40, 0.50, 0.60]      # exponent p in power‑law (arcsine =0.5)
    beta_ps = np.linspace(0.02, 0.80, 20)      # symmetric Beta(p,p)
    asym_q = 0.5                               # partner for asymmetric Beta tests
    edge_beta_exps = (0.3, 0.5, 0.7)            # Dirichlet edge‑bias
    mix_epsilons = (0.0, 0.05, 0.10, 0.15, 0.20)

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- uniform (reference) -----
        vec = _distribution_uniform(n)
        sc = _max_convolution_score(vec)
        if sc < best_score:
            best_score, best_seq, best_n = sc, vec.copy(), n
            print(f"[{time.time():.1f}] n={n} uniform      score={sc:.6f}")

        # ----- power‑law / arcsine families -----
        for p in power_exps:
            if time.time() >= deadline:
                break
            vec = _distribution_power_law(n, p)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n} power(p={p:.2f}) score={sc:.6f}")

            # mixtures with uniform
            for mix in mix_epsilons[1:]:   # skip 0.0 (already evaluated)
                vec_mix = _mix_with_uniform(vec, mix)
                sc = _max_convolution_score(vec_mix)
                if sc < best_score:
                    best_score, best_seq, best_n = sc, vec_mix.copy(), n
                    print(f"[{time.time():.1f}] n={n} power(p={p:.2f})+U(eps={mix:.2f}) "
                          f"score={sc:.6f}")

        # ----- symmetric Beta(p,p) -----
        for p in beta_ps:
            if time.time() >= deadline:
                break
            vec = _distribution_beta_sym(n, p)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n} Beta(p={p:.3f}) score={sc:.6f}")

        # ----- asymmetric Beta(p,q) and swapped -----
        for p in beta_ps:
            if time.time() >= deadline:
                break
            # left‑heavy (p small, q fixed)
            vec = _distribution_beta(n, p, asym_q)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n} Beta(p={p:.3f},q={asym_q}) score={sc:.6f}")

            # right‑heavy (p fixed, q small)
            vec = _distribution_beta(n, asym_q, p)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n} Beta(p={asym_q},q={p:.3f}) score={sc:.6f}")

        # ----- random Dirichlet draws (concentration 1) -----
        for _ in range(4):
            if time.time() >= deadline:
                break
            vec = np.random.dirichlet(np.full(n, 1.0))
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n} Dirichlet(1) score={sc:.6f}")

        # ----- edge‑biased Dirichlet draws -----
        for bex in edge_beta_exps:
            if time.time() >= deadline:
                break
            vec = _distribution_edge_dirichlet(n, bex)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n} edge‑Dirichlet(beta={bex}) score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Incorporate a previously discovered sequence if present
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score, best_seq, best_n = sc, prev.copy(), len(prev)
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Safeguard (should never trigger)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_n = 3000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Smooth‑max mirror descent refinement
    # ------------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]

    remaining = deadline - time.time()
    est_per_iter = 0.003                     # empirical estimate (seconds)
    max_iters = int(max(0, remaining / est_per_iter))
    total_iters = max(3000, min(20000, max_iters))

    alpha_start, alpha_end = 1.0, 8000.0    # temperature schedule
    lr_start, lr_end = 0.3, 0.001           # learning‑rate schedule

    for it in range(total_iters):
        if time.time() >= deadline:
            break
        t = it / total_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t

        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # smooth‑max weights
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth‑max w.r.t. a
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # exponentiated‑gradient (mirror descent) update
        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()

        if (it % max(1, total_iters // 10) == 0) or (it == total_iters - 1):
            cur_sc = _max_convolution_score(a)
            if cur_sc + 1e-12 < best_score:
                best_score, best_seq = cur_sc, a.copy()
                print(f"[{time.time():.1f}] mirror it={it}/{total_iters} new best {cur_sc:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Adam‑style projected sub‑gradient descent on the true max‑conv
    # ------------------------------------------------------------------
    a = best_seq.copy()
    velocity = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    step0 = 0.5
    sub_it = 0
    while time.time() < deadline:
        sub_it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        # active set (indices attaining the maximum within tolerance)
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam momentum update (bias‑corrected)
        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** sub_it)

        step = step0
        while True:
            a_cand = a - step * v_hat
            a_cand = _project_onto_simplex(a_cand, 1.0)
            a_cand = np.maximum(a_cand, eps)
            a_cand /= a_cand.sum()
            cand_sc = _max_convolution_score(a_cand)
            if cand_sc + 1e-12 < best_score or step < 1e-8:
                break
            step *= 0.5

        a = a_cand
        if cand_sc + 1e-12 < best_score:
            best_score, best_seq = cand_sc, a.copy()
            print(f"[{time.time():.1f}] subgrad it={sub_it} new best {cand_sc:.6f}")

        # occasional random jitter to escape shallow plateaus
        if sub_it % 800 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    # ------------------------------------------------------------------
    # 5️⃣  Simulated‑annealing / mass‑transfer polishing
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cur_score = _max_convolution_score(a)
    temperature = 0.02
    temperature_decay = 0.99995          # geometric cooling
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # locate the convolution peak and the pair (i,j) that contributes most
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.argmin(a)
        if target == source:
            # pick the second‑smallest if the absolute minimum coincides
            sorted_idx = np.argsort(a)
            target = sorted_idx[1]

        # propose moving a modest amount of mass
        max_delta = min(a[source] * 0.5, 0.03)      # keep steps small
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a_new = a.copy()
        a_new[source] -= delta
        a_new[target] += delta
        a_new = np.maximum(a_new, eps)
        a_new /= a_new.sum()

        new_score = _max_convolution_score(a_new)
        delta_score = new_score - cur_score

        # Metropolis acceptance
        if delta_score < 0 or random.random() < math.exp(-delta_score / temperature):
            a = a_new
            cur_score = new_score
            if cur_score + 1e-12 < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time():.1f}] SA it={iteration} new best {best_score:.6f}")

        # temperature schedule
        temperature *= temperature_decay
        # keep temperature from vanishing completely
        if temperature < 1e-8:
            temperature = 1e-8

        # occasional random jitter to avoid stagnation
        if iteration % 2500 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    # 6️⃣  Final reporting
    # ------------------------------------------------------------------
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
