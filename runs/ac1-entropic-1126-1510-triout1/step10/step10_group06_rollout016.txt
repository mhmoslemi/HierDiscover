"""
A refined optimizer for the step‑function “max‑convolution” problem.

The algorithm consists of three stages:

1. **Broad sweep** – evaluate a large collection of analytic families
   (uniform, arcsine, triangular, exponential, beta, sine‑power, Gaussian,
   inverse‑linear, double‑exponential, logarithmic) on several lengths and
   a few Dirichlet draws.  The best vector is kept as a seed.

2. **Heavy refinement** – a long optimisation pipeline (exponential‑gradient
   with a very slowly increasing smooth‑max parameter, projected gradient,
   Adam on the logits, and an aggressive hill‑climber) is applied to the
   current best vector.  All steps respect the simplex constraint
   (non‑negative, sum = 1).

3. **Fast restarts** – while time permits, new random Dirichlet vectors
   of various lengths are generated and a shortened version of the pipeline
   is run.  Any improvement updates the global best.

All helper functions are defined at module level (no closures).  The
objective is evaluated internally as `2·n·max(a*a)` which matches the
output of `evaluate_sequence` for a normalised vector (sum = 1).  The
function stops before the 1000 s budget expires and returns the best
sequence found as a list of floats.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

# ----------------------------------------------------------------------
# Objective -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) for a probability vector (sum = 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Weighting helpers for smooth‑max --------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max style weights approximating the max, parameterised by α."""
    maxc = conv.max()
    w = np.exp(alpha * (conv - maxc))
    s = w.sum()
    if s == 0:
        return np.full_like(w, 1.0 / w.size)
    return w / s

def _subgradient_weights(conv: np.ndarray, eps_rel: float = 1e-12) -> np.ndarray:
    """Sub‑gradient weights for the exact max (uniform over active indices)."""
    m = conv.max()
    w = np.zeros_like(conv)
    w[conv >= m - eps_rel * (abs(m) + 1.0)] = 1.0
    cnt = w.sum()
    if cnt == 0:
        return w
    return w / cnt

# ----------------------------------------------------------------------
# Gradient steps --------------------------------------------------------
def _exponential_gradient_step(a: np.ndarray, lr: float, alpha: float) -> np.ndarray:
    """Exponentiated‑gradient (EG) update on the simplex."""
    conv = _fft_self_convolve(a)
    w = _smooth_weights(conv, alpha)
    # gradient of the smooth max
    grad = 2.0 * _fft_convolve(a[::-1], w)[len(a) - 1 : 2 * len(a) - 1]
    a = a * np.exp(-lr * grad)
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def _projected_gradient_step(a: np.ndarray, eta: float, active_w: np.ndarray) -> np.ndarray:
    """Projected gradient descent step using a sub‑gradient of the exact max."""
    grad = 2.0 * _fft_convolve(a[::-1], active_w)[len(a) - 1 : 2 * len(a) - 1]
    a = a - eta * grad
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def _adam_step(logits: np.ndarray, m: np.ndarray, v: np.ndarray,
               t: int, grad: np.ndarray) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Adam update on the logits (unconstrained)."""
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    lr = 0.02 / (np.sqrt(v_hat) + eps)
    logits = logits - lr * m_hat
    logits = np.clip(logits, -30, 30)          # avoid extreme values
    return logits, m, v


# ----------------------------------------------------------------------
# Hill‑climber ---------------------------------------------------------
def _hillclimber_move(a: np.ndarray, conv: np.ndarray,
                      eps: float = 1e-12) -> tuple[np.ndarray, np.ndarray, bool]:
    """Transfer a tiny chunk of mass from a large‑gradient index to a small‑gradient one."""
    n = a.shape[0]
    w = _subgradient_weights(conv, eps_rel=eps)
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : 2 * n - 1]

    src = int(np.argmax(grad))      # biggest contributor to the max
    dst = int(np.argmin(grad))      # smallest contributor
    if src == dst:
        dst = random.randint(0, n - 1)

    max_delta = min(a[src] * 0.5, 0.02)
    delta = max_delta * random.random()
    if delta < eps:
        return a, conv, False

    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    new_conv = _fft_self_convolve(a_new)
    improved = new_conv.max() < conv.max() - 1e-13
    if improved:
        return a_new, new_conv, True
    else:
        return a, conv, False


# ----------------------------------------------------------------------
# Analytic families -----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.01) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_inverse_linear(n: int) -> np.ndarray:
    """a_i ∝ 1/(i+1)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = 1.0 / idx
    vals /= vals.sum()
    return vals


def _distribution_logarithmic(n: int) -> np.ndarray:
    """a_i ∝ -log((i+0.5)/n) (positive after shift)."""
    t = (np.arange(n, dtype=np.float64) + 0.5) / n
    vals = -np.log(t)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_double_exponential(n: int, lam: float = 0.02) -> np.ndarray:
    """Symmetric sum of two exponentials from the edges."""
    idx = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * idx) + np.exp(-lam * (n - 1 - idx))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Core optimisation routine ---------------------------------------------
def _refine_vector(a: np.ndarray,
                   best_seq: np.ndarray,
                   best_score: float,
                   best_n: int,
                   deadline: float,
                   short: bool = False) -> tuple[np.ndarray, float, int]:
    """
    Run the full optimisation pipeline on vector `a`.
    Returns possibly improved (best_seq, best_score, best_n).
    """
    n = a.shape[0]
    # ------------------------------------------------------------------
    # 1. Exponential‑gradient phase
    steps_eg = 4000 if not short else 800
    a_eg = a.copy()
    # smoothing alpha from 0.5 up to 1e6 (geometric)
    a0, a1 = 0.5, 1e6
    # learning‑rate from 0.30 down to 0.001
    lr0, lr1 = 0.30, 0.001
    for it in range(steps_eg):
        if time.time() >= deadline:
            break
        t = it / steps_eg
        alpha = a0 * (a1 / a0) ** t
        lr = lr0 * (lr1 / lr0) ** t
        a_eg = _exponential_gradient_step(a_eg, lr, alpha)

        if it % 400 == 0 or it == steps_eg - 1:
            sc = _objective(a_eg)
            if sc + 1e-13 < best_score:
                best_seq, best_score, best_n = a_eg.copy(), sc, n
                print(f"[{time.time():.1f}] EG it={it}  new best={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Projected‑gradient phase
    steps_pg = 2500 if not short else 500
    a_pg = a_eg.copy()
    eta0, eta1 = 0.50, 0.001
    for it in range(steps_pg):
        if time.time() >= deadline:
            break
        conv = _fft_self_convolve(a_pg)
        w = _subgradient_weights(conv, eps_rel=1e-12)
        eta = eta0 * (eta1 / eta0) ** (it / steps_pg)
        a_pg = _projected_gradient_step(a_pg, eta, w)

        if it % 250 == 0 or it == steps_pg - 1:
            sc = _objective(a_pg)
            if sc + 1e-13 < best_score:
                best_seq, best_score, best_n = a_pg.copy(), sc, n
                print(f"[{time.time():.1f}] PG it={it}  new best={sc:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam on logits (softmax parametrisation)
    logits = np.log(a_pg + 1e-12)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    steps_adam = 4000 if not short else 800
    for it in range(1, steps_adam + 1):
        if time.time() >= deadline:
            break
        a_adam = np.exp(logits)
        a_adam /= a_adam.sum()
        conv = _fft_self_convolve(a_adam)
        w = _subgradient_weights(conv, eps_rel=1e-12)

        # gradient w.r.t. a then map to logits (softmax Jacobian)
        grad_a = 2.0 * _fft_convolve(a_adam[::-1], w)[n - 1 : 2 * n - 1]
        grad_l = (grad_a - np.dot(a_adam, grad_a)) * a_adam
        logits, m, v = _adam_step(logits, m, v, it, grad_l)

        if it % 400 == 0 or it == steps_adam:
            a_adam = np.exp(logits)
            a_adam /= a_adam.sum()
            sc = _objective(a_adam)
            if sc + 1e-13 < best_score:
                best_seq, best_score, best_n = a_adam.copy(), sc, n
                print(f"[{time.time():.1f}] Adam it={it}  new best={sc:.6f}")

    # ------------------------------------------------------------------
    # 4. Hill‑climber (aggressive local search)
    a_hc = best_seq if best_seq.shape[0] == n else a_adam
    conv_hc = _fft_self_convolve(a_hc)
    hc_iters = 50000 if not short else 5000
    for it in range(hc_iters):
        if time.time() >= deadline:
            break
        a_hc, conv_hc, improved = _hillclimber_move(a_hc, conv_hc, eps=1e-12)
        if improved:
            sc = 2.0 * n * conv_hc.max()
            if sc + 1e-13 < best_score:
                best_seq, best_score, best_n = a_hc.copy(), sc, n
                if it % 200 == 0:
                    print(f"[{time.time():.1f}] hill it={it}  new best={sc:.6f}")

        # occasional random jitter to escape plateaus
        if it % 5000 == 0 and it > 0:
            jitter = np.random.randn(n) * 1e-5
            a_hc += jitter
            a_hc = np.maximum(a_hc, 1e-12)
            a_hc /= a_hc.sum()
            conv_hc = _fft_self_convolve(a_hc)

    return best_seq, best_score, best_n


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns the best sequence found within the allotted time.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # small safety margin
    eps = 1e-12

    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1. Global scan of analytic families and few random draws
    scan_lengths = [1500, 2000, 2500, 3000, 3500, 4000, 5000, 6000]
    for n in scan_lengths:
        if time.time() >= deadline:
            break
        cand = []

        # classic families
        cand.append(_distribution_uniform(n))
        cand.append(_distribution_arcsine(n))
        cand.append(_distribution_triangular(n))
        cand.append(_distribution_exponential(n, lam=0.004))
        cand.append(_distribution_exponential(n, lam=0.02))
        for p in (0.3, 0.6, 1.0, 2.0, 4.0):
            cand.append(_distribution_beta_sym(n, p))
        for pw in (0.5, 1.0, 2.0, 4.0):
            cand.append(_distribution_sine_power(n, pw))
        for sigma in (0.22, 0.33, 0.44):
            cand.append(_distribution_gaussian(n, sigma))

        # extra families
        cand.append(_distribution_inverse_linear(n))
        cand.append(_distribution_logarithmic(n))
        cand.append(_distribution_double_exponential(n, lam=0.015))

        # random Dirichlet draws (different concentrations)
        for alpha in (0.1, 0.3, 0.5, 0.8, 1.2):
            cand.append(np.random.dirichlet(np.full(n, alpha)))

        for vec in cand:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _objective(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] init n={n}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Pull existing global solution if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < best_score:
                best_seq = prev.copy()
                best_n = prev.size
                best_score = sc
                print(f"[{time.time():.1f}] global best {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never happen)
    if best_seq is None:
        best_n = 2500
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Interpolate the current best to a few larger lengths
    extra_lengths = [best_n + 500, best_n + 1000, best_n + 1500, best_n + 2000]
    for n in extra_lengths:
        if time.time() >= deadline:
            break
        x_old = np.linspace(0.0, 1.0, best_seq.size)
        x_new = np.linspace(0.0, 1.0, n)
        vec = np.interp(x_new, x_old, best_seq)
        vec = np.maximum(vec, eps)
        vec /= vec.sum()
        sc = _objective(vec)
        if sc < best_score:
            best_seq = vec.copy()
            best_n = n
            best_score = sc
            print(f"[{time.time():.1f}] interp n={n}  new best={sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Perform a heavy refinement on the current best vector
    best_seq, best_score, best_n = _refine_vector(best_seq,
                                                 best_seq,
                                                 best_score,
                                                 best_n,
                                                 deadline,
                                                 short=False)

    # ------------------------------------------------------------------
    # 6. Fast random restarts while time permits
    restart_lengths = [1500, 2000, 2500, 3000, 3500, 4000, 5000, 6000]
    while time.time() < deadline - 5.0:
        n = random.choice(restart_lengths)
        # Dirichlet with small concentration favours sparsity
        a0 = np.random.dirichlet(np.full(n, 0.5))
        a0 = np.maximum(a0, eps)
        a0 /= a0.sum()
        best_seq, best_score, best_n = _refine_vector(a0,
                                                     best_seq,
                                                     best_score,
                                                     best_n,
                                                     deadline,
                                                     short=True)

    # ------------------------------------------------------------------
    final_score = _objective(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
