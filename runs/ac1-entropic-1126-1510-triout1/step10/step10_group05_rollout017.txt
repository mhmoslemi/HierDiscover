"""
Propose a step‑function (list of non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

using a large pool of analytic families, a long multiplicative‑gradient
descent on the simplex, followed by pairwise Frank‑Wolfe style mass
transfer, random hill‑climbing and a final ordering sweep.  The routine
observes a 1000 s time budget, prints progress, and returns the best
vector it discovers.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
from scipy.special import i0  # Kaiser window

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution of a real vector via FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors via FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate when Σa = 1: 2·n·max(conv(a,a))."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of the max‑convolution functional on the simplex.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _gaussian_distribution(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    w /= w.sum()
    return w


def _cauchy_distribution(n: int, scale_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    scale = scale_factor * n
    w = 1.0 / (1.0 + ((idx - centre) / scale) ** 2)
    w /= w.sum()
    return w


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide='ignore', invalid='ignore'):
        w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    if np.isnan(w).any() or np.isinf(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Re‑ordering heuristics ------------------------------------------------
def order_alternate_extremes(desc_vals: np.ndarray) -> np.ndarray:
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    left, right = 0, n - 1
    toggle = True
    for v in desc_vals:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    return out


def order_center_outwards(desc_vals: np.ndarray) -> np.ndarray:
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    pos = []
    mid = (n - 1) // 2
    pos.append(mid)
    for d in range(1, n):
        l = mid - d
        if l >= 0:
            pos.append(l)
        r = mid + d
        if r < n:
            pos.append(r)
    for i, p in enumerate(pos[:n]):
        out[p] = desc_vals[i]
    return out


def order_binary_tree(desc_vals: np.ndarray) -> np.ndarray:
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    positions = []

    def recurse(l: int, r: int):
        if l > r:
            return
        m = (l + r) // 2
        positions.append(m)
        recurse(l, m - 1)
        recurse(m + 1, r)

    recurse(0, n - 1)
    out[positions] = desc_vals
    return out

def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 1_000_000,
                          eta0: float = 0.8) -> tuple[np.ndarray, float]:

    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        if it % 10 == 0:   # evaluate only every few steps
            cur_score = _max_convolution_score(a)
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_a = a.copy()
    return best_a, best_score


def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 500_000,
                  step0: float = 0.2,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _sgd_subgrad(a0: np.ndarray, deadline: float,
                 max_iters: int = 500_000,
                 step0: float = 0.5) -> tuple[np.ndarray, float]:

def _hill_climb(a: np.ndarray, deadline: float,
                max_iters: int = 300_000, epsilon: float = 0.02) -> tuple[np.ndarray, float]:
    best = a.copy()
    best_score = _max_convolution_score(best)
    n = best.size
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or best[i] < 1e-12:
            continue
        delta = min(best[i] * epsilon, 1.0 - best[j])
        if delta <= 0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = _max_convolution_score(cand)
        if sc < best_score:
            best = cand
            best_score = sc
    return best, best_score


def _frank_wolfe_opt(a0: np.ndarray, deadline: float,
                    max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _max_convolution_score(a)
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iters:

        i_src = int(np.argmax(grad))

        if delta <= 0:
            break
        step = 0.7 * delta
        a[i_src] -= step
        a[i_dst] += step
        cur_score = _max_convolution_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        if it % 4000 == 0 and np.random.rand() < 0.1:
            a = _project_onto_simplex(a + np.random.randn(a.size) * 1e-5, s=1.0)
            cur_score = _max_convolution_score(a)
    return best, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects a ~1000 s deadline and returns the best vector found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Choose a base length (use previous solution if available)
    try:
        base_n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        base_n = 5000

    # ------------------------------------------------------------------
    # Lengths to explore (keep them moderate)
    lengths = set()
    for factor in (0.8, 1.0, 1.2):
        n = int(base_n * factor)
        if 800 <= n <= 12000:
            lengths.add(n)
    for _ in range(3):
        lengths.add(np.random.randint(800, min(12000, base_n * 2) + 1))
    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # Helper to add a candidate (normalise to ℓ₁ = 1 then score)
    candidates: list[tuple[np.ndarray, float]] = []

    def _add_candidate(arr: np.ndarray):
        a = np.asarray(arr, dtype=np.float64)
        if a.ndim != 1 or a.size == 0:
            return
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            return
        a = a / s
        sc = _max_convolution_score(a)
        candidates.append((a, sc))

    # ------------------------------------------------------------------
    # 1. Broad sampling -------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # classic families
        _add_candidate(_arcsine_distribution(n))
        _add_candidate(_triangular_distribution(n))
        _add_candidate(_exponential_distribution(n, lam=0.005))
        _add_candidate(_gaussian_distribution(n, sigma_factor=0.15))
        _add_candidate(_cauchy_distribution(n, scale_factor=0.15))
        _add_candidate(_hann_window(n))
        _add_candidate(_hamming_window(n))
        _add_candidate(_blackman_window(n))
        for beta in (2.0, 4.0, 6.0, 8.0):
            _add_candidate(_kaiser_window(n, beta))

        # symmetric beta family
        for p in np.linspace(0.1, 5.0, 12):
            _add_candidate(_beta_symmetric_distribution(n, p))

        # unsymmetric beta (a few random draws)
        for _ in range(4):
            a = np.random.uniform(0.1, 5.0)
            b = np.random.uniform(0.1, 5.0)
            _add_candidate(_beta_distribution(n, a, b))

        # power‑transformed arcsine
        base = _arcsine_distribution(n)
        for pw in (0.5, 0.8, 1.2, 1.5, 2.0):
            a = np.power(base, pw)
            a /= a.sum()
            _add_candidate(a)

        # Dirichlet draws
        for alpha in (0.2, 0.5, 1.0, 2.0):
            _add_candidate(_random_dirichlet_alpha(n, alpha))

        # Re‑orderings of the *sorted* vector (helpful for many families)
        # Only once per n to keep time low
        if candidates:
            desc = np.sort(candidates[-1][0])[::-1]
            _add_candidate(order_alternate_extremes(desc))
            _add_candidate(order_center_outwards(desc))
            _add_candidate(order_binary_tree(desc))

    # ------------------------------------------------------------------
    # 2. Inject previous best if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        if prev.sum() > 0:
            prev = prev / prev.sum()
            _add_candidate(prev)
            print(f"Added previous best candidate, score={candidates[-1][1]:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Keep the best few seeds
    candidates.sort(key=lambda x: x[1])
    top_k = min(8, len(candidates))
    top_candidates = candidates[:top_k]
    print(f"Generated {len(candidates)} candidates, keeping top {top_k}.")
    print(f"Best seed score = {top_candidates[0][1]:.6f}")

    best_seq = top_candidates[0][0].copy()
    best_score = top_candidates[0][1]

    # ------------------------------------------------------------------
    # 4. Intensive refinement (exp‑gradient is the most effective)
    for idx, (seed_seq, _) in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        seq, sc = _exp_gradient_descent(seed_seq, deadline,
                                        max_iters=2_000_000,
                                        eta0=0.8)
        if sc < best_score - 1e-12:
            best_seq, best_score = seq.copy(), sc
            print(f"[{idx}] Exp‑grad improved to {best_score:.6f}")

        # A short Adam pass can sometimes break plateaus
        if time.time() < deadline:
            seq, sc = _adam_descent(best_seq, deadline,
                                    max_iters=300_000,
                                    step0=0.2)
            if sc < best_score - 1e-12:
                best_seq, best_score = seq.copy(), sc
                print(f"[{idx}] Adam improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Frank‑Wolfe style pairwise mass transfer (global polishing)
    if time.time() < deadline - 20:
        seq, sc = _frank_wolfe_opt(best_seq, deadline,
                                   max_iters=500_000)
        if sc < best_score - 1e-12:
            best_seq, best_score = seq.copy(), sc
            print(f"Frank‑Wolfe polishing improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Random hill‑climbing (fine‑grained local search)
    if time.time() < deadline - 10:
        seq, sc = _hill_climb(best_seq, deadline,
                              max_iters=300_000,
                              epsilon=0.01)
        if sc < best_score - 1e-12:
            best_seq, best_score = seq.copy(), sc
            print(f"Hill‑climb final improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final deterministic reorderings of the *sorted* best vector
    if time.time() < deadline:
        desc = np.sort(best_seq)[::-1]
        for fn in (order_alternate_extremes,
                   order_center_outwards,
                   order_binary_tree):
            cand = fn(desc)
            sc = _max_convolution_score(cand)
            if sc < best_score - 1e-12:
                best_seq, best_score = cand.copy(), sc
                print(f"Reordering {fn.__name__} improved to {best_score:.6f}")

    print(f"Final best surrogate score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
