"""Propose a step‑function sequence with a low value of
evaluate_sequence().

The optimizer works in several stages:

1. **Seed generation** – a collection of analytic families (uniform,
   arcsine, triangular, exponential, beta‑sym, sine‑power,
   Gaussian, power‑law, mixtures) and random Dirichlet draws are
   evaluated for several lengths.  The best few seeds are kept.

2. **Local refinement** – each seed is improved by a long optimisation
   pipeline:
      * Exponential‑gradient (smooth‑max) with exponentially decaying
        learning‑rate.
      * Projected sub‑gradient descent on the exact max‑convolution.
      * Adam on the soft‑max (logits) parametrisation.
      * An aggressive hill‑climber that repeatedly moves a small chunk
        of probability from the index with the largest gradient to the
        one with the smallest.

3. **Perturbation & re‑refinement** – while time permits the current
   best vector is jittered with a tiny Gaussian noise and the whole
   refinement pipeline is run again.  This helps escape shallow local
   minima.

4. **Symmetrisation & smoothing** – after the time is almost up we try
   a simple symmetrisation (average with its reversal) and a tiny
   moving‑average smoothing; whichever yields a lower objective replaces
   the best candidate.

All vectors are kept on the probability simplex (non‑negative,
sum = 1) using the Euclidean projection onto the simplex.  The objective
inside the optimiser is the scale‑invariant quantity

    2·n·max(a * a)      (a * a = full linear self‑convolution)

which matches ``evaluate_sequence`` after normalisation.

The total wall‑clock budget is respected (default 1000 s).
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a):
    """Return 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.01):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    """Discrete Gaussian centred at (n‑1)/2."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_powerlaw(n, p):
    """Power‑law a_i ∝ (i+1)^{-p} on [0,n)."""
    i = np.arange(n, dtype=np.float64) + 1.0
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _distribution_mixture_gauss(n, centers, sigma):
    """Mixture of a few Gaussian bumps."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.zeros_like(x)
    for c in centers:
        vals += np.exp(-0.5 * ((x - c) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    if vals.sum() == 0:
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _smooth_weights(conv, alpha):
    """Soft‑max style weights, approximating the smooth max."""
    maxc = conv.max()
    w = np.exp(alpha * (conv - maxc))
    s = w.sum()
    if s == 0:
        return np.full_like(w, 1.0 / w.size)
    return w / s


def _subgradient_weights(conv, eps_rel=1e-12):
    """Binary weights for a sub‑gradient of the max‑convolution."""
    m = conv.max()
    w = np.zeros_like(conv)
    w[conv >= m - eps_rel * (abs(m) + 1.0)] = 1.0
    cnt = w.sum()
    if cnt == 0:
        return w
    return w / cnt


# ----------------------------------------------------------------------
def _exponential_gradient_step(a, lr, alpha):
    """One EG step (a ≥ 0, Σa = 1)."""
    conv = _fft_self_convolve(a)
    w = _smooth_weights(conv, alpha)
    grad = 2.0 * _fft_convolve(a[::-1], w)[len(a) - 1 : len(a) - 1 + len(a)]
    a = a * np.exp(-lr * grad)
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def _projected_gradient_step(a, eta, active_w):
    """PGD step on the sub‑gradient defined by active_w."""
    grad = 2.0 * _fft_convolve(a[::-1], active_w)[len(a) - 1 : len(a) - 1 + len(a)]
    a = a - eta * grad
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def _adam_step(logits, m, v, t, step_size):
    """Adam update on logits (unconstrained)."""
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = beta1 * m + (1.0 - beta1) * step_size
    v = beta2 * v + (1.0 - beta2) * (step_size ** 2)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    lr = 0.02 / (np.sqrt(v_hat) + eps)
    logits = logits - lr * m_hat
    logits = np.clip(logits, -30, 30)
    return logits, m, v


# ----------------------------------------------------------------------
def _hillclimber_move(a, conv, eps=1e-12):
    """Move a tiny chunk from a large‑gradient index to a small‑gradient one."""
    n = a.shape[0]
    w = _subgradient_weights(conv, eps_rel=eps)
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

    src = int(np.argmax(grad))
    dst = int(np.argmin(grad))
    if src == dst:
        dst = random.randint(0, n - 1)

    # Adaptive delta – larger early, shrinking later
    max_delta = min(a[src] * 0.5, 0.03)
    delta = max_delta * random.random()
    if delta < eps:
        return a, conv, False

    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    new_conv = _fft_self_convolve(a_new)
    improved = new_conv.max() < conv.max() - 1e-13
    if improved:
        return a_new, new_conv, True
    else:
        return a, conv, False

def _refine_vector(a, n, deadline, eps=1e-12):
    """
    Run a long optimisation pipeline on a probability vector a of length n.
    Returns the (possibly) improved vector and its score.
    The global best is updated by the caller.
    """
    best_seq = a.copy()
    best_score = _max_convolution_score(best_seq)

    # ---- exponential‑gradient phase ---------------------------------
    a_eg = a.copy()
    steps_eg = 2500
    a0_alpha, a1_alpha = 1.0, 5000.0      # smoothing schedule
    lr0, lr1 = 0.18, 0.001                # learning‑rate schedule
    for it in range(steps_eg):
        if time.time() >= deadline:
            break
        t = it / steps_eg
        alpha = a0_alpha * (a1_alpha / a0_alpha) ** t
        lr = lr0 * (lr1 / lr0) ** t
        a_eg = _exponential_gradient_step(a_eg, lr, alpha)

        if it % 300 == 0 or it == steps_eg - 1:
            sc = _max_convolution_score(a_eg)
            if sc < best_score - 1e-13:
                best_seq, best_score = a_eg.copy(), sc
                print(f"[{time.time():.1f}] EG it={it}  best={sc:.6f}")

    # ---- projected‑gradient phase ----------------------------------
    a_pg = a_eg.copy()
    steps_pg = 1500
    eta0, eta1 = 0.28, 0.002
    for it in range(steps_pg):
        if time.time() >= deadline:
            break
        conv = _fft_self_convolve(a_pg)
        w = _subgradient_weights(conv, eps_rel=1e-12)
        eta = eta0 * (eta1 / eta0) ** (it / steps_pg)
        a_pg = _projected_gradient_step(a_pg, eta, w)

        if it % 200 == 0 or it == steps_pg - 1:
            sc = _max_convolution_score(a_pg)
            if sc < best_score - 1e-13:
                best_seq, best_score = a_pg.copy(), sc
                print(f"[{time.time():.1f}] PG it={it}  best={sc:.6f}")

    # ---- Adam on logits --------------------------------------------
    logits = np.log(a_pg + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    steps_adam = 2500
    for it in range(1, steps_adam + 1):
        if time.time() >= deadline:
            break
        a_adam = np.exp(logits)
        a_adam /= a_adam.sum()
        conv = _fft_self_convolve(a_adam)
        w = _subgradient_weights(conv, eps_rel=1e-12)
        grad_a = 2.0 * _fft_convolve(a_adam[::-1], w)[n - 1 : n - 1 + n]
        grad_l = (grad_a - np.dot(a_adam, grad_a)) * a_adam
        logits, m, v = _adam_step(logits, m, v, it, grad_l)

        if it % 400 == 0 or it == steps_adam:
            a_adam = np.exp(logits)
            a_adam /= a_adam.sum()
            sc = _max_convolution_score(a_adam)
            if sc < best_score - 1e-13:
                best_seq, best_score = a_adam.copy(), sc
                print(f"[{time.time():.1f}] Adam it={it}  best={sc:.6f}")

    # ---- aggressive hill‑climber ------------------------------------
    a_hc = best_seq.copy()
    conv = _fft_self_convolve(a_hc)
    hc_iters = 40000
    for it in range(hc_iters):
        if time.time() >= deadline:
            break
        a_hc, conv, improved = _hillclimber_move(a_hc, conv, eps=eps)
        if improved:
            sc = 2.0 * n * conv.max()
            if sc < best_score - 1e-13:
                best_seq, best_score = a_hc.copy(), sc
                if it % 500 == 0:
                    print(f"[{time.time():.1f}] hill it={it}  best={sc:.6f}")

        # occasional jitter to escape plateaus
        if it % 5000 == 0 and it > 0:
            jitter = np.random.randn(n) * 1e-5
            a_hc = np.maximum(a_hc + jitter, eps)
            a_hc /= a_hc.sum()
            conv = _fft_self_convolve(a_hc)

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found within the allotted time.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1. Load previous global solution if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_seq, best_score, best_n = prev.copy(), sc, prev.size
                print(f"[{time.time():.1f}] global best {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2. Scan a variety of analytic families and random Dirichlet draws
    lengths_to_try = [2000, 2500, 3000, 3500, 4000, 4500, 5000, 6000]
    candidate_pool = []  # keep up to 8 best seeds for refinement

    for n in lengths_to_try:
        if time.time() >= deadline:
            break
        families = []

        families.append(_distribution_uniform(n))
        families.append(_distribution_arcsine(n))
        families.append(_distribution_triangular(n))
        families.append(_distribution_exponential(n, lam=0.008))
        families.append(_distribution_exponential(n, lam=0.02))
        families.append(_distribution_gaussian(n, sigma=0.3))
        families.append(_distribution_gaussian(n, sigma=0.45))

        for p in (0.3, 0.6, 1.0, 2.0, 4.0):
            families.append(_distribution_beta_sym(n, p))

        for pw in (0.5, 1.0, 2.0, 4.0):
            families.append(_distribution_sine_power(n, pw))

        for pw in (0.5, 1.0, 2.0, 4.0):
            families.append(_distribution_powerlaw(n, pw))

        # a couple of two‑bump mixtures
        for sigma in (0.15, 0.25):
            centers = [int(0.25 * n), int(0.75 * n)]
            families.append(_distribution_mixture_gauss(n, centers, sigma))
            centers = [int(0.33 * n), int(0.66 * n)]
            families.append(_distribution_mixture_gauss(n, centers, sigma))

        # random Dirichlet draws with various concentrations
        for alpha in (0.1, 0.3, 0.6, 1.0, 2.0):
            families.append(np.random.dirichlet(np.full(n, alpha)))

        # evaluate all
        for vec in families:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq, best_score, best_n = vec.copy(), sc, n
                print(f"[{time.time():.1f}] init n={n}  score={sc:.6f}")
            # keep a short list of promising seeds
            if len(candidate_pool) < 8:
                candidate_pool.append((vec.copy(), n, sc))
            else:
                # replace worst if current is better
                worst_idx = max(range(len(candidate_pool)),
                               key=lambda i: candidate_pool[i][2])
                if sc < candidate_pool[worst_idx][2]:
                    candidate_pool[worst_idx] = (vec.copy(), n, sc)

    # ------------------------------------------------------------------
    # If nothing was found (unlikely) fallback to a uniform vector
    if best_seq is None:
        best_n = 2500
        best_seq = _distribution_uniform(best_n)
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Refine the top seeds
    # sort candidate pool by score (ascending)
    candidate_pool.sort(key=lambda x: x[2])
    top_candidates = candidate_pool[:4]  # keep a few for deep refinement

    for vec, n, sc in top_candidates:
        if time.time() >= deadline:
            break
        vec_ref, sc_ref = _refine_vector(vec, n, deadline, eps=eps)
        if sc_ref < best_score - 1e-13:
            best_seq, best_score, best_n = vec_ref.copy(), sc_ref, n
            print(f"[{time.time():.1f}] refined seed n={n}  new best={sc_ref:.6f}")

    # ------------------------------------------------------------------
    # 4. Perturbation / re‑refinement loop while time permits
    while time.time() < deadline - 5.0:
        # jitter the current best a little and try to improve it
        n = best_n
        a0 = best_seq.copy()
        jitter = np.random.randn(n) * 5e-5
        a0 += jitter
        a0 = np.maximum(a0, eps)
        a0 /= a0.sum()
        # run a *shorter* refinement (fewer iterations) to keep the loop fast
        a_ref, sc_ref = _refine_vector(a0, n, deadline, eps=eps)
        if sc_ref < best_score - 1e-13:
            best_seq, best_score, best_n = a_ref.copy(), sc_ref, n
            print(f"[{time.time():.1f}] perturb‑refine n={n}  new best={sc_ref:.6f}")

    # ------------------------------------------------------------------
    # 5. Simple post‑processing tricks (symmetrisation & tiny smoothing)
    #    These are cheap and sometimes lower the max‑convolution.
    a_sym = (best_seq + best_seq[::-1]) * 0.5
    a_sym = np.maximum(a_sym, eps)
    a_sym /= a_sym.sum()
    sc_sym = _max_convolution_score(a_sym)
    if sc_sym < best_score - 1e-13:
        best_seq, best_score = a_sym, sc_sym
        print(f"[{time.time():.1f}] symmetrised improvement {sc_sym:.6f}")

    # tiny moving‑average smoothing (width 3)
    kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
    a_sm = np.convolve(best_seq, kernel, mode='same')
    a_sm = np.maximum(a_sm, eps)
    a_sm /= a_sm.sum()
    sc_sm = _max_convolution_score(a_sm)
    if sc_sm < best_score - 1e-13:
        best_seq, best_score = a_sm, sc_sm
        print(f"[{time.time():.1f}] smoothing improvement {sc_sm:.6f}")

    # ------------------------------------------------------------------
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
