def propose_candidate(seed=..., budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # Determine lengths
    base_n = 4000
    lengths = [int(base_n*fac) for fac in [0.75,1.0,1.25,1.5,2.0] if 1000 <= fac*base_n <= 25000]
    # plus some random lengths
    for _ in range(5):
        lengths.append(np.random.randint(1500, min(25000,base_n*2)+1))
    lengths = sorted(set(lengths))

    # Predefine exponent parameter ranges:
    edge_exponents = np.arange(0.2, 1.6, 0.1)   # inclusive
    power_exponents = np.arange(0.2, 2.1, 0.2)
    beta_ps = np.arange(0.2, 2.1, 0.2)
    lam_vals = [0.001,0.003,0.005,0.008,0.01,0.02]
    # More.

    candidates = []
    # For each length, generate base families:
    for n in lengths:
        base_fams = []
        # uniform
        uniform = np.full(n, 1.0/n)
        base_fams.append(('uniform', uniform))

        # arcsine
        arcsine = _arcsine_distribution(n)
        base_fams.append(('arcsine', arcsine))

        # triangular
        tri = _triangular_distribution(n)
        base_fams.append(('triangular', tri))

        # add edge powers
        for s in edge_exponents:
            e = _edge_power_distribution(n,s)
            base_fams.append((f'edge_{s:.2f}', e))

        # power law decreasing (i^(-p))
        for p in power_exponents:
            pw = _power_law_distribution(n,p)
            base_fams.append((f'power_{p:.2f}', pw))

        # exponential tails
        for lam in lam_vals:
            expd = _exponential_distribution(n, lam)
            base_fams.append((f'exp_{lam:.3f}', expd))

        # also double bump
        for bf in (0.15,0.20,0.25):
            for gf in (0.05,0.10,0.15):
                db = _double_bump_distribution(n, bf, gf)
                base_fams.append((f'db_{bf}_{gf}', db))

        # Compute score for each base; add to candidates
        for name, a in base_fams:
            score = _max_convolution_score(a)
            candidates.append((a.copy(), score, name))

        # Random mixtures: 200 combos each.
        num_mixtures = 200
        for _ in range(num_mixtures):
            # pick two distinct families
            f1, f2 = np.random.choice(base_fams, size=2, replace=False)
            w = np.random.rand()

"""
Propose a new step‐function sequence.

The algorithm proceeds in three stages.

1. **Rich candidate generation** –  a modest set of lengths (≈2000–20000)
   is chosen.  For each length a library of analytic families
   (arcsine, edge‑power, power‑law, exponential, triangular, uniform,
   sinusoidal and a few double‑bump shapes) is built.  Random convex
   mixtures of two families and Dirichlet draws are then added.  All
   candidates are normalised to sum = 1 and scored with the exact
   surrogate objective 2·n·max(conv(a,a)).  The best ~30 seeds are kept.

2. **Local refinement** – each seed undergoes two first‑order methods
   on the probability simplex:
   * projected Adam,
   * exponentiated‑gradient (KL‑mirror) descent.
   The best point seen by either method replaces the current global
   best.

3. **Polishing** – a Frank–Wolfe style mass‑transfer step and a
   simulated‑annealing jitter phase are run until the 1000 s budget is
   exhausted.

The routine returns a plain list of floats; the external
``evaluate_sequence`` will clip, renormalise and compute the final
score.  The whole search fits comfortably into the 1000 s limit.
"""

import time

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate objective for a with Σa = 1: 2·n·max(conv(a,a))."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) under Σa = 1.
    Returns (gradient, cur_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # gradient of the (non‑smooth) max‑convolution:
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for i=j terms (when k even)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.30,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 2_000_000):
    """Adam on the simplex (projected). Returns best point seen."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          eta0: float = 0.9,
                          max_iters: int = 1_500_000):
    """Exponentiated‑gradient (KL‑mirror) descent on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score

def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with concentration ``alpha``."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _edge_power_distribution(n: int, s: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = (i * (n - i + 1)) ** (-s)
    vals /= vals.sum()
    return vals


def _double_bump_distribution(n: int,
                              block_frac: float = 0.25,
                              gap_frac: float = 0.1) -> np.ndarray:
    total_block = max(2, int(block_frac * n))
    m = total_block // 2
    g = int(gap_frac * n)
    a = np.zeros(n, dtype=np.float64)
    a[:m] = 1.0 / (2 * m)
    start2 = m + g
    if start2 + m > n:
        start2 = n - m
    a[start2:start2 + m] = 1.0 / (2 * m)
    return a


def _sinusoidal_distribution(n: int, power: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * i / n) ** power
    if vals.sum() == 0:
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _raised_cosine_distribution(n: int, power: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = (1.0 - np.cos(2.0 * np.pi * i / n)) ** power
    if vals.sum() == 0:
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals

        if fw_iter % 8000 == 0 and np.random.rand() < 0.14:
            jitter = np.random.randn(a.size) * 4e-5
            a = _project_onto_simplex(a + jitter, s=1.0)
            cur_score = _max_convolution_score(a)

    # --------------------------------------------------------------
    # Simulated‑annealing / jitter phase (final push)
    a = best_seq.copy()
    cur_score = best_score
    temp_start = 0.030
    temp_end = 0.0015
    anneal_iter = 0
    while time.time() < deadline and anneal_iter < 300_000:
        anneal_iter += 1
        # linear temperature schedule based on remaining fraction
        t_rem = deadline - time.time()
        fraction = max(0.0, t_rem / (budget_s - 0.5))
        temp = temp_end + (temp_start - temp_end) * fraction

        i = np.random.randint(a.size)
        j = np.random.randint(a.size - 1)
        if j >= i:
            j += 1
        max_delta = min(a[i], 1.0 - a[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * temp
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if cur_score < best_score - 1e-14:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time():.1f}] Anneal iter={anneal_iter} new best {best_score:.6f}")
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_score - cur_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                a = a_new
                cur_score = new_score

    # --------------------------------------------------------------
    print(f"Final best surrogate score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
