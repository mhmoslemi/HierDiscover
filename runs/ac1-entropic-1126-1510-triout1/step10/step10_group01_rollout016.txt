"""
Hybrid search for the step‑function attack (improved version).

The algorithm consists of three phases:

1. **Wide seed exploration** – generate many analytic families (uniform,
   arcsine, Beta‑U‑shaped, asymmetric Beta, power‑law, exponential,
   two‑block spikes, sparse, Dirichlet).  Seeds are evaluated with the
   exact objective `2·n·max(conv)/sum^2` (with `sum=1`).  The best few
   seeds are kept for refinement.

2. **Smooth sub‑gradient descent** – starting from each selected seed we
   perform exponentiated‑gradient descent on the simplex.  Instead of a
   plain uniform weight over the active convolution indices we use a
   *soft‑max* weighting (`w_i ∝ exp(β·(conv_i−max))`).  The temperature
   `β` is increased gradually, turning the soft‑max into a uniform
   active‑set weight as the optimization proceeds.  A diminishing
   stepsize `η_t = lr0/√t` is used, together with occasional random
   perturbations to escape plateaus.

3. **Greedy local polishing** – a cheap pairwise‑mass‑transfer walk
   (random “swap” of a small amount of probability mass) is applied
   until the time budget expires.

All phases respect a global deadline (≈ 1000 s).  The best sequence
found is returned as a plain list of Python floats, ready for
`evaluate_sequence`.
"""

import time
import math

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (assumes sum≈1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic families ------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """U‑shaped Beta(p,p) on [0,1] (p∈(0,1] gives heavier edges as p↓)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """General Beta(α,β) on [0,1]."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Few non‑zero entries placed at random positions."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals

def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Broad pool of seeds for a given length."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # Symmetric Beta (U‑shaped) – sweep many p values
    for p in np.logspace(-3, 0, num=15, base=10.0):   # 0.001 … 1
        cand.append(_distribution_beta_sym(n, float(p)))

    # Asymmetric Beta – random draws
    for _ in range(8):
        alpha = 10 ** rng.uniform(-3, 2)   # 1e‑3 … 1e2
        beta = 10 ** rng.uniform(-3, 2)
        cand.append(_distribution_beta_asym(n, float(alpha), float(beta)))

    # Power‑law growth
    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))

    # Exponential growth
    for lam in (0.001, 0.005, 0.01, 0.02, 0.05):
        cand.append(_distribution_exponential_increasing(n, lam))

    # Two‑block spikes
    for block_frac in (0.02, 0.05, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac))

    # Dirichlet samples – various concentration parameters
    for a in (0.1, 0.3, 0.5, 0.8, 2.0):
        cand.append(_distribution_dirichlet(n, rng, alpha=float(a)))

    # Sparse random vectors
    cand.append(_distribution_random_sparse(n, rng, density=0.005))
    cand.append(_distribution_random_sparse(n, rng, density=0.015))

    return cand

def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the sub‑gradient 2·a⋆w where w (length 2n‑1) is a weight
    vector (non‑negative, sum 1).  Uses FFT based cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reversed a
    size = 1 << ((n + len(w) - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1                       # position where shift 0 begins
    grad = 2.0 * conv[start:start + n]
    return grad

def _smooth_subgradient_descent(a: np.ndarray, deadline: float,
                               best_seq: np.ndarray, best_score: float,
                               rng: np.random.Generator,
                               lr0: float = 0.6,
                               beta0: float = 0.5,
                               beta_mul: float = 1.02,
                               beta_max: float = 200.0,
                               max_iter: int = 300000,
                               eval_interval: int = 200,
                               beta_update_interval: int = 5000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent where the sub‑gradient uses a soft‑max
    weighting over all convolution indices (the weights approximate the
    active‑set sub‑gradient).  The temperature β grows slowly, turning the
    soft‑max into a uniform active‑set weight.
    """
    n = a.shape[0]
    beta = beta0
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()

        # soft‑max weights (higher weight for entries close to the maximum)
        diff = conv - max_val                    # ≤ 0
        w = np.exp(beta * diff)                  # 0 < w_i ≤ 1, max‑entry = 1
        w_sum = w.sum()
        if w_sum == 0.0:
            # fallback – pick the true maximiser(s)
            mask = conv >= max_val - 1e-12
            w = mask.astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        grad = _gradient_from_weights(a, w)

        # stepsize schedule η_t = lr0 / √t
        eta = lr0 / math.sqrt(t)

        # mirror‑descent (exponentiated‑gradient) step
        a = a * np.exp(-eta * grad)

        # Renormalise onto the simplex (keeps sum = 1)
        s = a.sum()
        if s <= 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # temperature schedule
        if t % beta_update_interval == 0:
            beta = min(beta * beta_mul, beta_max)

        # occasional exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth‑GD it={t} β={beta:.2f} new best {best_score:.6f}")

        # occasional random jitter to escape plateaus
        if t % (eval_interval * 5) == 0:
            noise = rng.normal(scale=0.001, size=n)
            a = np.maximum(a + noise, 0.0)
            a_sum = a.sum()
            if a_sum > 0.0:
                a = a / a_sum
            else:

    return best_seq, best_score

def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer walk that accepts a move only if it improves the true objective."""
    n = a.shape[0]
    while time.time() < deadline:
        # pick a source index with non‑trivial mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        # pick a distinct destination index
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move a random fraction of the source mass
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score
