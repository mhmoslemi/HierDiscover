    conv = _fft_self_convolve(a)

        return np.zeros_like(a)

The code below implements a new search strategy that improves on the previous approach.

* **Rich initial pool** – we generate many analytic families (uniform, arcsine, symmetric‑β, edge‑weighted) for several lengths, also random Dirichlet and sparse vectors and keep the best few.
* **Smooth Lp‑annealing** – instead of pure sub‑gradient on the max‑convolution we minimise the \(p\)-norm of the self‑convolution  
  \[
      \|a*a\|_p = \bigl(\sum_i (a*a)_i^p\bigr)^{1/p},
  \]  
  whose gradient can be computed by a single FFT‑based convolution.  
  Starting with a small \(p\) (smooth landscape) and gradually increasing it (to \(p=32\)) gives a powerful “annealing” that escapes shallow local minima.
* **Adam optimiser** – we apply the Adam adaptive‑step method on the smoothed objective, projecting after every step onto the probability simplex (so the vector stays non‑negative and sums to 1).
* **Final greedy polishing** – a cheap pairwise mass‑transfer walk refines the best vector found.
* **Time‑aware** – the whole routine respects the 1000 s budget, splitting the remaining time among the top seeds.

The algorithm is fully deterministic given the seed, uses only `numpy` and the provided `evaluate_sequence`, and returns a Python list of non‑negative heights.

Propose a step‑function (non‑negative vector) that minimises evaluate_sequence.

Algorithm
---------
1. Build a diverse pool of analytic and random candidates for several
   lengths (up to ~30 k). Keep the best few.
2. Refine the top seeds with an Adam optimiser that minimises the

   gradient of the Lp‑norm is obtained via a single FFT‑based convolution.
   After each update the iterate is projected onto the simplex
   (non‑negative, sum = 1).
3. Run a cheap greedy pairwise mass‑transfer walk that accepts only

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolution utilities
def _fft_convolve(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Full linear convolution of u and v using a real FFT."""
    size = 1 << ((len(u) + len(v) - 1) - 1).bit_length()
    U = np.fft.rfft(u, n=size)
    V = np.fft.rfft(v, n=size)
    conv = np.fft.irfft(U * V, n=size)
    return conv[: len(u) + len(v) - 1]


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
# Exact objective (max‑convolution) – used for reporting
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.02
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Gradient of the Lp‑norm of the convolution
def _grad_lp(a: np.ndarray, p: float) -> np.ndarray:
    """Gradient of (∑(a*a)^p)^{1/p} w.r.t. a (a lives on the simplex)."""
    if p <= 1.0:
        return np.zeros_like(a)
    conv = _fft_self_convolve(a)
    # protect against underflow
    conv = np.maximum(conv, 1e-30)
    # weights w_i = conv_i^{p‑1}
    w = np.power(conv, p - 1, out=np.empty_like(conv))
    # convolution w * a_rev
    a_rev = a[::-1]
    conv_w = _fft_convolve(w, a_rev)            # length 3·n‑2
    n = a.shape[0]
    grad = 2.0 * conv_w[n - 1 : n - 1 + n]      # slice of length n
    # scaling factor based on the Lp‑norm
    norm_p = np.linalg.norm(conv, ord=p)
    if norm_p == 0.0 or not np.isfinite(norm_p):
        return np.zeros_like(a)
    factor = p / (norm_p ** (p - 1))
    grad *= factor
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed Lp‑objective
def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    p_schedule: list[float],
    lr: float = 0.05,
    max_iter_per_p: int = 400,
) -> tuple[np.ndarray, float]:
    """Adam on the Lp‑norm objective with simplex projection."""
    a = a0.copy()
    n = a.shape[0]
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = a.copy()
    best_score = _objective(a)

    for p in p_schedule:
        for _ in range(max_iter_per_p):
            if time.time() >= deadline:
                return best_a, best_score
            t += 1
            grad = _grad_lp(a, p)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            # projection onto the simplex (keeps sum = 1)
            a = _project_onto_simplex(a, 1.0)

            if t % 100 == 0:
                cur = _objective(a)
                if cur < best_score:
                    best_score = cur
                    best_a = a.copy()
                    print(
                        f"[{time.time():.1f}] Adam p={p:.1f} iter={t} new best {best_score:.6f}"
                    )
    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy pairwise mass transfer (local refinement)
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small amount of mass; keep move only if objective improves."""
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises evaluate_sequence.
    The routine respects a ~1000 s wall‑time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # small safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Assemble a rich pool of analytic and random vectors
    lengths = [2000, 4000, 8000, 16000, 32000]          # moderate to large
    p_vals = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]      # parameters for beta / edge families
    lam_vals = [0.0, 0.25, 0.5, 0.75, 1.0]              # mixing with uniform

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic bases
        uniform = _distribution_uniform(n)
        arcsine = _distribution_arcsine(n)

        for base in (uniform, arcsine):
            sc = _objective(base)
            top_candidates.append((sc, base.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = base.copy()
                print(f"[{time.time():.1f}] new global best len={n} (base) score={sc:.6f}")

        # parametric families + mixture with uniform
        for p in p_vals:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            for base in (beta, edge):
                for lam in lam_vals:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # random Dirichlet seeds
        for _ in range(8):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] new global best Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(8):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.02)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}")

        # keep a modest number of elite seeds
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:25]

    # ------------------------------------------------------------------
    # 2.  Bring in any previously discovered construction (if present)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback in the unlikely case we have no candidate
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Adam‑based Lp‑annealing on a handful of elite seeds
    remaining = deadline - time.time()
    if remaining > 30 and top_candidates:
        # pick the five currently best seeds
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        p_schedule = [2, 4, 8, 16, 32]          # smooth → sharp
        print(f"[{time.time():.1f}] Adam refinement on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            a_opt, sc_opt = _adam_optimize(
                seed_seq.copy(),
                sub_dead,
                p_schedule,
                lr=0.05,
                max_iter_per_p=400,
            )
            if sc_opt < overall_best_score:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] after Adam seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Final greedy polishing (pairwise swaps)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy refinement")
        a_opt, sc_opt = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq.copy(),
            best_score=overall_best_score,
            max_trials=500_000,
        )
        if sc_opt < overall_best_score:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain list – evaluate_sequence will apply its own clipping & normalisation.
    return overall_best_seq.tolist()
