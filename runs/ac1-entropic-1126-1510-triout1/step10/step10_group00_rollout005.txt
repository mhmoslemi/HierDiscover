Propose a non‑negative step function that (empirically) lowers the
evaluate_sequence objective below the previous 1.509957 limit.

Algorithm
---------
1. Enumerate a modest set of analytic families (uniform, arcsine,
   symmetric beta, U‑shaped “edge‑weighted”, and a two‑parameter
   custom family).  For each family we sweep a fine grid of shape
   parameters (including p up to 10) and also form convex mixtures
   with the uniform distribution.  Several random Dirichlet and
   sparse seeds are added.  All vectors are normalised to sum = 1
   and the exact objective `2·n·max(conv(a,a))` is evaluated.  The
   best ≈30 candidates are retained.

2. For each retained seed we run a *smooth* gradient descent on the
   true objective.  The non‑smooth max is replaced by a
   log‑sum‑exp (soft‑max) with a temperature β that is slowly
   increased (β starts at 0.2 and is multiplied by 1.002 each
   iteration, capped at 15).  The gradient of the soft‑max is
   obtained by a single FFT‑based correlation, and the parameters
   are updated with the Adam optimiser.  After each update the
   iterate is projected back onto the probability simplex.

3. Periodically the current best objective is recorded.  After the
   Adam phase we run a cheap greedy pairwise mass‑transfer walk to
   polish the solution.

The whole search respects the supplied time budget (default 1000 s)
and returns a plain Python list; `evaluate_sequence` will perform the
final normalisation and clipping.

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    # size must accommodate length 2n‑1
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_cross_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Correlation g_i = Σ_j a_j * w_{i+j} for i = 0…len(a)-1.
    Implemented via FFT: g = ifft(fft(a) * conj(fft(w))).
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    size = 1 << ((n + m - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w, n=size)
    corr = np.fft.irfft(A * np.conj(W), n=size)
    return corr[:n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_custom(n: int, p: float, q: float) -> np.ndarray:
    """
    Two‑parameter family:
        w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p) * (i+0.5 + n‑i‑0.5)^(−q)
    Includes the edge‑weighted family when q = 0.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p) * (i + j) ** (-q)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Adam optimisation on the smoothed max‑convolution objective ----------
def _adam_smooth_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr: float = 0.12,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    beta_start: float = 0.2,
    beta_max: float = 15.0,
    beta_mul: float = 1.002,
    eval_every: int = 120,
) -> Tuple[np.ndarray, float]:
    """Projected Adam on a soft‑max approximation of max(conv(a,a))."""
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta_temp = beta_start
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)                     # length 2n‑1
        # soft‑max weights (numerically stable)
        conv_shift = conv - conv.max()
        exp_weights = np.exp(beta_temp * conv_shift)
        w = exp_weights / exp_weights.sum()             # size 2n‑1, sums to 1

        # gradient = 2 * correlation(a, w)
        grad = 2.0 * _fft_cross_corr(a, w)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = -lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a, s=1.0)

        # temperature schedule
        beta_temp = min(beta_temp * beta_mul, beta_max)

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} β={beta_temp:.3f} new best {best_score:.6f}")

        # safety stop
        if t >= 300000:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise mass transfer (same as previous code) ---------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> Tuple[np.ndarray, float]:
    """Randomly relocate a small amount of mass; keep improvement."""
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative vector that minimises `evaluate_sequence`.
    The routine builds a diverse pool of analytic seeds, refines the
    most promising ones with a smoothed Adam optimisation, and finally
    applies a cheap greedy local walk.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # margin for tidy return

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: List[Tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Generate analytic seeds
    lengths = [2000, 4000, 6000, 8000, 12000]
    # p values covering a wide range (including >1)
    p_grid = np.concatenate(
        [
            np.logspace(-2, -1, 6),   # 0.01 … 0.1
            np.logspace(-1, 0, 8),    # 0.1 … 1
            np.logspace(0, 1, 6),     # 1 … 10
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 5)   # 0, .25, .5, .75, 1

    # a couple of (p,q) pairs for the custom family
    q_grid = [0.0, 0.2, 0.5]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        base_uniform = _distribution_uniform(n)
        base_arcsine = _distribution_arcsine(n)

        for base in (base_uniform, base_arcsine):
            sc = _objective(base)
            top_candidates.append((sc, base.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = base.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # parametrised families
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)

            for base in (beta, edge):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

            # custom two‑parameter family (p,q)
            for q in q_grid:
                if time.time() >= deadline:
                    break
                custom = _distribution_custom(n, p, q)
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(custom, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} q={q:.2g} lam={lam:.2f} score={sc:.6f}"
                        )

        # random Dirichlet seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.02, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] new global best random Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}")

        # prune to keep only the best few
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2. Incorporate any previously found solution (if the variable exists)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback – should never be needed
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Adam+soft‑max refinement on the few most promising seeds
    remaining = deadline - time.time()
    if remaining > 25.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] starting Adam refinement on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            a_opt, sc_opt = _adam_smooth_descent(
                seed_seq.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr=0.12,
                eval_every=150,
            )
            if sc_opt < overall_best_score:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] after Adam seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5. Final greedy local refinements
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy swaps")
        a_opt, sc_opt = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return plain list for the external evaluator.
    return overall_best_seq.tolist()
