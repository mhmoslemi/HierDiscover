Algorithm

   exponentiated‑gradient updates on the smoothed max
   `f_α(a)=log Σ_k exp(α·(a*a)_k)`.  The smoothing parameter α grows

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Fast evaluation of 2·n·max(a*a) for a vector normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * float(conv.max())

# Analytic families – each returns a *probability* vector (sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.01) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * (np.log(x) + np.log(1 - x))
    w = np.exp(logw)
    if w.sum() == 0 or np.any(np.isinf(w)):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals /= vals.sum()
    return vals


def _distribution_power_edge(n: int, alpha: float) -> np.ndarray:
    """
    Edge‑heavy family:
        w_i ∝ ((i+0.5)*(n‑i‑0.5))^{-α}
    """
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - i - 0.5)
    eps = 1e-12
    vals = np.power(denom + eps, -alpha)
    vals /= vals.sum()
    return vals


def _distribution_V_shape(n: int) -> np.ndarray:
    centre = (n - 1) / 2.0
    vals = np.abs(np.arange(n, dtype=np.float64) - centre)
    eps = 1e-12
    vals = np.maximum(vals, eps)
    vals /= vals.sum()
    return vals


def _distribution_power_abs(n: int, beta: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    vals = np.abs(np.arange(n, dtype=np.float64) - centre) ** beta

    vals = np.maximum(vals, eps)
    vals /= vals.sum()

    # ---------- fallback ----------
    if best_seq is None:
        best_n = 3000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3. smooth‑max mirror descent
    a = best_seq.copy()
    n = a.shape[0]

    total_steps = 25000
    for it in range(total_steps):
        if time.time() >= deadline:
            break
        t = it / total_steps
        # smoothing parameter rises to ≈500
        alpha = 0.1 * (5000.0 ** t)
        # learning rate decays from 10 to 0.01
        lr = 10.0 * (0.001 ** t)

        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # exponential weights (always ≤1, never overflow)
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of Σ w_k (a*a)_k  is 2·a * w  (convolution)
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]
        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()

        if (it % 1000 == 0) or (it == total_steps - 1):
            sc = _max_convolution_score(a)
            if sc < best_score - 1e-12:
                best_seq, best_score = a.copy(), sc
                print(f"[{time.time():.1f}] mirror it={it}   best={sc:.6f}")

    # --------------------------------------------------------------
    # 4. sub‑gradient flattening (very cheap per iteration)
    a = best_seq.copy()
    cur_score = best_score
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        k_max = int(conv.argmax())
        # gradient for the max entry only
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        g[idx] = 2.0 * a[k_max - idx]

        # diminishing step size η = 0.5 / sqrt(t+1)
        eta = 0.5 / math.sqrt(it + 1.0)
        a = a - eta * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # occasional cheap evaluation
        if it % 2000 == 0:
            sc = _max_convolution_score(a)
            if sc < cur_score - 1e-12:
                cur_score = sc
                if sc < best_score - 1e-12:
                    best_seq, best_score = a.copy(), sc
                    print(f"[{time.time():.1f}] subgrad it={it}   best={sc:.6f}")

    # --------------------------------------------------------------
    # 5. final targeted hill‑climber (greedy mass transfers)
    a = best_seq.copy()
    cur_score = best_score
    hill_it = 0
    while time.time() < deadline:
        hill_it += 1
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        products = a[idx] * a[k - idx]
        i_star = i_low + int(products.argmax())
        j_star = k - i_star
        source = i_star if a[i_star] >= a[j_star] else j_star
        target = random.randint(0, n - 1)
        if target == source:
            continue
        delta_max = min(a[source] * 0.4, 0.015)
        delta = delta_max * random.random()
        if delta < eps:
            continue
        a_new = a.copy()
        a_new[source] -= delta
        a_new[target] += delta
        a_new = np.maximum(a_new, eps)
        a_new /= a_new.sum()
        new_score = _max_convolution_score(a_new)
        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_seq, best_score = a.copy(), new_score
                print(f"[{time.time():.1f}] hill it={hill_it}   best={new_score:.6f}")

        # tiny jitter every 5000 iterations to escape flats
        if hill_it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()

    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
