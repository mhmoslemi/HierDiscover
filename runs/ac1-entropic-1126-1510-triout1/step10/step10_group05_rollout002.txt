def _exp_gradient_descent_reorder(a0, deadline, max_iters=..., eta0=0.8):
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        # renormalize
        s = a.sum()
        if s <= 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        # reorder
        desc = np.sort(a)[::-1]
        a = order_alternate_extremes(desc)  # reorder to put big at ends
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

"""
Propose a non‑negative step function (list of heights) that minimizes

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm proceeds in three stages:

1️⃣  Broad sampling – generate many analytic families (generalised arcsine,
    end‑concentrated, double‑exponential, beta, Dirichlet) for several
    lengths.  Every vector is ℓ₁‑normalised and reordered with
    ``order_alternate_extremes`` (largest values at the two ends) which is
    known to reduce the peak of the self‑convolution.

2️⃣  Intensive refinement – the best few seeds are polished with long runs
    of exponentiated‑gradient, Adam and SGD descent.  After every
    ``reorder_every`` iterations the vector is re‑ordered again (large
    coefficients stay at the ends).  This simple “re‑order‑guided”
    optimisation often escapes the plateau that plain gradient methods hit.

3️⃣  Global polishing – a short random hill‑climber and a Frank‑Wolfe‑style
    pairwise mass‑transfer finish the search.

All steps respect the supplied wall‑clock budget (≈ 1000 s).  The
returned list can be fed directly into ``evaluate_sequence``.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of the max‑convolution functional on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate for `evaluate_sequence` when Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Ordering heuristics ----------------------------------------------------
def order_alternate_extremes(desc_vals: np.ndarray) -> np.ndarray:
    """Place the largest, second‑largest, … alternately at the left‑most
    and right‑most positions, spreading large masses."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    left = 0
    right = n - 1
    toggle = True
    for v in desc_vals:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    return out


def _reorder_by_alternate_extremes(a: np.ndarray) -> np.ndarray:
    """Re‑order a vector so that its biggest entries sit at the two ends."""
    desc = np.sort(a)[::-1]
    return order_alternate_extremes(desc)


# ----------------------------------------------------------------------
# Analytic families -------------------------------------------------------
def _symmetric_power_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+0.5)*(n‑i‑0.5))^(‑p)  (generalised arcsine)."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-p)
    vals = np.maximum(vals, 0.0)
    total = vals.sum()
    if total == 0:
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals /= total
    return vals


def _end_concentrated_distribution(n: int, q: float) -> np.ndarray:
    """a_i ∝ max(i+1, n‑i) ^ q  – large mass on the two ends."""
    i = np.arange(n, dtype=np.float64)
    base = np.maximum(i + 1, n - i)
    vals = base ** q
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _double_exponential_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) + exp(-lam·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Unsymmetric Beta(α,β) on a discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide='ignore', invalid='ignore'):
        w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    if np.isnan(w).any() or np.isinf(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex from Dirichlet(α,…,α)."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Optimisers --------------------------------------------------------------
def _exp_gradient_descent_reorder(a0: np.ndarray, deadline: float,
                                 max_iters: int = 2_000_000,
                                 eta0: float = 0.8,
                                 reorder_every: int = 50) -> tuple[np.ndarray, float]:
    """Multiplicative (exponentiated) gradient descent with periodic re‑ordering."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s <= 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        if it % reorder_every == 0:
            a = _reorder_by_alternate_extremes(a)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _adam_descent_reorder(a0: np.ndarray, deadline: float,
                          max_iters: int = 600_000,
                          step0: float = 0.2,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          reorder_every: int = 100) -> tuple[np.ndarray, float]:
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        if t % reorder_every == 0:
            a = _reorder_by_alternate_extremes(a)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _sgd_subgrad_reorder(a0: np.ndarray, deadline: float,
                         max_iters: int = 500_000,
                         step0: float = 0.5,
                         reorder_every: int = 100) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        step = step0 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if it % reorder_every == 0:
            a = _reorder_by_alternate_extremes(a)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _hill_climb(a: np.ndarray, deadline: float,
                max_iters: int = 300_000,
                epsilon: float = 0.02) -> tuple[np.ndarray, float]:
    """Random mass transfer hill‑climber (accept‑only‑improve)."""
    best = a.copy()
    best_score = _max_convolution_score(best)
    n = best.size
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or best[i] < 1e-12:
            continue
        delta = min(best[i] * epsilon, 1.0 - best[j])
        if delta <= 0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _reorder_by_alternate_extremes(cand)  # keep large at ends
        sc = _max_convolution_score(cand)
        if sc < best_score:
            best = cand
            best_score = sc
    return best, best_score


def _frank_wolfe_opt(a0: np.ndarray, deadline: float,
                    max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Iterated pairwise transfer of mass from the index with largest
    sub‑gradient to the one with smallest."""
    a = a0.copy()
    cur_score = _max_convolution_score(a)
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        delta = min(a[i_src], 1.0 - a[i_dst])
        if delta <= 0:
            break
        step = 0.7 * delta
        a[i_src] -= step
        a[i_dst] += step
        a = _reorder_by_alternate_extremes(a)          # maintain good layout
        cur_score = _max_convolution_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        if it % 4000 == 0 and np.random.rand() < 0.1:
            a = _project_onto_simplex(a + np.random.randn(a.size) * 1e-5, s=1.0)
            a = _reorder_by_alternate_extremes(a)
    return best, best_score


# ----------------------------------------------------------------------
def _add_candidate(arr: np.ndarray, candidates: list):
    """Normalise, reorder, evaluate and store a candidate vector."""
    if arr.ndim != 1 or arr.size == 0:
        return
    a = np.maximum(arr, 0.0)
    s = a.sum()
    if s < 1e-12:
        return
    a = a / s
    a = _reorder_by_alternate_extremes(a)
    cand_score = _max_convolution_score(a)
    candidates.append((a, cand_score))

    return best_seq.tolist()
