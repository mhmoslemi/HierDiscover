def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    deadline=time.time()+budget_s-0.5
    base_n= 8000 # default
    try: base_n = len(height_sequence_1)
    except: pass
    # lengths: choose a variety of lengths
    lens=set()
    for factor in [0.8,1,1.2,1.5]:
        n=int(base_n*factor)
        if n>=800 and n<=20000:
            lens.add(n)
    # also random lengths
    while len(lens)<10:
        n=np.random.randint(800, min(20000, base_n*2)+1)
        lens.add(n)
    lens=sorted(lens)
    candidates=[] # (a, score)
    def add(a):
        a=np.asarray(a,float)
        if a.ndim!=1 or a.size==0: return
        a=np.maximum(a,0)
        s=a.sum()
        if s<1e-12: return
        a=a/s
        sc=_max_convolution_score(a)
        candidates.append((a,sc))
    for n in lens:
        if time.time()>=deadline: break
        # families
        a=_arcsine_distribution(n)
        add(a)
        desc=np.sort(a)[::-1]; add(order_alternate_extremes(desc))
        add(order_center_outwards(desc)); add(order_binary_tree(desc))
        # power transforms of arcsine
        for p in [1.2,1.5,2.0,2.5,3.0]:
            a2 = np.power(_arcsine_distribution(n), p)
            a2/=a2.sum()
            add(a2)
        # beta symmetric family
        for p in [0.005,0.01,0.02,0.05,0.1]:
            a=_beta_symmetric_distribution(n,p)
            add(a)
        # unsymmetric Beta with extreme parameters
        for a_param,b_param in [(0.1,5),(5,0.1),(0.2,5),(5,0.2)]:
            a=_beta_distribution(n,a_param,b_param)
            add(a)
        # exponential with smaller lam
        for lam in [0.001,0.005]:
            a=_exponential_distribution(n,lam)
            add(a)
        # Gaussian with small sigma factor
        for sigma in [0.1,0.15]:
            a=_gaussian_distribution(n,sigma_factor=sigma)
            add(a)
        # Dirichlet draws
        for alpha in [0.02,0.05,0.1,0.5,1]:
            a=_random_dirichlet_alpha(n,alpha)
            add(a)
        # random permutations of sorted uniform heights (makes random ordering)
        u=np.full(n,1.0/n)
        for _ in range(2):
            perm=np.random.permutation(n)
            a=u[perm]
            add(a)
    # add previous best
    try:
        prev=np.asarray(height_sequence_1,float)
        prev=prev/prev.sum()
        add(prev)
    except: pass

    # keep top seeds
    candidates.sort(key=lambda x:x[1])
    top_k = min(8, len(candidates))
    seeds = [c[0] for c in candidates[:top_k]]
    best_seq = seeds[0].copy()
    best_score = _max_convolution_score(best_seq)

    # iterative refinement
    while time.time()<deadline-5:
        # choose a seed among seeds at random or cycle
        for seq in seeds:
            if time.time()>=deadline-5:
                break
            seq, sc = _exp_gradient_descent(seq, deadline, max_iters=500000, eta0=0.9)
            if sc<best_score:
                best_score, best_seq = sc, seq.copy()
            seq, sc = _adam_descent(best_seq, deadline, max_iters=200000, step0=0.2)
            if sc<best_score:
                best_score, best_seq = sc, seq.copy()
            seq, sc = _sgd_subgrad(best_seq, deadline, max_iters=200000, step0=0.5)
            if sc<best_score:
                best_score, best_seq = sc, seq.copy()
        # after cycle, add a perturbation of best for diversity
        if time.time()<deadline-5:
            pert = best_seq * np.exp(0.05*np.random.randn(best_seq.size))
            pert = pert/pert.sum()
            # run short exp gradient
            pert, sc = _exp_gradient_descent(pert, deadline, max_iters=100000, eta0=0.8)
            if sc<best_score:
                best_score, best_seq = sc, pert.copy()
        # occasional hill climbing
        if time.time()<deadline-15:
            seq, sc = _hill_climb(best_seq, deadline, max_iters=200000, epsilon=0.015)
            if sc<best_score:
                best_score, best_seq = sc, seq.copy()
        # final Frank-Wolfe
        if time.time()<deadline-5:
            seq, sc = _frank_wolfe_opt(best_seq, deadline, max_iters=300000)
            if sc<best_score:
                best_score, best_seq = sc, seq.copy()
        # break if time low
        if time.time()>=deadline-2:
            break

    print(f"Final surrogate score: {best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()

"""
Propose a step‑function (non‑negative list of heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The routine follows a three‑phase strategy:

1. **Diverse seeding** – a large collection of analytically‑defined families
   (Beta, arcsine, Gaussian, exponential, power‑transforms, etc.) on
   several lengths, plus numerous random Dirichlet draws.  Each seed is
   ℓ₁‑normalised and scored with the cheap surrogate
   `2·n·max(conv(a,a))` (where `∑a = 1`).

2. **Intensive continuous optimisation** – the best few seeds are polished
   with long runs of exponentiated‑gradient descent, Adam and projected‑SGD.
   The remaining budget is devoted to repeatedly refining the current
   best candidate, occasionally perturbing it to escape shallow minima.

3. **Discrete polishing** – a cheap hill‑climber (random mass transfers)
   and a Frank‑Wolfe style pairwise mass transfer finish the search.

All steps respect the supplied time budget (≈ 1000 s).  The returned list
has been projected onto the simplex, so it can be passed directly to
`evaluate_sequence`.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate for `evaluate_sequence` when Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of the max‑convolution functional on the simplex.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even offsets (i = j)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _gaussian_distribution(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    w /= w.sum()
    return w


def _cauchy_distribution(n: int, scale_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    scale = scale_factor * n
    w = 1.0 / (1.0 + ((idx - centre) / scale) ** 2)
    w /= w.sum()
    return w


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Unsymmetric Beta(α,β) on a discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide='ignore', invalid='ignore'):
        w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    if np.isnan(w).any() or np.isinf(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex from Dirichlet(α,…,α)."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Re‑ordering heuristics -----------------------------------------------
def order_alternate_extremes(desc_vals: np.ndarray) -> np.ndarray:
    """Place the largest, second‑largest, … alternately at the left‑most
    and right‑most positions, spreading large masses."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    left = 0
    right = n - 1
    toggle = True
    for v in desc_vals:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    return out


def order_center_outwards(desc_vals: np.ndarray) -> np.ndarray:
    """Put the biggest value at the centre and then fill alternatingly
    to the left and to the right, moving outwards."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    positions = []
    mid = (n - 1) // 2
    positions.append(mid)
    for d in range(1, n):
        left = mid - d
        if left >= 0:
            positions.append(left)
        right = mid + d
        if right < n:
            positions.append(right)
    for i, pos in enumerate(positions[:n]):
        out[pos] = desc_vals[i]
    return out


def order_binary_tree(desc_vals: np.ndarray) -> np.ndarray:
    """Spread large coefficients by recursively filling the midpoint of
    the remaining interval (binary‑tree layout)."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    positions = []

    def recurse(l: int, r: int):
        if l > r:
            return
        m = (l + r) // 2
        positions.append(m)
        recurse(l, m - 1)
        recurse(m + 1, r)

    recurse(0, n - 1)
    out[positions] = desc_vals
    return out

# Mirror‑descent variants -----------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 500_000,
                  step0: float = 0.2,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 1_000_000,
                          eta0: float = 0.8) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _sgd_subgrad(a0: np.ndarray, deadline: float,
                 max_iters: int = 500_000,
                 step0: float = 0.5) -> tuple[np.ndarray, float]:

    best_score = _max_convolution_score(a)

            best_a = a.copy()

def _random_mass_transfer(a: np.ndarray, epsilon: float = 0.02) -> np.ndarray:
    """Move a random fraction ε of mass from a randomly chosen source to a
    randomly chosen destination (preserving the ℓ₁‑norm)."""
    n = a.shape[0]
    src = np.random.randint(n)
    dst = np.random.randint(n)
    while dst == src or a[src] < 1e-12:
        src = np.random.randint(n)
        dst = np.random.randint(n)
    delta = min(a[src] * epsilon, 1.0 - a[dst])
    if delta <= 0:
        return a
    b = a.copy()
    b[src] -= delta
    b[dst] += delta
    return b


def _hill_climb(a: np.ndarray, deadline: float,
                max_iters: int = 300_000, epsilon: float = 0.015) -> tuple[np.ndarray, float]:
    """Simple hill‑climber that repeatedly performs a random mass transfer.
    Only accepts moves that improve the max‑convolution score."""
    best = a.copy()
    best_score = _max_convolution_score(best)
    n = best.size
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or best[i] < 1e-12:
            continue
        delta = min(best[i] * epsilon, 1.0 - best[j])
        if delta <= 0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = _max_convolution_score(cand)
        if sc < best_score:
            best = cand
            best_score = sc
    return best, best_score

# Frank‑Wolfe style pairwise mass transfer ------------------------------
def _frank_wolfe_opt(a0: np.ndarray, deadline: float,
                    max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Iterated pairwise transfer of mass from the index with largest
    sub‑gradient to the one with smallest."""
    a = a0.copy()
    cur_score = _max_convolution_score(a)
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        delta = min(a[i_src], 1.0 - a[i_dst])
        if delta <= 0:
            break
        step = 0.7 * delta          # 70 % of admissible transfer
        a[i_src] -= step
        a[i_dst] += step
        cur_score = _max_convolution_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        # occasional jitter
        if it % 4000 == 0 and np.random.rand() < 0.1:
            a = _project_onto_simplex(a + np.random.randn(a.size) * 1e-5, s=1.0)
    return best, best_score
