"""
Propose a step‑function (list of non‑negative floats) that minimizes
``evaluate_sequence``.  The algorithm works in three stages:

1. **Broad sampling** –  a mixture of deterministic analytic families
   (uniform, arcsine, triangular, exponential, Beta, power‑edge,
   power‑abs, sine‑power, Gaussian, V‑shape) and many random families
   (Dirichlet draws, random mixtures) is generated for several
   lengths (2000‑15000).  For each candidate the cheap score
   ``2·n·max(a*a)`` is computed by an FFT‑based convolution and the
   best ``K`` candidates (K≈30) are kept.

2. **Local refinement** – each of the stored candidates is improved
   with a projected exponentiated‑gradient descent on the smooth‑max
   approximation of the matrix‑infinity norm of the self‑convolution.
   The update respects the simplex constraint via Euclidean projection.
   The schedule for the smoothness parameter and the learning rate is
   aggressive (α grows from 0.1 to >10 000, learning rate from 5 down
   to 0.02).  Periodic exact‑max steps and small random jitters help
   escape plateaus.

3. **Final hill‑climb** – the globally best vector from step 2 is
   polished by a cheap greedy mass‑transfer that repeatedly removes a
   little mass from the pair of indices that currently contributes most
   to the maximal convolution entry.

All helper functions are defined at module level (no closures,
no lambdas).  The routine respects a wall‑clock budget of 1000 s and
returns the best sequence found.
"""

import time
import random
import math

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Fast evaluation of 2·n·max(a*a) (a is assumed to sum to 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * float(conv.max())

# Analytic families – each returns a *probability* vector (sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.01) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * (np.log(x) + np.log(1 - x))
    w = np.exp(logw)
    if w.sum() == 0 or np.any(np.isinf(w)):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals /= vals.sum()
    return vals


def _distribution_power_edge(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - i - 0.5)
    eps = 1e-12
    vals = np.power(denom + eps, -alpha)
    vals /= vals.sum()
    return vals


def _distribution_V_shape(n: int) -> np.ndarray:
    centre = (n - 1) / 2.0
    vals = np.abs(np.arange(n, dtype=np.float64) - centre)
    vals = np.maximum(vals, 1e-12)
    vals /= vals.sum()
    return vals


def _distribution_power_abs(n: int, beta: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    vals = np.abs(np.arange(n, dtype=np.float64) - centre) ** beta
    vals = np.maximum(vals, 1e-12)
    vals /= vals.sum()
    return vals


def _random_symmetric_dirichlet(n: int, alpha: float) -> np.ndarray:
    half = (n + 1) // 2
    half_vec = np.random.dirichlet(np.full(half, alpha))
    if n % 2 == 0:
        full = np.concatenate([half_vec, half_vec[::-1]])
    else:
        full = np.concatenate([half_vec, half_vec[-2::-1]])
    full /= full.sum()
    return full


# ----------------------------------------------------------------------
def _add_candidate(cand_list: list, vec: np.ndarray, K: int, eps: float) -> None:
    """Insert vec (already a numpy array) into cand_list if it belongs to top‑K."""
    vec = np.maximum(vec, eps)
    total = vec.sum()
    if total < 1e-12:
        return
    vec = vec / total
    score = _max_convolution_score(vec)
    if len(cand_list) < K:
        cand_list.append((score, vec.copy()))
        cand_list.sort(key=lambda x: x[0])
    else:
        if score < cand_list[-1][0]:
            cand_list[-1] = (score, vec.copy())
            cand_list.sort(key=lambda x: x[0])


def _random_distribution(n: int) -> np.ndarray:
    """Return a random distribution constructed from several parametric families."""
    choice = random.randint(0, 9)
    if choice == 0:                     # exponential
        lam = 10 ** random.uniform(-4.0, -0.5)   # ~[1e-4, 0.3]
        return _distribution_exponential(n, lam)
    if choice == 1:                     # symmetric beta
        p = 10 ** random.uniform(-4.0, 2.0)    # ~[1e-4, 100]
        return _distribution_beta_sym(n, p)
    if choice == 2:                     # power‑edge
        alpha = random.uniform(0.55, 0.99)
        return _distribution_power_edge(n, alpha)
    if choice == 3:                     # power‑abs
        beta = random.uniform(0.5, 5.0)
        return _distribution_power_abs(n, beta)
    if choice == 4:                     # sine‑power
        power = random.uniform(0.2, 6.0)
        return _distribution_sine_power(n, power)
    if choice == 5:                     # Gaussian
        sigma = random.uniform(0.1, 0.6)
        return _distribution_gaussian(n, sigma)
    if choice == 6:                     # very sparse Dirichlet
        alpha = 10 ** random.uniform(-4.0, -0.2)   # ~[1e-4, 0.63]
        return _random_symmetric_dirichlet(n, alpha)
    if choice == 7:                     # mixture Uniform + Dirichlet
        lam = random.random()
        alpha = 10 ** random.uniform(-4.0, -0.2)
        dirv = _random_symmetric_dirichlet(n, alpha)
        uni = _distribution_uniform(n)
        return lam * uni + (1.0 - lam) * dirv
    if choice == 8:                     # V‑shape (deterministic)
        return _distribution_V_shape(n)
    # choice == 9
    return _distribution_triangular(n)


def _refine_vector(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected exponentiated‑gradient descent on the smooth‑max approximation.
    Returns the best vector seen before the deadline and its score.
    """
    eps = 1e-12
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_score(best)

    it = 0
    # total planned iterations – we keep iterating until time runs out.
    while time.time() < deadline:
        it += 1
        # convolution and smooth‑max weights
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # exponential schedule for smoothing
        tau = it / 20000.0                     # [0,1] slowly
        alpha = 0.1 * (20000.0 ** tau)         # goes from 0.1 up to >10 000
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth max
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        # learning‑rate schedule (aggressive early, tiny later)
        lr = 5.0 * (1.0 - tau) + 0.02

        # exponentiated‑gradient step (with clipping for safety)
        exp_factor = np.exp(-lr * grad)
        exp_factor = np.clip(exp_factor, 1e-12, 1e12)
        a = a * exp_factor
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()

        # occasional exact‑max step to sharpen the descent
        if it % 1000 == 0:
            mask = (conv >= maxc - 1e-12).astype(np.float64)
            grad_exact = 2.0 * _fft_convolve(a[::-1], mask)[n - 1:n - 1 + n]
            exp_fact2 = np.exp(-lr * grad_exact)
            exp_fact2 = np.clip(exp_fact2, 1e-12, 1e12)
            a = a * exp_fact2
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

        # tiny random jitter to escape flat regions
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()

        # evaluate periodically
        if it % 500 == 0:
            sc = _max_convolution_score(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best = a.copy()
                print(
                    f"[{time.time():.1f}] refinement n={n} it={it} best={sc:.6f}"
                )
    return best, best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Main entry point required by the evaluation harness.
    Returns a list of non‑negative floats (the step‑function) whose
    score is as low as possible within the given time budget.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin
    eps = 1e-12

    # --------------------------------------------------------------
    # 1.  Broad generation of candidates
    lengths = [2000, 2500, 3000, 3500, 4000, 4500, 5000,
               6000, 8000, 10000, 12000, 15000]
    K = 30                                 # keep top‑K candidates
    top_candidates: list[tuple[float, np.ndarray]] = []  # (score, vector)

    # deterministic families (dense sweep)
    for n in lengths:
        if time.time() >= deadline:
            break

        # uniform
        _add_candidate(top_candidates, _distribution_uniform(n), K, eps)
        # arcsine
        _add_candidate(top_candidates, _distribution_arcsine(n), K, eps)
        # triangular
        _add_candidate(top_candidates, _distribution_triangular(n), K, eps)
        # exponential (two lambdas)
        _add_candidate(top_candidates, _distribution_exponential(n, lam=0.008), K, eps)
        _add_candidate(top_candidates, _distribution_exponential(n, lam=0.02), K, eps)
        # V‑shape
        _add_candidate(top_candidates, _distribution_V_shape(n), K, eps)

        # sine‑power
        for pw in (0.5, 1.0, 2.0, 4.0):
            _add_candidate(top_candidates, _distribution_sine_power(n, pw), K, eps)

        # Gaussian
        for sigma in (0.2, 0.35, 0.5):
            _add_candidate(top_candidates, _distribution_gaussian(n, sigma), K, eps)

        # symmetric Beta
        for p in (0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.8,
                  1.0, 1.5, 2.0, 3.0, 4.0):
            _add_candidate(top_candidates, _distribution_beta_sym(n, p), K, eps)

        # power‑edge
        for a_edge in (0.55, 0.6, 0.7, 0.8, 0.9):
            _add_candidate(top_candidates, _distribution_power_edge(n, a_edge), K, eps)

        # power‑abs
        for beta in (0.5, 1.0, 2.0, 3.0):
            _add_candidate(top_candidates, _distribution_power_abs(n, beta), K, eps)

        # random Dirichlet draws (few per length)
        for alpha in (0.03, 0.05, 0.08, 0.1, 0.2, 0.5):
            _add_candidate(
                top_candidates,
                _random_symmetric_dirichlet(n, alpha),
                K,
                eps,
            )

    # --------------------------------------------------------------
    # 2.  Random exploration (mass sampling)
    # reserve a small window for the refinement phase
    while time.time() < deadline - 15.0:
        n = random.choice(lengths)
        vec = _random_distribution(n)
        _add_candidate(top_candidates, vec, K, eps)

    # --------------------------------------------------------------
    # 3.  Include any previously supplied sequence
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            _add_candidate(top_candidates, prev, K, eps)
    except Exception:
        pass

    # If nothing survived (unlikely), fall back to a uniform vector.
    if not top_candidates:
        n_fallback = 3000
        fallback = _distribution_uniform(n_fallback)
        _add_candidate(top_candidates, fallback, K, eps)

    # --------------------------------------------------------------
    # 4.  Local refinement of the stored candidates
    remaining = deadline - time.time()
    if remaining <= 0:
        # time exhausted – just return the current best
        best_vec = min(top_candidates, key=lambda x: x[0])[1]
        return best_vec.tolist()

    # allocate time roughly equally among the candidates
    per_candidate = max(5.0, remaining / len(top_candidates))
    global_best_vec = None
    global_best_score = float("inf")

    for score, vec in top_candidates:
        end_time = min(time.time() + per_candidate, deadline)
        refined_vec, refined_score = _refine_vector(vec, end_time)
        if refined_score < global_best_score:
            global_best_score = refined_score
            global_best_vec = refined_vec.copy()
            print(
                f"[{time.time():.1f}] new global best from refinement score={refined_score:.6f}"
            )
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # 5.  Final cheap hill‑climber on the best vector
    if global_best_vec is None:
        # fallback: pick the smallest‑score vector from the pool
        global_best_vec = min(top_candidates, key=lambda x: x[0])[1].copy()
        global_best_score = _max_convolution_score(global_best_vec)

    a = global_best_vec.copy()
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    hill_iters = 0

    while time.time() < deadline:
        hill_iters += 1
        k = int(conv.argmax())
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        # locate the pair (i,j) that gives the maximal product for this k
        best_i, best_j = i_low, k - i_low
        max_prod = a[best_i] * a[best_j]
        for i in range(i_low, i_high + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        src = best_i if a[best_i] >= a[best_j] else best_j
        tgt = random.randint(0, n - 1)
        if tgt == src:
            continue

        max_delta = min(a[src] * 0.5, 0.015)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a_snapshot = a.copy()
        a[src] -= delta
        a[tgt] += delta
        a = np.maximum(a, eps)
        a /= a.sum()
        conv = _fft_self_convolve(a)
        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
            sc = 2.0 * n * M
            if sc < global_best_score - 1e-12:
                global_best_score = sc
                global_best_vec = a.copy()
                print(f"[{time.time():.1f}] hill‑climb it={hill_iters}  best={sc:.6f}")
        else:
            a = a_snapshot
            conv = _fft_self_convolve(a)

        if hill_iters % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] jitter applied at hill iter {hill_iters}")

    final_score = 2.0 * global_best_vec.shape[0] * _fft_self_convolve(global_best_vec).max()
    print(f"Final best score ≈ {final_score:.6f} (length={global_best_vec.shape[0]})")

    return global_best_vec.tolist()
