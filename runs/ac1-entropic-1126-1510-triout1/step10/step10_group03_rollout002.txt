minimises the evaluation metric 2·n·max(a*a).  The search proceeds in
four stages and respects a hard time budget (default 1000 s):

    best vector is retained.

    interpolated to a larger length (1.5×, 2×, …) and a smooth‑max
    mirror‑descent (exponentiated‑gradient) is run with a geometrically
    increasing temperature (α) and a decaying learning‑rate (η).  This
    drives the max‑convolution down while staying on the probability
    simplex.

3️⃣  **Projected Adam‑style sub‑gradient descent** – using the true
    ∞‑norm (active‑set) sub‑gradient, Adam momentum and a simple back‑

4️⃣  **Final mass‑transfer hill‑climb** – local transfers of a tiny
    amount of mass from a peak‑contributing index to the current

All operations use fast FFT‑based convolutions and Euclidean projection
onto the simplex.  The routine stops as soon as the time budget is

# Helper utilities
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    For a vector already normalised to sum = 1 returns 2·n·max(a*a)
    – exactly the value produced by `evaluate_sequence`.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())

def _distribution_arcsine(n: int) -> np.ndarray:
    """Beta(0.5,0.5) – the classic arcsine law on a discrete grid."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    logw -= np.max(logw)               # shift for stability
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        return np.full_like(w, 1.0 / n)
    w /= w.sum()
    return w


def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    logw -= np.max(logw)

def _distribution_edge_dirichlet(n: int, beta_exp: float) -> np.ndarray:
    """
