"""
Hybrid evolutionary + smooth‑gradient optimiser for the step‑function attack.

*   Generates a diverse pool of analytic “seed” distributions for a few
    lengths (≈2000‑5000).  All seeds are normalised to sum = 1 (probability
    simplex).

*   Maintains a small population per length (best 10 candidates).

*   Repeatedly creates new candidates by
    – crossover (convex combination of two parents),
    – Gaussian mutation of the best parent,
    – a short “soft‑max gradient descent” that minimises a smooth
      approximation of  max(conv(a,a))  (exponentiated‑gradient update
      with projection onto the simplex).

*   Occasionally evaluates the exact objective (the same as
    evaluate_sequence after fixing the sum) and updates the global best.

*   After the time budget is exhausted a final local “pairwise‑swap”
    walk is performed to polish the best vector.

The optimiser uses only NumPy, no external solvers, and respects the
global time limit (≈ 1000 s).  The returned list of floats can be fed
directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a via a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of Σ_k w_k·(a*a)_k  w.r.t. a.
    Implements 2·a⋆w where ⋆ denotes cross‑correlation.
    """
    n = a.shape[0]
    m = w.shape[0]                # = 2·n‑1
    a_rev = a[::-1]
    size = 1 << ((n + m - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1                 # index where shift = 0 begins
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (sum(a) is assumed = 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    return 2.0 * n * max_conv

# Analytic families (identical to those used in the reference implementation)

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _initial_candidates_extended(n: int,
                                 rng: np.random.Generator) -> list[np.ndarray]:
    """Broad pool of analytic seeds for a given length."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    for p in (0.05, 0.1, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0):
        cand.append(_distribution_beta_sym(n, p))

    for _ in range(5):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))

    for lam in (0.001, 0.005, 0.01, 0.02, 0.05):
        cand.append(_distribution_exponential_increasing(n, lam))

    for block_frac in (0.05, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac))

    # a few Dirichlet draws with varying concentration
    for _ in range(4):
        alpha = 0.3 + 0.7 * rng.random()
        cand.append(_distribution_dirichlet(n, rng, alpha=alpha))

    cand.append(_distribution_random_sparse(n, rng, density=0.01))
    return cand

def _smooth_gradient_descent(a: np.ndarray,
                             deadline: float,
                             lr0: float = 0.5,
                             steps: int = 4000,
                             beta_start: float = 0.5,
                             beta_end: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a *smooth* approximation of max(conv(a,a)).
    Uses a soft‑max over the convolution entries with a temperature β
    that is increased from ``beta_start`` to ``beta_end``.
    The update is mirror‑descent with an exponentiated‑gradient step
    followed by a projection onto the probability simplex.
    Returns the best vector seen and its exact objective.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    for t in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # linear schedule for β
        beta = beta_start + (beta_end - beta_start) * (t - 1) / (steps - 1)

        conv = _fft_self_convolve(a)

        # soft‑max weights (stable)
        scaled = beta * conv
        max_scaled = np.max(scaled)
        w = np.exp(scaled - max_scaled)
        w /= w.sum()

        # sub‑gradient of the smooth max
        grad = _gradient_from_weights(a, w)

        # step size η_t = lr0 / √t
        eta = lr0 / math.sqrt(t)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad)

        # simplex projection (keeps sum = 1)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # occasional exact evaluation
        if t % 200 == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_a = a.copy()
                print(f"[{time.time():.1f}] smooth‑gd it={t} β={beta:.1f} new best {best_score:.6f}")

    return best_a, best_score

def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_score: float) -> tuple[np.ndarray, float]:
    """
    Greedy local mass‑transfer walk (same idea as the reference implementation).
    Moves a random fraction of mass from a source index to a different target
    index and keeps the move only if the exact objective improves.
    """
    n = a.shape[0]
    cur_seq = a.copy()
    cur_best = best_score

    while time.time() < deadline:
        src = np.where(cur_seq > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = cur_seq[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        cand = cur_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_onto_simplex(cand, s=1.0)

        sc = _objective(cand)
        if sc < cur_best - 1e-13:
            cur_seq = cand
            cur_best = sc
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {cur_best:.6f}")

    return cur_seq, cur_best


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine runs for up to ``budget_s`` seconds
    and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. problem sizes we will explore (moderate lengths keep evaluation cheap)
    lengths = [2000, 3000, 4000, 5000]

    # population per length: list of (score, vector)
    pop: dict[int, list[tuple[float, np.ndarray]]] = {}
    global_best_seq: np.ndarray | None = None
    global_best_score = float("inf")

    # ------------------------------------------------------------------
    # 2. initialise populations with analytic and random seeds
    for n in lengths:
        if time.time() >= deadline:
            break
        seeds = _initial_candidates_extended(n, rng)

        # add a few extra Dirichlet draws with very small concentration
        for _ in range(3):
            alpha = 10 ** rng.uniform(-3, -0.5)   # very sparse
            seeds.append(_distribution_dirichlet(n, rng, alpha=alpha))

        cur_pop: list[tuple[float, np.ndarray]] = []
        for cand in seeds:
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = _objective(cand)
            cur_pop.append((sc, cand.copy()))
            if sc < global_best_score:
                global_best_score = sc
                global_best_seq = cand.copy()
                print(f"[{time.time():.1f}] init len={n} new global best {sc:.6f}")

        cur_pop.sort(key=lambda x: x[0])
        pop[n] = cur_pop[:12]   # keep the best few

    # ------------------------------------------------------------------
    # 3. optionally incorporate a previously discovered sequence
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < global_best_score:
                global_best_score = sc
                global_best_seq = prev.copy()
                print(f"[{time.time():.1f}] imported previous best {sc:.6f}")
            # insert in each length’s population (if length matches)
            if prev.shape[0] in pop:
                pop[prev.shape[0]].append((sc, prev.copy()))
                pop[prev.shape[0]].sort(key=lambda x: x[0])
                pop[prev.shape[0]] = pop[prev.shape[0]][:12]
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. evolutionary / gradient optimisation loop
    while time.time() < deadline:
        # pick a length (biased towards those with better scores)
        n = rng.choice(lengths)
        cur_pop = pop[n]

        # favour better individuals when sampling parents
        scores = np.array([s for s, _ in cur_pop])
        probs = 1.0 / (scores + 1e-12)
        probs /= probs.sum()

        op = rng.choice(["crossover", "mutate", "smooth_gd"], p=[0.4, 0.3, 0.3])

        if op == "crossover" and len(cur_pop) >= 2:
            i, j = rng.choice(len(cur_pop), size=2, replace=False, p=probs)
            a = cur_pop[i][1]
            b = cur_pop[j][1]
            lam = rng.random()
            child = lam * a + (1.0 - lam) * b
            child = _project_onto_simplex(child, s=1.0)

        elif op == "mutate":
            # mutate the currently best individual of this length
            a = cur_pop[0][1]
            sigma = 0.02  # relative noise level
            child = a + sigma * rng.standard_normal(n)
            child = np.maximum(child, 0.0)
            child = _project_onto_simplex(child, s=1.0)

        else:   # smooth gradient descent on the best individual
            a = cur_pop[0][1].copy()
            # allocate a few seconds for a short descent
            sub_deadline = min(deadline, time.time() + 4.0)
            child, _ = _smooth_gradient_descent(
                a,
                sub_deadline,
                lr0=0.7,
                steps=5000,
                beta_start=0.5,
                beta_end=300.0,
            )

        # evaluate the new candidate
        sc = _objective(child)
        if sc < global_best_score - 1e-13:
            global_best_score = sc
            global_best_seq = child.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST {sc:.6f} (len={n}, op={op})")

        # keep the population trimmed to the top 12
        cur_pop.append((sc, child.copy()))
        cur_pop.sort(key=lambda x: x[0])
        pop[n] = cur_pop[:12]

    # ------------------------------------------------------------------
    # 5. final polishing with cheap local swaps (if time permits)
    if global_best_seq is not None and time.time() < deadline:
        global_best_seq, global_best_score = _pairwise_swap_local(
            global_best_seq.copy(),
            deadline,
            best_score=global_best_score,
        )

    # ------------------------------------------------------------------
    # 6. report and return
    final_val = _objective(global_best_seq) if global_best_seq is not None else float("inf")
    print(f"FINAL SCORE {final_val:.6f} (length={len(global_best_seq) if global_best_seq is not None else 0})")
    return global_best_seq.tolist() if global_best_seq is not None else []
