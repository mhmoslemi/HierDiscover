"""
Hybrid multi‑start optimiser for the step‑function attack.

The optimiser works in several stages:

1.  **Broad seed generation** – analytic families (uniform, arcsine,
    beta, power‑law, exponential, two‑block, sparse, Dirichlet) are
    instantiated for a few moderate lengths (2k‑8k).  Each seed is
    normalised to an L1‑norm of 1 and its exact objective
    `2·n·max(conv(a,a))/sum(a)**2` is evaluated.  The best ~12 seeds are
    kept for the next stage.

2.  **High‑precision continuous optimisation** – each retained seed is
    refined by a mirror‑descent scheme on the probability simplex.
    Two complementary gradient methods are employed:

        * ``_subgradient_multimax_descent`` – uses the full active set of
          convolution indices attaining the current maximum and an
          exponentiated‑gradient update with a √t step‑size schedule.

        * ``_adam_exponential_descent`` – a momentum‑based variant (Adam)
          that keeps exponential updates in the log‑domain, again
          projected onto the simplex.

    Both procedures run until the global time budget is exhausted or a
    preset maximum number of iterations is reached.  The best sequence
    seen so far (across all seeds and both methods) is memorised.

3.  **Scale‑up & re‑optimise** – the current best distribution is
    linearly interpolated to a larger length (up to 30 000 entries) and
    the same descent routines are applied again.  Upscaling allows the
    shape to be represented more finely, which often yields a modest
    reduction of the supremum of the self‑convolution.

4.  **Final polishing** – a cheap greedy “pairwise‑swap” walk moves a
    tiny amount of mass from one index to another whenever the true
    objective improves.  This step typically extracts the last few
    digits of improvement.

The whole pipeline respects the supplied ``budget_s`` (≈ 1000 s) and
returns the best coefficient list found (as plain Python ``float``s).
All numbers are non‑negative; the simplex projection guarantees that
the sum is exactly 1, which is invariant under the scaling performed by
``evaluate_sequence``.
"""

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (sum is forced to 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _smooth_distribution(a: np.ndarray, width: int) -> np.ndarray:
    """Average smoothing with a uniform kernel of odd ``width``."""
    if width <= 1:
        return a.copy()
    k = np.ones(width, dtype=np.float64) / width
    sm = np.convolve(a, k, mode='same')
    sm = np.maximum(sm, 0.0)
    sm /= sm.sum()
    return sm


def _interpolate_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(x_new, x_old, a)
    new_a = np.maximum(new_a, 0.0)
    new_a /= new_a.sum()
    return new_a

    cand = []

def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the sub‑gradient 2·a⋆w where w is a weight vector (length 2n‑1)
    that sums to 1.  Implemented via FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2·n‑1
    a_rev = a[::-1]
    size = 1 << ((n + m - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad

def _subgradient_multimax_descent(a: np.ndarray, deadline: float,
                                 best_seq: np.ndarray, best_score: float,
                                 lr0: float = 0.5, max_iter: int = 20000,
                                 eval_interval: int = 200) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) using the *full* active set.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()

        # active set – indices where convolution is within a tiny tolerance
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        # step‑size schedule η_t = lr0 / √t
        eta = lr0 / math.sqrt(t)

        # multiplicative (exponentiated) update and simplex projection
        a = a * np.exp(-eta * grad)
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] multigrad it={t} new best {best_score:.6f}")

    return best_seq, best_score

                              max_iter: int = 20000,
                              eval_interval: int = 250) -> tuple[np.ndarray, float]:
    """
    Adam‑style momentum optimiser, but the update is performed in the
    log‑domain (exponentiated gradient) and the vector is kept on the
    probability simplex.
    """
    n = a.shape[0]
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        # Adam updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule similar to subgradient
        lr = lr0 / math.sqrt(t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a * np.exp(-step)          # exponentiated‑gradient step
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """
    Simple greedy walk: pick a source index with non‑negligible mass,
    move a random fraction of it to a random destination, keep the move
    iff the exact objective improves.
    """
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # move at most 30 % of the source mass
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Entry point required by the evaluation harness.  Searches for a
    non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1.  Create an initial pool of seeds (moderate lengths)
    lengths = [2000, 4000, 8000]   # quick scan
    all_candidates: list[tuple[float, np.ndarray]] = []
    overall_best_seq = None
    overall_best_score = float("inf")

    print(f"[{time.time():.1f}] Starting seed generation")
    for n in lengths:
        if time.time() >= deadline:
            break
        for cand in _initial_candidates_extended(n, rng):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] *** new global best len={n} score={sc:.6f}")
            all_candidates.append((sc, cand.copy()))

    # Keep only the very best seeds (across all lengths)
    all_candidates.sort(key=lambda x: x[0])
    top_candidates = all_candidates[:12]      # 12 strongest starts

    # ------------------------------------------------------------------
    # 2.  If a historic solution exists, insert it as a seed
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s == 0.0:
                prev = _distribution_uniform(prev.shape[0])
                s = prev.sum()
            prev = prev / s
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous seed improved: {sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Main optimisation loop – gradient‑based refinements
    #    Allocate remaining time roughly equally among a few top seeds.
    remaining_time = deadline - time.time()
    num_seeds = min(5, len(top_candidates))
    time_per_seed = max(0.0, remaining_time / max(1, num_seeds))

    for idx, (_, seed_seq) in enumerate(sorted(top_candidates, key=lambda x: x[0])[:num_seeds]):
        if time.time() >= deadline:
            break
        sub_deadline = time.time() + time_per_seed

        # ----- sub‑gradient (multimax) phase
        seq, score = _subgradient_multimax_descent(
            seed_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.7,
            max_iter=40000,
            eval_interval=500,
        )
        if score < overall_best_score:
            overall_best_score = score
            overall_best_seq = seq.copy()
            print(f"[{time.time():.1f}] after multimax {score:.6f}")

        # ----- Adam‑style phase (only if we still have time)
        if time.time() < deadline:
            sub_deadline = time.time() + (deadline - time.time()) * 0.5
            seq, score = _adam_exponential_descent(
                seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.6,
                max_iter=30000,
                eval_interval=400,
            )
            if score < overall_best_score:
                overall_best_score = score
                overall_best_seq = seq.copy()
                print(f"[{time.time():.1f}] after Adam {score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Upscale the current best to a larger resolution (if time permits)
    upscale_targets = [12000, 20000, 30000]
    for new_n in upscale_targets:
        if time.time() >= deadline:
            break
        if new_n <= len(overall_best_seq):
            continue
        up_seq = _interpolate_sequence(overall_best_seq, new_n)
        # a quick short gradient phase on the larger grid
        sub_deadline = time.time() + (deadline - time.time()) * 0.5
        up_seq, up_score = _subgradient_multimax_descent(
            up_seq,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.5,
            max_iter=20000,
            eval_interval=300,
        )
        if up_score < overall_best_score:
            overall_best_score = up_score
            overall_best_seq = up_seq.copy()
            print(f"[{time.time():.1f}] upscaled to {new_n} score {up_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final polishing with cheap pairwise swaps
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    # 6.  Report final result (converted to plain Python list)
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
