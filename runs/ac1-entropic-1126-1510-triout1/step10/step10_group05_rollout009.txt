"""
Propose a new step‑function using an LP‑based “mass‑transfer” polishing
procedure followed by a short exponentiated‑gradient refinement.

The algorithm works on the simplex (ℓ₁‑normalised vectors).  For a given
candidate f it computes M = max(conv(f,f)).  An LP maximises Σg subject
to the linear constraints conv(f,g) ≤ M (only the most‑active
convolution indices are kept).  The resulting g is mixed into f;
the process is repeated a few times, gradually lowering M.
Finally a cheap exponentiated‑gradient descent is applied for fine
tuning.  All work respects the supplied time budget (≈ 1000 s) and
returns a plain list of non‑negative heights ready for
`evaluate_sequence`.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate for `evaluate_sequence` when Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a point on the simplex from Dirichlet(α,…,α)."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Re‑ordering heuristics -----------------------------------------------
def order_alternate_extremes(desc_vals: np.ndarray) -> np.ndarray:
    """Place the largest, second‑largest, … alternately at the left‑most
    and right‑most positions, spreading large masses."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    left = 0
    right = n - 1
    toggle = True
    for v in desc_vals:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    return out


def order_center_outwards(desc_vals: np.ndarray) -> np.ndarray:
    """Put the biggest value at centre and then fill alternatingly left/right."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    mid = (n - 1) // 2
    positions = [mid]
    for d in range(1, n):
        if mid - d >= 0:
            positions.append(mid - d)
        if mid + d < n:
            positions.append(mid + d)
    for i, pos in enumerate(positions[:n]):
        out[pos] = desc_vals[i]
    return out


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent (used for final polish) -------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 300_000,
                          eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Multiplicative (exponentiated) gradient descent on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # sub‑gradient of max‑convolution (same routine as below)
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-14:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient of the max‑convolution (required by exp‑grad)
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of the max‑convolution functional on the simplex.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
# LP‑based polishing ---------------------------------------------------
def _lp_refine_one_step(f: np.ndarray, M: float,
                        max_active: int = 150) -> np.ndarray | None:
    """
    One LP step: maximise Σg under conv(f,g) ≤ M for a selected set of
    active constraints.  Returns the normalised optimiser g, or None on
    failure.
    """
    n = f.size
    # convolution of f with itself already computed – we only need the active rows
    # conv_f already available in caller; here we build the constraint matrix.
    # Choose at most `max_active` indices with largest conv values.
    # This helper assumes the caller already selected `active_idx`.
    return None  # placeholder – actual building is done in the caller


def _lp_iterative_refine(f0: np.ndarray, deadline: float,
                         max_iter: int = 30, t: float = 0.3,
                         max_active: int = 150) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve a tiny LP that pushes mass to positions where the
    current convolution bound is active, and mix the result back into the
    current vector.  Returns the best vector seen and its surrogate score.
    """
    f = f0.copy()
    best_f = f.copy()
    best_score = _max_convolution_score(f)

    n = f.size
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        conv_f = _fft_self_convolve(f)
        M = conv_f.max()
        # pick the most‑binding constraints (top‑k)
        if conv_f.size > max_active:
            # argpartition is O(n), faster than full sort
            idx = np.argpartition(-conv_f, max_active - 1)[:max_active]
        else:
            idx = np.arange(conv_f.size, dtype=int)

        # Build sparse matrix A (|idx| × n) where A[k,i] = f[i] iff i contributes to (f*g)[k]
        row_ids = []
        col_ids = []
        data = []
        for r, k in enumerate(idx):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            cols = np.arange(i_start, i_end + 1, dtype=int)
            row_ids.extend([r] * cols.size)
            col_ids.extend(cols.tolist())
            data.extend(f[cols].tolist())
        A = scipy.sparse.csr_matrix((data, (row_ids, col_ids)),
                                    shape=(len(idx), n))
        b = np.full(len(idx), M, dtype=np.float64)

        # Linear programme: maximise Σg  ⇔ minimise –Σg
        c = -np.ones(n, dtype=np.float64)
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b,
                                    bounds=[(0, None)] * n,
                                    method='highs')
        if not res.success:
            # if the LP fails for numerical reasons we abort further refinement
            break

        g0 = res.x
        S = g0.sum()
        if S <= 1.0 + 1e-12:
            # No improvement possible – stop early
            break
        g = g0 / S                           # renormalise to a probability vector
        f = (1.0 - t) * f + t * g           # convex blend
        f = f / f.sum()                     # enforce exact simplex condition

        cur_score = _max_convolution_score(f)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_f = f.copy()
        # early stop if progress stalls
        if abs(cur_score - best_score) < 1e-12:
            break
    return best_f, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine respects the supplied time budget
    (≈ 1000 s) and returns the best vector discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose a base length.  If a previous solution exists we reuse its size,
    # otherwise we fall back to a moderate default (~3000) which keeps the LP fast.
    try:
        base_n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        base_n = 3000
    n = max(800, min(12000, base_n))

    # --------------------------------------------------------------
    # Helper to add a normalised candidate to the seed list.
    seeds: list[np.ndarray] = []

    def _add_candidate(vec: np.ndarray):
        a = np.asarray(vec, dtype=np.float64)
        if a.ndim != 1 or a.size == 0:
            return
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            return
        a /= s
        seeds.append(a)

    # --------------------------------------------------------------
    # Broad family sampling (few but diverse families)
    # 1) arcsine – the strongest performer so far
    a = _arcsine_distribution(n)
    _add_candidate(a)                                         # raw arcsine
    _add_candidate(order_alternate_extremes(np.sort(a)[::-1]))
    _add_candidate(order_center_outwards(np.sort(a)[::-1]))

    # 2) triangular
    a = _triangular_distribution(n)
    _add_candidate(a)
    _add_candidate(order_alternate_extremes(np.sort(a)[::-1]))

    # 3) symmetric beta with a range of shapes
    for p in (0.2, 0.5, 0.8, 1.2, 2.0, 5.0):
        a = _beta_symmetric_distribution(n, p)
        _add_candidate(order_alternate_extremes(np.sort(a)[::-1]))

    # 4) random Dirichlet draws – they occasionally stumble onto good patterns
    for alpha in (0.2, 0.6, 2.0):
        a = _random_dirichlet_alpha(n, alpha)
        _add_candidate(a)

    # 5) reuse the previous best if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        if prev.sum() > 0:
            prev = prev / prev.sum()
            _add_candidate(prev)
    except Exception:
        pass

    # --------------------------------------------------------------
    # Keep only the most promising seeds (based on cheap surrogate)
    def _surrogate(a):
        return _max_convolution_score(a)

    seed_scores = [(i, _surrogate(vec)) for i, vec in enumerate(seeds)]
    seed_scores.sort(key=lambda x: x[1])
    top_k = min(12, len(seed_scores))
    chosen_indices = [i for i, _ in seed_scores[:top_k]]
    seeds = [seeds[i] for i in chosen_indices]

    # --------------------------------------------------------------
    # Main optimisation loop
    best_seq = None
    best_score = np.inf

    for sid, seed_vec in enumerate(seeds):
        if time.time() >= deadline:
            break
        # Phase 1 – LP‑based polishing
        polished, lp_score = _lp_iterative_refine(seed_vec, deadline,
                                                  max_iter=30,
                                                  t=0.3,
                                                  max_active=150)
        # Phase 2 – short exponentiated‑gradient descent to clean up any remaining
        if time.time() < deadline:
            polished, eg_score = _exp_gradient_descent(polished, deadline,
                                                       max_iters=200_000,
                                                       eta0=0.5)
            final_score = _max_convolution_score(polished)
        else:
            final_score = lp_score

        if final_score < best_score:
            best_score = final_score
            best_seq = polished.copy()
            print(f"[{time.time():.1f}] new best from seed {sid+1}/{len(seeds)}: score={best_score:.6f}")

    # --------------------------------------------------------------
    # Fallback – if everything failed we return a uniform vector
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _max_convolution_score(best_seq)

    print(f"Final best surrogate score: {best_score:.6f} (length={best_seq.size})")
    # The evaluator rescales internally, so we can keep the ℓ₁‑norm equal to 1.
    return best_seq.tolist()
