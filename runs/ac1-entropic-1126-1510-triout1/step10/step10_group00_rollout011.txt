def _fft_convolve(a,b):
    size = 1 << ((len(a)+len(b)-2).bit_length())
    A = np.fft.rfft(a, n=size)

def random_distribution(n, rng):
    choice = rng.integers(0,5)
    if choice==0: # uniform
        return np.full(n, 1/n)
    elif choice==1: # arcsine
        vals = 1.0/np.sqrt((np.arange(n)+0.5)*(n - np.arange(n) - 0.5))
        vals /= vals.sum()
        return vals
    elif choice==2: # edge_weighted
        p = 10 ** rng.uniform(-2, 0.6) # random exponent
        vals = ( ( (np.arange(n)+0.5)*(n - np.arange(n) - 0.5) ) )**(-p)
        vals /= vals.sum()
        return vals
    elif choice==3: # beta sym
        p = 10 ** rng.uniform(-2, 0.6)
        x = (np.arange(n)+0.5)/n
        vals = np.exp((p-1)*(np.log(x) + np.log(1-x)))
        vals[~np.isfinite(vals)] = 0
        if vals.sum()==0:
            vals = np.full(n,1/n)
        else:
            vals/=vals.sum()
        return vals
    elif choice==4: # exponential decay
        alpha = rng.uniform(0.001, 0.5) # adjust
        d = np.minimum(np.arange(n)+0.5, n - np.arange(n) -0.5)
        vals = np.exp(-alpha * d)
        vals /= vals.sum()
        return vals
    elif choice==5: # triangular
        center = (n-1)/2.0
        vals = 1 - np.abs(np.arange(n)-center)/(center+1)
        vals = np.maximum(vals, 0)
        vals /= vals.sum()
        return vals
    else:
        # fallback random Dirichlet
        alpha = rng.uniform(0.01, 2)
        vals = rng.dirichlet(np.full(n, alpha))
        return vals
```

        gamma = init_gamma + (final_gamma - init_gamma) * min(1.0, t / 2000.0)  # schedule up to 2000 iters
        # to avoid overflow for large gamma, shift
        exp_term = np.exp(gamma * (conv - M))
        w = exp_term / exp_term.sum()
        # compute gradient = 2 * conv(w_rev, a)
        w_rev = w[::-1]
        grad_full = _fft_convolve(w_rev, a)
        L = len(w)
        grad = 2.0 * grad_full[L-1 : L-1 + n]
        # step size
        eta = lr0 / np.sqrt(t)
        # exponentiated update
        a = a * np.exp(-eta * grad)
        # renorm
        s = a.sum()
        if s == 0:
            a = np.full(n, 1.0/n)
        else:
            a = a / s

        # evaluation every 100 iterations
        if t % 100 == 0:
            cur = _objective(a)
            if cur < best_score - 1e-12:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth descent it={t} new best {best_score:.6f}")
        # safety break after many iterations
        if t >= 100000:
            break
    return best_seq, best_score

"""
propose_candidate – a time‑bounded stochastic optimiser.

The algorithm works in three phases:

1. **Broad random sampling** – many analytic families (uniform,
   arcsine, beta‑sym, edge‑weighted, exponential, triangular, sparse)
   are instantiated for a handful of lengths.  Random mixtures with the
   uniform distribution are also tried.  The 20‑30 best seeds are kept.

2. **Smooth exponentiated‑gradient descent** – for each seed we minimise
   the exact objective  
   `obj(a) = 2·n·max(conv(a,a))` (here `a` is a probability vector, i.e.
   `sum(a)=1`).  A smoothed version of the max (`log‑sum‑exp`) yields a
   tractable gradient that is computed by a single FFT‑based convolution.
   The update is `a ← a·exp(‑η·grad)` (mirror descent on the simplex)
   with a decreasing step‑size and a schedule that slowly sharpens the
   smoothing parameter `γ`.  Whenever a new best score is found it is
   recorded.

3. **Greedy pairwise mass transfer** – a cheap local walk swaps a small
   amount of mass between two positions whenever it improves the
   objective.  This polishes the final vector.

All work respects the 1000 s budget (2 s safety margin).  The returned
vector has non‑negative entries; scaling does not affect the score unless
clipping occurs, which we avoid by staying well below the 1000 limit.
"""

import time

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a*b (real) using FFT."""
    size = 1 << ((a.shape[0] + b.shape[0] - 2) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.shape[0] + b.shape[0] - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))  for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0) – clamp
    eps = np.finfo(float).tiny
    x = np.clip(x, eps, 1 - eps)
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = n - np.arange(n, dtype=np.float64) - 0.5
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential_decay(n: int, alpha: float) -> np.ndarray:
    """Mass decays exponentially from the centre towards the edges."""
    d = np.minimum(np.arange(n, dtype=np.float64) + 0.5,
                   n - np.arange(n, dtype=np.float64) - 0.5)
    w = np.exp(-alpha * d)
    w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    """Simple symmetric triangle (peak in the middle)."""
    center = (n - 1) / 2.0
    w = 1.0 - np.abs(np.arange(n, dtype=np.float64) - center) / (center + 1.0)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Smoothed gradient descent (mirror / exponentiated gradient) -------
def _smooth_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_gamma: float = 0.5,
    final_gamma: float = 20.0,
    lr0: float = 0.6,
    eval_interval: int = 100,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a smoothed version of the max‑convolution.
    Uses a log‑sum‑exp surrogate with a gradually increasing smoothing factor γ.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1

        # --- smoothed max -------------------------------------------------
        conv = _fft_self_convolve(a)                # length 2n‑1
        M = conv.max()
        # linear schedule for γ, saturating after ~2000 iterations
        gamma = init_gamma + (final_gamma - init_gamma) * min(1.0, t / 2000.0)

        # stable softmax
        shifted = conv - M
        exp_term = np.exp(gamma * shifted)
        w = exp_term / exp_term.sum()               # weights on convolution indices

        # --- gradient of the smoothed objective ---------------------------
        # grad_i = 2 * Σ_k w_k * a_{k-i}   (with zero‑padding)
        w_rev = w[::-1]
        grad_full = _fft_convolve(w_rev, a)        # length 3n‑2, we need the middle n entries
        L = len(w)                                 # = 2n‑1
        grad = 2.0 * grad_full[L - 1 : L - 1 + n]

        # --- exponentiated‑gradient update --------------------------------
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex
        s = a.sum()
        if s == 0.0:
            a = _distribution_uniform(n)
        else:
            a = a / s

        # --- bookkeeping --------------------------------------------------
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] smooth descent it={t} new global best {best_score:.6f}"
                )
        if t >= 150000:   # safety cap
            break
    return best_seq, best_score

# Greedy pairwise mass transfer ---------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small chunk of mass; keep move only if objective improves."""
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score
