"""
Propose a step‑function (list of non‑negative floats) with a low
``evaluate_sequence`` score.

Algorithm
---------

1. **Broad family scan** – for several lengths we generate many analytic
   families (uniform, arcsine, power‑edge, power‑abs, sine‑power,
   Gaussian, Dirichlet draws) and a few simple mixtures with a uniform
   component.  Every vector is projected onto the simplex (sum = 1) and
   the cheap fast score `2·n·max(a*a)` is evaluated via an FFT.  The best
   candidate (lowest score) is kept.

2. **Sub‑gradient exponential‑gradient descent** – starting from the best
   candidate we repeatedly:
        * compute the full self‑convolution,
        * locate the index *k* where the maximum occurs,
        * build the exact sub‑gradient of the max‑convolution (only the
          entries that contribute to *k* are non‑zero),
        * update the weights with a multiplicative (exponentiated)
          step, then project back onto the simplex.
   The learning rate decays from ≈5 to ≈0.01.  Every few hundred steps
   the true score is recomputed and the global best is recorded.

3. **Light simulated‑annealing fine‑tuning** – after the gradient phase
   a short temperature‑controlled random walk moves a tiny amount of
   mass between two positions; moves are accepted according to the
   Metropolis rule.  Improvements update the global best.

The whole routine respects the 1000 s wall‑clock budget; intermediate
best scores are printed.  The returned list is already normalised,
contains only non‑negative numbers (tiny eps added) and is ready for
``evaluate_sequence``.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector (sum = 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * float(conv.max())

def _dist_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

    i = np.arange(n, dtype=np.float64)
    w = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    w /= w.sum()
    return w

    w /= w.sum()
    return w

    w /= w.sum()
    return w

    return w

            sc = _max_convolution_score(mix)
            if sc < best_score:
                best_seq = mix.copy()
                best_score = sc
                best_n = n
                print(f"[{time.time():.1f}] n={n} mix‑uniform  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Incorporate previously computed sequence if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_seq = prev.copy()
                best_score = sc
                best_n = prev.size
                print(f"[{time.time():.1f}] previous best  score={sc:.6f}")
    except Exception:
        pass

    # If nothing was found (unlikely) fall back to uniform
    if best_seq is None:
        n = 3000
        best_seq = _dist_uniform(n)
        best_score = _max_convolution_score(best_seq)
        best_n = n
        print(f"[{time.time():.1f}] fallback uniform  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Sub‑gradient exponential‑gradient refinement
    a = best_seq.copy()
    n = a.shape[0]
    it = 0
    max_iters = 200_000

    while time.time() < deadline and it < max_iters:
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())

        # sub‑gradient for the max‑convolution entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        if i_end >= i_start:
            i_idx = np.arange(i_start, i_end + 1, dtype=int)
            j_idx = k - i_idx
            g[i_idx] = 2.0 * a[j_idx]

        # learning‑rate schedule (linear decay)
        frac = it / max_iters
        lr = 5.0 * (1.0 - frac) + 0.02

        # multiplicative update, projection onto simplex
        a = a * np.exp(-lr * g)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()

        # occasional exact evaluation
        if (it % 200) == 0:
            sc = _max_convolution_score(a)
            if sc < best_score - 1e-12:
                best_seq = a.copy()
                best_score = sc
                best_n = n
                print(f"[{time.time():.1f}] subgrad it={it}  new best={sc:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # 4. Light simulated‑annealing fine‑tuning (if time permits)
    remaining = deadline - time.time()
    if remaining > 5.0:
        a = best_seq.copy()
        cur_score = best_score
        T = max(cur_score * 0.02, 1e-6)
        T_min = cur_score * 1e-7
        decay = 0.9997
        while time.time() < deadline:
            # pick source proportionally to its mass
            src = np.random.choice(n, p=a)
            dst = random.randint(0, n - 1)
            if dst == src:
                continue
            delta = min(a[src] * 0.25, 0.015) * random.random()
            if delta < eps:
                continue
            a_new = a.copy()
            a_new[src] -= delta
            a_new[dst] += delta
            a_new = np.maximum(a_new, eps)
            a_new /= a_new.sum()
            new_score = _max_convolution_score(a_new)
            dE = new_score - cur_score
            if dE < 0 or random.random() < math.exp(-dE / T):
                a = a_new
                cur_score = new_score
                if new_score < best_score - 1e-12:
                    best_seq = a.copy()
                    best_score = new_score
                    best_n = n
                    print(f"[{time.time():.1f}] SA improve {new_score:.6f}")
            T = max(T * decay, T_min)

    # ------------------------------------------------------------------
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
