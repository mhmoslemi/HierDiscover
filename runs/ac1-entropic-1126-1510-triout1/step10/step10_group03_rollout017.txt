"""
Propose a step‑function (list of non‑negative floats) that minimises
`evaluate_sequence`.  The search consists of four stages, each respecting the
global time budget (default 1000 s).

1. **Broad exploration** – sweep a handful of lengths and several analytic
   families (uniform, symmetric Beta, asymmetric Beta, edge‑power
   distributions, random Dirichlet draws).  The best‑scoring candidate is kept.

2. **Smooth‑max mirror descent** – minimise a soft‑max approximation of the
   max‑convolution using an exponentiated‑gradient update.  The temperature
   α grows to 10⁶ while the learning rate decays, turning the smooth problem
   into the original one.

3. **Projected sub‑gradient descent with Adam‑style momentum** on the true
   max‑convolution (active‑set gradient) with a simple back‑tracking line
   search.

4. **Randomised mass‑transfer hill‑climber** – repeatedly move a small amount
   of probability mass from a high‑mass index to a random lower‑mass index,
   accepting moves that lower the peak of the convolution.

All helpers are defined at the module level (no closures) and only `numpy`
is required.  The final sequence is returned as a plain Python list.
"""

# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# FFT based convolutions ------------------------------------------------
def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a):
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score_normed(a):
    """
    For a vector already normalised to sum = 1 returns 2·n·max(a*a)
    – exactly the value produced by `evaluate_sequence` (apart from the
    scaling of the input, which does not affect the result).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families (all return a vector with sum = 1) -------------------
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta(n, p, q):
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_power(n, exponent, offset):
    """
    Deterministic edge‑biased shape:
        w_i ∝ ((i+0.5)*(n-i-0.5))^{-exponent} + offset
    """
    idx = np.arange(n, dtype=np.float64)
    base = ((idx + 0.5) * (n - idx - 0.5)) ** (-exponent)
    if offset > 0.0:
        base = base + offset
    vec = base
    vec = np.maximum(vec, 0.0)
    total = vec.sum()
    if total == 0 or not np.isfinite(total):
        vec = np.full_like(vec, 1.0 / n)
    else:
        vec /= total
    return vec


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The function respects the global time budget and
    returns the best vector found (as a plain list of floats).
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    eps = 1e-12

    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Helper to evaluate and possibly update the global best
    def _consider(vec: np.ndarray, n: int, family: str, params):
        nonlocal best_seq, best_score, best_n
        vec = np.maximum(vec, eps)          # avoid exact zeros
        vec /= vec.sum()
        sc = _max_convolution_score_normed(vec)
        if sc < best_score:
            best_seq = vec.copy()
            best_score = sc
            best_n = n
            print(f"[{time.time():.1f}] NEW BEST {sc:.6f} n={n} family={family} params={params}")

    # ------------------------------------------------------------------
    # 1️⃣  Broad grid search over lengths and analytic families
    lengths = [2000, 4000, 8000, 16000, 32000, 50000]

    # p values for symmetric / asymmetric Beta (fine near 0)
    p_grid = np.concatenate([
        np.linspace(0.01, 0.20, 40),
        np.linspace(0.21, 0.60, 40)
    ])

    edge_exps = np.linspace(0.2, 2.0, 19)          # step ≈0.1
    offsets = [0.0, 0.01, 0.05, 0.1]

    for n in lengths:
        if time.time() >= deadline:
            break

        # Uniform (baseline)
        _consider(_distribution_uniform(n), n, "uniform", None)

        # Symmetric Beta (including the classic arcsine p=0.5)
        for p in p_grid:
            if time.time() >= deadline:
                break
            _consider(_distribution_beta_sym(n, p), n, "beta_sym", (p,))

        # Asymmetric Beta (left‑heavy / right‑heavy, fixing the partner to 0.5)
        for p in p_grid:
            if time.time() >= deadline:
                break
            _consider(_distribution_beta(n, p, 0.5), n, "beta_asym_left", (p, 0.5))
            _consider(_distribution_beta(n, 0.5, p), n, "beta_asym_right", (0.5, p))

        # Edge‑biased deterministic power law
        for exp in edge_exps:
            if time.time() >= deadline:
                break
            for off in offsets:
                _consider(_distribution_edge_power(n, exp, off),
                          n, "edge_power", (exp, off))

        # A handful of random Dirichlet draws (concentration 1)
        for _ in range(5):
            if time.time() >= deadline:
                break
            vec = np.random.dirichlet(np.full(n, 1.0))
            _consider(vec, n, "dirichlet", None)

    # ------------------------------------------------------------------
    # 2️⃣  Incorporate previous best if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size:
            _consider(prev, len(prev), "previous", None)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Safeguard: if nothing was generated, fall back to uniform
    if best_seq is None:
        best_n = 2000
        best_seq = _distribution_uniform(best_n)
        best_score = _max_convolution_score_normed(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Smooth‑max mirror‑descent refinement (soft‑max of the convolution)
    a = best_seq.copy()
    n = a.shape[0]

    remaining = deadline - time.time()
    # Rough estimate: ~0.002 s per iteration (FFT + overhead)
    max_iter = int(max(0, remaining / 0.002))
    total_iter = max(5000, min(30000, max_iter))

    # exponential schedules for temperature and learning rate
    alpha_start, alpha_end = 0.5, 1e6
    lr_start, lr_end = 0.5, 0.001

    for it in range(total_iter):
        if time.time() >= deadline:
            break
        t = it / total_iter
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t

        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # soft‑max weights (numerically stable)
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth‑max objective
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1: n - 1 + n]

        # exponentiated‑gradient (mirror‑descent) update
        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()

        if (it % max(1, total_iter // 10) == 0) or (it == total_iter - 1):
            cur_sc = _max_convolution_score_normed(a)
            if cur_sc < best_score - 1e-12:
                best_seq = a.copy()
                best_score = cur_sc
                print(f"[{time.time():.1f}] mirror it={it}/{total_iter} new best {cur_sc:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Projected Adam‑style sub‑gradient descent on the true max‑convolution
    a = best_seq.copy()
    velocity = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    step0 = 0.5
    sub_it = 0

    while time.time() < deadline:
        sub_it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1: n - 1 + n]

        # Adam momentum (bias‑corrected)
        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** sub_it)

        step = step0
        while True:
            cand = a - step * v_hat
            cand = _project_onto_simplex(cand, 1.0)
            cand = np.maximum(cand, eps)
            cand /= cand.sum()
            cand_sc = _max_convolution_score_normed(cand)
            if cand_sc < best_score - 1e-13 or step < 1e-8:
                break
            step *= 0.5

        a = cand
        if cand_sc < best_score - 1e-13:
            best_seq = a.copy()
            best_score = cand_sc
            print(f"[{time.time():.1f}] subgrad it={sub_it} new best {cand_sc:.6f}")

        if sub_it % 1000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    # ------------------------------------------------------------------
    # 5️⃣  Random mass‑transfer hill‑climber (simple simulated‑annealing style)
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    best_M = conv.max()
    hill_it = 0

    while time.time() < deadline:
        hill_it += 1
        # Choose a source index proportional to its mass (more mass → more moves)
        src = np.random.choice(n, p=a)
        dst = np.random.randint(0, n)
        if src == dst:
            continue
        delta = a[src] * 0.5 * random.random()
        if delta < eps:
            continue

        a2 = a.copy()
        a2[src] -= delta
        a2[dst] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < best_M - 1e-12:
            a = a2
            conv = _fft_self_convolve(a)
            best_M = new_M
            best_seq = a.copy()
            best_score = 2.0 * n * best_M
            if hill_it % 100 == 0:
                print(f"[{time.time():.1f}] hill it={hill_it} new best {best_score:.6f}")

        if hill_it % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            best_M = conv.max()
            print(f"[{time.time():.1f}] hill jitter at it={hill_it}")

    final_score = _max_convolution_score_normed(best_seq)
    print(f"FINAL BEST ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
