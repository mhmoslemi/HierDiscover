def propose_candidate(seed=0, budget_s=1000):
    import time, random, math, numpy as np

    # set seeds
    np.random.seed(seed)
    random.seed(seed)
    eps = 1e-12

    start = time.time()
    deadline = start + budget_s - 0.3

    best_seq = None
    best_score = float('inf')
    best_n = None

    # Stage 1: search over lengths
    lengths = [1024, 2048, 4096, 8192, 16384, 32768]
    # optional random lengths up to 50000 as well
    for n in lengths:
        if time.time() >= deadline:
            break
        # deterministic families
        candidates = []
        # uniform
        candidates.append(_distribution_uniform(n))
        # arcsine
        candidates.append(_distribution_arcsine(n))
        # symmetric beta param grid
        p_grid = np.concatenate((np.linspace(0.02, 0.5, 15), np.logspace(-2, -0.1, 10)))
        for p in p_grid:
            candidates.append(_distribution_beta_sym(n, p))
        # asymmetric with q small 0.3 maybe
        for p in p_grid:
            candidates.append(_distribution_beta(n, p, 0.3))
            candidates.append(_distribution_beta(n, 0.3, p))
        # edge Dirichlet
        for bex in (0.3, 0.5, 0.7, 1.0):
            candidates.append(_distribution_edge_dirichlet(n, bex))
        # random Dirichlet draws
        for _ in range(5):
            conc = np.random.rand(n) * 5 + 0.001  # random concentration per coordinate
            vec = np.random.dirichlet(conc)
            candidates.append(vec)

        # evaluate each candidate
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.2f}] n={n} new best {sc:.6f}")

    # if previous solution available via global height_sequence_1
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.ndim == 1 and prev.size:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score, best_seq, best_n = sc, prev.copy(), len(prev)
                print(f"[{time.time():.2f}] previous best {sc:.6f}")
    except Exception:
        pass

    if best_seq is None:
        # fallback to uniform length 4096
        n = 4096
        best_seq = _distribution_uniform(n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        best_n = n
        print(f"[{time.time():.2f}] fallback uniform score {best_score:.6f}")

    # Stage 2: long mirror descent using smooth max exponent
    a = best_seq.copy()
    n = a.shape[0]

    remaining = deadline - time.time()
    # Rough cost per iteration for this n
    per_iter = 0.0015
    max_iters = int(max(0, remaining / per_iter))
    total_iters = max(20000, min(150000, max_iters))

    # schedule: temperature grows exponentially to raise max sharpness
    alpha_start, alpha_end = 1.0, 1e6
    lr_start, lr_end = 0.15, 0.0005

    for it in range(total_iters):
        if time.time() >= deadline:
            break
        t = it / total_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t

        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # compute softmax weights
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth max
        grad = 2.0 * _fft_convolve(a[::-1], w)[n-1:n-1+n]

        # exponentiated gradient update (mirror descent on simplex)
        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a, 1.0)
        # enforce numerical lower bound
        a = np.maximum(a, eps)
        a /= a.sum()

        if (it % max(1, total_iters // 20) == 0) or (it == total_iters-1):
            cur_score = _max_convolution_score(a)
            if cur_score < best_score - 1e-12:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time():.2f}] mirror it={it}/{total_iters} new best {cur_score:.6f}")

            tgt = sorted_idx[0]
        max_delta = min(a[src] * 0.5, 0.02)
        if max_delta < eps:
            # cannot move any mass, break
            break
        delta = max_delta
        improved = False
        # Try decreasing delta geometrically
        while delta > eps:
            a2 = a.copy()
            a2[src] -= delta
            a2[tgt] += delta
            # projection not needed as sum unchanged and entries remain >=0 (ensured)
            new_max = _fft_self_convolve(a2).max()
            if new_max < maxc - 1e-14:
                a = a2

                    best_score, best_seq = cur_score, a.copy()

                improved = True
                break
            delta *= 0.5
        if not improved:
            # random jitter to possibly escape plateau
            if random.random() < 0.01:
                jitter = np.random.randn(n) * 1e-5
                a = np.maximum(a + jitter, eps)
                a /= a.sum()
        # occasional recompute of gradient (already each loop)

    final_score = 2.0 * n * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()

# Simplex projection (Euclidean)
def _project_onto_simplex(v, s=1.0):
    """Project a vector onto the simplex {x ≥ 0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    """Full linear self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

    """Full linear convolution of two real vectors using a real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

    x = (np.arange(n, dtype=np.float64) + 0.5) / n
