def _smooth_weights(conv, alpha):

    max_conv = conv.max()
    # compute exponent weight relative to max: exp(alpha * (conv - max))
    # conv <= max, so exponent <= 0 => w <= 1.
    w = np.exp(alpha * (conv - max_conv))
    # Normalize sum to 1. But conv length is 2n-1; ensure w sum not zero.
    total = w.sum()
    if total == 0:
        # fallback to uniform weight on active set (single max)
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w = w / total
    return w

def _smooth_subgradient_descent(a, deadline, best_seq, best_score, lr0=0.7, alpha_max=30., max_iter=20000, eval_interval=200):

    n = a.shape[0]

    for t in range(1, max_iter+1):

        if time.time() >= deadline: break

        conv = _fft_self_convolve(a)  # length 2n-1

        maxc = conv.max()

        # compute objective (denominator sum = 1)
        obj = 2.0 * n * maxc

        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"[{time.time():.1f}] smooth descent iter={t} new best {best_score:.6f}")

        # compute smoothing weight
        # schedule alpha: increase linearly up to alpha_max
        alpha = alpha_max * min(1.0, t / (max_iter // 2))  # ramp up over half iterations

        # using more stable method: compute w = np.exp(alpha * (conv - maxc))
        # conv-maxc is non-positive.
        # If alpha=0 produce uniform weights.

        if alpha == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            # subtract maxc to avoid overflow of exponent large positive (cur conv-maxc <= 0)
            # compute exp(alpha * (conv - maxc))
            exp_vals = np.exp(alpha * (conv - maxc))
            w_sum = exp_vals.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                # fallback to active set uniform
                mask = conv >= maxc - 1e-14
                cnt = mask.sum()
                w = mask.astype(np.float64) / cnt
            else:
                w = exp_vals / w_sum

        # compute gradient
        grad = _gradient_from_weights(a, w)  # already includes factor 2

        # choose step size
        eta = lr0 / math.sqrt(t)

        # exponentiated update with clipping
        # Clip gradients to avoid overflow
        grad_clipped = np.clip(grad, -50.0, 50.0)
        a = a * np.exp(-eta * grad_clipped)
        total = a.sum()
        if total == 0 or not np.isfinite(total):
            a = np.full(n, 1.0 / n)
        else:
            a = a / total

    return best_seq, best_score

"""
Hybrid heuristic search for a step‑function minimizing the evaluation
objective.  The algorithm consists of three stages:
1) a broad sweep of analytic probability‐mass families (uniform, arcsine,
   beta‑sym/​asym, power, exponential, two‑block, Dirichlet,
   sparse) for several candidate lengths.  The best few seeds are kept.
2) a smooth sub‑gradient descent on the simplex.  The true max in the
   objective is approximated by a softmax (temperature controlled by a
   linearly increasing parameter α); the resulting sub‑gradient is
   computed via FFT‑based cross‑correlation and an exponentiated‑gradient
   (mirror‑descent) update is performed with a step‑size η_t=lr/√t.
   The best vector seen during the descent is retained.
3) a cheap greedy pairwise‑mass‑transfer “local swap’’ phase to polish
   the result.

All work respects the global deadline (≈ 1000 s).  The returned list
contains non‑negative numbers whose sum is 1, ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the sub‑gradient 2·a⋆w where w is a weight vector (length 2n‑1)
    that sums to 1. The operation is performed via an FFT‑based
    cross‑correlation: grad_i = 2·∑_j w_{i+j}·a_j.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2·n‑1
    a_rev = a[::-1]                    # reversed a
    size = 1 << ((n + m - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1                      # index where shift = 0 begins
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (sum is forced to 1 here)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _smooth_subgradient_descent(a: np.ndarray, deadline: float,
                               best_seq: np.ndarray, best_score: float,
                               lr0: float = 0.7, alpha_max: float = 30.0,
                               max_iter: int = 12000,
                               eval_interval: int = 200) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using a smoothed version of the
    max‑convolution objective.  The smoothing parameter α grows linearly
    from 0 to `alpha_max` during the run.
    """
    n = a.shape[0]
    t = 0
    log_a = np.log(a + 1e-300)      # start in log‑domain

        if t % eval_interval == 0 and time.time() + 5.0 >= deadline:

"""
Hybrid heuristic search for a step‑function minimizing the evaluation
objective.  The algorithm works in three stages:

1. **Broad analytic sweep** – generates a variety of probability‑mass
   vectors (uniform, arcsine, symmetric/asymmetric beta, power‑law,
   exponential, two‑block spikes, Dirichlet, sparse).  A handful of the
   best seeds across several lengths are retained.

2. **Smoothed sub‑gradient descent** – approximates the true
   `max(conv(a,a))` by a soft‑max with a temperature that grows linearly.
   The resulting (convex) sub‑gradient is obtained via FFT‑based
   cross‑correlation and an exponentiated‑gradient (mirror‑descent)
   update is performed on the simplex.  The best vector seen during the
   descent is recorded.

3. **Local polishing** – a cheap greedy pairwise‑mass‑transfer walk that
   accepts a move only when the exact objective improves.

All work respects a global deadline (≈ 1000 s).  The returned list
contains non‑negative numbers whose sum is 1, ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient 2·a⋆w where w (length 2n‑1) sums to 1.
    Computed via FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2·n‑1
    a_rev = a[::-1]
    size = 1 << ((n + m - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    return 2.0 * conv[start:start + n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (sum forced to 1 here)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals

def _smooth_subgradient_descent(a: np.ndarray, deadline: float,

    The smoothing parameter α grows linearly from 0 to `alpha_max`.

            print(f"[{time.time():.1f}] smooth descent it={t} new best {best_score:.6f}")

                lr0=0.5,
                alpha_max=20.0,
                max_iter=8000,
                eval_interval=200,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] perturbed descent new best {overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 7. Final polishing with cheap local swaps
    if time.time() < deadline and overall_best_seq is not None:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] after final swaps best score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
