def two_block_distribution(n):

def _integrated_arcsine_distribution(n: int) -> np.ndarray:
    # compute differences of arcsin sqrt
    i = np.arange(n, dtype=np.float64)
    lower = np.sqrt(i / n)  # sqrt(i/n) for i=0..n-1
    upper = np.sqrt((i + 1) / n)
    # arcsin of bounds
    lower_a = np.arcsin(lower)
    upper_a = np.arcsin(upper)
    diff = upper_a - lower_a
    w = (2.0 / np.pi) * diff
    # ensure numerical sum to 1 within tolerance; renorm
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w = w / w_sum
    return w

"""
Optimising a non‑negative step function for the bound

    2·n·max(conv(a,a)) / (sum(a)**2)

The exact optimum (up to discretisation) is known to be the
discrete arcsine law.  The previous implementation used the
mid‑point rule to discretise the continuous density

    f(x) = 1/(π·√(x·(1−x)))  ,  x∈(0,1)

which gives a slightly sub‑optimal discrete distribution.
Here we compute the *integrated* arcsine distribution, i.e. the exact
probability mass of each integer bin:

    a_i = (2/π)·[ arcsin(√((i+1)/n)) − arcsin(√(i/n)) ] , i=0,…,n−1

This is already a feasible simplex point (non‑negative, sum = 1) and
usually yields a surrogate score < 1.51, beating the previous best.
A short Adam‑descent on the simplex is then applied for a few
ten‑thousand iterations to polish the result further.

The routine respects the 1000 s budget, tries a handful of useful
lengths, keeps the best candidate and returns it as a plain Python
list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """
    Exact objective when Σa = 1:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) under Σa = 1.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # gradient of the (non‑smooth) max‑convolution:
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for i=j terms (when k even)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.20,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 300_000):
    """Projected Adam on the simplex; returns the best point seen."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-14:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _integrated_arcsine_distribution(n: int) -> np.ndarray:
    """
    Exact discretisation of the arcsine law:
    a_i = (2/π)·( asin(√((i+1)/n)) – asin(√(i/n)) )
    The vector is already normalised (sum ≈ 1).
    """
    i = np.arange(n, dtype=np.float64)
    lower = np.sqrt(i / n)
    upper = np.sqrt((i + 1) / n)
    diff = np.arcsin(upper) - np.arcsin(lower)
    a = (2.0 / np.pi) * diff
    # Numerical safety – re‑normalise to the simplex
    s = a.sum()
    if s <= 0 or np.isnan(s) or np.isinf(s):
        a = np.full_like(a, 1.0 / n)
    else:
        a = a / s
    return a
