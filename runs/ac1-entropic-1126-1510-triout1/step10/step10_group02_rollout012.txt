Propose a non‑negative step function (list of floats) that minimises
``evaluate_sequence``.

Algorithm outline
-----------------
1. Broad exploration – generate several analytic families (uniform,
   arcsine, beta, sine‑power, exponential, etc.) for a handful of
   lengths and keep the few elite candidates (lowest objective).

2. Frank‑Wolfe style refinement – for each elite start we repeatedly
   solve a small linear program

        max   Σ g_j
        s.t.  (f * g) ≤ M               (M = max(f*f))
              g ≥ 0

   where ``f`` is the current vector.  The LP gives a direction ``g``.
   We normalise ``g`` to sum = 1 and move a fraction ``t`` towards it
   (convex combination).  This step provably never increases the
   objective and usually reduces it noticeably.  A few iterations are
   enough to drive the score well below the previous best.

3. Light hill‑climber – after the LP phase we perform a cheap random
   mass‑transfer hill‑climbing that occasionally drops tiny entries.
   This can squeeze a little more out of the solution.

The whole search respects a 1000 s budget (2 s safety margin).  All
vectors are kept on the simplex (non‑negative, sum = 1), which makes

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) for a vector already normalised to sum = 1.
    This is exactly the value that evaluate_sequence would return.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

    vals /= vals.sum()

def _distribution_triangular(n):
    half = n // 2

    vals /= vals.sum()
    return vals

    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n

        w /= w.sum()
    return w

# Frank‑Wolfe LP refinement ------------------------------------------------
def _lp_step_max_sum(f, M):
    """
    Solve the LP  max Σ g_j   s.t.  f * g ≤ M  (componentwise), g ≥ 0.
    Returns the optimal vector g (may be all zeros if something goes wrong).
    """
    n = f.shape[0]
    # Build convolution matrix A where (A @ g)[k] = (f * g)[k]
    c = np.concatenate([f, np.zeros(n - 1)])
    r = np.concatenate([f[:1], np.zeros(n - 1)])
    A = toeplitz(c, r)                # shape (2n‑1, n)

    # Linear program: minimise -Σ g_j   (i.e. maximise Σ g_j)
    obj = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    # SciPy HiGHS (fast) – tolerates dense A
    res = linprog(obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})

    if not res.success:
        # fallback to zero vector (always feasible)
        return np.zeros_like(f)
    return res.x

                        a = a2
                        M = new_M
                        cur_sc = 2.0 * n * M
                        if cur_sc < best_score - 1e-12:
                            best_score = cur_sc
                            best_seq = a.copy()
                            print(f"[{time.time():.1f}] Hillclimb improved to {cur_sc:.6f}")

    # ------------------------------------------------------------------
    # 6. Final sanity check and return
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
