"""
Propose a non‑negative step function (list of floats) that minimises
`evaluate_sequence`.  The search runs under a 1000 s budget and consists of:

1. **Fast family sweep** – several lengths (≈ 2‑8 k) are examined with many
   analytic families (uniform, arcsine, Beta, exponential and power‑law
   distributions) plus a few random Dirichlet draws.  The best normalised
   vector is kept.

2. **Heavy‑weight smooth‑max mirror descent** – a soft‑max approximation of the
   max‑convolution is minimised with many (≈ 10⁵) multiplicative updates.
   The temperature grows to ~5·10⁵ while the learning rate decays, giving a
   long, fine‑grained optimisation phase.

3. **Projected Adam on the true max‑convolution** – after the smooth phase a
   sub‑gradient of the exact max‑convolution is used together with Adam‑style
   momentum to chase the remaining improvement.

4. **Targeted mass‑transfer hill‑climbing** – the pair that contributes the
   largest product to the current peak is identified and a small amount of
   mass is moved to the smallest entry; this locally reduces the peak.
   Random jitter is added occasionally to escape plateaus.

All steps respect the simplex constraint (non‑negative, sum = 1) by means of
Euclidean projection.  The algorithm is deliberately time‑aware; it never
exceeds the supplied budget.
"""

import time
import random

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of a vector onto the simplex {x ≥ 0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Returns 2·n·max(a*a) for a vector already normalised to sum = 1.
    This is exactly the value returned by ``evaluate_sequence`` (up to the
    clamping that never occurs for the vectors we generate).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families (all return a vector with sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    logw -= np.max(logw)               # stability shift
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_beta(n, p, q):
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_exponential(n, lam):
    """a_i ∝ exp(-lam·i)  (lam > 0, i = 0…n‑1)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-lam * i)
    w /= w.sum()
    return w


def _distribution_powerlaw(n, p):
    """a_i ∝ (i+1)^{-p}  (p > 0)."""
    i = np.arange(n, dtype=np.float64) + 1.0
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_dirichlet(n, conc):
    """Simple Dirichlet draw with given concentration vector."""
    vec = np.random.dirichlet(conc)
    # ensure no exact zeroes
    eps = 1e-15
    vec = np.maximum(vec, eps)
    vec /= vec.sum()
    return vec

    p_pow_grid = [0.8, 1.2, 1.6, 2.2]        # power‑law exponents
    conc_params = [0.5, 1.0, 2.0]            # Dirichlet concentrations

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- deterministic families -----------------------------------
        families = []

        families.append(_distribution_uniform(n))
        families.append(_distribution_arcsine(n))

        for p in p_grid:
            families.append(_distribution_beta_sym(n, p))

        for lam in lam_grid:
            families.append(_distribution_exponential(n, lam))

        for p in p_pow_grid:
            families.append(_distribution_powerlaw(n, p))

        # asymmetric beta (left‑heavy)
        for p in p_grid:
            families.append(_distribution_beta(n, p, 0.5))
            families.append(_distribution_beta(n, 0.5, p))

        # ----- random Dirichlet draws -----------------------------------
        for conc in conc_params:
            for _ in range(3):
                families.append(_distribution_dirichlet(n, np.full(n, conc)))

        # ------------------------------------------------------------------
        # evaluate all candidates for this length
        # ------------------------------------------------------------------
        for vec in families:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score - 1e-12:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n} new best {sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Incorporate a previously discovered sequence if it exists.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score - 1e-12:
                best_score, best_seq, best_n = sc, prev.copy(), len(prev)
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Safeguard – never return an empty list
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 3000
        best_seq = _distribution_uniform(n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        best_n = n
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Smooth‑max mirror descent – many cheap iterations
    # ------------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]

    remaining = deadline - time.time()
    # allocate roughly 70 % of remaining time to this phase
    mirror_budget = remaining * 0.70
    est_iter_time = 0.0012                     # measured on modest n
    max_iters = int(max(1, mirror_budget / est_iter_time))
    # enforce a lower bound so that we actually run a sizable optimisation
    if max_iters < 20000:
        max_iters = int(mirror_budget / 0.0015)

    # geometric schedules
    alpha_start, alpha_end = 0.5, 5e5          # temperature
    lr_start, lr_end = 0.12, 0.001            # learning rate

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        t = it / max_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t

        # ---- smooth‑max gradient ------------------------------------
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth‑max (mirrored via convolution)
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()

        # occasional exact‑score check
        if (it % max(1, max_iters // 10) == 0) or (it == max_iters - 1):
            cur_sc = _max_convolution_score(a)
            if cur_sc < best_score - 1e-12:
                best_score, best_seq = cur_sc, a.copy()
                print(f"[{time.time():.1f}] mirror it={it}/{max_iters} new best {cur_sc:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Projected Adam on the true max‑convolution (sub‑gradient)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    velocity = np.zeros_like(a)
    beta1 = 0.9
    step0 = 0.45
    sub_it = 0
    while time.time() < deadline - 5.0:   # keep a tiny buffer for the last phase
        sub_it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        # Use uniform weights over the active set (any convex combination works)
        w = np.zeros_like(conv)
        w[active] = 1.0 / max(1, active.size)

        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam‑style momentum (bias‑corrected)
        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** sub_it)

        # simple decay of the step size
        step = step0 / math.sqrt(sub_it)

        a_cand = a - step * v_hat
        a_cand = _project_onto_simplex(a_cand, 1.0)
        a_cand = np.maximum(a_cand, eps)
        a_cand /= a_cand.sum()
        cand_sc = _max_convolution_score(a_cand)

        if cand_sc < best_score - 1e-12:
            best_score, best_seq = cand_sc, a_cand.copy()
            print(f"[{time.time():.1f}] Adam subgrad it={sub_it} new best {cand_sc:.6f}")

        a = a_cand

        # occasional random jitter to avoid tiny plateaus
        if sub_it % 800 == 0:
            jitter = np.random.randn(n) * 5e-6
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

    # ------------------------------------------------------------------
    # 5️⃣  Targeted mass‑transfer hill‑climber (polish)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    hill_it = 0
    while time.time() < deadline:
        hill_it += 1
        # location of the current peak
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # pair (i,j) that actually contributes the largest product
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        # move a small chunk of mass from the heavier of the pair to the global minimum
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.argmin(a)
        if target == source:
            target = np.argsort(a)[1]   # second smallest

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()

        if new_M < M - 1e-12:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M
            best_seq = a.copy()
            best_score = 2.0 * n * M
            if hill_it % 100 == 0:
                print(f"[{time.time():.1f}] hill it={hill_it} new best {best_score:.6f}")

        # occasional jitter to keep the walk alive
        if hill_it % 2500 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] hill jitter at it={hill_it}")

    final_score = 2.0 * n * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
