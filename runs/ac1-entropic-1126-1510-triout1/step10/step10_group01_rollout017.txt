"""
Hybrid search for the step‑function optimisation problem.

The algorithm proceeds in three stages under a 1000 s budget:

1. **Broad seeding** – generate a modest collection of analytic families
   (uniform, beta, power‑law, exponential, sparse, two‑block, Dirichlet).
   All candidates are normalised to sum = 1 and evaluated with a fast
   FFT‑based objective ``2·n·max(conv(a,a))`` (the sum‑square factor is 1).

2. **Deterministic sub‑gradient descent** – the current best few seeds are
   refined with the “multimax’’ sub‑gradient method (active‑set
   weighting) from the reference solution.  Only a few thousand iterations
   are performed for each seed, keeping the cost low.

3. **Stochastic simulated‑annealing** – starting from the best sequence
   found so far, we repeatedly transfer a random fraction of mass from one
   index to another, accepting worsening moves with a temperature schedule.
   This helps escape the local minima that trap plain greedy swaps.
   After the annealing run a cheap greedy pairwise‑swap polish is applied.

All helper functions are top‑level; the main entry point is
``propose_candidate`` which returns the best list of non‑negative heights
found within the allotted time.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (result length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of `evaluate_sequence` for a normalised sequence."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Return a modest pool of analytic seeds for a given length."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    for p in (0.05, 0.1, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0):
        cand.append(_distribution_beta_sym(n, p))

    for _ in range(4):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))

    for lam in (0.001, 0.005, 0.01, 0.02, 0.05):
        cand.append(_distribution_exponential_increasing(n, lam))

    for block_frac in (0.05, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac))

    # a few Dirichlet draws
    for _ in range(3):
        cand.append(_distribution_dirichlet(n, rng, alpha=0.3 + 0.7 * rng.random()))

    cand.append(_distribution_random_sparse(n, rng, density=0.01))

    return cand

# Sub‑gradient descent (multimax) -------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Sub‑gradient 2·a⋆w for a weight vector w (length 2·n‑1)."""
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + (2 * n - 1) - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    return 2.0 * conv[start:start + n]


def _subgradient_multimax_descent(a: np.ndarray, deadline: float,
                                 best_seq: np.ndarray, best_score: float,
                                 lr0: float = 0.5, max_iter: int = 20000,
                                 eval_interval: int = 200) -> tuple[np.ndarray, float]:
    """Exponential‑gradient descent using the full active set."""
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt
        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad it={t} new best {best_score:.6f}")
    return best_seq, best_score

# Greedy local mass‑transfer swaps ---------------------------------------
def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Simple hill‑climbing moves: transfer a random fraction of mass."""
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score

# Simulated annealing ----------------------------------------------------
def _simulated_annealing(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float,
                         rng: np.random.Generator,
                         max_step: float = 0.30) -> tuple[np.ndarray, float]:
    """
    Random mass transfers with a temperature schedule.
    Moves that increase the objective are accepted with
    probability exp(-(Δ)/T).  The temperature decays exponentially.
    """
    n = a.shape[0]
    cur = a.copy()
    cur_score = _objective(cur)

    start = time.time()
    total = max(deadline - start, 1e-9)

    T0 = cur_score * 0.10          # initial temperature
    Tf = cur_score * 1e-6          # final temperature (practically zero)

    while time.time() < deadline:
        # temperature according to elapsed fraction
        elapsed = time.time() - start
        frac = elapsed / total
        T = T0 * ((Tf / T0) ** frac)

        # choose a source with positive mass
        src_candidates = np.where(cur > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        # choose a distinct destination
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction of the source mass
        delta = cur[i] * max_step * rng.random()
        if delta < 1e-15:
            continue

        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        new_score = _objective(new)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if T > 0.0:
                prob = math.exp(-(new_score - cur_score) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = cur.copy()
                print(f"[{time.time():.1f}] anneal new best {best_score:.6f}")

    return best_seq, best_score
