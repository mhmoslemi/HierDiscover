def propose_candidate(seed=1234, budget_s=1000):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2

    lengths = [2000, 4000, 8000, 12000, 16000]
    # allow about 5000 iterations for mixture search per length maybe.

    best_seq = None
    best_score = float('inf')
    best_n = None

    for n in lengths:
        if time.time() >= deadline:
            break
        families = _family_vectors(n)  # list of ndarray shape (n,).
        K = len(families)
        # Precompute families as 2D array for vectorized mixing? Maybe use matrix.
        # However mixing can be done via weighted sum.
        # Determine number of samples based on remaining time: ensure some min number.
        # For simplicity, we can set samples = int(5000 * (deadline - time.time()) / budget_s)
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # estimate per-sample time ~ 0.0015 sec per evaluation for n=12000 ~ maybe 0.004 sec. Estimate allow 1e-3.
        # Let's allocate about 5000 * remaining / (budget_s) samples, but cap.
        max_samples = 5000
        samples = int(min(max_samples, remaining * 50))  # approximate 50 eval/s.
        samples = max(samples, 200)  # at least some
        for i in range(samples):
            if time.time() >= deadline:
                break

            conc = random.choice([0.2, 0.5, 1.0, 2.0, 5.0])
            w_raw = np.random.dirichlet(np.full(K, conc))
            # Occasionally produce extreme weight: set one weight =1 (choose each uniformly)

                idx = random.randrange(K)
                w_raw = np.zeros(K)
                w_raw[idx] = 1.0
            mix = np.tensordot(w_raw, families, axes=1)  # shape (n,)
            # Optionally shuffle mix
            if random.random() < 0.1:
                np.random.shuffle(mix)
            # Ensure positivity
            mix = np.maximum(mix, 1e-12)
            mix /= mix.sum()
            score = _max_convolution_score(mix)
            if score < best_score - 1e-12:
                best_score = score
                best_seq = mix.copy()
                best_n = n
                print(f"[{time.time():.1f}] Length {n}, new best {score:.6f}")

        # Keep the best few for polishing
        # After loops store them maybe.

    # Now run polishing on best_seq
    if best_seq is not None and time.time() < deadline:
        # small number of polish steps, maybe subgradient
        a = best_seq.copy()
        # Mirror descent optional
        a, sc = _smooth_max_mirror_descent(a, deadline)

"""
Propose a non‑negative step function (list of floats) that minimises
`evaluate_sequence`.  The algorithm works in three phases:

1.  Broad exploration – a handful of analytic families (uniform, arcsine,
    triangular, exponential, beta, sine‑power, cosine‑window, gaussian,
    power‑law) together with many random Dirichlet draws are generated.
    The best few candidates are retained.

2.  Intensive refinement – the best candidates are refined with a
    smooth‑max mirror‑descent (exponentiated‑gradient) routine that
    gradually sharpens the smoothed max (parameter α grows from 1 to 1e6).
    Afterwards a true sub‑gradient Adam‑style step is applied.

3.  Light hill‑climber – a few random mass‑transfer moves are performed
    to escape tiny plateaus.

All calculations respect the 1000 s time budget.  The first candidate
found is returned as a plain Python list.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Core utilities ---------------------------------------------------------

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families – each returns a vector with sum = 1 ------------------
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_powerlaw(n, p):
    """Monotone decreasing power‑law a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _family_vectors(n):
    """A modest collection of analytic families, all normalised to sum = 1."""
    fam = [
        _distribution_uniform(n),
        _distribution_arcsine(n),
        _distribution_triangular(n),
        _distribution_exponential(n, lam=0.003),
        _distribution_exponential(n, lam=0.02),
        _distribution_beta_sym(n, p=0.5),
        _distribution_beta_sym(n, p=2.0),
        _distribution_sine_power(n, power=2.0),
        _distribution_cosine_window(n, coeff=0.5),
        _distribution_powerlaw(n, p=2.0),
        _distribution_powerlaw(n, p=2.0)[::-1],
    ]
    # a few random Dirichlet draws for diversity
    for a in (0.2, 0.5, 1.0, 2.0):
        fam.append(np.random.dirichlet(np.full(n, a)))
    return fam


# ----------------------------------------------------------------------
# Local optimisation -----------------------------------------------------

def _smooth_max_mirror_descent(a, deadline):
    """Exponentiated‑gradient mirror descent on a smoothed max."""
    n = a.shape[0]
    eps = 1e-12
    # schedule – (α0, α1, steps, lr0, lr1)
    schedule = [
        (1.0, 1e2, 3000, 0.20, 0.02),
        (1e2, 1e4, 3000, 0.02, 0.002),
        (1e4, 1e6, 3000, 0.002, 2e-04),
    ]

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    for (alpha0, alpha1, steps, lr0, lr1) in schedule:
        for it in range(steps):
            if time.time() >= deadline:
                break
            t = it / steps
            alpha = alpha0 * (alpha1 / alpha0) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of the smooth‑max
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if it % 500 == 0:
                sc = _max_convolution_score(a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] Mirror it={it} new best {sc:.6f}")

        if time.time() >= deadline:
            break

    return best_a, best_sc


def _projected_subgradient(a, deadline):
    """True sub‑gradient descent with Adam‑style momentum & back‑tracking."""
    n = a.shape[0]
    eps = 1e-12
    velocity = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    step0 = 0.5
    it = 0

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam‑style update
        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** it)

        # back‑tracking line search
        step = step0
        while True:
            a_candidate = a - step * v_hat
            a_candidate = _project_onto_simplex(a_candidate, 1.0)
            a_candidate = np.maximum(a_candidate, eps)
            a_candidate /= a_candidate.sum()
            cand_sc = _max_convolution_score(a_candidate)
            if cand_sc < best_sc - 1e-12 or step < 1e-8:
                break
            step *= 0.5

        a = a_candidate
        if cand_sc < best_sc - 1e-12:
            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] Subgrad it={it} new best {cand_sc:.6f}")

        if it % 2000 == 0:
            # tiny jitter to escape tiny plateaus
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    return best_a, best_sc


def _light_hillclimber(a, deadline, steps=2000):
    """Random mass‑transfer moves that keep the simplex constraint."""
    n = a.shape[0]
    eps = 1e-12
    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    for it in range(steps):
        if time.time() >= deadline:
            break
        # pick two distinct indices
        i, j = random.sample(range(n), 2)
        if a[i] < eps:
            continue
        delta = a[i] * 0.02 * random.random()   # move ≤2 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        sc = _max_convolution_score(a2)
        if sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
            a = a2
            # occasional log
            if it % 250 == 0:
                print(f"[{time.time():.1f}] Hillclimb it={it} new best {sc:.6f}")
        else:
            # revert to previous a (keeps simple)
            pass
    return best_a, best_sc


# ----------------------------------------------------------------------
# Candidate generation & main driver -------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 5.0   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. Broad exploration on a modest set of lengths
    lengths = [2000, 3000, 4000, 5000, 6000, 8000]
    elite = []                     # (score, vector) – keep top‑6 overall
    for n in lengths:
        if time.time() >= deadline:
            break
        families = _family_vectors(n)
        # add a few pure random Dirichlet draws for extra diversity
        for _ in range(30):
            families.append(np.random.dirichlet(np.full(n, random.choice([0.1, 0.3, 0.7, 1.5]))))

        for vec in families:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, 1e-12)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                print(f"[{time.time():.1f}] Length {n} new global best {sc:.6f}")

            # maintain a small elite list
            if len(elite) < 6:
                elite.append((sc, vec.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, vec.copy())
                    elite.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 2. If a previous best (height_sequence_1) is available, use it
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 1e-12)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                print(f"[{time.time():.1f}] Using previous best {sc:.6f}")
            if len(elite) < 6:
                elite.append((sc, prev.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, prev.copy())
                    elite.sort(key=lambda x: x[0])
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Refine a few elite candidates (up to 3 restarts)
    restarts = min(3, len(elite))
    for idx in range(restarts):
        if time.time() >= deadline:
            break
        a0 = elite[idx][1].copy()
        print(f"[{time.time():.1f}] Restart {idx+1}/{restarts} length={a0.shape[0]} start score={_max_convolution_score(a0):.6f}")

        # smooth‑max mirror descent
        a_mid, sc_mid = _smooth_max_mirror_descent(a0, deadline)

        # true sub‑gradient polishing
        a_mid, sc_mid = _projected_subgradient(a_mid, deadline)

        # light hill‑climber
        a_mid, sc_mid = _light_hillclimber(a_mid, deadline, steps=1500)

        if sc_mid < best_score - 1e-12:
            best_score = sc_mid
            best_seq = a_mid.copy()
            print(f"[{time.time():.1f}] New global best after restart {sc_mid:.6f}")

    # ------------------------------------------------------------------
    # 4. Final sanity – ensure non‑negativity, normalisation
    best_seq = np.maximum(best_seq, 1e-12)
    best_seq /= best_seq.sum()

    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
