def mirror_descent(a, deadline, best_score):

   n = a.shape[0]
   it = 0
   # schedule parameters
   total_est = 200000
   alpha_start = 0.1
   alpha_target = 500.0
   lr_start = 10.0
   lr_end = 0.001
   eps = 1e-12

   while time.time() < deadline and it < total_est:

       ratio = it / total_est
       alpha = alpha_start * (alpha_target / alpha_start) ** ratio
       lr = lr_start * (lr_end / lr_start) ** ratio

def simulated_annealing(a, deadline, best_score):

   n = a.shape[0]
   cur = a.copy()
   cur_score = best_score
   T = max(cur_score * 0.01, 1e-7)
   T_min = cur_score * 1e-7
   decay = 0.9995
   eps = 1e-12

   while time.time() < deadline:
       # source selection using probs a
       source = np.random.choice(n, p=cur)  # probability from cur (ensures > 0)
       target = random.randint(0, n - 1)
       if target == source: continue
       delta = min(cur[source] * 0.3, 0.02) * random.random()
       if delta < eps: continue
       new = cur.copy()
       new[source] -= delta
       new[target] += delta
       new = np.maximum(new, eps)
       new = new / new.sum()
       new_score = 2.0 * n * np.max(_fft_self_convolve(new))
       dE = new_score - cur_score
       if dE < 0 or random.random() < math.exp(-dE / T):
           cur = new
           cur_score = new_score
           if new_score < best_score - 1e-12:
               best_seq = cur.copy()
               best_score = new_score
               print(f"[{time.time():.1f}] SA new best {new_score}")
       T = max(T * decay, T_min)
   return best_seq, best_score, cur

def hill_climber(a, deadline, best_score):

   n = a.shape[0]
   a = a.copy()
   conv = _fft_self_convolve(a)
   M = conv.max()
   eps = 1e-12
   iter_cnt = 0

   while time.time() < deadline:
       iter_cnt += 1
       # find max index
       k = int(conv.argmax())
       # compute i range
       i_low = max(0, k - (n - 1))
       i_high = min(n - 1, k)
       # compute products vectorized
       i_range = np.arange(i_low, i_high + 1)
       j_range = k - i_range
       products = a[i_range] * a[j_range]
       idx_max = products.argmax()
       i_star = i_range[idx_max]
       j_star = j_range[idx_max]
       # choose source: the larger a among those two
       source = i_star if a[i_star] >= a[j_star] else j_star
       target = random.randint(0, n - 1)
       if target == source: continue
       delta = min(a[source] * 0.5, 0.015) * random.random()
       if delta < eps: continue
       # Attempt move
       a_new = a.copy()
       a_new[source] -= delta
       a_new[target] += delta
       a_new = np.maximum(a_new, eps)
       a_new = a_new / a_new.sum()
       conv_new = _fft_self_convolve(a_new)
       M_new = conv_new.max()
       if M_new < M - 1e-12:
           a = a_new
           conv = conv_new
           M = M_new
           score = 2.0 * n * M
           if score < best_score - 1e-12:
               best_seq = a.copy()

           a = a / a.sum()
           conv = _fft_self_convolve(a)
           M = conv.max()

"""
search for low‑score step function.

The algorithm consists of three phases:

1. **Broad family scan** – generate a variety of analytic and random
   symmetric distributions for several lengths (2000 … 10 000).  Each
   candidate is normalised to sum = 1 and a cheap FFT based upper‑bound
   ``2·n·max(a*a)`` is computed.  The best candidate is kept.

2. **Smooth‑max mirror descent** – starting from the current best,
   perform many projected exponentiated‑gradient steps on the simplex.
   A smoothed version of the max (log‑sum‑exp) is used; the smoothing
   parameter grows from 0.1 to 500 while the learning rate decays from
   10 to 1e‑3.  The iteration stops when the time budget is exhausted.
   The best vector seen during the descent is recorded.

3. **Refinement** – a quick simulated‑annealing walk followed by a
   deterministic hill‑climber that repeatedly attacks the pair of
   indices that contributes most to the current maximal convolution
   entry.  Both phases respect the remaining time.

All steps use only NumPy (FFT) and the helper routines defined below.
The returned list is a non‑negative probability vector (sum ≈ 1) and
conforms to the security checks of ``evaluate_sequence``.
"""
import time
import random

import numpy as np
from scipy.linalg import toeplitz  # optional, not used but allowed

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a: np.ndarray) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a vector already normalised to sum = 1.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * float(conv.max())

# Analytic families – each returns a *probability* vector (sum = 1)

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals

def _distribution_exponential(n: int, lam: float = 0.01) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals

def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * (np.log(x) + np.log(1 - x))
    w = np.exp(logw)
    if w.sum() == 0 or np.any(np.isinf(w)):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals /= vals.sum()
    return vals

def _distribution_power_edge(n: int, alpha: float) -> np.ndarray:
    """
    Edge‑heavy family:
        w_i ∝ ((i+0.5)*(n‑i‑0.5))^{-α}
    """
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - i - 0.5)
    eps = 1e-12
    vals = np.power(denom + eps, -alpha)
    vals /= vals.sum()

def _distribution_V_shape(n: int) -> np.ndarray:
    """
    Simple V‑shape: linear increase from centre to the edges, zero at centre
    (tiny epsilon added to avoid exact zeros).
    """
    centre = (n - 1) / 2.0
    vals = np.abs(np.arange(n, dtype=np.float64) - centre)
    eps = 1e-12
    vals = np.maximum(vals, eps)
    vals /= vals.sum()
    return vals

def _distribution_power_abs(n: int, beta: float) -> np.ndarray:
    """
    Symmetric power‑law: |i‑centre|^{β}
    """
    centre = (n - 1) / 2.0
    vals = np.abs(np.arange(n, dtype=np.float64) - centre) ** beta
    eps = 1e-12
    vals = np.maximum(vals, eps)
    vals /= vals.sum()

def _random_symmetric_dirichlet(n: int, alpha: float) -> np.ndarray:
    """
    Draw a symmetric vector from Dirichlet(α) on the half‑grid and mirror it.
    """
    half = (n + 1) // 2
    half_vec = np.random.dirichlet(np.full(half, alpha))
    if n % 2 == 0:
        full = np.concatenate([half_vec, half_vec[::-1]])
    else:
        full = np.concatenate([half_vec, half_vec[-2::-1]])
    full /= full.sum()
    return full

def _smooth_max_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Return the soft‑max weights for a convolution vector."""
    maxc = conv.max()
    w = np.exp(alpha * (conv - maxc))
    s = w.sum()
    if s == 0:
        return np.full_like(w, 1.0 / w.size)
    return w / s

def _mirror_descent(a: np.ndarray,
                    deadline: float,
                    best_seq: np.ndarray,
                    best_score: float) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Exponentiated‑gradient (mirror) descent on the smooth max of the
    self‑convolution, using an exponential schedule for the smoothing
    parameter (α) and a decaying learning rate.
    Returns the updated best sequence, its score and the final a.
    """
    n = a.shape[0]
    eps = 1e-12
    # schedule parameters – they have been tuned to explore a wide range
    total_steps = 200000
    alpha_start = 0.1
    alpha_target = 500.0
    lr_start = 10.0
    lr_end = 0.001

    best_local = best_seq
    best_local_score = best_score

    it = 0
    while time.time() < deadline and it < total_steps:
        ratio = it / total_steps
        # exponential growth/decay
        alpha = alpha_start * (alpha_target / alpha_start) ** ratio
        lr = lr_start * (lr_end / lr_start) ** ratio

        conv = _fft_self_convolve(a)
        w = _smooth_max_weights(conv, alpha)

        # gradient of the smoothed max w.r.t a
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        # multiplicative EG update + renormalisation
        a = a * np.exp(-lr * grad)
        a = np.maximum(a, eps)
        a = a / a.sum()

        # occasional true evaluation
        if (it % 1000 == 0) or (it == total_steps - 1):
            cur_score = _max_convolution_score(a)
            if cur_score < best_local_score - 1e-12:
                best_local = a.copy()
                best_local_score = cur_score
                print(f"[{time.time():.1f}] mirror it={it} new best {cur_score:.6f}")

        it += 1

    return best_local, best_local_score, a

def _simulated_annealing(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """
    Low‑temperature random walk that moves a small amount of mass from a
    source index to a target index.  Accepts worsening moves with a
    Metropolis probability.
    """
    n = a.shape[0]
    cur = a.copy()
    cur_score = best_score
    eps = 1e-12

    # initialise temperature as a small fraction of the current score
    T = max(cur_score * 0.01, 1e-7)
    T_min = cur_score * 1e-7
    decay = 0.9995

    best_local = best_seq
    best_local_score = best_score

    while time.time() < deadline:
        source = np.random.choice(n, p=cur)
        target = random.randint(0, n - 1)
        if target == source:
            continue

        delta = min(cur[source] * 0.3, 0.02) * random.random()
        if delta < eps:
            continue

        new = cur.copy()
        new[source] -= delta
        new[target] += delta
        new = np.maximum(new, eps)
        new = new / new.sum()
        new_score = _max_convolution_score(new)
        dE = new_score - cur_score

        if dE < 0 or random.random() < math.exp(-dE / T):
            cur = new
            cur_score = new_score
            if new_score < best_local_score - 1e-12:
                best_local = cur.copy()
                best_local_score = new_score
                print(f"[{time.time():.1f}] SA new best {new_score:.6f}")

        T = max(T * decay, T_min)

    return best_local, best_local_score

# ----------------------------------------------------------------------
def _hill_climber(a: np.ndarray,
                  deadline: float,
                  best_seq: np.ndarray,
                  best_score: float) -> tuple[np.ndarray, float]:
    """
    Deterministic greedy improvement.  Repeatedly finds the convolution
    index k that attains the current maximum, extracts the pair (i,j)
    that contributes the most to that entry and transfers a tiny chunk
    of mass from the larger contributor to a random slot.
    """
    n = a.shape[0]
    eps = 1e-12
    cur = a.copy()
    conv = _fft_self_convolve(cur)
    M = conv.max()
    best_local = best_seq
    best_local_score = best_score
    it = 0

    while time.time() < deadline:
        it += 1
        k = int(conv.argmax())
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)

        i_range = np.arange(i_low, i_high + 1, dtype=np.int64)
        j_range = k - i_range
        prods = cur[i_range] * cur[j_range]
        idx = int(prods.argmax())
        i_star = int(i_range[idx])
        j_star = int(j_range[idx])

        source = i_star if cur[i_star] >= cur[j_star] else j_star
        target = random.randint(0, n - 1)
        if target == source:
            continue

        delta = min(cur[source] * 0.5, 0.015) * random.random()
        if delta < eps:
            continue

        new = cur.copy()
        new[source] -= delta
        new[target] += delta
        new = np.maximum(new, eps)
        new = new / new.sum()

        conv_new = _fft_self_convolve(new)
        M_new = conv_new.max()

        if M_new < M - 1e-12:
            cur = new
            conv = conv_new
            M = M_new
            score = 2.0 * n * M
            if score < best_local_score - 1e-12:
                best_local = cur.copy()
                best_local_score = score
                print(f"[{time.time():.1f}] hill it={it} new best {score:.6f}")

        # occasional jitter to escape plateaus
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            cur = cur + jitter
            cur = np.maximum(cur, eps)
            cur = cur / cur.sum()
            conv = _fft_self_convolve(cur)
            M = conv.max()

    return best_local, best_local_score

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals /= vals.sum()
    return vals


def _distribution_power_edge(n: int, alpha: float) -> np.ndarray:
    """
    Edge‑heavy family:
        w_i ∝ ((i+0.5)*(n‑i‑0.5))^{-α}
    """
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - i - 0.5)
    eps = 1e-12
    vals = np.power(denom + eps, -alpha)
    vals /= vals.sum()
    return vals


def _distribution_V_shape(n: int) -> np.ndarray:
    """
    Simple V‑shape: linear increase from centre to the edges, zero at centre.
    """
    centre = (n - 1) / 2.0
    vals = np.abs(np.arange(n, dtype=np.float64) - centre)
    eps = 1e-12
    vals = np.maximum(vals, eps)
    vals /= vals.sum()
    return vals


def _distribution_power_abs(n: int, beta: float) -> np.ndarray:
    """
    Symmetric power‑law: |i‑centre|^{β}.
    """
    centre = (n - 1) / 2.0
    vals = np.abs(np.arange(n, dtype=np.float64) - centre) ** beta
    eps = 1e-12
    vals = np.maximum(vals, eps)
    vals /= vals.sum()
    return vals


def _random_symmetric_dirichlet(n: int, alpha: float) -> np.ndarray:
    """
    Draw a symmetric vector from Dirichlet(α) on the half‑grid and mirror it.
    """
    half = (n + 1) // 2
    half_vec = np.random.dirichlet(np.full(half, alpha))
    if n % 2 == 0:
        full = np.concatenate([half_vec, half_vec[::-1]])
    else:
        full = np.concatenate([half_vec, half_vec[-2::-1]])
    full /= full.sum()
    return full


# ----------------------------------------------------------------------
def _smooth_max_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights for a convolution vector at smoothing level α."""
    maxc = conv.max()
    w = np.exp(alpha * (conv - maxc))
    s = w.sum()
    if s == 0:
        return np.full_like(w, 1.0 / w.size)
    return w / s


# ----------------------------------------------------------------------
def _mirror_descent(a: np.ndarray,
                    deadline: float,
                    best_seq: np.ndarray,
                    best_score: float) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Exponentiated‑gradient (mirror) descent on the smooth max of the
    self‑convolution.  The smoothing α grows from 0.1 to ~500 while the
    learning rate decays from 10 to 1e‑3.
    """
    n = a.shape[0]
    eps = 1e-12
    total_steps = 200_000
    alpha_start = 0.1
    alpha_target = 500.0
    lr_start = 10.0
    lr_end = 0.001

    best_local = best_seq
    best_local_score = best_score

    it = 0
    while time.time() < deadline and it < total_steps:
        ratio = it / total_steps
        alpha = alpha_start * (alpha_target / alpha_start) ** ratio
        lr = lr_start * (lr_end / lr_start) ** ratio

        conv = _fft_self_convolve(a)
        w = _smooth_max_weights(conv, alpha)

        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        a = a * np.exp(-lr * grad)
        a = np.maximum(a, eps)
        a = a / a.sum()

        if (it % 1000 == 0) or (it == total_steps - 1):
            cur_score = _max_convolution_score(a)
            if cur_score < best_local_score - 1e-12:
                best_local = a.copy()
                best_local_score = cur_score
                print(f"[{time.time():.1f}] mirror it={it} new best {cur_score:.6f}")

        it += 1

    return best_local, best_local_score, a


# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """
    Low‑temperature random walk that moves a modest amount of mass from a
    source index to a target index, accepting worsening moves with a
    Metropolis probability.
    """
    n = a.shape[0]
    cur = a.copy()
    cur_score = best_score
    eps = 1e-12

    T = max(cur_score * 0.01, 1e-7)
    T_min = cur_score * 1e-7
    decay = 0.9995

    best_local = best_seq
    best_local_score = best_score

    while time.time() < deadline:
        source = np.random.choice(n, p=cur)
        target = random.randint(0, n - 1)
        if target == source:
            continue

        delta = min(cur[source] * 0.3, 0.02) * random.random()
        if delta < eps:
            continue

        new = cur.copy()
        new[source] -= delta
        new[target] += delta
        new = np.maximum(new, eps)
        new = new / new.sum()
        new_score = _max_convolution_score(new)
        dE = new_score - cur_score

        if dE < 0 or random.random() < math.exp(-dE / T):
            cur = new
            cur_score = new_score
            if new_score < best_local_score - 1e-12:
                best_local = cur.copy()
                best_local_score = new_score
                print(f"[{time.time():.1f}] SA new best {new_score:.6f}")

        T = max(T * decay, T_min)

    return best_local, best_local_score


# ----------------------------------------------------------------------
def _hill_climber(a: np.ndarray,
                  deadline: float,
                  best_seq: np.ndarray,
                  best_score: float) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer that repeatedly attacks the pair of indices that
    contributes most to the current maximal convolution entry.
    """
    n = a.shape[0]
    eps = 1e-12
    cur = a.copy()
    conv = _fft_self_convolve(cur)
    M = conv.max()

    best_local = best_seq
    best_local_score = best_score
    it = 0

    while time.time() < deadline:
        it += 1
        k = int(conv.argmax())
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)

        i_range = np.arange(i_low, i_high + 1, dtype=np.int64)
        j_range = k - i_range
        prods = cur[i_range] * cur[j_range]
        idx = int(prods.argmax())
        i_star = int(i_range[idx])
        j_star = int(j_range[idx])

        source = i_star if cur[i_star] >= cur[j_star] else j_star
        target = random.randint(0, n - 1)
        if target == source:
            continue

        delta = min(cur[source] * 0.5, 0.015) * random.random()
        if delta < eps:
            continue

        new = cur.copy()
        new[source] -= delta
        new[target] += delta
        new = np.maximum(new, eps)
        new = new / new.sum()

        conv_new = _fft_self_convolve(new)
        M_new = conv_new.max()

        if M_new < M - 1e-12:
            cur = new
            conv = conv_new
            M = M_new
            score = 2.0 * n * M
            if score < best_local_score - 1e-12:
                best_local = cur.copy()
                best_local_score = score
                print(f"[{time.time():.1f}] hill it={it} new best {score:.6f}")

        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            cur = cur + jitter
            cur = np.maximum(cur, eps)
            cur = cur / cur.sum()
            conv = _fft_self_convolve(cur)
            M = conv.max()

    return best_local, best_local_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) with a low
    ``evaluate_sequence`` score.  The function respects the given time
    budget and returns the best sequence found.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # leave a small safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Phase 1 – scan analytic families over several lengths
    lengths = [2000, 2500, 3000, 3500, 4000, 4500, 5000,
               6000, 8000, 10000]
    for n in lengths:
        if time.time() >= deadline:
            break

        # uniform
        vec = _distribution_uniform(n)
        vec = np.maximum(vec, eps)
        vec /= vec.sum()
        sc = _max_convolution_score(vec)
        if sc < best_score:
            best_seq = vec.copy()
            best_n = n
            best_score = sc
            print(f"[{time.time():.1f}] init n={n} uniform score={sc:.6f}")

        # arcsine
        vec = _distribution_arcsine(n)
        vec = np.maximum(vec, eps)
        vec /= vec.sum()
        sc = _max_convolution_score(vec)
        if sc < best_score:
            best_seq = vec.copy()
            best_n = n
            best_score = sc
            print(f"[{time.time():.1f}] init n={n} arcsine score={sc:.6f}")

        # triangular
        vec = _distribution_triangular(n)
        vec = np.maximum(vec, eps)
        vec /= vec.sum()
        sc = _max_convolution_score(vec)
        if sc < best_score:
            best_seq = vec.copy()
            best_n = n
            best_score = sc
            print(f"[{time.time():.1f}] init n={n} triangular score={sc:.6f}")

        # exponential (several λ)
        for lam in (0.008, 0.02, 0.05):
            vec = _distribution_exponential(n, lam=lam)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] init n={n} exp lam={lam} score={sc:.6f}")

        # V‑shape
        vec = _distribution_V_shape(n)
        vec = np.maximum(vec, eps)
        vec /= vec.sum()
        sc = _max_convolution_score(vec)
        if sc < best_score:
            best_seq = vec.copy()
            best_n = n
            best_score = sc
            print(f"[{time.time():.1f}] init n={n} V‑shape score={sc:.6f}")

        # sine‑power
        for pw in (0.5, 1.0, 2.0, 4.0):
            vec = _distribution_sine_power(n, pw)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] init n={n} sine‑pow={pw} score={sc:.6f}")

        # Gaussian
        for sigma in (0.2, 0.35, 0.5, 0.7):
            vec = _distribution_gaussian(n, sigma)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] init n={n} gaussian σ={sigma} score={sc:.6f}")

        # symmetric Beta
        for p in (0.05, 0.1, 0.15, 0.2, 0.3, 0.5,
                  0.8, 1.0, 1.5, 2.0, 3.0):
            vec = _distribution_beta_sym(n, p)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] init n={n} beta p={p} score={sc:.6f}")

        # power‑edge
        for a_edge in (0.55, 0.6, 0.7, 0.8, 0.9):
            vec = _distribution_power_edge(n, a_edge)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] init n={n} power‑edge α={a_edge} score={sc:.6f}")

        # power‑abs
        for beta in (0.5, 1.0, 2.0, 3.0):
            vec = _distribution_power_abs(n, beta)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] init n={n} power‑abs β={beta} score={sc:.6f}")

        # random low‑α Dirichlet draws
        for alpha in (0.01, 0.03, 0.05, 0.08):
            vec = _random_symmetric_dirichlet(n, alpha)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] init n={n} Dirichlet α={alpha} score={sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – incorporate a previously computed sequence if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_seq = prev.copy()
                best_n = prev.size
                best_score = sc
                print(f"[{time.time():.1f}] previous best imported score={sc:.6f}")
    except Exception:
        pass

    # Safety fallback
    if best_seq is None:
        best_n = 3000
        best_seq = _distribution_uniform(best_n)
        best_seq = best_seq / best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – intensive mirror descent
    best_seq, best_score, a_current = _mirror_descent(
        best_seq, deadline, best_seq, best_score
    )

    # ------------------------------------------------------------------
    # Phase 4 – simulated annealing refinement
    best_seq, best_score = _simulated_annealing(
        a_current, deadline, best_seq, best_score
    )

    # ------------------------------------------------------------------
    # Phase 5 – final cheap hill‑climber
    best_seq, best_score = _hill_climber(
        best_seq, deadline, best_seq, best_score
    )

    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
