"""
Propose a non‑negative step function (list of floats) that minimizes
`evaluate_sequence`.  

The algorithm works in three phases under a strict time budget:

1️⃣ **Broad family sweep** – test several lengths (2k‑8k) and many
   analytic families (arcsine, symmetric/asymmetric Beta, edge‑power‑law,
   exponential) plus a few random Dirichlet draws.  The best candidate
   (normalised to sum = 1) is kept.

2️⃣ **Projected sub‑gradient descent with momentum** – for the current
   best vector we minimise the true objective  
   `2·n·max(conv(a,a))` using the exact sub‑gradient of the max‑convolution.
   An Adam‑style momentum term and an adaptive back‑tracking line‑search
   give fast convergence.  This phase consumes most of the time budget.

3️⃣ **Simulated‑annealing polishing** – the remaining time is spent on
   random mass‑transfer moves accepted with a Metropolis criterion.
   Temperature decays exponentially, allowing the algorithm to escape
   shallow local minima and to improve the best score further.

All vectors are projected onto the probability simplex after every
update, guaranteeing the `evaluate_sequence` constraints (non‑negative,
sum ≥ 0.01).  The final sequence (still normalised to sum = 1) is
returned as a plain Python list.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a):
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a):
    """
    For a vector already normalised to sum = 1 the objective equals
    2·n·max(a*a).  This is exactly the value that `evaluate_sequence`
    returns for a normalised vector.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families (all return a vector with sum = 1)
# ----------------------------------------------------------------------


def _distribution_uniform(n):
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    """Discrete arcsine (Beta(0.5,0.5))."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_beta(n, p, q):
    """Asymmetric Beta(p,q) on the grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_edge_powerlaw(n, alpha):
    """Edge‑biased power‑law: a_i ∝ (i+½)^(-α)+(n-i-½)^(-α)."""
    idx = np.arange(n, dtype=np.float64)
    w = (idx + 0.5) ** (-alpha) + (n - idx - 0.5) ** (-alpha)
    w = np.maximum(w, 1e-16)
    w /= w.sum()
    return w


def _distribution_exponential(n, beta):
    """Sum of two exponentials decaying from the edges."""
    idx = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * idx) + np.exp(-beta * (n - 1 - idx))
    w = np.maximum(w, 1e-16)
    w /= w.sum()
    return w


def _distribution_dirichlet(n, conc):
    """Plain Dirichlet draw with given concentration vector."""
    vec = np.random.dirichlet(conc)
    vec = np.maximum(vec, 1e-16)
    vec /= vec.sum()
    return vec

# Core search routine
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000) -> list[float]:
    """Return a non‑negative step function (list of floats) that minimises
    ``evaluate_sequence`` within the supplied time budget.
    """
    # --------------------------------------------------------------
    # 0.  Setup & time budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    # keep a small safety margin for the caller
    deadline = start_time + budget_s - 0.5
    eps = 1e-12

    best_seq = None
    best_score = float("inf")
    best_n = None

    # --------------------------------------------------------------
    # 1. Broad exploration over lengths and analytic families
    # --------------------------------------------------------------
    # Lengths that give a reasonable trade‑off between speed and
    # expressive power.
    lengths = [2000, 4000, 8000]

    # Parameter grids (coarse – more than enough for the limited time)
    p_grid = np.linspace(0.02, 0.6, 25)          # symmetric Beta
    asym_p_grid = np.linspace(0.02, 0.6, 15)    # asymmetric Beta (q fixed)
    alpha_grid = np.linspace(0.2, 4.0, 20)      # edge power‑law
    beta_grid = np.linspace(0.001, 2.0, 20)    # exponential decay

    for n in lengths:
        if time.time() >= deadline:
            break

        # ---- deterministic families ---------------------------------
        families = []

        # arcsine (U‑shaped)
        families.append(_distribution_arcsine(n))

        # symmetric beta
        for p in p_grid:
            families.append(_distribution_beta_sym(n, p))

        # asymmetric beta (left‑heavy)
        q_fixed = 0.5
        for p in asym_p_grid:
            families.append(_distribution_beta(n, p, q_fixed))

        # asymmetric beta (right‑heavy)
        for p in asym_p_grid:
            families.append(_distribution_beta(n, q_fixed, p))

        # edge‑biased power law
        for a in alpha_grid:
            families.append(_distribution_edge_powerlaw(n, a))

        # exponential edge decay
        for b in beta_grid:
            families.append(_distribution_exponential(n, b))

        # ---- random Dirichlet draws ---------------------------------
        for _ in range(5):
            conc = np.full(n, 1.0)                       # concentration = 1 (uniform Dirichlet)
            families.append(_distribution_dirichlet(n, conc))

        for vec in families:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            score = _max_convolution_score(vec)
            if score < best_score:
                best_score = score
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] n={n} candidate score={score:.6f}")

    # --------------------------------------------------------------
    # 2. Incorporate a previously supplied sequence if available
    # --------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            # If lengths differ, resample to the best length seen so far
            if best_n is not None and prev.size != best_n:
                x_old = np.linspace(0.0, 1.0, prev.size, endpoint=False)
                x_new = np.linspace(0.0, 1.0, best_n, endpoint=False)
                prev = np.interp(x_new, x_old, prev)
            prev /= prev.sum()
            score = _max_convolution_score(prev)
            if score < best_score:
                best_score = score
                best_seq = prev.copy()
                best_n = len(prev)
                print(f"[{time.time():.1f}] previous sequence improved score to {score:.6f}")
    except Exception:
        pass

    # If for any reason we still have nothing, fall back to a uniform vector.
    if best_seq is None:
        best_n = 4000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Projected sub‑gradient descent with momentum
    # --------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]

    # momentum term (simple heavy‑ball style)
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    # Learning‑rate schedule – start relatively large and decay
    step0 = 0.5
    sub_it = 0

    # Allocate roughly 60 % of the remaining time to this phase
    subgrad_deadline = start_time + 0.7 * budget_s

    while time.time() < subgrad_deadline:
        sub_it += 1

        # ----- convolution & active set -----
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        # ----- sub‑gradient of max‑convolution -----
        w = np.zeros_like(conv)
        w[active] = 1.0                     # each active index contributes equally
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1: n - 1 + n]

        # ----- momentum update -----
        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad

        # ----- back‑tracking line search -----
        step = step0 / math.sqrt(sub_it)    # simple decay
        while True:
            a_cand = a - step * momentum
            a_cand = _project_onto_simplex(a_cand, 1.0)
            a_cand = np.maximum(a_cand, eps)
            a_cand /= a_cand.sum()
            cand_score = _max_convolution_score(a_cand)

            if cand_score <= best_score + 1e-12 or step < 1e-8:
                break
            step *= 0.5

        a = a_cand
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad it={sub_it} new best {cand_score:.6f}")

        # occasional jitter to shake out plateaus (every 500 iters)
        if sub_it % 500 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

    # --------------------------------------------------------------
    # 4. Simulated‑annealing polishing (mass‑transfer moves)
    # --------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_score = best_score

    # exponential cooling from a fraction of the current score down to
    # a very low temperature.
    anneal_start = cur_score * 0.02
    anneal_end = cur_score * 1e-5

    while time.time() < deadline:
        # fraction of annealing budget already consumed
        elapsed = time.time() - subgrad_deadline
        total_anneal = deadline - subgrad_deadline
        frac = min(1.0, elapsed / total_anneal)
        T = anneal_start * (anneal_end / anneal_start) ** frac

        # ---- propose a random mass‑transfer move ----
        i = random.randrange(n)
        j = random.randrange(n)
        if i == j:
            continue
        # move a portion of the source mass (up to 10 % of its value)
        delta = cur_seq[i] * 0.1 * random.random()
        if delta < eps:
            continue

        cand = cur_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_onto_simplex(cand, 1.0)      # enforce sum‑1
        cand = np.maximum(cand, eps)
        cand /= cand.sum()

        cand_score = _max_convolution_score(cand)
        dE = cand_score - cur_score

        # Metropolis acceptance
        if dE < 0 or random.random() < math.exp(-dE / (T + 1e-18)):
            cur_seq, cur_score = cand, cand_score
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time():.1f}] anneal new best {cand_score:.6f}")

        # very occasional large jump to explore new basins
        if random.random() < 0.0005:
            # full random Dirichlet perturbation
            perturb = np.random.dirichlet(np.full(n, 0.3))
            cur_seq = 0.9 * cur_seq + 0.1 * perturb
            cur_seq = _project_onto_simplex(cur_seq, 1.0)

    # --------------------------------------------------------------
    # 5. Final reporting
    # --------------------------------------------------------------
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    # The returned vector does **not** need to be scaled – any positive scaling
    # leaves `evaluate_sequence` unchanged.
    return best_seq.tolist()
