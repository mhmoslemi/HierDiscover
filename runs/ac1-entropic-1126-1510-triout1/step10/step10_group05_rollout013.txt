def _optimize_beta(beta0, C, n, deadline, max_iters=...):
    beta = beta0.copy()
    m = np.zeros_like(beta)
    v = np.zeros_like(beta)
    best_beta = beta.copy()
    best_seq = None; best_score = np.inf
    t = 0
    factor = 2 * n
    while time.time() < deadline and t < max_iters:
        t += 1
        # compute a = exp(C.T @ beta), normalize.
        logits = C.T @ beta
        max_log = np.max(logits) # for numerical stable exponent
        exp_logits = np.exp(logits - max_log)
        a = exp_logits / exp_logits.sum()
        # compute subgrad for max conv:
        grad_max, cur_max, _ = _subgrad(a)
        grad_surrogate = factor * grad_max
        # compute gradient w.r.t beta
        ga = grad_surrogate * a
        term1 = C @ ga   # shape (K,)
        term2_scalar = np.dot(grad_surrogate, a)
        term2 = term2_scalar * (C @ a)
        grad_beta = term1 - term2

        # Adam update
        beta_m = 0.9 * m + 0.1 * grad_beta
        beta_v = 0.999 * v + 0.001 * (grad_beta ** 2)
        m_hat = beta_m / (1 - 0.9 ** t)
        v_hat = beta_v / (1 - 0.999 ** t)
        eta = 0.02 / (np.sqrt(v_hat) + 1e-8)
        beta = beta - eta * m_hat
        m, v = beta_m, beta_v

        # Evaluate current surrogate: sc = 2 * n * cur_max
        sc = factor * cur_max
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
    # after loop return best_seq, best_score
    if best_seq is None:
        # fallback: compute a from final beta
        logits = C.T @ beta
        exp_logits = np.exp(logits - np.max(logits))
        best_seq = exp_logits / exp_logits.sum()
        best_score = _max_convolution_score(best_seq)
    return best_seq, best_score

def _hill_climb_simple(a, deadline, iters=100000, epsilon=0.02):
    best = a.copy()
    best_score = _max_convolution_score(best)
    n = a.size
    while time.time() < deadline and iters>0:
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        if best[i] < 1e-12:
            continue
        delta = min(best[i] * epsilon, 1.0 - best[j])
        if delta <= 0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = _max_convolution_score(cand)
        if sc < best_score:
            best = cand
            best_score = sc
        iters -= 1
    return best, best_score

"""
Search for a non‑negative step function minimizing

    2·n·max(conv(a,a)) / (sum(a)²)

using a low‑dimensional cosine/sine basis (soft‑max parameterisation) and
Adam optimisation.  The approach builds a smooth symmetric shape that can be
refined by many fast iterations, then applies a short hill‑climber for a final
polish.  The routine respects the supplied time budget.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (gradient of max_b)
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of the functional max_i (a*a)_i.
    Returns (grad_of_max, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices (k = 2*i)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
# Surrogate score (exact when sum(a)=1)
def _max_convolution_score(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families for initial seeds (copied from the previous script)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _gaussian_distribution(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    w /= w.sum()
    return w


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex from Dirichlet(α,…,α)."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Simple hill‑climber (random mass transfers)
def _hill_climb(a: np.ndarray, deadline: float,
                max_iters: int = 200_000,
                epsilon: float = 0.02) -> tuple[np.ndarray, float]:
    """Randomly move a small fraction of mass; keep improving moves."""
    best = a.copy()
    best_score = _max_convolution_score(best)
    n = best.size
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        if best[i] < 1e-12:
            continue
        delta = min(best[i] * epsilon, 1.0 - best[j])
        if delta <= 0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = _max_convolution_score(cand)
        if sc < best_score:
            best = cand
            best_score = sc
    return best, best_score


# ----------------------------------------------------------------------
# Adam optimisation in the cosine/sine basis
def _optimize_beta(beta0: np.ndarray,
                   C: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 150_000,
                   factor: float | None = None) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the surrogate 2·n·max(conv(a,a)) where
    a_i = softmax( (Cᵀβ)_i ).
    Returns the best a seen and its surrogate score.
    """
    if factor is None:
        factor = 2.0 * n
    beta = beta0.copy()
    m = np.zeros_like(beta)
    v = np.zeros_like(beta)
    best_a = None
    best_score = np.inf
    t = 0
    beta1, beta2 = 0.9, 0.999
    step0 = 0.02

    while time.time() < deadline and t < max_iters:
        t += 1
        # soft‑max mapping
        logits = C.T @ beta
        # shift for numerical stability
        logits -= np.max(logits)
        exp_logits = np.exp(logits)
        a = exp_logits / exp_logits.sum()

        # sub‑gradient of max convolution
        grad_max, cur_max, _ = _subgrad(a)
        grad_surrogate = factor * grad_max  # ∇ of 2·n·max

        # gradient w.r.t. β (chain rule)
        ga = grad_surrogate * a                     # element‑wise
        term1 = C @ ga                              # shape (R,)
        term2_scalar = np.dot(grad_surrogate, a)    # scalar
        term2 = term2_scalar * (C @ a)              # shape (R,)
        grad_beta = term1 - term2

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_beta
        v = beta2 * v + (1.0 - beta2) * (grad_beta ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        beta = beta - eta * m_hat

        # current surrogate score
        cur_score = factor * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    if best_a is None:
        # fallback (should not happen)
        logits = C.T @ beta
        logits -= np.max(logits)
        exp_logits = np.exp(logits)
        best_a = exp_logits / exp_logits.sum()
        best_score = _max_convolution_score(best_a)

    return best_a, best_score
