"""
Hybrid search for a low‑convolution step‑function.

The algorithm consists of three stages:

1. **Wide seed generation** – several analytic families (uniform,
   arcsine, beta, power‑law, exponential, sparse, two‑block and a
   Gaussian shape) are normalized to sum = 1 for a handful of lengths.
   The best few seeds are kept.

2. **Smooth sub‑gradient descent** – for each selected seed we minimise
   the exact objective `2·n·max(conv(a,a))/sum(a)^2` by mirror‑descent
   on the probability simplex.  Instead of using only the active set,
   a soft‑max over all convolution entries is employed:
   `w_i ∝ exp(α·conv_i)`.  The resulting gradient
   `∇ = 2·a⋆w` is used in an exponentiated‑gradient update with a
   momentum term and a decaying step‑size `η_t = lr0/√t`.  The number
   of iterations is limited by the global time budget.

3. **Greedy pairwise mass‑transfer polishing** – after the continuous
   phase a cheap hill‑climbing walk repeatedly moves a random fraction
   of mass from one index to another, keeping a move only if the true
   objective improves.

All heavy work (convolutions, gradients) uses real FFTs, so the method
scales to tens of thousands of coefficients within the 1000 s budget.
The final sequence is returned as a plain list of floats ready for
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (sum forced to 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:          # safety – should never happen
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# ---------- Analytic families (used as seeds) -------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma_frac: float = 0.2) -> np.ndarray:
    """Centered Gaussian‑like shape, truncated to the interval."""
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * n
    vals = np.exp(-0.5 * (i / sigma) ** 2)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Broad pool of seeds for a given length."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_gaussian(n))

    for p in (0.05, 0.1, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0):
        cand.append(_distribution_beta_sym(n, p))

    for _ in range(5):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))

    for lam in (0.001, 0.005, 0.01, 0.02, 0.05):
        cand.append(_distribution_exponential_increasing(n, lam))

    for block_frac in (0.05, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac))

    for _ in range(3):
        cand.append(_distribution_dirichlet(n, rng, alpha=0.3 + 0.7 * rng.random()))

    cand.append(_distribution_random_sparse(n, rng, density=0.01))

    return cand


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the sub‑gradient 2·a⋆w where w is a weight vector (length 2n‑1)
    that sums to 1. Implemented via FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2·n‑1
    a_rev = a[::-1]
    size = 1 << ((n + m - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad

    a: np.ndarray,

    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.7,
    alpha: float = 12.0,
    momentum: float = 0.9,
    max_iter: int = 60000,
    eval_interval: int = 300,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex with a *soft‑max* weighting over the
    convolution entries.  A momentum term smoothes the noisy sub‑gradient.
    The routine is bounded by `deadline` and returns the best vector seen.
    """
    n = a.shape[0]
    # momentum accumulator
    g_avg = np.zeros_like(a)
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        # --- soft‑max weights on the convolution (stabilised) ---
        conv_max = conv.max()
        # numerical stability: shift by the max before exponentiation
        exp_vals = np.exp(alpha * (conv - conv_max))
        w = exp_vals / exp_vals.sum()          # length 2n‑1, sums to 1

        # gradient of the max‑conv objective
        grad = _gradient_from_weights(a, w)

        # momentum update (exponential moving average)
        g_avg = momentum * g_avg + (1.0 - momentum) * grad

        # step size schedule η_t = lr0 / √t
        eta = lr0 / math.sqrt(t)

        # exponentiated‑gradient update (mirror‑descent w.r.t. entropy)
        a = a * np.exp(-eta * g_avg)

        # renormalise to the simplex (keeps sum = 1)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # occasional exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth‑grad it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Simple greedy mass‑transfer walk: repeatedly move a random fraction of
    mass from a non‑zero coordinate `i` to a random different coordinate
    `j`.  A move is kept only if the true objective improves.
    """
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        # random transfer magnitude (up to 30 % of a[i])
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects a total time budget (default 1000 s) and returns
    the best sequence found as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # 1. Choose lengths – larger lengths give a finer discretisation.
    lengths = [4000, 8000, 15000, 25000, 35000]

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 2. Scan analytic families for each length
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        for cand in _initial_candidates_extended(n, rng):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] **** new global best len={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))

        # keep only the best few seeds globally
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:12]

    # ------------------------------------------------------------------
    # 3. Optionally import a previously discovered sequence.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s == 0.0:
                prev = _distribution_uniform(prev.shape[0])
                s = prev.sum()
            prev = prev / s
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best improved: {sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Fallback – should never be needed
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Main continuous optimisation phase – smooth sub‑gradient descent
    remaining = deadline - time.time()
    if remaining > 80.0:
        # pick a few top seeds for an intensive descent phase
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]]
        # allocate about 80 % of the remaining budget to this phase
        descent_budget = remaining * 0.80
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            # split the remaining descent budget fairly among the yet‑to‑run seeds
            seeds_left = len(seeds) - idx
            per_seed_deadline = time.time() + descent_budget / seeds_left
            seq_out, score_out = _subgradient_smooth_descent(
                seed_seq.copy(),
                sub_deadline := per_seed_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.8,
                alpha=15.0,
                momentum=0.85,
                max_iter=40000,
                eval_interval=400,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after smooth descent new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final polishing with greedy pairwise swaps
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # Convert to plain Python list (evaluate_sequence expects that)
    return overall_best_seq.tolist()
