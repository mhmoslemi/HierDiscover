"""
Hybrid search for a low‑peak step‑function.

*   A fast analytic sweep (edge‑power family, uniform, arcsine) gives a
    good starting point for a range of lengths.
*   The best seed is refined by an exponentiated‑gradient
    (mirror‑descent) that uses the *full* active set of convolution
    entries attaining the current maximum.  The step size decays as
    ηₜ = η₀ / √t.
*   After the continuous phase a cheap simulated‑annealing / hill‑climbing
    walk performs random mass transfers a[i] → a[j] and accepts
    improvements (or occasional uphill moves) while the temperature
    slowly cools.
*   All work respects a global deadline (≈ 1000 s).  The routine returns
    the best sequence found as a plain list of floats ready for
    `evaluate_sequence`.

The objective that is minimised is the exact value returned by
`evaluate_sequence` (the denominator is forced to 1, i.e. the
distribution is normalised to unit sum, so the objective reduces to
`2·n·max(conv(a,a))`).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """ Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}. """
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """ Full linear self‑convolution a*a using a real FFT (length 2·n‑1). """
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective used by `evaluate_sequence` for a *unit‑sum* vector.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    # a is assumed to sum to 1; the denominator is therefore 1.
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
# Analytic families ------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+½)(n‑i‑½))^{‑p},  p>0 makes the distribution more
    edge‑heavy (p=0.5 reproduces the arcsine shape)."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-p)
    vals /= vals.sum()
    return vals


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient 2·a⋆w  (cross‑correlation) needed for the exponentiated‑gradient.
    a has length n, w length 2·n‑1 and ∑w = 1.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2·n‑1
    a_rev = a[::-1]
    size = 1 << ((n + m - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _exponential_gradient_descent(a0: np.ndarray,
                                 deadline: float,
                                 rng: np.random.Generator,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.7,
                                 eval_interval: int = 200) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using the full active set.
    Stops when the time budget expires.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()

        # active set (all indices attaining the maximum up to tiny tolerance)
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)

        # exponentiated‑gradient step (mirror‑descent w.r.t. KL)
        a = a * np.exp(-eta * grad)

        # projection back onto the probability simplex (keeps sum = 1)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # occasional exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] gd it={t}   new global best {best_score:.6f}")

        # occasional random perturbation to escape tiny plateaus
        if t % 5000 == 0:
            perturb = np.exp(0.001 * rng.normal(size=n))
            a = a * perturb
            a = a / a.sum()
    return best_seq, best_score


# ----------------------------------------------------------------------
def _hill_climb_random_moves(a0: np.ndarray,
                             deadline: float,
                             rng: np.random.Generator,
                             best_seq: np.ndarray,
                             best_score: float) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing style walk that transfers a small random
    amount of probability mass from one position to another.
    Accepts improvements always and occasional uphill moves while the
    temperature slowly cools.
    """
    a = a0.copy()
    n = a.shape[0]
    temperature = 0.02            # initial temperature
    it = 0
    while time.time() < deadline:
        it += 1

        # pick a source index with non‑negligible mass
        src_idx = rng.choice(np.where(a > 1e-12)[0])

        # pick a distinct destination
        dst_idx = rng.integers(n - 1)
        if dst_idx >= src_idx:
            dst_idx += 1

        # amount to move: a fraction of the source mass
        max_move = a[src_idx] * 0.30
        delta = max_move * rng.random()
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[src_idx] -= delta
        new_a[dst_idx] += delta

        cur = _objective(new_a)

        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = new_a.copy()
            print(f"[{time.time():.1f}] hill‑climb it={it}   new best {best_score:.6f}")
        else:
            # accept uphill with Boltzmann probability
            if rng.random() < math.exp(-(cur - best_score) / max(temperature, 1e-12)):
                a = new_a

        # geometric cooling
        temperature *= 0.9999
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimising

    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1.  Scan a variety of analytic families for several lengths.
    lengths = [2000, 4000, 8000, 12000, 20000]   # larger n → lower lower‑bound
    top_candidates: list[tuple[float, np.ndarray]] = []

    best_seq = None
    best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # a handful of generic families
        families = [
            _distribution_uniform,
            _distribution_arcsine,
            lambda N: _distribution_edge_power(N, 0.5),   # standard arcsine
            lambda N: _distribution_edge_power(N, 0.7),
            lambda N: _distribution_edge_power(N, 0.9),
            lambda N: _distribution_edge_power(N, 1.1),
            lambda N: _distribution_power_increasing(N, 1.0),
            lambda N: _distribution_power_increasing(N, 2.0),
            lambda N: _distribution_exponential_increasing(N, 0.005),
            lambda N: _distribution_exponential_increasing(N, 0.02),
            lambda N: _distribution_random_sparse(N, rng, density=0.015),
        ]

        for fam in families:
            if time.time() >= deadline:
                break
            a = fam(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()                     # normalise to unit sum
            sc = _objective(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[{time.time():.1f}] ***** new global best (len={n})  score={sc:.6f}")
            top_candidates.append((sc, a.copy()))

        # keep only a few very promising seeds for the later continuous phase
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:12]

    # ------------------------------------------------------------------
    # 2.  Optionally import a previously discovered sequence.
    try:

        if prev.ndim == 1 and prev.size > 0:

            sc = _objective(prev)

            top_candidates.append((sc, prev.copy()))
            top_candidates.sort(key=lambda x: x[0])

    except Exception:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # 4.  Continuous optimisation – exponentiated‑gradient descent.
    if time.time() < deadline:
        # allocate roughly two‑thirds of the remaining budget for the GD phase
