"""
optimize_step_function.py

A fast, multi‑phase optimizer for the “max‑self‑convolution” objective used
by `evaluate_sequence`.  The algorithm consists of:

1. **Exploratory sweep** – a rich pool of analytic families (uniform,
   arcsine, symmetric β, U‑shaped `(i·(n‑i))^{‑p}` with an offset) is
   evaluated for several lengths up to 100 000.  The best ~20 candidates
   are retained.

2. **Mirror‑descent refinement** – an exponentiated‑gradient (mirror)
   descent that uses the *full* sub‑gradient coming from every
   convolution entry that is within a small fraction of the current
   maximum.  The step‑size follows a 1/√t schedule and an Adam‑style
   momentum is added for robustness.

3. **Greedy pairwise mass transfer** – a cheap stochastic local search
   that repeatedly moves a random amount of mass from one coordinate to
   another if it strictly decreases the objective.

All operations respect the 1000 s time budget, fall back to the previously
known `height_sequence_1` if it exists, and finally return a plain
`list[float]` ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    # size = next power of two ≥ 2*n‑1
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
#  Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_u_shape(n: int, p: float, delta: float = 0.0) -> np.ndarray:
    """
    U‑shaped family: w_i ∝ ((i+0.5+δ)*(n‑i‑0.5+δ))^{‑p}.
    The offset δ smooths the singularity at the boundaries.
    """
    i = np.arange(n, dtype=np.float64) + 0.5 + delta
    j = (n - np.arange(n, dtype=np.float64) - 0.5 + delta)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base (U = uniform)."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
#  Mirror‑descent on the true max‑convolution objective ---------------
def _subgradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.6,
    eval_interval: int = 150,
    tol_frac: float = 0.001,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent using all near‑max indices."""
    n = a.shape[0]
    t = 0
    # Adam‑style moment estimates
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        # indices whose value lies within tol_frac of the max
        thr = max_val * (1.0 - tol_frac)
        idxs = np.where(conv >= thr)[0]

        # accumulate gradient from all quasi‑max indices
        grad = np.zeros_like(a)
        rng = np.arange(n, dtype=np.int64)
        for k in idxs:
            ind = k - rng
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * a[ind[mask]]
        grad /= max(len(idxs), 1)   # average

        # Adam momentum update (on the log‑space, because we use exponentiated gradient)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        # effective learning rate
        eta = lr0 / math.sqrt(t)
        # exponentiated gradient step
        a = a * np.exp(-eta * m_hat / (np.sqrt(v_hat) + eps))

        # re‑normalize to the simplex (sum = 1)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] subgrad it={t} new global best {best_score:.6f}"
                )
        if t >= 200_000:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Greedy pairwise mass transfer ----------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass; keep the move only if the objective improves."""
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score
