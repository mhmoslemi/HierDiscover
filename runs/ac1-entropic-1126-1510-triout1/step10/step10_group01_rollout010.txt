"""
Hybrid multi‑scale search for the step‑function minimising

    2·n·max(conv(a,a)) / (sum(a)**2),

where a is a non‑negative vector (clipping to [0,1000] is applied by
evaluate_sequence).  The algorithm works in three stages:

1. **Exploration** on small lengths (512–2048).  A diversified pool of
   analytic seeds is created, each seed is refined by exponentiated‑
   gradient descent on the active set of convolution indices.  The best
   few candidates are kept.

2. **Upsampling & fine tuning** – the best small‑scale shape is linearly
   interpolated to a larger length (8k‑32k) and further optimised with the
   same sub‑gradient method plus a stochastic “mass‑transfer” walk
   (simulated‑annealing style).

3. **Polishing** – a cheap greedy pairwise‑swap local search is run
   until the time budget (≈ 1000 s) expires.

All updates keep the vector on the probability simplex (non‑negative,
unit L1‑norm).  The routine returns a plain Python list ready for
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective used in the competition (sum is forced to 1 for the
    optimisation, but any positive scale gives the same value).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:

    Sub‑gradient 2·a⋆w where w is a vector of length 2·n‑1 that sums to 1.
    Computed via an FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    return 2.0 * conv[start:start + n]

# Analytic families (same as in the reference implementation)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Broad pool of analytic seeds for a given length."""
    cand = []
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    for p in (0.05, 0.1, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0):
        cand.append(_distribution_beta_sym(n, p))

    for _ in range(5):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))

    for lam in (0.001, 0.005, 0.01, 0.02, 0.05):
        cand.append(_distribution_exponential_increasing(n, lam))

    for block_frac in (0.05, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac))

    for _ in range(3):
        cand.append(_distribution_dirichlet(n, rng,
                                            alpha=0.3 + 0.7 * rng.random()))
    cand.append(_distribution_random_sparse(n, rng, density=0.01))
    return cand


# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         lr0: float = 0.7,
                         max_iter: int = 30000,
                         eval_interval: int = 500) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the active set of maximal convolution
    indices.  Returns the best sequence seen during the run.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        # learning rate schedule η_t = lr0 / sqrt(t)
        eta = lr0 / math.sqrt(t)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad)

        # simplex projection
        a = _project_onto_simplex(a)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] iter={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """
    Simple greedy mass‑transfer walk.  A move is kept only if the true
    objective strictly improves.  Very cheap – runs until time runs out.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3g} score {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         start_temp: float = 1e-2,
                         end_temp: float = 1e-6,
                         max_moves: int = 20000) -> tuple[np.ndarray, float]:
    """
    Very light‑weight annealing that perturbs two coordinates.
    Acceptance follows the Metropolis rule with a linearly decaying
    temperature.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    cur_a = a.copy()
    cur_score = best_score
    move = 0
    while time.time() < deadline and move < max_moves:
        move += 1
        # linear temperature schedule
        T = start_temp + (end_temp - start_temp) * move / max_moves
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # draw a fraction of the source mass
        delta = cur_a[i] * rng.random() * 0.15
        if delta < 1e-12:
            continue
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = _objective(new_a)
        diff = new_score - cur_score
        accept = diff < 0 or rng.random() < math.exp(-diff / max(T, 1e-12))
        if accept:
            cur_a = new_a
            cur_score = new_score
            if cur_score < best_score - 1e-13:
                best_score = cur_score
                best_seq = cur_a.copy()
                print(f"[{time.time():.1f}] anneal accept move={move} score={best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger size, followed by simplex renormalisation."""
    old_n = seq.shape[0]
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    up /= up.sum()
    return up

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for the best non‑negative step‑function under the evaluation
    budget.  The function respects the global time limit and returns a
    plain list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Small‑scale exploration
    small_lengths = [512, 1024, 2048]
    top_seeds: list[np.ndarray] = []
    best_overall = None
    best_score = float("inf")

    for n in small_lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] exploring length {n}")
        for cand in _initial_candidates_extended(n, rng):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = _objective(cand)
            if sc < best_score:
                best_score = sc
                best_overall = cand.copy()
                print(f"   *** new global best len={n} score={sc:.6f}")
            top_seeds.append(cand.copy())

        # keep only a modest number of promising seeds for the next stage
        top_seeds.sort(key=lambda x: _objective(x))
        top_seeds = top_seeds[:20]

    # ------------------------------------------------------------------
    # 2. Sub‑gradient refinement on small lengths
    for seq in top_seeds:
        if time.time() >= deadline:
            break
        seq_out, score_out = _subgradient_descent(
            seq.copy(),
            deadline,
            best_seq=best_overall,
            best_score=best_score,
            lr0=0.8,
            max_iter=25000,
            eval_interval=300,
        )
        if score_out < best_score:
            best_score = score_out
            best_overall = seq_out.copy()
            print(f"[{time.time():.1f}] after small‑scale descent best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Upsample to a medium size and continue optimisation
    medium_n = 12000
    if time.time() < deadline:
        print(f"[{time.time():.1f}] upsampling to {medium_n}")
        up_seq = _upsample(best_overall, medium_n)

        # a few more sub‑gradient steps at the medium scale
        up_seq, mid_score = _subgradient_descent(
            up_seq,
            deadline,
            best_seq=best_overall,
            best_score=best_score,
            lr0=0.6,
            max_iter=40000,
            eval_interval=500,
        )
        if mid_score < best_score:
            best_score = mid_score
            best_overall = up_seq.copy()
            print(f"[{time.time():.1f}] after medium‑scale descent best={best_score:.6f}")

        # stochastic annealing to escape shallow local minima
        best_overall, best_score = _simulated_annealing(
            best_overall,
            deadline,
            best_seq=best_overall,
            best_score=best_score,
            start_temp=1e-2,
            end_temp=1e-6,
            max_moves=20000,
        )

    # ------------------------------------------------------------------
    # 4. Final large‑scale polishing (if time permits)
    large_n = 30000
    if time.time() < deadline and best_overall.shape[0] != large_n:
        print(f"[{time.time():.1f}] final upsample to {large_n}")
        best_overall = _upsample(best_overall, large_n)

    if time.time() < deadline:
        best_overall, best_score = _subgradient_descent(
            best_overall,
            deadline,
            best_seq=best_overall,
            best_score=best_score,
            lr0=0.5,
            max_iter=50000,
            eval_interval=600,
        )

    # cheap greedy swaps – final cleanup
    if time.time() < deadline:
        best_overall, best_score = _pairwise_swap_local(
            best_overall.copy(),
            deadline,
            best_seq=best_overall,
            best_score=best_score,
        )

    # ------------------------------------------------------------------
    # 5. Incorporate the previously discovered sequence if it is better
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < best_score:
                best_score = sc
                best_overall = prev.copy()
                print(f"[{time.time():.1f}] previous sequence improved best to {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    print(f"Finished – best score {best_score:.6f} with length {len(best_overall)}")
    return best_overall.tolist()
