evaluate_sequence().  The algorithm is a hybrid of:

  1.  A large pool of analytic candidates (arcsine, beta, exponential, etc.)

  3.  A final polishing phase that combines a Frank‑Wolfe “mass‑transfer’’

All steps respect the 1000 s time limit and use only pure NumPy + SciPy
(High‑s) linear algebra.  The returned list is already normalised to sum‑1,

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
# Objective utilities (sum = 1 → surrogate = 2·n·max(conv))
# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a that already sums to 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


def _smooth_grad(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smoothed max objective:
        M_α(a) = (1/α)·log Σ_k exp(α·conv_k(a))
    using the standard identity grad M = Σ p_k ∇conv_k,
    where p_k = softmax(α·conv_k).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)                  # length 2n‑1
    # softmax of α·conv  (numerically stable)
    scaled = alpha * conv
    maxv = np.max(scaled)
    w = np.exp(scaled - maxv)
    w /= w.sum()                                   # weights p_k, Σw = 1

    # 2·conv_rev part
    rev = _fft_convolve(a[::-1], w)                # length 3n‑2
    grad = 2.0 * rev[n - 1 : n - 1 + n]            # slice of length n

    # diagonal correction for i=j terms (k even)
    even = (np.arange(w.size) % 2 == 0)
    diag_idx = np.arange(w.size)[even] // 2       # indices 0 … n‑1
    grad[diag_idx] -= w[even] * a[diag_idx]        # subtract the over‑counted part
    return grad

def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _edge_power_distribution(n: int, s: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = (i * (n - i + 1)) ** (-s)
    vals /= vals.sum()
    return vals


def _double_bump_distribution(n: int,
                              block_frac: float = 0.25,
                              gap_frac: float = 0.1) -> np.ndarray:
    total_block = max(2, int(block_frac * n))
    m = total_block // 2
    g = int(gap_frac * n)
    a = np.zeros(n, dtype=np.float64)
    a[:m] = 1.0 / (2 * m)
    start2 = m + g
    if start2 + m > n:
        start2 = n - m
    a[start2:start2 + m] = 1.0 / (2 * m)
    return a


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Optimisation primitives -----------------------------------------------
# ----------------------------------------------------------------------
def _adam_smooth(a0: np.ndarray, deadline: float,
                 step0: float = 0.55,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 max_alpha: float = 80.0) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smoothed max objective.
    Alpha grows linearly from 1.0 to max_alpha during the run.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _max_convolution_score(a)

    t = 0
    start = time.time()
    total_budget = deadline - start
    while time.time() < deadline:
        t += 1
        elapsed = time.time() - start
        # linear schedule for alpha
        alpha = 1.0 + (max_alpha - 1.0) * (elapsed / total_budget)

        grad = _smooth_grad(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


def _frank_wolfe_polish(a: np.ndarray, deadline: float,
                       fractions=(1.0, 0.75, 0.5, 0.25, 0.125)) -> tuple[np.ndarray, float]:
    """
    Simple Frank‑Wolfe style “mass‑transfer’’ step.
    Repeatedly moves mass from the index with largest gradient to the one with
    smallest gradient, accepting the move if the surrogate improves.
    """
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    cur = best_score
    n = a.size
    it = 0
    while time.time() < deadline and it < 200_000:
        it += 1
        grad, _, _ = _subgrad(best_a)          # uses the exact sub‑gradient for max
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        max_transfer = min(best_a[i_src], 1.0 - best_a[i_dst])
        if max_transfer < 1e-15:
            break
        improved = False
        for f in fractions:
            delta = max_transfer * f
            a_new = best_a.copy()
            a_new[i_src] -= delta
            a_new[i_dst] += delta
            sc = _max_convolution_score(a_new)
            if sc < cur - 1e-13:
                best_a = a_new
                cur = sc
                improved = True
                break
        if improved and cur < best_score - 1e-14:
            best_score, best_a = cur, best_a.copy()
    return best_a, best_score


def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) under Σa = 1.
    Returns (gradient, current_max, active_indices) – identical to the
    version used in the original baseline.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # gradient of the (non‑smooth) max‑convolution:
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for i=j terms (when k even)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
# Main search routine ----------------------------------------------------
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a low‑score non‑negative step function.
    The routine respects the 1000 s deadline and returns the best list found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin
    start_global = time.time()

    # ------------------------------------------------------------------
    # Choose a base length (prefer the previous best if it exists)
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 6000
    base_n = max(2000, min(base_n, 25000))

    # ------------------------------------------------------------------
    # Set of problem sizes to explore (including a few randoms)
    lengths = set()
    for fac in (0.6, 0.8, 1.0, 1.2, 1.5, 2.0):
        n = int(base_n * fac)
        if 1500 <= n <= 25000:
            lengths.add(n)
    # a few random lengths up to 20 000
    for _ in range(4):
        lengths.add(np.random.randint(1500, min(20000, base_n * 2) + 1))
    lengths = sorted(lengths)

    print(f"[{time.time():.1f}] Exploring {len(lengths)} lengths (base≈{base_n})")

    # ------------------------------------------------------------------
    # Parameter grids for families (moderate size – we keep the total count
    # manageable because we will later run a heavy optimisation on the top
    # few candidates).
    beta_ps = np.arange(0.1, 1.01, 0.1)          # symmetric beta
    asym_grid = [(0.3, 0.7), (0.5, 0.5), (0.7, 0.3),
                 (0.2, 1.0), (1.0, 0.2), (0.4, 0.8), (0.8, 0.4)]
    power_ps = [0.4, 0.8, 1.2, 1.5, 2.0]
    lam_vals = [0.001, 0.003, 0.005, 0.01]
    dirichlet_alphas = [0.05, 0.1, 0.3, 0.5, 1.0, 2.0]
    edge_ps = [0.30, 0.40, 0.50, 0.60, 0.70, 0.80]   # edge‑power exponents
    double_blocks = [(0.15, 0.05), (0.20, 0.10), (0.25, 0.15)]

    # ------------------------------------------------------------------
    # Generate a large pool of analytic/random candidates
    candidates = []   # (array, surrogate_score)

    for n in lengths:
        if time.time() >= deadline:
            break

        # deterministic families ------------------------------------------------
        families = [
            (_arcsine_distribution, "arcsine"),
            (_triangular_distribution, "triangular"),
            (lambda m: _exponential_distribution(m, lam=0.005), "exp0.005")
        ]
        for gen, name in families:
            a = gen(n)
            candidates.append((a, _max_convolution_score(a)))

        # exponential tails
        for lam in lam_vals:
            a = _exponential_distribution(n, lam)
            candidates.append((a, _max_convolution_score(a)))

        # symmetric beta
        for p in beta_ps:
            a = _beta_symmetric_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))

        # asymmetric beta
        for a_alpha, a_beta in asym_grid:
            a = _beta_asymmetric_distribution(n, a_alpha, a_beta)
            candidates.append((a, _max_convolution_score(a)))

        # power‑law tails
        for p in power_ps:
            a = _power_law_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))

        # edge‑power (generalised arcsine)
        for s in edge_ps:
            a = _edge_power_distribution(n, s)
            candidates.append((a, _max_convolution_score(a)))

        # double‑bump blocks
        for block_frac, gap_frac in double_blocks:
            a = _double_bump_distribution(n, block_frac, gap_frac)
            candidates.append((a, _max_convolution_score(a)))

        # random Dirichlet draws
        for alpha in dirichlet_alphas:
            a = _random_dirichlet_alpha(n, alpha)
            candidates.append((a, _max_convolution_score(a)))

    # ------------------------------------------------------------------
    # If a previous best sequence exists, seed it and a few perturbed copies
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            if prev.sum() == 0:
                prev = np.full_like(prev, 1.0 / prev.size)
            else:
                prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
            for _ in range(6):
                jitter = np.random.randn(prev.size) * 4e-5
                noisy = _project_onto_simplex(prev + jitter, s=1.0)
                candidates.append((noisy, _max_convolution_score(noisy)))
            print(f"[{time.time():.1f}] added previous best (+perturbations)")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Keep a modest number of the best seeds for heavy refinement
    candidates.sort(key=lambda x: x[1])
    top_k = min(20, len(candidates))
    top_seeds = candidates[:top_k]
    print(f"Generated {len(candidates)} candidates, keeping top {top_k} for refinement.")

    best_seq = top_seeds[0][0].copy()
    best_score = top_seeds[0][1]
    print(f"Initial best surrogate score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Intensive refinement of each seed (smooth Adam + optional FW polish)
    for idx, (seed_seq, seed_score) in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        # allocate a fraction of the remaining budget to this seed
        remaining = deadline - time.time()
        # give each seed at most 0.12 · remaining seconds
        seed_deadline = time.time() + min(remaining * 0.12, 70.0)

        # --- smooth Adam ----------------------------------------------------
        seq_adam, sc_adam = _adam_smooth(seed_seq, seed_deadline)
        if sc_adam < best_score - 1e-14:
            best_score, best_seq = sc_adam, seq_adam.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} Adam‑smooth improved to {best_score:.6f}")

        # --- short Frank‑Wolfe polish (optional) ---------------------------
        # Only attempt if we still have time left after Adam
        if time.time() < deadline:
            polish_deadline = time.time() + min(remaining * 0.03, 15.0)
            seq_fw, sc_fw = _frank_wolfe_polish(best_seq, polish_deadline)
            if sc_fw < best_score - 1e-14:
                best_score, best_seq = sc_fw, seq_fw.copy()
                print(f"[{time.time():.1f}] FW polish after seed {idx+1} new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing: long Frank‑Wolfe phase followed by a short anneal
    # (these phases are cheap compared to the Adam runs)
    a = best_seq.copy()
    cur_score = best_score
    # Frank‑Wolfe style mass‑transfer (more iterations than above)
    fw_deadline = time.time() + min(120.0, (deadline - time.time()) * 0.15)
    a, cur_score = _frank_wolfe_polish(a, fw_deadline)

    if cur_score < best_score - 1e-14:
        best_score, best_seq = cur_score, a.copy()
        print(f"[{time.time():.1f}] Final FW phase improved to {best_score:.6f}")

    # Simulated‑annealing / jitter push (very short)
    a = best_seq.copy()
    cur_score = best_score
    temp_start = 0.025
    temp_end = 0.0012
    anneal_iter = 0
    while time.time() < deadline and anneal_iter < 150_000:
        anneal_iter += 1
        fraction = max(0.0, (deadline - time.time()) / (budget_s - 0.5))
        temp = temp_end + (temp_start - temp_end) * fraction

        i = np.random.randint(a.size)
        j = np.random.randint(a.size - 1)
        if j >= i:
            j += 1
        max_delta = min(a[i], 1.0 - a[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * temp
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if cur_score < best_score - 1e-14:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time():.1f}] Anneal iter={anneal_iter} new best {best_score:.6f}")
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_score - cur_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                a = a_new
                cur_score = new_score

    # ------------------------------------------------------------------
    elapsed_total = time.time() - start_global
    print(f"Finished after {elapsed_total:.1f}s – final surrogate score: {best_score:.6f}"
          f" (length={len(best_seq)})")
    # Return a plain Python list; the external evaluate_sequence will do the final scaling.
    return best_seq.tolist()
