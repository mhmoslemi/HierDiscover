def build_conv_matrix(f):
    n = f.shape[0]
    # rows 0..2n-2, cols 0..n-1
    # For each column j, the non-zero rows k are j..j+n-1
    rows = []
    cols = []
    data = []
    for j in range(n):
        # rows k from j to j+n-1
        i_range = np.arange(j, j+n)   # size n
        coeffs = f[i_range - j]  # equals f[i] for i in 0..n-1
        rows.extend(i_range.tolist())
        cols.extend([j]*n)
        data.extend(coeffs.tolist())
    # Convert to csr
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

def lp_optimize(f, M):
    n = len(f)
    A_ub = build_conv_matrix(f)
    b_ub = np.full(2*n-1, M, dtype=np.float64)

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0:
        return None
    g = g / sum_g
    return g

4. Perform a greedy hill‑climber that repeatedly removes mass from the
   pair (i,j) contributing most to the current maximal convolution entry.
5. Finally, solve a linear‑program (maximise ∑g subject to f∗g≤M) which
   often yields a further reduction of the maximal self‑convolution.
All steps respect the 1000 s wall‑clock limit.  The best vector found is

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a: np.ndarray) -> float:
    """Fast evaluation of 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * float(conv.max())

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * (np.log(x) + np.log(1 - x))
    w = np.exp(logw)
    if w.sum() == 0 or np.any(np.isinf(w)):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals /= vals.sum()
    return vals


def _distribution_power_edge(n: int, alpha: float) -> np.ndarray:
    """
    Edge‑heavy family:
        w_i ∝ ((i+0.5)*(n-i-0.5))^{-α}
    """

    eps = 1e-12

    vals /= vals.sum()

    vals = np.abs(np.arange(n, dtype=np.float64) - centre)
    eps = 1e-12

def _random_symmetric_dirichlet(n: int, alpha: float) -> np.ndarray:
    half = (n + 1) // 2
    half_vec = np.random.dirichlet(np.full(half, alpha))
    if n % 2 == 0:

    else:
        full = np.concatenate([half_vec, half_vec[-2::-1]])

def _build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build the (2n‑1)×n matrix A such that A·g = f∗g (full linear convolution).
    Returns a CSR sparse matrix suitable for scipy.optimize.linprog.
    """
    n = f.shape[0]
    # Toeplitz construction: first column = [f, zeros]; first row = [f[0], zeros]
    col = np.concatenate([f, np.zeros(n - 1, dtype=np.float64)])
    row = np.concatenate([f[:1], np.zeros(n - 1, dtype=np.float64)])
    A_dense = toeplitz(col, row)          # shape (2n-1, n)
    return csr_matrix(A_dense)


def _lp_refine(f: np.ndarray, M: float, time_deadline: float):
    """
    Solve the linear program:
        maximise   Σ g_j
        subject to conv(f,g) ≤ M , g ≥ 0 .
    Returns a new vector (normalised to sum = 1) if improvement is found,
    otherwise returns None.
    """
    n = f.shape[0]
    A_ub = _build_conv_matrix(f)          # (2n‑1, n) sparse
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)     # maximise Σg ⇒ minimise -Σg
    # Using HiGHS (default) – it accepts sparse A_ub.
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': max(0.0, time_deadline - time.time())})
    except Exception:
        return None

    if not res.success or res.x is None:
        return None

    g = np.maximum(res.x, 0.0)
    s = g.sum()
    if s <= 0:
        return None
    g = g / s
    # compute new maximal convolution
    conv_new = _fft_self_convolve(g)
    M_new = conv_new.max()
    if M_new < M - 1e-14:
        return g
    return None

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

    n = a.shape[0]
    conv = _fft_self_convolve(a)

    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)

    vals /= vals.sum()
    return vals

def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n

def _build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build the (2n‑1)×n matrix A that satisfies A·g = f∗g (full linear convolution).
    Returns a CSR sparse matrix suitable for scipy.optimize.linprog.
    """
    n = f.shape[0]
    col = np.concatenate([f, np.zeros(n - 1, dtype=np.float64)])
    row = np.concatenate([f[:1], np.zeros(n - 1, dtype=np.float64)])
    A_dense = toeplitz(col, row)          # shape (2n‑1, n)
    return csr_matrix(A_dense)


def _lp_refine(f: np.ndarray, M: float, deadline: float):
    """
    Linear‑programming refinement:
        maximise   Σ g_j
        subject to   f∗g ≤ M ,   g ≥ 0 .
    Returns a new normalised vector if it improves the maximal convolution,
    otherwise returns None.
    """
    n = f.shape[0]
    A_ub = _build_conv_matrix(f)          # (2n‑1, n) sparse
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)     # maximise Σg  ⇒ minimise -Σg

    # time left for the LP (HiGHS respects a time_limit option)
    time_left = max(0.0, deadline - time.time())
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_left})
    except Exception:
        return None

    if not res.success or res.x is None:
        return None

    g = np.maximum(res.x, 0.0)
    s = g.sum()
    if s <= 0:
        return None
    g /= s
    conv_new = _fft_self_convolve(g)
    M_new = conv_new.max()
    if M_new < M - 1e-14:
        return g
    return None

        i_high = min(n - 1, k)
        best_i = i_low
        best_j = k - i_low
        max_prod = a[best_i] * a[best_j]
        for i in range(i_low, i_high + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue
        max_delta = min(a[source] * 0.5, 0.01)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a_snapshot = a.copy()
        a[source] -= delta
        a[target] += delta
        a = np.maximum(a, eps)
        a /= a.sum()
        conv = _fft_self_convolve(a)
        new_M = conv.max()
        if new_M < M - 1e-13:
            M = new_M
            sc = 2.0 * n * M
            if sc < best_score - 1e-12:
                best_seq = a.copy()
                best_score = sc
                best_n = n
                print(f"[{time.time():.1f}] hill it={hill_iters}  best={sc:.6f}")
        else:
            a = a_snapshot
            conv = _fft_self_convolve(a)

        if hill_iters % 5_000 == 0:
            jitter = np.random.randn(n) * 1e-6
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] jitter applied at hill iter {hill_iters}")

    # ------------------------------------------------------------------
    # 5. Linear‑programming refinement (if time permits)
    for step in range(3):
        if time.time() >= deadline:
            break
        conv_cur = _fft_self_convolve(best_seq)
        M_cur = conv_cur.max()
        g = _lp_refine(best_seq, M_cur, deadline)
        if g is None:
            break
        new_score = _max_convolution_score(g)
        if new_score < best_score - 1e-12:
            best_seq = g.copy()
            best_score = new_score
            best_n = g.shape[0]
            print(f"[{time.time():.1f}] LP refinement {step+1}  best={new_score:.6f}")
        else:
            break

    final_score = 2.0 * best_seq.shape[0] * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
