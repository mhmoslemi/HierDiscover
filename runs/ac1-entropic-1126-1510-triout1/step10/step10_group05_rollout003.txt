"""
Improved search for a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of three phases:

1. **Broad Sampling** – a very large collection of analytic families
   (arcsine, power‑law, symmetric Beta, unsymmetric Beta, Gaussian,
   Cauchy, exponential) on many lengths together with a handful of
   deterministic reorderings.  Every candidate is ℓ₁‑normalised and
   scored with the exact surrogate 2·n·max(conv(a,a)).

2. **Intensive Refinement** – the best dozens of seeds are polished
   with several convex‑optimisation style methods:
   * exponentiated‑gradient (mirror‑descent on the simplex),
   * Adam,
   * a Frank‑Wolfe style pairwise mass transfer,
   * a short SGD pass.
   Each optimiser runs only as long as the remaining budget allows.

3. **Final Polishing** – the current best vector is further improved by
   (i) a random mass‑transfer hill‑climber, and (ii) a *swap* hill‑climber
   that only re‑orders the existing multiset of heights (large values are
   pushed far apart).  Both accept only improving moves and stop when the
   time budget is exhausted.

All heavy work uses an FFT‑based convolution, so even a few‑thousand‑item
vector can be evaluated millions of times within the 1000 s limit.

The function `propose_candidate` returns the best vector found (as a plain
Python list) ready for `evaluate_sequence`.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate (the objective for Σa = 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of the max‑convolution functional on the simplex.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices (self‑product terms)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active

# Analytic families (all return vectors that sum to 1)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+0.5)*(n-i-0.5))^{-p}  (p = 0.5 gives arcsine)."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-p)
    vals[vals == 0] = 1e-30
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide="ignore", invalid="ignore"):
        w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    if np.isnan(w).any() or np.isinf(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _gaussian_distribution(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    w /= w.sum()
    return w


def _cauchy_distribution(n: int, scale_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    scale = scale_factor * n
    w = 1.0 / (1.0 + ((idx - centre) / scale) ** 2)
    w /= w.sum()
    return w


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Sample from Dirichlet(α,…,α)."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Re‑ordering heuristics -------------------------------------------------
def order_alternate_extremes(desc_vals: np.ndarray) -> np.ndarray:
    """Largest, second‑largest, … alternately at left‑most / right‑most."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    left, right, toggle = 0, n - 1, True
    for v in desc_vals:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    return out


def order_center_outwards(desc_vals: np.ndarray) -> np.ndarray:
    """Put the biggest value at the centre and then fill expanding outwards."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    mid = (n - 1) // 2
    positions = [mid]
    for d in range(1, n):
        l = mid - d
        if l >= 0:
            positions.append(l)
        r = mid + d
        if r < n:
            positions.append(r)
    for i, pos in enumerate(positions[:n]):
        out[pos] = desc_vals[i]
    return out


def order_binary_tree(desc_vals: np.ndarray) -> np.ndarray:
    """Recursive binary‑tree layout (large values spread uniformly)."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    positions = []

    def recurse(l: int, r: int):
        if l > r:
            return
        m = (l + r) // 2
        positions.append(m)
        recurse(l, m - 1)
        recurse(m + 1, r)

    recurse(0, n - 1)
    out[positions] = desc_vals
    return out

def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 1_000_000,
                          eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror descent on the simplex)."""
    a = a0.copy()
    best = a.copy()
    best_sc = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        sc = _max_convolution_score(a)
        if sc < best_sc:
            best_sc, best = sc, a.copy()
    return best, best_sc


def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 500_000,
                  step0: float = 0.2,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best = a.copy()
    best_sc = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        sc = _max_convolution_score(a)
        if sc < best_sc:
            best_sc, best = sc, a.copy()
    return best, best_sc


def _sgd_subgrad(a0: np.ndarray, deadline: float,
                 max_iters: int = 500_000,
                 step0: float = 0.5) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best = a.copy()
    best_sc = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        step = step0 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        sc = _max_convolution_score(a)
        if sc < best_sc:
            best_sc, best = sc, a.copy()
    return best, best_sc


def _frank_wolfe_opt(a0: np.ndarray, deadline: float,
                    max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Iterated pairwise mass transfer from steepest to flattest coordinate."""
    a = a0.copy()
    best = a.copy()
    best_sc = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        delta = min(a[i_src], 1.0 - a[i_dst])
        if delta <= 0:
            break
        step = 0.7 * delta
        a[i_src] -= step
        a[i_dst] += step
        sc = _max_convolution_score(a)
        if sc < best_sc:
            best_sc, best = sc, a.copy()
        if it % 4000 == 0 and np.random.rand() < 0.1:
            a = _project_onto_simplex(a + np.random.randn(a.size) * 1e-5, s=1.0)
    return best, best_sc


def _hill_climb(a0: np.ndarray, deadline: float,
                max_iters: int = 300_000,
                epsilon: float = 0.02) -> tuple[np.ndarray, float]:
    """Random mass‑transfer hill‑climber that only accepts improvements."""
    a = a0.copy()
    best = a.copy()
    best_sc = _max_convolution_score(a)
    n = a.size
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or best[i] < 1e-12:
            continue
        delta = min(best[i] * epsilon, 1.0 - best[j])
        if delta <= 0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = _max_convolution_score(cand)
        if sc < best_sc:
            best, best_sc = cand, sc
    return best, best_sc


def _swap_hill_climb(a0: np.ndarray, deadline: float,
                    max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Swap‑only hill‑climber (keeps the multiset of values, only re‑orders)."""
    a = a0.copy()
    best = a.copy()
    best_sc = _max_convolution_score(a)
    n = a.size
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        i, j = np.random.choice(n, 2, replace=False)
        # swapping
        cand = best.copy()
        cand[i], cand[j] = cand[j], cand[i]
        sc = _max_convolution_score(cand)
        if sc < best_sc:
            best, best_sc = cand, sc
    return best, best_sc

            if time.time() >= deadline:
                break
            seq, sc = _exp_gradient_descent(seed_seq, deadline,
                                            max_iters=1_200_000,
                                            eta0=eta0)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} ExpGrad η₀={eta0} -> {best_score:.6f}")

        # ---- Adam ----------------------------------------------------
        if time.time() < deadline:
            seq, sc = _adam_descent(best_seq, deadline,
                                    max_iters=400_000,
                                    step0=0.25)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} Adam -> {best_score:.6f}")

        # ---- Frank‑Wolfe pairwise transfer ---------------------------
        if time.time() < deadline:
            seq, sc = _frank_wolfe_opt(best_seq, deadline,
                                       max_iters=300_000)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} FW -> {best_score:.6f}")

        # ---- Plain SGD (short) ---------------------------------------
        if time.time() < deadline:
            seq, sc = _sgd_subgrad(best_seq, deadline,
                                   max_iters=200_000,
                                   step0=0.4)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} SGD -> {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Global hill‑climbing on the current best ----------------------
    if time.time() < deadline - 30:
        seq, sc = _hill_climb(best_seq, deadline,
                              max_iters=400_000,
                              epsilon=0.015)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"Hill‑climb mass‑transfer improvement -> {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Swap‑only hill‑climber (re‑ordering) -------------------------
    if time.time() < deadline - 20:
        seq, sc = _swap_hill_climb(best_seq, deadline,
                                   max_iters=300_000)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"Swap‑only hill‑climb improvement -> {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best surrogate score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
