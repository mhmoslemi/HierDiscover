    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5

    eps = 1e-12

    best_seq = None
    best_score = float('inf')
    best_n = None

    # define lengths to try
    lengths = [2000, 3000, 4000, 5000, 7000, 10000]

    # define distribution generator functions
    # We'll define them above at top-level

    for n in lengths:
        if time.time() >= deadline: break

        # Build basis vectors
        basis = []
        # Uniform
        basis.append(_distribution_uniform(n))
        # arcsine
        basis.append(_distribution_arcsine(n))
        # triangular
        basis.append(_distribution_triangular(n))
        # exponential with lam 0.01, 0.02
        basis.append(_distribution_exponential(n, lam=0.008))
        basis.append(_distribution_exponential(n, lam=0.02))
        # beta sym p=0.3,0.6,1,2,4
        for p in (0.3,0.6,1.0,2.0,4.0):
            basis.append(_distribution_beta_sym(n, p))
        # sine power pw=0.5,1,2,4
        for pw in (0.5,1.0,2.0,4.0):
            basis.append(_distribution_sine_power(n, pw))
        # gaussian sigma 0.2, 0.35, 0.5
        for sigma in (0.2,0.35,0.5):
            basis.append(_distribution_gaussian(n, sigma))
        # add previous global best if exists (interpolated)
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.ndim == 1 and prev.size > 0:
                # interpolate to n
                x_old = np.linspace(0.0, 1.0, prev.size)
                x_new = np.linspace(0.0, 1.0, n)
                vec = np.interp(x_new, x_old, prev)
                vec = np.maximum(vec, eps)
                vec /= vec.sum()
                basis.append(vec)
        except Exception:
            pass

        # Normalize each basis (should already be sum 1 but ensure)
        basis = [np.maximum(v, eps) for v in basis]
        basis = [v / v.sum() for v in basis]

        # Compute convolutions
        convs = [_fft_self_convolve(v) for v in basis]
        m = len(convs[0])  # 2n-1

        # Build LP matrices
        k = len(basis)
        # We'll create arrays
        # A_ub shape (m, k+1)
        A_ub = np.empty((m, k+1), dtype=np.float64)
        for i, conv in enumerate(convs):
            A_ub[:, i] = conv
        A_ub[:, -1] = -1.0  # coefficient for t
        b_ub = np.zeros(m, dtype=np.float64)

        # Equality: sum w_i = 1
        A_eq = np.zeros((1, k+1), dtype=np.float64)
        A_eq[0, :k] = 1.0
        b_eq = np.array([1.0], dtype=np.float64)

        # variable bounds: w_i >=0, t >=0
        bounds = [(0, None)] * k + [(0, None)]

        # Solve LP
        res = linprog(c=np.concatenate([np.zeros(k), [1.0]]), A_ub=A_ub, b_ub=b_ub,

                      options={'presolve': True})
        if not res.success:
            # fallback: skip LP, just try individual basis vectors
            for v in basis:
                sc = _max_convolution_score(v)
                if sc < best_score:
                    best_seq = v.copy()
                    best_n = n
                    best_score = sc
            continue
        # Extract solution
        w = res.x[:k]
        # Probably many zero entries; compute mixture
        a_mix = np.zeros(n, dtype=np.float64)

        best_score = _max_convolution_score(best_seq)

"""
Propose a step‑function sequence that minimises ``evaluate_sequence``.

Algorithm outline
----------------
1. **Analytic basis** – a handful of classic probability families
   (uniform, arcsine, triangular, exponential, symmetric beta,
   sine‑power, Gaussian) are instantiated for a candidate length `n`.
   The previously‑found global best (if any) is also added as a basis.

2. **Linear‑programming mix** – for each length we solve a tiny LP:
      minimise  t
      s.t.      Σ_i w_i·conv_i[j] ≤ t   (for every convolution index j)
                Σ_i w_i = 1,   w_i ≥ 0
   where `conv_i` is the self‑convolution of basis `i`.  The LP returns a
   convex combination of the bases that yields the smallest possible maximum
   convolution entry among the span of the bases.

3. **Gradient‑based refinement** – the best mixed vector is refined with
   several optimisation phases:
      * Exponentiated‑gradient descent on a smooth‑max approximation.
      * Projected gradient descent on the true sub‑gradient.
      * Adam on the log‑it (soft‑max) parametrisation.
      * A simple hill‑climber that moves a tiny mass from the worst to the
        best gradient coordinate, with occasional random jitter.

4. **Random restarts** – while time remains we draw fresh Dirichlet
   vectors of various lengths and run a shortened version of the refinement
   pipeline, keeping any improvement.

All vectors are kept on the probability simplex (sum = 1) and all entries
are forced to be non‑negative, so the ``evaluate_sequence`` wrapper will not
trigger its security checks.  The whole search respects the 1000 s wall‑clock
budget.

The function returns the best sequence found as a plain list of floats.
"""

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families – each returns a probability vector (sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.01):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    """Discrete Gaussian centred at (n‑1)/2."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _smooth_weights(conv, alpha):
    """Smooth max weights: w_j ∝ exp(alpha * (conv_j – max(conv)))."""
    maxc = conv.max()
    w = np.exp(alpha * (conv - maxc))
    s = w.sum()
    if s == 0:
        return np.full_like(w, 1.0 / w.size)
    return w / s


def _subgradient_weights(conv, eps_rel=1e-12):
    """Uniform weights over entries that are (almost) maximal."""
    m = conv.max()
    w = np.zeros_like(conv)
    w[conv >= m - eps_rel * (abs(m) + 1.0)] = 1.0
    cnt = w.sum()
    if cnt == 0:
        return w
    return w / cnt

def _exponential_gradient_step(a, lr, alpha):
    """Exponentiated‑gradient step on the simplex."""
    conv = _fft_self_convolve(a)
    w = _smooth_weights(conv, alpha)
    grad = 2.0 * _fft_convolve(a[::-1], w)[len(a) - 1 : len(a) - 1 + len(a)]
    a = a * np.exp(-lr * grad)
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def _projected_gradient_step(a, eta, active_w):
    """Projected gradient step on the true sub‑gradient."""
    grad = 2.0 * _fft_convolve(a[::-1], active_w)[len(a) - 1 : len(a) - 1 + len(a)]
    a = a - eta * grad
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def _adam_step(logits, m, v, t, step_size):
    """Adam optimizer on logits (unconstrained)."""
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = beta1 * m + (1.0 - beta1) * step_size
    v = beta2 * v + (1.0 - beta2) * (step_size ** 2)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    lr = 0.02 / (np.sqrt(v_hat) + eps)
    logits = logits - lr * m_hat
    logits = np.clip(logits, -30, 30)
    return logits, m, v

def _hillclimber_move(a, conv, eps=1e-12):
    """Move a tiny mass from a “bad’’ index to a “good’’ one."""
    n = a.shape[0]
    w = _subgradient_weights(conv, eps_rel=eps)
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    src = int(np.argmax(grad))
    dst = int(np.argmin(grad))
    if src == dst:
        dst = random.randint(0, n - 1)
    max_delta = min(a[src] * 0.5, 0.02)
    delta = max_delta * random.random()
    if delta < eps:
        return a, conv, False
    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    new_conv = _fft_self_convolve(a_new)
    improved = new_conv.max() < conv.max() - 1e-13
    return (a_new, new_conv, True) if improved else (a, conv, False)

        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform n={best_n} score={best_score:.6f}")

    # ----- refinement of the current best candidate -----
    best_seq = _refine_vector(best_seq, best_n, deadline)

    # ----- random restarts while time permits -----
    restart_lengths = [2000, 3000, 4000, 5000, 7000, 10000]
    while time.time() < deadline - 5.0:
        n = random.choice(restart_lengths)
        a0 = np.random.dirichlet(np.full(n, 0.5))
        a0 = np.maximum(a0, eps)
        a0 /= a0.sum()
        a0 = _refine_vector(a0, n, deadline)
        sc = _max_convolution_score(a0)
        if sc < best_score:
            best_seq, best_n, best_score = a0.copy(), n, sc
            print(f"[{time.time():.1f}] restart n={n} improved to {sc:.6f}")

    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
