"""
Optimiser for the step‑function problem.

The algorithm works in three stages:

1. **Exploration** – evaluate a wide variety of analytic families
   (uniform, arcsine, beta, sine‑power, exponential, Gaussian) and a few
   random Dirichlet draws for several lengths (up to 10000).  The best
   few candidates are kept.

2. **Refinement** – for each promising candidate run an intensive
   optimisation pipeline:
        * Exponential‑gradient descent with a smooth‑max surrogate.
        * Projected‑gradient descent on the sub‑gradient of the max‑convolution.
        * Adam on the log‑its (simple way to stay on the simplex).
        * A fast hill‑climber that moves a tiny amount of mass from the
          worst‑gradient index to the best‑gradient index.
   The number of iterations is adapted to the vector length and the
   remaining time budget.

3. **Random restarts** – while time permits spawn new Dirichlet vectors,
   run a shortened refinement and keep the global best.

All vectors are kept normalised to sum = 1, so the objective to minimise
is `score = 2·n·max(conv(a,a))`.  The implementation uses FFT‑based
convolution (`O(n log n)`) and only NumPy, SciPy (for the Toeplitz helper)
and the Python standard library.
"""

import time
import random
import math
import numpy as np
from scipy.linalg import toeplitz  # only for a tiny legacy helper (no LP used)


# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a):
    """Score = 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.01):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    if vals.sum() == 0:
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    """Discrete Gaussian centred at (n‑1)/2."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    if vals.sum() == 0:
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _smooth_weights(conv, alpha):
    """Soft‑max style weights for a smooth approximation of the max."""
    maxc = conv.max()
    w = np.exp(alpha * (conv - maxc))
    s = w.sum()
    if s == 0:
        return np.full_like(w, 1.0 / w.size)
    return w / s


def _subgradient_weights(conv, eps_rel=1e-12):
    """Weights for a sub‑gradient of the max‑convolution."""
    m = conv.max()
    w = np.zeros_like(conv)
    w[conv >= m - eps_rel * (abs(m) + 1.0)] = 1.0
    cnt = w.sum()
    if cnt == 0:
        return w
    return w / cnt


# ----------------------------------------------------------------------
def _exponential_gradient_step(a, lr, alpha):
    """One EG step (a ≥ 0, Σa = 1)."""
    conv = _fft_self_convolve(a)
    w = _smooth_weights(conv, alpha)
    grad = 2.0 * _fft_convolve(a[::-1], w)[len(a) - 1 : len(a) - 1 + len(a)]
    a = a * np.exp(-lr * grad)
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def _projected_gradient_step(a, eta, active_w):
    """PGD step on the sub‑gradient defined by active_w."""
    grad = 2.0 * _fft_convolve(a[::-1], active_w)[len(a) - 1 : len(a) - 1 + len(a)]
    a = a - eta * grad
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def _adam_step(logits, m, v, t, grad):
    """Adam update for logits (softmax parametrisation)."""
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    lr = 0.02 / (np.sqrt(v_hat) + eps)
    logits = logits - lr * m_hat
    logits = np.clip(logits, -30.0, 30.0)      # keep exp stable
    return logits, m, v


def _hillclimber_move(a, conv, eps=1e-12):
    """Move a tiny amount of mass from the worst to the best index."""
    n = a.shape[0]
    w = _subgradient_weights(conv, eps_rel=eps)
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    src = int(np.argmax(grad))
    dst = int(np.argmin(grad))
    if src == dst:
        dst = random.randint(0, n - 1)
    max_delta = min(a[src] * 0.5, 0.02)
    delta = max_delta * random.random()
    if delta < eps:
        return a, conv, False
    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    new_conv = _fft_self_convolve(a_new)
    improved = new_conv.max() < conv.max() - 1e-13
    return (a_new, new_conv, True) if improved else (a, conv, False)


# ----------------------------------------------------------------------
def _refine_vector(a, n, deadline):
    """
    Run a full optimisation pipeline on vector a (already normalised to sum=1).
    Returns the best vector found during the run and its score.
    """
    best_vec = a.copy()
    best_sc = _max_convolution_score(best_vec)

    # ----- Exponential‑gradient phase -----
    steps_eg = 2500 if n <= 4000 else 1200
    a_eg = a.copy()
    for it in range(steps_eg):
        if time.time() >= deadline:
            break
        t = it / steps_eg
        # smoothing goes from large to small
        alpha = 400.0 * (0.1 / 400.0) ** t          # 400 → 0.1
        lr = 0.6 * (0.001 / 0.6) ** t               # 0.6 → 0.001
        a_eg = _exponential_gradient_step(a_eg, lr, alpha)
        if it % 250 == 0 or it == steps_eg - 1:
            sc = _max_convolution_score(a_eg)
            if sc < best_sc - 1e-13:
                best_sc, best_vec = sc, a_eg.copy()
                print(f"[{time.time():.1f}] EG it={it}  best={sc:.6f}")

    # ----- Projected‑gradient phase -----
    steps_pg = 1500 if n <= 4000 else 600
    a_pg = a_eg.copy()
    for it in range(steps_pg):
        if time.time() >= deadline:
            break
        conv = _fft_self_convolve(a_pg)
        w = _subgradient_weights(conv, eps_rel=1e-12)
        eta = 0.4 * (0.01 / 0.4) ** (it / steps_pg)   # 0.4 → 0.01
        a_pg = _projected_gradient_step(a_pg, eta, w)
        if it % 200 == 0 or it == steps_pg - 1:
            sc = _max_convolution_score(a_pg)
            if sc < best_sc - 1e-13:
                best_sc, best_vec = sc, a_pg.copy()
                print(f"[{time.time():.1f}] PG it={it}  best={sc:.6f}")

    # ----- Adam on logits -----
    logits = np.log(a_pg + 1e-14)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    steps_adam = 3500 if n <= 4000 else 1500
    for it in range(1, steps_adam + 1):
        if time.time() >= deadline:
            break
        a_adam = np.exp(logits)
        a_adam /= a_adam.sum()
        conv = _fft_self_convolve(a_adam)
        w = _subgradient_weights(conv, eps_rel=1e-12)
        grad_a = 2.0 * _fft_convolve(a_adam[::-1], w)[n - 1 : n - 1 + n]
        # gradient w.r.t logits under softmax
        grad_l = (grad_a - np.dot(a_adam, grad_a)) * a_adam
        logits, m, v = _adam_step(logits, m, v, it, grad_l)
        if it % 400 == 0 or it == steps_adam:
            a_adam = np.exp(logits)
            a_adam /= a_adam.sum()
            sc = _max_convolution_score(a_adam)
            if sc < best_sc - 1e-13:
                best_sc, best_vec = sc, a_adam.copy()
                print(f"[{time.time():.1f}] Adam it={it}  best={sc:.6f}")

    # ----- Hill‑climber -----
    a_hc = best_vec.copy()
    conv = _fft_self_convolve(a_hc)
    hc_iters = 25000 if n <= 4000 else 10000
    for it in range(hc_iters):
        if time.time() >= deadline:
            break
        a_hc, conv, improved = _hillclimber_move(a_hc, conv)
        if improved:
            sc = 2.0 * n * conv.max()
            if sc < best_sc - 1e-13:
                best_sc, best_vec = sc, a_hc.copy()
                if it % 500 == 0:
                    print(f"[{time.time():.1f}] Hill it={it}  best={sc:.6f}")
        # occasional tiny jitter to escape plateaus
        if (it + 1) % 4000 == 0:
            jitter = np.random.randn(n) * 2e-5
            a_hc += jitter
            a_hc = np.maximum(a_hc, 1e-14)
            a_hc /= a_hc.sum()
            conv = _fft_self_convolve(a_hc)

    return best_vec, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns the best sequence found within the wall‑clock budget.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    eps = 1e-14
    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1.  Scan analytic families and a few Dirichlet draws
    scan_lengths = [2000, 3000, 4000, 6000, 8000, 10000]
    for n in scan_lengths:
        if time.time() >= deadline:
            break
        candidates = []

        # Analytic families
        candidates.append(_distribution_uniform(n))
        candidates.append(_distribution_arcsine(n))
        candidates.append(_distribution_triangular(n))
        candidates.append(_distribution_exponential(n, lam=0.006))
        candidates.append(_distribution_exponential(n, lam=0.02))
        for p in (0.2, 0.5, 1.0, 2.0, 4.0):
            candidates.append(_distribution_beta_sym(n, p))
        for pw in (0.3, 0.7, 1.5, 3.0):
            candidates.append(_distribution_sine_power(n, pw))
        for sigma in (0.15, 0.30, 0.45):
            candidates.append(_distribution_gaussian(n, sigma))

        # Random Dirichlet draws (low concentration favours sparsity)
        for alpha in (0.2, 0.5, 1.0, 2.0):
            candidates.append(np.random.dirichlet(np.full(n, alpha)))

        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] init n={n}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  Incorporate a possibly existing global solution `height_sequence_1`
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_seq = prev.copy()
                best_n = prev.size
                best_score = sc
                print(f"[{time.time():.1f}] global best {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback – uniform if everything else failed
    if best_seq is None:
        best_n = 3000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Keep the top‑few candidates for deep refinement
    # (sorted by score, keep up to 3)
    top_candidates = [(best_score, best_seq.copy(), best_n)]
    # we also keep the second‑best among the scanned families if any
    # (a cheap way: re‑scan and retain a few additional low‑scoring vectors)
    extra = []
    for n in scan_lengths:
        if time.time() >= deadline:
            break
        # take a random Dirichlet for each length (different seed)
        a = np.random.dirichlet(np.full(n, 0.5))
        a = np.maximum(a, eps)
        a /= a.sum()
        sc = _max_convolution_score(a)
        extra.append((sc, a, n))
    extra.sort(key=lambda x: x[0])
    for sc, a, n in extra[:2]:
        top_candidates.append((sc, a, n))

    # ------------------------------------------------------------------
    # 5.  Deep refinement of the best few vectors
    for _, vec, n in sorted(top_candidates, key=lambda x: x[0])[:3]:
        if time.time() >= deadline:
            break
        refined_vec, refined_sc = _refine_vector(vec, n, deadline)
        if refined_sc < best_score:
            best_seq, best_score, best_n = refined_vec.copy(), refined_sc, n
            print(f"[{time.time():.1f}] refined n={n}  new best {refined_sc:.6f}")

    # ------------------------------------------------------------------
    # 6.  Random restarts while time permits
    restart_lengths = [2000, 2500, 3000, 3500, 4000, 5000, 8000, 10000]
    while time.time() < deadline - 5.0:
        n = random.choice(restart_lengths)
        a0 = np.random.dirichlet(np.full(n, 0.5))
        a0 = np.maximum(a0, eps)
        a0 /= a0.sum()
        # a shortened but still effective refinement
        a0_ref, sc0 = _refine_vector(a0, n, deadline)
        if sc0 < best_score:
            best_seq, best_score, best_n = a0_ref.copy(), sc0, n
            print(f"[{time.time():.1f}] restart n={n}  new best {sc0:.6f}")

    # ------------------------------------------------------------------
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    # Return as plain Python list; evaluate_sequence will handle normalisation.
    return best_seq.tolist()
