"""
Propose a step‑function (list of non‑negative floats) with a low
``evaluate_sequence`` score.

The algorithm consists of three stages:

1. **Diverse initialisation** – generate many analytic families (Beta,
   power‑edge, V‑shape, arcsine, etc.) for several lengths (powers of two).
   Keep the best vector according to a fast proxy
   ``score = 2·n·max(conv)`` (the true objective with sum = 1).

2. **Large‑scale Adam mirror‑descent** – minimise a
   smoothed version of the max‑convolution using an adaptive
   exponential‑gradient (Adam).  The smoothing parameter α grows from
   0.01 to ≈30 while the learning‑rate decays, and after each step the
   vector is projected back onto the probability simplex.

3. **Fine‑grained hill‑climber** – repeatedly identify the pair of indices
   that contributes most to the current maximal convolution value and
   transfer a tiny amount of mass away from that pair.  Moves that lower
   the true max are kept; otherwise they are discarded.

All work respects the 1000 s wall‑clock budget; intermediate best scores
are printed.  The returned list is already normalised and contains only
non‑negative numbers (a tiny epsilon avoids exact zeros).
"""

import time

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _fast_score(a: np.ndarray) -> float:
    """Fast proxy of evaluate_sequence when sum(a)=1:
       2·n·max(conv) .
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * float(conv.max())

# Analytic families – each returns a *probability* vector (sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.01) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * (np.log(x) + np.log(1 - x))
    w = np.exp(logw)
    if w.sum() == 0 or np.any(np.isinf(w)):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals /= vals.sum()
    return vals


def _distribution_power_edge(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - i - 0.5)
    eps = 1e-12
    vals = np.power(denom + eps, -alpha)
    vals /= vals.sum()
    return vals


def _distribution_V_shape(n: int) -> np.ndarray:
    centre = (n - 1) / 2.0
    vals = np.abs(np.arange(n, dtype=np.float64) - centre)
    eps = 1e-12
    vals = np.maximum(vals, eps)
    vals /= vals.sum()
    return vals


def _distribution_power_abs(n: int, beta: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    vals = np.abs(np.arange(n, dtype=np.float64) - centre) ** beta
    eps = 1e-12
    vals = np.maximum(vals, eps)
    vals /= vals.sum()
    return vals


def _random_symmetric_dirichlet(n: int, alpha: float) -> np.ndarray:
    half = (n + 1) // 2
    half_vec = np.random.dirichlet(np.full(half, alpha))
    if n % 2 == 0:
        full = np.concatenate([half_vec, half_vec[::-1]])
    else:
        full = np.concatenate([half_vec, half_vec[-2::-1]])
    full /= full.sum()
    return full


# ----------------------------------------------------------------------
def _initial_scan(budget_s: float, deadline: float):
    """
    Scan a rich collection of analytic families for several lengths.
    Returns (best_vector, best_score, best_n)
    """
    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    best_n = None

    # Lengths – powers of two are cheap for FFT.
    lengths = [2048, 4096, 8192]

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- generate candidate vectors for this n -----
        candidates = []

        # basic families
        candidates.append(_distribution_uniform(n))
        candidates.append(_distribution_arcsine(n))
        candidates.append(_distribution_triangular(n))
        candidates.append(_distribution_V_shape(n))

        # exponential
        for lam in (0.001, 0.005, 0.01, 0.02, 0.05):
            candidates.append(_distribution_exponential(n, lam))

        # sine‑power
        for pw in (0.5, 1.0, 2.0, 4.0, 8.0):
            candidates.append(_distribution_sine_power(n, pw))

        # Gaussian
        for sigma in (0.1, 0.2, 0.3, 0.5):
            candidates.append(_distribution_gaussian(n, sigma))

        # symmetric Beta
        for p in (0.001, 0.005, 0.01, 0.02, 0.05,
                  0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
            candidates.append(_distribution_beta_sym(n, p))

        # power‑edge (allow α>1 as well)
        for a_edge in (0.5, 0.7, 0.9, 1.0, 1.2, 1.5, 2.0):
            candidates.append(_distribution_power_edge(n, a_edge))

        # power‑abs
        for beta in (0.5, 1.0, 2.0, 3.0, 5.0, 8.0):
            candidates.append(_distribution_power_abs(n, beta))

        # random Dirichlet draws (few per α)
        for alpha in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1):
            for _ in range(3):
                candidates.append(_random_symmetric_dirichlet(n, alpha))

        # evaluate all candidates
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _fast_score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_score = sc
                best_n = n
                print(f"[{time.time():.1f}] init n={n}  score={sc:.6f}")

    return best_seq, best_score, best_n


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, time_limit: float):
    """
    Run Adam‑style mirror descent on the smoothed max‑convolution.
    Returns the best vector found and its proxy score.
    """
    eps = 1e-12
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # schedule
    total_steps = 80000
    alpha_start = 0.01
    alpha_end = 30.0
    lr_start = 3.0
    lr_end = 0.01

    best_vec = a.copy()
    best_sc = _fast_score(a)

    for it in range(total_steps):
        if time.time() >= time_limit:
            break

        t = it / total_steps
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t

        # convolution and smoothed weights
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # soft‑max over the 2n‑1 entries
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient (see analysis notes)
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        # Adam update (additive version)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = a - step

        # project back onto the simplex and enforce epsilon
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()

        # occasional reporting
        if it % 2000 == 0 or it == total_steps - 1:
            sc = _fast_score(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_vec = a.copy()
                print(f"[{time.time():.1f}] Adam it={it}  best_proxy={sc:.6f}")

    return best_vec, best_sc


# ----------------------------------------------------------------------
def _final_hill_climber(a0: np.ndarray, time_limit: float):
    """
    Tiny greedy mass‑transfer that focuses on the pair (i,j) that
    contributes most to the current maximal convolution entry.
    """
    eps = 1e-12
    a = a0.copy()
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_vec = a.copy()
    best_sc = 2.0 * n * M

    it = 0
    while time.time() < time_limit:
        it += 1

        # locate the index where the maximum occurs
        k = int(conv.argmax())
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)

        # find the pair (i,j) that maximises a_i * a_j for this k
        best_i, best_j = i_low, k - i_low
        max_prod = a[best_i] * a[best_j]
        for i in range(i_low, i_high + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        max_delta = min(a[source] * 0.5, 0.015)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a_new = a.copy()
        a_new[source] -= delta
        a_new[target] += delta
        a_new = np.maximum(a_new, eps)
        a_new /= a_new.sum()

        conv_new = _fft_self_convolve(a_new)
        M_new = conv_new.max()

        if M_new < M - 1e-13:
            a = a_new
            conv = conv_new
            M = M_new
            sc = 2.0 * n * M
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_vec = a.copy()
                print(f"[{time.time():.1f}] hill it={it}  best={sc:.6f}")
        else:
            # occasionally shake the system to escape plateaus
            if it % 5000 == 0:
                jitter = np.random.randn(n) * 1e-5
                a += jitter
                a = np.maximum(a, eps)
                a /= a.sum()
                conv = _fft_self_convolve(a)
                M = conv.max()
                print(f"[{time.time():.1f}] jitter applied at hill iter {it}")

    return best_vec, best_sc

def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function vector that minimises ``evaluate_sequence``.
    Returns the best sequence found within the given time budget.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # keep a safety margin for the caller
    global_deadline = start_time + budget_s - 2.0
    print(f"[{time.time():.1f}] starting search (budget {budget_s}s)")

    # ------------------------------------------------------------------
    # 1.  Broad family scan
    best_seq, best_score, best_n = _initial_scan(budget_s, global_deadline)

    # ------------------------------------------------------------------
    # 2.  If a previous sequence exists, try it as a seed
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 1e-12)
            prev /= prev.sum()
            sc = _fast_score(prev)
            if sc < best_score:
                best_seq = prev.copy()
                best_score = sc
                best_n = prev.size
                print(f"[{time.time():.1f}] using previous sequence, score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Adam mirror‑descent on the smoothed max‑convolution
    deadline_adam = global_deadline - 30.0  # leave a little room for hill‑climb
    a_adam, sc_adam = _adam_descent(best_seq, deadline_adam)
    if sc_adam < best_score:
        best_seq = a_adam.copy()
        best_score = sc_adam
        best_n = best_seq.shape[0]
        print(f"[{time.time():.1f}] post‑Adam best proxy score={sc_adam:.6f}")

    # ------------------------------------------------------------------
    # 4.  Final greedy hill‑climber
    deadline_hill = global_deadline
    a_hill, sc_hill = _final_hill_climber(best_seq, deadline_hill)
    if sc_hill < best_score:
        best_seq = a_hill.copy()
        best_score = sc_hill
        best_n = best_seq.shape[0]
        print(f"[{time.time():.1f}] final hill‑climber best proxy={sc_hill:.6f}")

    # ------------------------------------------------------------------
    # Compute the *true* evaluation score for reporting (uses the provided
    # evaluate_sequence function).  This may be slightly more expensive
    # because np.convolve is used internally, but we only call it once.
    try:
        true_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        true_score = float("nan")
    print(f"[{time.time():.1f}] DONE – length={best_n}  proxy={best_score:.6f}  true={true_score:.6f}")

    # return a plain Python list as required
    return best_seq.tolist()
