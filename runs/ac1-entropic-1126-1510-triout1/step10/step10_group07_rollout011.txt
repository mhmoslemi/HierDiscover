"""
propose_candidate – 1000 s heuristic search for a low‑score non‑negative step function.

The routine builds a modest set of analytic families (beta, edge‑power,
Gaussian, double‑bump, exponential, Dirichlet draws, etc.) for several
lengths (from a few thousand up to ≈ 20000).  Each candidate is normalised
to sum 1 and its exact surrogate score 2·n·max(conv) is evaluated via FFT.
The best *k* seeds are refined by two first‑order methods on the simplex
(Adam and exponentiated‑gradient) and then polished with a Frank‑Wolfe style
mass‑transfer plus a short simulated‑annealing jitter phase.  The
implementation stays fully vectorised, respects the 1000 s limit and returns
the best list of floats found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """
    Exact surrogate objective when Σa = 1:
        2·n·max(conv(a,a))   == evaluate_sequence(a)   (since Σa=1)
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) under Σa = 1.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # gradient of the (non‑smooth) max‑convolution:
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for i=j terms (when k even)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.35,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 2_000_000):
    """Adam on the simplex (projected). Returns best point seen."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          eta0: float = 0.9,
                          max_iters: int = 1_500_000):
    """Exponentiated‑gradient (KL‑mirror) descent on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score

def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with concentration ``alpha``."""
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0:
        return np.full_like(x, 1.0 / n)
    return x / x.sum()

# Analytic families ----------------------------------------------------
def _uniform_distribution(n: int) -> np.ndarray:
    a = np.full(n, 1.0 / n)
    return a


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # w ∝ x^{p‑1} (1‑x)^{p‑1}
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _edge_power_distribution(n: int, s: float) -> np.ndarray:
    """
    w_i ∝ (i·(n‑i+1))^{-s},  i = 1,…,n.
    Generalises the arcsine case (s=0.5).
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = (i * (n - i + 1)) ** (-s)
    vals /= vals.sum()
    return vals


def _double_bump_distribution(n: int,
                              block_frac: float = 0.25,
                              gap_frac: float = 0.1) -> np.ndarray:
    """Two equal uniform blocks separated by a gap."""
    total_block = max(2, int(block_frac * n))
    m = total_block // 2
    g = int(gap_frac * n)
    a = np.zeros(n, dtype=np.float64)
    a[:m] = 1.0 / (2 * m)
    start2 = m + g
    if start2 + m > n:
        start2 = n - m
    a[start2:start2 + m] = 1.0 / (2 * m)
    return a


def _gaussian_distribution(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    """Truncated Gaussian centred in the middle, sigma = factor·n."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    vals /= vals.sum()
    return vals

        lengths.add(n)
    lengths = sorted(lengths)

    print(f"[{time.time():.1f}] Exploring {len(lengths)} lengths (base≈{base_n})")

    # ------------------------------------------------------------------
    # Hyper‑parameter grids (expanded compared to the baseline)
    beta_ps = np.arange(0.10, 5.01, 0.10)            # symmetric beta
    asym_grid = [(0.2, 0.8), (0.3, 0.7), (0.5, 0.5),
                 (0.7, 0.3), (0.8, 0.2)]
    power_ps = [0.3, 0.5, 0.8, 1.0, 1.5, 2.0]
    lam_vals = [0.001, 0.003, 0.006, 0.01, 0.02, 0.04]
    dirichlet_alphas = [0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    edge_ps = np.arange(0.10, 0.91, 0.05)            # generalized arcsine
    block_fracs = [0.10, 0.15, 0.20, 0.25]
    gap_fracs = [0.05, 0.10, 0.15]
    sigma_factors = [0.10, 0.15, 0.20, 0.25, 0.30]

    # ------------------------------------------------------------------
    # Build candidates
    candidates = []   # list of (array, surrogate_score)

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- deterministic families ---------------------------------
        a = _uniform_distribution(n)
        candidates.append((a, _max_convolution_score(a)))
        a = _triangular_distribution(n)
        candidates.append((a, _max_convolution_score(a)))
        a = _arcsine_distribution(n)
        candidates.append((a, _max_convolution_score(a)))
        a = _exponential_distribution(n, lam=0.005)
        candidates.append((a, _max_convolution_score(a)))

        # ----- parametric families ------------------------------------
        for lam in lam_vals:
            a = _exponential_distribution(n, lam=lam)
            candidates.append((a, _max_convolution_score(a)))

        for p in beta_ps:
            a = _beta_symmetric_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))

        for a_alpha, a_beta in asym_grid:
            a = _beta_asymmetric_distribution(n, a_alpha, a_beta)
            candidates.append((a, _max_convolution_score(a)))

        for p in power_ps:
            a = _power_law_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))

        for s in edge_ps:
            a = _edge_power_distribution(n, s)
            candidates.append((a, _max_convolution_score(a)))

        for bf in block_fracs:
            for gf in gap_fracs:
                a = _double_bump_distribution(n, block_frac=bf, gap_frac=gf)
                candidates.append((a, _max_convolution_score(a)))

        for sf in sigma_factors:
            a = _gaussian_distribution(n, sigma_factor=sf)
            candidates.append((a, _max_convolution_score(a)))

        # ----- random Dirichlet draws ---------------------------------
        for alpha in dirichlet_alphas:
            a = _random_dirichlet_alpha(n, alpha)
            candidates.append((a, _max_convolution_score(a)))

    # ------------------------------------------------------------------
    # Seed from previous best if it exists (plus small jittered copies)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            if prev.sum() == 0:
                prev = np.full_like(prev, 1.0 / prev.size)
            else:
                prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
            for _ in range(6):
                jitter = np.random.randn(prev.size) * 5e-5
                noisy = _project_onto_simplex(prev + jitter, s=1.0)
                candidates.append((noisy, _max_convolution_score(noisy)))
            print(f"[{time.time():.1f}] added previous best (+perturbations)")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Keep the best k seeds for intensive local refinement
    candidates.sort(key=lambda x: x[1])
    top_k = min(30, len(candidates))
    top_seeds = candidates[:top_k]
    print(f"Generated {len(candidates)} candidates, keeping top {top_k} for refinement.")

    best_seq = top_seeds[0][0].copy()
    best_score = top_seeds[0][1]
    print(f"Initial best surrogate = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Refine each seed (Adam → ExpGradient)
    for idx, (seed_seq, _) in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        n = seed_seq.shape[0]

        # Adam descent
        seq_adam, sc_adam = _adam_descent(
            seed_seq, deadline, max_iters=1_200_000)
        if sc_adam < best_score - 1e-14:
            best_score, best_seq = sc_adam, seq_adam.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} Adam → {best_score:.6f}")

        # Exponentiated‑gradient descent
        seq_eg, sc_eg = _exp_gradient_descent(
            seed_seq, deadline, max_iters=1_000_000)
        if sc_eg < best_score - 1e-14:
            best_score, best_seq = sc_eg, seq_eg.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} ExpGrad → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Frank‑Wolfe style “mass‑transfer’’ polishing
    a = best_seq.copy()
    cur_score = best_score
    fw_iter = 0
    fractions = (1.0, 0.75, 0.5, 0.25, 0.125)

    while time.time() < deadline and fw_iter < 300_000:
        fw_iter += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-15:
            break
        improved = False
        for f in fractions:
            delta = max_transfer * f
            a_new = a.copy()
            a_new[i_src] -= delta
            a_new[i_dst] += delta
            sc = _max_convolution_score(a_new)
            if sc < cur_score - 1e-13:
                a = a_new
                cur_score = sc
                improved = True
                break
        if improved and cur_score < best_score - 1e-14:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time():.1f}] FW iter={fw_iter} new best {best_score:.6f}")

        # occasional random jitter to escape tiny plateaus
        if fw_iter % 9_000 == 0 and np.random.rand() < 0.13:
            jitter = np.random.randn(a.size) * 4e-5
            a = _project_onto_simplex(a + jitter, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    # Simulated‑annealing / jitter final push
    a = best_seq.copy()
    cur_score = best_score
    temp_start = 0.028
    temp_end = 0.0012
    anneal_iter = 0

    while time.time() < deadline and anneal_iter < 300_000:
        anneal_iter += 1
        # linear temperature schedule (remaining fraction of budget)
        t_rem = deadline - time.time()
        fraction = max(0.0, t_rem / (budget_s - 0.5))
        temp = temp_end + (temp_start - temp_end) * fraction

        i = np.random.randint(a.size)
        j = np.random.randint(a.size - 1)
        if j >= i:
            j += 1
        max_delta = min(a[i], 1.0 - a[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * temp
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if cur_score < best_score - 1e-14:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time():.1f}] Anneal iter={anneal_iter} new best {best_score:.6f}")
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_score - cur_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                a = a_new
                cur_score = new_score

    # ------------------------------------------------------------------
    print(f"Final best surrogate = {best_score:.6f} (length={len(best_seq)})")
    # The external ``evaluate_sequence`` will apply the necessary scaling.
    return best_seq.tolist()
