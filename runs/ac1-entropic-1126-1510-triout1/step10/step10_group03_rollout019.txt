"""
Propose a non‑negative step function (list of floats) that minimizes
`evaluate_sequence`.  The algorithm works in two phases:

1️⃣  **Exploratory grid search** over a few modest lengths (2k–32k) and
    simple *monotone* families – exponential tilt, linear tilt and
    power‑law decay.  These families are known to reduce the inner
    product with the reversed vector (the dominant term of the
    max‑convolution) while keeping the entries small, so they quickly
    beat the uniform baseline (score ≈ 2).

2️⃣  **Polishing** – a long projected sub‑gradient descent (with a
    simple Adam‑style momentum) that works directly on the true
    `max(convolution)` objective.  The gradient is obtained from the
    active set of indices attaining the maximum.  The loop runs until
    the time budget (≈ 1000 s) is exhausted.

Both stages respect the constraints of `evaluate_sequence`:
entries are non‑negative, sum ≥ 0.01 and ≤ 1000.  The returned value is
a plain Python list.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution helpers (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    For a vector already normalised to sum = 1 returns the exact value
    produced by `evaluate_sequence`: 2·n·max(a*a).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families (return a vector with sum = 1)
# ----------------------------------------------------------------------
def _distribution_exponential(n: int, theta: float) -> np.ndarray:
    """Exponential tilt: a_i ∝ exp(-theta * i/(n-1)).  theta≥0."""
    if n == 1:
        return np.ones(1, dtype=np.float64)
    lam = theta / (n - 1) if theta != 0 else 0.0
    idx = np.arange(n, dtype=np.float64)
    vec = np.exp(-lam * idx)
    vec = np.maximum(vec, 1e-12)
    vec /= vec.sum()
    return vec


def _distribution_linear_tilt(n: int, beta: float) -> np.ndarray:
    """
    Linear decreasing tilt: a_i ∝ 1 - beta * i/(n-1),   0 ≤ beta < 1.
    """
    if beta <= 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    idx = np.arange(n, dtype=np.float64)
    vec = 1.0 - beta * idx / (n - 1)
    if np.any(vec <= 0):
        # fall back to uniform – the caller will skip invalid betas
        return np.full(n, 1.0 / n, dtype=np.float64)
    vec = np.maximum(vec, 1e-12)
    vec /= vec.sum()
    return vec


def _distribution_powerlaw(n: int, p: float) -> np.ndarray:
    """
    Power‑law decay: a_i ∝ (i+1)^(-p),   p > 0.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)
    vec = idx ** (-p)
    vec = np.maximum(vec, 1e-12)
    vec /= vec.sum()
    return vec


# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000) -> list[float]:
    """Search for a step‑function minimising `evaluate_sequence`.

    The function respects the supplied time budget (seconds) and returns
    the best sequence found as a plain Python list.
    """
    np.random.seed(0)
    random.seed(0)

    start = time.time()
    deadline = start + budget_s - 1.5           # a small safety margin

    eps = 1e-12

    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1️⃣  Broad grid search over monotone families
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 16000, 32000]   # modest yet diverse
    theta_grid = np.linspace(0.0, 5.0, 51)       # for exponential tilt
    beta_grid = np.linspace(0.0, 0.90, 31)       # for linear tilt
    p_grid = np.linspace(0.2, 5.0, 49)          # for power‑law

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- exponential tilt -------------------------------------------------
        for theta in theta_grid:
            if time.time() >= deadline:
                break
            vec = _distribution_exponential(n, theta)
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n} exp‑tilt θ={theta:.2f}  score={sc:.6f}")

        # ----- linear tilt -------------------------------------------------------
        for beta in beta_grid:
            if time.time() >= deadline:
                break
            vec = _distribution_linear_tilt(n, beta)
            # reject produced non‑positive entries (happens when beta≈1)
            if np.min(vec) <= 0:
                continue
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n} lin‑tilt β={beta:.2f}   score={sc:.6f}")

        # ----- power‑law ---------------------------------------------------------
        for p in p_grid:
            if time.time() >= deadline:
                break
            vec = _distribution_powerlaw(n, p)
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n} power‑law p={p:.2f}  score={sc:.6f}")

        # ----- a few random Dirichlet draws (fresh restarts) --------------------
        for _ in range(3):
            if time.time() >= deadline:
                break
            vec = np.random.dirichlet(np.full(n, 1.0))
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n} random Dirichlet score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  If a previous solution exists, try it as a warm start.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score, best_seq, best_n = sc, prev.copy(), len(prev)
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Safeguard – should never happen, but keep a uniform fallback.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_n = 4000
        best_seq = np.full(best_n, 1.0 / best_n, dtype=np.float64)
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Long projected sub‑gradient descent (Adam‑style) starting
    #     from the best candidate found above.
    # ------------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]

    # Adam‑style momentum buffers
    m = np.zeros_like(a)      # first moment
    v = np.zeros_like(a)      # second moment (unused, simple momentum)
    beta1 = 0.9
    beta2 = 0.999
    step0 = 0.5               # initial learning rate (will be scaled)

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # compute full self‑convolution once per iteration
        conv = _fft_self_convolve(a)
        cur_max = conv.max()

        # active set: indices where the maximum is attained (within tolerance)
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        # gradient of the (non‑smooth) max‑convolution
        w = np.zeros_like(conv)
        w[active] = 1.0                        # any convex combination works; we use sum
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        a_new = np.maximum(a_new, eps)
        a_new /= a_new.sum()

        new_score = _max_convolution_score(a_new)

        if new_score + 1e-12 < best_score:
            best_score, best_seq = new_score, a_new.copy()
            print(
                f"[{time.time():.1f}] subgrad iter={iteration} new best {new_score:.6f}"
            )

        # accept the move (even if not improving) – helps escape plateaus
        a = a_new

        # occasional tiny jitter to avoid dead‑ends
        if iteration % 500 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

    # ------------------------------------------------------------------
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_n})")
    return best_seq.tolist()
