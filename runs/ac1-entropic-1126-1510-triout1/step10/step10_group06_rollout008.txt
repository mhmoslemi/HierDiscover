    while time.time() < deadline and iter < max_iter:
        conv = np.convolve(f, f)
        M = conv.max()
        # build matrix A
        A = np.zeros((2*n-1, n))
        for i in range(n):
            A[i:i+n, i] = f[i]
        b = np.full(2*n-1, M)
        c = -np.ones(n)  # maximize sum(g)
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')

            # reduce step size

def hillclimb(a, n, deadline, max_iter=50000, step=0.001):
    best_a = a.copy()
    best_conv = np.convolve(best_a, best_a)
    best_max = best_conv.max()
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # choose source and target random
        i, j = random.sample(range(n), 2)
        delta = step * best_a[i]  # moving fraction

        m = conv.max()
        if m < best_max:
            best_a = a_new
            best_max = m
            # optionally decrease step
    return best_a

def _lp_solve(f, M):
    n = len(f)
    # Build A
    rows = 2 * n -1
    A = np.empty((rows, n), dtype=np.float64)
    # Fill quickly:
    for i in range(n):
        A[i:i+n, i] = f[i]
    b = np.full(rows, M, dtype=np.float64)
    res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return None
    return res.x

def _lp_refine(f, n, deadline, max_iter=30, base_t=0.3):
    best_f = f
    best_score = _score(f)
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        M = np.convolve(best_f, best_f).max()
        g0 = _lp_solve(best_f, M)
        if g0 is None:
            break
        sum_g = g0.sum()
        if sum_g <= 1 + 1e-8:
            break
        g = g0 / sum_g
        # try update with t
        t = base_t
        improved=False
        while t > 1e-4:
            f_new = (1 - t) * best_f + t * g
            sc_new = _score(f_new)
            if sc_new < best_score - 1e-12:
                best_f, best_score = f_new, sc_new
                improved=True
                break
            else:
                t *= 0.5
        if not improved:
            # no improvement; break loop
            break
        # optional: print iteration progress
        print(f"[{time.time():.1f}] LP it={it} best_score={best_score:.6f}")
    return best_f

"""
LP‑refine step‑function optimizer.

Idea
----
1. Scan a few classic families (arcsine, beta, random Dirichlet) on several
   lengths, keep the best.
2. Starting from that vector, run a small number of iterations of a
   *linear‑programming refinement*:
       given f (∑f = 1) and M = max(f∗f), solve
            max ∑g   s.t.   f∗g ≤ M ,   g ≥ 0 .
       The optimum gives a vector g0 whose ℓ1‑norm is ≥ 1.  Rescaling to unit
       sum yields a candidate that can strictly reduce the convolution peak.
       A convex combination f←(1‑t)f + t g (t≈0.3) is accepted if it improves the
       objective.
3. Finish with a cheap hill‑climber that moves a tiny mass from the index
   with the largest gradient of the active‐max entry to the one with the
   smallest gradient.
4. Return the best sequence found before the budget expires.

The objective for a unit‑sum vector a of length n is
        score(a) = 2·n·max(a∗a) .
Only this value is used for internal bookkeeping; the final list is
returned unchanged (the evaluation routine will normalise it itself).
"""

import time
import random
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _conv_self(a):
    """Full linear self‑convolution a*a."""
    return np.convolve(a, a)


def _score(a):
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def _subgradient_weights(conv, eps_rel=1e-12):
    """Weights for a (sub)gradient of the max‑convolution."""
    m = conv.max()
    w = np.zeros_like(conv)
    w[conv >= m - eps_rel * (abs(m) + 1.0)] = 1.0
    cnt = w.sum()
    if cnt == 0:
        return w
    return w / cnt


def _gradient_of_max(a, w):
    """∇ max(a∗a) w.r.t. a when the max‑entry weights are w."""
    n = a.shape[0]
    full = np.convolve(a[::-1], w)          # length = 3n‑2
    return 2.0 * full[n - 1 : n - 1 + n]    # exactly n components


def _hillclimber_move(a, conv, eps=1e-12):
    """Tiny mass transfer from a large‑gradient index to a small‑gradient one."""
    n = a.shape[0]
    w = _subgradient_weights(conv, eps_rel=eps)
    grad = _gradient_of_max(a, w)
    src = int(np.argmax(grad))
    dst = int(np.argmin(grad))
    if src == dst:
        dst = random.randrange(n)
    max_delta = min(a[src] * 0.5, 0.02)
    delta = max_delta * random.random()
    if delta < eps:
        return a, conv, False
    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    new_conv = _conv_self(a_new)
    improved = new_conv.max() < conv.max() - 1e-13
    return a_new, new_conv, improved


def _hillclimber_fine(a, deadline, max_iters=20000):
    """Simple stochastic hill‑climber that stops when time runs out."""
    n = a.shape[0]
    conv = _conv_self(a)
    best = a.copy()
    best_conv = conv
    best_max = conv.max()
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        best, best_conv, improved = _hillclimber_move(best, best_conv)
        if improved:
            best_max = best_conv.max()
            if it % 500 == 0:
                print(f"[{time.time():.1f}] hill it={it} score={2*n*best_max:.6f}")
        # occasional jitter to escape plateaus
        if it % 5000 == 0 and it > 0:
            jitter = np.random.randn(n) * 1e-5
            best += jitter
            best = np.maximum(best, 1e-12)
            best /= best.sum()
            best_conv = _conv_self(best)
    return best


# ----------------------------------------------------------------------
def _build_toeplitz(f):
    """Build the (2n‑1)×n matrix T such that T·g = f∗g."""
    n = f.shape[0]
    rows = 2 * n - 1
    T = np.empty((rows, n), dtype=np.float64)
    for i, fi in enumerate(f):
        T[i:i + n, i] = fi
    return T


def _lp_solve(f, M):
    """
    Solve   max ∑g   s.t.   f∗g ≤ M , g ≥ 0.
    Returns the optimal g (not normalised).  None on failure.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    # scale to avoid tiny numbers: (f/M)·g ≤ 1
    f_scaled = f / M
    A = np.empty((rows, n), dtype=np.float64)
    for i, fi in enumerate(f_scaled):
        A[i:i + n, i] = fi
    b = np.ones(rows, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # linprog does min, so -∑g
    res = linprog(c, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n,
                  method='highs')
    if not res.success:
        return None
    return res.x


def _lp_refine(a, deadline, max_iter=30, t=0.3):
    """
    Repeatedly apply the LP step to the current vector a.
    Returns an improved vector (still ℓ1‑normalised).
    """
    best = a.copy()
    best_sc = _score(best)
    n = best.shape[0]
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        conv = _conv_self(best)
        M = conv.max()
        g0 = _lp_solve(best, M)
        if g0 is None:
            break
        sum_g0 = g0.sum()
        if sum_g0 <= 1.0 + 1e-12:
            break
        g = g0 / sum_g0                # normalised to ℓ1‑norm 1
        # try a convex update; shrink step if it does not improve
        cur_t = t
        improved = False
        while cur_t > 1e-4:
            cand = (1.0 - cur_t) * best + cur_t * g
            sc = _score(cand)
            if sc < best_sc - 1e-13:
                best, best_sc = cand, sc
                improved = True
                print(f"[{time.time():.1f}] LP it={it} t={cur_t:.3f} score={sc:.6f}")
                break
            cur_t *= 0.5
        if not improved:
            break
    return best

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:

    random.seed(seed)
