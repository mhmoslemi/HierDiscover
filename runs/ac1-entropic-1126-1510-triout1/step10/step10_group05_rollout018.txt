"""
A fast “soft‑max” driven optimiser for the step‑function bound

    2·n·max(conv(a,a)) / (∑a)²

The key ideas
----------------
* The objective depends only on the *shape* of a (scaling cancels), so we keep
  ∑a = 1 and work on the simplex.
* Instead of a hard max we minimise a *smooth* approximation
      τ·log Σₖ exp(convₖ/τ) .
  Its gradient can be obtained in closed form and uses the same FFT‑based
  convolutions as the exact sub‑gradient.
* An exponentiated‑gradient (mirror‑descent) update is performed on the
  simplex, with a decreasing step size ηₜ = η₀/√t.
* The temperature τ is annealed (τ←τ·γ) so that the optimisation first sees a
  smooth landscape and gradually focuses on the true max.
* After the annealing we polish the best vector with a short hill‑climb and a
  Frank‑Wolfe style pairwise mass‑transfer.

The routine respects the supplied time budget (≈ 1000 s) and prints a short
progress log.  The returned list can be fed directly to `evaluate_sequence`.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate for `evaluate_sequence` when Σa = 1 (i.e. 2·n·max(conv))."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families – all vectors are ℓ₁‑normalised (∑a = 1)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _gaussian_distribution(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    w /= w.sum()
    return w


def _cauchy_distribution(n: int, scale_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    scale = scale_factor * n
    w = 1.0 / (1.0 + ((idx - centre) / scale) ** 2)
    w /= w.sum()
    return w


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Unsymmetric Beta(α,β) on a discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide='ignore', invalid='ignore'):
        w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    if np.isnan(w).any() or np.isinf(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Dirichlet(α,…,α) draw."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Smooth‑max gradient (softmax over the convolution entries)
def _smooth_gradient(a: np.ndarray, tau: float):
    """
    Returns (grad, current_max) where grad is the gradient of
    τ·log Σ exp(conv/τ) and current_max = max(conv).
    """
    n = a.shape[0]
    # self‑convolution
    conv = _fft_self_convolve(a)            # length 2n‑1
    cur_max = conv.max()

    # numerically stable softmax weights
    shifted = (conv - cur_max) / tau
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # convolution of reversed a with the weight vector
    conv_rev = _fft_convolve(a[::-1], w)    # length 2n‑1
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]

    # diagonal correction for even indices (k = 2j)
    diag = w[::2] * a                       # w[0], w[2], …, w[2n‑2]
    grad -= diag
    return grad, cur_max


# ----------------------------------------------------------------------
# Annealed exponentiated‑gradient optimiser (mirror‑descent on simplex)
def _annealed_exp_gradient(
    a0: np.ndarray,
    deadline: float,
    eta0: float = 0.8,
    tau_factor: float = 0.85,
    inner_steps: int = 1500,
    tau_min: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Optimise the max‑convolution objective by minimising a smooth
    approximation.  Temperature τ is decreased geometrically; at each τ we
    perform a few exponentiated‑gradient steps.
    Returns the best vector (ℓ₁‑normalised) and its surrogate score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    # initial temperature – a fraction of the current max‑convolution value
    _, init_max = _smooth_gradient(a, tau=1.0)   # just to compute max
    tau = init_max * 0.6 if init_max > 0 else 0.5

    step = 0
    while time.time() < deadline:
        # ----- inner loop at fixed τ -----
        for _ in range(inner_steps):
            if time.time() >= deadline:
                break
            step += 1
            grad, cur_max = _smooth_gradient(a, tau)

            # exponentiated‑gradient update on the simplex (mirror descent)
            eta = eta0 / np.sqrt(step)
            a = a * np.exp(-eta * grad)
            # normalise (mirrors Euclidean projection onto simplex)
            a = a / a.sum()

            # enforce exact symmetry – this greatly stabilises the search
            a = (a + a[::-1]) / 2.0
            a = a / a.sum()

            cur_score = _max_convolution_score(a)
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_a = a.copy()
        # ----- temperature annealing -----
        tau = max(tau * tau_factor, tau_min)
        # stop early if τ is already tiny (the smooth objective ≈ max)
        if tau <= tau_min:
            break
    return best_a, best_score

# Simple hill‑climber (random mass transfers that improve the hard max)
def _hill_climb(a: np.ndarray, deadline: float,
                max_iters: int = 300_000, epsilon: float = 0.02) -> tuple[np.ndarray, float]:
    """Randomly move a small fraction of mass between two coordinates if it improves."""
    best = a.copy()
    best_score = _max_convolution_score(best)
    n = best.size
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or best[i] < 1e-12:
            continue
        delta = min(best[i] * epsilon, 1.0 - best[j])
        if delta <= 0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = _max_convolution_score(cand)
        if sc < best_score:
            best = cand
            best_score = sc
    return best, best_score


# ----------------------------------------------------------------------
# Frank‑Wolfe style pairwise mass transfer
def _frank_wolfe_opt(a0: np.ndarray, deadline: float,
                    max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Iteratively shift mass from the index with largest sub‑gradient to the smallest."""
    a = a0.copy()
    cur_score = _max_convolution_score(a)
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)               # reuse exact sub‑gradient
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        delta = min(a[i_src], 1.0 - a[i_dst])
        if delta <= 0:
            break
        step = 0.7 * delta
        a[i_src] -= step
        a[i_dst] += step
        cur_score = _max_convolution_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        # occasional jitter
        if it % 4000 == 0 and np.random.rand() < 0.1:
            a = _project_onto_simplex(a + np.random.randn(a.size) * 1e-5, s=1.0)
            cur_score = _max_convolution_score(a)
    return best, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient (used by the Frank‑Wolfe helper above)
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of the max‑convolution functional on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]

    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the supplied time budget (≈ 1000 s) and returns the
    best vector discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.8   # safety margin

    # ------------------------------------------------------------------
    # Choose a base length – we keep the size moderate for speed.
    try:
        base_n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        base_n = 5000
    # Clamp to a reasonable window
    n = int(min(max(3000, base_n), 8000))
    print(f"[{time.time():.1f}] Using length n = {n}")

    # ------------------------------------------------------------------
    # Build a modest pool of analytic seeds (all ℓ₁‑normalised).
    candidates = []

    def _add(arr: np.ndarray):
        a = np.asarray(arr, dtype=np.float64)
        if a.ndim != 1 or a.size == 0:
            return
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            return
        a = a / s
        score = _max_convolution_score(a)
        candidates.append((a, score))

    # a handful of families (including extra extreme β‑parameters)
    families = [
        _arcsine_distribution,
        _triangular_distribution,
        lambda nn: _exponential_distribution(nn, lam=0.005),
        lambda nn: _gaussian_distribution(nn, sigma_factor=0.15),
        lambda nn: _cauchy_distribution(nn, scale_factor=0.15),
    ]

    for func in families:
        a = func(n)
        _add(a)

    # extra symmetric beta shapes (including very small p)
    for p in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        a = _beta_symmetric_distribution(n, p)
        _add(a)

    # unsymmetric beta draws (coarse grid)
    for alpha in [0.2, 0.5, 1.0, 2.0]:
        for beta in [0.2, 0.5, 1.0, 2.0]:
            a = _beta_distribution(n, alpha, beta)
            _add(a)

    # random Dirichlet with several concentrations
    for a in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        _add(_random_dirichlet_alpha(n, a))

    # ------------------------------------------------------------------
    # Inject the previously best vector if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        if prev.sum() > 0:
            prev = prev / prev.sum()
            _add(prev)
            print(f"Added previous best seed, score={candidates[-1][1]:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Keep the top few seeds
    candidates.sort(key=lambda x: x[1])
    top_k = min(5, len(candidates))
    seeds = [c[0] for c in candidates[:top_k]]
    print(f"Generated {len(candidates)} seeds, keeping top {top_k}.")
    print(f"Best seed surrogate = {candidates[0][1]:.6f}")

    # ------------------------------------------------------------------
    # Allocate time: 80 % for annealed optimisation, rest for polishing
    time_now = time.time()
    remaining = deadline - time_now
    anneal_deadline = time.time() + 0.80 * remaining

    best_seq = seeds[0].copy()
    best_score = _max_convolution_score(best_seq)

    # ------------------------------------------------------------------
    # Annealed soft‑max optimisation (run on the best seed, then refine others if time permits)
    for idx, seed in enumerate(seeds):
        if time.time() >= anneal_deadline:
            break
        print(f"[{time.time():.1f}] Annealed optimisation on seed {idx + 1}/{len(seeds)}")
        seq, sc = _annealed_exp_gradient(
            seed,
            deadline=anneal_deadline,
            eta0=0.8,
            tau_factor=0.85,
            inner_steps=1800,
            tau_min=1e-9,
        )
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"   --> new best after annealing: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing – hill‑climb then Frank‑Wolfe (if time permits)
    if time.time() < deadline - 30:
        print(f"[{time.time():.1f}] Hill‑climb polishing …")
        seq, sc = _hill_climb(best_seq, deadline, max_iters=400_000, epsilon=0.015)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"   hill‑climb improved to {best_score:.6f}")

    if time.time() < deadline - 20:
        print(f"[{time.time():.1f}] Frank‑Wolfe polishing …")
        seq, sc = _frank_wolfe_opt(best_seq, deadline, max_iters=400_000)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"   Frank‑Wolfe improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final surrogate score = {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
