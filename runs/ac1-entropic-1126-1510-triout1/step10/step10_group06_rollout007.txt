"""
A fresh optimizer for the step‑function problem.

The algorithm proceeds as follows:

1. **Initial scan** – evaluate several analytic families (uniform,
   arcsine, triangular, exponential, symmetric beta with a wide range of
   shape parameters, sine‑power) over a handful of lengths and keep the
   best distribution (all vectors are normalised to sum = 1).

2. **Refinement** – the current best vector is refined by a long
   exponential‑gradient phase (smooth approximation of the max‑convolution
   with a decreasing learning‑rate), a projected‑gradient phase,
   an Adam phase on the logits, and a large simulated‑annealing / hill‑climber
   phase.  After each update the vector is symmetrised (a ← (a+a[::-1])/2)
   and projected back onto the probability simplex.

3. **Random restarts** – while time permits new random Dirichlet
   initialisers are drawn (different lengths, different concentration)
   and a shortened refinement pipeline is executed.  Any improvement updates
   the global best.

All heavy work is done with FFT‑based convolutions, and the wall‑clock is
kept under the supplied 1000 s budget.  The final best sequence (a list of
floats) is returned.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities ------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _symmetrize(a):
    """Enforce a[i] = a[-1‑i] (average with its reverse)."""
    a = (a + a[::-1]) / 2.0
    return a


# ----------------------------------------------------------------------
# FFT convolution helpers ------------------------------------------------
def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a):
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
# Objective and gradient helpers ----------------------------------------
def _objective(a):
    """2·n·max(a*a) for a probability vector (sum≈1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


def _grad_from_weights(a, w):
    """Gradient of Σ w_k (a*a)_k w.r.t. a (w has length 2·n‑1)."""
    n = a.shape[0]
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    return grad


def _smooth_weights(conv, alpha):
    """Soft‑max weights approximating the max (used in EG)."""
    m = conv.max()
    w = np.exp(alpha * (conv - m))          # conv‑m ≤ 0 ⇒ stable
    s = w.sum()
    if s == 0:
        return np.full_like(w, 1.0 / w.size)
    return w / s


def _subgradient_weights(conv, eps_rel=1e-12):
    """Weights for a (sub)gradient of the true max (active set)."""
    m = conv.max()
    w = np.zeros_like(conv)
    w[conv >= m - eps_rel * (abs(m) + 1.0)] = 1.0
    cnt = w.sum()
    if cnt == 0:
        return w
    return w / cnt


# ----------------------------------------------------------------------
# Optimisation steps ----------------------------------------------------
def _exponential_gradient_step(a, lr, alpha):
    """One step of the Exponential‑Gradient update (a stays on simplex)."""
    conv = _fft_self_convolve(a)
    w = _smooth_weights(conv, alpha)
    grad = _grad_from_weights(a, w)
    a = a * np.exp(-lr * grad)
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def _projected_gradient_step(a, eta, active_w):
    """Projected‑gradient step using the true sub‑gradient."""
    grad = _grad_from_weights(a, active_w)
    a = a - eta * grad
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def _adam_step(logits, m, v, t, step):
    """Adam update (logits are unconstrained)."""
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = beta1 * m + (1.0 - beta1) * step
    v = beta2 * v + (1.0 - beta2) * (step ** 2)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    lr = 0.02 / (np.sqrt(v_hat) + eps)          # modest step‑size
    logits = logits - lr * m_hat
    logits = np.clip(logits, -30, 30)           # avoid overflow
    return logits, m, v

# Analytic families (all normalised to sum = 1) -------------------------
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.01):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑form to avoid overflow for tiny p
    logw = (p - 1) * (np.log(x) + np.log(1.0 - x))
    w = np.exp(logw - np.max(logw))          # shift for stability
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals
