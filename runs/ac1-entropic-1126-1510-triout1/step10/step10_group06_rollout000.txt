    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    t = 0
    best_local = a.copy()
    best_local_score = score_of(a)
    # schedule for alpha: start at 1, increase to maybe 5000 linearly or exponentially
    total_iter = 5000
    for i in range(total_iter):
        if time.time() >= time_limit: break
        # compute a
        a = np.exp(logits)
        a = a / a.sum()
        conv = _fft_self_convolve(a)
        # compute smooth max weight
        alpha = 1.0 * (10000.0/1.0) ** (i/total_iter) # maybe exponential from 1 to 1e4
        w = np.exp(alpha * (conv - conv.max()))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(conv, 1.0/conv.size)
        else:
            w = w / w_sum
        # gradient: g = 2 * conv_a_rev * w (like earlier)
        grad_a = 2.0 * _fft_convolve(a[::-1], w)[n-1:n-1+n]
        # gradient wrt logits under softmax:
        grad_l = (grad_a - np.dot(a, grad_a)) * a
        # adam update

def hillclimber(a, n, deadline):

    conv = _fft_self_convolve(a)
    best_a = a.copy()
    best_score = 2.0*n*conv.max()

            break

        w = _subgradient_weights(conv, eps_rel=eps)

        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        max_delta = min(a[src]*0.5, 0.02)
        delta = max_delta * random.random()
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        a_new = np.maximum(a_new, eps)
        a_new = a_new / a_new.sum()
        conv_new = _fft_self_convolve(a_new)

        if new_score + 1e-12 < best_score:

            best_a = a_new
            a = a_new
            conv = conv_new
        # occasional random jitter to escape
        if it % 5000 == 0 and it>0:
            jitter = np.random.randn(n) * 1e-5
            a = a + jitter
            a = np.maximum(a, eps)
            a = a / a.sum()
            conv = _fft_self_convolve(a)
    return best_a, best_score

"""
Propose a low‑score step‑function.

The algorithm consists of four stages.

1️⃣  **Broad analytic scan** – evaluate a large collection of classic
    families (uniform, arcsine, triangular, exponential, symmetric &
    asymmetric Beta, sine‑power, Gaussian) on several lengths and a
    handful of random Dirichlet draws.  The best vector is kept.

2️⃣  **Size extrapolation** – linearly interpolate the current best to
    a few larger lengths (up to 10 000) and keep any improvement.

3️⃣  **Refinement** – minimise the smooth approximation of the
    max‑convolution by Adam on the soft‑max (logits) parametrisation.
    The smoothness parameter α is annealed from 0.5 to 5 000, which
    allows the optimiser to first see a smooth landscape and then
    focus on the true maximum.

4️⃣  **Local search & restarts** – a cheap hill‑climber that moves a
    small mass from the most “harmful’’ index to the most “helpful’’
    one, plus many short random‑restart runs (random Dirichlet
    initialisations) that each perform a reduced refinement.  All
    improvements update a global best, which is finally returned.

The whole search respects the 1000 s wall‑clock budget.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a):
    """Score = 2·n·max(a*a) assuming Σa = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families (all return a vector with Σ = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.01):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if np.isnan(w).any() or w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n, p, q):
    """Asymmetric Beta(p,q) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1 - x))
    if np.isnan(w).any() or w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    """Discrete Gaussian centred at (n‑1)/2."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _subgradient_weights(conv, eps_rel=1e-12):
    """Weights for a (sub)gradient of the max‑convolution."""
    m = conv.max()
    w = np.zeros_like(conv)
    w[conv >= m - eps_rel * (abs(m) + 1.0)] = 1.0
    cnt = w.sum()
    if cnt == 0:
        return w
    return w / cnt


# ----------------------------------------------------------------------
def _adam_refine(a, n, deadline):
    """
    Adam on the logits with an annealed smooth‑max.
    Returns an improved (a, score) pair.
    """
    eps = 1e-12
    logits = np.log(a + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    best_a = a.copy()
    best_score = _max_convolution_score(a)

    max_iter = 3000                     # enough for convergence on moderate sizes
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # ---------- current vector ----------
        a = np.exp(logits)
        a = a / a.sum()

        # ---------- smooth max ----------
        # α grows exponentially from 0.5 to 5 000
        alpha = 0.5 * (5000.0 / 0.5) ** (it / max_iter)

        conv = _fft_self_convolve(a)

        # compute weights w ≈ soft‑max(conv, α)
        shift = conv.max()
        arg = alpha * (conv - shift)
        arg = np.clip(arg, -700, 700)               # avoid overflow
        w = np.exp(arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / w_sum

        # ---------- gradient w.r.t. a ----------
        grad_a = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # ---------- gradient w.r.t. logits (softmax) ----------
        dot = np.dot(a, grad_a)
        grad_l = (grad_a - dot) * a

        # ---------- Adam update ----------
        beta1, beta2 = 0.9, 0.999
        lr_const = 0.02

        m = beta1 * m + (1 - beta1) * grad_l
        v = beta2 * v + (1 - beta2) * (grad_l ** 2)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        lr = lr_const / (np.sqrt(v_hat) + 1e-8)
        logits = logits - lr * m_hat
        logits = np.clip(logits, -30, 30)           # keep numerics sane

        # ---------- record improvement ----------
        if it % 200 == 0 or it == max_iter:
            cur_score = _max_convolution_score(a)
            if cur_score < best_score - 1e-13:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] Adam it={it:4d}  best={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
def _hillclimber(a, n, deadline):
    """
    Tiny mass transfer from the index that contributes most to the
    current max to the index that contributes the least.
    """
    eps = 1e-12
    conv = _fft_self_convolve(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    it = 0
    while it < 20000 and time.time() < deadline:
        # active‑set weights for the (sub)gradient of the max
        w = _subgradient_weights(conv, eps_rel=eps)
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        src = int(np.argmax(grad))   # most harmful index
        dst = int(np.argmin(grad))   # most helpful index
        if src == dst:
            dst = random.randint(0, n - 1)

        max_delta = min(a[src] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta < eps:
            it += 1
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        a_new = np.maximum(a_new, eps)
        a_new /= a_new.sum()

        conv_new = _fft_self_convolve(a_new)
        new_score = 2.0 * n * conv_new.max()
        if new_score + 1e-13 < best_score:
            best_score = new_score
            best_a = a_new
            a = a_new
            conv = conv_new
            # occasional logging
            if it % 500 == 0:
                print(f"[{time.time():.1f}] hill it={it:5d}  best={new_score:.6f}")

        # small jitter every few thousand steps to escape plateaus
        if it % 5000 == 0 and it > 0:
            jitter = np.random.randn(n) * 1e-5
            a = a + jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)

        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
