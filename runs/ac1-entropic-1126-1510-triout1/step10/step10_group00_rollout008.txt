"""
Hybrid analytic/discrete search for a non‑negative step‑function that
minimises ``evaluate_sequence``.

Algorithm
---------
1.  Generate many candidate shapes on several lengths.
    • Classical families: uniform, arcsine, symmetric β, edge‑weighted.
    • Additional signal‑processing windows (Chebyshev, Kaiser) when SciPy
      is available.
    • Random Dirichlet and sparse seeds.
    • Convex mixes with the uniform vector.

2.  Keep the ~30 best candidates (objective = 2·n·max(conv) assuming sum=1).

3.  Refine the top‑few seeds with an accelerated mirror‑descent.
    Gradient is taken from the *k* largest convolution entries
    (k=3 by default).  Exponentiated‑gradient updates are combined with a
    decaying learning‑rate and a simplex projection.

4.  Final greedy polishing: random pairwise mass transfers that are
    kept only when they improve the exact objective.

The whole routine respects the 1000 s budget (a 2 s safety margin) and
returns a plain Python list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1.
    The input is assumed non‑negative.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# Optional windows from SciPy (if available) ---------------------------
try:
    from scipy.signal import chebwin, kaiser

    _HAS_SCI_SIX = True
except Exception:  # pragma: no cover
    _HAS_SCI_SIX = False

def _distribution_chebyshev(n: int, atten: float) -> np.ndarray:
    """
    Chebyshev window (Dolph–Chebyshev) with sidelobe attenuation ``atten`` dB.
    Returns a non‑negative vector normalized to sum = 1.
    """
    w = chebwin(n, at=atten, sym=True).astype(np.float64)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_kaiser(n: int, beta: float) -> np.ndarray:
    """
    Kaiser window with shape parameter ``beta``.
    Returns a non‑negative vector normalized to sum = 1.
    """
    w = kaiser(n, beta, sym=True).astype(np.float64)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

# Sub‑gradient / mirror descent ----------------------------------------
def _subgradient_descent_multi(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.7,
    top_k: int = 3,
    eval_interval: int = 120,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution objective.
    Gradient is accumulated from the ``top_k`` largest convolution entries.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        # indices of the largest entries (including the zero shift)
        idxs = np.argpartition(-conv, range(top_k))[:top_k]

        grad = np.zeros_like(a)
        for idx in idxs:
            ind = idx - np.arange(n)
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * a[ind[mask]]

        eta = lr0 / math.sqrt(t)            # decaying stepsize
        a = a * np.exp(-eta * grad)        # exponentiated‑gradient update
        # re‑normalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad it={t} new global best {best_score:.6f}")

        # safety cap
        if t >= 300000:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise mass transfer ----------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small amount of mass; keep move only if objective improves."""
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)

        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.4   # up to 40 % of source mass
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine combines analytic families, refined gradient descent and a
    final greedy polish, all within the allotted time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # small safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Scan a rich pool of analytic families.
    # ------------------------------------------------------------------
    lengths = [2000, 5000, 8000, 12000, 20000, 40000]   # larger lengths for finer discretisation
    # p‑grid: denser close to zero where the optimum lives
    p_grid = np.concatenate(
        [
            np.logspace(-2, -0.5, 12, base=10),
            np.logspace(-0.5, 0.0, 12, base=10),
            np.logspace(0.0, 0.7, 10, base=10),
        ]
    )
    lambda_grid = np.linspace(0.0, 1.0, 6)                # 0, .2, …, 1

    # optional windows already have their own parameter grids
    atten_grid = np.linspace(30, 150, 7)                # Chebyshev attenuation (dB)
    beta_grid = np.linspace(0.0, 14.0, 8)               # Kaiser beta

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # ---- deterministic families ------------------------------------
        uniform = _distribution_uniform(n)
        arcsine = _distribution_arcsine(n)
        for cand in (uniform, arcsine):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # ---- parameterised families ------------------------------------
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)

            for base in (beta, edge):
                for lam in lambda_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # ---- optional SciPy windows -----------------------------------
        if _HAS_SCI_SIX:
            for atten in atten_grid:
                if time.time() >= deadline:
                    break
                cand = _distribution_chebyshev(n, atten)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(f"[{time.time():.1f}] new Cheb len={n} att={atten:.0f} score={sc:.6f}")

            for beta_par in beta_grid:
                if time.time() >= deadline:
                    break
                cand = _distribution_kaiser(n, beta_par)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(f"[{time.time():.1f}] new Kaiser len={n} beta={beta_par:.2f} score={sc:.6f}")

        # ---- random Dirichlet seeds -----------------------------------
        for _ in range(6):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] new global best random Dirichlet len={n} score={sc:.6f}")

        # ---- sparse random seeds ---------------------------------------
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.01)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}")

        # keep only a modest number of best seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2.  Incorporate a previously discovered construction if present.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback in the unlikely case no candidate was generated.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 5000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Sub‑gradient descent on a few top seeds.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] starting subgradient descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _subgradient_descent_multi(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.8,
                top_k=3,
                eval_interval=120,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] after subgrad seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Final greedy local refinement.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final pairwise‑swap polishing")
        a_opt, sc_opt = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)): {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    # Return a plain Python list – evaluate_sequence will handle any needed normalisation.
    return overall_best_seq.tolist()
