def _adam_optimize(a, deadline, max_iters=200000, lr0=0.2, beta1=0.9, beta2=0.999, eps_adam=1e-8):

    n = len(a)

    m = np.zeros(n)

    v = np.zeros(n)

    best_a = a.copy()

    best_sc = _max_convolution_score(a)

    for t in range(1, max_iters+1):

        if time.time() >= deadline: break

        conv = _fft_self_convolve(a)

        maxc = conv.max()

        # active indices tolerance relative to max
        # else 0? Use small absolute tolerance.
        tol = maxc * 1e-12 + 1e-12

        active = np.where(conv >= maxc - tol)[0]

        w = np.zeros_like(conv)

        w[active] = 1.0

        grad = 2.0 * _fft_convolve(a[::-1], w)[n-1:n-1+n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad

        v = beta2 * v + (1 - beta2) * (grad**2)

        m_hat = m / (1 - beta1**t)

        v_hat = v / (1 - beta2**t)

        step = lr0 / (np.sqrt(v_hat) + eps_adam)

        a = a - step * m_hat

        # Project onto simplex

        a = _project_onto_simplex(a, 1.0)

        a = np.maximum(a, 1e-12)

        a /= a.sum()

        sc = _max_convolution_score(a)

        if sc < best_sc:

            best_sc = sc

            best_a = a.copy()

            print(f"[{time.time():.1f}] Adam iter={t} new best {sc:.6f}")

        # occasional random jitter

        if t % 5000 == 0:

            jitter = np.random.randn(n) * 1e-5

            a = np.maximum(a + jitter, 1e-12)

            a /= a.sum()

    # return best found

    return best_a, best_sc

def _hill_climb(a, deadline, max_attempts=5000):

    n = len(a)

    best_a = a.copy()

    best_sc = _max_convolution_score(a)

    for it in range(max_attempts):

        if time.time() >= deadline: break

        # Select source index with largest a

        source = np.argmax(a)

        # Choose random target different from source

        target = random.randint(0, n-1)

        if target == source: continue

        # delta up to some fraction of source

        max_delta = min(a[source] * 0.5, 0.01)

        delta = max_delta * random.random()

        if delta <= 0: continue

        a2 = a.copy()

        a2[source] -= delta

        a2[target] += delta

        a2 = np.maximum(a2, 1e-12)

        a2 /= a2.sum()

        sc2 = _max_convolution_score(a2)

        if sc2 < best_sc:

            best_sc = sc2

            best_a = a2.copy()

            a = a2  # accept

            # print improvement occasionally

            if it % 100 == 0:

                print(f"[{time.time():.1f}] Hill climb iter={it} new best {sc2:.6f}")

        else:

            # maybe accept small uphill with some prob? Not needed

            pass

    return best_a, best_sc

"""
Improved search for a non‑negative step function minimising
`evaluate_sequence`.  The algorithm consists of:

1️⃣  Broad generation of candidate vectors from many analytic families
    (Uniform, Arcsine, Triangular, Exponential, Beta, Sine‑power,
    Cosine‑window, Gaussian, Power‑law, Double‑Power‑law) together
    with a large number of random Dirichlet draws (tiny concentration
    parameters) and a few sparse/randomly shuffled variants.

2️⃣  Keep a small «elite» list (best 20) of candidates (by the objective
    `2·n·max(a*a)`, i.e. the value returned by `evaluate_sequence` for a
    normalised vector).

3️⃣  Polishing of each elite vector with a fast ADAM‑style projected
    sub‑gradient optimisation of the true `max`‑convolution.  The
    update uses the exact sub‑gradient (indicator of the active
    convolution indices) and projects onto the probability simplex.
    A few thousand iterations are performed, with a geometrically
    decreasing learning rate and occasional jitter to escape plateaus.

4️⃣  A cheap hill‑climber moves a small amount of mass from the largest‑product
    pair to a random location, accepting any improvement.  This step
    can also prune tiny entries.

5️⃣  Finally a short simulated‑annealing‑like random‑mass‑transfer loop
    explores the neighbourhood of the current best sequence.

All steps respect the 1000 s budget (a 2 s safety margin).  The best
sequence found (as a plain Python list) is returned.
"""

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) for a vector already normalised to sum = 1.
    This is exactly the value that evaluate_sequence would return.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families – each returns a vector with sum = 1                --
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:                      # pad if needed
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑space to avoid under/overflow
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_powerlaw(n, p):
    """Monotone decreasing power‑law a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _distribution_double_powerlaw(n, p, delta=0.0):
    """a_i ∝ (i+δ)^{-p}·(n+1‑i+δ)^{-p}, symmetric."""
    i = np.arange(1, n + 1, dtype=np.float64)
    if delta == 0.0:
        vals = (i ** (-p)) * ((n + 1 - i) ** (-p))
    else:
        vals = ((i + delta) ** (-p)) * ((n + 1 - i + delta) ** (-p))
    vals /= vals.sum()
    return vals

def _adam_optimize(a0, deadline, max_iters=80000,
                   lr0=0.3, beta1=0.9, beta2=0.999, eps_adam=1e-8):
    """Projected ADAM on the true max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_sc = _max_convolution_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # use a tiny absolute + relative tolerance
        tol = maxc * 1e-12 + 1e-12
        active = np.where(conv >= maxc - tol)[0]

        w = np.zeros_like(conv)
        w[active] = 1.0                # all active constraints equally weighted

        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # ADAM updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule (geometric decay)
        lr = lr0 * (0.9995 ** it)

        step = lr / (np.sqrt(v_hat) + eps_adam)
        a = a - step * m_hat

        # projection onto the simplex, enforce positivity
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, 1e-12)
        a /= a.sum()

        sc = _max_convolution_score(a)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] ADAM it={it} new best {sc:.6f}")

        # occasional jitter to shake out plateaus
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 5e-6
            a = np.maximum(a + jitter, 1e-12)
            a /= a.sum()

    return best_a, best_sc


def _hill_climb(a0, deadline, max_attempts=3000):
    """Simple mass‑transfer hill‑climber that moves a tiny chunk of weight."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = _max_convolution_score(a)

    for it in range(max_attempts):
        if time.time() >= deadline:
            break

        # pick source = index with currently largest contribution to max‑conv
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # find the pair (i,j) that realises the max product a_i*a_j for this k
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= 1e-15:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, 1e-12)
        a2 /= a2.sum()

        sc2 = _max_convolution_score(a2)
        if sc2 < best_sc - 1e-12:
            best_sc = sc2
            best_a = a2.copy()
            a = a2
            if it % 100 == 0:
                print(f"[{time.time():.1f}] Hill‑climb it={it} new best {sc2:.6f}")

    return best_a, best_sc


def _random_mass_anneal(a0, deadline, max_iter=4000):
    """Very light simulated‑annealing style perturbation near the current best."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = _max_convolution_score(a)

    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # propose a tiny random transfer
        i = random.randint(0, n - 1)
        j = random.randint(0, n - 1)
        if i == j:
            continue
        delta = a[i] * 0.05 * random.random()
        if delta <= 1e-15:
            continue
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        a2 = np.maximum(a2, 1e-12)
        a2 /= a2.sum()
        sc2 = _max_convolution_score(a2)
        if sc2 < best_sc - 1e-12:
            best_sc = sc2
            best_a = a2.copy()
            a = a2
            print(f"[{time.time():.1f}] Anneal it={it} new best {sc2:.6f}")

def _generate_candidates_for_length(n):
    """Return a list of candidate vectors (sum = 1) for a given length."""
    cand = []

    # Analytic families ---------------------------------------------------
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))
    cand.append(_distribution_exponential(n, lam=0.001))
    cand.append(_distribution_exponential(n, lam=0.02))

    # Beta‑sym with a fine grid of shape parameters (including extremely small)
    beta_ps = [0.001, 0.002, 0.004, 0.006, 0.008, 0.01,
               0.02, 0.04, 0.06, 0.08, 0.1, 0.12, 0.15,
               0.2, 0.25, 0.3, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0]
    for p in beta_ps:
        cand.append(_distribution_beta_sym(n, p))

    # Double‑power‑law (symmetric) – a flexible edge‑heavy family
    for p in [0.5, 0.8, 1.0, 1.2, 1.5, 2.0]:
        cand.append(_distribution_double_powerlaw(n, p))
        cand.append(_distribution_double_powerlaw(n, p, delta=0.5))

    # Sine power family
    for pw in [0.5, 1.0, 2.0, 4.0, 8.0, 12.0]:
        cand.append(_distribution_sine_power(n, pw))

    # Cosine window
    for coeff in [0.3, 0.5, 0.7, 0.9]:
        cand.append(_distribution_cosine_window(n, coeff))

    # Gaussian
    for sigma in [0.12, 0.18, 0.25, 0.35, 0.45]:
        cand.append(_distribution_gaussian(n, sigma))

    # Power‑law (monotone decreasing) and its reverse
    for p in [1.2, 1.5, 2.0, 2.5, 3.0]:
        pl = _distribution_powerlaw(n, p)
        cand.append(pl)
        cand.append(pl[::-1])

    # Random Dirichlet draws – many of them, especially with tiny α
    alphas = [0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1]
    draws_per_alpha = 30  # enough to stay within time budget
    for alpha in alphas:
        for _ in range(draws_per_alpha):
            vec = np.random.dirichlet(np.full(n, alpha))
            cand.append(vec)

    # Sparse random start (few non‑zeros)
    k = max(2, n // 500)
    sparse = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    sparse[idx] = np.random.rand(k)
    sparse /= sparse.sum()
    cand.append(sparse)

    # Mix a few families with a small uniform component (adds asymmetry)
    mix_beta = 0.08
    for p in [0.001, 0.005, 0.01]:
        mixed = (1 - mix_beta) * _distribution_beta_sym(n, p) + mix_beta * _distribution_uniform(n)
        cand.append(mixed)

    # Random shuffles of a few base families
    for base in (cand[0], cand[1], cand[4]):
        shuf = base.copy()
        np.random.shuffle(shuf)
        cand.append(shuf)

    return cand

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:

    start = time.time()

    elite_sz = 20

    for n in lengths:

                elite.sort(key=lambda x: x[0])
