"""
Propose a non‑negative step function that (almost) minimizes
`evaluate_sequence`.  

The search consists of four phases, each respecting the time budget:

1️⃣  **Broad exploration** – generate many candidate vectors of several
    lengths (up to 65 536) from rich analytic families
    (uniform, arcsine, symmetric/asymmetric Beta, Dirichlet,
    edge‑biased Dirichlet).  The best candidate (by the exact max‑convolution
    score) is kept.

2️⃣  **Smooth‑max mirror descent with Adam‑style momentum** – minimise a
    soft‑max approximation of the max‑convolution.  The temperature is
    increased geometrically while the learning‑rate is decreased; a
    momentum vector is kept for faster convergence.

3️⃣  **Projected Adam sub‑gradient descent on the true max‑convolution** –
    uses the exact active set (indices attaining the current maximum) and
    standard Adam update with back‑tracking line search.

4️⃣  **Targeted mass‑transfer hill‑climbing** – repeatedly moves a tiny
    amount of probability mass from the pair contributing the largest
    product at the current peak to the currently smallest entry.  Small
    random jitter is added occasionally to avoid plateaus.

All vectors are kept on the simplex (non‑negative, sum = 1) and clipped
to avoid exact zeros.  The final sequence is returned as a plain Python
list of floats.
"""

import time
import random
import numpy as np
import math

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Exact score used by `evaluate_sequence` for a vector already
    normalised to sum = 1 (i.e. 2·n·max(a*a)).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families (all return a vector with sum = 1)

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    w /= w.sum()
    return w


def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    w /= w.sum()
    return w


def _distribution_edge_dirichlet(n: int, beta_exp: float) -> np.ndarray:
    """
    Dirichlet draw with concentration concentrated at the edges:
    c_i ∝ ((i+0.5)*(n-i-0.5))^{-beta_exp}  +  0.1 .
    """
    idx = np.arange(n, dtype=np.float64)
    conc = ((idx + 0.5) * (n - idx - 0.5)) ** (-beta_exp) + 0.1
    vec = np.random.dirichlet(conc)
    return vec


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a step‑function that minimises `evaluate_sequence`."""
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a small safety margin
    eps = 1e-12

    best_seq = None
    best_score = float("inf")
    best_n = None

    # --------------------------------------------------------------
    # 1️⃣  Broad exploration over many lengths and families
    lengths = [1024, 2048, 4096, 8192, 16384, 32768, 65536]

    sym_beta_ps = np.linspace(0.02, 0.6, 25)          # symmetric Beta
    asym_beta_ps = np.linspace(0.02, 0.6, 15)        # p for asymmetric
    edge_exps = np.linspace(0.2, 1.2, 7)             # edge‑biased Dirichlet

    for n in lengths:
        if time.time() >= deadline:
            break

        # ---- deterministic families ---------------------------------
        # uniform (useful as a cheap baseline)
        vec = _distribution_uniform(n)
        sc = _max_convolution_score(vec)
        if sc < best_score:
            best_score, best_seq, best_n = sc, vec.copy(), n
            print(f"[{time.time():.1f}] n={n:<5} uniform        score={sc:.6f}")

        # arcsine (Beta(0.5,0.5))
        vec = _distribution_arcsine(n)
        vec = np.maximum(vec, eps)
        vec /= vec.sum()
        sc = _max_convolution_score(vec)
        if sc < best_score:
            best_score, best_seq, best_n = sc, vec.copy(), n
            print(f"[{time.time():.1f}] n={n:<5} arcsine        score={sc:.6f}")

        # symmetric Beta(p,p)
        for p in sym_beta_ps:
            if time.time() >= deadline:
                break
            vec = _distribution_beta_sym(n, p)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n:<5} Beta_sym(p={p:.3f}) score={sc:.6f}")

        # asymmetric Beta(p,q) – left‑heavy and right‑heavy
        for p in asym_beta_ps:
            if time.time() >= deadline:
                break
            # left‑heavy: small p, q fixed moderate
            q = 0.6
            vec = _distribution_beta(n, p, q)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n:<5} Beta(p={p:.3f},q={q}) score={sc:.6f}")

            # right‑heavy: p fixed, q small
            p_fixed = 0.6
            vec = _distribution_beta(n, p_fixed, p)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n:<5} Beta(p={p_fixed},q={p:.3f}) score={sc:.6f}")

        # ---- random draws --------------------------------------------
        # plain Dirichlet with concentration 1
        for _ in range(4):
            if time.time() >= deadline:
                break
            vec = np.random.dirichlet(np.full(n, 1.0))
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n:<5} Dirichlet(1)   score={sc:.6f}")

        # Dirichlet with random concentration (range 0.1‑3)
        for _ in range(3):
            if time.time() >= deadline:
                break
            conc = np.random.uniform(0.1, 3.0, size=n)
            vec = np.random.dirichlet(conc + 0.01)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n:<5} Dirichlet(rnd) score={sc:.6f}")

        # edge‑biased Dirichlet draws
        for bex in edge_exps:
            if time.time() >= deadline:
                break
            vec = _distribution_edge_dirichlet(n, bex)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n:<5} edge‑Dir(beta={bex:.2f}) score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Incorporate a previously discovered sequence if it exists.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score, best_seq, best_n = sc, prev.copy(), len(prev)
                print(f"[{time.time():.1f}] previous best      score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Safeguard: if nothing was generated (should never happen) fall back.
    if best_seq is None:
        best_n = 4096
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform    score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Smooth‑max mirror descent with momentum (Adam‑style)
    a = best_seq.copy()
    n = a.shape[0]

    # Estimate remaining time and decide iteration budget.
    remaining = deadline - time.time()
    # Rough estimate of time per iteration (empirically ~0.003 s for n≈30 k)
    est_per_iter = 0.004
    max_iters = max(5000, int(min(50000, remaining / est_per_iter)))
    print(f"[{time.time():.1f}] mirror‑descent iters={max_iters}")

    # schedules (geometric)
    alpha_start, alpha_end = 0.5, 2e4
    lr_start, lr_end = 0.2, 0.001
    beta_mom = 0.9
    m = np.zeros_like(a)

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        t = it / max_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t

        conv = _fft_self_convolve(a)
        maxc = conv.max()
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth‑max (2 * a_rev * w)
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        m = beta_mom * m + (1 - beta_mom) * grad

        # exponentiated‑gradient update
        a = a * np.exp(-lr * m)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)   # avoid exact zeros
        a /= a.sum()

        if (it % max(1, max_iters // 10) == 0) or (it == max_iters - 1):

                best_score, best_seq = cur_sc, a.copy()
                print(f"[{time.time():.1f}] mirror it={it}/{max_iters} new best {cur_sc:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Projected Adam sub‑gradient descent on the true max‑convolution
    a = best_seq.copy()
    n = a.shape[0]
    beta1, beta2 = 0.9, 0.999
    m1 = np.zeros_like(a)
    m2 = np.zeros_like(a)
    t = 0
    sub_iter = 0
    while time.time() < deadline:
        sub_iter += 1
        t += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()

        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = conv >= cur_max - tol

        # distribute weight uniformly over the active set (any sub‑gradient works)
        w[active] = 1.0 / active.sum()
        # true sub‑gradient of max‑convolution
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam update

        m2 = beta2 * m2 + (1 - beta2) * (grad ** 2)

        cand = _project_onto_simplex(cand, 1.0)
        cand = np.maximum(cand, eps)
        cand /= cand.sum()
        cand_sc = _max_convolution_score(cand)

            print(f"[{time.time():.1f}] subgrad it={sub_iter} new best {cand_sc:.6f}")

        # Accept the candidate if it does not worsen the current max (monotone descent)
        cur_sc = _max_convolution_score(a)
        if cand_sc <= cur_sc:
            a = cand

        # occasional random jitter to escape shallow plateaus
        if sub_iter % 500 == 0:
            jitter = np.random.randn(n) * 5e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        # sanity break if almost out of time
        if time.time() + 0.3 >= deadline:
            break

    # ------------------------------------------------------------------
    # 5️⃣  Targeted mass‑transfer hill‑climber
    a = best_seq.copy()
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    hill_iter = 0
    while time.time() < deadline:
        hill_iter += 1
        # index where convolution attains its maximum
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # Find the pair (i,j) among the contributing indices with the largest product
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        # Choose source as the larger of the two entries in the maximal product pair
        source = best_i if a[best_i] >= a[best_j] else best_j
        # Choose target as the index with currently minimal mass (different from source)
        target = np.argmin(a)
        if target == source:

            target = np.argsort(a)[1]

        # Amount of mass to move: limited by source value and a small absolute cap
        max_delta = min(a[source] * 0.4, 0.03)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()

        new_M = _fft_self_convolve(a2).max()
        if new_M + 1e-12 < M:
            # accept improvement
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M
            best_seq = a.copy()
            best_score = 2.0 * n * M
            if hill_iter % 100 == 0:

        # occasional jitter to avoid stagnation

            a = np.maximum(a + jitter, eps)

            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] hill jitter at it={hill_iter}")

    final_score = 2.0 * n * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
