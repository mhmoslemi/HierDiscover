The objective to minimise is  

\[
F(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge0,
\]

where \((a*a)_k\) is the full linear convolution of the step‑function with itself.
Because the expression is homogeneous we may (and will) normalise
\(\sum_i a_i = 1\); the factor \(2n\) then stays constant for a fixed length
\(n\).

The problem is non‑convex because of the `max`.  A well‑known trick is to
replace the hard `max` by a smooth **soft‑max**:

\[
w_k = \frac{e^{\alpha\,(c_k-\max c)}}{\sum_j e^{\alpha\,(c_j-\max c)}} ,
\qquad c = a*a .
\]

For a large smoothing parameter \(\alpha\) the weights concentrate on the
active (maximal) indices, but they stay differentiable.  With these weights

\[
\frac{\partial}{\partial a_i}\sum_k w_k\,c_k
    = 2\;\sum_k w_k\,a_{k-i}
\]

i.e. the gradient is a cross‑correlation between the current distribution
and the weight vector.  This gives a *full‑active‑set* gradient that is
much less noisy than using a single active index.

The optimisation loop consists of three stages

1. **Smooth exponential‑gradient descent** – mirror‑descent on the
   probability simplex with a decaying step size \(\eta_t=\frac{\eta_0}{\sqrt t}\).
2. **Greedy pairwise mass transfers** – the classic “swap” move from the
   reference implementation, applied until no further improvement is found.
3. **Simulated‑annealing walk** – occasional non‑improving moves are accepted
   with a temperature schedule, allowing the search to escape shallow local
   minima.

All operations are performed on the probability simplex, therefore the
sequence always remains non‑negative and its sum stays equal to 1.
A modest number of analytic seeds (uniform, arcsine, beta, power‑law,
exponential, sparse, two‑block, Dirichlet) are generated for several
different lengths; the best few seeds are refined with the three stages
above.  The whole routine respects the global time budget (≈ 1000 s).

The implementation is completely self‑contained, uses only `numpy`,
`math`, `time` and the already provided `evaluate_sequence`, and returns a
plain list of floats ready for the evaluator.

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective (sum is forced to 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp(
        (alpha - 1) * np.log(np.clip(x, eps, 1.0))
        + (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0))
    )
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals

def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient = 2 * a ⋆ w, where w has length 2n‑1 and sums to 1.
    Implemented via FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    m = w.shape[0]               # = 2n‑1
    a_rev = a[::-1]
    size = 1 << ((n + m - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad

def _smooth_exponential_descent(
        a: np.ndarray,
        deadline: float,
        best_seq: np.ndarray,
        best_score: float,
        lr0: float = 0.8,
        max_iter: int = 30000,
        eval_interval: int = 200,
        alpha: float = 2500.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) using a soft‑max weight vector.
    The step size decays like η_t = lr0 / sqrt(t).
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()

        # soft‑max weights (numerically stable)
        w = np.exp(alpha * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0.0:
            # fall back to a pure active set (very unlikely)
            mask = conv >= max_val - 1e-14
            cnt = mask.sum()
            w = mask.astype(np.float64) / max(cnt, 1)
        else:
            w /= w_sum

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)          # exponentiated‑gradient step
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s                         # simplex projection

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth descent it={t} new best {best_score:.6f}")

    return best_seq, best_score

def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """
    Greedy local mass‑transfer moves (same idea as the reference code).
    """
    n = a.shape[0]
    while time.time() < deadline:
        src_cand = np.where(a > 1e-12)[0]
        if src_cand.size == 0:
            break
        i = np.random.choice(src_cand)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float,
                         start_temp: float = 1e-4,
                         end_temp: float = 1e-7,
                         max_moves: int = 200000) -> tuple[np.ndarray, float]:
    """
    Light simulated‑annealing walk.  Moves that increase the objective are
    accepted with probability exp(−Δ/T).  Temperature cools exponentially.
    """
    n = a.shape[0]
    cur_seq = a.copy()
    cur_score = _objective(cur_seq)

    # temperature schedule
    if max_moves > 0:
        cooling = (end_temp / start_temp) ** (1.0 / max_moves)
    else:
        cooling = 1.0
    temp = start_temp
    moves = 0

    while time.time() < deadline and moves < max_moves:
        moves += 1
        # donor (must have some mass) and recipient
        donor_idx = np.random.choice(np.where(cur_seq > 1e-12)[0])
        recip_idx = np.random.randint(n)
        if recip_idx == donor_idx:
            continue

        # propose moving a random fraction (up to 30 %) of donor mass
        frac = 0.30 * np.random.rand()
        delta = cur_seq[donor_idx] * frac
        if delta < 1e-14:
            continue

        new_seq = cur_seq.copy()
        new_seq[donor_idx] -= delta
        new_seq[recip_idx] += delta
        new_score = _objective(new_seq)
        dE = new_score - cur_score

        accept = False
        if dE < 0:
            accept = True
        else:
            prob = math.exp(-dE / max(temp, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_seq = new_seq
            cur_score = new_score
            if cur_score < best_score - 1e-13:
                best_score = cur_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.1f}] anneal move new best {best_score:.6f}")

        temp *= cooling

    return best_seq, best_score


# ----------------------------------------------------------------------
def _resize_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a probability vector to a new length.
    The result is renormalised to sum‑1 and forced to be non‑negative.
    """
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    # centre of each bin
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False) + 0.5 / old_n
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False) + 0.5 / new_n
    new_vals = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s == 0.0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The routine respects a global time budget (≈ 1000 s) and returns the best
    sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Choose a handful of lengths (powers of two are FFT‑friendly)
    lengths = [1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # analytic seeds
        seeds = _initial_candidates_extended(n, rng)

        # bring the (possible) previous solution to this length
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
            if prev.ndim == 1 and prev.size > 0:
                prev_resized = _resize_distribution(prev, n)
                seeds.append(prev_resized)
        except Exception:
            pass

        for cand in seeds:
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] **** new global best len={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))

        # keep only a few very promising seeds for the intensive phase
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:12]

    # ------------------------------------------------------------------
    # 2. Continuous phase – smooth exponential‑gradient descent
    for sc, seed_seq in top_candidates:
        if time.time() >= deadline:
            break
        n = seed_seq.shape[0]
        # allocate an equal share of the remaining time to each seed
        remaining = deadline - time.time()
        time_per_seed = remaining / max(1, len(top_candidates))
        sub_deadline = time.time() + time_per_seed

        seq, new_score = _smooth_exponential_descent(
            seed_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.8,
            max_iter=30000,
            eval_interval=400,
            alpha=2500.0,
        )
        if new_score < overall_best_score:
            overall_best_score = new_score
            overall_best_seq = seq.copy()
            print(f"[{time.time():.1f}] after smooth descent new best {overall_best_score:.6f}")

        # ------------------------------------------------------------------
        # 3. Greedy local swaps
        seq, new_score = _pairwise_swap_local(
            seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if new_score < overall_best_score:
            overall_best_score = new_score
            overall_best_seq = seq.copy()
            print(f"[{time.time():.1f}] after swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final polishing – simulated annealing (light)
    if overall_best_seq is not None and time.time() < deadline:
        overall_best_seq, overall_best_score = _simulated_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            start_temp=1e-4,
            end_temp=1e-7,
            max_moves=150000,
        )
        print(f"[{time.time():.1f}] after annealing best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence (clipped to the limits enforced by
    #    `evaluate_sequence` – 0 ≤ height ≤ 1000)
    if overall_best_seq is None:
        # fallback – uniform distribution of modest size
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    overall_best_seq = np.clip(overall_best_seq, 0.0, 1000.0)
    overall_best_seq = np.maximum(overall_best_seq, 0.0)

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
