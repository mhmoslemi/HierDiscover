"""
Improved 1000 s search for a low‑score step function.

The algorithm proceeds in four phases:

1. **Broad exploration** – generate many analytic families (arcsine, beta,
   edge‑power, two‑tail, exponential, power‑law, sinusoidal‑edge, block)
   for a diverse set of lengths and parameters.  Their surrogate score
   (2·n·max(conv)) is computed immediately.

2. **Seed selection** – keep the best *K* candidates (default 60) for an
   intensive refinement stage.

3. **Refinement** – each seed is run through two first‑order methods
   (Adam and exponentiated‑gradient) on the simplex, using a generous
   iteration budget that adapts to the remaining time.  The best vector
   seen any‑where is tracked.

4. **Polishing** – a Frank‑Wolfe “mass transfer’’ step followed by a
   simulated‑annealing jitter finishes the search, always respecting the
   deadline.  The returned list is a plain Python list of floats; the
   external ``evaluate_sequence`` will apply the final scaling/validation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _surrogate_score(a: np.ndarray) -> float:
    """
    Exact surrogate when Σa = 1:
        2·n·max(conv(a,a)).
    This is exactly the value that ``evaluate_sequence`` would return after
    normalising the input to sum = 1.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) under Σa = 1.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # grad = 2·conv_rev where rev is a reversed version of a and w is the weight
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (pairs i=j)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.30,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 2_000_000):
    """Adam on the simplex (projected) – returns best point seen."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _surrogate_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur = _surrogate_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          eta0: float = 0.9,
                          max_iters: int = 1_500_000):
    """Exponentiated‑gradient (KL‑mirror) descent on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _surrogate_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s

        cur = _surrogate_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score

def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with concentration ``alpha``."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _edge_power_distribution(n: int, s: float) -> np.ndarray:
    """w_i ∝ (i·(n‑i+1))^{-s},  i = 1,…,n.  Generalises the arcsine case (s=0.5)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = (i * (n - i + 1)) ** (-s)
    vals /= vals.sum()
    return vals


def _two_tail_distribution(n: int, a: float) -> np.ndarray:
    """Mixture of two symmetric power‑law tails."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-a) + (n - i + 1) ** (-a)
    vals /= vals.sum()
    return vals


def _double_bump_distribution(n: int,
                              block_frac: float = 0.25,
                              gap_frac: float = 0.1) -> np.ndarray:
    """Two equal uniform blocks separated by a gap."""
    total_block = max(2, int(block_frac * n))
    m = total_block // 2
    g = int(gap_frac * n)
    a = np.zeros(n, dtype=np.float64)
    a[:m] = 1.0 / (2 * m)
    start2 = m + g
    if start2 + m > n:
        start2 = n - m
    a[start2:start2 + m] = 1.0 / (2 * m)
    return a


def _sinusoidal_edge_distribution(n: int) -> np.ndarray:
    """
    Edge‑heavy shape built from the inverse sine of a half‑period.
    w_i ∝ 1 / sqrt( sin( π·(i+0.5)/n ) ).
    The sine term is tiny near the boundaries, giving a strong U‑shape.
    """
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    # add a tiny epsilon to avoid division by zero exactly at the edges
    eps = 1e-12
    vals = 1.0 / np.sqrt(np.sin(theta) + eps)
    vals /= vals.sum()
    return vals


def _logistic_distribution(n: int, steep: float = 6.0) -> np.ndarray:
    """Symmetric logistic (sigmoidal) shape – steep parameter controls edge sharpness."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    vals = 1.0 / (1.0 + np.exp(steep * (x - 0.5)))
    vals = vals + 1.0 / (1.0 + np.exp(-steep * (x - 0.5)))  # mirror, gives symmetry
    vals /= vals.sum()
    return vals


def _random_block_distribution(n: int, blocks: int = 2,
                              min_frac: float = 0.01,
                              max_frac: float = 0.25) -> np.ndarray:
    """Place a few uniform blocks at random positions."""
    a = np.zeros(n, dtype=np.float64)
    mass_per_block = 1.0 / blocks
    for _ in range(blocks):
        blk_len = max(1, int(np.random.uniform(min_frac, max_frac) * n))
        start = np.random.randint(0, n - blk_len + 1)
        a[start:start + blk_len] = mass_per_block / blk_len
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the 1000 s deadline and returns the best list found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Determine a sensible base length (use previous best if available)
    try:
        base_n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        base_n = 4000
    base_n = max(2000, min(base_n, 25000))

    # ------------------------------------------------------------------
    # Choose a diverse set of lengths (including a few random large ones)
    lengths = set()
    for fac in (0.6, 0.8, 0.9, 1.0, 1.2, 1.4, 1.6, 2.0):
        n = int(base_n * fac)
        if 1500 <= n <= 30000:
            lengths.add(n)
    # a handful of completely random lengths
    for _ in range(8):
        lengths.add(np.random.randint(1500, 30001))
    lengths = sorted(lengths)

    print(f"[{time.time():.1f}] Exploring {len(lengths)} lengths (base≈{base_n})")

    # ------------------------------------------------------------------
    # Parameter grids – extended compared with the reference implementation
    beta_ps = np.arange(0.02, 1.02, 0.02)                 # finer grid symmetric beta
    asym_grid = [(0.2, 0.8), (0.3, 0.7), (0.5, 0.5),
                 (0.7, 0.3), (0.8, 0.2), (0.4, 0.6), (0.6, 0.4)]
    power_ps = np.arange(0.3, 2.01, 0.2)                  # power‑law exponents
    lam_vals = [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02]
    edge_ps = np.arange(0.10, 0.91, 0.05)                 # generalized arcsine exponents
    tail_as = np.arange(0.1, 2.01, 0.2)                    # two‑tail mixture
    logistic_steeps = [4.0, 6.0, 8.0]                     # logistic sharpness

    candidates = []   # list of (array, surrogate_score)

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- deterministic analytic families ---------------------------------
        families = [
            (_arcsine_distribution, "arcsine"),
            (_triangular_distribution, "triangular"),
            (lambda m: _exponential_distribution(m, lam=0.005), "exp0.005"),
        ]
        for gen, name in families:
            a = gen(n)
            candidates.append((a, _surrogate_score(a)))

        # ----- exponential tails ------------------------------------------------
        for lam in lam_vals:
            a = _exponential_distribution(n, lam=lam)
            candidates.append((a, _surrogate_score(a)))

        # ----- symmetric beta ---------------------------------------------------
        for p in beta_ps:
            a = _beta_symmetric_distribution(n, p)
            candidates.append((a, _surrogate_score(a)))

        # ----- asymmetric beta --------------------------------------------------
        for a_alpha, a_beta in asym_grid:
            a = _beta_asymmetric_distribution(n, a_alpha, a_beta)
            candidates.append((a, _surrogate_score(a)))

        # ----- power‑law --------------------------------------------------------
        for p in power_ps:
            a = _power_law_distribution(n, p)
            candidates.append((a, _surrogate_score(a)))

        # ----- edge‑power family ------------------------------------------------
        for s in edge_ps:
            a = _edge_power_distribution(n, s)
            candidates.append((a, _surrogate_score(a)))

        # ----- two‑tail mixture -------------------------------------------------
        for a_param in tail_as:
            a = _two_tail_distribution(n, a_param)
            candidates.append((a, _surrogate_score(a)))

        # ----- double‑bump (large blocks) --------------------------------------
        for block_frac in (0.20, 0.30, 0.40):
            for gap_frac in (0.02, 0.05, 0.10):
                a = _double_bump_distribution(n, block_frac, gap_frac)
                candidates.append((a, _surrogate_score(a)))

        # ----- sinusoidal‑edge --------------------------------------------------
        a = _sinusoidal_edge_distribution(n)
        candidates.append((a, _surrogate_score(a)))

        # ----- logistic families -------------------------------------------------
        for steep in logistic_steeps:
            a = _logistic_distribution(n, steep=steep)
            candidates.append((a, _surrogate_score(a)))

        # ----- random block constructions ---------------------------------------
        for blocks in (2, 3):
            for _ in range(3):
                a = _random_block_distribution(n, blocks=blocks)
                candidates.append((a, _surrogate_score(a)))

        # ----- random Dirichlet draws -------------------------------------------
        for alpha in [0.01, 0.05, 0.1, 0.3, 0.7, 1.0]:
            a = _random_dirichlet_alpha(n, alpha)
            candidates.append((a, _surrogate_score(a)))

    # ------------------------------------------------------------------
    # Seed from a previous best (if it exists)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            if prev.sum() == 0:
                prev = np.full_like(prev, 1.0 / prev.size)
            else:
                prev = prev / prev.sum()
            candidates.append((prev, _surrogate_score(prev)))
            for _ in range(8):
                jitter = np.random.randn(prev.size) * 2e-5
                noisy = _project_onto_simplex(prev + jitter, s=1.0)
                candidates.append((noisy, _surrogate_score(noisy)))
            print(f"[{time.time():.1f}] added previous best (+perturbations)")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Keep the best *K* seeds for intensive refinement
    candidates.sort(key=lambda x: x[1])
    top_k = min(60, len(candidates))
    top_seeds = candidates[:top_k]
    print(f"Generated {len(candidates)} candidates, keeping top {top_k} for refinement.")

    best_seq = top_seeds[0][0].copy()
    best_score = top_seeds[0][1]
    print(f"Initial best surrogate score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Intensive refinement (Adam + Exp‑Grad) – time‑aware
    for idx, (seed_seq, seed_score) in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        n = seed_seq.shape[0]

        # Allocate a fraction of the remaining budget to this seed
        # (roughly equal for all seeds but leaving a buffer for polishing)
        per_seed_deadline = time.time() + (deadline - time.time()) * 0.03

        seq_adam, sc_adam = _adam_descent(seed_seq, per_seed_deadline,
                                          max_iters=800_000)
        if sc_adam < best_score - 1e-14:
            best_score, best_seq = sc_adam, seq_adam.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} Adam improved to {best_score:.6f}")

        seq_eg, sc_eg = _exp_gradient_descent(seed_seq, per_seed_deadline,
                                              max_iters=500_000)
        if sc_eg < best_score - 1e-14:
            best_score, best_seq = sc_eg, seq_eg.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} ExpGrad improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Frank‑Wolfe style mass‑transfer polishing (more iterations)
    a = best_seq.copy()
    cur_score = best_score
    fw_iter = 0
    fractions = (1.0, 0.75, 0.5, 0.25, 0.125)
    while time.time() < deadline and fw_iter < 400_000:
        fw_iter += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-15:
            break
        improved = False
        for f in fractions:
            delta = max_transfer * f
            a_new = a.copy()
            a_new[i_src] -= delta
            a_new[i_dst] += delta
            sc = _surrogate_score(a_new)
            if sc < cur_score - 1e-13:
                a = a_new
                cur_score = sc
                improved = True
                break
        if improved and cur_score < best_score - 1e-14:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time():.1f}] FW iter={fw_iter} new best {best_score:.6f}")

        # occasional jitter to break tiny plateaus
        if fw_iter % 10000 == 0 and np.random.rand() < 0.15:
            jitter = np.random.randn(a.size) * 3e-5
            a = _project_onto_simplex(a + jitter, s=1.0)
            cur_score = _surrogate_score(a)

    # ------------------------------------------------------------------
    # Simulated‑annealing / jitter phase (final push)
    a = best_seq.copy()
    cur_score = best_score
    temp_start = 0.035
    temp_end = 0.0012
    anneal_iter = 0
    while time.time() < deadline and anneal_iter < 350_000:
        anneal_iter += 1
        # linear temperature schedule based on remaining time
        t_rem = deadline - time.time()
        fraction = max(0.0, t_rem / (budget_s - 0.5))
        temp = temp_end + (temp_start - temp_end) * fraction

        i = np.random.randint(a.size)
        j = np.random.randint(a.size - 1)
        if j >= i:
            j += 1
        max_delta = min(a[i], 1.0 - a[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * temp
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _surrogate_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if cur_score < best_score - 1e-14:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time():.1f}] Anneal iter={anneal_iter} new best {best_score:.6f}")
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_score - cur_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                a = a_new
                cur_score = new_score

    # ------------------------------------------------------------------
    print(f"Final best surrogate score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
