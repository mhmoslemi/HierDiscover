"""
Improved search for a non‑negative step function (vector a) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of three phases:

1.  **Seed generation** – several analytic families (arcsine, symmetric
    Beta, power‑transformed arcsine) are instantiated for a few moderate
    lengths (≈ 2000‑6000).  Each seed is ℓ₁‑normalised (sum = 1) and
    several deterministic reorderings are added (alternating extremes,
    centre‑outwards, binary‑tree layout).

2.  **Pairwise Frank‑Wolfe with exact line‑search** – starting from each
    seed we repeatedly move a small amount of mass from the index with
    largest sub‑gradient (worst for the objective) to the one with
    smallest sub‑gradient.  The amount transferred is chosen by a cheap
    ternary search on the convex one‑dimensional function

        f(α) = 2·n·max( b + α·L + α²·Q ),

    where `b` is the current self‑convolution, `L` and `Q` are the linear
    and quadratic contributions of the transfer.  Updating `b` can be
    done in O(n) without recomputing an FFT, therefore thousands of
    iterations are cheap.

3.  **Polishing** – the best vector found is finetuned with a few
    exponentiated‑gradient steps (multiplicative updates on the simplex)
    and a final random hill‑climber that swaps tiny mass chunks between
    two random coordinates.

All phases respect the global time budget (≈ 1000 s).  The returned
sequence is ℓ₁‑normalised, non‑negative and ready for `evaluate_sequence`.
"""

import time

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate (2·n·max(conv(a,a))) for a vector with ℓ₁‑norm = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of the max‑convolution functional on the simplex.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives raw gradient
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_transform(base: np.ndarray, p: float) -> np.ndarray:
    a = np.power(base, p)
    a /= a.sum()
    return a

# Re‑ordering heuristics -----------------------------------------------
def order_alternate_extremes(desc_vals: np.ndarray) -> np.ndarray:
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    left, right = 0, n - 1
    toggle = True
    for v in desc_vals:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    return out


def order_center_outwards(desc_vals: np.ndarray) -> np.ndarray:
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    mid = (n - 1) // 2
    pos = [mid]
    for d in range(1, n):
        left = mid - d
        if left >= 0:
            pos.append(left)
        right = mid + d
        if right < n:
            pos.append(right)
    for i, p in enumerate(pos[:n]):
        out[p] = desc_vals[i]
    return out


def order_binary_tree(desc_vals: np.ndarray) -> np.ndarray:
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    positions = []

    def recurse(l: int, r: int):
        if l > r:
            return
        m = (l + r) // 2
        positions.append(m)
        recurse(l, m - 1)
        recurse(m + 1, r)

    recurse(0, n - 1)
    out[positions] = desc_vals
    return out

def _pairwise_line_search_opt(a0: np.ndarray,
                              deadline: float,
                              max_iters: int = 100_000,
                              step_check: int = 200) -> tuple[np.ndarray, float]:

    Frank‑Wolfe style pairwise mass transfer with exact convex line‑search.
    Returns the best vector seen and its surrogate score.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _fft_self_convolve(a)              # current self‑convolution
    best_score = 2.0 * n * b.max()

        it += 1

        # compute sub‑gradient on the current point

        src = int(np.argmax(grad))          # mass to remove
        dst = int(np.argmin(grad))          # mass to add
        if src == dst or a[src] <= 1e-14:
            break

        # maximum admissible transfer (keep non‑negativity and ℓ₁‑norm)
        delta = min(a[src], 1.0 - a[dst])
        if delta <= 1e-14:
            break

        # Build linear (L) and quadratic (Q) contributions for this pair
        L = np.zeros_like(b)
        L[src:src + n] -= 2.0 * a
        L[dst:dst + n] += 2.0 * a

        Q = np.zeros_like(b)
        Q[2 * src] += 1.0
        Q[2 * dst] += 1.0
        Q[src + dst] -= 2.0

        # Convex ternary search for optimal α ∈ [0, delta]
        lo, hi = 0.0, delta
        for _ in range(30):                 # enough for double precision
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            val1 = np.max(b + L * m1 + Q * (m1 * m1))
            val2 = np.max(b + L * m2 + Q * (m2 * m2))
            if val1 < val2:
                hi = m2
            else:
                lo = m1
        alpha = (lo + hi) * 0.5

        if alpha < 1e-12:
            break

        # Apply the transfer
        a[src] -= alpha
        a[dst] += alpha

        # Update convolution cheaply
        b += L * alpha + Q * (alpha * alpha)

        # Keep track of the best point

            best_a = a.copy()

        # Occasionally recompute `b` from scratch to avoid drift
        if it % step_check == 0:
            b = _fft_self_convolve(a)

    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray,

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:

            best_a = a.copy()
    return best_a, best_score


def _hill_climb(a0: np.ndarray,
                deadline: float,

    a = a0.copy()

        cand[j] += delta
