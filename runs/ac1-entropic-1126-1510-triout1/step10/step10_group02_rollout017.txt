Propose a non‑negative step function (list of floats) that minimises
`evaluate_sequence`.  The algorithm consists of:

1. **Wide exploration** – generate many analytic families (uniform,
   arcsine, triangular, exponential, asymmetric beta, power‑law …) for a
   handful of lengths.  Keep the best few (elite) candidates.

2. **Linear‑programming improvement** – for each elite vector solve the LP

        maximise   ∑ b_j
        subject to (a * b)_k ≤ M  for all k,
                    b_j ≥ 0

   where `M = max(a*a)`.  The optimiser returns a vector `b` with a larger
   ℓ₁‑norm, and after normalising we obtain a new direction that strictly
   reduces the convolution maximum.  This step is optional (skipped if
   `cvxpy` is unavailable).

3. **Smooth‑max mirror descent** – a long exponentiated‑gradient stage with
   a temperature schedule that first explores the simplex and then

4. **Projected sub‑gradient polishing** – true sub‑gradient updates with a
   back‑tracking line‑search (Adam‑style momentum).

5. **Hill‑climber & pruning** – local mass transfers and support pruning
   to squeeze out the last bits of improvement.

All stages respect a global 1000 s budget (2 s safety margin).  The

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x≥0,∑x=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Score = 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families – each returns a vector with sum = 1 -----------------
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_asym(n, a, b):
    """Beta(a,b) on the discrete grid (asymmetric)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide='ignore', invalid='ignore'):
        logw = (a - 1) * np.log(x) + (b - 1) * np.log(1 - x)
        w = np.exp(logw)
    w = np.where(np.isfinite(w), w, 0.0)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_powerlaw(n, p):
    """Monotone decreasing power‑law a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Candidate generator ----------------------------------------------------
def _generate_candidates_for_length(n):
    """Return a list of candidate vectors (sum = 1) for a specific length."""
    cand = []

    # analytic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))
    cand.append(_distribution_exponential(n, lam=0.003))
    cand.append(_distribution_exponential(n, lam=0.02))

    # asymmetric beta – a small, b large gives a decreasing shape
    for a, b in ((0.5, 2.0), (0.2, 1.5), (0.1, 1.2), (0.05, 1.0)):
        cand.append(_distribution_beta_asym(n, a, b))

    # power‑law (decreasing) and its reverse
    for p in (1.2, 1.5, 2.0, 2.5, 3.0):
        pl = _distribution_powerlaw(n, p)
        cand.append(pl)          # decreasing
        cand.append(pl[::-1])    # increasing (sometimes useful)

    # cosine window (different coeff)
    for coeff in (0.3, 0.5, 0.7, 0.9):
        cand.append(_distribution_cosine_window(n, coeff))

    # sine‑power
    for pw in (0.5, 1.0, 2.0, 4.0, 8.0):
        cand.append(_distribution_sine_power(n, pw))

    # Gaussian
    for sigma in (0.12, 0.18, 0.25, 0.35, 0.45):
        cand.append(_distribution_gaussian(n, sigma))

    # Random Dirichlet draws (various concentration)
    for alpha in (0.2, 0.5, 1.0, 2.0):
        cand.append(np.random.dirichlet(np.full(n, alpha)))

    # Sparse start – a few non‑zeros
    k = max(2, n // 300)
    sparse = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    sparse[idx] = np.random.rand(k)
    sparse /= sparse.sum()
    cand.append(sparse)

    # Random shuffles of a few base families (add asymmetry)
    for base in (cand[0], cand[1], cand[4]):
        shuf = base.copy()
        np.random.shuffle(shuf)
        cand.append(shuf)

    return cand

# Linear‑programming improvement (optional) -------------------------------
def _lp_improve(a, deadline):
    """Perform one LP improvement step if cvxpy is available."""
    if not _CVX_AVAILABLE:
        return a
    n = a.shape[0]
    # current worst convolution value
    M = _fft_self_convolve(a).max()

    # Build Toeplitz matrix T so that T @ b = a * b (full linear conv)
    # each diagonal offset i (0 ≤ i < n) is constant a[i]
    offsets = np.arange(n, dtype=int)            # 0 … n‑1
    data = [np.full(n, a[i]) for i in range(n)]  # each diagonal length = n
    T = diags(data, offsets, shape=(2 * n - 1, n), format='csr')

    # Linear program: maximise sum(b) s.t. T b ≤ M, b ≥ 0
    b = cp.Variable(n, nonneg=True)
    constraints = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                   eps=1e-6, feastol=1e-6)   # fast LP solver
    except Exception:  # pragma: no cover
        return a

    if b.value is None:
        return a

    b_val = np.maximum(b.value, 0.0)
    s = b_val.sum()
    if s <= 0:
        return a

    # Normalise and move a a fraction towards the LP solution
    g = b_val / s
    t = 0.5   # mixing parameter (aggressive yet stable)
    a_new = (1.0 - t) * a + t * g
    a_new = np.maximum(a_new, 1e-12)
    a_new /= a_new.sum()
    # ensure we stay inside the time budget
    if time.time() < deadline:
        return a_new
    else:
        return a

# Mirror‑descent (smooth max) -------------------------------------------
def _smooth_max_mirror_descent(a, deadline):
    """Exponentiated‑gradient descent on the soft‑max of the convolution."""
    n = a.shape[0]
    eps = 1e-12

    # Temperature schedule – more aggressive than the baseline
    schedule = [
        (1.0, 200.0, 3000, 0.25, 0.02),
        (200.0, 2000.0, 3000, 0.02, 0.001),
        (2000.0, 20000.0, 4000, 0.001, 2e-04),
        (20000.0, 80000.0, 5000, 2e-04, 5e-05),
    ]

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    for (alpha0, alpha1, steps, lr0, lr1) in schedule:
        if time.time() >= deadline:
            break
        for it in range(steps):
            if time.time() >= deadline:
                break
            t = it / steps
            alpha = alpha0 * (alpha1 / alpha0) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of smooth‑max
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if it % 800 == 0:
                sc = _max_convolution_score(a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] Mirror it={it} new best {sc:.6f}")

    return best_a, best_sc

# Projected sub‑gradient --------------------------------------------------
def _projected_subgradient(a, deadline):
    """True sub‑gradient descent with Adam‑style momentum & back‑tracking."""
    n = a.shape[0]
    eps = 1e-12
    velocity = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    step0 = 0.5
    it = 0

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam‑style update
        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** it)

        step = step0
        while True:
            a_candidate = a - step * v_hat
            a_candidate = _project_onto_simplex(a_candidate, 1.0)
            a_candidate = np.maximum(a_candidate, eps)
            a_candidate /= a_candidate.sum()
            cand_sc = _max_convolution_score(a_candidate)
            if cand_sc < best_sc - 1e-12 or step < 1e-8:
                break
            step *= 0.5

        a = a_candidate
        if cand_sc < best_sc - 1e-12:
            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] Subgrad it={it} new best {cand_sc:.6f}")

        if it % 1500 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    return best_a, best_sc

# Hill‑climber & support pruning -----------------------------------------
def _hill_climb_prune(a, deadline):
    """Final polishing: tiny mass transfers + support pruning."""
    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_sc = 2.0 * n * M
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- mass‑transfer hill climbing ---------------------------------
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target != source:
            max_delta = min(a[source] * 0.5, 0.02)
            delta = max_delta * random.random()
            if delta > eps:
                a2 = a.copy()
                a2[source] -= delta
                a2[target] += delta
                a2 = np.maximum(a2, eps)
                a2 /= a2.sum()
                new_M = _fft_self_convolve(a2).max()
                if new_M < M - 1e-12:
                    a = a2
                    conv = _fft_self_convolve(a)
                    M = new_M
                    best_a = a.copy()
                    best_sc = 2.0 * n * M
                    if it % 200 == 0:
                        print(f"[{time.time():.1f}] Hill it={it} new best {best_sc:.6f}")

        # ---- occasional pruning of tiny entries -------------------------
        if it % 1500 == 0:
            for keep_frac in (0.99, 0.97, 0.95, 0.90):
                k_keep = max(2, int(keep_frac * n))
                idx = np.argpartition(a, -k_keep)[-k_keep:]
                a_pruned = np.zeros_like(a)
                a_pruned[idx] = a[idx]
                a_pruned /= a_pruned.sum()
                sc_pruned = _max_convolution_score(a_pruned)
                if sc_pruned < best_sc - 1e-12:
                    a = a_pruned
                    conv = _fft_self_convolve(a)
                    M = conv.max()
                    n = a.shape[0]
                    best_a = a.copy()
                    best_sc = sc_pruned
                    print(f"[{time.time():.1f}] Prune it={it} keep={keep_frac:.2f} new best {best_sc:.6f}")

        # ---- random jitter to keep search alive -------------------------
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

    return best_a, best_sc

# Main entry point -------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`
    and return the best vector (as a plain list).
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0   # small safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1. Broad exploration – keep the top 8 candidates overall
    lengths = [1500, 2500, 4000, 6000, 9000, 12000]   # a spread of sizes
    elite = []   # list of (score, vector)

    for n in lengths:
        if time.time() >= deadline:
            break
        candidates = _generate_candidates_for_length(n)
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] Length {n} new global best {sc:.6f}")

            # maintain elite list (max size 8)
            if len(elite) < 8:
                elite.append((sc, vec.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, vec.copy())
                    elite.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 2. Include the previously known best (if available)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = prev.size
                print(f"[{time.time():.1f}] Using previous best {sc:.6f}")
            if len(elite) < 8:
                elite.append((sc, prev.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, prev.copy())
                    elite.sort(key=lambda x: x[0])
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should virtually never be taken)
    if best_seq is None:
        best_n = 4000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] Fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Refine elite candidates
    # we allow at most 4 restarts – each gets its own optimisation path
    restarts = min(4, len(elite))
    for idx in range(restarts):
        if time.time() >= deadline:
            break
        a0 = elite[idx][1].copy()
        n0 = a0.shape[0]
        print(f"[{time.time():.1f}] Restart {idx+1}/{restarts} (len={n0}) score {elite[idx][0]:.6f}")

        # ---- optional LP step (fast improvement) -----------------------
        a_mid = _lp_improve(a0, deadline)

        # ---- smooth‑max mirror descent ---------------------------------
        a_mid, sc_mid = _smooth_max_mirror_descent(a_mid, deadline)

        # ---- projected sub‑gradient polishing ---------------------------
        a_mid, sc_mid = _projected_subgradient(a_mid, deadline)

        # ---- final hill‑climber & pruning -------------------------------
        a_mid, sc_mid = _hill_climb_prune(a_mid, deadline)

        if sc_mid < best_score - 1e-12:
            best_score = sc_mid
            best_seq = a_mid.copy()
            best_n = a_mid.shape[0]
            print(f"[{time.time():.1f}] New global best after restart {sc_mid:.6f}")

    # ------------------------------------------------------------------
    # 5. Final sanity checks and return
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()

    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
