"""Search for step function minimising evaluate_sequence.

The algorithm:
- Scan several analytic families over a range of lengths.
- Keep the best candidate (sum‑normalised to 1).
- Enforce symmetry about the centre, because the objective is symmetric.
- Refine the best candidate with a multi‑phase gradient scheme:
  * exponential‑gradient on a smooth‑max approximation,
  * projected sub‑gradient descent,
  * Adam on the soft‑max logits,
  * a simple hill‑climber that moves mass from the “worst’’ index to the
    “best’’ index.
- While time permits, launch many random‑restart refinements on
  different lengths.
All steps respect the 1000 s wall‑clock budget.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
EPS = 1e-12

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _fft_self_convolve(a):
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_convolve(a, b):
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())

def _smooth_weights(conv, alpha):
    """Return a probability vector approximating the smooth max."""
    maxc = conv.max()
    w = np.exp(alpha * (conv - maxc))
    s = w.sum()
    if s == 0:
        return np.full_like(w, 1.0 / w.size)
    return w / s

def _subgradient_weights(conv, eps_rel=1e-12):
    """Weights for a (sub)gradient of the max‑convolution."""
    m = conv.max()
    w = np.zeros_like(conv)
    w[conv >= m - eps_rel * (abs(m) + 1.0)] = 1.0
    cnt = w.sum()
    if cnt == 0:
        return w
    return w / cnt

def _exponential_gradient_step(a, lr, alpha):
    """One EG step (a ≥ 0, Σa = 1)."""
    conv = _fft_self_convolve(a)
    w = _smooth_weights(conv, alpha)
    grad = 2.0 * _fft_convolve(a[::-1], w)[len(a) - 1 : len(a) - 1 + len(a)]
    a = a * np.exp(-lr * grad)
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, EPS)
    a /= a.sum()
    return a

def _projected_gradient_step(a, eta, active_w):
    """PGD step on the sub‑gradient defined by active_w."""
    grad = 2.0 * _fft_convolve(a[::-1], active_w)[len(a) - 1 : len(a) - 1 + len(a)]
    a = a - eta * grad
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, EPS)
    a /= a.sum()
    return a

def _adam_step(logits, m, v, t, step_size):
    """Adam update on logits (unconstrained)."""
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = beta1 * m + (1.0 - beta1) * step_size
    v = beta2 * v + (1.0 - beta2) * (step_size ** 2)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    lr = 0.02 / (np.sqrt(v_hat) + eps)
    logits = logits - lr * m_hat
    logits = np.clip(logits, -30, 30)   # avoid overflow/underflow
    return logits, m, v

def _enforce_symmetry_and_normalise(a):
    """Make a symmetric (a[i]=a[-i-1]) and re‑normalise to sum = 1."""
    a = (a + a[::-1]) * 0.5
    a = np.maximum(a, EPS)
    a /= a.sum()
    return a

def _hillclimber_move(a, conv, eps=EPS):
    """Move a tiny chunk from a large‑gradient index to a small‑gradient one."""
    n = a.shape[0]
    w = _subgradient_weights(conv, eps_rel=eps)
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    src = int(np.argmax(grad))
    dst = int(np.argmin(grad))
    if src == dst:
        dst = random.randint(0, n - 1)
    max_delta = min(a[src] * 0.5, 0.02)
    delta = max_delta * random.random()
    if delta < eps:
        return a, conv, False
    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta
    a_new = _enforce_symmetry_and_normalise(a_new)
    new_conv = _fft_self_convolve(a_new)
    # improvement if the maximum shrinks (with a tiny tolerance)
    improved = new_conv.max() < conv.max() - 1e-14
    if improved:
        return a_new, new_conv, True
    else:
        return a, conv, False

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)

def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals

def _distribution_exponential(n, lam=0.01):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals

def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _distribution_gaussian(n, sigma):
    """Discrete Gaussian centred at (n‑1)/2."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _refine_vector(a, n, deadline):
    """Run EG → PGD → Adam → hill‑climber on a vector of length n."""
    best_local = a.copy()
    best_score_local = _max_convolution_score(best_local)

    # Phase 1: exponential‑gradient on smooth max
    steps_eg = 1800
    a_eg = a.copy()

    lr0, lr1 = 0.15, 0.001       # learning‑rate range
    for it in range(steps_eg):
        if time.time() >= deadline:
            break

        alpha = a0 * (a1 / a0) ** t
        lr = lr0 * (lr1 / lr0) ** t
        a_eg = _exponential_gradient_step(a_eg, lr, alpha)
        a_eg = _enforce_symmetry_and_normalise(a_eg)
        if it % 300 == 0 or it == steps_eg - 1:
            sc = _max_convolution_score(a_eg)
            if sc < best_score_local - 1e-13:
                best_local = a_eg.copy()
                best_score_local = sc
                # printing is done by caller
    # Phase 2: projected sub‑gradient descent on the true max
    steps_pg = 1200
    a_pg = a_eg.copy()
    eta0, eta1 = 0.28, 0.0032
    for it in range(steps_pg):
        if time.time() >= deadline:
            break
        conv = _fft_self_convolve(a_pg)
        w = _subgradient_weights(conv, eps_rel=1e-12)
        eta = eta0 * (eta1 / eta0) ** (it / steps_pg)
        a_pg = _projected_gradient_step(a_pg, eta, w)
        a_pg = _enforce_symmetry_and_normalise(a_pg)
        if it % 200 == 0 or it == steps_pg - 1:
            sc = _max_convolution_score(a_pg)
            if sc < best_score_local - 1e-13:
                best_local = a_pg.copy()
                best_score_local = sc
    # Phase 3: Adam on logits
    logits = np.log(a_pg + EPS)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    steps_adam = 2500
    for it in range(1, steps_adam + 1):
        if time.time() >= deadline:
            break
        a_adam = np.exp(logits)
        a_adam /= a_adam.sum()
        conv = _fft_self_convolve(a_adam)
        w = _subgradient_weights(conv, eps_rel=1e-12)
        grad_a = 2.0 * _fft_convolve(a_adam[::-1], w)[n - 1 : n - 1 + n]
        grad_l = (grad_a - np.dot(a_adam, grad_a)) * a_adam
        logits, m, v = _adam_step(logits, m, v, it, grad_l)
        if it % 400 == 0 or it == steps_adam:
            a_adam = np.exp(logits)
            a_adam /= a_adam.sum()
            a_adam = _enforce_symmetry_and_normalise(a_adam)
            sc = _max_convolution_score(a_adam)
            if sc < best_score_local - 1e-13:
                best_local = a_adam.copy()
                best_score_local = sc
    # Phase 4: simple hill‑climber
    a_hc = best_local.copy()
    conv_hc = _fft_self_convolve(a_hc)
    hc_iters = 40000
    for it in range(hc_iters):
        if time.time() >= deadline:
            break
        a_hc, conv_hc, improved = _hillclimber_move(a_hc, conv_hc, eps=EPS)
        if improved:
            sc = 2.0 * n * conv_hc.max()
            if sc < best_score_local - 1e-13:
                best_local = a_hc.copy()
                best_score_local = sc
        # occasional jitter to escape flat plateaux
        if it % 5000 == 0 and it > 0:
            jitter = np.random.randn(n) * 1e-5

"""
Search for a step function that minimizes ``evaluate_sequence``.

The algorithm works as follows:

1. **Analytic scan** – evaluate several classical families (uniform,
   arcsine, triangular, exponential, beta, sine‑power, Gaussian) on a
   range of lengths up to 10 000.  All candidates are forced to be
   symmetric (a[i]=a[n‑1‑i]) because the objective is symmetric.
2. **Interpolation** – linearly interpolate the current best vector to a few
   larger lengths (up to 20 000); this often reduces the constant factor.
3. **Refinement pipeline** – for a given vector perform four phases:
   * exponential‑gradient on a smooth‑max approximation,
   * projected sub‑gradient descent on the true max‑convolution,
   * Adam optimisation on the soft‑max logits,
   * a cheap hill‑climber that moves a tiny chunk of mass from the
     index with the largest gradient contribution to the one with the
     smallest contribution.  After every update symmetry is re‑imposed.
4. **Random restarts** – while time permits draw fresh Dirichlet vectors,
   apply a shortened version of the pipeline and keep the globally best
   sequence.

All steps respect the 1000 s wall‑clock budget and the function returns
the best sequence found as a plain list of floats.

EPS = 1e-12


def _project_onto_simplex(v, s=1.0):
    """Project ``v`` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a):
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


def _smooth_weights(conv, alpha):
    """Probability vector approximating the smooth max."""
    maxc = conv.max()
    w = np.exp(alpha * (conv - maxc))
    s = w.sum()
    if s == 0:
        return np.full_like(w, 1.0 / w.size)
    return w / s


def _subgradient_weights(conv, eps_rel=1e-12):
    """Weights for a (sub)gradient of the max‑convolution."""
    m = conv.max()
    w = np.zeros_like(conv)
    w[conv >= m - eps_rel * (abs(m) + 1.0)] = 1.0
    cnt = w.sum()
    if cnt == 0:
        return w
    return w / cnt


def _exponential_gradient_step(a, lr, alpha):
    """One exponential‑gradient step (a ≥ 0, Σa = 1)."""
    conv = _fft_self_convolve(a)
    w = _smooth_weights(conv, alpha)
    grad = 2.0 * _fft_convolve(a[::-1], w)[len(a) - 1 : len(a) - 1 + len(a)]
    a = a * np.exp(-lr * grad)
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, EPS)
    a /= a.sum()
    return a


def _projected_gradient_step(a, eta, active_w):
    """Projected gradient step on the sub‑gradient defined by ``active_w``."""
    grad = 2.0 * _fft_convolve(a[::-1], active_w)[len(a) - 1 : len(a) - 1 + len(a)]
    a = a - eta * grad
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, EPS)
    a /= a.sum()
    return a


def _adam_step(logits, m, v, t, step_size):
    """Adam update on unconstrained logits."""
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = beta1 * m + (1.0 - beta1) * step_size
    v = beta2 * v + (1.0 - beta2) * (step_size ** 2)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    lr = 0.02 / (np.sqrt(v_hat) + eps)
    logits = logits - lr * m_hat
    logits = np.clip(logits, -30, 30)
    return logits, m, v


def _enforce_symmetry_and_normalise(a):
    """Make the vector symmetric and re‑normalise it to sum = 1."""
    a = (a + a[::-1]) * 0.5
    a = np.maximum(a, EPS)
    a /= a.sum()
    return a


def _hillclimber_move(a, conv, eps=EPS):
    """Move a tiny chunk from a large‑gradient index to a small‑gradient one."""
    n = a.shape[0]
    w = _subgradient_weights(conv, eps_rel=eps)
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    src = int(np.argmax(grad))
    dst = int(np.argmin(grad))
    if src == dst:
        dst = random.randint(0, n - 1)
    max_delta = min(a[src] * 0.5, 0.02)
    delta = max_delta * random.random()
    if delta < eps:
        return a, conv, False
    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta
    a_new = _enforce_symmetry_and_normalise(a_new)
    new_conv = _fft_self_convolve(a_new)
    improved = new_conv.max() < conv.max() - 1e-14
    if improved:
        return a_new, new_conv, True
    else:
        return a, conv, False

# Analytic families (each returns a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.01):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    """Discrete Gaussian centred at (n‑1)/2."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

            a_eg = _exponential_gradient_step(a_eg, lr, alpha)
            a_eg = _enforce_symmetry_and_normalise(a_eg)

        # quick PGD
        steps_pg = 600
        a_pg = a_eg.copy()
        eta0, eta1 = 0.25, 0.004
        for it in range(steps_pg):
            if time.time() >= deadline:
                break
            conv = _fft_self_convolve(a_pg)
            w = _subgradient_weights(conv, eps_rel=1e-12)
            eta = eta0 * (eta1 / eta0) ** (it / steps_pg)
            a_pg = _projected_gradient_step(a_pg, eta, w)
            a_pg = _enforce_symmetry_and_normalise(a_pg)

        # brief hill‑climber
        a_hc = a_pg.copy()
        conv_hc = _fft_self_convolve(a_hc)
        for it in range(5000):
            if time.time() >= deadline:
                break
            a_hc, conv_hc, _ = _hillclimber_move(a_hc, conv_hc, eps=EPS)

        sc = _max_convolution_score(a_hc)
        if sc < best_score:
            best_seq = a_hc.copy()
            best_n = a_hc.size
            best_score = sc
            print(f"[{time.time():.1f}] restart n={n}  new best {sc:.6f}")

    # ------------------------------------------------------------------
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
