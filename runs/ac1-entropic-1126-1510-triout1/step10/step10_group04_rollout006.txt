"""
Propose a non‑negative step‑function (list of floats) that minimizes
``evaluate_sequence`` (i.e. 2·n·max(conv(a,a)) with sum(a)=1).

Algorithm
---------

1. **Broad analytic scan** – for several lengths (2000‑5000) we evaluate
   many parametric families (uniform, arcsine, triangular, exponential,
   Gaussian, sine‑power (both positive and negative exponents),
   symmetric Beta, power‑edge, power‑abs, simple V‑shape and very sparse
   Dirichlet draws).  Each candidate is projected onto the simplex and
   scored with a fast FFT based estimator 2·n·max(conv).  The best vector
   is kept.

2. **Smooth‑max mirror descent** – starting from the best vector we run
   an exponentiated‑gradient (mirror) descent on the simplex, where the
   smoothed max is obtained with a soft‑max weight vector.  The smoothing
   parameter grows from 0.1 to ~20 and the learning rate decays.  The
   best iterate seen so far is stored.

3. **Deterministic gradient hill‑climber** – after mirror descent we
   repeatedly move a tiny amount of mass from the index with the largest
   sub‑gradient (most responsible for the current maximum convolution
   entry) to the index with the smallest sub‑gradient.  The step size is
   a small fraction of the source mass.  After each move we recompute
   the exact maximum of the convolution and update the global best if
   an improvement is observed.

4. **Final jitter** – a tiny random perturbation is added (if time
   allows) to shake the solution out of possible plateaus.

All steps respect the wall‑clock budget (default 1000 s).  The function
returns a list that sums to 1 and contains only non‑negative numbers.
"""

import time
import random
import math

import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a vector already normalised to sum = 1.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic families – each returns a *probability* vector (sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.01) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # work in log‑space to avoid overflow for tiny p
    logw = (p - 1) * (np.log(x) + np.log(1 - x))
    w = np.exp(logw)
    if w.sum() == 0 or np.any(np.isinf(w)):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_sine_power_neg(n: int, power: float) -> np.ndarray:
    """Negative power makes edges heavier (power must be <0)."""
    idx = np.arange(n, dtype=np.float64)
    s = np.sin(np.pi * (idx + 0.5) / n)
    eps = 1e-12
    s = np.maximum(s, eps)                # avoid division by zero
    vals = s ** power                     # power is negative
    vals = np.maximum(vals, eps)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals /= vals.sum()
    return vals


def _distribution_power_edge(n: int, alpha: float) -> np.ndarray:
    """
    Edge‑heavy family:
        w_i ∝ ((i+0.5)*(n‑i‑0.5))^{-α}
    """
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - i - 0.5)
    eps = 1e-12
    vals = np.power(denom + eps, -alpha)
    vals /= vals.sum()
    return vals


def _distribution_V_shape(n: int) -> np.ndarray:
    centre = (n - 1) / 2.0
    vals = np.abs(np.arange(n, dtype=np.float64) - centre)
    eps = 1e-12
    vals = np.maximum(vals, eps)
    vals /= vals.sum()
    return vals


def _distribution_power_abs(n: int, beta: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    vals = np.abs(np.arange(n, dtype=np.float64) - centre) ** beta
    eps = 1e-12
    vals = np.maximum(vals, eps)
    vals /= vals.sum()
    return vals


def _random_symmetric_dirichlet(n: int, alpha: float) -> np.ndarray:
    """
    Draw a symmetric vector from Dirichlet(α) on the half‑grid and mirror it.
    """
    half = (n + 1) // 2
    half_vec = np.random.dirichlet(np.full(half, alpha))
    if n % 2 == 0:
        full = np.concatenate([half_vec, half_vec[::-1]])
    else:
        full = np.concatenate([half_vec, half_vec[-2::-1]])
    full /= full.sum()
    return full

def _mirror_descent(a: np.ndarray, deadline: float,
                    best_seq: np.ndarray, best_score: float):
    """
    Smoothed‑max mirror descent (exponentiated‑gradient) on the simplex.
    Returns the (possibly improved) a, best_seq and best_score.
    """
    n = a.shape[0]
    total_steps = 25000
    eps = 1e-12
    for it in range(total_steps):
        if time.time() >= deadline:
            break
        t = it / total_steps
        alpha = 0.1 * (200.0 ** t)          # grows from ~0.1 to ~20
        lr = 5.0 * (1.0 - t) + 0.05        # decays from ~5 to ~0.05

        conv = _fft_self_convolve(a)
        maxc = conv.max()
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()

        if it % 500 == 0:
            sc = _max_convolution_score(a)
            if sc < best_score - 1e-12:
                best_seq = a.copy()
                best_score = sc
                print(f"[{time.time():.1f}] mirror it={it}  score={sc:.6f}")

    return a, best_seq, best_score


def _gradient_hillclimb(a: np.ndarray, deadline: float,
                        best_seq: np.ndarray, best_score: float):
    """
    Deterministic hill‑climbing that moves mass from the index with the
    largest sub‑gradient to the index with the smallest sub‑gradient.
    Returns the (possibly improved) a, best_seq and best_score.
    """
    n = a.shape[0]
    eps = 1e-12
    it = 0
    while time.time() < deadline:
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # sub‑gradient of the exact max (mask of active constraints)
        mask = np.abs(conv - maxc) <= 1e-14
        if not np.any(mask):
            break
        w = mask.astype(np.float64)
        w /= w.sum()
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        src = int(np.argmax(grad))
        tgt = int(np.argmin(grad))
        if src == tgt:
            break

        # take a modest fraction of the source mass
        delta = min(a[src] * 0.5, 0.01)
        if delta <= eps:
            break

        a[src] -= delta
        a[tgt] += delta
        a = np.maximum(a, eps)
        a /= a.sum()

        it += 1
        if it % 100 == 0:
            sc = _max_convolution_score(a)
            if sc < best_score - 1e-12:
                best_seq = a.copy()
                best_score = sc
                print(f"[{time.time():.1f}] grad_hill it={it}  best={sc:.6f}")

    return a, best_seq, best_score
