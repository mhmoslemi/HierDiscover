"""
Propose a low‑score step function.

The routine builds a diverse set of analytic and random candidates, keeps the
best few, and refines each of them with two complementary strategies:

1.  A continuation‑Adam optimizer that minimises a smooth approximation of the
    “max‑convolution’’ objective.  The smoothness is controlled by a temperature
    parameter α that starts small (≈1) and is geometrically increased up to a
    large value (≈128).  The optimizer works directly on the simplex by
    projecting after each Adam step; the gradient of the smoothed objective
    is obtained analytically via FFT‑based convolution.

2.  A short Frank‑Wolfe / mass‑transfer polishing phase (identical to the
    baseline) followed by a simulated‑annealing jitter phase.

Both phases respect the 1000 s wall‑clock budget.  The final best sequence
(together with its length) is returned as a plain list of floats.  The global
variable ``height_sequence_1`` (if present) is used as a warm start.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate objective (∑a=1) => 2·n·max(conv(a,a))."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _smooth_subgrad(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smoothed max‑convolution:
        f_α(a) = (1/α)·log∑_k exp(α·(a*a)_k)

    Returns the gradient ∇f_α(a) (same shape as ``a``) for a vector that
    already satisfies Σa = 1 and a≥0.
    """
    # self‑convolution
    conv = _fft_self_convolve(a)          # length 2n-1
    # stable softmax weights
    max_c = conv.max()
    scaled = alpha * (conv - max_c)       # avoid overflow
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()    # w_i = softmax_i(α·conv_i)

    # gradient = 2 * (a_rev * w)_{n-1:n-1+n}
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]
    return grad

def _adam_smooth(a0: np.ndarray, deadline: float,
                 step0: float = 0.30,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 max_iters: int = 1_500_000,
                 alpha_start: float = 1.0,
                 alpha_factor: float = 1.4,
                 max_alpha: float = 128.0,
                 alpha_inc_iter: int = 3_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex that minimises a smoothed max‑convolution.
    The smoothing parameter α is increased geometrically (continuation) from
    ``alpha_start`` up to ``max_alpha``.
    Returns the best point seen (as a NumPy array) and its surrogate score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    t = 0
    alpha = alpha_start
    while time.time() < deadline and t < max_iters:
        t += 1
        grad = _smooth_subgrad(a, alpha)

        # Adam updates (gradient descent)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        # increase smoothing temperature periodically
        if (t % alpha_inc_iter == 0) and (alpha < max_alpha):
            alpha = min(alpha * alpha_factor, max_alpha)

    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          eta0: float = 0.9,
                          max_iters: int = 1_500_000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (KL‑mirror) descent on the simplex (baseline)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)          # reuse sub‑gradient of the true max
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score

# Analytic families (extended)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _edge_power_distribution(n: int, s: float) -> np.ndarray:
    """w_i ∝ (i·(n‑i+1))^{-s},  i = 1,…,n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = (i * (n - i + 1)) ** (-s)
    vals /= vals.sum()
    return vals


def _double_bump_distribution(n: int,
                              block_frac: float = 0.25,
                              gap_frac: float = 0.1) -> np.ndarray:
    """Two disjoint uniform blocks."""
    total_block = max(2, int(block_frac * n))
    m = total_block // 2                     # size of each block
    g = max(1, int(gap_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:m] = 1.0 / (2 * m)
    start2 = m + g
    if start2 + m > n:
        start2 = n - m
    a[start2:start2 + m] = 1.0 / (2 * m)
    return a


def _random_block_distribution(n: int, blocks: int = 2,
                              min_frac: float = 0.01,
                              max_frac: float = 0.25) -> np.ndarray:
    """Place a few uniform blocks at random positions."""
    a = np.zeros(n, dtype=np.float64)
    mass_per_block = 1.0 / blocks
    for _ in range(blocks):
        blk_len = max(1, int(np.random.uniform(min_frac, max_frac) * n))
        start = np.random.randint(0, n - blk_len + 1)
        a[start:start + blk_len] = mass_per_block / blk_len
    a /= a.sum()
    return a


def _hann_distribution(n: int) -> np.ndarray:
    """Classic Hann window (non‑negative)."""
    i = np.arange(n, dtype=np.float64)
    vals = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    vals /= vals.sum()
    return vals


def _hamming_distribution(n: int) -> np.ndarray:
    """Hamming window."""
    i = np.arange(n, dtype=np.float64)
    vals = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _mixture_distribution(a: np.ndarray, b: np.ndarray, w: float) -> np.ndarray:
    """Convex mixture w·a + (1‑w)·b, renormalised."""
    mix = w * a + (1.0 - w) * b
    if mix.sum() == 0:
        mix = np.full_like(mix, 1.0 / mix.size)
    else:
        mix /= mix.sum()
    return mix


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with concentration ``alpha``."""
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0:
        x = np.full_like(x, 1.0 / n)
    else:
        x = x / x.sum()
    return x

        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            if prev.sum() == 0:
                prev = np.full_like(prev, 1.0 / prev.size)
            else:
                prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
            for _ in range(6):
                jitter = np.random.randn(prev.size) * 5e-5
                noisy = _project_onto_simplex(prev + jitter, s=1.0)
                candidates.append((noisy, _max_convolution_score(noisy)))
            print(f"[{time.time():.1f}] added previous best (+perturbations)")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Keep the top‑k seeds for intensive refinement
    candidates.sort(key=lambda x: x[1])
    top_k = min(20, len(candidates))
    top_seeds = candidates[:top_k]
    print(f"Generated {len(candidates)} candidates, keeping top {top_k} for refinement.")

    best_seq = top_seeds[0][0].copy()
    best_score = top_seeds[0][1]
    print(f"Initial surrogate score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Intensive refinement of each seed
    for idx, (seed_seq, seed_score) in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        n = seed_seq.shape[0]

        # ----- continuation‑Adam on smooth max ----------
        seq_smooth, sc_smooth = _adam_smooth(seed_seq, deadline,
                                             step0=0.30,
                                             alpha_start=1.0,
                                             alpha_factor=1.5,
                                             max_alpha=256.0,
                                             alpha_inc_iter=2500)
        if sc_smooth < best_score:
            best_score, best_seq = sc_smooth, seq_smooth.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} smooth‑Adam → {best_score:.6f}")

        # ----- baseline Adam (same as before) ----------
        seq_adam, sc_adam = _adam_smooth(seed_seq, deadline,  # reuse the same implementation
                                         step0=0.30,
                                         alpha_start=64.0,     # start already “sharp”
                                         alpha_factor=1.0,
                                         max_alpha=64.0,
                                         alpha_inc_iter=1)
        if sc_adam < best_score:
            best_score, best_seq = sc_adam, seq_adam.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} sharp‑Adam → {best_score:.6f}")

        # ----- exponentiated‑gradient (baseline) ----------
        seq_eg, sc_eg = _exp_gradient_descent(seed_seq, deadline)
        if sc_eg < best_score:
            best_score, best_seq = sc_eg, seq_eg.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} ExpGrad → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Frank‑Wolfe‑style mass‑transfer polishing (as in the baseline)
    a = best_seq.copy()
    cur_score = best_score
    fw_iter = 0
    fractions = (1.0, 0.75, 0.5, 0.25, 0.125)

    while time.time() < deadline and fw_iter < 300_000:
        fw_iter += 1
        grad, _, _ = _subgrad(a)                     # sub‑gradient of true max
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-15:
            break
        improved = False
        for f in fractions:
            delta = max_transfer * f
            a_new = a.copy()
            a_new[i_src] -= delta
            a_new[i_dst] += delta
            sc = _max_convolution_score(a_new)
            if sc < cur_score - 1e-13:
                a = a_new
                cur_score = sc
                improved = True
                break
        if improved and cur_score < best_score - 1e-14:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time():.1f}] FW iter={fw_iter} new best {best_score:.6f}")

        # occasional jitter to escape tiny plateaus
        if fw_iter % 8_000 == 0 and np.random.rand() < 0.15:
            jitter = np.random.randn(a.size) * 4e-5
            a = _project_onto_simplex(a + jitter, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    # Simulated‑annealing / jitter phase (final push)
    a = best_seq.copy()
    cur_score = best_score
    temp_start = 0.030
    temp_end = 0.0015
    anneal_iter = 0
    while time.time() < deadline and anneal_iter < 400_000:
        anneal_iter += 1
        # linear temperature schedule (remaining fraction)
        t_rem = deadline - time.time()
        fraction = max(0.0, t_rem / (budget_s - 0.5))
        temp = temp_end + (temp_start - temp_end) * fraction

        i = np.random.randint(a.size)
        j = np.random.randint(a.size - 1)
        if j >= i:
            j += 1
        max_delta = min(a[i], 1.0 - a[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * temp
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if cur_score < best_score - 1e-14:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time():.1f}] Anneal iter={anneal_iter} new best {best_score:.6f}")
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_score - cur_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                a = a_new
                cur_score = new_score

    # ------------------------------------------------------------------
    print(f"Final best surrogate score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()

# Helper needed for the sub‑gradient used in the polishing phases.
def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) under Σa = 1.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # gradient of the (non‑smooth) max‑convolution:
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for i=j terms (when k even)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active
