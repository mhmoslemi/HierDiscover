"""
Propose a non‑negative step function (list of floats) that minimizes
`evaluate_sequence`.  The search uses three phases:

1. **Extensive family sweep** – for several lengths we evaluate many
   analytic families (arcsine, symmetric & asymmetric Beta, power‑law,
   exponential, edge‑biased Dirichlet).  The best raw candidate is kept.

2. **Smooth‑max mirror descent** – starting from the best candidate we run
   exponentiated‑gradient (mirror) descent on a soft‑max approximation of the
   max‑convolution.  The temperature is increased and the learning‑rate is
   decreased geometrically, allowing a careful transition from exploration to
   fine tuning.

3. **Random mass‑transfer hill‑climber** – short stochastic polishing step
   that moves a tiny amount of probability mass between two coordinates and
   accepts the move if the true max‑convolution decreases.

All operations respect the constraints of `evaluate_sequence`
(non‑negative entries, sum ≥ 0.01, each entry ≤ 1000).  The function returns
the best sequence found before the time budget expires.
"""

# Utility functions ----------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    For a vector already normalised to sum = 1 returns 2·n·max(a*a),
    i.e. the exact value produced by `evaluate_sequence`.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    logw -= np.max(logw)  # for numerical stability
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_powerlaw(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-alpha}, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    w = 1.0 / np.power(idx, alpha)
    w /= w.sum()
    return w


def _distribution_exponential(n: int, r: float) -> np.ndarray:
    """
    Geometric/exponential decay a_i ∝ r**i, normalised to sum = 1.
    r must be in (0,1); for r very close to 1 the distribution is almost uniform.
    """
    i = np.arange(n, dtype=np.float64)
    # unnormalised weights (1‑r)·r**i
    base = (1.0 - r) * np.power(r, i)
    # truncation correction because sum_{i=0}^{n‑1} (1‑r)·r**i = 1‑r**n
    norm = 1.0 - r ** n
    if norm <= 0:
        # fall back to uniform if numerical issue
        return _distribution_uniform(n)
    w = base / norm
    return w


def _distribution_edge_dirichlet(n: int, beta_exp: float) -> np.ndarray:
    """
    Dirichlet draw with concentration concentrated at the edges:
    c_i ∝ ((i+0.5)*(n‑i‑0.5))^{-beta_exp}  + 0.1 .
    """
    idx = np.arange(n, dtype=np.float64)
    conc = ((idx + 0.5) * (n - idx - 0.5)) ** (-beta_exp) + 0.1
    vec = np.random.dirichlet(conc)
    return vec

    # asymmetric beta parameters (p small, q fixed) and vice‑versa
    asym_q = 0.5
    asym_p_grid = np.linspace(0.01, 0.5, 25)

    edge_beta_exps = (0.4, 0.6, 0.8, 1.0)

    # Helper to evaluate and maybe update the global best
    def _maybe_update(vec: np.ndarray, n: int, label: str):
        nonlocal best_seq, best_score, best_n
        if time.time() >= deadline:
            return
        vec = np.maximum(vec, eps)
        vec /= vec.sum()
        sc = _max_convolution_score(vec)
        if sc < best_score - 1e-12:
            best_score, best_seq, best_n = sc, vec.copy(), n
            print(f"[{time.time():.1f}] n={n} {label:<30} score={sc:.6f}")

    # --------------------------------------------------------------
    # Iterate over lengths (including larger ones at the end)
    for n in lengths + large_lengths:
        if time.time() >= deadline:
            break

        # ----- deterministic families ---------------------------------
        _maybe_update(_distribution_arcsine(n), n, "arcsine")
        _maybe_update(_distribution_uniform(n), n, "uniform")

        # symmetric Beta
        for p in p_grid:
            if time.time() >= deadline:
                break
            _maybe_update(_distribution_beta_sym(n, p), n,
                          f"Beta_sym(p={p:.3f})")

        # asymmetric Beta (left‑heavy)
        for p in asym_p_grid:
            if time.time() >= deadline:
                break
            _maybe_update(_distribution_beta_asym(n, p, asym_q), n,
                          f"Beta(p={p:.3f},q={asym_q})")
            _maybe_update(_distribution_beta_asym(n, asym_q, p), n,
                          f"Beta(p={asym_q},q={p:.3f})")

        # power‑law
        for alpha in alpha_grid:
            if time.time() >= deadline:
                break
            _maybe_update(_distribution_powerlaw(n, alpha), n,
                          f"powerlaw(alpha={alpha:.3f})")

        # exponential family (via c = n*(1‑r))
        for c in c_grid:
            if time.time() >= deadline:
                break
            r = 1.0 - c / n
            if r <= 0.0:
                continue
            _maybe_update(_distribution_exponential(n, r), n,
                          f"exp(r={r:.6f})")

        # edge‑biased Dirichlet draws
        for bex in edge_beta_exps:
            if time.time() >= deadline:
                break
            _maybe_update(_distribution_edge_dirichlet(n, bex), n,
                          f"edge‑Dirichlet(beta={bex})")

        # a few random Dirichlet draws (concentration 1)
        for _ in range(3):
            if time.time() >= deadline:
                break
            vec = np.random.dirichlet(np.ones(n))
            _maybe_update(vec, n, "Dirichlet(1)")

    # ------------------------------------------------------------------
    # If nothing was generated (extremely unlikely) fall back to arcsine.
    if best_seq is None:
        best_n = 4000
        best_seq = _distribution_arcsine(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback arcsine score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Refine with smooth‑max mirror descent (exponentiated gradient)
    a = best_seq.copy()
    n = a.shape[0]

    # Estimate how many iterations we can afford
    remaining = deadline - time.time()
    est_per_iter = 0.004              # seconds per iteration (FFT + overhead)
    max_iters = max(0, int(remaining / est_per_iter))
    total_iters = max(5000, min(25000, max_iters))

    # Temperature (soft‑max) schedule: start low, end high
    temp_start, temp_end = 0.5, 5000.0
    # Learning‑rate schedule (geometric decay)
    lr_start, lr_end = 0.3, 0.001

    for it in range(total_iters):
        if time.time() >= deadline:
            break
        t = it / total_iters
        alpha = temp_start * (temp_end / temp_start) ** t      # smooth‑max temperature
        lr = lr_start * (lr_end / lr_start) ** t

        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # soft‑max weights (the gradient of the smooth‑max)
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth‑max w.r.t. a (via convolution with reversed a)
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # exponentiated‑gradient (mirror descent) update
        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()

        if (it % max(1, total_iters // 10) == 0) or (it == total_iters - 1):
            cur_sc = _max_convolution_score(a)
            if cur_sc < best_score - 1e-12:
                best_score, best_seq = cur_sc, a.copy()
                print(f"[{time.time():.1f}] mirror it={it}/{total_iters} new best {cur_sc:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Projected Adam‑style sub‑gradient descent on the true max‑convolution
    a = best_seq.copy()
    velocity = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    step0 = 0.5
    sub_it = 0

    while time.time() < deadline:
        sub_it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        # active set (indices attaining the maximum up to tolerance)
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam momentum update (bias‑corrected)
        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** sub_it)

        step = step0
        # simple back‑tracking line search
        while True:
            a_cand = a - step * v_hat
            a_cand = _project_onto_simplex(a_cand, 1.0)
            a_cand = np.maximum(a_cand, eps)
            a_cand /= a_cand.sum()
            cand_sc = _max_convolution_score(a_cand)
            if cand_sc < best_score - 1e-12 or step < 1e-8:
                break
            step *= 0.5

        a = a_cand
        if cand_sc < best_score - 1e-12:
            best_score, best_seq = cand_sc, a.copy()
            print(f"[{time.time():.1f}] subgrad it={sub_it} new best {cand_sc:.6f}")

        # occasional random jitter to escape shallow plateaus
        if sub_it % 500 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    # ------------------------------------------------------------------
    # 4️⃣  Final stochastic hill‑climber (mass‑transfer) polish
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    hill_it = 0
    while time.time() < deadline:
        hill_it += 1
        # pick source index (one of the larger entries)
        src = np.argmax(a)                     # biggest entry
        # pick a target index that is currently small
        tgt = np.argmin(a)                     # smallest entry
        if src == tgt:
            # pick second smallest if necessary
            sorted_idx = np.argsort(a)
            tgt = sorted_idx[1]

        # propose moving a small delta from src to tgt
        max_delta = min(a[src] * 0.4, 0.02)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a2 = a.copy()
        a2[src] -= delta
        a2[tgt] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-13:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M
            best_seq = a.copy()
            best_score = 2.0 * n * M
            if hill_it % 50 == 0:
                print(f"[{time.time():.1f}] hill it={hill_it} new best {best_score:.6f}")

        # periodic jitter to avoid stagnation
        if hill_it % 1000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] hill jitter at it={hill_it}")

    final_score = 2.0 * n * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
