def _smoothmax_descent(a0, deadline, best_seq, best_score,
                       lr0=0.6, max_iter=20000,
                       alpha_start=0.2, alpha_end=2000,
                       eval_interval=200):
    a = a0.copy()
    n = a.shape[0]
    t=0
    while time.time()<deadline and t<max_iter:
        t+=1
        # compute conv
        conv = _fft_self_convolve(a)
        # compute beta=alpha schedule
        alpha = alpha_start * (alpha_end/alpha_start) ** (t / max_iter)  # geometric
        # compute softmax weights
        # shift for numeric stability
        maxc = conv.max()
        exp_vals = np.exp(alpha * (conv - maxc))
        w = exp_vals / exp_vals.sum()
        grad = _gradient_from_weights(a, w)   # shape n

        # step size schedule
        eta = lr0 / math.sqrt(t)

        # exponentiated gradient update
        a = a * np.exp(-eta * grad)

        # renormalize to simplex
        s = a.sum()
        if s==0:
            a = np.full(n, 1.0/n)
        else:
            a = a / s

        # occasional evaluation
        if t%eval_interval==0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth it={t} new best {best_score:.6f}")
    return best_seq, best_score
```

def upsample_distribution(a, new_n):
    old_n = a.shape[0]
    x_old = np.arange(old_n) / (old_n-1) if old_n>1 else np.array([0.0])
    x_new = np.arange(new_n) / (new_n-1)
    a_new = np.interp(x_new, x_old, a)
    # adjust to maintain sum 1 (or give weight of interpolation)
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    return a_new

"""
Propose a non‑negative step function that minimises
`evaluate_sequence`.  The algorithm builds on the previous
implementation but introduces two key ideas:

1. **Smooth max gradient descent** – instead of a hard
   `max` we use a softmax (log‑sum‑exp) with an annealing
   temperature.  This gives a smoother objective, lets the
   optimiser see *all* convolution entries and avoids getting
   stuck on a small active set.

2. **Multi‑scale seed generation** – we start from a diverse
   pool of analytic families, add the best known sequence
   (if available) and also up‑sample it to larger lengths.
   Each seed is refined with the smooth‑max descent and finally
   polished with cheap greedy pairwise mass swaps.

The whole search respects the given time budget (≈ 1000 s) and
returns the best sequence found as a plain list of floats.
"""

import time

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  max(conv(a,a))  at a given weight vector w.
    Implements 2·a⋆w  via an FFT‑based cross‑correlation (see the
    original code for details).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    return 2.0 * conv[start:start + n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (a is assumed to sum to 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic seed families (identical to the previous implementation) -----
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Generate a diverse pool of analytic seeds for a given length."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    for p in (0.05, 0.1, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0):
        cand.append(_distribution_beta_sym(n, p))

        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))

    for lam in (0.001, 0.005, 0.01, 0.02, 0.05):
        cand.append(_distribution_exponential_increasing(n, lam))

    for block_frac in (0.05, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac))

    for _ in range(3):
        cand.append(_distribution_dirichlet(n, rng,
                                            alpha=0.3 + 0.7 * rng.random()))

    cand.append(_distribution_random_sparse(n, rng, density=0.01))

    return cand

def _smoothmax_descent(a0: np.ndarray, deadline: float,
                       best_seq: np.ndarray, best_score: float,
                       lr0: float = 0.6,
                       max_iter: int = 25000,
                       alpha_start: float = 0.2,
                       alpha_end: float = 2000.0,
                       eval_interval: int = 250) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a *smooth* approximation of the max:
        smooth_max = (1/α)·log(∑_k exp(α·conv_k))

    The weight vector w = softmax(α·conv) yields the sub‑gradient
    2·a⋆w.  The step uses exponentiated‑gradient (mirror descent)
    on the probability simplex.  α is annealed geometrically from
    `alpha_start` to `alpha_end`.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1

        # self‑convolution
        conv = _fft_self_convolve(a)

        # geometric annealing of the temperature
        progress = t / max_iter
        alpha = alpha_start * ((alpha_end / alpha_start) ** progress)

        # numerically stable softmax
        maxc = conv.max()
        exp_vals = np.exp(alpha * (conv - maxc))
        w = exp_vals / exp_vals.sum()

        # gradient of the smooth max
        grad = _gradient_from_weights(a, w)

        # learning‑rate schedule
        eta = lr0 / math.sqrt(t)

        # exponentiated‑gradient update (mirror descent on KL)
        a = a * np.exp(-eta * grad)

        # projection onto the simplex (normalisation)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # occasional exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth it={t:5d}  new best {best_score:.6f}")

    return best_seq, best_score

def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Cheap greedy local mass‑transfer walk."""
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i:4d}→j={j:4d} δ={delta:.3g}  best {best_score:.6f}")
    return best_seq, best_score

def _upsample_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger size."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new = np.full(new_n, 1.0 / new_n)
    else:
        a_new /= a_new.sum()
    return a_new
