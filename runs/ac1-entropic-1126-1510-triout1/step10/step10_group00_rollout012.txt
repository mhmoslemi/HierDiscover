"""
propose_candidate – a fast multi‑stage search for the step‑function that
minimises evaluate_sequence.

Stage 1 – Analytic families:
    * Uniform, arcsine‑like (“edge weighted”) and a mixture with the
      uniform distribution are tried for several lengths (up to 80 000)
      and a dense grid of the shape parameter p and mixing weight λ.
    * The best few seeds (≤30) are kept.

Stage 2 – Refine the best seeds with Adam‑style sub‑gradient descent
    on the exact objective 2·n·max(conv(a,a)), keeping the iterate on the
    probability simplex by Euclidean projection.

Stage 3 – Final cheap greedy mass‑transfer walk (pairwise swaps) to polish
    the current best vector.

All work respects the given time budget (default 1000 s).  The function
returns a plain Python list; evaluate_sequence will handle the required
normalisation, clipping, etc.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (fast for all supported lengths)."""
    n = a.shape[0]
    # choose a power‑of‑two size that fits 2·n‑1 (full convolution length)
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector whose entries sum to 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_b = conv.max()
    # a is kept normalised (sum≈1) but we keep the denominator for safety
    sum_a = a.sum()
    if sum_a == 0:
        return float("inf")
    return 2.0 * n * max_b / (sum_a ** 2)


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Adam‑style sub‑gradient descent on the exact max‑convolution objective --
def _adam_subgradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr: float = 0.12,
    eval_interval: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser applied to the true sub‑gradient of max(conv(a,a)).
    The iterate is projected onto the probability simplex after each step.
    """
    n = a.shape[0]
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        idx = int(conv.argmax())

        # sub‑gradient from the maximal index
        ind = idx - np.arange(n)
        mask = (ind >= 0) & (ind < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[ind[mask]]

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep on the simplex (non‑negative & sum = 1)
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] adam it={t} new global best {best_score:.6f}"
                )
        if t >= 300_000:                # safety upper bound
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise mass transfer ---------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small amount of mass; keep move only if the objective improves."""
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score
