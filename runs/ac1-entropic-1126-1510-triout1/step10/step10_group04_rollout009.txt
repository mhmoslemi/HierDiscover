"""
Propose a step‑function (list of non‑negative floats) with a low
evaluate_sequence() score.

Algorithm
---------

1. **Broad analytic scan** – For several lengths (2000‑6000) generate a
   large collection of deterministic and random distributions:
   uniform, arcsine, triangular, exponential, Gaussian, sine‑power,
   symmetric Beta(p,p), power‑edge, power‑abs and plain Dirichlet draws.
   All vectors are normalised to sum = 1 and the cheap bound
   ``2·n·max(a*a)`` (computed via a real FFT) is evaluated.
   The best vector from this scan becomes the starting point.

2. **Adam‑style smooth‑max descent** – The non‑smooth objective
   ``max_k (a*a)_k`` is approximated by a soft‑max with a temperature
   parameter α that grows from ≈0.1 to ≈30.  The gradient of the
   smoothed maximum is
       grad = 2·conv_rev(a, w)   with   w_i ∝ exp(α·(conv_i−max)).
   An Adam optimiser updates the vector multiplicatively
   (exponentiated‑gradient) and the result is projected onto the
   probability simplex after each step.  The best run‑time candidate
   (according to the cheap bound) is kept.

3. **Targeted pairwise mass transfer** – Afterwards the current
   maximum‑convolution entry is identified (index k).  Among all
   contributing pairs (i, j) with i+j=k the pair with the largest
   product a_i·a_j is selected.  A fraction of the mass of both
   indices is moved to two random positions.  This directly reduces
   the dominant contribution to the maximum.  The step size is
   decreased slowly; improvements are recorded.

All phases respect a global wall‑clock budget (default 1000 s) and
intermediate best scores are printed.  The returned list is already
normalised, contains only non‑negative numbers (tiny ε added to avoid
exact zeros) and can be fed directly into ``evaluate_sequence``.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT (full length)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors (full length)."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """
    Fast evaluation of ``2·n·max(a*a)`` for a vector already normalised
    to sum = 1.  This equals the value returned by ``evaluate_sequence``
    up to the harmless scaling invariance.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * float(conv.max())

# Analytic families – each returns a *probability* vector (sum = 1)

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.01) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # work in log‑space for stability
    logw = (p - 1.0) * (np.log(x) + np.log(1.0 - x))
    # shift to avoid overflow
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or np.any(np.isinf(w)):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals /= vals.sum()
    return vals


def _distribution_power_edge(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - i - 0.5)
    eps = 1e-12
    vals = np.power(denom + eps, -alpha)
    vals /= vals.sum()
    return vals


def _distribution_power_abs(n: int, beta: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    vals = np.abs(i - centre) ** beta
    eps = 1e-12
    vals = np.maximum(vals, eps)
    vals /= vals.sum()
    return vals
