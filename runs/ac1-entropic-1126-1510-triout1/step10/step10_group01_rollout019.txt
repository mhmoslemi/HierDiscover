"""search_stepfn.py
A lightweight stochastic optimiser for the “step‑function” challenge.

The objective that we minimise is

    2·n·max(conv(a,a))/(sum(a)^2)

where ``conv`` is the ordinary (non‑reversed) convolution.  The
objective is scale‑invariant, therefore the vector is always kept
normalised to sum = 1.

The routine works in three stages

1. **Analytic seeds** – a small pool of families (uniform, arcsine,
   beta‑symmetric, beta‑asymmetric and simple power‑law / exponential
   decays) is generated for a few moderate lengths (≈ 2000‑4000).  The
   best few seeds are retained.

2. **Exponential‑gradient polishing** – for each seed an
   exponentiated‑gradient (mirror‑descent) step is performed using the
   full active set of convolution indices that attain the current
   maximum.  This is exactly the “multimax” update used in the reference
   implementation but with a slower learning‑rate schedule and many
   more iterations (the whole remaining time budget is spent here).

3. **Simulated‑annealing local walk** – the best vector from the
   previous stage is further refined by a cheap random mass‑transfer
   walk that occasionally accepts uphill moves (temperature linearly
   decreasing to 0).  This step is able to escape shallow local minima
   and typically yields a small but consistent improvement.

All heavy work respects the global time budget (``budget_s``).  The
function finally returns the best sequence found as a plain list of
floats, ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum(a)^2 (sum is forced to 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_decrease(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n-i)^p (decreasing power‐law)."""
    vals = (np.arange(n, 0, -1, dtype=np.float64) ** p)
    vals /= vals.sum()
    return vals


def _distribution_power_increase(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^p (increasing power‐law)."""
    vals = (np.arange(1, n + 1, dtype=np.float64) ** p)
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    """Two equal spikes at the ends."""
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the sub‑gradient 2·a⋆w where w is a weight vector (length 2n‑1)
    that sums to 1.  The operation is performed via an FFT‑based
    cross‑correlation: grad_i = 2·∑_j w_{i+j}·a_j.
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reversed a
    size = 1 << ((n + len(w) - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


def _multimax_subgradient_descent(a: np.ndarray, deadline: float,
                                 best_seq: np.ndarray, best_score: float,
                                 lr0: float = 0.5, max_iter: int = 40000,
                                 eval_interval: int = 500) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) using the *full active set*
    of convolution indices attaining the maximal value.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)          #  η_t = lr0 / √t

        a = a * np.exp(-eta * grad)       # exponentiated‑gradient step
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _annealing_local_search(a: np.ndarray, deadline: float,
                            best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """
    Cheap random mass‑transfer walk with a simple linear cooling schedule.
    The move is accepted if it improves the objective or with probability
    exp(-(Δ)/T) otherwise.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = _objective(cur_a)

    start_time = time.time()
    total_time = deadline - start_time
    iteration = 0
    # initial temperature (chosen so that the first few uphill moves have
    # a reasonable acceptance probability)
    T0 = 1.0

    while time.time() < deadline:
        iteration += 1
        elapsed = time.time() - start_time
        T = T0 * (1.0 - elapsed / total_time)  # linear cooling

        # pick a source with non‑zero mass
        src_candidates = np.where(cur_a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)

        # choose a distinct target
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction (up to 30 %) of the source mass
        delta = cur_a[i] * np.random.rand() * 0.30
        if delta <= 1e-14:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # re‑project to the simplex – protects against rounding drift
        new_a = _project_onto_simplex(new_a, s=1.0)

        new_score = _objective(new_a)

        # Metropolis accept/reject
        if new_score < cur_score - 1e-13 or np.exp(-(new_score - cur_score) / max(T, 1e-15)) > np.random.rand():
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score - 1e-13:
                best_score = new_score
                best_seq = new_a.copy()
                print(f"[{time.time():.1f}] anneal it={iteration} Δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    The routine runs for at most ``budget_s`` seconds and returns the best
    sequence found (as a Python list of floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # a tiny safety margin

    # ------------------------------------------------------------------
    # 1. Choose a modest length – the objective is scale‑invariant, so
    #    n≈2000 offers a good trade‑off between accuracy and speed.
    n = 2000

    # ------------------------------------------------------------------
    # 2. Build a small pool of analytic seeds.
    seeds = []

    seeds.append(_distribution_uniform(n))
    seeds.append(_distribution_arcsine(n))

    for p in (0.05, 0.1, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0):
        seeds.append(_distribution_beta_sym(n, p))

    for _ in range(5):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        seeds.append(_distribution_beta_asym(n, alpha, beta))

    for p in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_distribution_power_decrease(n, p))

    for p in (0.5, 1.0, 2.0):
        seeds.append(_distribution_power_increase(n, p))

    seeds.append(_distribution_two_blocks(n, block_frac=0.07))

    # ------------------------------------------------------------------
    # 3. Evaluate all seeds, keep the few best for intensive polishing.
    top = []                      # list of (score, vector)
    best_seq = None
    best_score = float("inf")

    for cand in seeds:
        if time.time() >= deadline:
            break
        cand = np.asarray(cand, dtype=np.float64)
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()                     # safety normalisation
        sc = _objective(cand)
        top.append((sc, cand))
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time():.1f}] seed new best {best_score:.6f}")

    # keep the 5 best seeds
    top.sort(key=lambda x: x[0])
    top = top[:5]

    # ------------------------------------------------------------------
    # 4. Intensive multimax sub‑gradient descent on the elite seeds.
    remaining = deadline - time.time()
    if remaining > 30:                 # give the optimiser a non‑trivial window
        per_seed = remaining / max(1, len(top))
        for sc, seed_vec in top:
            sub_deadline = time.time() + per_seed
            seq, sc_new = _multimax_subgradient_descent(
                seed_vec.copy(),
                sub_deadline,
                best_seq=best_seq,
                best_score=best_score,
                lr0=0.6,
                max_iter=60000,
                eval_interval=800,
            )
            if sc_new < best_score:
                best_score = sc_new
                best_seq = seq.copy()
                print(f"[{time.time():.1f}] after sub‑grad new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final annealing walk for a final polishing.
    if time.time() < deadline:
        best_seq, best_score = _annealing_local_search(
            best_seq.copy(),
            deadline,
            best_seq=best_seq,
            best_score=best_score,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
