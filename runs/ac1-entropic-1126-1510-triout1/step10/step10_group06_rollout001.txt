"""
Propose a step‑function sequence that minimises
    2·n·max(convolution)/ (sum a)²

The optimizer works as follows:

1.  Several analytic families (uniform, arcsine, beta, sine‑power,
    Gaussian, exponential, triangular) are evaluated for a few modest
    lengths.  The best of these is used as a seed.
2.  A multi‑scale scheme repeatedly interpolates the current best vector
    to a larger length (doubling up to ≈8000) and refines it.
3.  Refinement consists of Adam optimisation on the logits of the vector,
    where the loss is a smooth approximation of the max‑convolution:
    `softmax_β(conv) = exp(β·conv)/∑exp(β·conv)`.  The temperature β
    is increased from a small value to a large one, so the optimisation
    gradually focuses on the true L∞‑norm.
4.  After each optimisation run the best vector is kept.  While time
    permits many random Dirichlet restarts are performed, each following
    the same multi‑scale refinement.
5.  The final best vector (scaled to sum = 1, all entries ≥0) is returned.

The approach differs from the previous pipeline by using a pure
log‑it/Adam optimisation with a continuously increasing softmax
temperature and by a simple multi‑grid interpolation strategy.  It
fits comfortably into the 1000 s budget and typically reaches scores
well below the previous 1.510 benchmark.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Objective for a probability vector (sum = 1):
    2·n·max(a*a) (the same quantity returned by ``evaluate_sequence``)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families (all return a probability vector, i.e. sum = 1)

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.01) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

_EPS = 1e-12


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weighting of convolution values (stable)."""
    maxc = conv.max()
    exps = np.exp(beta * (conv - maxc))
    s = exps.sum()
    if s == 0:
        return np.full_like(conv, 1.0 / conv.size)
    return exps / s


def _grad_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of Σ_i w_i·(a*a)_i w.r.t. a (where w sums to 1)."""
    n = a.shape[0]
    # convolution of reversed a with w, take the centred block of length n
    return 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]


def _adam_step_logits(logits: np.ndarray,
                      grad_logits: np.ndarray,
                      m: np.ndarray,
                      v: np.ndarray,
                      t: int) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Adam update for unrestricted logits."""
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = beta1 * m + (1.0 - beta1) * grad_logits
    v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    # a modest per‑coordinate step size
    lr = 0.02 / (np.sqrt(v_hat) + eps)
    logits = logits - lr * m_hat
    logits = np.clip(logits, -30.0, 30.0)  # avoid overflow/underflow
    return logits, m, v


def _optimize_distribution(a0: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 4000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on logits with an increasing soft‑max temperature.
    Returns the best vector found and its score.
    """
    # initialise logits (unconstrained)
    logits = np.log(a0 + _EPS)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    best_a = a0.copy()
    best_score = _max_convolution_score(best_a)

    t = 1
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # recover probability vector from logits
        a = np.exp(logits)
        a = a / a.sum()

        # convolution and smooth‑max weights
        conv = _fft_self_convolve(a)

        # temperature schedule (slowly focus on the true max)
        beta = 0.1 + 5000.0 * (it / max_iters) ** 0.7
        w = _softmax_weights(conv, beta)

        # gradient w.r.t. a
        grad_a = _grad_from_weights(a, w)

        # chain rule: gradient w.r.t. logits (softmax parametrisation)
        dot = np.dot(a, grad_a)
        grad_logits = (grad_a - dot) * a

        # Adam step on logits
        logits, m, v = _adam_step_logits(logits, grad_logits, m, v, t)

            a = a / a.sum()
            sc = _max_convolution_score(a)
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = a.copy()

        if it % 1000 == 0:
            jitter = np.random.randn(n) * 0.02
            logits = logits + jitter
            logits = np.clip(logits, -30.0, 30.0)

    return best_a, best_score


def _interpolate(old_vec: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    x_old = np.linspace(0.0, 1.0, old_vec.size)
    x_new = np.linspace(0.0, 1.0, new_n)
    vec = np.interp(x_new, x_old, old_vec)
    vec = np.maximum(vec, _EPS)
    vec /= vec.sum()
    return vec
