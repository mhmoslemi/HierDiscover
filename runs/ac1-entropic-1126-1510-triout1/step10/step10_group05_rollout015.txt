"""
Propose a high‑quality non‑negative step function.

Algorithm
---------
1. **Broad seeding** – generate a few analytically‑motivated candidates:
   the arcsine distribution (the known near‑optimal shape) on several
   lengths up to the allowed maximum (≈ 12000).  All candidates are
   ℓ₁‑normalised.

2. **Select the best seed** using the exact surrogate
   `score = 2·n·max(conv(a,a))` (identical to `evaluate_sequence`
   because the vectors are normalised).

3. **Iterative local optimisation** – repeatedly move a chunk of mass
   from the coordinate with the largest sub‑gradient to the coordinate
   with the smallest sub‑gradient.  For each pair we compute the
   optimal transfer amount by solving a cheap one‑dimensional problem
   that exactly evaluates the new maximal convolution value (only three
   convolution entries can change).  The move is performed only if it
   strictly reduces the objective.

4. **Random hill‑climb** – after the deterministic phase a short random
   mass‑transfer hill‑climber is applied to escape possible local minima.

The whole routine respects the supplied time budget (≈ 1000 s) and
returns a list ready for `evaluate_sequence`.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate for `evaluate_sequence` when Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad_from_conv(a: np.ndarray, b: np.ndarray):
    """
    Sub‑gradient of `max(conv(a,a))` given the full convolution `b`.
    Returns (gradient, cur_max, active_indices).
    """
    n = a.shape[0]
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    # weight vector for the active constraints
    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives the summed gradient
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices (i = j)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def order_alternate_extremes(desc_vals: np.ndarray) -> np.ndarray:
    """Spread the largest values alternatingly to the left‑most and right‑most positions."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    left, right = 0, n - 1
    toggle = True
    for v in desc_vals:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    return out


# ----------------------------------------------------------------------
# Random hill‑climber --------------------------------------------------
def _hill_climb(a: np.ndarray, deadline: float,
                max_iters: int = 300_000, epsilon: float = 0.02) -> tuple[np.ndarray, float]:
    """Simple hill‑climber that repeatedly moves a random fraction ε of mass."""
    best = a.copy()
    best_score = _max_convolution_score(best)
    n = best.size
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or best[i] < 1e-12:
            continue
        delta = min(best[i] * epsilon, 1.0 - best[j])
        if delta <= 0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = _max_convolution_score(cand)
        if sc < best_score:
            best = cand
            best_score = sc
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:

    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the supplied time budget (≈ 1000 s) and returns the best
    vector discovered.
    """

            return

        s = a.sum()
        if s < 1e-12:
            return

        score = _max_convolution_score(a)
        candidates.append((a, score))

    # ------------------------------------------------------------------

        idx_ij = i_src + i_dst
        n = best_seq.size

        # values of the three entries that can change
        b_2i = b[idx_2i] if 0 <= idx_2i < b.size else 0.0
        b_2j = b[idx_2j] if 0 <= idx_2j < b.size else 0.0
        b_ij = b[idx_ij] if 0 <= idx_ij < b.size else 0.0
