"""
Hybrid grid‑search + local‑refinement for the step‑function problem.

The algorithm proceeds in three stages:

1. **Broad candidate generation** – a collection of analytic families
   (arcsine, β‑distributions, edge‑power, double‑bump, etc.) is sampled
   for a handful of sequence lengths.  For the edge‑power and symmetric
   β families a fine grid over the shape parameter is explored (step
   0.01/0.02).  The best ~20 candidates (by the exact surrogate
   `2·n·max(conv(a,a))` with Σa=1) are retained.

2. **Intensive local optimisation** – each retained seed is refined by
   three first‑order methods that respect the simplex constraint:
   * Adam on the Euclidean projection,
   * Exponentiated‑gradient (mirror descent w.r.t. KL),
   * A Frank–Wolfe “mass‑transfer’’ step that moves probability mass
     from the largest to the smallest gradient coordinate.

3. **Final simulated‑annealing push** – a cheap annealing loop with
   random pairwise transfers is run on the current best point until the
   time budget (≈1000 s) expires.

All operations are performed on the simplex (Σa=1, a_i≥0); the
`evaluate_sequence` function is scale‑invariant, so the returned list can
be used directly.  The code respects the deadline and prints progress
messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """
    Exact surrogate objective when Σa = 1:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) under Σa = 1.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # gradient of the (non‑smooth) max‑convolution:
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for i=j terms (when k even)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.30,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 2_000_000):
    """Projected Adam on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        sc = _max_convolution_score(a)
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          eta0: float = 0.9,
                          max_iters: int = 1_500_000):
    """Exponentiated‑gradient descent (mirror w.r.t. KL) on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s

        sc = _max_convolution_score(a)
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score

def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with concentration ``alpha``."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Analytic families (all return a vector with sum = 1)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _edge_power_distribution(n: int, s: float) -> np.ndarray:
    """
    w_i ∝ (i·(n‑i+1))^{-s}, i = 1,…,n.
    Generalises the arcsine case (s=0.5).
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = (i * (n - i + 1)) ** (-s)
    vals /= vals.sum()
    return vals


def _double_bump_distribution(n: int,
                              block_frac: float = 0.25,
                              gap_frac: float = 0.1) -> np.ndarray:
    """Two equal uniform blocks separated by a gap."""
    total_block = max(2, int(block_frac * n))
    m = total_block // 2                 # size of each block
    g = int(gap_frac * n)
    a = np.zeros(n, dtype=np.float64)
    a[:m] = 1.0 / (2 * m)
    start2 = m + g
    if start2 + m > n:
        start2 = n - m
    a[start2:start2 + m] = 1.0 / (2 * m)
    return a


def _random_block_distribution(n: int, blocks: int = 2,
                              min_frac: float = 0.01,
                              max_frac: float = 0.25) -> np.ndarray:
    """Place a few uniform blocks at random positions."""
    a = np.zeros(n, dtype=np.float64)
    mass_per_block = 1.0 / blocks
    for _ in range(blocks):
        blk_len = max(1, int(np.random.uniform(min_frac, max_frac) * n))
        start = np.random.randint(0, n - blk_len + 1)
        a[start:start + blk_len] = mass_per_block / blk_len
    a /= a.sum()
    return a

def _maybe_add(candidate_vec: np.ndarray,
               cand_score: float,
               top_list: list,
               max_keep: int = 20):
    """Maintain a sorted list of the best `max_keep` candidates."""
    if len(top_list) < max_keep:
        top_list.append((cand_score, candidate_vec.copy()))
        top_list.sort(key=lambda x: x[0])
    else:
        if cand_score < top_list[-1][0]:
            top_list[-1] = (cand_score, candidate_vec.copy())
            top_list.sort(key=lambda x: x[0])

                _maybe_add(a, _max_convolution_score(a), top_candidates)

        # ---- Mixtures of arcsine and edge‑power ------------------------------------
        arcs = _arcsine_distribution(n)
        for s in (0.40, 0.45, 0.50, 0.55, 0.60):
            edge = _edge_power_distribution(n, s)
            for w in (0.2, 0.4, 0.6, 0.8):
                a = w * arcs + (1.0 - w) * edge
                a /= a.sum()                       # renormalise (should stay 1)
                _maybe_add(a, _max_convolution_score(a), top_candidates)

        # ---- add a few jittered variants of the best candidate we have so far -----
        if top_candidates:
            best_vec = top_candidates[0][1]
            for _ in range(2):
                jitter = np.random.randn(n) * 5e-5
                noisy = _project_onto_simplex(best_vec + jitter, s=1.0)
                _maybe_add(noisy, _max_convolution_score(noisy), top_candidates)

    # ------------------------------------------------------------------
    # Report the best surrogate score from the grid phase
    best_score = top_candidates[0][0]
    best_seq = top_candidates[0][1].copy()
    print(f"[{time.time():.1f}] Grid phase best surrogate = {best_score:.6f}
          (length={len(best_seq)})")
    # ------------------------------------------------------------------
    # 2️⃣  Local refinement of the retained seeds
    # ------------------------------------------------------------------
    # we will refine at most the top 10 seeds to stay within the time budget
    seeds_to_refine = top_candidates[:10]

    for idx, (seed_score, seed_vec) in enumerate(seeds_to_refine):
        if time.time() >= deadline:
            break
        n = seed_vec.shape[0]

        # Adam
        a_adam, sc_adam = _adam_descent(seed_vec, deadline,
                                        step0=0.25,
                                        max_iters=1_000_000)
        if sc_adam < best_score - 1e-14:
            best_score, best_seq = sc_adam, a_adam.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(seeds_to_refine)} Adam → {best_score:.6f}")

        # Exponentiated‑gradient
        a_eg, sc_eg = _exp_gradient_descent(seed_vec, deadline,
                                            eta0=0.9,
                                            max_iters=800_000)
        if sc_eg < best_score - 1e-14:
            best_score, best_seq = sc_eg, a_eg.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(seeds_to_refine)} EG → {best_score:.6f}")

        # Frank‑Wolfe mass‑transfer polishing (a few hundred iterations only)
        a = best_seq.copy()
        cur = best_score
        fw_iter = 0
        fractions = (1.0, 0.5, 0.25, 0.1)
        while time.time() < deadline and fw_iter < 200_000:
            fw_iter += 1
            grad, _, _ = _subgrad(a)
            i_src = int(np.argmax(grad))
            i_dst = int(np.argmin(grad))
            if i_src == i_dst:
                break
            max_transfer = min(a[i_src], 1.0 - a[i_dst])
            if max_transfer < 1e-15:
                break
            improved = False
            for fct in fractions:
                delta = max_transfer * fct
                a_new = a.copy()
                a_new[i_src] -= delta
                a_new[i_dst] += delta
                sc_new = _max_convolution_score(a_new)
                if sc_new < cur - 1e-13:
                    a = a_new
                    cur = sc_new
                    improved = True
                    break
            if improved and cur < best_score - 1e-14:
                best_score, best_seq = cur, a.copy()
                print(f"[{time.time():.1f}] FW after seed {idx+1} → {best_score:.6f}")
                break   # stop this inner FW once we have an improvement

    # ------------------------------------------------------------------
    # 3️⃣  Final annealing push (run until time expires)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cur = best_score
    anneal_iter = 0
    temp_start = 0.025
    temp_end = 0.001
    while time.time() < deadline and anneal_iter < 400_000:
        anneal_iter += 1
        # linear temperature schedule based on remaining fraction
        fraction = max(0.0, (deadline - time.time()) / (budget_s - 0.2))
        temp = temp_end + (temp_start - temp_end) * fraction

        n = a.size
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        max_delta = min(a[i], 1.0 - a[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * np.random.rand() * temp
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_sc = _max_convolution_score(a_new)

        if new_sc < cur - 1e-15:
            a = a_new
            cur = new_sc
            if cur < best_score - 1e-14:
                best_score, best_seq = cur, a.copy()
                print(f"[{time.time():.1f}] Anneal iter {anneal_iter} → {best_score:.6f}")
        else:
            # Metropolis step
            prob = np.exp(-(new_sc - cur) / (temp + 1e-12))
            if np.random.rand() < prob:
                a = a_new
                cur = new_sc

    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] Finished. Best surrogate score = {best_score:.6f}
          (length={len(best_seq)})")
    return best_seq.tolist()
