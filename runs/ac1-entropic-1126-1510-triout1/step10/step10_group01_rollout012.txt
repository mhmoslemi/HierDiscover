"""
Annealed exponential‑gradient descent for the max‑convolution problem.

We minimise the evaluation function

    f(a) = 2·n·max(conv(a,a)) / (sum(a)²)

for a non‑negative step‑function a.  The algorithm works in three stages:

1.  **Seed generation** – a diverse pool of analytic and random candidates
    (uniform, arcsine, Beta, power‑law, exponential, sparse, two‑block,
    Dirichlet).  The best few seeds for several lengths are kept.

2.  **Annealed sub‑gradient descent** – we smooth the max by a
    soft‑max (log‑sum‑exp) with a temperature α that is increased
    gradually.  The gradient of the smoothed objective is
    `2·a⋆w` where the weight vector w is the normalised soft‑max of the
    convolution values.  An exponentiated‑gradient (mirror descent on the
    simplex) update `a ← a·exp(-η·grad)` is performed and the vector is
    projected back onto the probability simplex.

    The step‑size η follows ηₜ = η₀/√t.  The annealing schedule
    (α from α₀ to α_max) turns the smooth problem into the original
    non‑smooth one, allowing large early steps and fine‑grained final
    adjustments.

3.  **Polishing** – a cheap greedy pairwise‑mass‑transfer walk that
    accepts any move that strictly improves the exact objective.

The whole search respects a global time budget (≈ 1000 s).  The best
sequence found (as a plain list of floats) is returned.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a via a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient 2·a⋆w for weight vector w (length 2·n‑1, sum(w)=1).
    Implemented via FFT cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)² (assumes sum(a)>0)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _initial_candidates_extended(n: int,
                                 rng: np.random.Generator) -> List[np.ndarray]:
    """Return a broad list of seed vectors of length n (all sum to 1)."""
    cand = []
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    for p in (0.05, 0.1, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0):
        cand.append(_distribution_beta_sym(n, p))

    for _ in range(5):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))

    for lam in (0.001, 0.005, 0.01, 0.02, 0.05):
        cand.append(_distribution_exponential_increasing(n, lam))

    for block_frac in (0.05, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac))

    # a few Dirichlet draws with random concentration
    for _ in range(3):
        cand.append(_distribution_dirichlet(n, rng,
                                            alpha=0.3 + 0.7 * rng.random()))

    cand.append(_distribution_random_sparse(n, rng, density=0.01))

    return cand


# ----------------------------------------------------------------------
def _annealed_subgradient_descent(a: np.ndarray,
                                  deadline: float,
                                  best_seq: np.ndarray,
                                  best_score: float,
                                  eta0: float = 0.7,
                                  alpha0: float = 0.5,
                                  alpha_max: float = 50.0,
                                  max_iter: int = 30000,
                                  eval_interval: int = 400) -> Tuple[np.ndarray, float]:
    """
    Annealed exponential‑gradient descent.
    – soft‑max temperature α grows from alpha0 -> alpha_max,
    – step size ηₜ = eta0/√t.
    Returns the best sequence seen.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        # current convolution
        conv = _fft_self_convolve(a)
        max_conv = conv.max()

        # annealed soft‑max weights over the 2n‑1 convolution values
        alpha = alpha0 + (alpha_max - alpha0) * (t / max_iter)
        # shift for numerical stability
        shifted = conv - max_conv
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            # fallback to a uniform active set (should not happen)
            w = np.ones_like(w) / w.size
        else:
            w = w / w_sum

        # sub‑gradient
        grad = _gradient_from_weights(a, w)

        # step‑size
        eta = eta0 / math.sqrt(t)

        # mirror‑descent (exponential) update
        a = a * np.exp(-eta * grad)

        # projection onto the simplex (keeps sum = 1)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # occasional exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)                # exact objective
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] annealed it={t:5d} α={alpha:.2f} "
                      f"new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Greedy mass‑transfer walk that keeps any strictly improving move."""
    n = a.shape[0]
    while time.time() < deadline:
        # pick a source index with non‑negligible mass
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        # destination index (different from i)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # amount to move (up to 30 % of source)
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} "
                  f"new best {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Runs for at most ``budget_s`` seconds and
    returns the best sequence found as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # 1.  Choose a handful of lengths (larger = better asymptotics)
    lengths = [4000, 8000, 16000, 32000]

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: List[Tuple[float, np.ndarray]] = []   # (score, vec)

    # ------------------------------------------------------------------
    # 2.  Scan analytic families for each length and keep the best few
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] --- scanning length {n} ---")

        for cand in _initial_candidates_extended(n, rng):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)  # safety
            cand /= cand.sum()            # enforce sum=1
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] **** new global best "
                      f"len={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))

        # keep a small slate of the very best seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:12]

    # ------------------------------------------------------------------
    # 3.  Incorporate a previously discovered sequence if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s == 0.0:
                prev = _distribution_uniform(prev.shape[0])
                s = prev.sum()
            prev = prev / s
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best improved: {sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4.  Fallback (should never trigger)
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Intensive annealed descent on the best seeds
    remaining = deadline - time.time()
    if remaining > 60:
        # pick a few top seeds for dedicated optimisation
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        for seed_seq in seeds:
            if time.time() >= deadline:
                break
            # give each seed roughly an equal fraction of the remaining time
            sub_deadline = time.time() + (deadline - time.time()) / max(1, len(seeds))
            seq_out, score_out = _annealed_subgradient_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.65,
                alpha0=0.4,
                alpha_max=80.0,
                max_iter=25000,
                eval_interval=300,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after annealed descent new best "
                      f"{overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Final polishing with cheap local swaps
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
