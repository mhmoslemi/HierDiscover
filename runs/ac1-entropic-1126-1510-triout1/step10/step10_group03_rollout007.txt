"""
Propose a non‑negative step function (list of floats) that minimises
`evaluate_sequence`.  The optimiser works in five stages:

1️⃣  Coarse exploration of several lengths (up to 64 k) using many
    analytic families (arcsine, symmetric / asymmetric Beta,
    edge‑biased Dirichlet).  The best candidate is kept.

2️⃣  Upsampling of the best candidate to a larger length (≈ 150 k)
    by linear interpolation – this gives a finer discretisation of the
    same shape and lets the subsequent optimisation achieve a smaller
    ∞‑norm of the self‑convolution.

3️⃣  Smooth‑max mirror descent with a *gradient* update (instead of the
    exponentiated‑gradient of the original code).  The soft‑max
    temperature is raised geometrically up to 1e5 while the learning
    rate decays, allowing a fast transition from exploration to a
    precise optimisation of the true max‑convolution.

4️⃣  Projected Adam‑style sub‑gradient descent on the *exact* max‑convolution
    (active‑set gradient) with an adaptive back‑tracking line‑search.

5️⃣  Targeted mass‑transfer hill‑climbing (plus occasional random jitter)
    to polish the final vector.

All operations respect the constraints of `evaluate_sequence`
(non‑negative entries, sum ≥ 0.01, each entry ≤ 1000).  The function
stops as soon as the 1000 s budget elapses and returns the best
vector found.

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Score = 2·n·max(a*a) for a already normalised (∑a = 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families (all return a vector with sum = 1)

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_edge_dirichlet(n: int, beta_exp: float) -> np.ndarray:
    """
    Dirichlet draw with concentration concentrated at the edges:
    c_i ∝ ((i+0.5)*(n-i-0.5))^{-beta_exp} + 0.1 .
    """
    idx = np.arange(n, dtype=np.float64)
    conc = ((idx + 0.5) * (n - idx - 0.5)) ** (-beta_exp) + 0.1
    vec = np.random.dirichlet(conc)
    return vec


def _interpolate_distribution(old: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = old.shape[0]
    old_x = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    new_x = (np.arange(new_len, dtype=np.float64) + 0.5) / new_len
    new = np.interp(new_x, old_x, old)
    new = np.maximum(new, 0.0)
    new /= new.sum()
    return new

        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t

        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # soft‑max weights (numerically stable)
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth‑max
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Euclidean projected gradient step
        a_new = a - lr * grad
        a_new = _project_onto_simplex(a_new, 1.0)
        a_new = np.maximum(a_new, eps)
        a_new /= a_new.sum()
        a = a_new

        # occasional reporting
        if (it % max(1, total_iters // 10) == 0) or (it == total_iters - 1):
            cur_score = _max_convolution_score(a)
            if cur_score < best_score - 1e-12:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time():.1f}] smooth‑max it={it}/{total_iters} new best {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Projected Adam‑style sub‑gradient descent on the *true* max‑conv
    a = best_seq.copy()
    velocity = np.zeros_like(a)
    beta1, beta2 = 0.85, 0.995
    step0 = 0.8
    sub_it = 0

    while time.time() < deadline:
        sub_it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        # active‑set gradient
        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam‑style momentum
        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** sub_it)

        step = step0
        while True:
            a_cand = a - step * v_hat
            a_cand = _project_onto_simplex(a_cand, 1.0)
            a_cand = np.maximum(a_cand, eps)
            a_cand /= a_cand.sum()
            cand_score = _max_convolution_score(a_cand)
            if cand_score < best_score - 1e-12 or step < 1e-8:
                break
            step *= 0.5

        a = a_cand
        if cand_score < best_score - 1e-12:
            best_score, best_seq = cand_score, a.copy()
            print(f"[{time.time():.1f}] subgrad it={sub_it} new best {cand_score:.6f}")

        # jitter every 800 iterations to escape plateaus
        if sub_it % 800 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    # ------------------------------------------------------------------
    # 6️⃣  Focused mass‑transfer hill‑climber
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    hill_it = 0

    while time.time() < deadline:
        hill_it += 1
        # index where convolution attains its maximum
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # find the pair (i,j) giving the largest product for this peak
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.argmin(a)
        if target == source:
            # avoid moving mass onto itself
            sorted_idx = np.argsort(a)
            target = sorted_idx[1]

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M
            best_seq = a.copy()
            best_score = 2.0 * n * M
            if hill_it % 100 == 0:
                print(f"[{time.time():.1f}] hill it={hill_it} new best {best_score:.6f}")

        # occasional random jitter to break possible stagnation
        if hill_it % 2500 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] hill jitter at it={hill_it}")

    final_score = 2.0 * n * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
