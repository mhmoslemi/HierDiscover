"""
Implements a global search for a step‑function minimizing
evaluate_sequence.  The algorithm scans analytic families,
keeps the best seeds, then runs an active‑set sub‑gradient
descent (multiplicative update) on a smooth approximation of
the max‑convolution objective, followed by a greedy pairwise
mass‑transfer polishing.
"""

import time, math, numpy as np
from scipy.signal import correlate

def _project_onto_simplex(v, s=1.0):
    (implementation as above)

def _fft_convolve(x, y):
    ...

def _fft_self_convolve(a):
    return _fft_convolve(a, a)

def _objective_from_conv(conv, n):
    return 2.0 * n * conv.max()

def _objective(a):
    conv = _fft_self_convolve(a)
    return _objective_from_conv(conv, a.shape[0])

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)

def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _distribution_beta_sym(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * (np.log(x) + np.log(1.0 - x))
    w = np.exp(logw)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _distribution_edge_weighted(n, p):
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _distribution_sparse_random(n, rng, density=0.01):
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals

def _mix_with_uniform(base, lam):
    n = base.shape[0]
    return lam * (1.0 / n) + (1.0 - lam) * base

# active‑set sub‑gradient descent
def _active_set_descent(a, deadline, best_seq, best_score, lr0=0.6, eps=1e-12, eval_interval=150):
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        # active mask (indices where conv is within eps of max)
        mask = conv >= max_val - eps
        w = mask.astype(np.float64)
        total_w = w.sum()
        if total_w == 0.0:
            w = np.ones_like(conv)  # fallback to uniform
            total_w = w.size
        # gradient = 2 * correlate(w, a, mode='valid') / total_w
        grad = 2.0 * correlate(w, a, mode='valid')
        grad = grad / total_w
        # multiplicative (exponentiated) update
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        # renormalise onto simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        if t % eval_interval == 0:
            cur_score = 2.0 * n * max_val
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] active descent it={t} new global best {best_score:.6f}")
        if t >= 200000:
            break
    return best_seq, best_score

def _pairwise_swap_local(a, deadline, best_seq, best_score, max_trials=None):
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * np.random.rand() * 0.4
        if delta < 1e-14:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # evaluate objective
        conv = _fft_self_convolve(a_new)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-14:
            a = a_new
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step‑function that minimises evaluate_sequence."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 4.0   # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates = []   # list of (score, array)

    # ------ 1. Grid scan of analytic families ----------
    lengths = [2000, 3000, 5000, 8000, 12000, 20000]
    p_grid = np.concatenate((
        np.logspace(-2, -0.5, 6, base=10),    # very small
        np.logspace(-0.5, 0.5, 12, base=10), # around 1
        np.logspace(0.5, 1.5, 12, base=10),  # up to ~30
    ))
    lambda_grid = np.linspace(0.0, 1.0, 7)   # more points

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # uniform and arcsine
        for base in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(base)
            top_candidates.append((sc, base.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = base.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # parameterised families
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            for base in (beta, edge):
                for lam in lambda_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # random Dirichlet seeds (8 per length)
        for _ in range(8):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 3.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] new global best random Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds (6 per length)
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = _distribution_sparse_random(n, rng, density=0.01)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}")

        # keep only top 40 overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:40]

    # ----- 2. Include previously discovered construction -----
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ----- 3. Ensure a fallback seed -----
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    remaining = deadline - time.time()

        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:8]]
        per_seed = remaining / max(1, len(seeds))

        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:

            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _active_set_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,

            )
            if sc_opt < overall_best_score:

                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] after active descent seed {idx} new global best {overall_best_score:.6f}")

    # ----- 5. Final greedy refinement -----

            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()

"""
Search for a non‑negative step‑function that minimizes
evaluate_sequence.  The algorithm proceeds in three stages:

1.  Scan a rich pool of analytic families (uniform, arcsine,
    symmetric β, U‑shaped “edge‑weighted”, mixtures with uniform,
    random Dirichlet and sparse vectors) for several lengths and keep
    the best few seeds.

2.  Refine each of the top seeds with an *active‑set* sub‑gradient
    descent.  Instead of using only a single maximal convolution
    index, the whole active set (all indices within a tiny tolerance
    of the maximum) is used, giving a more informative sub‑gradient.
    The update is multiplicative (exponentiated‑gradient) and the
    vector is re‑projected onto the probability simplex.

3.  Finish with a cheap greedy pairwise mass‑transfer walk that moves
    a small amount of probability between two positions whenever the
    objective improves.

The whole search respects the 1000 s budget and returns a plain
Python list which can be fed directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.signal import correlate

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution using a real FFT (fast for large vectors)."""
    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:n]


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * (np.log(x) + np.log(1.0 - x))
    # avoid overflow – work in log‑space
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sparse_random(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, U being the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Active‑set sub‑gradient descent ---------------------------------------
def _active_set_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.6,
    eps: float = 1e-12,
    eval_interval: int = 120,
) -> tuple[np.ndarray, float]:
    """Multiplicative (exponentiated‑gradient) descent using the whole
    active set of maximal convolution entries."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - eps          # active indices
        w = mask.astype(np.float64)
        total_w = w.sum()
        if total_w == 0.0:
            total_w = 1.0
            w = np.ones_like(conv)

        # gradient = 2 * correlate(w, a) / |active|
        grad = 2.0 * correlate(w, a, mode="valid") / total_w

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)           # exponentiated‑gradient step
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s                           # re‑project onto simplex

        if t % eval_interval == 0:
            # evaluate the true objective for the updated vector
            conv_new = _fft_self_convolve(a)
            cur_score = 2.0 * n * conv_new.max()
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] active descent it={t} new global best {best_score:.6f}"
                )
        if t >= 200000:      # safety guard
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise mass transfer -----------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Move a random amount of mass from i to j if the objective improves."""
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur_score = _objective(new_a)
        if cur_score < best_score - 1e-14:
            a = new_a
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score
