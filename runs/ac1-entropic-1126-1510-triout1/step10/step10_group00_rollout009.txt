def _smooth_mirror_descent(a0, deadline, best_seq, best_score, lr0=0.3, T0=0.2, T_decay=0.9995, eval_interval=200):
    a = a0.copy()
    t = 0
    T = T0
    while time.time() < deadline:
        t += 1
        # compute convolution a*a
        conv = _fft_self_convolve(a)
        # compute smoothed gradient weights
        maxc = conv.max()
        # Temperature schedule maybe gradually decreasing: T = T0 * (T_decay ** t)
        T = max(1e-6, T0 * (T_decay ** t))
        w = np.exp((conv - maxc) / T)  # shape 2n-1
        # compute gradient g = 2 * w * a contributions
        a_rev = a[::-1]
        conv_wa_rev = _fft_convolve(w, a_rev)  # length 3n-2
        g = conv_wa_rev[:len(a)][::-1] * 2.0
        # update multiplicative
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(- eta * g)
        # renorm to sum=1
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / len(a))
        else:
            a = a / s
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth it={t} T={T:.6g} new best {best_score:.6f}")
        if t >= 500000: # safety
            break
    return best_seq, best_score

"""
Propose a step‑function that minimizes ``evaluate_sequence``.

Algorithm
---------
1.  Generate a diverse pool of analytic distributions (uniform,
    arcsine, symmetric beta, “edge‑weighted’’ U‑shapes and mixtures with
    uniform) for several lengths.  Keep the best few candidates.
2.  Refine each candidate with a *smooth* mirror‑descent that minimises
    a temperature‑smoothed version of the max‑convolution objective.
    The temperature is tied to the current maximal convolution value,
    which gives a gradually tighter focus on the true max.
    The update is multiplicative (exponentiated‑gradient) and the vector
    is renormalised to the simplex after every step.
3.  Finish with a cheap greedy pairwise mass‑transfer walk that accepts
    any move decreasing the exact objective.
4.  Return the best sequence found (as a plain Python list).

The smooth descent uses two FFT‑based convolutions per iteration:
   • a*a   – the current self‑convolution,
   • w*aᵣ – the weighted gradient, where w are soft‑max
            weights derived from the smoothed max and aᵣ is a reversed copy
            of a.  Both convolutions run in O(n log n) time, allowing many
            thousands of iterations within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full (real) convolution of two vectors using FFT."""
    n = len(x) + len(y) - 1
    size = 1 << ((n) - 1).bit_length()
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:n]


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
def _smooth_mirror_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.4,
    eval_interval: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on a temperature‑smoothed max‑convolution objective.
    The temperature is set to a small fraction of the current maximal
    convolution value, yielding a softmax over all convolution entries.
    """
    a = a0.copy()
    t = 0
    n = len(a)

    while time.time() < deadline:
        t += 1

        # 1) current self‑convolution
        conv = _fft_self_convolve(a)              # length 2n‑1
        maxc = conv.max()
        # temperature proportional to the scale of the convolution
        T = maxc * 0.12 + 1e-12                    # 12 % of the peak, never zero
        # soft‑max weights (higher weight → larger contribution to gradient)
        w = np.exp((conv - maxc) / T)             # still length 2n‑1, max weight = 1

        # 2) weighted gradient  g_s = 2 * Σ_i w_i a_{i‑s}
        a_rev = a[::-1]                           # reversed a
        conv_wa_rev = _fft_convolve(w, a_rev)    # length 3n‑2
        g = conv_wa_rev[:n][::-1] * 2.0           # keep only the first n and flip

        # 3) multiplicative (exponentiated‑gradient) step
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * g)

        # 4) re‑normalise to the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] smooth it={t:5d} T={T:.3e} new best {best_score:.6f}"
                )

        # safety stop
        if t >= 300000:
            break

    return best_seq, best_score

def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small amount of mass; keep move only if the objective improves."""
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        i = np.random.choice(src)

        j = np.random.randint(n - 1)

            j += 1

            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score
