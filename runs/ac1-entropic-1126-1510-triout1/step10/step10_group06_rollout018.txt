def refine_by_lp(a, M=None):
    n = len(a)
    if M is None:
        conv = fft_self_convolve(a) # shape (2n-1)
        M = conv.max()
    # Build A_ub sparse matrix for constraints a * g <= M
    rows = []
    cols = []
    data = []
    for i in range(2*n-1):
        # j range for which a[i-j] defined
        j_min = max(0, i-(n-1))
        j_max = min(n-1, i)
        # For each j in range, weight = a[i-j]
        # We'll vectorize: generate indices arrays
        # Could use loops.
        for j in range(j_min, j_max+1):
            rows.append(i)
            cols.append(j)
            data.append(a[i - j])
    A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

    # objective maximize sum(g) => minimize -sum
    c = -np.ones(n)
    b_ub = M * np.ones(2*n-1)
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        g = res.x
    else:
        g = a # fallback

    # Normalize g to sum 1
    if g.sum() > 0:
        g = g / g.sum()
    else:
        g = a
    return g

            if sc < best_score:

    ...

    # Stage 2: refinement for top candidates (maybe best_seq only)
    # We'll do SA on best_seq, also try a couple of random restarts

    def simulated_annealing(a):
        n = len(a)
        a = np.array(a, copy=True)
        current_score = evaluate(a)
        best_local = a.copy()
        best_local_score = current_score
        # initial temperature: maybe (current_score - 1) * 0.2
        T0 = max(0.1, (current_score - 1.0) * 0.2)
        T = T0
        iter = 0
        while time.time() < deadline - 0.1:
            # choose move
            if random.random() < 0.7:  # gradient-guided
                conv = _fft_self_convolve(a)
                w = _subgradient_weights(conv, eps_rel=1e-12)
                grad = 2.0 * _fft_convolve(a[::-1], w)[n-1 : n-1+n]
                src = int(np.argmax(grad))
                dst = int(np.argmin(grad))
            else:
                src = random.randrange(n)
                dst = random.randrange(n)
                while dst == src:
                    dst = random.randrange(n)
            # decide delta fraction
            max_delta = min(a[src]*0.5, 0.2)  # up to 20% of src mass
            delta = max_delta * random.random()
            if delta < 1e-12:
                continue
            a_new = a.copy()
            a_new[src] -= delta
            a_new[dst] += delta
            a_new = np.maximum(a_new, 1e-12)
            a_new /= a_new.sum()
            new_score = evaluate(a_new)
            # acceptance
            if new_score < current_score or random.random() < math.exp(-(new_score - current_score) / T):
                a = a_new
                current_score = new_score
                if new_score < best_local_score:
                    best_local = a.copy()
                    best_local_score = new_score
            # cooling
            iter += 1
            # update temperature schedule after some steps
            # simple exponential cooling
            if iter % 100 == 0:
                T *= 0.99
        return best_local, best_local_score

    # apply SA to the current best
    best_seq, best_score = simulated_annealing(best_seq)

    # Multiple restarts:

Search for a non‑negative step‑function minimising `evaluate_sequence`.

The algorithm proceeds in several phases:

1. Scan a variety of analytic families (uniform, arcsine, beta, exponential,
   sine‑power, Gaussian) and random Dirichlet draws for several lengths.
2. If a previously‑found global vector (`height_sequence_1`) exists, it is
   incorporated.
3. A cheap Frank‑Wolfe style refinement is performed on a moderate size
   (≤2500) using a linear programme that maximises the ℓ₁‑norm of a vector
   under the current convolution bound.
4. The best candidate is then run through a simulated‑annealing hill‑climber
   that repeatedly moves a small chunk of mass from a high‑gradient index to
   a low‑gradient one.  The temperature is slowly decreased, allowing the
   search to escape shallow local minima.
5. While time permits, a handful of fresh random restarts are optimised
   with the same annealer.

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _fft_self_convolve(a):
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_convolve(a, b):
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.01):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta(n, p, q):
    """Beta(p,q) on a regular grid (non‑symmetric)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1 - x))
    if w.sum() == 0:

    else:
        w /= w.sum()
    return w

    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _smooth_weights(conv, alpha):
    """Return a probability vector approximating the smooth max."""
    maxc = conv.max()
    w = np.exp(alpha * (conv - maxc))
    s = w.sum()
    if s == 0:
        return np.full_like(w, 1.0 / w.size)
    return w / s


def _subgradient_weights(conv, eps_rel=1e-12):
    """Weights for a (sub)gradient of the max‑convolution."""
    m = conv.max()
    w = np.zeros_like(conv)
    w[conv >= m - eps_rel * (abs(m) + 1.0)] = 1.0
    cnt = w.sum()
    if cnt == 0:
        return w
    return w / cnt


def _exponential_gradient_step(a, lr, alpha):
    """One EG step (a ≥ 0, Σa = 1)."""
    conv = _fft_self_convolve(a)
    w = _smooth_weights(conv, alpha)
    grad = 2.0 * _fft_convolve(a[::-1], w)[len(a) - 1 : len(a) - 1 + len(a)]
    a = a * np.exp(-lr * grad)
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def _projected_gradient_step(a, eta, active_w):
    """PGD step on the sub‑gradient defined by active_w."""
    grad = 2.0 * _fft_convolve(a[::-1], active_w)[len(a) - 1 : len(a) - 1 + len(a)]
    a = a - eta * grad
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def _adam_step(logits, m, v, t, step_size):
    """Adam update on logits (unconstrained)."""
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = beta1 * m + (1.0 - beta1) * step_size
    v = beta2 * v + (1.0 - beta2) * (step_size ** 2)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    lr = 0.02 / (np.sqrt(v_hat) + eps)
    logits = logits - lr * m_hat
    logits = np.clip(logits, -30, 30)
    return logits, m, v


def _hillclimber_move(a, conv, eps=1e-12):
    """Move a tiny chunk from a large‑gradient index to a small‑gradient one."""
    n = a.shape[0]
    w = _subgradient_weights(conv, eps_rel=eps)
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    src = int(np.argmax(grad))
    dst = int(np.argmin(grad))
    if src == dst:
        dst = random.randint(0, n - 1)
    max_delta = min(a[src] * 0.5, 0.02)
    delta = max_delta * random.random()
    if delta < eps:
        return a, conv, False
    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    new_conv = _fft_self_convolve(a_new)
    improved = new_conv.max() < conv.max() - 1e-13
    if improved:
        return a_new, new_conv, True
    else:
        return a, conv, False

def _solve_lp_maxsum(a, M):
    """
    Solve the LP  max Σ g_j   s.t.   a * g ≤ M   and   g ≥ 0.

    Returns the optimal g (may be all‑zeros if LP fails).
    """
    n = a.shape[0]
    rows = []
    cols = []
    data = []
    # each diagonal offset d ∈ [0, n‑1] carries the weight a[d]
    for d in range(n):
        i = np.arange(d, d + n)          # row indices for this diagonal
        j = np.arange(n)                 # column indices
        rows.append(i)
        cols.append(j)
        data.append(np.full(n, a[d]))
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)
    A_ub = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)                      # maximise Σg ⇔ minimise –Σg
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method="highs")
    if res.success:
        return res.x
    else:
        return a.copy()


def _frank_wolfe_refine(a, max_iters=5, t_mix=0.25):
    """
    Frank‑Wolfe style refinement using the LP above.
    Repeatedly mixes the current vector with the LP solution.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    for _ in range(max_iters):
        conv = _fft_self_convolve(a)
        M = conv.max()
        g = _solve_lp_maxsum(a, M)
        sum_g = g.sum()
        if sum_g <= 1 + 1e-12:
            break
        g_norm = g / sum_g
        a = (1 - t_mix) * a + t_mix * g_norm
        a = np.maximum(a, 1e-12)
        a /= a.sum()
    return a


def _simulated_annealing(a, deadline):
    """
    Temperature‑driven hill‑climber.  Moves mass from a high‑gradient index to
    a low‑gradient one, accepting occasional uphill steps.
    """
    n = a.shape[0]
    eps = 1e-12
    a = np.asarray(a, dtype=np.float64).copy()
    cur_score = _max_convolution_score(a)
    best_a, best_score = a.copy(), cur_score
    # initial temperature (scale with how far we are from the theoretical lower bound)
    T = max(0.1, (cur_score - 1.0) * 0.2)
    it = 0
    while time.time() < deadline - 0.1:
        # decide whether to use a gradient‑directed move or a random one
        if random.random() < 0.7:
            conv = _fft_self_convolve(a)
            w = _subgradient_weights(conv, eps_rel=1e-12)
            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
            src = int(np.argmax(grad))
            dst = int(np.argmin(grad))
        else:
            src = random.randrange(n)
            dst = random.randrange(n)
            while dst == src:
                dst = random.randrange(n)

        max_delta = min(a[src] * 0.5, 0.1)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        a_new = np.maximum(a_new, eps)
        a_new /= a_new.sum()

        new_score = _max_convolution_score(a_new)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if random.random() < prob:
                accept = True

        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_a, best_score = a_new.copy(), new_score

        # cooling schedule
        it += 1
        if it % 250 == 0:
            T *= 0.95
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns the best sequence found within the allotted time.
    """
    random.seed(seed)
    np.random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5  # spare margin for clean exit
    eps = 1e-12

    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1. Scan analytic families and random Dirichlet draws
    scan_lengths = [1500, 2000, 2500, 3000, 3500, 4000, 5000]
    for n in scan_lengths:
        if time.time() >= deadline:
            break
        cand = []

        # analytic families
        cand.append(_distribution_uniform(n))
        cand.append(_distribution_arcsine(n))
        cand.append(_distribution_triangular(n))
        cand.append(_distribution_exponential(n, lam=0.005))
        cand.append(_distribution_exponential(n, lam=0.02))

        for p in (0.3, 0.6, 1.0, 2.0, 4.0):
            cand.append(_distribution_beta_sym(n, p))
        for pw in (0.5, 1.0, 2.0, 4.0):
            cand.append(_distribution_sine_power(n, pw))
        for sigma in (0.2, 0.35, 0.5):
            cand.append(_distribution_gaussian(n, sigma))

        # a few asymmetric beta shapes
        for p, q in ((0.3, 0.6), (0.6, 0.3), (0.5, 2.0), (2.0, 0.5)):
            cand.append(_distribution_beta(n, p, q))

        # random Dirichlet draws
        for alpha in (0.2, 0.5, 1.0, 2.0):
            cand.append(np.random.dirichlet(np.full(n, alpha)))

        for vec in cand:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq, best_n, best_score = vec.copy(), n, sc
                print(f"[{time.time():.1f}] init n={n}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Incorporate a previously stored global solution if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_seq, best_n, best_score = prev.copy(), prev.size, sc
                print(f"[{time.time():.1f}] global best {sc:.6f}")
    except Exception:
        pass

    if best_seq is None:
        best_n = 2000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)

    # ------------------------------------------------------------------
    # 3. Frank‑Wolfe / LP refinement on a moderate size (≤2500)
    if best_seq.shape[0] > 2500:
        # down‑sample to 2500 for the LP stage, preserving the shape by linear interpolation
        n_lp = 2500
        x_old = np.linspace(0.0, 1.0, best_seq.size)
        x_new = np.linspace(0.0, 1.0, n_lp)
        a_lp = np.interp(x_new, x_old, best_seq)
    else:
        n_lp = best_seq.shape[0]
        a_lp = best_seq.copy()
    a_lp = np.maximum(a_lp, eps)
    a_lp /= a_lp.sum()
    a_lp = _frank_wolfe_refine(a_lp, max_iters=6, t_mix=0.25)
    # up‑sample back to the original length if we had down‑sampled
    if n_lp != best_seq.shape[0]:
        x_old = np.linspace(0.0, 1.0, n_lp)
        x_new = np.linspace(0.0, 1.0, best_seq.shape[0])
        a_up = np.interp(x_new, x_old, a_lp)
        a_up = np.maximum(a_up, eps)
        a_up /= a_up.sum()
        a_lp = a_up
    sc_lp = _max_convolution_score(a_lp)
    if sc_lp < best_score:
        best_seq, best_n, best_score = a_lp.copy(), a_lp.shape[0], sc_lp
        print(f"[{time.time():.1f}] after LP   n={best_n}  score={sc_lp:.6f}")

    # ------------------------------------------------------------------
    # 4. Simulated‑annealing fine‑tuning on the current best
    best_seq, best_score = _simulated_annealing(best_seq, deadline)

    # ------------------------------------------------------------------
    # 5. Additional random restarts (short SA runs) while time permits
    restart_lengths = [1500, 2000, 2500, 3000, 3500, 4000]
    while time.time() < deadline - 5.0:
        n = random.choice(restart_lengths)
        a0 = np.random.dirichlet(np.full(n, 0.5))
        a0 = np.maximum(a0, eps)
        a0 /= a0.sum()
        a_opt, sc_opt = _simulated_annealing(a0, deadline)
        if sc_opt < best_score:
            best_seq, best_n, best_score = a_opt, n, sc_opt
            print(f"[{time.time():.1f}] restart n={n}  new best {sc_opt:.6f}")

    # ------------------------------------------------------------------
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
