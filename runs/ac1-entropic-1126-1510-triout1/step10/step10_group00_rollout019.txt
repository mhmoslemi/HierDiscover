def _distribution_geometric(n, c): # param c = n*(1-r)
    # r = 1 - c / n
    r = 1 - c/n
    # ensure r >0
    if r <= 0:
        r = 0.0
    # compute raw weights as r**np.arange(n)
    # Then normalize to sum 1.
    # To avoid underflow for large n and r<1, we can compute with exponentials.
    i = np.arange(n, dtype=np.float64)
    if r <= 0:
        w = np.zeros(n)
        w[0] = 1.0
    else:
        # using log: log w_i = i * log(r)
        logw = i * np.log(r)
        # offset to prevent underflow: subtract max
        maxlog = np.max(logw)
        w = np.exp(logw - maxlog)
        w /= w.sum()
    return w

"""
propose_candidate
=================
Search for a non‑negative step‑function (a list of floats) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

where conv is the full discrete convolution (np.convolve).  The
objective is homogeneous, so we restrict the search to the simplex
∑a_i = 1.  The algorithm works in three phases

1. **Exploratory scan** – generate candidates from several analytic
   families (uniform, arcsine, symmetric beta, power‑law,
   geometric‑decay, and mixtures with the uniform distribution).  The
   most promising 30 candidates are kept.

2. **Mirror‑descent refinement** – a few top seeds are improved with an
   exponentiated‑gradient (mirror) descent on the exact max‑convolution
   objective.  The simplex constraint is enforced by Euclidean projection.

3. **Local polishing** – a cheap random “swap” walk moves a small amount
   of mass between two indices whenever it reduces the objective.

The whole routine respects a user‑supplied time budget (default 1000 s)
and returns the best vector found as a plain Python list.  The code
contains no closures, no lambda expressions and only uses the standard
library together with ``numpy``.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (fast for large n)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2).  The vector
    is assumed non‑negative; no clipping is performed here.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max() / (a.sum() ** 2)

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        return np.full_like(w, 1.0 / n)
    return w / w.sum()


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        return np.full_like(w, 1.0 / n)
    return w / w.sum()


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    """Monotone decreasing: w_i ∝ (i+1)^(−p)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = 1.0 / (i ** p)
    return w / w.sum()


def _distribution_geometric(n: int, c: float) -> np.ndarray:
    """
    Geometric decay a_i ∝ r**i with r = 1 - c/n  (c>0).
    The distribution is normalised to sum 1.
    """
    r = 1.0 - c / n
    if r <= 0.0:
        # degenerate case – all mass at the first cell
        w = np.zeros(n, dtype=np.float64)
        w[0] = 1.0
        return w
    i = np.arange(n, dtype=np.float64)
    # work in log‑space for stability
    logw = i * np.log(r)
    logw -= np.max(logw)               # shift to avoid overflow
    w = np.exp(logw)
    return w / w.sum()


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Mirror‑descent on the true max‑convolution objective ---------------
def _subgradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.6,
    eval_interval: int = 150,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror) descent on the exact objective."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        idx = int(conv.argmax())               # index attaining the max
        # gradient of the selected entry w.r.t. a
        ind = idx - np.arange(n)
        mask = (ind >= 0) & (ind < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[ind[mask]]         # ∂/∂a_i of a_i·a_{idx-i}
        eta = lr0 / math.sqrt(t)               # diminishing stepsize
        a = a * np.exp(-eta * grad)            # mirror step (exponential)
        # re‑normalise to the simplex
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad it={t} new best {best_score:.6f}")

        if t >= 200000:               # safety guard
            break
    return best_seq, best_score

# Simple greedy local improvement --------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly transfer a small chunk of mass; keep it if the objective improves."""
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")

    return best_seq, best_score

# Helper used only inside the scanning loop – kept at top level as required.
def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    return vals / vals.sum()
