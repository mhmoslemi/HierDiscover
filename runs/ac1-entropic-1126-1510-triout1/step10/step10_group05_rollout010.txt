"""
Propose a step‑function (non‑negative vector) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

under the simple‐x≤1000, x≥0 preprocessing of ``evaluate_sequence``.
The algorithm consists of three stages:

1. **Broad seeding** – generate many analytic candidate shapes
   (arcsine, triangular, exponential, Gaussian, Cauchy, symmetric / asymmetric
   Beta, power‑transforms, Dirichlet draws) on a handful of lengths.
   Each candidate is ℓ₁‑normalised (∑a=1) and scored by the exact surrogate
   ``_max_convolution_score`` (identical to the true value when ∑a=1).

2. **Intensive refinement** – start from the best seed and optimise it on the
   simplex with three complementary first‑order methods:
   * exponentiated‑gradient (mirror descent on the simplex),
   * Adam,
   * projected SGD.
   The budget is split automatically; each method runs until the
   remaining time is exhausted.

3. **Local permutation / hill‑climbing** – on the remaining time
   repeatedly perform cheap random moves:
   – swap two entries,
   – transfer a small random chunk of mass between two positions,
   – occasional jitter.
   Only moves that improve the surrogate are kept.
   This exploits the fact that the ordering of the same multiset of
   heights strongly influences the convolution peak.

All operations respect the time budget (≈ 1000 s).  The returned list is
ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolutions -----------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
# Exact surrogate (for ∑a = 1) ----------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """
    Exact objective when the vector is ℓ₁‑normalised.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution functional -----------------------
def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of the max‑convolution functional.
    Returns (grad, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with the weight mask
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # correct diagonal terms for even active indices (i=j)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _gaussian_distribution(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    w /= w.sum()
    return w


def _cauchy_distribution(n: int, scale_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    scale = scale_factor * n
    w = 1.0 / (1.0 + ((idx - centre) / scale) ** 2)
    w /= w.sum()
    return w


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0) – clip safely
    x = np.clip(x, 1e-15, 1 - 1e-15)
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Unsymmetric Beta(α,β) on a discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    x = np.clip(x, 1e-15, 1 - 1e-15)
    with np.errstate(divide='ignore', invalid='ignore'):
        w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    if np.isnan(w).any() or np.isinf(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex from Dirichlet(α,…,α)."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Ordering heuristics ---------------------------------------------------
def order_alternate_extremes(desc_vals: np.ndarray) -> np.ndarray:
    """Spread large values alternately to the left‑most and right‑most slots."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    left, right = 0, n - 1
    toggle = True
    for v in desc_vals:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    return out


def order_center_outwards(desc_vals: np.ndarray) -> np.ndarray:
    """Place biggest value at centre then fill left/right alternately."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    positions = []
    mid = (n - 1) // 2
    positions.append(mid)
    for d in range(1, n):
        l = mid - d
        if l >= 0:
            positions.append(l)
        r = mid + d
        if r < n:
            positions.append(r)
    for i, pos in enumerate(positions[:n]):
        out[pos] = desc_vals[i]
    return out


def order_binary_tree(desc_vals: np.ndarray) -> np.ndarray:
    """Recursive midpoint filling – spreads large values evenly."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    positions = []

    def recurse(l: int, r: int):
        if l > r:
            return
        m = (l + r) // 2
        positions.append(m)
        recurse(l, m - 1)
        recurse(m + 1, r)

    recurse(0, n - 1)
    out[positions] = desc_vals
    return out

def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 2_000_000,
                          eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """Mirrored (exponentiated) gradient descent on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / a_sum
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-14:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 600_000,
                  step0: float = 0.2,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam optimiser on the simplex (projected after every step)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-14:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


def _sgd_subgrad(a0: np.ndarray, deadline: float,
                 max_iters: int = 500_000,
                 step0: float = 0.5) -> tuple[np.ndarray, float]:
    """Plain projected sub‑gradient descent."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        step = step0 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-14:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Local permutation / hill‑climbing ------------------------------------
def _perm_hill_climb(a_start: np.ndarray, deadline: float,
                     max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Random swaps, mass‑transfers and tiny jitter, keeping improvements."""
    a = a_start.copy()
    best_score = _max_convolution_score(a)
    n = a.size
    it = 0
    # decide a schedule for the transfer magnitude
    while time.time() < deadline and it < max_iters:
        it += 1
        if np.random.rand() < 0.5:
            # ----- swap two entries -------------------------------------------------
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # ----- transfer a small random chunk of mass ----------------------------
            src = np.random.randint(n)
            dst = np.random.randint(n)
            if src == dst or a[src] < 1e-12:
                continue
            # linearly decay the transfer amount as we progress
            decay = 1.0 - (time.time() - (deadline - (deadline - time.time()))) / (deadline - time.time() + 1e-12)
            eps = 0.02 * max(0.1, decay)          # slowly shrink
            delta = min(a[src] * eps, 1.0 - a[dst])
            if delta <= 0.0:
                continue
            cand = a.copy()
            cand[src] -= delta
            cand[dst] += delta
        sc = _max_convolution_score(cand)
        if sc < best_score - 1e-14:
            a = cand
            best_score = sc
        # occasional jitter to escape plateaus
        if it % 4000 == 0:
            jitter = np.random.randn(n) * 1e-6
            cand = _project_onto_simplex(a + jitter, s=1.0)
            sc = _max_convolution_score(cand)
            if sc < best_score:
                a = cand
                best_score = sc
    return a, best_score

# Main driver -----------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the given time budget.
    Returns the best sequence found (as a Python list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Pick a handful of lengths to explore (moderate, to keep FFT cheap)
    # The list deliberately spans small to relatively large sizes.
    lengths = [800, 1600, 3000, 5000, 8000, 12000]
    lengths = [n for n in lengths if 500 <= n <= 12000]

    # ------------------------------------------------------------------
    # Candidate storage (vector, surrogate score)
    candidates: list[tuple[np.ndarray, float]] = []

    def _add_candidate(arr: np.ndarray):
        a = np.asarray(arr, dtype=np.float64)
        if a.ndim != 1 or a.size == 0:
            return
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            return
        a = a / s
        sc = _max_convolution_score(a)
        candidates.append((a, sc))

    # ------------------------------------------------------------------
    # 1. Broad sampling -------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # ---- basic analytic families ------------------------------------
        # arcsine + reorderings
        a = _arcsine_distribution(n)
        _add_candidate(a)
        desc = np.sort(a)[::-1]
        _add_candidate(order_alternate_extremes(desc))
        _add_candidate(order_center_outwards(desc))
        _add_candidate(order_binary_tree(desc))

        # triangular
        a = _triangular_distribution(n)
        _add_candidate(a)
        desc = np.sort(a)[::-1]
        _add_candidate(order_alternate_extremes(desc))
        _add_candidate(order_center_outwards(desc))
        _add_candidate(order_binary_tree(desc))

        # exponential
        a = _exponential_distribution(n, lam=0.008)
        _add_candidate(a)
        desc = np.sort(a)[::-1]
        _add_candidate(order_alternate_extremes(desc))
        _add_candidate(order_center_outwards(desc))
        _add_candidate(order_binary_tree(desc))

        # Gaussian
        a = _gaussian_distribution(n, sigma_factor=0.18)
        _add_candidate(a)
        desc = np.sort(a)[::-1]
        _add_candidate(order_alternate_extremes(desc))
        _add_candidate(order_center_outwards(desc))
        _add_candidate(order_binary_tree(desc))

        # Cauchy
        a = _cauchy_distribution(n, scale_factor=0.18)
        _add_candidate(a)
        desc = np.sort(a)[::-1]
        _add_candidate(order_alternate_extremes(desc))
        _add_candidate(order_center_outwards(desc))
        _add_candidate(order_binary_tree(desc))

        # ---- symmetric Beta family (dense grid around 0.5) -----------------
        for p in np.arange(0.4, 1.61, 0.05):
            if time.time() >= deadline:
                break
            a = _beta_symmetric_distribution(n, p)
            _add_candidate(a)

        # ---- unsymmetric Beta (coarse grid) -------------------------------
        beta_params = [0.3, 0.6, 1.0, 1.5, 2.5, 4.0]
        for alpha in beta_params:
            for beta in beta_params:
                if time.time() >= deadline:
                    break
                a = _beta_distribution(n, alpha, beta)
                _add_candidate(a)

        # ---- power‑transformed arcsine ------------------------------------
        base = _arcsine_distribution(n)
        for pwr in np.arange(0.8, 1.31, 0.05):
            a = np.power(base, pwr)
            a /= a.sum()
            _add_candidate(a)

        # ---- Dirichlet draws (different concentration) -------------------
        for alpha in [0.2, 0.5, 1.0, 2.0, 5.0]:
            a = _random_dirichlet_alpha(n, alpha)
            _add_candidate(a)

        # ---- a few random permutations of the best arcsine shape ----------
        for _ in range(3):
            perm = np.random.permutation(a)
            _add_candidate(perm)

    # ------------------------------------------------------------------
    # 2. Use previous best if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        if prev.sum() > 0:
            prev = prev / prev.sum()
            _add_candidate(prev)
            print(f"Added previous best candidate, surrogate={candidates[-1][1]:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Keep a modest number of top seeds (they will be refined intensively)
    candidates.sort(key=lambda x: x[1])
    top_k = min(8, len(candidates))
    top_candidates = candidates[:top_k]
    print(f"Generated {len(candidates)} candidates, keeping top {top_k}.")
    print(f"Best seed surrogate = {top_candidates[0][1]:.6f}")

    # ------------------------------------------------------------------
    # 3. Intensive refinement – start from the global best seed
    best_seq = top_candidates[0][0].copy()
    best_score = top_candidates[0][1]

    # allocate sub‑deadlines for the three descent phases (respect remaining time)
    remaining = deadline - time.time()
    # 40 % for exponentiated gradient, 30 % for Adam, 30 % for SGD
    t_exp = time.time() + 0.40 * remaining
    t_adam = time.time() + 0.30 * remaining
    t_sgd = time.time() + 0.30 * remaining

    # -------- exponentiated gradient ------------------------------------
    if time.time() < deadline:
        seq, sc = _exp_gradient_descent(best_seq, t_exp, max_iters=5_000_000, eta0=0.85)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"Exp‑grad improvement → {best_score:.6f}")

    # -------- Adam -------------------------------------------------------
    if time.time() < deadline:
        seq, sc = _adam_descent(best_seq, t_adam, max_iters=800_000, step0=0.22)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"Adam improvement → {best_score:.6f}")

    # -------- SGD --------------------------------------------------------
    if time.time() < deadline:
        seq, sc = _sgd_subgrad(best_seq, t_sgd, max_iters=600_000, step0=0.55)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"SGD improvement → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Local permutation / hill‑climbing phase (uses the remaining budget)
    if time.time() < deadline - 5.0:
        seq, sc = _perm_hill_climb(best_seq, deadline, max_iters=800_000)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"Permutation hill‑climb → {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final surrogate score: {best_score:.6f} (length = {len(best_seq)})")
    return best_seq.tolist()
