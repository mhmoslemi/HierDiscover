def solve_lp_max_sum(a, M):
    import scipy.sparse as sp
    from scipy.optimize import linprog
    n = a.shape[0]
    # construct T as sparse diag matrix
    diags = [np.full(n, a[i]) for i in range(n)]  # each diag length n
    offsets = np.arange(n)
    T = sp.diags(diags, offsets, shape=(2*n-1, n), format='csc')
    c = -np.ones(n)  # maximize sum = minimize -sum
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    # Use high quality method 'highs'
    res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        b = res.x
        b = np.maximum(b, 0)
        b /= b.sum()
        return b
    else:
        return None

   repeatedly identify the convolution index attaining the current
   maximum, compute the sub‑gradient of this max‑entry (which is
   ``2·a_{k‑i}``), take a step in the opposite direction and project
   back onto the probability simplex.  A simple back‑tracking line search
   guarantees a non‑increasing max‑convolution value.

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """
    Fast exact evaluation of 2·n·max(a*a) for a probability vector (sum = 1).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * float(conv.max())

# Analytic families – each returns a *probability* vector (sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.01) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * (np.log(x) + np.log(1 - x))
    w = np.exp(logw)
    if w.sum() == 0 or np.any(np.isinf(w)):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals /= vals.sum()
    return vals


def _distribution_power_edge(n: int, alpha: float) -> np.ndarray:
    """Edge‑heavy family: w_i ∝ ((i+0.5)*(n-i-0.5))^{-α}."""
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - i - 0.5)
    eps = 1e-12
    vals = np.power(denom + eps, -alpha)
    vals /= vals.sum()
    return vals


def _distribution_V_shape(n: int) -> np.ndarray:
    """Simple V‑shape: linear increase from centre to the edges."""
    centre = (n - 1) / 2.0
    vals = np.abs(np.arange(n, dtype=np.float64) - centre)
    eps = 1e-12
    vals = np.maximum(vals, eps)
    vals /= vals.sum()
    return vals


def _distribution_power_abs(n: int, beta: float) -> np.ndarray:
    """Symmetric power law: |i‑centre|^{β}."""
    centre = (n - 1) / 2.0
    vals = np.abs(np.arange(n, dtype=np.float64) - centre) ** beta
    eps = 1e-12
    vals = np.maximum(vals, eps)
    vals /= vals.sum()
    return vals


def _random_symmetric_dirichlet(n: int, alpha: float) -> np.ndarray:
    """Draw a symmetric vector from Dirichlet(α) on the half‑grid and mirror it."""
    half = (n + 1) // 2
    half_vec = np.random.dirichlet(np.full(half, alpha))
    if n % 2 == 0:
        full = np.concatenate([half_vec, half_vec[::-1]])
    else:
        full = np.concatenate([half_vec, half_vec[-2::-1]])
    full /= full.sum()
    return full

                b /= b.sum()
                # mix a and b (convex combination) and project back
                t_mix = 0.5
                a = _project_onto_simplex((1 - t_mix) * a + t_mix * b, 1.0)
                conv = _fft_self_convolve(a)
                maxc = conv.max()
                sc = 2.0 * n * maxc
                if sc < best_score - 1e-12:
                    best_seq = a.copy()
                    best_score = sc
                    print(f"[{time.time():.1f}] LP refinement  best={sc:.6f}")
        except Exception as e:
            # SciPy not available or LP failed – simply skip
            pass

    # ------------------------------------------------------------------
    # 5.  Fast hill‑climber with O(n) convolution updates
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    hill_iters = 0

    while time.time() < deadline:
        hill_iters += 1

        # index achieving the current maximum
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        prod = a[idx] * a[k - idx]
        max_pos = np.argmax(prod)
        i = idx[max_pos]
        j = k - i

        source = i if a[i] >= a[j] else j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        # limit the transferred mass (keep a[source] ≥ eps)
        max_delta = min(a[source] * 0.5, 0.015)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        # --- incremental convolution update --------------------------------
        # store current a for the slice updates
        a_src = a[source]
        a_tgt = a[target]

        # subtract contribution of removed mass
        lo_src = source
        hi_src = source + n
        conv[lo_src:hi_src] -= 2.0 * delta * a

        # add contribution of added mass
        lo_tgt = target
        hi_tgt = target + n
        conv[lo_tgt:hi_tgt] += 2.0 * delta * a

        # quadratic (order‑δ²) corrections
        conv[2 * source] += delta * delta
        conv[2 * target] += delta * delta
        conv[source + target] -= 2.0 * delta * delta

        # finally update the probability vector
        a[source] = a_src - delta
        a[target] = a_tgt + delta

        # recompute max and possibly record improvement
        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
            sc = 2.0 * n * M
            if sc < best_score - 1e-12:
                best_seq = a.copy()
                best_score = sc
                print(f"[{time.time():.1f}] hill it={hill_iters}  best={sc:.6f}")
        else:
            # occasionally apply a tiny jitter to escape stagnation
            if hill_iters % 5000 == 0:
                jitter = np.random.randn(n) * 1e-5
                a += jitter
                a = np.maximum(a, eps)
                a /= a.sum()
                conv = _fft_self_convolve(a)
                M = conv.max()

    # ------------------------------------------------------------------
    final_score = 2.0 * best_seq.shape[0] * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
