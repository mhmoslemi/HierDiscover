Propose a low‑score step‑function for ``evaluate_sequence``.

Algorithm
---------

1. **Initial scan** – for several lengths (2000‑5000) we generate a modest
   collection of analytic families (uniform, arcsine, triangular, V‑shape,
   exponential, Gaussian, Beta, power‑edge, power‑abs) together with a
   monotone linear “ramp’’ and random Dirichlet draws.  For each

   * the vector is normalised to L1 = 1,
   * a cheap surrogate score ``2·n·max(conv)`` (where ``conv`` is the full
     self‑convolution via FFT) is evaluated,
   * additionally the *sorted* (non‑increasing) permutation of the vector is
     examined – this breaks symmetry and often reduces the maximum
     convolution.

   The best vector found is kept as the starting point.

2. **Adam‑exponentiated gradient** – we minimise the smoothed maximum of the
   convolution using a multiplicative Adam update on the simplex.
   The smoothing parameter ``α`` grows exponentially from 0.1 to ≈30,
   while the learning‑rate decays from 5 to 0.05.  After each update we
   project back onto the simplex and keep the best iterate seen.

3. **Linear‑programming boost** – given the current vector ``a`` and its
   current peak ``M = max(a*a)``, we solve a small LP that *maximises* the
   total mass of a new vector ``g`` under the linear constraints
   ``(a * g)[k] ≤ M`` for the 200 convolution indices with the largest
   values.  The solution is mixed into ``a`` with a step size ``t=0.2``.
   This step is performed a few times (while time permits) and often
   yields a noticeable drop in the true score.

   random amount of mass from a source index (chosen proportionally to its
   current weight) to a randomly selected target.  The temperature starts

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap surrogate for the true objective (sum = 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * float(conv.max())

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.01) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * (np.log(x) + np.log(1 - x))
    w = np.exp(logw)
    if w.sum() == 0 or np.any(np.isinf(w)):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals /= vals.sum()
    return vals


def _distribution_power_edge(n: int, alpha: float) -> np.ndarray:
    """
    Edge‑heavy family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^{-α}
    """
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - i - 0.5)
    eps = 1e-12
    vals = np.power(denom + eps, -alpha)
    vals /= vals.sum()
    return vals


def _distribution_V_shape(n: int) -> np.ndarray:

    vals = np.abs(np.arange(n, dtype=np.float64) - centre)
    vals = np.maximum(vals, 1e-12)
    vals /= vals.sum()

def _distribution_power_abs(n: int, beta: float) -> np.ndarray:

    vals = np.abs(np.arange(n, dtype=np.float64) - centre) ** beta

def _distribution_linear_ramp(n: int) -> np.ndarray:

    i = np.arange(n, dtype=np.float64)

def _random_symmetric_dirichlet(n: int, alpha: float) -> np.ndarray:
    """Symmetric Dirichlet draw (used for a quick isotropic baseline)."""
    half = (n + 1) // 2
    half_vec = np.random.dirichlet(np.full(half, alpha))

def _lp_step(a: np.ndarray, M: float, top_k: int = 200) -> np.ndarray:
    """
    Solve a small linear programme:
        maximise   Σ g_j
        subject to (a * g)[k] ≤ M  for the `top_k` indices with largest (a*a)[k]
    Returns a feasible vector g (normalised to sum = 1).  If the LP fails,
    the original a is returned.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    # indices of the largest convolution values
    idx = np.argpartition(-conv, top_k)[:top_k]

    rows = []
    cols = []
    data = []
    for row, k in enumerate(idx):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            rows.append(row)
            cols.append(j)
            data.append(a[i])

    A_ub = coo_matrix((data, (rows, cols)), shape=(len(idx), n)).tocsr()
    b_ub = np.full(len(idx), M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)       # maximise Σ g  <=> minimise -Σ g
    bounds = [(0, None) for _ in range(n)]

    try:
        res = linprog(
            c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
            method='highs', options={'presolve': True, 'r': True}
        )
        if not res.success:
            raise RuntimeError("LP failed")
        g = res.x
        eps = 1e-12
        g = np.maximum(g, eps)
        g /= g.sum()
        return g
    except Exception:
        # any failure – fall back to the original vector
        return a.copy()

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a = a * np.exp(-step)               # multiplicative update
        a = _project_onto_simplex(a, 1.0)    # keep on the simplex

        if (it % 500 == 0) or (it == max_iters - 1):
            sc = _max_convolution_score(a)
            if sc < best_score - 1e-12:
                best_seq = a.copy()
                best_score = sc
                best_n = n
                print(f"[{time.time():.1f}] Adam it={it}  best={sc:.6f}")

    # ------------------------------------------------------------------
    # 3.  Linear‑programming boost (few repetitions while time permits)
    # ------------------------------------------------------------------
    lp_reps = 5
    for rep in range(lp_reps):
        if time.time() >= deadline:
            break
        conv = _fft_self_convolve(a)
        M = conv.max()
        g = _lp_step(a, M, top_k=200)
        # mix a little of the LP solution back in
        t_mix = 0.2
        a = (1.0 - t_mix) * a + t_mix * g
        a = _project_onto_simplex(a, 1.0)

        sc = _max_convolution_score(a)
        if sc < best_score - 1e-12:
            best_seq = a.copy()
            best_score = sc
            best_n = n
            print(f"[{time.time():.1f}] LP step {rep}  best={sc:.6f}")

    # ------------------------------------------------------------------
    # 4.  Simulated annealing (mass‑transfer moves)
    # ------------------------------------------------------------------
    cur_score = best_score
    T = max(cur_score * 0.01, 1e-6)
    T_min = cur_score * 1e-7
    decay = 0.9995
    max_sa_iters = 15000

    for it in range(max_sa_iters):
        if time.time() >= deadline:
            break
        source = np.random.choice(n, p=a)                # bias by mass
        target = random.randint(0, n - 1)
        if target == source:
            continue
        delta = min(a[source] * 0.3, 0.02) * random.random()
        if delta < eps:
            continue
        a_new = a.copy()
        a_new[source] -= delta
        a_new[target] += delta
        a_new = np.maximum(a_new, eps)
        a_new /= a_new.sum()
        new_score = _max_convolution_score(a_new)
        dE = new_score - cur_score
        if dE < 0 or random.random() < math.exp(-dE / T):
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_seq = a.copy()
                best_score = new_score
                best_n = n
                print(f"[{time.time():.1f}] SA it={it}  best={new_score:.6f}")
        T = max(T * decay, T_min)

    # ------------------------------------------------------------------
    # 5.  Final cheap hill‑climber
    # ------------------------------------------------------------------
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    hill_iters = 0
    while time.time() < deadline:
        hill_iters += 1
        # locate the index of the current maximum of the convolution
        k = int(conv.argmax())
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        # find (i,j) producing the largest product for this k
        best_i, best_j = i_low, k - i_low
        max_prod = a[best_i] * a[best_j]
        for i in range(i_low, i_high + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        max_delta = min(a[source] * 0.5, 0.015)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a_snapshot = a.copy()
        a[source] -= delta
        a[target] += delta
        a = np.maximum(a, eps)
        a /= a.sum()
        conv = _fft_self_convolve(a)
        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
            sc = 2.0 * n * M                # exact objective because sum=1
            if sc < best_score - 1e-12:
                best_seq = a.copy()
                best_score = sc
                print(f"[{time.time():.1f}] hill it={hill_iters}  best={sc:.6f}")
        else:
            a = a_snapshot
            conv = _fft_self_convolve(a)

        if hill_iters % 5000 == 0:
            # tiny jitter to avoid stagnation
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] jitter applied at hill iter {hill_iters}")

    final_score = 2.0 * best_seq.shape[0] * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")

    # Return as plain Python list (the caller will pass it to evaluate_sequence)
    return best_seq.tolist()
