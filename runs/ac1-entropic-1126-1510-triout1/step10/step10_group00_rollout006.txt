def _linear_corr(a, w):
    n = len(a)
    m = len(w)
    size = 1 << ((n + m - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w, n=size)

    # linear correlation length = n + m -1
    out = corr[: n + m - 1]

    Its gradient is `2·corr(a,w)` where `w` are the soft‑max weights.
    Starting from a low temperature we increase θ gradually (continuation
    method).  After each iteration the iterate is projected onto the
    probability simplex by normalising.
4.  After the time budget for the smooth descent is exhausted,
    a cheap greedy “pairwise mass‑transfer” walk is performed to polish
    the incumbent (only accepting moves that strictly improve the true
    max‑convolution objective).
5.  The final sequence (as a plain Python list) is returned.  All numbers
    are non‑negative; `evaluate_sequence` will handle the final scaling.

The method differs from the previous solution by using a smoothed
objective, a temperature continuation schedule and a correlation‑based
gradient (computed efficiently with FFTs).  This provides more stable
updates and helps escape shallow local minima, giving a better bound
within the same 1000 s budget.
"""

from __future__ import annotations

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (not needed with exponentiated gradient but kept)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _linear_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Linear correlation (a ⋆ w)[k] = Σ_i a[i]·w[i+k]   (k ≥ 0).

    Returns the first len(a) entries (k = 0 … n‑1).
    """
    n = a.shape[0]
    m = w.shape[0]
    size = _next_pow2(n + m - 1)
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w, n=size)
    corr = np.fft.irfft(np.conj(A) * W, n=size)
    # Linear correlation length = n+m-1; we need the first n elements.
    return corr[:n]


# ----------------------------------------------------------------------
# Exact objective (used for monitoring and final evaluation)
def _objective(a: np.ndarray) -> float:
    """Exact `evaluate_sequence` objective for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic distribution families ----------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑density to avoid underflow
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))  # shift for stability
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smoothed exponentiated‑gradient descent -------------------------------
def _smooth_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    theta_start: float = 5.0,
    theta_max: float = 2000.0,
    lr0: float = 0.4,
    eval_interval: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Minimise `max(conv(a,a))` by a temperature‑controlled soft‑max
    approximation.  The iterate stays on the probability simplex via
    exponentiated updates.  Returns the best sequence seen and its
    exact objective value.
    """
    n = a.shape[0]
    theta = theta_start
    t = 0
    while time.time() < deadline:
        t += 1

        # --- 1. compute exact self‑convolution -------------------------
        conv = _fft_self_convolve(a)           # length 2n‑1
        # --- 2. soft‑max weights --------------------------------------
        conv_max = conv.max()
        # stable exponentiation
        exps = np.exp(theta * (conv - conv_max))
        w = exps / exps.sum()                 # sum(w)=1, w ≥0

        # --- 3. gradient of the smoothed objective --------------------
        # grad = 2 * (a ⋆ w)
        grad = 2.0 * _linear_corr(a, w)       # length n, non‑negative

        # --- 4. exponentiated‑gradient update -------------------------
        eta = lr0 / math.sqrt(t)               # diminishing stepsize
        a = a * np.exp(-eta * grad)

        # --- 5. renormalise to simplex --------------------------------
        s = a.sum()
        if s == 0.0:
            a = _distribution_uniform(n)
        else:
            a = a / s

        # --- 6. temperature schedule ----------------------------------
        if t % 200 == 0 and theta < theta_max:
            theta = min(theta_max, theta * 1.15)

        # --- 7. occasional exact evaluation ---------------------------
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] it={t} θ={theta:.1f} new best {best_score:.6f}")
        # safety break
        if t >= 300000:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise mass transfer (polishing) ----------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small amount of mass; keep move only if it improves the exact objective."""
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        # pick a different destination
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.

    The routine respects the given time budget (seconds) and returns the
    best vector found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # a tiny safety margin

    # ------------------------------------------------------------------
    # 1. Choose a length that balances quality and speed.
    n = 20000                     # works well for the continuous limit

    # ------------------------------------------------------------------
    # 2. Build a small set of diverse initial seeds.
    seeds: list[np.ndarray] = []

    # a) classic analytic families
    seeds.append(_distribution_arcsine(n))
    seeds.append(_distribution_edge_weighted(n, p=0.7))
    seeds.append(_mix_with_uniform(seeds[0], lam=0.15))

    # b) a few Dirichlet draws (concentrated mass)
    for _ in range(2):
        alpha = rng.uniform(0.02, 0.2)
        a = rng.dirichlet(np.full(n, alpha))
        seeds.append(a.astype(np.float64))

    # c) previous best if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            seeds.append(prev.copy())
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Evaluate the raw seeds and keep the best incumbent.
    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    for idx, cand in enumerate(seeds):
        sc = _objective(cand)
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = cand.copy()
        print(f"[{time.time():.1f}] seed {idx}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 4. Smooth exponentiated‑gradient descent on the most promising seeds.
    remaining = deadline - time.time()
    if remaining > 30 and overall_best_seq is not None:
        per_seed = remaining / max(1, len(seeds))
        print(
            f"[{time.time():.1f}] start smooth descent – {per_seed:.1f}s per seed"
        )
        for idx, seed_vec in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            best_seq, best_sc = _smooth_gradient_descent(
                seed_vec.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                theta_start=5.0,
                theta_max=2500.0,
                lr0=0.45,
                eval_interval=150,
            )
            if best_sc < overall_best_score:
                overall_best_score = best_sc
                overall_best_seq = best_seq.copy()
                print(
                    f"[{time.time():.1f}] after seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5. Final greedy polishing (pairwise swaps)
    if overall_best_seq is not None and time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy refinement")
        best_seq, best_sc = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=600000,
        )
        if best_sc < overall_best_score:
            overall_best_score = best_sc
            overall_best_seq = best_seq.copy()
            print(f"[{time.time():.1f}] after polishing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Return the result
    if overall_best_seq is None:
        # fallback – should never happen
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)

    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Convert to plain Python list; `evaluate_sequence` will handle the rest.
    return overall_best_seq.tolist()
