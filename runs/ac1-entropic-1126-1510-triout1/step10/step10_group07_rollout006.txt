Pseudo:

```
def conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        # col j runs from i_start .. i_end
        # coefficient = a[k-j]
        for j in range(i_start, i_end+1):
            coeff = a[k - j]
            rows.append(k)
            cols.append(j)
            data.append(coeff)
    return scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
```

"""
Propose a low‑score non‑negative step function.

The pipeline:
1. Initialise a simplex point a (sum = 1) – either the previous best
   (`height_sequence_1`) or an arcsine‑distribution.
2. Repeatedly solve a linear programme:
       maximise Σ g_i   subject to   (a * g) ≤ M   and   g ≥ 0,
   where M = max(a*a).  The LP finds a vector g with a larger total
   mass while staying inside the current convolution bound.
   Mixing a with the normalised g (a ← (1‑t)a + t g/Σg) strictly reduces
   the objective whenever Σg > 1.
3. After a handful of LP‑updates, polish the result with a few
   projected‑Adam / exponentiated‑gradient steps (sub‑gradient of the
   max‑convolution) and a short Frank‑Wolfe “mass‑transfer’’ phase.
4. Return the best sequence found (already normalised, sum = 1).
All numbers are non‑negative; the outer `evaluate_sequence` will take care
of the final scaling.
"""

import time
import numpy as np

# -------------------------------------------------------------
# basic geometry helpers -------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
# convolution helpers -------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Surrogate objective for Σa = 1  →  2·n·max(a*a)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# -------------------------------------------------------------
# sub‑gradient of the max‑convolution (used by Adam/EG) --------------
def _subgrad(a: np.ndarray):
    """Sub‑gradient of max(conv(a,a)) with Σa = 1.
    Returns (grad, cur_max, active_indices)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # gradient of the (non‑smooth) max‑convolution:
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for i=j terms (when k even)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# -------------------------------------------------------------
# Adam and exponentiated‑gradient descent (projection onto simplex) ---
def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.30,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 500_000):
    """Adam on the simplex (projected). Returns best point seen."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          eta0: float = 0.9,
                          max_iters: int = 400_000):
    """Exponentiated‑gradient (mirror descent) on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


# -------------------------------------------------------------
# analytic seed – arcsine law (strictly positive, good start) ----------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine pdf on {0,…,n‑1} (all entries >0)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# -------------------------------------------------------------
# Build the convolution matrix A such that (A g)_k = (a * g)_k ----------
def _build_conv_matrix(a: np.ndarray):
    """
    Returns a sparse CSR matrix A with shape (2n‑1, n) satisfying:
        (A @ g) = a * g   (full linear convolution).
    Each diagonal d (0 ≤ d < n) contains the constant a[d].
    """
    from scipy.sparse import diags
    n = a.shape[0]
    # one diagonal per offset d, each filled with a[d]
    diag_data = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n)                     # 0,1,…,n‑1
    A = diags(diag_data, offsets, shape=(2 * n - 1, n), format='csr')
    return A

        if not res.success:

            break

        g = res.x
        S = g.sum()
        print(f"  LP solution sum(g) = {S:.6f}")

        # No improvement possible if total mass does not exceed 1

        a_new = (1.0 - t) * best_a + t * g_norm

        new_score = _max_convolution_score(a_new)

            print(f"  ** improved to {best_score:.6f}")
        else:
