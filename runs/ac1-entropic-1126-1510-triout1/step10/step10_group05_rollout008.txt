def _adam_smooth(a0, deadline, mu, max_iters, step0=0.2, beta1=0.9,beta2=0.999):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a=a.copy()
    best_score = _max_convolution_score(a)  # actual surrogate, not smooth
    t=0
    while time.time()<deadline and t<max_iters:
        t+=1
        obj, grad = _smooth_obj_and_grad(a, mu)
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        eta = step0/(np.sqrt(v_hat) + 1e-8)
        a = a - eta*m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

where `conv` is the full self‑convolution.
The algorithm consists of three phases:

1. **Broad seeding** – generate a handful of sensible initial vectors
   (arcsine distribution, random Dirichlet draws, and a previous best if
   available) on several lengths.

2. **Smooth Adam optimisation** – minimise a *smoothed* version of the
   max‑convolution objective
        f_μ(a) = 2·n·μ·log ∑ₖ exp((a*a)ₖ/μ) .
   For a large smoothing parameter μ the landscape is gentle, allowing the
   optimiser to move away from poor local minima; μ is decreased
   geometrically (5·max b, 2·max b, max b, …) so that the final stages approach
   the true max‑convolution.  Adam with simplex projection is used at every
   stage.

3. **Local polishing** – a short random hill‑climber and a Frank‑Wolfe‑style
   pairwise mass‑transfer finish the search.

All steps respect the given time budget (≈ 1000 s).  The returned list is
ready for `evaluate_sequence`.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate for `evaluate_sequence` when the vector is ℓ₁‑normalised."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex from Dirichlet(α,…,α)."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, mu: float):
    """
    Smooth approximation of the max‑convolution objective and its gradient.
    Returns (objective, gradient) with respect to a.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)               # length 2n‑1
    max_b = b.max()

    # stable soft‑max weights w_k = exp((b_k-max_b)/mu) / Σ exp(...)
    shifted = b - max_b                     # ≤ 0
    exps = np.exp(shifted / mu)            # safe because shifted ≤ 0
    sum_exps = exps.sum()
    w = exps / sum_exps                     # shape (2n‑1,)

    # weighted gradient of the quadratic form a*a
    conv_rev = _fft_convolve(a[::-1], w)    # length 3n‑2
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]  # slice to length n

    # diagonal correction for terms a_i² (they appear once, not twice)
    even_idx = np.arange(0, 2 * n - 1, 2)    # positions 0,2,…,2n‑2
    diag_pos = even_idx // 2
    grad[diag_pos] -= a[diag_pos] * w[even_idx]

    # overall scaling (objective contains factor 2·n)
    grad *= 2.0 * n

    # smooth objective: 2·n·(max_b + mu·log Σ exp((b‑max_b)/mu))
    obj = 2.0 * n * (max_b + mu * np.log(sum_exps))
    return obj, grad


# ----------------------------------------------------------------------
def _adam_smooth(a0: np.ndarray, deadline: float, mu: float,
                 max_iters: int = 200_000,
                 step0: float = 0.2,
                 beta1: float = 0.9,
                 beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed objective.
    Returns the best vector (ℓ₁‑normalised) seen and its surrogate score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        _, grad = _smooth_obj_and_grad(a, mu)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        # occasional evaluation to keep the best found
        if t % 10 == 0:
            cur = _max_convolution_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 200_000,
                          eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """Simple multiplicative (exponentiated) gradient descent."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # sub‑gradient of the max‑convolution
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps)

        w = np.zeros_like(b)
        w[active] = 1.0
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[a.size - 1:a.size - 1 + a.size]

        even_mask = (active % 2 == 0)
        diag_idx = (active[even_mask] // 2)
        if diag_idx.size > 0:
            grad[diag_idx] -= a[diag_idx]

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s

        cur_score = _max_convolution_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray, deadline: float,
                max_iters: int = 300_000,
                epsilon: float = 0.02) -> tuple[np.ndarray, float]:
    """Random mass‑transfer hill‑climber (accepts only improving moves)."""
    best = a.copy()
    best_score = _max_convolution_score(best)
    n = best.size
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or best[i] < 1e-12:
            continue
        delta = min(best[i] * epsilon, 1.0 - best[j])
        if delta <= 0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = _max_convolution_score(cand)
        if sc < best_score:
            best = cand
            best_score = sc
    return best, best_score


# ----------------------------------------------------------------------
def _frank_wolfe_opt(a0: np.ndarray, deadline: float,
                    max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Pairwise mass transfer from the coordinate with largest sub‑gradient to
    the one with smallest, with occasional jitter.
    """
    a = a0.copy()
    cur_score = _max_convolution_score(a)
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # sub‑gradient of the max‑convolution
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps)

        w = np.zeros_like(b)
        w[active] = 1.0
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[a.size - 1:a.size - 1 + a.size]
        even_mask = (active % 2 == 0)
        diag_idx = (active[even_mask] // 2)
        if diag_idx.size > 0:
            grad[diag_idx] -= a[diag_idx]

        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        delta = min(a[i_src], 1.0 - a[i_dst])
        if delta <= 0:
            break
        step = 0.7 * delta          # move 70 % of admissible mass
        a[i_src] -= step
        a[i_dst] += step
        cur_score = _max_convolution_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        # occasional jitter to avoid stagnation
        if it % 4000 == 0 and np.random.rand() < 0.1:
            a = _project_onto_simplex(a + np.random.randn(a.size) * 1e-5, s=1.0)
            cur_score = _max_convolution_score(a)
    return best, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the given time budget (≈ 1000 s) and returns the best
    vector discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Pick a base size (try to reuse a previously found vector)
    try:
        base_n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        base_n = 5000

    # ------------------------------------------------------------------
    # Lengths to explore (moderate range, keep within ~15 k)
    lengths = set()
    for factor in (0.8, 1.0, 1.25, 1.5):
        n = int(base_n * factor)
        if 800 <= n <= 15000:
            lengths.add(n)
    for _ in range(3):
        lengths.add(np.random.randint(800, min(15000, base_n * 2) + 1))
    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # Simple seeding on each length
    best_seq = None
    best_score = np.inf

    for n in lengths:
        if time.time() >= deadline:
            break

        # 1) deterministic arcsine seed
        a = _arcsine_distribution(n)
        sc = _max_convolution_score(a)
        if sc < best_score:
            best_seq, best_score = a.copy(), sc

        # 2) a few random Dirichlet seeds
        for _ in range(3):
            a = _random_dirichlet_alpha(n, np.random.choice([0.2, 0.5, 1.0, 2.0]))
            sc = _max_convolution_score(a)
            if sc < best_score:
                best_seq, best_score = a.copy(), sc

    # incorporate a previously known good vector (if any) by resampling it
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.size > 0:
            # resample to the current base length by linear interpolation of the CDF
            n = base_n
            cdf = np.cumsum(prev)
            cdf /= cdf[-1]
            target = np.linspace(0.0, 1.0, n, endpoint=False) + 0.5 / n
            a = np.interp(target, cdf, prev, left=0.0, right=0.0)
            a = np.maximum(a, 0.0)
            if a.sum() > 0:
                a /= a.sum()
                sc = _max_convolution_score(a)
                if sc < best_score:
                    best_seq, best_score = a.copy(), sc
    except Exception:
        pass

    if best_seq is None:
        best_seq = _arcsine_distribution(base_n)

    print(f"Initial best surrogate score after seeding: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Smooth Adam optimisation with decreasing μ
    #   (large μ → smoother, small μ → true max).  We start from the current
    #   best vector and keep the best improvement across all stages.
    mu_factors = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.03, 0.01, 0.003, 0.001]

    for factor in mu_factors:
        if time.time() >= deadline:
            break
        # current max‑convolution value (needed to set μ)
        cur_max = _fft_self_convolve(best_seq).max()
        mu = max(cur_max * factor, 1e-12)

        # number of Adam iterations for this stage – adapt to remaining time
        remaining = deadline - time.time()
        max_iters = int(min(200_000, max(5_000, remaining * 150)))  # approx. 150 it/s

        seq, sc = _adam_smooth(best_seq, deadline, mu,
                               max_iters=max_iters, step0=0.2)
        if sc < best_score:
            best_seq, best_score = seq.copy(), sc
            print(f"Smoothing μ={factor:.3f} improved score to {best_score:.6f}")

        # a short exponentiated‑gradient refinement after each smoothing stage
        if time.time() < deadline:
            seq, sc = _exp_gradient_descent(best_seq, deadline,
                                            max_iters=100_000, eta0=0.8)
            if sc < best_score:
                best_seq, best_score = seq.copy(), sc
                print(f"Post‑smooth ExpGrad improved score to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Final local polishing
    if time.time() < deadline - 30:
        seq, sc = _hill_climb(best_seq, deadline,
                              max_iters=300_000, epsilon=0.02)
        if sc < best_score:
            best_seq, best_score = seq.copy(), sc
            print(f"Hill‑climb final improvement → {best_score:.6f}")

    if time.time() < deadline - 15:
        seq, sc = _frank_wolfe_opt(best_seq, deadline,
                                   max_iters=500_000)
        if sc < best_score:
            best_seq, best_score = seq.copy(), sc
            print(f"Frank‑Wolfe final polishing → {best_score:.6f}")

    print(f"Finished search. Best surrogate score = {best_score:.6f} (len={len(best_seq)})")
    # Ensure the output respects the non‑negative, finite constraints.
    final = np.maximum(best_seq, 0.0).tolist()
    return final
