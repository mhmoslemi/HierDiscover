"""
Hybrid optimizer for the step‑function attack (target < 1.503).

The algorithm proceeds in three phases:

1) **Seed generation** – a large pool of analytic and random distributions
   (uniform, arcsine, beta, power‑law, exponential, two‑block, Dirichlet,
   sparse) is created for several problem sizes.  The exact objective
   `2·n·max(conv(a,a))/sum(a)^2` is evaluated and a small set of the best
   seeds (across all lengths) is kept.

2) **Power‑weighted mirror descent** – each retained seed is refined with a
   constrained mirror‑descent (exponentiated‑gradient) on the simplex.
   Instead of the hard active‑set used previously, a *smooth* weight vector
   `w ∝ conv(a,a)^p` (p grows from 1 to ≈12) is employed, which yields a
   gradient `∇ = 2·a⋆w`.  The update uses Adam‑style momentum and an
   exponentially decreasing step‑size.  The iteration is stopped when the
   global time budget expires.

3) **Greedy pairwise mass swaps** – a cheap local search repeatedly transfers
   a random fraction of mass from one index to another and keeps the move
   only if the true (non‑smoothed) objective improves.

All arithmetic is performed in double precision, the vector is kept
non‑negative and normalised to total mass = 1, and the final sequence is
returned as a plain list of floats ready for `evaluate_sequence`.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (expects sum(a)=1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic families -------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals

    cand = []

def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient 2·a⋆w, where w is a weight vector (length 2·n‑1)
    that sums to 1. Implemented via FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2·n‑1
    a_rev = a[::-1]                    # reversed a
    size = 1 << ((n + m - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1                      # index where shift = 0 begins
    grad = 2.0 * conv[start:start + n]
    return grad

def _refine_sequence_power_descent(seed: np.ndarray,
                                  sub_deadline: float,
                                  best_seq: np.ndarray,
                                  best_score: float,
                                  lr0: float = 0.6,

                                  eval_interval: int = 500) -> tuple[np.ndarray, float]:

    focus onto the active set.
    Adam‑style momentum is used for the updates.
    """

    if a.sum() == 0.0:
        a = _distribution_uniform(len(a))
    a = a / a.sum()

    n = a.shape[0]

    v = np.zeros_like(a)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_local_seq = a.copy()
    best_local_score = best_score

    for t in range(1, max_iter + 1):
        if time.time() >= sub_deadline:
            break

        conv = _fft_self_convolve(a)
        # schedule for p (start gentle, finish aggressive)

    if best_local_score < best_score:

    else:
        return best_seq, best_score

def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """
    Simple greedy mass‑transfer walk.  A random source index i transfers a
    random fraction (up to 30%) of its mass to a random target j.
    The move is kept only if the true objective improves.
    """
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        # choose a different target
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} Δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.6,
                max_iter=200000,
                eval_interval=400,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after refinement new best {overall_best_score:.6f}")

    # --- 5. Final cheap local polishing ---------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] Starting final greedy swaps")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
