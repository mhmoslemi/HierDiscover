"""
Step‑function optimizer.

The goal is to minimise
    2·n·max(conv(a,a)) / (sum(a)**2)
for a non‑negative vector a (the sum is scale‑invariant, we keep sum(a)=1).

The algorithm works in three stages:

1. **Broad scan** – generate many analytic families (uniform, arcsine,
   Beta, power‑edge, Dirichlet draws, …) for several lengths (2000‑5000).
   The fast FFT based score 2·n·max(conv) is evaluated and the best
   vector is kept.

2. **Smoothed‑max mirror descent** – exponentiated‑gradient descent on
   the simplex with a growing smooth‑max temperature reduces the L∞
   norm of the self‑convolution.  This yields a good continuous
   improvement.

3. **Linear‑programming refinement** – for the current vector `a`
   compute `M = max(conv(a,a))`.  Solve the LP

        maximise   Σ b_j
        subject to  conv(a,b) ≤ M   (component‑wise)
                    b ≥ 0

   The optimum gives a vector `b` with Σb = S > 1 (otherwise `a` is already
   optimal).  Re‑normalising `b` to sum = 1 reduces the maximal
   convolution by the factor `1/S`.  Repeating this step a few times
   drives the score further down.

Finally a short hill‑climber (mass‑transfer) cleans up tiny residues.
All steps respect the 1000 s wall‑clock budget and print intermediate
scores.
"""

import time

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Fast evaluation of 2·n·max(conv(a,a)) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * float(conv.max())

# Analytic distribution families – each returns a probability vector (sum = 1)

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.01) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * (np.log(x) + np.log(1 - x))
    w = np.exp(logw)
    if w.sum() == 0 or np.any(np.isinf(w)):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals /= vals.sum()
    return vals


def _distribution_power_edge(n: int, alpha: float) -> np.ndarray:
    """Edge‑heavy family: w_i ∝ ((i+½)*(n‑i‑½))^{-α}."""
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - i - 0.5)
    eps = 1e-12
    vals = np.power(denom + eps, -alpha)
    vals /= vals.sum()
    return vals


def _distribution_V_shape(n: int) -> np.ndarray:
    centre = (n - 1) / 2.0
    vals = np.abs(np.arange(n, dtype=np.float64) - centre)
    eps = 1e-12
    vals = np.maximum(vals, eps)
    vals /= vals.sum()
    return vals


def _distribution_power_abs(n: int, beta: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    vals = np.abs(np.arange(n, dtype=np.float64) - centre) ** beta
    eps = 1e-12
    vals = np.maximum(vals, eps)
    vals /= vals.sum()
    return vals


def _random_symmetric_dirichlet(n: int, alpha: float) -> np.ndarray:
    half = (n + 1) // 2
    half_vec = np.random.dirichlet(np.full(half, alpha))
    if n % 2 == 0:
        full = np.concatenate([half_vec, half_vec[::-1]])
    else:
        full = np.concatenate([half_vec, half_vec[-2::-1]])
    full /= full.sum()
    return full


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step‑function with a low evaluation score."""
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1. Broad analytic scan
    lengths = [2000, 2500, 3000, 3500, 4000, 4500, 5000]
    for n in lengths:
        if time.time() >= deadline:
            break

        candidates = []

        # deterministic families
        candidates.append(_distribution_uniform(n))
        candidates.append(_distribution_arcsine(n))
        candidates.append(_distribution_triangular(n))
        candidates.append(_distribution_exponential(n, lam=0.008))
        candidates.append(_distribution_exponential(n, lam=0.02))
        candidates.append(_distribution_V_shape(n))

        # sine‑power
        for pw in (0.5, 1.0, 2.0, 4.0):
            candidates.append(_distribution_sine_power(n, pw))

        # Gaussian
        for sigma in (0.2, 0.35, 0.5):
            candidates.append(_distribution_gaussian(n, sigma))

        # symmetric beta
        for p in (0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.8,
                  1.0, 1.5, 2.0, 3.0, 4.0):
            candidates.append(_distribution_beta_sym(n, p))

        # power‑edge
        for a_edge in (0.55, 0.6, 0.7, 0.8, 0.9):
            candidates.append(_distribution_power_edge(n, a_edge))

        # power‑abs
        for beta in (0.5, 1.0, 2.0, 3.0):
            candidates.append(_distribution_power_abs(n, beta))

        # random Dirichlet draws
        for alpha in (0.03, 0.05, 0.08, 0.1, 0.2, 0.5):
            candidates.append(_random_symmetric_dirichlet(n, alpha))

        # evaluate all vectors for this length
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] init n={n} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Use previously found global best if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_seq = prev.copy()
                best_n = prev.size
                best_score = sc
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # fallback
    if best_seq is None:
        best_n = 3000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Smoothed‑max mirror descent (exponentiated gradient)
    a = best_seq.copy()
    n = a.shape[0]
    total_steps = 20000
    for it in range(total_steps):
        if time.time() >= deadline:
            break
        frac = it / total_steps
        alpha = 0.1 * (200.0 ** frac)          # smooth‑max temperature
        lr = 5.0 * (1.0 - frac) + 0.05         # learning rate

        conv = _fft_self_convolve(a)
        maxc = conv.max()
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed max
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()

        if (it % 500 == 0) or (it == total_steps - 1):
            sc = _max_convolution_score(a)
            if sc < best_score - 1e-12:
                best_seq = a.copy()
                best_score = sc
                best_n = n
                print(f"[{time.time():.1f}] mirror it={it} best={sc:.6f}")

    # ------------------------------------------------------------------
    # 4. Linear‑programming refinement
    #   maximise Σb   s.t. conv(a,b) ≤ M   and b≥0.
    #   If Σb = S > 1 then a←b/S improves the max‑convolution.
    a = best_seq.copy()
    n = a.shape[0]
    lp_iter = 0
    while time.time() < deadline and lp_iter < 15:
        conv = _fft_self_convolve(a)
        M = conv.max()
        L = 2 * n - 1

        # Build sparse matrix for constraints conv(a,b) ≤ M
        rows = []
        cols = []
        data = []
        for k in range(L):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
            j_vals = k - i_vals
            rows.extend([k] * i_vals.size)
            cols.extend(j_vals.tolist())
            data.extend(a[i_vals].tolist())
        A_ub = sp.csr_matrix((data, (rows, cols)), shape=(L, n))
        b_ub = np.full(L, M, dtype=np.float64)

        c = -np.ones(n, dtype=np.float64)      # maximise Σb
        bounds = [(0, None)] * n

        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=bounds, method='highs',
                          options={'presolve': True})
        if not res.success:
            print(f"[{time.time():.1f}] LP #{lp_iter} failed: {res.message}")
            break

        b_opt = res.x
        S = b_opt.sum()
        if S <= 1.0 + 1e-9:
            print(f"[{time.time():.1f}] LP #{lp_iter} no improvement (S={S:.6f})")
            break

        a = b_opt / S
        a = np.maximum(a, eps)
        a /= a.sum()
        new_score = _max_convolution_score(a)
        print(f"[{time.time():.1f}] LP #{lp_iter}  S={S:.6f}  score={new_score:.6f}")

        if new_score < best_score - 1e-12:
            best_seq = a.copy()
            best_score = new_score
            best_n = n
        else:
            # no progress – stop early
            print(f"[{time.time():.1f}] LP #{lp_iter} did not improve score")
            break

        lp_iter += 1

    # ------------------------------------------------------------------
    # 5. Final cheap hill‑climber (mass transfers)
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    hill_iters = 0
    while time.time() < deadline:
        hill_iters += 1
        k = int(conv.argmax())
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)

        # find the pair (i,j) producing k with largest product a_i*a_j
        best_i, best_j = i_low, k - i_low
        max_prod = a[best_i] * a[best_j]
        for i in range(i_low, i_high + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        max_delta = min(a[source] * 0.5, 0.015)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a_snapshot = a.copy()
        a[source] -= delta
        a[target] += delta
        a = np.maximum(a, eps)
        a /= a.sum()
        conv = _fft_self_convolve(a)
        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
            sc = 2.0 * n * M
            if sc < best_score - 1e-12:
                best_seq = a.copy()
                best_score = sc
                print(f"[{time.time():.1f}] hill it={hill_iters} best={sc:.6f}")
        else:
            a = a_snapshot
            conv = _fft_self_convolve(a)
            M = conv.max()

        if hill_iters % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] jitter applied at hill iter {hill_iters}")

    final_score = 2.0 * best_seq.shape[0] * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
