"""
Propose a step‑function (non‑negative sequence) that minimises
evaluate_sequence().  The algorithm consists of:

1.  A quick scan of many analytic families (uniform, arcsine, triangular,
    exponential, Beta(p,p) with many p, inverse‑power, two‑block,
    sine‑power) for several lengths.  The best candidate so far is kept.
2.  If a previous global solution (height_sequence_1) is available it is
    considered.
3.  The current best vector is refined by a multi‑phase optimisation:
      – exponential‑gradient descent on a smooth approximation of max(conv)
      – Lp‑gradient descent for p = 2,4,8,16,32
      – Adam on the logits (soft‑max parametrisation)
      – a stochastic hill‑climber with occasional jitter.
4.  While time permits, many random Dirichlet restarts are generated and
    refined with a shortened version of the pipeline.
5.  The best vector found is returned as a plain list of floats.  All
    entries are non‑negative, ≤1000 and the sum is ≈1, so the score equals
    2·n·max(a∗a) as used inside evaluate_sequence().
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Score = 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Softmax weights approximating a smooth max."""
    maxc = conv.max()
    w = np.exp(alpha * (conv - maxc))
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / s


def _subgradient_weights(conv: np.ndarray, eps_rel: float = 1e-12) -> np.ndarray:
    """Weights for a (sub)gradient of the max‑convolution."""
    m = conv.max()
    w = np.zeros_like(conv)
    w[conv >= m - eps_rel * (abs(m) + 1.0)] = 1.0
    cnt = w.sum()
    if cnt == 0:
        return w
    return w / cnt


def _exponential_gradient_step(a: np.ndarray, lr: float, alpha: float) -> np.ndarray:
    """One EG step (mirror descent with KL) on the smooth‑max objective."""
    conv = _fft_self_convolve(a)
    w = _smooth_weights(conv, alpha)
    n = a.shape[0]
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    a = a * np.exp(-lr * grad)
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def _lp_gradient_step(a: np.ndarray, lr: float, p: int) -> np.ndarray:
    """Gradient descent on the p‑norm of the self‑convolution (p≥2)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    # derivative of Σ conv^p w.r.t. conv is p·conv^{p‑1}
    w = (conv ** (p - 1)) * p
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    a = a - lr * grad
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def _adam_step(logits: np.ndarray, m: np.ndarray, v: np.ndarray,
               t: int, step: np.ndarray):
    """Adam update on logits (unconstrained)."""
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = beta1 * m + (1.0 - beta1) * step
    v = beta2 * v + (1.0 - beta2) * (step ** 2)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    lr = 0.02 / (np.sqrt(v_hat) + eps)
    logits = logits - lr * m_hat
    logits = np.clip(logits, -30, 30)          # avoid overflow
    return logits, m, v


def _hillclimber_move(a: np.ndarray, conv: np.ndarray, eps: float = 1e-12):
    """Move a tiny chunk from a large‑gradient index to a small‑gradient one."""
    n = a.shape[0]
    w = _subgradient_weights(conv, eps_rel=eps)
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    src = int(np.argmax(grad))
    dst = int(np.argmin(grad))
    if src == dst:
        dst = random.randint(0, n - 1)
    max_delta = min(a[src] * 0.5, 0.02)
    delta = max_delta * random.random()
    if delta < eps:
        return a, conv, False
    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    new_conv = _fft_self_convolve(a_new)
    improved = new_conv.max() < conv.max() - 1e-13
    return (a_new, new_conv, improved) if improved else (a, conv, False)


# ----------------------------------------------------------------------
# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.01) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_inverse_power(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i+½)^{-p} + (n‑i‑½)^{-p}, symmetric, sum = 1."""
    i = np.arange(n, dtype=np.float64)
    rev = (n - 1 - i).astype(np.float64)
    vals = (i + 0.5) ** (-p) + (rev + 0.5) ** (-p)
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.1) -> np.ndarray:
    """Mass 0.5 on a left block and 0.5 on a right block (both of same length)."""
    block_len = max(1, int(n * block_frac))
    a = np.zeros(n, dtype=np.float64)
    a[:block_len] = 0.5 / block_len
    a[-block_len:] = 0.5 / block_len
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a step‑function minimising evaluate_sequence().

    The function respects a wall‑clock budget (default 1000 s) and returns
    the best sequence discovered as a list of floats.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin
    eps = 1e-12

    best_seq: np.ndarray | None = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1. Scan analytic families for a few lengths
    scan_lengths = [1500, 2000, 2500, 3000, 3500, 4000, 4500]
    beta_ps = [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.8,
               1.0, 2.0, 4.0]
    inv_ps = [0.5, 1.0, 2.0, 3.0, 4.0]
    block_fracs = [0.05, 0.1, 0.15, 0.2]
    sine_powers = [0.5, 1.0, 2.0, 4.0]

    for n in scan_lengths:
        if time.time() >= deadline:
            break
        # generate candidates one‑by‑one to keep memory low
        cand_funcs = [
            lambda: _distribution_uniform(n),
            lambda: _distribution_arcsine(n),
            lambda: _distribution_triangular(n),
            lambda: _distribution_exponential(n, lam=0.008),
            lambda: _distribution_exponential(n, lam=0.02),
        ]
        # beta family
        for p in beta_ps:
            cand_funcs.append(lambda p=p: _distribution_beta_sym(n, p))
        # inverse power family
        for p in inv_ps:
            cand_funcs.append(lambda p=p: _distribution_inverse_power(n, p))
        # two‑block family
        for f in block_fracs:
            cand_funcs.append(lambda f=f: _distribution_two_blocks(n, f))
        # sine‑power family
        for pw in sine_powers:
            cand_funcs.append(lambda pw=pw: _distribution_sine_power(n, pw))
        # a few random Dirichlet draws
        for alpha in (0.1, 0.3, 0.5, 0.8, 1.0, 2.0):
            cand_funcs.append(lambda alpha=alpha: np.random.dirichlet(np.full(n, alpha)))

        for gen in cand_funcs:
            if time.time() >= deadline:
                break
            vec = gen()
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.2f}] init n={n}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Include previously found global solution if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_seq = prev.copy()
                best_n = prev.size
                best_score = sc
                print(f"[{time.time():.2f}] global best {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback if nothing succeeded
    if best_seq is None:
        best_n = 2000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.2f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Refinement pipeline (full, expensive) ---------------------------------
    def _refine_vector(a: np.ndarray, n: int) -> np.ndarray:
        """Run a multi‑phase optimisation on vector a (already normalised)."""
        local_best = a.copy()
        local_best_score = _max_convolution_score(a)

        # ----- Phase A: exponential‑gradient on smooth max --------------------
        steps_eg = 3000
        a_cur = a.copy()
        for it in range(steps_eg):
            if time.time() >= deadline:
                break
            t = it / steps_eg
            # α grows from 1 to 5000, lr shrinks from 0.5 to 0.001
            alpha = 1.0 * (5000.0) ** t
            lr = 0.5 * (0.001 / 0.5) ** t
            a_cur = _exponential_gradient_step(a_cur, lr, alpha)
            if it % 300 == 0 or it == steps_eg - 1:
                sc = _max_convolution_score(a_cur)
                if sc < local_best_score:
                    local_best = a_cur.copy()
                    local_best_score = sc
                    print(f"[{time.time():.2f}] EG it={it}  best={sc:.6f}")

        # ----- Phase B: Lp‑gradient descent for several p --------------------
        a_cur = local_best.copy()
        for p in (2, 4, 8, 16, 32):
            steps_lp = 150
            for it in range(steps_lp):
                if time.time() >= deadline:
                    break
                lr = 0.2 / (1.0 + it * 0.001)          # small decay
                a_cur = _lp_gradient_step(a_cur, lr, p)
                if it % 30 == 0 or it == steps_lp - 1:
                    sc = _max_convolution_score(a_cur)
                    if sc < local_best_score:
                        local_best = a_cur.copy()
                        local_best_score = sc
                        print(f"[{time.time():.2f}] Lp{p} it={it}  best={sc:.6f}")

        # ----- Phase C: Adam on logits -----------------------------------------
        a_cur = local_best.copy()
        logits = np.log(a_cur + eps)
        m = np.zeros_like(logits)
        v = np.zeros_like(logits)
        steps_adam = 2000
        for it in range(1, steps_adam + 1):
            if time.time() >= deadline:
                break
            a_adam = np.exp(logits)
            a_adam /= a_adam.sum()
            conv = _fft_self_convolve(a_adam)
            w = _subgradient_weights(conv, eps_rel=1e-12)
            grad_a = 2.0 * _fft_convolve(a_adam[::-1], w)[n - 1 : n - 1 + n]
            # gradient w.r.t. logits under softmax
            grad_log = (grad_a - np.dot(a_adam, grad_a)) * a_adam
            logits, m, v = _adam_step(logits, m, v, it, grad_log)
            if it % 400 == 0 or it == steps_adam:
                a_adam = np.exp(logits)
                a_adam /= a_adam.sum()
                sc = _max_convolution_score(a_adam)
                if sc < local_best_score:
                    local_best = a_adam.copy()
                    local_best_score = sc
                    print(f"[{time.time():.2f}] Adam it={it}  best={sc:.6f}")

        # ----- Phase D: Simple hill‑climber ------------------------------------
        a_cur = local_best.copy()
        conv = _fft_self_convolve(a_cur)
        hc_iters = 20000
        for it in range(hc_iters):
            if time.time() >= deadline:
                break
            a_cur, conv, improved = _hillclimber_move(a_cur, conv, eps=eps)
            if improved:
                sc = 2.0 * n * conv.max()
                if sc < local_best_score:
                    local_best = a_cur.copy()
                    local_best_score = sc
                    if it % 500 == 0:
                        print(f"[{time.time():.2f}] hill it={it}  best={sc:.6f}")
            # occasional jitter to escape plateaus
            if it % 5000 == 0 and it > 0:
                jitter = np.random.randn(n) * 1e-5
                a_cur += jitter
                a_cur = np.maximum(a_cur, eps)
                a_cur /= a_cur.sum()
                conv = _fft_self_convolve(a_cur)

        return local_best

    # perform a full refinement on the current best vector
    best_seq = _refine_vector(best_seq, best_n)

    # ------------------------------------------------------------------
    # 5. Multiple random restarts (shorter pipelines) while time permits
    restart_lengths = [1500, 2000, 2500, 3000, 3500, 4000, 4500]
    while time.time() < deadline - 5.0:
        n = random.choice(restart_lengths)
        a0 = np.random.dirichlet(np.full(n, 0.3))
        a0 = np.maximum(a0, eps)
        a0 /= a0.sum()

        # shortened refinement: fewer EG steps, no Lp, tiny Adam, short hillclimb
        def _short_refine(vec: np.ndarray, size: int) -> np.ndarray:
            # EG only (1500 steps)
            a_tmp = vec.copy()
            steps = 1500
            for it in range(steps):
                if time.time() >= deadline:
                    break
                t = it / steps
                alpha = 1.0 * (3000.0) ** t
                lr = 0.5 * (0.001 / 0.5) ** t
                a_tmp = _exponential_gradient_step(a_tmp, lr, alpha)
                if it % 250 == 0:
                    sc = _max_convolution_score(a_tmp)
                    if sc < best_score:
                        best_seq = a_tmp.copy()
                        best_score = sc
                        best_n = size
                        print(f"[{time.time():.2f}] restart EG n={size}  score={sc:.6f}")
            # tiny hillclimb (5000 iters)
            a_tmp = a_tmp.copy()
            conv = _fft_self_convolve(a_tmp)
            for it in range(5000):
                if time.time() >= deadline:
                    break
                a_tmp, conv, improved = _hillclimber_move(a_tmp, conv, eps=eps)
                if improved:
                    sc = 2.0 * size * conv.max()
                    if sc < best_score:
                        best_seq = a_tmp.copy()
                        best_score = sc
                        best_n = size
                        print(f"[{time.time():.2f}] restart HC n={size}  score={sc:.6f}")
                if it % 1000 == 0 and it > 0:
                    jitter = np.random.randn(size) * 1e-5
                    a_tmp += jitter
                    a_tmp = np.maximum(a_tmp, eps)
                    a_tmp /= a_tmp.sum()
                    conv = _fft_self_convolve(a_tmp)
            return a_tmp

        _ = _short_refine(a0, n)

    # ------------------------------------------------------------------
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
