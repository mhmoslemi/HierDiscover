"""
Propose a step‑function (list of non‑negative floats) that minimises
``evaluate_sequence``.  The algorithm consists of three phases:

1. **Broad analytic scan** – for several lengths (2000‑10000) we
   generate a variety of symmetric analytic families (uniform,
   arcsine, triangular, exponential, Beta, sine‑power, Gaussian,
   power‑edge, power‑abs) and a few low‑α Dirichlet draws.
   Each candidate is normalised to sum = 1 and scored with the cheap
   fast metric ``2·n·max(a*a)`` (exactly the same quantity used by
   ``evaluate_sequence`` when the sum is 1).  The best vector from this
   scan is kept as a warm‑start.

2. **Adamised smooth‑max mirror descent** – the warm‑start is
   re‑parameterised as ``a = softmax(y)`` (so the simplex constraint is
   satisfied automatically).  In each iteration we evaluate the
   self‑convolution, build a soft‑max weighting of the entries
   ``w ∝ exp(α·(conv−max(conv)))`` and compute the gradient of the
   smoothed maximum.  The gradient on the probability simplex is
   transformed to a gradient on ``y`` and an Adam optimiser updates
   ``y``.  The smoothing parameter α grows exponentially from 0.1 to
   ≈40, while the Adam learning‑rate decays.  The best vector seen
   during this phase is recorded.

3. **Targeted hill‑climber** – starting from the best vector found
   so far we repeatedly move a small amount of mass away from the
   index‑pair that contributes most to the current maximal convolution
   entry.  Moves are accepted only if they strictly decrease the
   maximal convolution value.  Occasionally a tiny random jitter is
   added to escape shallow plateaus.

All phases respect the global 1000 s wall‑clock budget.  The returned
list contains only non‑negative numbers and is normalised to sum = 1.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolve_score(a: np.ndarray) -> float:
    """Exact objective for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    e_sum = e.sum()
    return e / e_sum

# Analytic families – each returns a *probability* vector (sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * (np.log(x) + np.log(1 - x))
    w = np.exp(logw)
    if w.sum() == 0 or np.any(np.isinf(w)):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals /= vals.sum()
    return vals


def _distribution_power_edge(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - i - 0.5)
    eps = 1e-12
    vals = np.power(denom + eps, -alpha)
    vals /= vals.sum()
    return vals


def _distribution_V_shape(n: int) -> np.ndarray:
    centre = (n - 1) / 2.0
    vals = np.abs(np.arange(n, dtype=np.float64) - centre)
    eps = 1e-12
    vals = np.maximum(vals, eps)
    vals /= vals.sum()
    return vals


def _distribution_power_abs(n: int, beta: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    vals = np.abs(np.arange(n, dtype=np.float64) - centre) ** beta
    eps = 1e-12
    vals = np.maximum(vals, eps)
    vals /= vals.sum()
    return vals


def _random_symmetric_dirichlet(n: int, alpha: float) -> np.ndarray:
    half = (n + 1) // 2
    half_vec = np.random.dirichlet(np.full(half, alpha))
    if n % 2 == 0:
        full = np.concatenate([half_vec, half_vec[::-1]])
    else:
        full = np.concatenate([half_vec, half_vec[-2::-1]])
    full /= full.sum()
    return full

        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolve_score(prev)
            if sc < best_score:
                best_seq, best_score, best_n = prev.copy(), sc, prev.size
                print(f"[{time.time():.1f}] imported previous  score={sc:.6f}")
    except Exception:
        pass

    # fallback (should never happen)
    if best_seq is None:
        best_n = 4000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolve_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Adamised smooth‑max mirror descent
    n = best_seq.shape[0]
    y = np.log(best_seq + eps)          # work in log‑space (softmax)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8

    total_steps = 20000                # may be cut short by the time budget
    for it in range(total_steps):
        if time.time() >= deadline:
            break

        # softmax -> probability vector
        a = _softmax(y)

        # full self‑convolution
        conv = _fft_self_convolve(a)
        maxc = conv.max()

        # smooth‑max weighting
        t = it / total_steps
        alpha = 0.1 * (400.0 ** t)      # grows from ~0.1 to ~40
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed max (w.r.t. a)
        grad_a = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        # convert to gradient w.r.t. y (softmax Jacobian)
        grad_y = a * (grad_a - np.dot(grad_a, a))

        # Adam moments update
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # learning‑rate schedule (starts around 0.5, ends near 0.02)
        lr = 0.5 * (1.0 - t) + 0.02

        y -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # occasional true‑objective check
        if (it % 400 == 0) or (it == total_steps - 1):
            a_check = _softmax(y)
            sc = _max_convolve_score(a_check)
            if sc < best_score - 1e-12:
                best_seq = a_check.copy()
                best_score = sc
                best_n = n
                print(f"[{time.time():.1f}] Adam descent it={it}  best={sc:.6f}")

    # ------------------------------------------------------------------
    # 4) Focused hill‑climber (deterministic mass‑transfer)
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    hill_iters = 0
    while time.time() < deadline:
        hill_iters += 1

        # locate the index k attaining the current maximum
        k = int(conv.argmax())

        # feasible i range for i+j=k
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)

        # find the pair (i,j) that contributes the largest product
        best_i, best_j = i_low, k - i_low
        max_prod = a[best_i] * a[best_j]
        for i in range(i_low, i_high + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        max_delta = min(a[source] * 0.5, 0.015)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a_snapshot = a.copy()
        a[source] -= delta
        a[target] += delta
        a = np.maximum(a, eps)
        a /= a.sum()

        new_conv = _fft_self_convolve(a)
        new_M = new_conv.max()
        if new_M < M - 1e-13:
            M = new_M
            sc = 2.0 * n * M
            if sc < best_score - 1e-12:
                best_seq = a.copy()
                best_score = sc
                best_n = n
                print(f"[{time.time():.1f}] hill it={hill_iters}  best={sc:.6f}")
        else:
            a = a_snapshot
            # no need to recompute conv – keep old one

        # occasional jitter to shake the system
        if hill_iters % 4000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] jitter applied at hill iter {hill_iters}")

    final_score = 2.0 * best_seq.shape[0] * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
```

This implementation expands the search space (up to length 10 000), uses a richer set of analytic families, applies an Adam optimiser on a smooth‑max surrogate, and finishes with a deterministic hill‑climber to polish the solution, all while respecting the 1000 s budget.
