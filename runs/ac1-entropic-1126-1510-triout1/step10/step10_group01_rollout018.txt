def _adam_exponential_descent(a0: np.ndarray, deadline: float,
                             best_seq: np.ndarray, best_score: float,
                             lr0: float = 0.15, beta1: float = 0.9,
                             beta2: float = 0.999, eps: float = 1e-12,
                             max_iter: int = 25000,
                             eval_interval: int = 300,
                             alpha_max: float = 8.0) -> tuple[np.ndarray, float]:
    """
    Adam‑style mirror descent on the smoothed max‑convolution.
    The weight vector w is a soft‑max over the convolution values,
    with an temperature parameter α that grows linearly to α_max.
    """
    n = a0.shape[0]
    a = a0.copy()
    # initialise Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # convolution of current iterate
        conv = _fft_self_convolve(a)
        max_conv = conv.max()

        # soft‑max over conv – progressive sharpening
        alpha = alpha_max * min(1.0, t / max_iter)  # linear schedule
        w = np.exp(alpha * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:
            # fallback to uniform active set (same as original)
            mask = conv >= max_conv - 1e-14
            cnt = mask.sum()
            w = mask.astype(np.float64) / max(cnt, 1)
        else:
            w = w / w_sum

        # gradient of the surrogate max (2·a⋆w)
        grad = _gradient_from_weights(a, w)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule – decay like 1/√t
        eta = lr0 / math.sqrt(t)

        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        # Optional clipping to keep exponent stable
        step = np.clip(step, -20.0, 20.0)

        # mirror‑descent update (exponentiated gradient)
        a = a * np.exp(-step)

        # re‑normalise to the simplex (guarantees sum=1)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # occasional exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} new best {best_score:.6f}")

    return best_seq, best_score

def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer walk – keeps a change only if the true objective improves."""
    n = a.shape[0]
    while time.time() < deadline:
        # pick source (must have some mass)
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)

        # pick a distinct target
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # move a random fraction (up to 30%) of the source mass
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

                             max_iter: int = 25000, eval_interval: int = 300,
                             alpha_max: float = 8.0) -> tuple[np.ndarray, float]:
    """
    Adam‑style mirror descent on a softened max‑convolution.
    The weight vector ``w`` is a soft‑max over the convolution values
    with a temperature that linearly increases to ``alpha_max``.
    """
    n = a0.shape[0]
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _fft_self_convolve(a)
        max_conv = conv.max()

        # soft‑max over the convolution – sharpening schedule
        alpha = alpha_max * min(1.0, t / max_iter)
        w = np.exp(alpha * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:
            mask = conv >= max_conv - 1e-14
            cnt = mask.sum()
            w = mask.astype(np.float64) / max(cnt, 1)
        else:
            w = w / w_sum

        grad = _gradient_from_weights(a, w)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = lr0 / math.sqrt(t)            # 1/√t decay
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        # clip step to avoid under/overflow in the exponent
        step = np.clip(step, -20.0, 20.0)

        # exponentiated‑gradient (mirror‑descent) update
        a = a * np.exp(-step)

        # re‑project to the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer walk – keeps a move only if the true objective improves."""
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects a global time‑budget
    and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # 1. Lengths to explore (larger ≈ finer discretisation)
    lengths = [2000, 4000, 8000, 12000, 20000, 30000]

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 2. Scan analytic families for each length
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        for cand in _initial_candidates_extended(n, rng):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                continue
            cand = cand / s
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] **** new global best len={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))

        # retain a modest number of elite seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:20]

    # ------------------------------------------------------------------
    # 3. Incorporate a previously discovered construction if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s == 0.0:
                prev = _distribution_uniform(prev.shape[0])
                s = prev.sum()
            prev = prev / s
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best improved: {sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Fallback (should never happen)
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Intensive optimisation on a handful of elite seeds
    if time.time() < deadline - 120:   # reserve ≈2 min for final polishing
        elite_seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]]
        per_seed_budget = (deadline - time.time()) / max(1, len(elite_seeds))

        for seed_seq in elite_seeds:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed_budget
            seq_out, score_out = _adam_exponential_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.18,
                max_iter=25000,
                eval_interval=250,
                alpha_max=10.0,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after Adam descent new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final cheap greedy polishing
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
