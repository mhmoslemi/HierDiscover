"""

``evaluate_sequence`` score.

Algorithm
---------

   triangular, exponential, V‑shape, sine‑power, Gaussian, symmetric
   Beta, power‑edge, power‑absolute) for several lengths (1500–3000)
   together with a few random Dirichlet draws.  Keep the best candidate
   w.r.t. the cheap proxy  ``2·n·max(a*a)`` (a is normalised to sum = 1).

2. **Exact‑max sub‑gradient descent** – repeat many steps:
   * find the convolution index attaining the current maximum,
   * compute the sub‑gradient of that max (simple O(n) loop),
   * apply an exponentiated‑gradient (mirror descent) update,
   * project back onto the probability simplex.
   The learning‑rate decays from 5 down to 0.02.

3. **Smooth‑max mirror descent** – a short fine‑tuning phase identical
   to the one used previously but with a tighter schedule.

4. **Greedy hill‑climber** – repeatedly move a tiny amount of mass
   away from the pair that most contributes to the current peak.
   Accept only moves that strictly lower the peak.

All steps respect the 1000 s wall‑clock budget and intermediate best
scores are printed.  The returned list is normalised (sum = 1) and
contains only non‑negative numbers (a tiny epsilon is added to avoid
exact zeros).

import time
import random

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a vector already normalised to sum = 1.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic families – each returns a *probability* vector (sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.01) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * (np.log(x) + np.log(1 - x))
    w = np.exp(logw)
    if w.sum() == 0 or np.any(np.isinf(w)):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals /= vals.sum()
    return vals


def _distribution_power_edge(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - i - 0.5)
    eps = 1e-12
    vals = np.power(denom + eps, -alpha)
    vals /= vals.sum()
    return vals


def _distribution_V_shape(n: int) -> np.ndarray:
    centre = (n - 1) / 2.0
    vals = np.abs(np.arange(n, dtype=np.float64) - centre)
    eps = 1e-12
    vals = np.maximum(vals, eps)
    vals /= vals.sum()
    return vals


def _distribution_power_abs(n: int, beta: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    vals = np.abs(np.arange(n, dtype=np.float64) - centre) ** beta
    eps = 1e-12
    vals = np.maximum(vals, eps)
    vals /= vals.sum()
    return vals


def _random_symmetric_dirichlet(n: int, alpha: float) -> np.ndarray:
    """
    Draw a symmetric vector from Dirichlet(α) on the half‑grid and mirror it.
    """
    half = (n + 1) // 2
    half_vec = np.random.dirichlet(np.full(half, alpha))
    if n % 2 == 0:
        full = np.concatenate([half_vec, half_vec[::-1]])
    else:
        full = np.concatenate([half_vec, half_vec[-2::-1]])
    full /= full.sum()
    return full


def _random_dirichlet(n: int, alpha: float) -> np.ndarray:
    """Unsymmetric Dirichlet draw."""
    vec = np.random.dirichlet(np.full(n, alpha))
    vec /= vec.sum()
    return vec


# ----------------------------------------------------------------------
def _subgradient_step(a: np.ndarray, lr: float) -> np.ndarray:
    """
    Perform one sub‑gradient step that minimizes the current maximal
    convolution entry.  Returns the updated vector (still on the simplex
    after projection).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(conv.argmax())
    # gradient of the max‑entry (sub‑gradient)
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    if i_start <= i_end:
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        mask_eq = i_vals == j_vals
        # diagonal entries (i == j) get factor 2
        grad[i_vals[mask_eq]] = 2.0 * a[i_vals[mask_eq]]
        # off‑diagonal entries
        grad[i_vals[~mask_eq]] = a[j_vals[~mask_eq]]
    # exponentiated (mirror) update
    a = a * np.exp(-lr * grad)
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function vector that minimises ``evaluate_sequence``.
    Returns the best sequence found within the given time budget.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # tiny safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1. Scan analytic families + random draws
    lengths = [1500, 1800, 2000, 2500, 3000]
    for n in lengths:
        if time.time() >= deadline:
            break

        # --- deterministic families ------------------------------------
        families = []

        families.append(_distribution_uniform(n))
        families.append(_distribution_arcsine(n))
        families.append(_distribution_triangular(n))
        families.append(_distribution_exponential(n, lam=0.008))
        families.append(_distribution_exponential(n, lam=0.02))
        families.append(_distribution_V_shape(n))

        for pw in (0.5, 1.0, 2.0, 4.0):
            families.append(_distribution_sine_power(n, pw))

        for sigma in (0.2, 0.35, 0.5):
            families.append(_distribution_gaussian(n, sigma))

        for p in (0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 4.0):
            families.append(_distribution_beta_sym(n, p))

        for a_edge in (0.55, 0.6, 0.7, 0.8, 0.9):
            families.append(_distribution_power_edge(n, a_edge))

        for beta in (0.5, 1.0, 2.0, 3.0):
            families.append(_distribution_power_abs(n, beta))

        # --- random draws ---------------------------------------------
        # symmetric Dirichlet
        for alpha in (0.02, 0.03, 0.05, 0.08, 0.1):
            families.append(_random_symmetric_dirichlet(n, alpha))
        # unsymmetric Dirichlet (adds variety)
        for alpha in (0.02, 0.05, 0.1):
            families.append(_random_dirichlet(n, alpha))

        # evaluate all candidates for this length
        for vec in families:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] init n={n}  score={sc:.6f}")

        # extra random symmetric Dirichlet draws (more exploration)
        for _ in range(120):
            if time.time() >= deadline:
                break
            vec = _random_symmetric_dirichlet(n, alpha=random.choice([0.005, 0.01, 0.02, 0.05]))
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] rand n={n}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  Re‑use a previously computed sequence if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_seq = prev.copy()
                best_n = prev.size
                best_score = sc
                print(f"[{time.time():.1f}] global best from previous  score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback (should never happen)
    if best_seq is None:
        best_n = 2000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Exact‑max sub‑gradient descent
    a = best_seq.copy()
    n = a.shape[0]

    # number of steps limited by remaining time (rough estimate of 0.8 ms per step)
    est_step_time = 0.001
    max_iters = int(min(20000, max(0, (deadline - time.time()) / est_step_time)))
    if max_iters < 1:
        max_iters = 0

    init_lr = 5.0
    final_lr = 0.02
    if max_iters > 1:
        lr_decay = (final_lr / init_lr) ** (1.0 / (max_iters - 1))
    else:
        lr_decay = 1.0

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        lr = init_lr * (lr_decay ** it)
        a = _subgradient_step(a, lr)

        if it % 500 == 0:
            conv = _fft_self_convolve(a)
            sc = 2.0 * n * conv.max()
            if sc < best_score - 1e-12:
                best_seq = a.copy()
                best_score = sc
                best_n = n
                print(f"[{time.time():.1f}] subgrad it={it}  best={sc:.6f}")

    # ------------------------------------------------------------------
    # 5.  Smooth‑max mirror descent (short fine‑tuning)
    total_smooth = int(min(6000, max(0, (deadline - time.time()) / 0.001)))
    for it in range(total_smooth):
        if time.time() >= deadline:
            break
        t = it / max(1, total_smooth - 1)
        alpha = 0.1 * (200.0 ** t)          # ≈0.1 → 20
        lr = 5.0 * (1.0 - t) + 0.05         # ≈5 → 0.05

        conv = _fft_self_convolve(a)
        maxc = conv.max()
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]
        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()

        if it % 500 == 0:
            conv = _fft_self_convolve(a)
            sc = 2.0 * n * conv.max()
            if sc < best_score - 1e-12:
                best_seq = a.copy()
                best_score = sc
                print(f"[{time.time():.1f}] smooth it={it}  best={sc:.6f}")

    # ------------------------------------------------------------------
    # 6.  Final cheap hill‑climber
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    hill_iters = 0
    while time.time() < deadline:
        hill_iters += 1
        # index where the current maximum occurs
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        prods = a[i_vals] * a[j_vals]
        idx_max = int(prods.argmax())
        best_i = i_vals[idx_max]
        best_j = j_vals[idx_max]

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        max_delta = min(a[source] * 0.5, 0.015)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a_new = a.copy()
        a_new[source] -= delta
        a_new[target] += delta
        a_new = np.maximum(a_new, eps)
        a_new /= a_new.sum()
        conv_new = _fft_self_convolve(a_new)
        new_M = conv_new.max()
        if new_M < M - 1e-12:
            a = a_new
            conv = conv_new
            M = new_M
            sc = 2.0 * n * M
            if sc < best_score - 1e-12:
                best_seq = a.copy()
                best_score = sc
                print(f"[{time.time():.1f}] hill it={hill_iters}  best={sc:.6f}")
        else:
            # occasional jitter to escape plateaus
            if hill_iters % 5000 == 0:
                jitter = np.random.randn(n) * 1e-5
                a += jitter
                a = np.maximum(a, eps)
                a /= a.sum()
                conv = _fft_self_convolve(a)
                M = conv.max()
                print(f"[{time.time():.1f}] jitter applied at hill iter {hill_iters}")

    final_score = 2.0 * best_seq.shape[0] * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
