def _refine_sequence(seq, new_n):
    n = len(seq)
    # underlying density approximated as seq * n
    dens = seq * n  # approximates f_i values times interval length
    # create x positions of current points (midpoints)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(new_n) + 0.5) / new_n
    dens_new = np.interp(x_new, x_old, dens, left=0.0, right=0.0)
    # ensure non-negative
    dens_new = np.maximum(dens_new, 0.0)
    # convert back to sequence: a_new = dens_new / new_n
    a_new = dens_new / new_n
    a_new_sum = a_new.sum()
    if a_new_sum == 0:
        return _distribution_uniform(new_n)
    a_new = a_new / a_new_sum
    return a_new

"""
Hybrid Simulated‑Annealing + Sub‑gradient optimizer for the step‑function attack.

The algorithm works in three stages:

1. **Seed generation** – a large pool of analytic distributions (uniform,
   arcsine, symmetric/asymmetric Beta, power‑law, exponential, sparse,
   two‑block and Dirichlet) is created for a chosen length *n*.
   The best few seeds are kept.

2. **Exploratory annealing** – starting from the current best seed we perform
   many tiny mass‑transfer moves (source → target).  Moves are accepted if
   they improve the objective or with a Metropolis probability that decays
   with a temperature schedule.  This allows the search to escape local
   minima.  After a configurable fraction of the total budget a short
   exponentiated‑gradient (multimax) descent is invoked to accelerate
   convergence.

3. **Polishing** – a cheap greedy pairwise‑swap walk is applied, followed
   by a *refinement* step that interpolates the current distribution to a
   larger resolution (reducing discretisation error).  The refined sequence
   is again polished with swaps.

All work respects the global time budget (≈ 1000 s).  The function returns
the best sequence (list of non‑negative floats) found.
"""

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic families (identical to those used in the reference code)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals

def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Sub‑gradient 2·a⋆w (cross‑correlation of a with weight vector w)."""
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.size - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


def _subgradient_multimax_descent(a: np.ndarray, deadline: float,
                                 best_seq: np.ndarray, best_score: float,
                                 lr0: float = 0.5, max_iter: int = 20000,
                                 eval_interval: int = 200) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent where the sub‑gradient is averaged over
    all convolution indices attaining the maximal value (the active set).
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt
        grad = _gradient_from_weights(a, w)
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] multimax it={t} new best {best_score:.6f}")
    return best_seq, best_score


def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer walk that keeps a move only if it truly improves the objective."""
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score


def _refine_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Upsample a distribution by linear interpolation of the underlying density."""
    n = len(seq)
    # underlying density f(x) ≈ seq_i * n  (since seq_i ≈ f(x_i)/n)
    dens = seq * n
    x_old = (np.arange(n, dtype=np.float64) + 0.5) / n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    dens_new = np.interp(x_new, x_old, dens, left=0.0, right=0.0)
    dens_new = np.maximum(dens_new, 0.0)
    a_new = dens_new / new_n
    total = a_new.sum()
    if total == 0.0:
        return _distribution_uniform(new_n)
    a_new /= total
    return a_new
