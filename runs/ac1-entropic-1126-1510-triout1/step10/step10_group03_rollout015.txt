def small_opt(k, max_iters_per_restart=2000, restarts=20):

    best_local_score = inf; best_local_p = None

    for r in range(restarts):
        p = np.random.dirichlet(np.ones(k))  # sum to 1
        # momentum variables
        m = np.zeros(k)
        v = np.zeros(k)
        beta1=0.9; beta2=0.999
        for it in range(max_iters_per_restart):
            conv = _fft_self_convolve(p)
            max_c = conv.max()
            score = 2*k * max_c
            if score < best_local_score - 1e-12:
                best_local_score=score; best_local_p = p.copy()
            # approx subgradient
            tol = max_c*1e-12 + 1e-12
            active = (conv >= max_c - tol).astype(float)
            # gradient: g_i = 2 * sum_{t} active_t * partial(convolution_t)/partial p_i
            # Use convolution computed via FFT: gradient = 2 * (p_rev convolve active)[k-1:k-1+k]
            grad = 2.0 * _fft_convolve(p[::-1], active)[k-1:k-1+k]
            # Adam update
            m = beta1*m + (1-beta1)*grad
            v = beta2*v + (1-beta2)*(grad**2)

def _upsample_distribution(p, n):

    base = n // k
    rem = n % k
    a = []

"""
Propose a high‑quality non‑negative step function.

The algorithm works in three conceptual stages:

1️⃣  **Low‑dimensional discovery** –  
    Search for a probability vector `p` of small length `k`
    (k∈[4,12]) that minimises the true objective
    `2·k·max(p∗p)`.  Because `k` is tiny we can afford many
    random restarts and dozens of Adam‑style sub‑gradient steps.
    The best `p` found for each `k` is kept.

2️⃣  **Upsampling** –  
    The chosen `p` is broadcast to a much longer sequence
    (up to 40 000 entries) by replicating each component a
    (almost) equal number of times: a block of length `m`
    receives weight `p_j/m`.  This keeps the ℓ₁–norm equal to 1
    while dramatically smoothing the autocorrelation
    (the peak of `a∗a` roughly scales as `max(p∗p)/m`).

3️⃣  **Fine‑tuning** –  
    On the full‑length vector a few hundred (or a few thousand)
    projected Adam sub‑gradient steps are performed directly on
    the *true* max‑convolution (no smooth approximation).  Small
    random jitter is injected periodically to escape shallow
    plateaus.

All steps respect the time budget (`budget_s`, default 1000 s) and
the constraints checked by ``evaluate_sequence``.  The function
returns the best sequence found as a plain Python list of floats.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:       # should not happen for a feasible vector
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    For a vector already normalised to sum = 1 returns
    2·n·max(a*a) – exactly the value produced by `evaluate_sequence`.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


def _distribution_uniform(n: int) -> np.ndarray:
    """Uniform distribution on n points (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# 1️⃣  Low‑dimensional optimisation (small k)
# ----------------------------------------------------------------------


def _optimize_small_k(k: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Find a probability vector `p` of length k that minimises
    `2·k·max(p∗p)` using many random restarts and Adam‑style
    sub‑gradient steps.  Returns the best vector and its score.
    """
    eps = 1e-12
    best_p = None
    best_sc = float("inf")

    # hyper‑parameters for the Adam updates
    beta1, beta2 = 0.9, 0.999
    max_outer_iters = 200          # number of random restarts (limited by time)

    for outer in range(max_outer_iters):
        if time.time() >= deadline - 1.0:
            break

        # simple Dirichlet start (concentration 1)
        p = np.random.dirichlet(np.ones(k))
        p = np.maximum(p, eps)
        p /= p.sum()

        # Adam momentum buffers
        m = np.zeros(k)
        v = np.zeros(k)

        # inner optimisation – a few hundred cheap steps are enough for tiny k
        for it in range(500):
            conv = _fft_self_convolve(p)
            cur_max = conv.max()
            sc = 2.0 * k * cur_max
            if sc < best_sc - 1e-12:
                best_sc, best_p = sc, p.copy()
                print(
                    f"[{time.time():.1f}] k={k} new best_small {best_sc:.6f} (restart {outer})"
                )

            # sub‑gradient of the max‑convolution
            tol = cur_max * 1e-12 + 1e-12
            w = np.where(conv >= cur_max - tol, 1.0, 0.0)
            grad = 2.0 * _fft_convolve(p[::-1], w)[k - 1 : k - 1 + k]

            # Adam update (bias‑corrected)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            it1 = it + 1
            m_hat = m / (1 - beta1 ** it1)
            v_hat = v / (1 - beta2 ** it1)

            # step size – modest learning‑rate that decays with the
            # second moment (standard Adam schedule)
            step = 0.05 / (np.sqrt(v_hat) + 1e-8)
            p_new = p - step * m_hat

            # project back onto the simplex and keep positivity
            p_new = _project_onto_simplex(p_new, 1.0)
            p_new = np.maximum(p_new, eps)
            p = p_new / p_new.sum()

        # occasional large random jump to avoid getting stuck
        if outer % 20 == 19:
            jitter = np.random.randn(k) * 1e-5
            p = np.maximum(p + jitter, eps)
            p /= p.sum()

    return best_p, best_sc


# ----------------------------------------------------------------------
# 2️⃣  Upsampling of a low‑dimensional candidate
# ----------------------------------------------------------------------


def _upsample_distribution(p: np.ndarray, n: int) -> np.ndarray:
    """
    Replicate each entry of `p` a (nearly) equal number of times
    so that the resulting vector `a` has length `n` and sum = 1.
    """
    k = len(p)
    base = n // k
    rem = n % k
    blocks = []
    for idx, prob in enumerate(p):
        blk_len = base + (1 if idx < rem else 0)
        if blk_len == 0:
            continue
        # each copy gets an equal share of the block’s mass
        blocks.append(np.full(blk_len, prob / blk_len, dtype=np.float64))
    a = np.concatenate(blocks) if blocks else np.array([], dtype=np.float64)
    # numerical safety – re‑normalise to exact 1
    total = a.sum()
    if not np.isclose(total, 1.0):
        a = a / total
    return a

# 3️⃣  Fine‑tuning on the full‑size vector
# ----------------------------------------------------------------------


def _fine_tune_sequence(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Run a short Adam‑style sub‑gradient descent on the *true*
    max‑convolution of `a`.  Returns the best vector seen and its score.
    """
    eps = 1e-12
    n = a.shape[0]
    best = a.copy()
    best_sc = _max_convolution_score(best)

    # Adam buffers
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    step0 = 0.3  # base learning rate
    it = 0

    while time.time() < deadline - 0.5:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        # active set (indices attaining the max up to tolerance)
        tol = cur_max * 1e-12 + 1e-12
        w = np.where(conv >= cur_max - tol, 1.0, 0.0)
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = step0 / (np.sqrt(v_hat) + 1e-8)
        a_cand = a - step * m_hat
        a_cand = _project_onto_simplex(a_cand, 1.0)
        a_cand = np.maximum(a_cand, eps)
        a_cand /= a_cand.sum()

        sc_cand = _max_convolution_score(a_cand)
        if sc_cand < best_sc - 1e-12:
            best_sc, best = sc_cand, a_cand.copy()
            print(f"[{time.time():.1f}] fine‑tune it={it} new best {best_sc:.6f}")

        a = a_cand

        # occasional tiny random jitter to provoke new active sets
        if it % 150 == 0:
            jitter = np.random.randn(n) * 1e-6
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

    return best, best_sc

            a_fine, sc_fine = _fine_tune_sequence(a, deadline)

            if sc_fine < best_score - 1e-12:
                best_score, best_seq, best_len = sc_fine, a_fine.copy(), n_target
                print(
                    f"[{time.time():.1f}] NEW GLOBAL BEST {best_score:.6f} (len={best_len})"
                )

    # ------------------------------------------------------------------
    # Safety net – if nothing succeeded, fall back to a uniform vector.
    if best_seq is None:
        n_fallback = 5000
        best_seq = _distribution_uniform(n_fallback)
        best_seq = best_seq / best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        best_len = n_fallback
        print(
            f"[{time.time():.1f}] fallback uniform seq, score≈{best_score:.6f} (len={best_len})"
        )

    final_score = _max_convolution_score(best_seq)
    print(
        f"Final best score ≈ {final_score:.6f} (length={best_len}) – returning sequence"
    )
    return best_seq.tolist()
