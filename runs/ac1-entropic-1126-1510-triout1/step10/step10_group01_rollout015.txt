"""
Hybrid analytic‐seed + smooth‑max Adam optimizer for the step‑function
attack.  The search works as follows:

1.  Build a diverse library of analytic start‑vectors (uniform, arcsine,
    many Beta‑shapes, power‑laws, exponentials and a specially designed
    inverse‑square‑root shape that makes the self‑convolution almost
    flat).

2.  Keep the ~20 most promising seeds (lowest exact objective).

3.  For each seed run a short Adam optimisation on the *smooth* max of
    the convolution:
        t(τ) = τ·log Σ exp(conv/τ) .
    The gradient of t(τ) is an average of all convolution coefficients
    weighted by a soft‑max (temperature τ).  τ is slowly annealed so
    that the optimisation first sees a smooth landscape and finally
    focuses on the true ∞‑norm.

    The variables are the softmax parameters of the step‑function,
    guaranteeing non‑negativity and Σaᵢ=1 by construction.
    Adam’s moments are kept per‑coordinate and a simple learning‑rate
    schedule is employed.

4.  After the continuous phase a cheap greedy “pairwise‑swap” walk
    (mass transfer between two positions) is applied to polish the
    result.

The whole routine respects a global time budget (≈ 1000 s) and returns
the best sequence found as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx = s}."""
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a∗a (real FFT, length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of Σᵏ wₖ·(a∗a)ₖ w.r.t. a.
    Implements   grad_i = 2·∑ₖ wₖ·a_{k-i}.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2·n‑1
    a_rev = a[::-1]
    size = 1 << ((n + m - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    return 2.0 * conv[start:start + n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)² (sum forced to 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * (np.log(x) + np.log(1.0 - x))
    w = np.exp(logw)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    logw = ((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
            (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    w = np.exp(logw)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_inverse_sqrt_left(n: int) -> np.ndarray:
    """a_i ∝ 1/√(i+½)  (mass concentrated on the left)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt(i + 0.5)
    vals /= vals.sum()
    return vals


def _distribution_inverse_sqrt_symmetric(n: int) -> np.ndarray:
    """a_i ∝ 1/√(i+½) + 1/√(n‑i‑½) – heavy ends, light centre."""
    i = np.arange(n, dtype=np.float64)
    vals = (1.0 / np.sqrt(i + 0.5) +
            1.0 / np.sqrt(n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _smooth_max_grad(a: np.ndarray, tau: float):
    """
    Computes a smooth approximation of the max and the corresponding
    sub‑gradient.
    Returns (conv, w, grad) where
        conv = a∗a,
        w    = soft‑max(conv / τ),
        grad = 2·a⋆w (cross‑correlation).
    """
    conv = _fft_self_convolve(a)
    # stable soft‑max
    m = conv.max()
    exps = np.exp((conv - m) / tau)
    w = exps / exps.sum()
    grad = _gradient_from_weights(a, w)
    return conv, w, grad


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float,
                   best_seq: np.ndarray, best_score: float,
                   lr: float = 0.08,
                   max_iter: int = 15000,
                   temp_start: float = None,
                   temp_decay: float = 0.9995) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate.
    The optimisation lives on the simplex via a softmax parametrisation.
    """
    n = a0.shape[0]
    # initialise log‑probabilities (add a tiny epsilon to avoid log(0))
    p = np.log(np.maximum(a0, 1e-12))
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # initialise temperature (if not supplied)
    if temp_start is None:
        conv0 = _fft_self_convolve(a0)
        temp_start = max(conv0) * 0.2 + 1e-4
    tau = temp_start

    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        # current vector on the simplex
        a = np.exp(p)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum

        # smooth‑max gradient
        _, _, grad_a = _smooth_max_grad(a, tau)

        # gradient wrt softmax parameters p
        avg = np.dot(a, grad_a)
        grad_p = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        p -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        tau *= temp_decay

        # occasional exact evaluation
        if t % 200 == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer walk that accepts only improving swaps."""
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score
