"""
Step‑function optimizer (≈ 1000 s budget)

The optimiser works in three phases:

1. **Broad scan** – a collection of analytic families (uniform,
   arcsine, triangular, exponential, beta, sine‑power, Gaussian) and
   random Dirichlet draws are evaluated for a handful of lengths
   (up to ≈ 8000).  The best candidate found is kept.

2. **Intensive refinement** – starting from the current best vector a,
   we apply a cascade of updates:
      * exponential‑gradient descent with a smoothly‑approximated max,
      * projected‑gradient descent on the sub‑gradient of the true max,
      * Adam optimisation on the unconstrained logits (soft‑max
        parametrisation),
      * a cheap hill‑climber (mass transfer from the “worst’’ index to
        the “best’’ one) together with occasional jitter.
   All updates keep a on the probability simplex, so the objective
   2·n·max(a*a) is exactly the value returned by ``evaluate_sequence``
   for a normalised vector.

3. **Random restarts** – while time permits we draw fresh Dirichlet
   vectors of random length, perform a shortened version of the above
   pipeline, and keep any improvement.

The code is deliberately written with only top‑level helper functions
(no closures or lambdas) and makes heavy use of FFT‑based convolution,
which is both fast and numerically stable.  The final sequence is
returned as a plain list of floats.
"""

import time
import random
import math
import numpy as np

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _score_normalised(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Smooth approximation of the max‑operator (soft‑max)."""
    maxc = conv.max()
    w = np.exp(alpha * (conv - maxc))
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / s


def _subgradient_weights(conv: np.ndarray, eps_rel: float = 1e-12) -> np.ndarray:
    """Binary weights for a sub‑gradient of the (non‑smooth) max."""
    m = conv.max()
    w = np.zeros_like(conv)
    w[conv >= m - eps_rel * (abs(m) + 1.0)] = 1.0
    cnt = w.sum()
    if cnt == 0:
        return w
    return w / cnt


def _exponential_gradient_step(a: np.ndarray, lr: float, alpha: float) -> np.ndarray:
    """One EG update (maintains a ≥ 0, Σa = 1)."""
    conv = _fft_self_convolve(a)
    w = _smooth_weights(conv, alpha)
    grad = 2.0 * _fft_convolve(a[::-1], w)[len(a) - 1: len(a) - 1 + len(a)]
    a = a * np.exp(-lr * grad)
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def _projected_gradient_step(a: np.ndarray, eta: float,
                             active_w: np.ndarray) -> np.ndarray:
    """PGD step on the sub‑gradient defined by active_w."""
    grad = 2.0 * _fft_convolve(a[::-1], active_w)[len(a) - 1: len(a) - 1 + len(a)]
    a = a - eta * grad
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def _adam_step(logits: np.ndarray, m: np.ndarray, v: np.ndarray,
               t: int, step_size: np.ndarray):
    """Adam update on logits (unconstrained)."""
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = beta1 * m + (1.0 - beta1) * step_size
    v = beta2 * v + (1.0 - beta2) * (step_size ** 2)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    lr = 0.02 / (np.sqrt(v_hat) + eps)
    logits = logits - lr * m_hat
    logits = np.clip(logits, -30, 30)      # avoid overflow
    return logits, m, v


def _hillclimber_move(a: np.ndarray, conv: np.ndarray, eps: float = 1e-12):
    """Move a tiny chunk of mass from the worst to the best index."""
    n = a.shape[0]
    w = _subgradient_weights(conv, eps_rel=eps)
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1: n - 1 + n]
    src = int(np.argmax(grad))
    dst = int(np.argmin(grad))
    if src == dst:
        dst = random.randint(0, n - 1)
    max_delta = min(a[src] * 0.5, 0.02)
    delta = max_delta * random.random()
    if delta < eps:
        return a, conv, False
    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    new_conv = _fft_self_convolve(a_new)
    improved = new_conv.max() < conv.max() - 1e-13
    if improved:
        return a_new, new_conv, True
    else:
        return a, conv, False

# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.01) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at (n‑1)/2."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _refine_vector(a: np.ndarray, n: int, deadline: float,
                  best_info: dict, eps: float = 1e-12) -> None:
    """
    Intensive refinement pipeline.
    Updates best_info in place (keys: 'seq', 'score', 'n').
    """
    # ---- exponential‑gradient phase ---------------------------------
    a_eg = a.copy()
    steps_eg = 3000
    for it in range(steps_eg):
        if time.time() >= deadline:
            break
        t = it / steps_eg
        alpha = 0.5 * (2000.0 ** t)            # grows from 0.5 to ~2000
        lr = 0.15 * ((0.001 / 0.15) ** t)       # decays from 0.15 to 0.001
        a_eg = _exponential_gradient_step(a_eg, lr, alpha)
        if it % 500 == 0 or it == steps_eg - 1:
            sc = _score_normalised(a_eg)
            if sc < best_info['score'] - 1e-13:
                best_info.update({'seq': a_eg.copy(),
                                  'score': sc,
                                  'n': n})
                print(f"[{time.time():.1f}] EG it={it:+5d}  score={sc:.6f}")

    # ---- projected‑gradient phase ----------------------------------
    a_pg = a_eg.copy()
    steps_pg = 2000
    for it in range(steps_pg):
        if time.time() >= deadline:
            break
        conv = _fft_self_convolve(a_pg)
        w = _subgradient_weights(conv, eps_rel=1e-12)
        eta = 0.25 * ((0.001 / 0.25) ** (it / steps_pg))
        a_pg = _projected_gradient_step(a_pg, eta, w)
        if it % 400 == 0 or it == steps_pg - 1:
            sc = _score_normalised(a_pg)
            if sc < best_info['score'] - 1e-13:
                best_info.update({'seq': a_pg.copy(),
                                  'score': sc,
                                  'n': n})
                print(f"[{time.time():.1f}] PG it={it:+5d}  score={sc:.6f}")

    # ---- Adam on logits --------------------------------------------
    logits = np.log(a_pg + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    steps_adam = 3000
    for it in range(1, steps_adam + 1):
        if time.time() >= deadline:
            break
        a_adam = np.exp(logits)
        a_adam /= a_adam.sum()
        conv = _fft_self_convolve(a_adam)
        w = _subgradient_weights(conv, eps_rel=1e-12)
        grad_a = 2.0 * _fft_convolve(a_adam[::-1], w)[n - 1: n - 1 + n]
        grad_l = (grad_a - np.dot(a_adam, grad_a)) * a_adam
        logits, m, v = _adam_step(logits, m, v, it, grad_l)
        if it % 500 == 0 or it == steps_adam:
            a_adam = np.exp(logits)
            a_adam /= a_adam.sum()
            sc = _score_normalised(a_adam)
            if sc < best_info['score'] - 1e-13:
                best_info.update({'seq': a_adam.copy(),
                                  'score': sc,
                                  'n': n})
                print(f"[{time.time():.1f}] Adam it={it:+5d}  score={sc:.6f}")

    # ---- simple hill‑climber ----------------------------------------
    a_hc = best_info['seq'] if best_info['seq'].shape[0] == n else a_adam
    conv = _fft_self_convolve(a_hc)
    hc_iters = 50000
    for it in range(hc_iters):
        if time.time() >= deadline:
            break
        a_hc, conv, improved = _hillclimber_move(a_hc, conv, eps)
        if improved:
            sc = 2.0 * n * conv.max()
            if sc < best_info['score'] - 1e-13:
                best_info.update({'seq': a_hc.copy(),
                                  'score': sc,
                                  'n': n})
                if it % 500 == 0:
                    print(f"[{time.time():.1f}] HC it={it:+5d}  score={sc:.6f}")
        # occasional jitter to escape shallow plateaus
        if it % 4000 == 0 and it > 0:
            jitter = np.random.randn(n) * 1e-5
            a_hc += jitter
            a_hc = np.maximum(a_hc, eps)
            a_hc /= a_hc.sum()
            conv = _fft_self_convolve(a_hc)

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found within the
    allotted wall‑clock budget.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    eps = 1e-12
    best_info = {'seq': None, 'score': float('inf'), 'n': None}

    # ------------------------------------------------------------------
    # 1. Broad analytic scan (several lengths, many families)
    scan_lengths = [2000, 3000, 4000, 5000, 6000, 8000]
    for n in scan_lengths:
        if time.time() >= deadline:
            break
        cand = []
        # deterministic families
        cand.append(_distribution_uniform(n))
        cand.append(_distribution_arcsine(n))
        cand.append(_distribution_triangular(n))
        for lam in (0.004, 0.008, 0.012, 0.02):
            cand.append(_distribution_exponential(n, lam))
        for p in (0.1, 0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 5.0, 8.0):
            cand.append(_distribution_beta_sym(n, p))
        for pw in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 4.0):
            cand.append(_distribution_sine_power(n, pw))
        for sigma in (0.12, 0.18, 0.24, 0.30, 0.38):
            cand.append(_distribution_gaussian(n, sigma))
        # random Dirichlet draws with varied concentration
        for alpha in (0.1, 0.3, 0.5, 0.9, 1.5):
            cand.append(np.random.dirichlet(np.full(n, alpha)))

        for vec in cand:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _score_normalised(vec)
            if sc < best_info['score']:
                best_info.update({'seq': vec.copy(),
                                  'score': sc,
                                  'n': n})
                print(f"[{time.time():.1f}] init n={n} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Use the previous global solution if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _score_normalised(prev)
            if sc < best_info['score']:
                best_info.update({'seq': prev.copy(),
                                  'score': sc,
                                  'n': prev.size})
                print(f"[{time.time():.1f}] global best {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never be needed)
    if best_info['seq'] is None:
        n = 4000
        vec = _distribution_uniform(n)
        vec /= vec.sum()
        sc = _score_normalised(vec)
        best_info.update({'seq': vec, 'score': sc, 'n': n})
        print(f"[{time.time():.1f}] fallback uniform score={sc:.6f}")

    # ------------------------------------------------------------------
    # 4. Interpolation to a few larger lengths (might lower the max‑conv)
    for factor in (1.25, 1.5, 1.75, 2.0):
        if time.time() >= deadline:
            break
        n_new = int(best_info['n'] * factor)
        if n_new > 20000:
            continue
        x_old = np.linspace(0.0, 1.0, best_info['seq'].size)
        x_new = np.linspace(0.0, 1.0, n_new)
        vec = np.interp(x_new, x_old, best_info['seq'])
        vec = np.maximum(vec, eps)
        vec /= vec.sum()
        sc = _score_normalised(vec)
        if sc < best_info['score']:
            best_info.update({'seq': vec.copy(),
                              'score': sc,
                              'n': n_new})
            print(f"[{time.time():.1f}] interp factor={factor:.2f} n={n_new} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Full refinement on the current best vector
    _refine_vector(best_info['seq'], best_info['n'], deadline, best_info, eps)

    # ------------------------------------------------------------------
    # 6. Random restarts with a lighter refinement pipeline while time permits
    restart_lengths = [2000, 3000, 4000, 5000, 6000, 8000]
    while time.time() < deadline - 5.0:

        a0 = np.random.dirichlet(np.full(n, 0.5))
        a0 = np.maximum(a0, eps)
        a0 /= a0.sum()
        # a truncated version of the refinement (fewer iterations)
        # we simply call the same routine – it will terminate early because of the deadline
        _refine_vector(a0, n, deadline, best_info, eps)

    final_score = best_info['score']
    print(f"Final best score ≈ {final_score:.6f} (length={best_info['n']})")
    return best_info['seq'].tolist()
```
