def _subgradient_active_descent(a, deadline, best_seq, best_score, lr0=0.7, epsilon=1e-3, eval_interval=200):
    t=0
    n = len(a)
    while time.time() < deadline:
        t+=1
        conv = _fft_self_convolve(a)
        M=conv.max()
        if M <= 0:
            break
        tol = epsilon * M
        active = np.where(conv >= M - tol)[0]
        # compute gradient:
        grad = np.zeros_like(a)
        for k in active:
            start_i = max(0, k - (n - 1))
            end_i = min(k, n - 1)
            idx = np.arange(start_i, end_i + 1, dtype=int)
            grad[idx] += 2 * a[k - idx]
        if active.size > 0:
            grad /= active.size

        eta = lr0 / math.sqrt(t)
        a_new = a - eta * grad
        # project onto simplex
        a_new = _project_onto_simplex(a_new, s=1.0)   # sum = 1
        a = a_new

        # evaluate in intervals
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] active sd it={t} new best {best_score:.6f}")

        if t >= 400000:
            break
    return best_seq, best_score

"""
Propose a step‑function (non‑negative vector) that minimises
`evaluate_sequence`.  The search consists of three phases:

1️⃣  Scan a rich family of analytic distributions (uniform, arcsine,
    symmetric beta, edge‑weighted, triangular, cosine, exponential) for
    several lengths.  For each family a modest grid of shape parameters
    (p, decay, …) and a mixing weight with the uniform distribution is
    explored.  The best ~20 candidates (by the exact max‑convolution
    objective) are kept.

2️⃣  Refine each top candidate with a *multiple‑active‑set* mirror‑descent.
    At every iteration the maximal convolution value `M` is found; all
    indices whose value is within a relative tolerance `ε·M` are taken as
    active constraints.  The sub‑gradient is the average over these
    constraints and is computed efficiently via FFTs.  After a gradient
    step the vector is projected onto the probability simplex
    (`∑a_i = 1`, `a_i ≥ 0`).  The procedure is run until the time budget
    is exhausted.

3️⃣  A cheap greedy pairwise mass‑transfer walk finalises the best vector.

All operations work on the normalised shape (sum = 1); the evaluation
function is scale‑invariant, so the returned list can be used directly.
The whole routine stops well before the 1000 s limit, printing progress
information along the way.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    """Arcsine law on the discrete grid."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    """Symmetric triangle (ramp up then down)."""
    i = np.arange(n, dtype=np.float64) + 1
    w = np.minimum(i, n - i + 1)
    w = w.astype(np.float64)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    """Cosine‑shaped density on (0,π)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_exponential(n: int, decay: float) -> np.ndarray:
    """One‑sided exponential decaying from the left."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-decay * i)
    w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _mix_two(b1: np.ndarray, b2: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·b1 + (1‑lam)·b2."""
    return lam * b1 + (1.0 - lam) * b2


# ----------------------------------------------------------------------
def _active_subgradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.6,
    eps0: float = 5e-4,
    eval_interval: int = 250,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent using all near‑max constraints.
    The sub‑gradient is the average over every index whose convolution value
    lies within `eps·M` of the current maximum `M`.
    """
    n = a.shape[0]
    size = _next_pow2(3 * n)          # length ≥ n + (2n‑1) – 1
    t = 0
    while time.time() < deadline:
        t += 1
        # FFT of the current vector (re‑used for both steps)
        A = np.fft.rfft(a, n=size)

        # ---------- convolution + max ----------
        conv_full = np.fft.irfft(A * A, n=size)
        conv = conv_full[: 2 * n - 1]
        M = conv.max()
        # active set (relative tolerance)
        tol = eps0 * M
        active = np.where(conv >= M - tol)[0]

        # ---------- gradient from active set ----------
        mask = np.zeros(2 * n - 1, dtype=np.float64)
        mask[active] = 1.0
        M_fft = np.fft.rfft(mask, n=size)
        corr_full = np.fft.irfft(A * np.conj(M_fft), n=size)
        # correlation result needed for indices 0 … n‑1
        start = (2 * n - 1) - 1          # = 2n‑2
        grad = 2.0 * corr_full[start:start + n]
        if active.size > 0:
            grad /= float(active.size)

        # ---------- gradient step and projection ----------
        eta = lr0 / math.sqrt(t)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)

        # ---------- occasional evaluation ----------
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] active‑sgd it={t} new best {best_score:.6f}"
                )
        # safety break
        if t >= 300000:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Random mass transfers (i→j) that improve the objective."""
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * np.random.rand() * 0.35
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score

        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return plain Python list; evaluate_sequence will handle any needed clipping.
    return overall_best_seq.tolist()
