---------

   The best vector found in this phase is kept.

   The objective is a smooth approximation of the max‑convolution,

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Fast evaluation of 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * float(conv.max())

# Analytic families – each returns a *probability* vector (sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.01) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * (np.log(x) + np.log(1 - x))
    w = np.exp(logw)
    if w.sum() == 0 or np.any(np.isinf(w)):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_general(n: int, p: float, q: float) -> np.ndarray:
    """Beta(p,q) on a regular grid – p,q>0."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    w = np.exp(logw)
    # Guard against under‑/overflow
    if w.sum() == 0 or np.any(np.isinf(w)) or np.any(np.isnan(w)):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals /= vals.sum()
    return vals


def _distribution_power_edge(n: int, alpha: float) -> np.ndarray:
    """
    Edge‑heavy family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^{-α}.
    """
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - i - 0.5)
    eps = 1e-12
    vals = np.power(denom + eps, -alpha)
    vals /= vals.sum()
    return vals


def _distribution_V_shape(n: int) -> np.ndarray:
    centre = (n - 1) / 2.0
    vals = np.abs(np.arange(n, dtype=np.float64) - centre)
    eps = 1e-12
    vals = np.maximum(vals, eps)
    vals /= vals.sum()
    return vals


def _distribution_power_abs(n: int, beta: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    vals = np.abs(np.arange(n, dtype=np.float64) - centre) ** beta
    eps = 1e-12
    vals = np.maximum(vals, eps)
    vals /= vals.sum()
    return vals


def _distribution_power_left(n: int, alpha: float) -> np.ndarray:
    """Monotonically decreasing power law from left to right."""
    i = np.arange(n, dtype=np.float64)
    vals = (i + 1.0) ** (-alpha)
    vals = np.maximum(vals, 1e-12)
    vals /= vals.sum()
    return vals


def _distribution_power_right(n: int, alpha: float) -> np.ndarray:
    """Monotonically decreasing power law from right to left."""
    i = np.arange(n, dtype=np.float64)[::-1]
    vals = (i + 1.0) ** (-alpha)
    vals = np.maximum(vals, 1e-12)
    vals /= vals.sum()
    return vals


def _random_symmetric_dirichlet(n: int, alpha: float) -> np.ndarray:
    half = (n + 1) // 2
    half_vec = np.random.dirichlet(np.full(half, alpha))
    if n % 2 == 0:
        full = np.concatenate([half_vec, half_vec[::-1]])
    else:
        full = np.concatenate([half_vec, half_vec[-2::-1]])
    full /= full.sum()
    return full


def _random_asymmetric_dirichlet(n: int, alpha: float) -> np.ndarray:
    """Draw an (asymmetric) Dirichlet vector."""
    vec = np.random.dirichlet(np.full(n, alpha))
    vec /= vec.sum()
    return vec

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function vector that minimises ``evaluate_sequence``.
    Returns the best sequence found within the given time budget.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin
    eps = 1e-12

    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1. Broad analytic scan
    lengths = [2000, 2500, 3000, 3500, 4000, 4500, 5000, 6000, 8000, 10000]
    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- families ------------------------------------------------
        # uniform
        vec = _distribution_uniform(n)
        vec = np.maximum(vec, eps)
        vec /= vec.sum()
        sc = _max_convolution_score(vec)
        if sc < best_score:
            best_seq, best_score, best_n = vec.copy(), sc, n
            print(f"[{time.time():.1f}] n={n:5d} uniform      score={sc:.6f}")

        # arcsine
        vec = _distribution_arcsine(n)
        vec = np.maximum(vec, eps)
        vec /= vec.sum()
        sc = _max_convolution_score(vec)
        if sc < best_score:
            best_seq, best_score, best_n = vec.copy(), sc, n
            print(f"[{time.time():.1f}] n={n:5d} arcsine      score={sc:.6f}")

        # triangular
        vec = _distribution_triangular(n)
        vec = np.maximum(vec, eps)
        vec /= vec.sum()
        sc = _max_convolution_score(vec)
        if sc < best_score:
            best_seq, best_score, best_n = vec.copy(), sc, n
            print(f"[{time.time():.1f}] n={n:5d} triangular   score={sc:.6f}")

        # V‑shape
        vec = _distribution_V_shape(n)
        vec = np.maximum(vec, eps)
        vec /= vec.sum()
        sc = _max_convolution_score(vec)
        if sc < best_score:
            best_seq, best_score, best_n = vec.copy(), sc, n
            print(f"[{time.time():.1f}] n={n:5d} V‑shape     score={sc:.6f}")

        # exponential – several λ
        for lam in (0.005, 0.015, 0.03):
            vec = _distribution_exponential(n, lam=lam)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq, best_score, best_n = vec.copy(), sc, n
                print(f"[{time.time():.1f}] n={n:5d} exp(λ={lam})   score={sc:.6f}")

        # Gaussian – several σ
        for sigma in (0.15, 0.25, 0.4):
            vec = _distribution_gaussian(n, sigma=sigma)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq, best_score, best_n = vec.copy(), sc, n
                print(f"[{time.time():.1f}] n={n:5d} gauss(σ={sigma})score={sc:.6f}")

        # sine‑power
        for pw in (0.5, 1.0, 2.0, 4.0, 8.0):
            vec = _distribution_sine_power(n, pw)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq, best_score, best_n = vec.copy(), sc, n
                print(f"[{time.time():.1f}] n={n:5d} sine^({pw})   score={sc:.6f}")

        # power‑edge
        for a_edge in (0.55, 0.65, 0.8, 0.9, 1.0):
            vec = _distribution_power_edge(n, a_edge)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq, best_score, best_n = vec.copy(), sc, n
                print(f"[{time.time():.1f}] n={n:5d} power_edge({a_edge}) score={sc:.6f}")

        # power‑abs (edge‑heavy)
        for beta in (0.5, 1.0, 2.0, 3.0):
            vec = _distribution_power_abs(n, beta)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq, best_score, best_n = vec.copy(), sc, n
                print(f"[{time.time():.1f}] n={n:5d} power_abs({beta})  score={sc:.6f}")

        # asymmetric power‑law (left‑heavy)
        for alpha in (0.5, 1.0, 1.5, 2.0):
            vec = _distribution_power_left(n, alpha)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq, best_score, best_n = vec.copy(), sc, n
                print(f"[{time.time():.1f}] n={n:5d} left_power({alpha}) score={sc:.6f}")

        # asymmetric power‑law (right‑heavy)
        for alpha in (0.5, 1.0, 1.5, 2.0):
            vec = _distribution_power_right(n, alpha)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq, best_score, best_n = vec.copy(), sc, n
                print(f"[{time.time():.1f}] n={n:5d} right_power({alpha})score={sc:.6f}")

        # beta(p,q) with p≠q – a few skewed shapes
        for (p, q) in ((0.05, 2.0), (0.1, 3.0), (0.2, 5.0), (2.0, 0.05)):
            vec = _distribution_beta_general(n, p, q)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq, best_score, best_n = vec.copy(), sc, n
                print(f"[{time.time():.1f}] n={n:5d} Beta({p},{q}) score={sc:.6f}")

        # random Dirichlet draws – tiny concentration gives edge spikes
        for alpha in (0.02, 0.05, 0.08):
            vec = _random_asymmetric_dirichlet(n, alpha)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq, best_score, best_n = vec.copy(), sc, n
                print(f"[{time.time():.1f}] n={n:5d} Dirichlet(α={alpha}) score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  Use a previously stored sequence when available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_seq, best_score, best_n = prev.copy(), sc, prev.size
                print(f"[{time.time():.1f}] previous global best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback (should never happen)
    if best_seq is None:
        best_n = 3000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Adam‑mirrored descent on the smoothed max
    a = best_seq.copy()
    n = a.shape[0]
    total_iters = 150_000                     # adjustable; will stop early if out of time
    # Adam moments
    m_mom = np.zeros_like(a)
    v_mom = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-12

    for it in range(total_iters):
        if time.time() >= deadline:
            break

        # schedule for smoothing α (from ~0.05 to ~150)
        t_frac = it / total_iters
        alpha = 0.05 * (3000.0 ** t_frac)          # exponential growth
        # learning‑rate schedule (from ~10 down to ~0.05)
        lr = 10.0 * (1.0 - t_frac) + 0.05

        # current convolution and its maximum
        conv = _fft_self_convolve(a)
        maxc = conv.max()

        # soft‑max weights over convolution indices
        w = np.exp(alpha * (conv - maxc))           # non‑positive exponent
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed max (see analysis)
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        # Adam update (clipping to avoid huge exponentials)
        m_mom = beta1 * m_mom + (1.0 - beta1) * grad
        v_mom = beta2 * v_mom + (1.0 - beta2) * (grad ** 2)
        m_hat = m_mom / (1.0 - beta1 ** (it + 1))
        v_hat = v_mom / (1.0 - beta2 ** (it + 1))

        # element‑wise exponentiated‑gradient step
        step = -lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        step = np.clip(step, -50.0, 50.0)          # keep exponent in a safe range
        a = a * np.exp(step)

        # simplex projection and positivity safeguard
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()

        # occasional exact scoring
        if (it % 2000 == 0) or (it == total_iters - 1):
            sc = _max_convolution_score(a)
            if sc < best_score - 1e-12:
                best_seq, best_score = a.copy(), sc
                print(f"[{time.time():.1f}] Adam it={it:6d}  new best={sc:.6f}")

    # ------------------------------------------------------------------
    # 5.  Simulated annealing
    a = best_seq.copy()
    cur_score = best_score
    T = max(cur_score * 0.01, 1e-6)               # start temperature ~1 % of current score
    T_min = cur_score * 1e-8
    decay = 0.9996
    max_sa_iters = 30_000

    for it in range(max_sa_iters):
        if time.time() >= deadline:
            break
        # pick source proportional to current mass (more mass → higher chance)
        source = np.random.choice(n, p=a)
        target = random.randint(0, n - 1)
        if target == source:
            continue

        delta = min(a[source] * 0.3, 0.02) * random.random()
        if delta < eps:
            continue

        a_new = a.copy()
        a_new[source] -= delta
        a_new[target] += delta
        a_new = np.maximum(a_new, eps)
        a_new /= a_new.sum()

        new_score = _max_convolution_score(a_new)
        dE = new_score - cur_score

        if dE < 0.0 or random.random() < math.exp(-dE / T):
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_seq, best_score = a.copy(), new_score
                print(f"[{time.time():.1f}] SA it={it:5d}  new best={new_score:.6f}")

        T = max(T * decay, T_min)

    # ------------------------------------------------------------------
    # 6.  Fast hill‑climber (mass‑transfer moves)
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    hill_iters = 0
    while time.time() < deadline:
        hill_iters += 1
        # index where the current maximum occurs
        k = int(conv.argmax())
        # admissible i range for i + j = k
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        best_i, best_j = i_low, k - i_low
        max_prod = a[best_i] * a[best_j]
        for i in range(i_low, i_high + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod, best_i, best_j = prod, i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        max_delta = min(a[source] * 0.5, 0.015)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a_snapshot = a.copy()
        a[source] -= delta
        a[target] += delta
        a = np.maximum(a, eps)
        a /= a.sum()
        conv = _fft_self_convolve(a)
        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
            sc = 2.0 * n * M
            if sc < best_score - 1e-12:
                best_seq, best_score = a.copy(), sc
                print(f"[{time.time():.1f}] hill it={hill_iters:6d} best={sc:.6f}")
        else:
            a = a_snapshot
            conv = _fft_self_convolve(a)

        if hill_iters % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] jitter applied at hill iter {hill_iters}")

    final_score = 2.0 * best_seq.shape[0] * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
