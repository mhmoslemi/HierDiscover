def _subgradient_descent(a, deadline):
    n = a.shape[0]
    eps = 1e-12
    current_a = a.copy()
    current_score = _max_convolution_score(current_a)

    best_a = current_a.copy()
    best_score = current_score

    # Use Adam-like momentum
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(current_a)
        maxc = conv.max()
        tol = 1e-12 * (abs(maxc) + 1.0)
        active = np.flatnonzero(conv >= maxc - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0  # simple subgradient

        grad = 2.0 * _fft_convolve(current_a[::-1], w)[n-1:n-1+n]

        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)

        m_hat = m / (1-beta1**t)
        v_hat = v / (1-beta2**t)

        # learning rate schedule: start with lr0 maybe 0.5 and decay
        lr = 0.5 / np.sqrt(t)  # simple

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # take step (gradient descent) with projection onto simplex
        a_new = current_a - step
        a_new = _project_onto_simplex(a_new, 1.0)
        a_new = np.maximum(a_new, eps)
        a_new /= a_new.sum()

        new_score = _max_convolution_score(a_new)

        # Accept if not worse by more than small epsilon
        if new_score <= current_score + 1e-12:
            current_a = a_new
            current_score = new_score

            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = current_a.copy()
                print(f"[{time.time():.1f}] Subgrad step new best {new_score:.6f}")

        # occasional random restart from current with jitter to avoid stuck
        if t % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            current_a = np.maximum(current_a + jitter, eps)
            current_a /= current_a.sum()
            current_score = _max_convolution_score(current_a)
```

"""
Hybrid search for a non‑negative step function that minimises
`evaluate_sequence`. The algorithm proceeds in three stages:

1. **Broad exploration** – for several lengths we build many analytic
   families (uniform, arcsine, triangular, several beta‑distributions,
   sine‑power, cosine‑window, Gaussian, power‑law) together with a few
   random Dirichlet draws and sparse starts.  The best few candidates
   (lowest objective) are kept in an elite pool.

2. **Local optimisation** – each elite vector is refined by a
   *smooth‑max mirror descent* (exponentiated‑gradient on a soft‑max of
   the self‑convolution) followed by an *adaptive sub‑gradient descent*
   (Adam‑style, accepting any improvement over the current iterate) and
   finally a *multi‑target hill‑climber* that moves tiny mass from the
   most offending pair to several random locations.

3. **Finalisation** – the best vector found is normalised to sum = 1,
   clipped to non‑negative values and returned as a plain Python list.

All steps respect a total wall‑clock budget of 1000 s (a small safety
margin is reserved).  The implementation uses FFT‑based convolutions
(`O(n log n)`) and works for vectors of up to a few · 10⁴ entries.
"""

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) for a vector already normalised to sum = 1.
    This is exactly the value returned by `evaluate_sequence`.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential_centered(n, lam=0.005):
    """Symmetric exponential decay from the centre."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n, p, q):
    """Beta(p,q) on the grid, not forced symmetric."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid 0/1 exactly – the 0.5 offset already protects us
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_powerlaw(n, p):
    """Monotone decreasing a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _distribution_edges_exp(n, beta=1.0):
    """Edge‑heavy: a_i ∝ exp(beta * |i-centre|)."""
    centre = (n - 1) / 2.0
    idx = np.arange(n, dtype=np.float64)
    vals = np.exp(beta * np.abs(idx - centre) / n)
    vals /= vals.sum()
    return vals

def _smooth_max_mirror_descent(a, deadline):
    """
    Exponentiated‑gradient (mirror) descent on the smooth‑max of the
    self‑convolution.  The temperature α is increased progressively.
    """
    n = a.shape[0]
    eps = 1e-12

    # (α_start, α_end, steps, lr_start, lr_end)
    schedule = [
        (1.0, 200.0, 3000, 0.25, 0.03),
        (200.0, 5e3, 3000, 0.03, 0.003),
        (5e3, 1e5, 4000, 0.003, 3e-4),
        (1e5, 5e5, 5000, 3e-4, 5e-5),
    ]

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    for (a0, a1, steps, lr0, lr1) in schedule:
        if time.time() >= deadline:
            break
        for it in range(steps):
            if time.time() >= deadline:
                break
            t = it / steps
            alpha = a0 * (a1 / a0) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            mx = conv.max()
            w = np.exp(alpha * (conv - mx))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if it % 1000 == 0:
                sc = _max_convolution_score(a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] Mirror it={it} new best {sc:.6f}")

    return best_a, best_sc


def _adaptive_subgradient_descent(a, deadline):
    """
    Adam‑style sub‑gradient descent on the true max‑norm.
    Accepts any improvement over the *current* score (not just the global best).
    """
    n = a.shape[0]
    eps = 1e-12
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    cur_a = a.copy()
    cur_sc = _max_convolution_score(cur_a)

    best_a = cur_a.copy()
    best_sc = cur_sc

    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(cur_a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(cur_a[::-1], w)[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # learning‑rate schedule roughly 0.5 / sqrt(t)
        lr = 0.5 / math.sqrt(t)

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        a_new = cur_a - step
        a_new = _project_onto_simplex(a_new, 1.0)
        a_new = np.maximum(a_new, eps)
        a_new /= a_new.sum()

        new_sc = _max_convolution_score(a_new)

        # accept if no worse (tiny tolerance)
        if new_sc <= cur_sc + 1e-14:
            cur_a = a_new
            cur_sc = new_sc
            if new_sc < best_sc - 1e-12:
                best_sc = new_sc
                best_a = cur_a.copy()
                print(f"[{time.time():.1f}] Subgrad new best {new_sc:.6f}")

        # occasional jitter to escape plateaus
        if t % 2500 == 0:
            jitter = np.random.randn(n) * 1e-5
            cur_a = np.maximum(cur_a + jitter, eps)
            cur_a /= cur_a.sum()
            cur_sc = _max_convolution_score(cur_a)

        if time.time() + 0.5 >= deadline:
            break

    return best_a, best_sc


def _hill_climb_prune(a, deadline):
    """
    Final polishing: move a tiny amount of mass from the pair that
    attains the current maximum convolution to several random locations,
    keep the best improvement.  Also tries occasional support pruning.
    """
    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_sc = 2.0 * n * M
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- mass‑transfer step --------------------------------------
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # locate the pair (i,j) giving the largest product a_i * a_j
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j

        # try a handful of random targets and keep the best move
        max_delta = min(a[source] * 0.5, 0.03)
        best_local = None
        best_local_sc = best_sc
        for _ in range(5):
            target = random.randint(0, n - 1)
            if target == source:
                continue
            delta = max_delta * random.random()
            if delta < eps:
                continue
            a2 = a.copy()
            a2[source] -= delta
            a2[target] += delta
            a2 = np.maximum(a2, eps)
            a2 /= a2.sum()
            new_M = _fft_self_convolve(a2).max()
            new_sc = 2.0 * n * new_M
            if new_sc < best_local_sc - 1e-12:
                best_local_sc = new_sc
                best_local = a2

        if best_local is not None:
            a = best_local
            conv = _fft_self_convolve(a)
            M = conv.max()
            best_a = a.copy()
            best_sc = best_local_sc
            if it % 200 == 0:
                print(f"[{time.time():.1f}] Hill it={it} new best {best_sc:.6f}")

        # ---- occasional pruning of tiny entries ---------------------
        if it % 1500 == 0:
            # try several keep‑fraction thresholds, keep at least 2 non‑zeros
            for keep_frac in (0.99, 0.97, 0.95, 0.90):
                k_keep = max(2, int(keep_frac * n))
                idx = np.argpartition(a, -k_keep)[-k_keep:]
                a_pruned = np.zeros_like(a)
                a_pruned[idx] = a[idx]
                a_pruned /= a_pruned.sum()
                sc_pruned = _max_convolution_score(a_pruned)
                if sc_pruned < best_sc - 1e-12:
                    a = a_pruned
                    conv = _fft_self_convolve(a)
                    M = conv.max()
                    n = a.shape[0]
                    best_a = a.copy()
                    best_sc = sc_pruned

        # ---- keep the search alive ------------------------------------
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

def _generate_candidates_for_length(n):
    """Return a list of candidate vectors (sum = 1) for a given length."""
    cand = []

    # Analytic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))
    cand.append(_distribution_exponential_centered(n, lam=0.003))
    cand.append(_distribution_exponential_centered(n, lam=0.02))

    # Symmetric Beta(p,p)
    for p in (0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
        cand.append(_distribution_beta_sym(n, p))

    # Asymmetric Beta(p,q) – a few promising combos
    for p, q in ((0.02, 0.5), (0.02, 1.0), (0.05, 0.5),
                 (0.05, 2.0), (0.1, 1.0), (0.1, 3.0), (0.2, 5.0)):
        cand.append(_distribution_beta_asym(n, p, q))
        cand.append(_distribution_beta_asym(n, q, p))   # reversed version

    # Sine‑power
    for pw in (0.5, 1.0, 2.0, 4.0, 8.0, 12.0):
        cand.append(_distribution_sine_power(n, pw))

    # Cosine window
    for coeff in (0.3, 0.5, 0.7, 0.9):
        cand.append(_distribution_cosine_window(n, coeff))

    # Gaussian
    for sigma in (0.12, 0.18, 0.25, 0.35, 0.45):
        cand.append(_distribution_gaussian(n, sigma))

    # Power‑law (monotone decreasing) and its reverse
    for p in (0.8, 1.2, 1.5, 2.0, 2.5):
        pl = _distribution_powerlaw(n, p)
        cand.append(pl)
        cand.append(pl[::-1])

    # Edge‑heavy exponential
    for beta in (0.8, 1.2, 1.5):
        cand.append(_distribution_edges_exp(n, beta))

    # Random Dirichlet draws (different concentration)
    for alpha in (0.2, 0.5, 1.0, 2.0):
        vec = np.random.dirichlet(np.full(n, alpha))
        cand.append(vec)

    # Sparse random start (few non‑zeros)
    k = max(2, n // 400)
    sparse = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    sparse[idx] = np.random.rand(k)
    sparse /= sparse.sum()
    cand.append(sparse)

    # Random shuffles of a few base families – add asymmetry
    for base in (cand[0], cand[1], cand[4]):
        shuf = base.copy()
        np.random.shuffle(shuf)
        cand.append(shuf)

    return cand
