"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.  

The algorithm works in three stages:

1. **Rich initial pool** – analytic families (uniform, arcsine, symmetric
   Beta, edge‑weighted, sine‑weighted) are sampled for several lengths and
   a few random Dirichlet / sparse vectors.  The best ~30 seeds are kept.

2. **Multi‑active mirror descent** – on each of the top seeds we run an
   exponentiated‑gradient (mirror) descent on the exact objective.  Instead
   of using only the single maximal convolution index we take all
   *nearly maximal* indices (within a small fraction of the maximum) as an
   active set; the resulting sub‑gradient pushes the vector towards a flatter
   self‑convolution.  The step size decays as 1/√t and a simple jitter
   mechanism is used to escape plateaus.

3. **Final polishing** – a cheap stochastic mass‑transfer walk (pairwise
   swaps) and a simulated‑annealing style local search try to improve the
   best vector found.

The whole search respects the supplied time budget (default 1000 s).  The
returned list is a plain Python list; ``evaluate_sequence`` will take care
of normalisation, clipping, etc.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1.
    The function is homogeneous of degree 0, therefore we keep the sum
    normalised to 1 during optimisation.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight proportional to sin(π (i+½)/n)^p, vanishing at the ends."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Multi‑active mirror descent -------------------------------------------
def _subgradient_descent_multi(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.6,
    eval_interval: int = 200,
    active_frac: float = 0.01,
    max_active: int = 12,
    patience: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent where the sub‑gradient is the sum of the
    gradients belonging to all convolution indices whose value lies within
    ``active_frac`` of the current maximum.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    while time.time() < deadline and t < 300_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()
        # indices with value close to the maximum
        thresh = max_c * (1.0 - active_frac)
        act = np.where(conv >= thresh)[0]

        if act.shape[0] > max_active:
            # keep the top‑max_active entries
            top_idx = np.argsort(conv[act])[-max_active:]
            act = act[top_idx]

        # accumulate gradient from all active indices
        grad = np.zeros_like(a)
        ind_grid = np.arange(n, dtype=np.int32)
        for idx in act:
            ind = idx - ind_grid
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * a[ind[mask]]

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] subgrad_multi it={t} new global best {best_score:.6f}")
            else:
                no_improve += 1

        # if we have stagnated for a while, inject a tiny random jitter
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.005, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small mass; keep move only if objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score

        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            # metropolis acceptance
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function respects the supplied time budget (default 1000 s) and
    returns a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0           # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.   Build a diverse pool of analytic / random seeds
    lengths = [3000, 5000, 8000, 12000, 20000]
    # parameter grids
    p_grid = np.concatenate(
        [
            np.logspace(-2, -1, 9, base=10),   # 0.01 .. 0.1
            np.logspace(-1, 0, 9, base=10),    # 0.1 .. 1
            np.logspace(0, 0.5, 7, base=10),   # 1 .. 3.16
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 7)          # 0, 0.166,…,1

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families (no mixing)
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # parametrised families + mixing with uniform
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            sine = _distribution_sine_weighted(n, p)

            for base in (beta, edge, sine):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # random Dirichlet seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(
                    f"[{time.time():.1f}] new global best Dirichlet len={n} score={sc:.6f}"
                )

        # sparse random seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(
                    f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}"
                )

        # keep only the strongest seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:40]

    # ------------------------------------------------------------------
    # 2.   Incorporate a previously discovered construction if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.   Fallback guard
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.   Multi‑active mirror descent on the most promising seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] starting multi‑active descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _subgradient_descent_multi(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.55,
                eval_interval=120,
                active_frac=0.008,
                max_active=10,
                patience=3000,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after descent seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.   Greedy pairwise swaps
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=200_000,
        )

    # ------------------------------------------------------------------
    # 6.   Final stochastic local search (simulated annealing style)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final random‑local search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list; evaluate_sequence will handle further checks.
    return overall_best_seq.tolist()
