#!/usr/bin/env python
"""
Propose a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

by a three‑stage pipeline:

1.  Broad sampling – analytic families (arcsine, Beta, triangular,
    exponential, Gaussian, Cauchy, Dirichlet) on a few moderate lengths,
    together with simple deterministic re‑orderings.  All candidates are
    ℓ₁‑normalised and scored with the cheap surrogate
    `2·n·max(conv(a,a))`.

2.  Smooth Adam descent – a differentiable surrogate of the max (soft‑max
    of the convolution) is minimised with an Adam optimiser.  The
    “softness’’ parameter β is increased geometrically during optimisation,
    so the method starts with a smooth landscape and ends by accurately
    approximating the true ℓ∞‑norm.  Only a projection onto the simplex
    (non‑negative, unit sum) is required after each update.

3.  Final polishing – a short random hill‑climber followed by the
    Frank‑Wolfe pairwise mass‑transfer from the previous solution.

The whole routine respects the supplied time budget (≈ 1000 s) and returns
the best vector discovered.  It can be called repeatedly; the global
variable `height_sequence_1` (if present) is used as an additional seed.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate for `evaluate_sequence` when Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _gaussian_distribution(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    w /= w.sum()
    return w


def _cauchy_distribution(n: int, scale_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    scale = scale_factor * n
    w = 1.0 / (1.0 + ((idx - centre) / scale) ** 2)
    w /= w.sum()
    return w


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex from Dirichlet(α,…,α)."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Deterministic re‑orderings --------------------------------------------
def order_alternate_extremes(desc_vals: np.ndarray) -> np.ndarray:
    """Place the largest, second‑largest, … alternately at the left‑most
    and right‑most positions, spreading large masses."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    left = 0
    right = n - 1
    toggle = True
    for v in desc_vals:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    return out


def order_center_outwards(desc_vals: np.ndarray) -> np.ndarray:
    """Put the biggest value at the centre and then fill alternatingly
    to the left and to the right, moving outwards."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    positions = []
    mid = (n - 1) // 2
    positions.append(mid)
    for d in range(1, n):
        left = mid - d
        if left >= 0:
            positions.append(left)
        right = mid + d
        if right < n:
            positions.append(right)
    for i, pos in enumerate(positions[:n]):
        out[pos] = desc_vals[i]
    return out


# ----------------------------------------------------------------------
# Smooth (soft‑max) gradient --------------------------------------------
def _smooth_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smoothed objective
        Lβ(a) = 2·n·log‑sum‑exp(β·conv(a,a)) / β
    with respect to a, evaluated on the probability simplex.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)                # full self‑convolution
    max_b = b.max()
    # stable soft‑max weights
    w = np.exp(beta * (b - max_b))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # conv of reversed a with the weight vector
    conv_rev = _fft_convolve(a[::-1], w)
    # slice corresponds to the n coordinates of a
    grad = 4.0 * n * conv_rev[n - 1 : n - 1 + n]   # factor 4n = 2n * 2 (∂b/∂a)
    return grad


def _smooth_adam_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 5_000_000,
                         beta0: float = 0.5,
                         beta_factor: float = 1.05,
                         beta_max: float = 200.0,
                         beta_step: int = 5000,
                         step0: float = 0.2,
                         beta1: float = 0.9,
                         beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.  The inverse temperature β
    starts at `beta0` and is multiplied by `beta_factor` every `beta_step`
    iterations (capped at `beta_max`).  The routine respects `deadline`
    and returns the best vector seen (according to the exact surrogate).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best = a.copy()
    best_score = _max_convolution_score(best)
    t = 0
    beta = beta0
    while time.time() < deadline and t < max_iters:
        t += 1
        if t % beta_step == 0:
            beta = min(beta * beta_factor, beta_max)

        grad = _smooth_gradient(a, beta)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        # occasional exact evaluation to keep the best vector
        if t % 1000 == 0:
            cur = _max_convolution_score(a)
            if cur < best_score:
                best_score = cur
                best = a.copy()
    return best, best_score


# ----------------------------------------------------------------------
# Simple hill‑climbing (mass transfers) ---------------------------------
def _hill_climb(a: np.ndarray, deadline: float,
                max_iters: int = 300_000,
                epsilon: float = 0.015) -> tuple[np.ndarray, float]:
    """Randomly move a small fraction of mass between two coordinates,
    accepting only improvements of the exact surrogate."""
    best = a.copy()
    best_score = _max_convolution_score(best)
    n = best.size
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or best[i] < 1e-12:
            continue
        delta = min(best[i] * epsilon, 1.0 - best[j])
        if delta <= 0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = _max_convolution_score(cand)
        if sc < best_score:
            best = cand
            best_score = sc
    return best, best_score


# ----------------------------------------------------------------------
# Frank‑Wolfe style polishing -------------------------------------------
def _frank_wolfe_opt(a0: np.ndarray, deadline: float,
                    max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Iterated pairwise transfer from the coordinate with largest
    sub‑gradient to the one with smallest."""
    a = a0.copy()
    cur_score = _max_convolution_score(a)
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # sub‑gradient of the true max‑convolution (the same as in the
        # original code but without the diagonal correction)
        b = _fft_self_convolve(a)
        cur_max = b.max()
        active = np.flatnonzero(b >= cur_max - 1e-12 * (abs(cur_max) + 1.0))
        w = np.zeros_like(b)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[a.size - 1 : a.size - 1 + a.size]

        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        delta = min(a[i_src], 1.0 - a[i_dst])
        if delta <= 0:
            break
        step = 0.7 * delta
        a[i_src] -= step
        a[i_dst] += step
        cur_score = _max_convolution_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        # occasional jitter
        if it % 4000 == 0 and np.random.rand() < 0.1:
            a = _project_onto_simplex(a + np.random.randn(a.size) * 1e-5, s=1.0)
            cur_score = _max_convolution_score(a)
    return best, best_score

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Length(s) to consider
    try:
        base_n = len(height_sequence_1)        # type: ignore[name-defined]
    except Exception:
        base_n = 5000

    lengths = sorted({
        max(800, min(12000, int(base_n * f)))
        for f in (0.8, 1.0, 1.2)
    })

    # --------------------------------------------------------------
    # Helper to normalise and store a candidate
    candidates: list[tuple[np.ndarray, float]] = []

    def _add_candidate(arr: np.ndarray):
        a = np.asarray(arr, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            return
        a = a / s
        sc = _max_convolution_score(a)
        candidates.append((a, sc))

    # --------------------------------------------------------------
    # 1. Broad sampling
    for n in lengths:
        if time.time() >= deadline:
            break

        # arcsine
        a = _arcsine_distribution(n)
        _add_candidate(a)
        d = np.sort(a)[::-1]
        _add_candidate(order_alternate_extremes(d))
        _add_candidate(order_center_outwards(d))

        # triangular
        a = _triangular_distribution(n)
        _add_candidate(a)
        d = np.sort(a)[::-1]
        _add_candidate(order_alternate_extremes(d))
        _add_candidate(order_center_outwards(d))

        # exponential
        a = _exponential_distribution(n, lam=0.01)
        _add_candidate(a)
        d = np.sort(a)[::-1]
        _add_candidate(order_alternate_extremes(d))
        _add_candidate(order_center_outwards(d))

        # Gaussian
        a = _gaussian_distribution(n, sigma_factor=0.15)
        _add_candidate(a)
        d = np.sort(a)[::-1]
        _add_candidate(order_alternate_extremes(d))
        _add_candidate(order_center_outwards(d))

        # Cauchy
        a = _cauchy_distribution(n, scale_factor=0.15)
        _add_candidate(a)
        d = np.sort(a)[::-1]
        _add_candidate(order_alternate_extremes(d))
        _add_candidate(order_center_outwards(d))

        # symmetric Beta (a few representative p‑values)
        for p in (0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0):
            if time.time() >= deadline:
                break
            a = _beta_symmetric_distribution(n, p)
            _add_candidate(a)
            d = np.sort(a)[::-1]
            _add_candidate(order_alternate_extremes(d))

        # random Dirichlet draws
        for alpha in (0.1, 0.5, 1.0, 2.0):
            if time.time() >= deadline:
                break
            a = _random_dirichlet_alpha(n, alpha)
            _add_candidate(a)

    # --------------------------------------------------------------
    # Include previous global best if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        if prev.sum() > 0:
            prev = prev / prev.sum()
            _add_candidate(prev)
            print(f"Added previous best candidate (surrogate={candidates[-1][1]:.6f})")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Keep the best seeds
    if not candidates:
        raise RuntimeError("No candidates generated")
    candidates.sort(key=lambda x: x[1])
    top_k = min(10, len(candidates))
    top_candidates = candidates[:top_k]
    print(f"Generated {len(candidates)} candidates, keeping top {top_k}.")
    print(f"Best seed surrogate = {top_candidates[0][1]:.6f}")

    best_seq = top_candidates[0][0].copy()
    best_score = top_candidates[0][1]

    # --------------------------------------------------------------
    # 2. Smooth Adam descent (majority of the remaining budget)
    print("Running smooth Adam descent on the global best...")
    seq, sc = _smooth_adam_descent(best_seq, deadline,
                                   max_iters=5_000_000,
                                   beta0=0.5, beta_factor=1.05,
                                   beta_max=200.0, step0=0.2,
                                   beta_step=5000)
    if sc < best_score:
        best_seq, best_score = seq.copy(), sc
        print(f"Improved after smooth descent: {best_score:.6f}")

    # --------------------------------------------------------------
    # Small random perturbations + short refinements (if time permits)
    while time.time() < deadline - 30:
        pert = best_seq + np.random.randn(best_seq.size) * 0.001
        pert = np.maximum(pert, 0.0)
        pert /= pert.sum()
        seq, sc = _smooth_adam_descent(pert, deadline,
                                       max_iters=800_000,
                                       beta0=1.0, beta_factor=1.05,
                                       beta_max=200.0, step0=0.1,
                                       beta_step=4000)
        if sc < best_score:
            best_seq, best_score = seq.copy(), sc
            print(f"Perturbation improvement: {best_score:.6f}")
        else:
            break

    # --------------------------------------------------------------
    # 3. Final local search
    if time.time() < deadline - 20:
        seq, sc = _hill_climb(best_seq, deadline,
                              max_iters=400_000, epsilon=0.015)
        if sc < best_score:
            best_seq, best_score = seq.copy(), sc
            print(f"Hill‑climb improvement -> {best_score:.6f}")

    if time.time() < deadline - 10:
        seq, sc = _frank_wolfe_opt(best_seq, deadline,
                                   max_iters=300_000)
        if sc < best_score:
            best_seq, best_score = seq.copy(), sc
            print(f"Frank‑Wolfe polishing -> {best_score:.6f}")

    print(f"Final best surrogate: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
