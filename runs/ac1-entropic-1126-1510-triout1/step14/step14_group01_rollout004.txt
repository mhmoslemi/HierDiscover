The program searches for a non‑negative step‑function `a = (a₀,…,a_{n‑1})` (normalised to sum = 1) that minimises  

\[
J(a)=2\,n\;\max\bigl((a*a)_k\bigr),
\]

i.e. the exact value returned by `evaluate_sequence`.  
The search consists of three stages:

1. **Broad seeding** – families of analytic distributions (uniform, arcsine, symmetric / asymmetric Beta,
   edge‑weighted, edge‑inverse‑power, sine‑weighted, random Dirichlet, sparse random) are sampled for
   several lengths.  A dense grid for the edge‑weight exponent `p∈[10⁻³,10]` and a small mixing with the
   uniform distribution are explored.  The 15 best seeds are kept.

2. **Sub‑gradient optimisation** – a *projected exponentiated‑gradient* method that uses the *exact*
   sub‑gradient of the non‑smooth objective (the gradient of the active `max` entry).  
   Momentum, an annealing step‑size `η_t≈η₀/√t` and occasional jitter avoid stagnation.  The best seeds are
   refined for a share of the remaining budget.

3. **Upsampling & final polishing** – the current best vector is linearly interpolated to a larger
   length (up to ≈ 18000), refined again with the sub‑gradient optimiser, then improved by a deterministic
   pairwise‑mass‑transfer sweep and a cheap simulated‑annealing clean‑up.

All operations respect the 1000 s wall‑clock limit; the routine returns a plain Python list that can be
directly handed to `evaluate_sequence`.  The algorithm differs from the previous approach by (i) using the
exact `max`‑sub‑gradient instead of a log‑sum‑exp surrogate, (ii) extending the parameter search for
edge‑weighted families, and (iii) allocating more time to a high‑resolution up‑sampled refinement.

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0 , Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # Fast path – already feasible
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using real FFT."""
    n = a.shape[0]
    m = b.shape[0]
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a simplex vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric Beta(a,b)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    logw -= np.max(logw)          # numerical stability
    w = np.exp(logw)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i = (i·j)^(-p) , i = i+0.5 , j = n-i-0.5."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    """w_i = i^(-p) + j^(-p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    """k non‑zero entries uniformly placed, values drawn from Uniform(0,1)."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base , 0≤lam≤1."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Exact sub‑gradient of max(conv) and optimiser -------------------------
def _max_subgradient(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Returns (grad, max_conv) where grad = ∇_a max(conv(a,a)).
    The sub‑gradient corresponds to the average over all active
    convolution positions.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)                   # length 2n‑1
    max_c = conv.max()
    # indicator of active positions
    mask = (conv == max_c).astype(np.float64)
    cnt = mask.sum()
    if cnt == 0:
        mask[np.argmax(conv)] = 1.0
        cnt = 1.0
    mask /= cnt                                   # normalise → probability vector
    # gradient = 2 * (a_rev * mask) restricted to centre block
    a_rev = a[::-1]
    grad_rev = _fft_convolve(a_rev, mask)          # length 3n‑2
    grad = 2.0 * grad_rev[n - 1 : n - 1 + n]       # central block of length n
    return grad, max_c


def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    beta: float = 0.9,
    eval_interval: int = 120,
    patience: int = 1800,
) -> tuple[np.ndarray, float]:
    """
    Projected exponentiated‑gradient descent on the *exact* max‑objective.
    Returns the best sequence seen and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()

    while time.time() < deadline and t < 500_000:
        t += 1
        grad, max_c = _max_subgradient(a)

        # momentum (exponential moving average)
        momentum = beta * momentum + (1.0 - beta) * grad

        # step‑size schedule
        eta = lr0 / math.sqrt(t)

        # multiplicative update → stays on simplex after renormalisation
        a = a * np.exp(-eta * momentum)
        a = np.maximum(a, 1e-30)      # avoid exact zeros
        a /= a.sum()

        # periodic exact evaluation
        if t % eval_interval == 0:
            cur_score = 2.0 * n * max_c
            if cur_score + 1e-14 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] subgrad it={t:,} new best {best_score:.6f}")
            else:
                no_improve += 1

        # if stuck for a while, inject a tiny random jitter
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.002, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Deterministic pairwise mass‑transfer sweep (cheap local search)."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e}  new best {best_score:.6f}")

    return best_seq, best_score


def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Very cheap simulated‑annealing clean‑up."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e}  best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Multi‑start search for a step‑function that minimises
    evaluate_sequence(...).  Uses a broad family scan followed by
    exact sub‑gradient descent, up‑sampling, and final local polishing.
    Returns a plain Python list of non‑negative heights.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0         # small safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    elite: list[tuple[float, np.ndarray]] = []   # (score, vector)

    # ------------------------------------------------------------------
    # 1️⃣  Seed generation over several lengths
    lengths = [2000, 3000, 4000, 6000, 8000]
    # exponent grid – extended up to 10 (log‑scale)
    p_grid = np.concatenate(
        [
            np.logspace(-3, -1, 9),    # 0.001 … 0.1
            np.logspace(-1, 0, 9),     # 0.1 … 1
            np.logspace(0, 1, 12),    # 1 … 10
        ]
    )
    lam_grid = np.linspace(0.0, 0.6, 7)   # mixing with uniform (0…0.6)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] ** seed generation, n={n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            elite.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (base) {sc:.6f}")

        # parametrised families -------------------------------------------------
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            edge2 = _distribution_edge_inversepow(n, p)
            sine = _distribution_sine_weighted(n, p)

            for base in (beta, edge, edge2, sine):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    elite.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(f"[{time.time():.1f}] new best (param) n={n} p={p:.3g} lam={lam:.2f} {sc:.6f}")

        # asymmetric beta (few random draws)
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = rng.uniform(0.001, 5.0)
            b = rng.uniform(0.001, 5.0)
            cand = _distribution_beta_asym(n, a, b)
            sc = _objective(cand)
            elite.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best asym‑beta n={n} a={a:.3g} b={b:.3g} {sc:.6f}")

        # random Dirichlet
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_dirichlet(n, rng)
            sc = _objective(cand)
            elite.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best Dirichlet n={n} {sc:.6f}")

        # sparse random
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(cand)
            elite.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best sparse n={n} {sc:.6f}")

        # keep only the best few (elitism)
        elite.sort(key=lambda x: x[0])
        elite = elite[:30]

    # ------------------------------------------------------------------
    # 2️⃣  Incorporate any previously discovered construction (if present)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev = prev / prev.sum()
            sc = _objective(prev)
            elite.append((sc, prev.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3️⃣  Heavy sub‑gradient refinement on a few elite seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and elite:
        seeds = [seq for _, seq in sorted(elite, key=lambda x: x[0])[:7]]
        per_seed = remaining * 0.55 / max(1, len(seeds))
        print(f"[{time.time():.1f}] Sub‑gradient refinement on {len(seeds)} seeds, ~{per_seed:.1f}s each")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            best_seq, best_sc = _subgradient_descent(
                seed_seq.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.85,
                beta=0.9,
                eval_interval=100,
                patience=1500,
            )
            if best_sc < overall_best_score:
                overall_best_score, overall_best_seq = best_sc, best_seq.copy()
                print(f"[{time.time():.1f}] after seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Upsample to a larger length (if time permits)
    remaining = deadline - time.time()
    if remaining > 80.0:
        # choose a length up to ~18000 (2× previous, but cap)
        target_len = min(18000, max(8000, int(len(overall_best_seq) * 2)))
        print(f"[{time.time():.1f}] Upsampling best sequence to {target_len}")
        up_seq = _upsample(overall_best_seq, target_len)

        sub_dead = time.time() + remaining * 0.65
        overall_best_seq, overall_best_score = _subgradient_descent(
            up_seq,
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.65,
            beta=0.92,
            eval_interval=120,
            patience=2000,
        )
        print(f"[{time.time():.1f}] after up‑sampling best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Deterministic pairwise refinement (mass‑transfer sweep)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] deterministic pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=500_000,
        )

    # ------------------------------------------------------------------
    # 6️⃣  Final cheap simulated‑annealing clean‑up
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list – evaluate_sequence will perform final clipping
    return overall_best_seq.tolist()
