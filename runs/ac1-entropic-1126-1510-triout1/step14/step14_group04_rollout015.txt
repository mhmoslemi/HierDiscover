"""
Hybrid optimization for the “step‑function” problem.

The program first loads the best known construction (global variable
`height_sequence_1`).  It normalises the vector to the simplex, then runs
a fast, *exact* hill‑climber:

* The self‑convolution `c = a*a` is kept explicitly.
* In each iteration the sub‑gradient of the L∞‑objective
  (the set of indices where `c` attains its maximum) is obtained by a
  short FFT convolution.
* The index with the largest sub‑gradient is taken as a source of mass,
  the index with the smallest sub‑gradient as a sink.  A fraction of the
  source mass is transferred; the effect on the convolution is updated
  analytically (no full FFT per move).
* If the move strictly decreases the peak of the convolution it is
  accepted; otherwise a small random move is tried.  Periodically the
  convolution is recomputed from scratch to avoid drift.
* After the directed phase a few random pairwise swaps are performed
  for polishing.

The whole procedure respects the 1000 s budget and returns the best
sequence (as a plain list) found within that time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolutions (real inputs)
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors via a real FFT."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())  # next power of two
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a * a."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
# Simple analytic families (only used if no previous solution is present)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Edge‑weighted family:   w_i ∝ (i·(n‑i))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    s = w.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    w /= s
    return w


def _random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet sample with a random concentration."""
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a / a.sum()


# ----------------------------------------------------------------------
# Sub‑gradient computation (binary weights on the current peak)
# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Returns the (averaged) sub‑gradient of max(conv) at the current
    point `a`.  The result has the same shape as `a`.
    """
    n = a.shape[0]
    L = conv.shape[0]               # = 2n‑1
    max_val = conv.max()
    max_pos = np.where(conv == max_val)[0]

    # binary weight vector on the active positions (average over them)
    w = np.zeros(L, dtype=np.float64)
    if max_pos.size:
        w[max_pos] = 1.0 / max_pos.size

    a_rev = a[::-1]
    grad_conv = _fft_convolve(w, a_rev)      # length 3n‑2
    grad = 2.0 * grad_conv[n - 1 : n - 1 + n]   # size n
    return grad


# ----------------------------------------------------------------------
# Apply a single transfer of mass (i → j) to the convolution vector
# ----------------------------------------------------------------------
def _apply_move(conv: np.ndarray, a: np.ndarray,
                i: int, j: int, delta: float) -> np.ndarray:
    """
    Returns a new convolution vector after moving `delta` mass from `i`
    to `j`.  The formula is derived analytically, avoiding a full FFT.
    """
    n = a.shape[0]
    L = conv.shape[0]          # = 2n‑1
    new = conv.copy()

    # linear part  –2·δ·(a_shift_i – a_shift_j)
    # a_shift_i occupies indices i … i+n‑1
    new[i : i + n] -= 2.0 * delta * a
    new[j : j + n] += 2.0 * delta * a

    # quadratic correction δ² on three special positions
    new[2 * i] += delta * delta
    new[2 * j] += delta * delta
    new[i + j] -= 2.0 * delta * delta
    return new


# ----------------------------------------------------------------------
# Main optimisation loop for a single starting vector
# ----------------------------------------------------------------------
def _improve_sequence(a0: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Directed greedy descent + random polishing.
    Returns the best vector found (still normalised to sum = 1) and its
    objective value 2·n·max(conv).
    """
    a = a0.copy()
    n = a.shape[0]
    L = 2 * n - 1
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_score = 2.0 * n * M
    best_M = M

    it = 0
    no_improve = 0
    rng = np.random.default_rng(0)

    while time.time() < time_limit:
        it += 1

        # ---- sub‑gradient -------------------------------------------------
        grad = _subgradient(a, conv)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))

        if grad[i] - grad[j] < 1e-12:
            # gradient stalled – break to random polishing fallback
            break

        # ---- try a few deterministic fractions ----------------------------
        improved = False
        # fractions from large to small – give the algorithm a chance to take
        # bold steps early, then finer adjustments later.
        for frac in (0.8, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01):
            if a[i] <= 0.0:
                break
            delta = a[i] * frac
            if delta <= 0.0:
                continue

            conv_new = _apply_move(conv, a, i, j, delta)
            new_M = conv_new.max()
            if new_M < M - 1e-15:          # strict improvement
                a[i] -= delta
                a[j] += delta
                conv = conv_new
                M = new_M
                improved = True
                if 2.0 * n * M < best_score - 1e-14:
                    best_score = 2.0 * n * M
                    best_a = a.copy()
                    best_M = M
                    print(f"[{time.time():.1f}] iter {it} new best {best_score:.6f}")
                break   # accept first improving fraction

        # ---- if nothing worked, try a tiny random move --------------------
        if not improved:
            # choose random source with some mass
            src_candidates = np.where(a > 1e-14)[0]
            if src_candidates.size == 0:
                break
            i_rand = rng.choice(src_candidates)
            # choose a different target
            j_rand = rng.integers(n - 1)
            if j_rand >= i_rand:
                j_rand += 1

            delta = a[i_rand] * rng.random() * 0.03   # tiny perturbation
            conv_new = _apply_move(conv, a, i_rand, j_rand, delta)
            new_M = conv_new.max()
            if new_M < M - 1e-15:
                a[i_rand] -= delta
                a[j_rand] += delta
                conv = conv_new
                M = new_M
                if 2.0 * n * M < best_score - 1e-14:
                    best_score = 2.0 * n * M
                    best_a = a.copy()
                    best_M = M
                    print(f"[{time.time():.1f}] random tweak iter {it} best {best_score:.6f}")
                improved = True

        # ---- housekeeping -------------------------------------------------
        if it % 2000 == 0:
            # recompute convolution from scratch to curb numerical drift
            conv = _fft_self_convolve(a)
            M = conv.max()
            # also renormalise (tiny drift may happen after many moves)
            s = a.sum()
            if s != 0.0 and abs(s - 1.0) > 1e-12:
                a /= s

        if not improved:
            no_improve += 1
            if no_improve > 5000:
                # give up the directed phase after a long plateau
                break
        else:
            no_improve = 0

    # ------------------------------------------------------------------
    # Random pairwise polishing (very cheap hill‑climber)
    # ------------------------------------------------------------------
    while time.time() < time_limit:
        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.04
        if delta <= 0.0:
            continue
        conv_new = _apply_move(conv, a, i, j, delta)
        new_M = conv_new.max()
        if new_M < M - 1e-15:
            a[i] -= delta
            a[j] += delta
            conv = conv_new
            M = new_M
            if 2.0 * n * M < best_score - 1e-14:
                best_score = 2.0 * n * M
                best_a = a.copy()
                best_M = M
                print(f"[{time.time():.1f}] final swap i={i}→j={j} best {best_score:.6f}")

    return best_a, best_score

        if time.time() >= deadline:
            break
        vec = _random_dirichlet(len(seq), rng)
        seeds.append(vec)

    # ------------------------------------------------------------------
    # 3️⃣  Run the directed greedy optimiser on each seed (time sharing)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    remaining = deadline - time.time()
    if remaining <= 0:
        return seq.tolist()   # safety fallback

    per_seed = remaining / max(1, len(seeds))
    for idx, seed_vec in enumerate(seeds):
        if time.time() >= deadline:
            break
        sub_deadline = time.time() + per_seed
        a_opt, sc_opt = _improve_sequence(seed_vec, sub_deadline)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt
            print(f"[{time.time():.1f}] seed {idx} yielded new global best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Return the best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = seq
    # The evaluation routine auto‑scales, so we can keep the simplex normalisation.
    return best_seq.tolist()
```
