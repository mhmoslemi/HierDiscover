    no_improve = 0

        eta = lr0 / math.sqrt(t)

                best_seq = a.copy()

                no_improve+=1

            # jitter

            a = a + jitter

"""
Optimisation of a non‑negative step function for the objective

    J(a) = 2·n·max(conv(a,a))   with   a_i ≥ 0, Σ a_i = 1

The code follows a multi‑scale strategy:

1.  generate a diverse pool of seeds using many analytic families
    (uniform, arcsine, symmetric/asymmetric Beta, edge‑/center‑weighted,
    Gaussian, Laplace, sine‑weighted, random sparse, Dirichlet) and a
    grid of mixture parameters.

2.  keep the best few seeds and run a long annealed exponentiated‑gradient
    descent (smooth log‑sum‑exp surrogate) on each.

3.  progressively up‑sample the current best sequence to larger lengths
    (up to ~18000 entries) and continue the annealed descent – this lets
    the shape refine while the discretisation becomes finer.

4.  a final sub‑gradient (exact max) phase is performed to sharpen the
    solution once the surrogate has become very steep.

5.  deterministic pairwise mass transfers and a cheap simulated‑annealing
    sweep clean‑up any remaining local irregularities.

All steps respect a 1000 s budget; intermediate progress is printed.
The returned list is a plain Python list of non‑negative heights – the
evaluate_sequence wrapper will perform the final clipping and normalisation.
"""

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b (real FFT)."""
    n = a.shape[0]
    m = b.shape[0]
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective value 2·n·max(conv(a,a)) for a simplex vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric Beta(a,b)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^(-p), heavy at the edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_center_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{p}, heavy in the centre."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Gaussian centred at the middle, σ expressed as a fraction of n."""
    sigma = sigma_frac * n
    x = np.arange(n, dtype=np.float64) + 0.5
    centre = n / 2.0
    w = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_laplace(n: int, scale_frac: float) -> np.ndarray:
    """Laplace distribution centred in the middle."""
    scale = scale_frac * n
    x = np.arange(n, dtype=np.float64) + 0.5
    centre = n / 2.0
    w = np.exp(-np.abs(x - centre) / scale)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, where U is the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Gradient utilities ----------------------------------------------------
def _smooth_gradient(a: np.ndarray, lam: float):
    """
    Returns (grad, max_conv) where grad = ∇_a φ_λ(a)
    and max_conv = max(conv(a,a)).  a is assumed to sum to 1.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)          # length 2n‑1
    max_c = conv.max()
    z = lam * conv
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)            # stable softmax
    s = exp_z / exp_z.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, s)   # length 3n‑2
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, max_c


def _exact_subgradient(a: np.ndarray):
    """
    Sub‑gradient of the exact max‑convolution objective.
    Returns (grad, max_conv) where grad = 2 * (a_rev * w) and
    w is the (normalized) indicator of the arg‑max positions.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_c = conv.max()
    mask = (conv >= max_c - 1e-14).astype(np.float64)
    # distribute weight uniformly over all maximisers
    w = mask / mask.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, max_c

def _smooth_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    lam_start_factor: float = 0.12,
    lam_inc: float = 1.0018,
    beta: float = 0.9,
    eval_interval: int = 150,
    patience: int = 1800,
):
    """
    Annealed exponentiated‑gradient descent on the smooth log‑sum‑exp
    surrogate.  Returns the best sequence and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    lam = lam_start_factor * n
    rng = np.random.default_rng()

    while time.time() < deadline and t < 1_200_000:
        t += 1
        grad, max_conv = _smooth_gradient(a, lam)

        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # learning rate schedule
        eta = lr0 / math.sqrt(t)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * momentum)

        # keep positivity and renormalise
        a = np.maximum(a, 1e-30)
        a /= a.sum()

        # anneal λ
        lam *= lam_inc

        # periodic exact evaluation
        if t % eval_interval == 0:
            cur_score = 2.0 * n * max_conv
            if cur_score + 1e-14 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,}  new best {best_score:.6f}")
            else:
                no_improve += 1

        # jitter on plateaus
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score


def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.5,
    eval_interval: int = 200,
    patience: int = 800,
):
    """
    Mirror‑descent with the exact sub‑gradient (max‑convolution).
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        grad, max_conv = _exact_subgradient(a)
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-30)
        a /= a.sum()

        if t % eval_interval == 0:
            cur_score = 2.0 * n * max_conv
            if cur_score + 1e-14 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] sub‑grad it={t}  new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            jitter = rng.normal(scale=0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Deterministic pairwise mass‑transfer sweep."""
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e}  new best {best_score:.6f}"
            )
    return best_seq, best_score


def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Cheap simulated‑annealing clean‑up."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e}  best {best_score:.6f}"
            )
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq

                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new best (gauss/lap) n={n} σ={sigma:.3g} lam={lam:.2f} {sc:.6f}"
                        )

        # asymmetric beta --------------------------------------------------------------
        for _ in range(8):
            if time.time() >= deadline:
                break
            a = rng.uniform(0.001, 5.0)
            b = rng.uniform(0.001, 5.0)
            cand = _distribution_beta_asym(n, a, b)
            sc = _objective(cand)
            elite.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new best asym‑beta n={n} a={a:.3g} b={b:.3g} {sc:.6f}"
                )

        # random Dirichlet -------------------------------------------------------------
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_dirichlet(n, rng)
            sc = _objective(cand)
            elite.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best Dirichlet n={n} {sc:.6f}")

        # sparse random ---------------------------------------------------------------
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(cand)
            elite.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new best sparse n={n} {sc:.6f}"
                )

        # keep only the top performers
        elite.sort(key=lambda x: x[0])
        elite = elite[:30]

    # --------------------------------------------------------------
    # 2.  Include any previous solution if present
    # --------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
            elite.append((sc, prev.copy()))
    except Exception:
        pass

    if overall_best_seq is None:
        # fallback to a simple uniform seed
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform {overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 3.  Heavy annealed descent on elite seeds (short to medium lengths)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 50.0 and elite:
        seeds = [seq for _, seq in sorted(elite, key=lambda x: x[0])[:7]]
        per_seed = remaining * 0.55 / max(1, len(seeds))
        print(f"[{time.time():.1f}] annealed descent on {len(seeds)} seeds, ≈{per_seed:.1f}s each")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            a_opt, sc_opt = _smooth_descent(
                seed_seq.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                lam_start_factor=0.10,
                lam_inc=1.002,
                beta=0.9,
                eval_interval=120,
                patience=1500,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after descent seed {idx} new best {overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 4.  Progressive up‑sampling to larger lengths
    # --------------------------------------------------------------
    max_len = 18000
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 30.0:
            break
        cur_len = len(overall_best_seq)
        # target length roughly doubles each loop, but never exceed max_len
        target_len = min(max_len, int(cur_len * 1.8))
        if target_len <= cur_len:
            break
        print(f"[{time.time():.1f}] up‑sampling from {cur_len} → {target_len}")
        up_seq = _upsample(overall_best_seq, target_len)

        sub_dead = time.time() + remaining * 0.45
        up_opt, up_sc = _smooth_descent(
            up_seq,
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.6,
            lam_start_factor=0.12,
            lam_inc=1.0015,
            beta=0.92,
            eval_interval=120,
            patience=2000,
        )
        if up_sc < overall_best_score:
            overall_best_score, overall_best_seq = up_sc, up_opt.copy()
            print(f"[{time.time():.1f}] up‑sampled descent new best {overall_best_score:.6f}")

        # after the smooth phase, run a short exact sub‑gradient refinement
        sub_dead2 = time.time() + (deadline - time.time()) * 0.4
        overall_best_seq, overall_best_score = _subgradient_descent(
            overall_best_seq,
            sub_dead2,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.45,
            eval_interval=100,
            patience=600,
        )
        # loop again – the upsampling factor will shrink as we approach max_len

    # --------------------------------------------------------------
    # 5.  Deterministic pairwise refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] deterministic pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=400_000,
        )

    # --------------------------------------------------------------
    # 6.  Final cheap simulated‑annealing clean‑up
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9995,
        )

    # --------------------------------------------------------------
    # 7.  Report and return
    # --------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
