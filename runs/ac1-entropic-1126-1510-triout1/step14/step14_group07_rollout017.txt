def _mutate(seq, rng):
    a = seq.copy()
    idx_nonzero = np.where(a > 1e-12)[0]
    if idx_nonzero.size == 0:
        return a
    src = rng.choice(idx_nonzero)
    dst = rng.integers(len(a) - 1)
    if dst >= src:
        dst += 1
    delta = a[src] * rng.random() * 0.3
    a[src] -= delta
    a[dst] += delta
    # possible small negative due to rounding; enforce bounds
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

"""
Propose a non‑negative step‑function (vector of heights) that minimises
``evaluate_sequence``.

Algorithm
---------
1️⃣  **Broad seeding** –  deterministic analytic families (uniform,
    sine‑weighted, exponential, triangular, Gaussian, raised‑cosine) and
    random Dirichlet / sparse vectors are sampled for several lengths.
    The 12 best seeds per length are kept.

2️⃣  **Genetic optimisation** –  for each length a small population is
    evolved using crossover, simple mass‑transfer mutation and random
    newcomers.  The fitness is the exact objective
    ``2·n·max(conv(a,a))`` (the sum of the vector is normalised to 1).  The
    population size is modest (≈30) so many generations fit easily into
    the time budget and many candidate vectors are inspected.

3️⃣  **Adam fine‑tuning** –  the overall best vector from the GA is
    refined with a mirror‑descent Adam optimiser on a soft‑max
    parametrisation.  The true sub‑gradient of the max‑convolution is used
    (active set of the largest entries).

4️⃣  **Local polishing** –  greedy pairwise mass‑transfer swaps followed
    by a cheap simulated‑annealing pass clean up any remaining spikes.

All phases respect the global 1000 s budget.  The returned list is already
normalised (sum = 1) – ``evaluate_sequence`` will re‑normalise it anyway.
"""


import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """Exact objective for a vector normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families – each returns a probability vector (sum = 1)

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    w /= w.sum()
    return w


def _distribution_exponential_center(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(i - half))
    w /= w.sum()
    return w


def _distribution_exponential_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-rate * i)
    w /= w.sum()
    return w


def _distribution_exponential_right(n: int, rate: float) -> np.ndarray:
    return _distribution_exponential_left(n, rate)[::-1]


def _distribution_triangle(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.minimum(i + 1, n - i)
    w /= w.sum()
    return w


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    w /= w.sum()
    return w


def _distribution_raised_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / n
    w = (1.0 + np.cos(theta)) / 2.0
    w[w < 0] = 0.0
    w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator, conc: float) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a.astype(np.float64)


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Genetic operators

def _crossover(p1: np.ndarray, p2: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Binary convex crossover – result automatically sums to 1."""
    lam = rng.random()
    child = lam * p1 + (1.0 - lam) * p2
    return child  # already on simplex


def _mutate(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Simple mass‑transfer mutation."""
    a = seq.copy()
    src_candidates = np.where(a > 1e-12)[0]
    if src_candidates.size == 0:
        return a
    src = rng.choice(src_candidates)
    dst = rng.integers(len(a) - 1)
    if dst >= src:
        dst += 1
    delta = a[src] * rng.random() * 0.35
    if delta <= 0:
        return a
    a[src] -= delta
    a[dst] += delta
    # Project back onto simplex (numerical drift may appear)
    a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
# Adam‑based mirror descent (same as in the reference code)

def _adam_refine(
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    lr0: float = 0.9,
    active_frac: float = 0.02,
    max_active: int = 30,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_interval: int = 250,
    best_seq: np.ndarray | None = None,
    best_score: float = float("inf"),
) -> tuple[np.ndarray, float]:
    """
    Optimise a probability vector using Adam on log‑softmax parameters.
    The exact objective (max‑convolution) is used.
    """
    n = init_a.shape[0]
    x = np.log(np.maximum(init_a, 1e-12))          # log‑weights
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    ind_grid = np.arange(n, dtype=np.int32)
    t = 0
    if best_seq is None:
        best_seq = init_a.copy()
        best_score = _objective(init_a)

    while time.time() < deadline and t < max_iters:
        t += 1
        a = np.exp(x - np.max(x))
        a /= a.sum()
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)
        act = np.where(conv >= thresh)[0]
        # keep only the largest few active indices
        if act.size > max_active:
            top = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top]

        # sub‑gradient of the max (sum over active indices)
        grad_a = np.zeros_like(a)
        for idx_k in act:
            offset = idx_k - ind_grid
            mask = (offset >= 0) & (offset < n)
            grad_a[mask] += 2.0 * a[offset[mask]]

        # transform gradient to log‑softmax space
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = lr0 / math.sqrt(t)
        x -= eta * m_hat / (np.sqrt(v_hat) + eps)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] adam it={t:,} new best {best_score:.6f}")

    best_seq = _project_onto_simplex(best_seq, s=1.0)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Local refinements (pairwise swap + simulated annealing)

def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.01,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Entry point – returns a plain Python list of non‑negative heights."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣  Seed generation for several lengths
    lengths = [2000, 3000, 4000, 6000, 8000, 12000]
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] seeding length {n}")

        seeds = []

        # deterministic families
        seeds.append(_distribution_uniform(n))
        seeds.append(_distribution_sine_weighted(n, p=1.0))
        seeds.append(_distribution_exponential_center(n, rate=0.005))
        seeds.append(_distribution_exponential_left(n, rate=0.003))
        seeds.append(_distribution_exponential_right(n, rate=0.003))
        seeds.append(_distribution_triangle(n))
        seeds.append(_distribution_gaussian(n, sigma=n / 6.0))
        seeds.append(_distribution_raised_cosine(n))

        # parametric sweeps
        for p in np.logspace(-0.5, 1.0, 5):
            seeds.append(_distribution_sine_weighted(n, p))
        for rate in np.linspace(0.001, 0.02, 5):
            seeds.append(_distribution_exponential_center(n, rate))
            seeds.append(_distribution_exponential_left(n, rate))
            seeds.append(_distribution_exponential_right(n, rate))
        for sigma in np.linspace(n / 12.0, n / 3.0, 4):
            seeds.append(_distribution_gaussian(n, sigma))

        # random Dirichlet with various concentrations
        for conc in [0.05, 0.2, 1.0, 5.0]:
            for _ in range(2):
                seeds.append(_distribution_dirichlet(n, rng, conc))

        # sparse random
        for _ in range(2):
            seeds.append(_distribution_random_sparse(n, rng, density=0.004))

        # evaluate all seeds and keep the 12 best for this length
        evaluated = []
        for s in seeds:
            s = s / s.sum()
            sc = _objective(s)
            evaluated.append((sc, s))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, s.copy()
                print(f"[{time.time():.1f}] new global best (seed) len={n} score={sc:.6f}")

        evaluated.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in evaluated[:12]]

        # ------------------------------------------------------------------
        # 2️⃣  Genetic optimisation (population ≈30)
        pop_size = 30
        population = top_seeds[:]
        # pad population with random Dirichlet vectors if needed
        while len(population) < pop_size and time.time() < deadline:
            conc = rng.uniform(0.05, 5.0)
            population.append(_distribution_dirichlet(n, rng, conc))

        # main GA loop – allocate a slice of the remaining budget for this length
        slice_deadline = deadline - 30.0  # reserve time for later refinements
        while time.time() < slice_deadline:
            # ---- create children -------------------------------------------------
            children = []

            # crossover children (pop_size of them)
            for _ in range(pop_size):
                i, j = rng.choice(len(population), size=2, replace=False)
                child = _crossover(population[i], population[j], rng)
                children.append(child)

            # mutation children (pop_size // 2)
            for _ in range(pop_size // 2):
                i = rng.integers(len(population))
                child = _mutate(population[i], rng)
                children.append(child)

            # fresh random individuals (pop_size // 3)
            for _ in range(pop_size // 3):
                conc = rng.uniform(0.05, 5.0)
                child = _distribution_dirichlet(n, rng, conc)
                children.append(child)

            # ---- evaluate & select elite -----------------------------------------
            candidates = population + children
            cand_scores = [_objective(c) for c in candidates]

            # keep the best pop_size vectors
            idx_keep = np.argsort(cand_scores)[:pop_size]
            population = [candidates[i] for i in idx_keep]

            # update global best
            if cand_scores[idx_keep[0]] < overall_best_score - 1e-12:
                overall_best_score = cand_scores[idx_keep[0]]
                overall_best_seq = population[0].copy()
                print(f"[{time.time():.1f}] GA new global best len={n} score={overall_best_score:.6f}")

            # early exit if we beat the target comfortably
            if overall_best_score < 1.503:
                break

        # stop early if target hit
        if overall_best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # 3️⃣  Adam fine‑tuning of the overall best vector (if any)
    if overall_best_seq is not None and time.time() < deadline:
        remain = deadline - time.time()
        # keep ~20 s for Adam, the rest for polishing
        adam_deadline = time.time() + min(20.0, max(5.0, remain - 30.0))
        print(f"[{time.time():.1f}] final Adam refinement (≈{adam_deadline - time.time():.1f}s left)")
        overall_best_seq, overall_best_score = _adam_refine(
            overall_best_seq.copy(),
            adam_deadline,
            max_iters=250_000,
            lr0=0.9,
            active_frac=0.015,
            max_active=30,
            beta1=0.9,
            beta2=0.999,
            eval_interval=200,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] after Adam score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Greedy pairwise refinement
    if overall_best_seq is not None and time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=200_000,
        )
        print(f"[{time.time():.1f}] after swaps score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final cheap simulated‑annealing polishing
    if overall_best_seq is not None and time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing polishing")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.015,
            decay=0.9997,
        )
        print(f"[{time.time():.1f}] final best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    if overall_best_seq is None:
        # fallback – return a simple uniform vector (should never happen)
        overall_best_seq = _distribution_uniform(2000)

    # ``evaluate_sequence`` will renormalise, but we keep a proper probability.
    return overall_best_seq.tolist()
