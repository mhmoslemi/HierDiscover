def _smooth_gradient_descent(a, deadline, best_seq, best_score):
    n = a.shape[0]
    t = 0
    beta = 0.5
    eta0 = 1.0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        # compute softmax weights
        # subtract max_conv for numerical stability
        exps = np.exp(beta * (conv - max_conv))
        sum_exps = exps.sum()
        w = exps / sum_exps  # shape 2n-1

        # compute gradient using correlation
        w_rev = w[::-1]
        # maybe use FFT convolution for speed
        # compute conv_rev = np.convolve(a, w_rev) works but may be O(n^2). Use FFT:
        size = 1 << ((n + (2*n-1) - 1) - 1).bit_length()
        A = np.fft.rfft(a, n=size)
        W_rev = np.fft.rfft(w_rev, n=size)
        conv2 = np.fft.irfft(A * W_rev, n=size)
        # conv2 length size, we need length n + (2n-1) -1 = 3n-2; but rfft size yields length=size, but we have zero-padded. We'll get correct values.
        # now extract gradient

        g = 2.0 * conv2[idx_start]
        # exponentiated update
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)
        # renorm
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0/n)
        else:
            a = a / s

        # schedule increase of beta:
        if t % 1000 == 0:

                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth EG it={t} beta={beta:.3f} new best {best_score:.6f}")

    return best_seq, best_score

def _distribution_mixed_edges(n, p_left, p_right, alpha_left, alpha_right, lam):
    left = (np.arange(n) + 0.5) ** (-p_left)
    right = ((n - np.arange(n) - 0.5)) ** (-p_right)
    mix = alpha_left * left + alpha_right * right

        uniform = np.full(n, 1.0/n)
        mix = lam * uniform + (1 - lam) * mix
    mix_sum = mix.sum()
    mix = mix / mix_sum
    return mix

"""Step‑function optimizer for ``evaluate_sequence``.

The search proceeds in three stages:
1.  Build a diverse pool of “seed” distributions using several analytic families
    (uniform, arcsine, edge‑weighted, sum‑edge‑weighted, generalized‑normal,
    exponential, asymmetric left/right mixtures, random sparse Dirichlet, …).
    The 20 best seeds are retained.
2.  Refine each retained seed with a *smooth* exponentiated‑gradient descent.
    The objective 2·n·max(conv)/sum² is approximated by a
    log‑sum‑exp of the convolution entries; the temperature (β) is slowly
    increased so that the method starts from a smooth landscape and ends
    targeting the true max.  The update is mirror‑descent on the simplex
    (multiplicative weights) which guarantees non‑negativity.
3.  A short pairwise‑mass‑transfer simulated annealing polish finishes the run.

The whole procedure respects the 1000 s wall‑time limit; the best sequence
found is returned as a plain ``list[float]`` (its scale is irrelevant because
the objective is scale‑invariant)."""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using the next power‑of‑two FFT size."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """Exact objective used by the evaluator (scale‑invariant)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
#  Analytic families (all return a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i·(n‑i))^{‑p}   with i = distance to the left edge (½‑based)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = n - np.arange(n, dtype=np.float64) - 0.5
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_sum_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ i^{‑p} + (n‑i)^{‑p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = n - np.arange(n, dtype=np.float64) - 0.5
    w = (i ** (-p)) + (j ** (-p))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(‑|x‑½|^q) on the discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_exponential(n: int, lam: float) -> np.ndarray:
    """Exponential decay w_i ∝ exp(‑lam·i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-lam * i)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_mixed_edges(
    n: int,
    p_left: float,
    p_right: float,
    alpha_left: float,
    alpha_right: float,
    lam: float,
) -> np.ndarray:
    """Asymmetric mixture:
       w_i ∝ α_left·(i+½)^{-p_left} + α_right·(n‑i‑½)^{-p_right},
       optionally mixed with a uniform part."""
    i = np.arange(n, dtype=np.float64) + 0.5
    left = i ** (-p_left)
    right = (n - i) ** (-p_right)
    w = alpha_left * left + alpha_right * right
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    if lam > 0.0:
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        w = lam * uniform + (1.0 - lam) * w
        w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1‑lam) + uniform·lam."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
#  Smooth (log‑sum‑exp) gradient of the max‑convolution term
# ----------------------------------------------------------------------
def _smooth_exponential_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    beta0: float = 0.5,
    beta_max: float = 30.0,
    beta_factor: float = 1.002,
    eval_every: int = 150,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smooth approximation
        f_β(a) = (1/β)·log Σ_k exp(β·conv_k(a))
    of the max‑convolution.  ``β`` is increased gradually (annealing)
    so the method first sees a smooth landscape and ends up optimising the
    true max.  The update is multiplicative, keeping a on the probability simplex.
    """
    n = a.shape[0]
    size = 1 << ((3 * n - 2) - 1).bit_length()   # FFT size ≥ 3n‑2
    beta = beta0
    t = 0

    while time.time() < deadline:
        t += 1

        # FFT of the current vector – reuse in both convolutions
        A = np.fft.rfft(a, n=size)

        # 1) self‑convolution a*a  (for the soft‑max weights)
        conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]

        # soft‑max over convolution entries
        m = conv.max()
        exp_vals = np.exp(beta * (conv - m))          # stable exp
        w = exp_vals / exp_vals.sum()                 # shape 2n‑1, Σ w = 1

        # 2) gradient of the smooth max:
        #    g = Σ_k w_k * ∇ conv_k  = 2 * (a  ⋆  w_rev)  on the interval [0,n‑1]
        w_rev = w[::-1]
        W_rev = np.fft.rfft(w_rev, n=size)
        conv_full = np.fft.irfft(A * W_rev, n=size)   # length ≥ 3n‑2
        idx = (2 * n - 2) - np.arange(n)              # indices picking the right slice
        grad = 2.0 * conv_full[idx]

        # exponentiated‑gradient step on the simplex
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # renormalise (stay on the simplex)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # schedule increase of β (more focus on the true max)
        if beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        # occasional evaluation of the true objective
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth EG it={t} β={beta:.3f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
#  Pairwise‑mass‑transfer simulated annealing (final polish)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass from one index to another,
    accepting worsened moves with a decreasing temperature."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time

    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score

                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best mix L={p_left:.2f} R={p_right:.2f} lam={lam:.2f} score={sc:.6f}")

        # random sparse Dirichlet seeds ---------------------------------------------
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = np.zeros(n, dtype=np.float64)
            k = max(2, int(n * 0.005))                 # 0.5 % non‑zeros
            idx = rng.choice(n, size=k, replace=False)
            cand[idx] = rng.random(k)
            cand /= cand.sum()
            sc = _objective(cand)
            top.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best sparse score={sc:.6f}")

        # keep only the very best seeds (size 20)
        top.sort(key=lambda x: x[0])
        top = top[:20]

    # ------------------------------------------------------------------
    # 2.  Include previously submitted construction if it exists
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.abs(prev)
            if prev.sum() == 0.0:
                prev = np.full_like(prev, 1.0 / prev.shape[0])
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < best_score:
                best_score, best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Guard against an empty result
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _distribution_uniform(5000)
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Smooth exponentiated‑gradient refinement on the retained seeds
    # ------------------------------------------------------------------
    top.sort(key=lambda x: x[0])
    while time.time() < deadline - 10.0 and top:
        # allocate a modest chunk of time to each remaining seed
        remaining = deadline - time.time()
        sub_time = min(60.0, remaining / len(top))
        if sub_time < 5.0:
            break

        _, seed_seq = top.pop(0)
        sub_deadline = time.time() + sub_time
        print(f"[{time.time():.1f}] smooth EG on seed len={len(seed_seq)} time={sub_time:.1f}s")
        refined_seq, refined_score = _smooth_exponential_gradient_descent(
            seed_seq.copy(),
            sub_deadline,
            best_seq,
            best_score,
            eta0=0.9,
            beta0=0.5,
            beta_max=25.0,
            beta_factor=1.002,
            eval_every=150,
        )
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq.copy()
            print(f"[{time.time():.1f}] new global best after smooth EG {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing polish")
        best_seq, best_score = _pairwise_annealing(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            rng,
        )
        print(f"[{time.time():.1f}] after annealing best score {best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective = 2·n·max(conv)/sum²): {best_score:.6f} "
        f"(length={len(best_seq)})"
    )
    return best_seq.tolist()
