def _fft_convolve(a, b):
    # returns full convolution of real vectors a,b.
    na = len(a); nb = len(b)
    need = na + nb - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[:need]
```

"""
Propose a non‑negative step‑function (a probability vector) that minimises
``evaluate_sequence``.

Algorithm
---------
1️⃣  **Seed generation** – A handful of analytically‑motivated families
    (uniform, sine‑weighted, centre‑exponential, left/right‑skewed
    exponentials, quadratic, Dirichlet, sparse random) are sampled for
    several lengths.  The best ~12 candidates (by the exact objective)
    are retained.

2️⃣  **Smooth‑max Adam** – Each retained seed is short‑run optimised with
    Adam on the *smooth* approximation of the max‑convolution:
        sβ(a)= (1/β)·log∑exp(β·(a*a)).
    The gradient of sβ is a soft‑weighted sum of the individual convolution
    gradients and can be computed by a single extra FFT.  The temperature
    β is increased (2, 5, 10, 20, 40, 80) so the optimisation first sees
    a smooth landscape and then focuses on the true peak.

3️⃣  **Long exact Adam** – The best sequence from step 2 is refined with the
    original sub‑gradient Adam (active‑set gradient) for the remaining
    time budget.

4️⃣  **Local polishing** – A greedy pairwise mass‑transfer step followed by
    a cheap simulated‑annealing style search removes any remaining spikes.

All steps obey the global 1000 s budget.  The returned list is a plain
Python list; ``evaluate_sequence`` will re‑normalise it automatically.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


# ----------------------------------------------------------------------
# FFT‑based convolutions
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b (real input, real output)."""
    na, nb = a.shape[0], b.shape[0]
    need = na + nb - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[:need]


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a."""
    n = a.shape[0]
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
# Exact objective (assumes Σ a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact evaluation: 2·n·max(conv(a,a))."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector – sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    w /= w.sum()
    return w


def _distribution_exponential_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-rate * i)
    w /= w.sum()
    return w


def _distribution_exponential_right(n: int, rate: float) -> np.ndarray:
    return _distribution_exponential_left(n, rate)[::-1]


def _distribution_exponential_center(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(i - half))
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    """Symmetric quadratic bump: a_i ∝ (i+1)*(n-i)."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 1) * (n - i)
    w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator, conc: float) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a.astype(np.float64)


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Adam optimiser on the *smooth* max (log‑sum‑exp) objective
# ----------------------------------------------------------------------
def _smoothmax_adam(
    init_a: np.ndarray,
    deadline: float,
    betas: list[float],
    max_iters: int = 250_000,
    lr0: float = 0.6,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_interval: int = 300,
    best_seq: np.ndarray | None = None,
    best_score: float = float("inf"),
) -> tuple[np.ndarray, float]:
    """
    Optimise a probability vector using Adam on the *smooth* max objective

        sβ(a) = (1/β)·log∑_k exp(β·(a*a)_k)

    The gradient of sβ is a soft‑weighted sum of the individual
    convolution gradients and can be obtained by a single extra FFT.
    The schedule of β values (starting small, ending large) turns the
    problem from a smooth one into the original max‑convolution problem.
    """
    n = init_a.shape[0]
    # Initialise log‑weights
    x = np.log(np.maximum(init_a, 1e-12))
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    best_seq_local = init_a.copy() if best_seq is None else best_seq.copy()
    best_score_local = _objective(best_seq_local) if best_seq is None else best_score

    # Pre‑allocate a_rev (reversed a) because it will be needed each iteration
    for beta in betas:
        # Allocate a time slice for this beta (fairly generous)
        slice_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.2)
        t = 0
        while time.time() < slice_deadline and t < max_iters:
            t += 1
            a = _softmax(x)                     # stays on the simplex
            conv = _fft_self_convolve(a)         # length 2n‑1

            # ------- smooth max weights -------
            y = conv * beta
            y_max = np.max(y)
            expy = np.exp(y - y_max)            # stabilise
            w = expy / expy.sum()               # softmax over convolution entries

            # ------- gradient wrt a -------
            a_rev = a[::-1]
            full = _fft_convolve(w, a_rev)      # length 3n‑2
            grad_a = 2.0 * full[n - 1 : 2 * n - 1]   # slice of length n

            # ------- convert to gradient wrt log‑weights -------
            dot = np.dot(a, grad_a)
            grad_x = a * (grad_a - dot)

            # ------- Adam update -------
            m = beta1 * m + (1.0 - beta1) * grad_x
            v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            eta = lr0 / math.sqrt(t)
            x -= eta * m_hat / (np.sqrt(v_hat) + eps)

            # ------- occasional exact evaluation -------
            if t % eval_interval == 0:
                cur = _objective(a)
                if cur < best_score_local - 1e-14:
                    best_score_local = cur
                    best_seq_local = a.copy()
                    print(f"[{time.time():.1f}] β={beta:.1f} it={t:,} new best {best_score_local:.6f}")

    best_seq_local = _project_onto_simplex(best_seq_local, s=1.0)
    return best_seq_local, best_score_local


# ----------------------------------------------------------------------
# Adam optimiser on the *true* max‑convolution (active‑set sub‑gradient)
# ----------------------------------------------------------------------
def _adam_refine(
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    lr0: float = 0.8,
    active_frac: float = 0.025,
    max_active: int = 30,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_interval: int = 250,
    best_seq: np.ndarray | None = None,
    best_score: float = float("inf"),
) -> tuple[np.ndarray, float]:
    """Exact‑max Adam (the routine from the reference implementation)."""
    n = init_a.shape[0]
    x = np.log(np.maximum(init_a, 1e-12))
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    ind_grid = np.arange(n, dtype=np.int32)

    if best_seq is None:
        best_seq = init_a.copy()
        best_score = _objective(init_a)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        a = _softmax(x)
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            top = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top]

        # sub‑gradient from the active set
        grad_a = np.zeros_like(a)
        for idx_k in act:
            offset = idx_k - ind_grid
            mask = (offset >= 0) & (offset < n)
            grad_a[mask] += 2.0 * a[offset[mask]]

        # convert to gradient wrt log‑weights
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update with decaying step‑size
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = lr0 / math.sqrt(t)
        x -= eta * m_hat / (np.sqrt(v_hat) + eps)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] adam it={t:,} new best {best_score:.6f}")

    best_seq = _project_onto_simplex(best_seq, s=1.0)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Local pairwise mass‑transfer refinement
# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Cheap simulated‑annealing polishing
# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

                    print(f"[{time.time():.1f}] new global best Dirichlet conc={conc:.3g} score={sc:.6f}")

        # sparse random seeds
        for _ in range(5):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.004)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}")

        # keep only the very best seeds (12 of them)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:12]

    # ------------------------------------------------------------------
    # Include a previously saved construction if available
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2️⃣  Smooth‑max Adam refinement on the retained seeds
    # ------------------------------------------------------------------
    refined: list[tuple[float, np.ndarray]] = []
    beta_schedule = [2.0, 5.0, 10.0, 20.0, 40.0, 80.0]

    for sc, cand in top_candidates:
        if time.time() >= deadline:
            break
        slice_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.1)
        best_seq, best_sc = _smoothmax_adam(
            cand.copy(),
            slice_deadline,
            betas=beta_schedule,
            max_iters=20_000,
            lr0=0.6,
            beta1=0.9,
            beta2=0.999,
            eval_interval=200,
            best_seq=cand,
            best_score=sc,
        )
        refined.append((best_sc, best_seq))

    if refined:
        overall_best_score, overall_best_seq = min(refined, key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 3️⃣  Long exact‑max Adam descent from the overall best seed
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        long_deadline = time.time() + max(0.0, remaining - 30.0)
        print(f"[{time.time():.1f}] start long Adam descent (≈{remaining-30:.1f}s left)")
        overall_best_seq, overall_best_score = _adam_refine(
            overall_best_seq.copy(),
            long_deadline,
            max_iters=500_000,
            lr0=0.9,
            active_frac=0.02,
            max_active=30,
            beta1=0.9,
            beta2=0.999,
            eval_interval=250,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] after long descent best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Greedy pairwise refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 5️⃣  Final simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.01,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(f"Final best score (exact) ≈ {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # ``evaluate_sequence`` will re‑normalise, but we keep the vector a true probability.
    return overall_best_seq.tolist()
