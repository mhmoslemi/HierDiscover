"""
Optimise a non‑negative step function (list of heights) for the
``evaluate_sequence`` objective.

Strategy
--------
1️⃣  Broad seeding – generate many analytically motivated families
    (exponential, double‑ended, power‑law, reciprocal‑min,
    beta‑U‑shape, logistic, uniform, Dirichlet, sparse).  The 20
    best seeds (lowest exact objective) are retained.

2️⃣  Mirror‑descent (Adam) – each retained seed is refined with a
    projected Adam optimiser that works directly on the exact
    objective.  The optimizer uses the active set of convolution
    entries that are within a few percent of the current maximum,
    i.e. a sub‑gradient of the max‑operator.  Short runs are performed
    for every seed, then a long final run from the overall best seed.

3️⃣  Local polishing – greedy pairwise mass‑transfer moves followed by
    a cheap simulated‑annealing (SA) search remove remaining
    irregularities.

All steps respect a global 1000 s budget.  The returned list is a
proper probability vector (sum = 1) – the evaluator will re‑normalise
anyway.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Simplex projection (used after the Adam optimiser)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector that sums to 1."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective (sum of a is assumed to be 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# -------------------  Analytic families ---------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_exponential_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-rate * i)
    w /= w.sum()
    return w


def _distribution_exponential_right(n: int, rate: float) -> np.ndarray:
    return _distribution_exponential_left(n, rate)[::-1]


def _distribution_exponential_double_end(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_powerlaw_left(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 1.0
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_reciprocal_min(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i + 1.0, n - i)  # 1‑based distance to nearest edge
    w = 1.0 / d
    w /= w.sum()
    return w


def _distribution_beta_u_shape(n: int, a: float, b: float) -> np.ndarray:
    """U‑shaped Beta pdf on the grid 0,…,n−1 (a,b∈(0,1) give high ends)."""
    # use half‑step points to avoid exact zeros at the boundaries
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = (x ** (a - 1.0)) * ((1.0 - x) ** (b - 1.0))
    # guard against NaNs when a or b are very small
    w[np.isnan(w)] = 0.0
    if w.sum() == 0.0:
        w[0] = 1.0
    w /= w.sum()
    return w


def _distribution_logistic(n: int, scale: float) -> np.ndarray:
    """Monotonically decreasing logistic shape."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = 1.0 / (1.0 + np.exp((i - centre) / scale))
    w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator, conc: float) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a.astype(np.float64)


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is the uniform vector."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    """Cyclic shift (roll) of a vector."""
    return np.roll(v, shift)


# ----------------------------------------------------------------------
# Adam based mirror descent on the exact objective
def _adam_refine(
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    lr0: float = 0.8,
    active_frac: float = 0.025,
    max_active: int = 30,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_interval: int = 250,
    best_seq: np.ndarray | None = None,
    best_score: float = float("inf"),
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑softmax (mirror descent) parametrisation.
    Sub‑gradient is built from the active set of convolution entries that
    are within ``active_frac`` of the current maximum.
    """
    n = init_a.shape[0]
    # initialise log‑weights (avoid log(0))
    x = np.log(np.maximum(init_a, 1e-12))
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    ind_grid = np.arange(n, dtype=np.int32)
    t = 0
    if best_seq is None:
        best_seq = init_a.copy()
        best_score = _objective(init_a)

    while time.time() < deadline and t < max_iters:
        t += 1
        a = _softmax(x)                # lies on the simplex
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        # ---- active set (sub‑gradient of the max) --------------------
        thresh = max_c * (1.0 - active_frac)
        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            # keep only the largest max_active entries
            top = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top]

        # gradient w.r.t. a (sub‑gradient of the max)
        grad_a = np.zeros_like(a)
        for idx_k in act:
            offset = idx_k - ind_grid        # vectorised offsets
            mask = (offset >= 0) & (offset < n)
            grad_a[mask] += 2.0 * a[offset[mask]]
        # ---- convert to gradient w.r.t. x (log‑softmax Jacobian) -----
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # ---- Adam update --------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = lr0 / math.sqrt(t)
        x -= eta * m_hat / (np.sqrt(v_hat) + eps)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t:,} new best {best_score:.6f}")

    # final projection (soft‑max already respects simplex, keep guard)
    best_seq = _project_onto_simplex(best_seq, s=1.0)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise swap local improvement
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Cheap simulated‑annealing style local search
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def _add_candidate(candidates: list[tuple[float, np.ndarray]],
                   score: float,
                   vec: np.ndarray,
                   limit: int = 20):
    """Insert a candidate keeping the list sorted by score and trimmed."""
    candidates.append((score, vec.copy()))
    candidates.sort(key=lambda x: x[0])
    if len(candidates) > limit:
        del candidates[limit:]


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Entry‑point used by the evaluator – returns a plain Python list."""
    rng = np.random.default_rng(seed)

    deadline = start + budget_s - 2.0          # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------

    lengths = [2000, 4000, 8000, 12000, 16000]          # a few representative sizes

    p_grid = np.logspace(-0.5, 1.2, 7)                # power‑law exponents

    scale_grid = np.logspace(-0.5, 1.0, 6)            # logistic scales
    lam_grid = np.linspace(0.0, 0.8, 5)                # uniform mixing fractions

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Seeding length {n}")

        # -- deterministic families ------------------------------------
        uniform = _distribution_uniform(n)
        _add_candidate(top_candidates, _objective(uniform), uniform)

        # left / right exponentials
        for rate in rate_grid:
            left = _distribution_exponential_left(n, rate)
            right = _distribution_exponential_right(n, rate)
            _add_candidate(top_candidates, _objective(left), left)
            _add_candidate(top_candidates, _objective(right), right)

            double_end = _distribution_exponential_double_end(n, rate)
            _add_candidate(top_candidates, _objective(double_end), double_end)

        # power‑law decreasing (left)
        for p in p_grid:
            pl = _distribution_powerlaw_left(n, p)
            _add_candidate(top_candidates, _objective(pl), pl)

        # reciprocal‑min shape (high ends, low centre)
        rec = _distribution_reciprocal_min(n)
        _add_candidate(top_candidates, _objective(rec), rec)

        # beta‑U shape (a,b<1 give high ends)
        for a in beta_grid:
            for b in beta_grid:
                beta_u = _distribution_beta_u_shape(n, a, b)
                _add_candidate(top_candidates, _objective(beta_u), beta_u)

        # logistic decreasing
        for scale in scale_grid:
            logi = _distribution_logistic(n, scale)
            _add_candidate(top_candidates, _objective(logi), logi)

        # mixing a few of the above with uniform
        for base_func in [uniform, rec]:
            for lam in lam_grid:
                mixed = _mix_with_uniform(base_func, lam)
                _add_candidate(top_candidates, _objective(mixed), mixed)

        # -- random families -----------------------------------------
        for _ in range(3):
            conc = rng.choice([0.02, 0.1, 0.5, 2.0])
            dirich = _distribution_dirichlet(n, rng, conc)
            _add_candidate(top_candidates, _objective(dirich), dirich)

        for _ in range(4):
            sparse = _distribution_random_sparse(n, rng, density=0.008)
            _add_candidate(top_candidates, _objective(sparse), sparse)

        # random cyclic rotations of a few good bases

            base = rng.choice([rec, uniform, left if 'left' in locals() else uniform])
            shift = rng.integers(n)

          f"(length={len(overall_best_seq)})")
    # ``evaluate_sequence`` will re‑normalise, but we keep a proper PMF.
    return overall_best_seq.tolist()
