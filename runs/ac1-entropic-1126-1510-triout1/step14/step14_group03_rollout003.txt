"""
Propose a step‑function (list of non‑negative heights) that minimizes
the provided ``evaluate_sequence`` objective.

Algorithm
---------
1. Generate a diverse pool of candidate distributions (uniform, arcsine,
   edge‑weighted, sum‑edge‑weighted, generalized‑normal, random Dirichlet).
   All candidates are normalised to sum = 1 (the evaluator rescales anyway).

2. Keep the ~15 best candidates according to the exact objective
   (fast to compute via FFT).

3. For each of those seeds run a *smooth* mirror‑descent optimisation.
   The non‑smooth max‑convolution term is approximated by a soft‑max
   (log‑sum‑exp) with a continuation schedule on the temperature β.
   The gradient of the smooth loss is obtained analytically and the
   update is performed with a multiplicative (exponentiated‑gradient)
   step followed by re‑normalisation onto the probability simplex.

4. After the smooth phase a short pairwise‑mass‑transfer simulated‑annealing
   polish refines the solution further.

All work respects the 1000 s wall‑time limit; intermediate progress and
objective values are printed for monitoring.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full, length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using real FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:n]


# ----------------------------------------------------------------------
#  Exact objective used by the evaluator (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a))   (a is assumed non‑negative and normalised)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
#  Analytic families (return probability vectors, i.e. sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i·(n−i))^{−p}   with i = distance to the left edge (½‑shifted)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_sum_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ i^{−p} + (n−i)^{−p} (sum of two one‑sided tails)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (-p)) + (j ** (-p))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(−|x−½|^{q}) on the discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1‑lam) + uniform·lam."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
#  Smooth (log‑sum‑exp) approximation of max(conv) and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, beta: float):
    """
    Returns (smooth_loss, gradient, max_convolution) where
    smooth_loss = (1/β)·log Σ exp(β·conv_i)  (≈ max conv for large β)
    gradient = ∂ smooth_loss / ∂ a   (shape = len(a))
    """
    # convolve a with itself
    b = _fft_self_convolve(a)
    maxb = b.max()
    # stable log‑sum‑exp
    shifted = b - maxb               # ≤ 0
    exp_beta = np.exp(beta * shifted)
    Z = exp_beta.sum()
    w = exp_beta / Z                 # softmax weights, sum = 1

    # smooth loss = maxb + (1/β)·log Z
    smooth_loss = maxb + (1.0 / beta) * np.log(Z)

    # gradient: 2·conv(w, a_rev) sliced to length n
    a_rev = a[::-1]
    conv_res = _fft_convolve(w, a_rev)          # length = 3·n‑2
    n = a.shape[0]
    grad = 2.0 * conv_res[(n - 1):(2 * n - 1)]
    return smooth_loss, grad, maxb


# ----------------------------------------------------------------------
#  Mirror‑descent (exponentiated‑gradient) with smooth loss
# ----------------------------------------------------------------------
def _smooth_mirror_descent(
    a: np.ndarray,
    deadline: float,
    cur_best_seq: np.ndarray,
    cur_best_score: float,
    eta0: float = 0.5,
    eval_every: int = 30,
) -> tuple[np.ndarray, float]:
    """
    Runs exponentiated‑gradient on the smooth loss for a given time budget.
    Returns the best sequence seen (according to the exact objective).
    """
    n = a.shape[0]
    t = 0
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    # continuation schedule – start smooth, finish with a sharp approximation
    beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0]

    for beta in beta_schedule:
        # a few hundred steps per β (or until time runs out)
        for _ in range(300):
            if time.time() >= deadline:
                return best_seq, best_score

            t += 1
            _, grad, _ = _smooth_obj_and_grad(a, beta)

            eta = eta0 / math.sqrt(t)
            a = a * np.exp(-eta * grad)          # multiplicative update
            s = a.sum()
            if s == 0.0:
                a = np.full(n, 1.0 / n, dtype=np.float64)
            else:
                a = a / s

            if t % eval_every == 0:
                cur = _objective(a)
                if cur < best_score - 1e-14:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] β={beta:.1f} it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
#  Pairwise‑mass‑transfer simulated‑annealing polish
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    cur_best_seq: np.ndarray,
    cur_best_score: float,
    rng: np.random.Generator,

    """
    Randomly move a fraction of mass between two bins, accepting
    improvements (and occasional uphill moves while temperature > 0).
    """
    n = a.shape[0]
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    start = time.time()
    total = deadline - start

    while time.time() < deadline:
        # pick a non‑zero source
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        # pick a distinct target
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move a random fraction of source mass
        delta = a[i] * rng.random() * 0.5   # ≤ 50 % of source
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.02

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}→j={j} δ={delta:.2e} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
#  Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns a plain ``list[float]`` ready for the evaluator.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0                     # small safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")
    top: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Broad seed generation (probability vectors)
    # ------------------------------------------------------------------
    lengths = [3000, 5000, 8000]                         # moderate sizes
    p_grid = np.concatenate((
        np.logspace(-2.0, -0.5, 8, base=10),           # 0.01 … 0.316
        np.logspace(-0.5,  0.5, 8, base=10),           # 0.316 … 3.16
        np.logspace( 0.5,  1.5, 6, base=10),           # 3.16 … 31.6
    ))
    lam_grid = np.linspace(0.0, 1.0, 9)                 # 0,0.125,…,1
    q_grid = np.linspace(0.5, 3.0, 9)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for cand in (_distribution_uniform(n),
                     _distribution_arcsine(n)):
            sc = _objective(cand)
            top.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (uniform/arc) score={sc:.6f}")

        # product edge‑weighted + mixtures
        for p in p_grid:
            if time.time() >= deadline:
                break
            base = _distribution_edge_weighted(n, float(p))
            sc = _objective(base)
            top.append((sc, base.copy()))
            if sc < best_score:
                best_score, best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new best prod p={p:.3g} score={sc:.6f}")

            for lam in lam_grid:
                cand = _mix_with_uniform(base, float(lam))
                sc = _objective(cand)
                top.append((sc, cand.copy()))
                if sc < best_score:
                    best_score, best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new best prod p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # sum‑edge‑weighted family (new ingredient)
        for p in p_grid:
            if time.time() >= deadline:
                break
            base = _distribution_edge_sum_weighted(n, float(p))
            sc = _objective(base)
            top.append((sc, base.copy()))
            if sc < best_score:
                best_score, best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new best sum p={p:.3g} score={sc:.6f}")

            for lam in lam_grid:
                cand = _mix_with_uniform(base, float(lam))
                sc = _objective(cand)
                top.append((sc, cand.copy()))
                if sc < best_score:
                    best_score, best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new best sum p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # generalized normal family
        for q in q_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_generalized_normal(n, float(q))
            sc = _objective(cand)
            top.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best gn q={q:.2f} score={sc:.6f}")

        # random sparse Dirichlet seeds (few non‑zeros)
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = np.zeros(n, dtype=np.float64)
            k = max(2, int(n * 0.01))
            idx = rng.choice(n, size=k, replace=False)
            cand[idx] = rng.random(k)
            cand /= cand.sum()
            sc = _objective(cand)
            top.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best sparse score={sc:.6f}")

        # keep only the very best seeds (size 15)
        top.sort(key=lambda x: x[0])
        top = top[:15]

    # ------------------------------------------------------------------
    # 2.  Include the previously submitted construction if available
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.abs(prev)
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < best_score:
                best_score, best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Guard against an empty result
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _distribution_uniform(5000)
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Smooth mirror‑descent refinement on the top seeds
    # ------------------------------------------------------------------
    for _, seed_vec in top:
        if time.time() >= deadline:
            break
        # allow a few seconds per seed (adjust if we are running out of time)
        remaining = deadline - time.time()
        per_seed = min(7.0, remaining / (len(top) + 1))
        sub_deadline = time.time() + per_seed
        print(f"[{time.time():.1f}] smooth descent on seed (len={len(seed_vec)}) budget={per_seed:.1f}s")
        best_seq, best_score = _smooth_mirror_descent(
            best_seq.copy(),
            sub_deadline,
            best_seq,
            best_score,
            eta0=0.5,
            eval_every=30,
        )
        print(f"[{time.time():.1f}] after smooth descent best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing")
        best_seq, best_score = _pairwise_annealing(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            rng,
        )
        print(f"[{time.time():.1f}] after annealing best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Return plain Python list (the evaluator will clip/scale as needed)
    # ------------------------------------------------------------------
    print(
        f"Final best score (objective = 2·n·max(conv)/sum²): {best_score:.6f} "
        f"(length={len(best_seq)})"
    )
    return best_seq.tolist()
