def _smooth_max_descent_long(a0, time_limit, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    t=0
    step_base = 0.2  # maybe bigger
    # iterate over schedule gradually increasing beta
    for beta in [0.5,1,2,5,10,20,50,100,200,400,800]:
        while time.time() < time_limit:
            t += 1
            conv = _fft_self_convolve(a)
            max_val = conv.max()
            shifted = conv - max_val
            # compute w in double precision careful
            exp_beta = np.exp(beta * shifted)  # no overflow because shifted <=0
            w = exp_beta / exp_beta.sum()
            a_rev = a[::-1]
            grad_conv = _fft_convolve(w, a_rev)
            grad = 2.0 * grad_conv[n-1 : n-1 + n]
            grad_J = 2.0 * n * grad
            m = beta1 * m + (1-beta1) * grad_J
            v = beta2 * v + (1-beta2) * (grad_J**2)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            step = step_base / (np.sqrt(v_hat) + eps)
            a -= step * m_hat
            a = _project_onto_simplex(a, s=1.0)

            if t % 100 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-14:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] smooth beta={beta:.1f} it={t} new best {best_score:.6f}")
            # Additional early termination: if a changes very little relative to norm, break perhaps.
            if t % 5000 == 0:
                # optional need
                pass
        if time.time() >= time_limit:
            break
    return best_seq, best_score

"""
Hybrid optimisation for the step‑function objective

The pipeline consists of

  1️⃣  A dense scan over many analytically‑parameterised families
      (uniform, arcsine, sine, cosine, quadratic, beta, edge‑heavy,
      exponential‑edge, Gaussian‑center, polynomial‑center).  Each
      candidate is normalised to a probability simplex and evaluated
      with the exact objective 2·n·max(conv(a,a)).

  2️⃣  A long smoothed‑max mirror‑descent (Adam) run on the best few
      seeds.  The smoothing (log‑sum‑exp) parameter β is increased
      gradually (0.5 → 800) and the optimiser is allowed to perform
      many thousands of updates – the previous code stopped after
      ~2500 per β, which limited convergence.

  3️⃣  An exact exponentiated‑gradient sub‑gradient descent phase
      (as in the prior state‑of‑the‑art) for further polishing.

  4️⃣  A final hill‑climber that repeatedly moves a random fraction
      of mass between two positions, accepting only strict
      improvements.

The whole routine respects the supplied wall‑clock budget
(`budget_s`) – any part of the algorithm stops as soon as the
deadline is reached, guaranteeing a returned sequence.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  Utility: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
#  FFT‑based (full) convolution utilities
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a * a."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
#  Exact objective (sum(a) = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
#  Analytic families (all return a normalised probability vector)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    w /= w_sum
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    total = w.sum()
    if total == 0.0:
        return _distribution_uniform(n)
    w /= total
    return w


def _distribution_gaussian_center(n: int, sigma: float) -> np.ndarray:
    """Symmetric Gaussian centred at (n‑1)/2."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    w /= w_sum
    return w


def _distribution_polynomial_center(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i·(n‑1‑i))^p, centralising mass."""
    i = np.arange(n, dtype=np.float64)
    w = (i * (n - 1 - i)) ** p
    s = w.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    w /= s
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
#  Smoothed‑max mirror descent (Adam) – long version
# ----------------------------------------------------------------------
def _smooth_max_descent_long(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of the
    L∞‑objective, with an extended schedule of β values and a
    generous iteration budget.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    # β schedule – we push a little farther than before
    beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0,
                     50.0, 100.0, 200.0, 400.0, 800.0]

    step_base = 0.25  # a slightly larger base step size

    for beta in beta_schedule:
        # run until time runs out or an optional max‑iteration count
        while time.time() < time_limit:
            t += 1

            conv = _fft_self_convolve(a)      # length 2n‑1
            max_val = conv.max()
            shifted = conv - max_val          # ≤ 0, stabilises exp
            exp_beta = np.exp(beta * shifted)
            w = exp_beta / exp_beta.sum()     # softmax over indices, Σw = 1

            a_rev = a[::-1]
            grad_conv = _fft_convolve(w, a_rev)    # length 3n‑2
            grad = 2.0 * grad_conv[n - 1 : n - 1 + n]   # size n

            grad_J = 2.0 * n * grad           # gradient of 2·n·smooth_max

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_J
            v = beta2 * v + (1.0 - beta2) * (grad_J ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            step = step_base / (np.sqrt(v_hat) + eps)
            a -= step * m_hat
            a = _project_onto_simplex(a, s=1.0)

            # occasional exact evaluation
            if t % 200 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-13:
                    best_score, best_seq = cur, a.copy()
                    print(
                        f"[{time.time():.1f}] smooth β={beta:.1f} it={t} new best {best_score:.6f}"
                    )

            # safety break if the remaining budget is tiny
            if time.time() + 0.05 >= time_limit:
                break

        if time.time() >= time_limit:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
#  Exact sub‑gradient descent using exponentiated gradient (EG)
# ----------------------------------------------------------------------
def _subgradient_descent_exact(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true L∞‑objective."""
    a = a0.copy()
    n = a.shape[0]
    t = 0

    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        # sub‑gradient for the max‑entry (sum of shifted copies)
        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)          # EG keeps positivity
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-13:
                best_score, best_seq = cur, a.copy()
                print(
                    f"[{time.time():.1f}] exact subgrad it={t} new best {best_score:.6f}"
                )
        # early exit if almost out of time
        if time.time() + 0.05 >= time_limit:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
#  Greedy pairwise mass‑transfer hill‑climber
# ----------------------------------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly relocate a fraction of mass; accept only strict improvements."""
    a = a0.copy()
    n = a.shape[0]
    trials = 0

    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45  # move ≤45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score, best_seq = cur, a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
        # safety break
        if time.time() + 0.05 >= time_limit:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
#  Small random perturbation (used for restarts)
# ----------------------------------------------------------------------
def _perturb(a: np.ndarray, rng: np.random.Generator, sigma: float = 0.02) -> np.ndarray:
    b = a + sigma * rng.standard_normal(a.shape)
    b = np.clip(b, 0.0, None)
    s = b.sum()
    if s == 0.0:
        return np.full_like(a, 1.0 / a.shape[0])
    return b / s

# ----------------------------------------------------------------------
#  Helper: sparse random seeds (used in the analytic scan)
# ----------------------------------------------------------------------
def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals
