"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (∑a)²

using a large and diverse seed pool, fast sub‑gradient (exponentiated)
descent, progressive up‑sampling and a biased pairwise‑mass‑transfer
annealing.  The implementation expands on the previous 1000 s search by
adding monotone decreasing families (power‑law, exponential, logistic),
fixing a bug in the up‑sampling loop (so the length truly grows) and
biasing the annealing moves towards entries that contribute most to the
current maximal convolution entry.  All operations keep the vector on the
probability simplex (∑a = 1) to avoid the scaling checks of
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based self‑convolution (exact, O(n log n))
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two ≥ 2·n‑1
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Objective for a probability vector (∑a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) for Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families (all return probability vectors)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i+0.5)^{-p}·(n‑i‑0.5)^{-p} (symmetric, high at edges)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w /= w.sum()
    return w


def _distribution_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) on cell centres."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # Guard against overflow/underflow
    w = np.exp((alpha - 1) * np.log(x) + (alpha - 1) * np.log1p(-x))
    w[~np.isfinite(w)] = 0.0
    w /= w.sum()
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(‑|x‑½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse vector with a fixed density (mass placed on few bins)."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_power_law_decreasing(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i+1)^{‑p}, monotone decreasing, non‑symmetric."""
    i = np.arange(n, dtype=np.float64) + 1.0
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_exponential_decreasing(n: int, lam: float) -> np.ndarray:
    """w_i ∝ exp(‑lam·i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-lam * i)
    w /= w.sum()
    return w


def _distribution_logistic_decreasing(n: int, steep: float, centre: float) -> np.ndarray:
    """w_i ∝ 1 / (1 + exp(steep·(i‑centre))) – monotone S‑shape."""
    i = np.arange(n, dtype=np.float64)
    w = 1.0 / (1.0 + np.exp(steep * (i - centre)))
    w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1‑λ) + uniform·λ."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Returns (grad, k) where k is the index attaining max(conv(a,a)).
    grad_i = 2·a_{k‑i} for all i that participate in that term,
    otherwise 0.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))
    grad = np.zeros_like(a)
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
# Adaptive exponentiated‑gradient descent (mirrored sub‑gradient)
# ----------------------------------------------------------------------
def _adaptive_exp_gradient(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    beta: float = 0.9,
    eval_every: int = 50,
) -> tuple[np.ndarray, float]:
    """
    Mirror descent on the exact objective.
    Uses EMA of the sub‑gradient (momentum) and an adaptive step size.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)          # EMA of gradients
    eta = eta0
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        m = beta * m + (1.0 - beta) * grad   # EMA

        # multiplicative (exponential) update
        a = a * np.exp(-eta * m)
        # re‑normalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                eta = min(eta0, eta * 1.02)          # reward good step
                print(f"[{time.time():.1f}] GD it={t:,} new best {best_score:.6f}")
            else:
                eta *= 0.995                         # gentle decay

        if t % (eval_every * 10) == 0:
            a = _project_onto_simplex(a, s=1.0)      # numerical safety
    return best_seq, best_score


# ----------------------------------------------------------------------
# Biased pairwise mass‑transfer simulated annealing
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Moves a random fraction of mass from a high‑gradient index to a low‑gradient one.
    The temperature linearly cools from a small fraction of best_score.
    """
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    total = deadline - start

    while time.time() < deadline:
        # gradient tells which positions matter for the current max entry
        grad, _ = _grad_of_max_conv(a)

        # pick source among the top‑10 % by gradient magnitude
        src_candidates = np.where(grad > np.percentile(grad, 90))[0]
        if src_candidates.size == 0:
            src = rng.choice(np.where(a > 1e-14)[0])
        else:
            src = rng.choice(src_candidates)

        # pick destination among the bottom‑10 % (or any other index)
        dst_candidates = np.where(grad < np.percentile(grad, 10))[0]
        if dst_candidates.size == 0:
            dst = rng.integers(n - 1)
            if dst >= src:
                dst += 1
        else:
            dst = rng.choice(dst_candidates)
            # ensure different indices
            if dst == src:
                continue

        delta = a[src] * rng.random() * 0.25   # up to 25 % of the source mass
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.02

        accept = False
        if cur < best_score - 1e-15:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal {src}->{dst} Δ={delta:.1e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling that keeps Σa = 1
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new

    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_dead = time.time() + (deadline - time.time()) * 0.35
        a_opt, sc_opt = _adaptive_exp_gradient(
            overall_best_seq,
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.5,
            beta=0.9,
            eval_every=60,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] long GD improved best to {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Progressive up‑sampling (double length while time permits)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n > 65536:
            break
        print(f"[{time.time():.1f}] up‑sampling {cur_n} → {target_n}")
        up = _upsample(overall_best_seq, target_n)
        up_sc = _objective(up)

        # keep the longer vector for further work even if its score is slightly worse
        overall_best_seq = up
        if up_sc < overall_best_score:
            overall_best_score = up_sc
            print(f"[{time.time():.1f}] up‑sampled new best {overall_best_score:.6f}")

        # medium‑length GD on the up‑sampled vector
        sub_dead = time.time() + (deadline - time.time()) * 0.2
        a_opt, sc_opt = _adaptive_exp_gradient(
            overall_best_seq,
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.45,
            beta=0.9,
            eval_every=50,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] GD after up‑sample new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣  Final biased annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing phase")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq,
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after annealing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)) = {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    # evaluate_sequence expects a plain Python list
    return overall_best_seq.tolist()
