    beta1=0.9; beta2=0.999; eps=1e-8

    for beta in beta_schedule:

        for _ in range(5000):
            if time.time() >= time_limit:
                break
            t+=1
            conv = _fft_self_convolve(a)
            max_val = conv.max()
            shifted = conv - max_val
            exp_beta = np.exp(beta * shifted)

            w = exp_beta / sum_exp
            # compute gradient

def _subgradient_descent_exact(...):
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t+=1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]
        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n-1))
            i_end = min(n-1, idx)
            i = np.arange(i_start, i_end+1)
            j = idx - i
            grad[i] += 2.0 * a[j]
        eta = 0.9 / math.sqrt(t)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0:
            a = np.full(n, 1.0/n)
        else:
            a /= s
        # symmetrize optionally?

        if t % 200 == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad it={t} new global best {best_score:.6f}")

def _upsample(seq, new_n):
    old_n = len(seq)
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    new_seq = np.interp(new_x, old_x, seq, left=0.0, right=0.0)
    new_seq = np.maximum(new_seq, 0.0)
    total = new_seq.sum()
    if total == 0:
        new_seq = np.full(new_n, 1.0 / new_n)
    else:
        new_seq /= total
    return new_seq

"""
Hybrid optimizer for the step‑function objective

    F(a) = 2·n·max(conv(a,a))      (a ≥ 0, Σa = 1)

The algorithm proceeds in four phases:

1️⃣  **Analytic seeds** – a quick sweep over many parametric families
    (uniform, arcsine, sine, cosine, quadratic, beta, edge‑weighted,
    exponential‑edge, random Dirichlet / sparse).  The best few seeds are
    kept.

2️⃣  **Smoothed‑max mirror descent** – the non‑smooth max is replaced by the
    log‑sum‑exp surrogate.  An Adam optimiser with a continuation on the
    smoothing parameter β (0.5 → 1000) drives the iterate towards a local
    minimum.  After each update the vector is projected onto the simplex
    and symmetrised (a = (a+a[::-1])/2), which is natural for the problem.

3️⃣  **Exact sub‑gradient refinement** – exponentiated‑gradient descent on the
    true L∞‑objective (the “sub‑gradient” step used in the previous state‑of‑the‑art).

4️⃣  **Greedy pairwise polishing** – random mass transfers between two positions
    are accepted only if they strictly improve the exact objective.

If time permits the best seed is up‑sampled to a larger support (e.g. 20000
entries) before phase 2, giving a finer discretisation of the limiting
continuous problem.  The function returns a plain list of non‑negative
floats; the supplied ``evaluate_sequence`` routine will re‑scale it
automatically.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
#  Simplex projection
# ------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of `v` onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
#  FFT based convolutions (real inputs)
# ------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    return _fft_convolve(a, a)


# ------------------------------------------------------------
#  Exact objective (a is already normalised, Σa = 1)
# ------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a))."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ------------------------------------------------------------
#  Analytic families (always return a normalised probability vector)
# ------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    s = w.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    w /= s
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    s = w.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    w /= s
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Enforce a_i = a_{n-1-i} by simple averaging."""
    return (v + v[::-1]) * 0.5


# ------------------------------------------------------------
#  Smoothed‑max mirror descent (Adam + β‑continuation)
# ------------------------------------------------------------
def _smooth_max_descent(a0: np.ndarray,
                        time_limit: float,
                        best_seq: np.ndarray,
                        best_score: float) -> tuple[np.ndarray, float]:
    """Run Adam on the log‑sum‑exp surrogate of max(conv(a,a))."""
    a = a0.copy()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    # β values – gradually become larger, ending at 1000
    beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0,
                     200.0, 500.0, 1000.0]

    for beta in beta_schedule:
        # up to a few thousand iterations per β unless time runs out
        max_iter_beta = 4000
        for _ in range(max_iter_beta):
            if time.time() >= time_limit:
                break
            t += 1

            # self‑convolution
            conv = _fft_self_convolve(a)          # length 2n‑1
            max_val = conv.max()
            # soft‑max weights (stable)
            shifted = conv - max_val
            exp_beta = np.exp(beta * shifted)
            w = exp_beta / exp_beta.sum()          # Σ w = 1

            # gradient of the smooth max
            a_rev = a[::-1]
            grad_conv = _fft_convolve(w, a_rev)   # length 3n‑2
            grad = 2.0 * grad_conv[n - 1: n - 1 + n]

            # gradient of the (smooth) objective J = 2·n·smooth_max
            grad_J = 2.0 * n * grad

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_J
            v = beta2 * v + (1.0 - beta2) * (grad_J ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            step = 0.08 / (np.sqrt(v_hat) + eps)   # base step size tuned experimentally
            a -= step * m_hat

            # project onto the simplex and re‑symmetrise
            a = _project_onto_simplex(a, s=1.0)
            a = _symmetrize(a)

            # occasional exact evaluation
            if t % 200 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-14:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] β={beta:g} it={t} new best {best_score:.6f}")

        if time.time() >= time_limit:
            break

    return best_seq, best_score


# ------------------------------------------------------------
#  Exact sub‑gradient descent (exponentiated gradient)
# ------------------------------------------------------------
def _subgradient_descent_exact(a0: np.ndarray,
                               time_limit: float,
                               best_seq: np.ndarray,
                               best_score: float,
                               lr0: float = 0.9,
                               eval_every: int = 200) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true L∞‑objective."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        # exact sub‑gradient (sum of shifted copies of a)
        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)               # EG keeps positivity
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s
        a = _symmetrize(a)

        if t % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] exact subgrad it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ------------------------------------------------------------
#  Greedy pairwise mass‑transfer hill‑climber
# ------------------------------------------------------------
def _pairwise_swap(a0: np.ndarray,
                   time_limit: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   max_trials: int | None = None) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass; accept only strict improvements."""
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")

    return best_seq, best_score


# ------------------------------------------------------------
#  Upsampling helper (interpolates a probability vector to a larger support)
# ------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    up = np.interp(new_x, old_x, seq, left=0.0, right=0.0)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s == 0.0:
        up = _distribution_uniform(new_n)
    else:
        up /= s
    return up
