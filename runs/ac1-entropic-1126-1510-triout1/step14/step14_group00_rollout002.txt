assistantfinal**Solution Explanation**

The goal is to minimise  

\[
E(a)=\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2}
\]

for a non‑negative step‑function `a`.  
Because the numerator and denominator are both quadratic in a scaling factor,
the objective is *scale invariant*.  
Hence we may restrict the search to *probability vectors* (`∑a_i = 1`);
the objective then simplifies to  

\[
E(a)=2\,n\;\max (a*a)
\]

where `a*a` is the discrete self‑convolution of `a`.

The search follows four phases

1. **Rich seeding** – many analytic families (uniform, arcsine, Hann,
   edge‑weighted, exponential‑decay, β‑distribution, Dirichlet, sparse)
   are sampled for several lengths. The best 30 candidates are kept.
2. **Smoothed exponentiated‑gradient descent** –  
   instead of the non‑smooth `max`, we minimise a *log‑sum‑exp* surrogate  

   \[
   M_\tau(a)=\frac{1}{\tau}\log\!\Bigl(\sum_k e^{\tau (a*a)_k}\Bigr)
   \]

   whose gradient is a weighted average of the gradients of all convolution
   entries.  
   The algorithm uses an entropic (multiplicative) update, an EMA momentum,
   an adaptive step size, and a temperature schedule that slowly raises `τ`
   from a smooth regime (`τ≈0.5`) to the true `max` (`τ≈30`).  
   Each seed receives a short run, the current global best receives a
   longer run.
3. **Progressive up‑sampling** – the best vector is repeatedly doubled in
   length by linear interpolation, each up‑sampled vector is again polished
   with the smoothed gradient method. This exploits the fact that a finer
   discretisation can only improve the bound.
4. **Final pairwise annealing** – a simple simulated‑annealing that moves a
   random fraction of mass from one index to another, accepting non‑improving
   moves with a linearly cooling temperature. This helps to escape tiny
   local minima left by the gradient phase.

All steps respect the 1000 s budget (a 2 s safety margin).  
The function `propose_candidate` returns a plain Python list ready for
`evaluate_sequence`.

"""
Multi‑phase optimisation for the self‑convolution objective

    2·n·max(conv(a,a)) / (∑a)²

The algorithm

1. builds a diverse pool of analytic seeds (uniform, arcsine, Hann,
   edge‑weighted, exponential‑decay, β‑distribution, Dirichlet, sparse);
   the 30 best are retained.
2. optimises every seed by a *smoothed* exponentiated‑gradient descent.
   The surrogate objective is the log‑sum‑exp of the convolution; the
   temperature τ is increased from 0.5 to ~30 during optimisation.
3. repeatedly up‑samples the current best vector (doubling its length)
   and re‑optimises it.
4. finishes with a pairwise mass‑transfer simulated annealing.

All variables stay non‑negative; after each update the vector is projected
onto the probability simplex. The routine respects a 1000 s wall‑clock
budget and prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based self‑convolution (full length 2·n‑1)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective for a probability vector (∑a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families (all return probability vectors)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted_product(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{-p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted_sum(n: int, p: float) -> np.ndarray:
    """Weight ∝ i^{-p}+j^{-p} (separate distance to each edge)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (-p)) + (j ** (-p))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_exponential_decay(n: int, alpha: float) -> np.ndarray:
    """Symmetric exponential decay w_i ∝ exp(-α·dist_to_edge)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    d = np.minimum(i, n + 1 - i)
    w = np.exp(-alpha * d)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) sampled on cell mid‑points."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # safe computation for extreme α
    w = np.where(
        (x == 0) | (x == 1),
        0.0,
        np.exp((alpha - 1) * np.log(x) + (alpha - 1) * np.log1p(-x))
    )
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.02
) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1‑λ) + uniform·λ."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Gradient of the smoothed objective (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _grad_weighted(a: np.ndarray, w_norm: np.ndarray) -> np.ndarray:
    """
    Compute Σ_k (w_norm[k])·∇(conv_k)  efficiently.
    Resulting vector has length n and is always non‑negative.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    # convolution a_rev * w_norm  (length n + (2n‑1)‑1 = 3n‑2)
    size = 1 << ((len(a_rev) + len(w_norm) - 1 - 1).bit_length())
    A = np.fft.rfft(a_rev, n=size)
    W = np.fft.rfft(w_norm, n=size)
    conv = np.fft.irfft(A * W, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    # numerical safety
    grad = np.maximum(grad, 0.0)
    return grad


# ----------------------------------------------------------------------
# Smoothed exponentiated‑gradient descent (mirrored gradient)
# ----------------------------------------------------------------------
def _exp_grad_smoothed(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    beta: float = 0.9,
    eval_every: int = 50,
    tau_start: float = 0.5,
    tau_max: float = 30.0,
    tau_factor: float = 1.001,
) -> tuple[np.ndarray, float]:
    """
    Entropic gradient descent on the log‑sum‑exp surrogate.
    The temperature τ is increased gradually (tau_factor > 1) so that the
    optimisation starts smooth and ends close to the true max.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)          # EMA of gradients
    eta = eta0
    tau = tau_start
    it = 0
    while time.time() < deadline:
        it += 1
        # 1) smoothed weights
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        shifted = tau * (conv - max_c)          # ≤ 0, numerically safe
        w_exp = np.exp(shifted)                 # max entry = 1
        Z = w_exp.sum()
        w_norm = w_exp / Z

        # 2) gradient of the surrogate
        grad = _grad_weighted(a, w_norm)

        # 3) EMA momentum
        m = beta * m + (1.0 - beta) * grad

        # 4) multiplicative (entropic) update
        a = a * np.exp(-eta * m)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # 5) temperature schedule
        if tau < tau_max:
            tau = min(tau_max, tau * tau_factor)

        # 6) occasional exact evaluation
        if it % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                eta = min(eta0, eta * 1.02)      # reward improvement
                print(f"[{time.time():.1f}] exp‑grad it={it:,} new best {best_score:.6f}")
            else:
                eta *= 0.995                      # mild decay

        # 7) re‑project periodically to avoid drift
        if it % (eval_every * 10) == 0:
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling while preserving total mass (=1)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new

                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal {src}->{dst} Δ={delta:.1e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a step‑function (list of non‑negative heights) that minimises
    ``evaluate_sequence``.  The routine uses the four phases described
    above and respects the given wall‑clock budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    cand_pool: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1️⃣  Seed generation (multiple lengths)
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096, 8192, 16384]
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] seed generation, length={n}")

        # ----- deterministic families -----
        for base in (
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_hann(n),
        ):
            sc = _objective(base)
            cand_pool.append((sc, base.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new global best (det) {sc:.6f}")

        # ----- edge‑weighted product -----
        p_vals = np.concatenate((
            np.logspace(-3.0, -0.5, 8, base=10),   # 0.001 → 0.316
            np.logspace(-0.5, 0.0, 8, base=10),    # 0.316 → 1
            np.linspace(0.1, 2.0, 10),             # linear region
            np.logspace(0.0, 0.7, 6, base=10)      # 1 → 5
        ))
        lam_vals = np.linspace(0.0, 1.0, 5)
        for p in p_vals:
            if time.time() >= deadline:
                break
            base = _distribution_edge_weighted_product(n, float(p))
            for lam in lam_vals:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(base, float(lam))
                sc = _objective(cand)
                cand_pool.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] edge‑prod p={p:.3g} λ={lam:.2f} new best {sc:.6f}")

        # ----- edge‑weighted sum (separate distances) -----
        for p in np.linspace(0.2, 4.0, 12):
            if time.time() >= deadline:
                break
            cand = _distribution_edge_weighted_sum(n, float(p))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] edge‑sum p={p:.2f} new best {sc:.6f}")

        # ----- exponential decay -----
        for alpha in np.linspace(0.1, 4.0, 10):
            if time.time() >= deadline:
                break
            cand = _distribution_exponential_decay(n, float(alpha))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] exp‑dec α={alpha:.2f} new best {sc:.6f}")

        # ----- symmetric Beta family -----
        for alpha in np.concatenate((
            np.logspace(-2.0, -0.2, 7, base=10),   # heavy tails
            np.linspace(0.2, 2.0, 9),             # moderate
            np.linspace(2.0, 5.0, 5)              # central
        )):
            if time.time() >= deadline:
                break
            cand = _distribution_beta(n, float(alpha))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] beta α={alpha:.3g} new best {sc:.6f}")

        # ----- random Dirichlet seeds -----
        for _ in range(10):
            if time.time() >= deadline:
                break
            log_alpha = rng.uniform(-2.0, 1.0)          # α ∈ [0.01,10]
            alpha = 10 ** log_alpha
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] Dirichlet α={alpha:.3g} new best {sc:.6f}")

        # ----- sparse random seeds -----
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.02)
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] sparse new best {sc:.6f}")

        # keep only the top‑30 seeds overall
        cand_pool.sort(key=lambda x: x[0])
        cand_pool = cand_pool[:30]

    # ------------------------------------------------------------------
    # 2️⃣  Incorporate a possibly existing construction (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.abs(prev)                 # safety: non‑negative
            s = prev.sum()
            if s == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            else:
                prev = prev / s
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
            cand_pool.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3️⃣  Ensure we have a fallback
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 2048
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Short smoothed GD on every seed (quick polishing)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 15.0 and cand_pool:
        per_seed = max(0.25, remaining * 0.2 / max(len(cand_pool), 1))
        print(f"[{time.time():.1f}] quick smoothed GD on {len(cand_pool)} seeds (~{per_seed:.2f}s each)")
        for idx, (sc, a0) in enumerate(cand_pool):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            a_opt, sc_opt = _exp_grad_smoothed(
                a0.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.8,
                beta=0.9,
                eval_every=30,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after quick GD seed {idx} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Longer smoothed GD on the current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_dead = time.time() + (deadline - time.time()) * 0.35
        a_opt, sc_opt = _exp_grad_smoothed(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.6,
            beta=0.9,
            eval_every=60,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] long smoothed GD improved best to {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Progressive up‑sampling (doubling length while time permits)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n > 65536:
            break
        print(f"[{time.time():.1f}] up‑sampling {cur_n} → {target_n}")
        up = _upsample(overall_best_seq, target_n)
        up_sc = _objective(up)
        if up_sc < overall_best_score:
            overall_best_seq, overall_best_score = up, up_sc
            print(f"[{time.time():.1f}] up‑sampled new best {overall_best_score:.6f}")

        # medium‑length GD on the up‑sampled vector
        sub_dead = time.time() + (deadline - time.time()) * 0.2
        a_opt, sc_opt = _exp_grad_smoothed(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.5,
            beta=0.9,
            eval_every=50,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
            print(f"[{time.time():.1f}] GD after up‑sample new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣  Final pairwise‑mass‑transfer annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing phase")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
            print(f"[{time.time():.1f}] after annealing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)) = {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
