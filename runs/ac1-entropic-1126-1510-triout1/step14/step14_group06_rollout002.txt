"""
Proposes a non‑negative step‑function (probability vector) that aims to
minimise ``evaluate_sequence``.  The search proceeds as follows:

1.  Generate a diverse pool of analytic and random seeds for several
    lengths (up to 65 536).  Keep the best few candidates.

2.  Form convex mixtures of the top candidates (including mixes with the
    uniform distribution) – a cheap way to discover improved shapes.

3.  Upsample the current best vector repeatedly (doubling the length each
    time) and run a temperature‑smoothed Adam optimiser on each new size.
    The temperature is cooled aggressively so that the optimiser finally
    targets the true ``max`` of the convolution.

4.  Refine the resulting vector with a sub‑gradient (exponentiated‑gradient)
    descent on the exact objective, followed by a short Frank‑Wolfe style
    mass‑transfer step.

5.  Finish with a long pair‑wise “annealing” phase that moves mass between
    bins, accepting improving moves and, early on, occasional worsening
    moves with a temperature that decays linearly.

All steps respect the overall 1000 s budget and always return a plain
``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (unused but kept for safety)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation of a (len n) with w (len m) for non‑negative shifts.
    Returns an array of length n where element i equals Σ_j a_j·w_{j+i}.
    Implemented via FFT using convolution with the reversed weight vector.
    """
    n = a.shape[0]
    m = w.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()
    rev_w = w[::-1]
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_w, n=size)
    conv = np.fft.irfft(A * B, n=size)
    start = m - 1
    return conv[start:start + n]


# ----------------------------------------------------------------------
# Exact objective used by ``evaluate_sequence`` (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return the exact value of ``evaluate_sequence`` for a (unnormalised)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic families – always return a probability vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{-p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(-|x-½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    """Monotone decreasing power‑law (p>0)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_sqrt(n: int) -> np.ndarray:
    """Shape ∝ sqrt(i·(n‑i)) – puts more mass in the centre."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sqrt(i * (n - i))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Sub‑gradient of the active convolution entry (used by EG and FW)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Compute the sub‑gradient of the currently maximal convolution entry.
    Returns (grad, active_index).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))
    grad = np.zeros_like(a)
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent (multiplicative mirror descent)
# ----------------------------------------------------------------------
def _exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """Perform exponentiated‑gradient descent on the exact objective."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Frank‑Wolfe style refinement (pairwise mass transfer guided by the gradient)
# ----------------------------------------------------------------------
def _frank_wolfe_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """Simple Frank‑Wolfe iteration: move mass towards the coordinate with the
    smallest gradient (the least harmful direction)."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        i_min = int(np.argmin(grad))
        gamma = 2.0 / (t + 2.0)          # classic diminishing step size
        a = a * (1.0 - gamma)
        a[i_min] += gamma
        if t % eval_every == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] FW it={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise‑annealing polishing (simulated annealing style)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    max_trials: int | None = None,
    temp_factor: float = 0.05,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins; accept improving
    moves or, early on, worsening moves with a temperature‑dependent probability."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        # pick a distinct destination
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move a random fraction (up to 40 % of source mass)
        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * temp_factor

        if cur + 1e-14 < best_score:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur + 1e-14 < best_score:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling while preserving the simplex (used after a good candidate)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Adam optimisation of a temperature‑smoothed version of the objective
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    tau_start: float = 0.2,
    tau_end: float = 0.001,
    tau_decay: float = 0.999,
    eval_every: int = 200,
    step_size0: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smoothed objective
        f(a) = 2·n·τ·log‑sum‑exp(conv(a)/τ) .
    The temperature τ is exponentially cooled from ``tau_start`` to ``tau_end``.
    """
    n = a0.shape[0]
    x = np.log(a0 + 1e-15)               # initialise logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    tau = tau_start
    while time.time() < deadline and tau > tau_end:
        t += 1
        # softmax probabilities (ensures non‑negativity and sum‑1)
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # convolution of a with itself
        conv = _fft_self_convolve(a)
        max_conv = conv.max()

        # τ‑softmax over convolution entries
        exp_vals = np.exp((conv - max_conv) / tau)
        w = exp_vals / exp_vals.sum()

        # gradient w.r.t. a (4·n·correlation of a and w)
        corr = _fft_corr(a, w)                 # length n
        grad_a = 4.0 * n * corr

        # chain rule to logits
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = step_size0 / (np.sqrt(v_hat) + eps)
        x = x - lr * m_hat

        # periodic evaluation on the *exact* objective
        if t % eval_every == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} τ={tau:.6f} new best {best_score:.6f}")

        # cool the temperature
        tau = max(tau * tau_decay, tau_end)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the total 1000 s budget
    and returns a plain ``list[float]`` ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1.  Populate a diverse pool of seeds (large lengths)
    # ------------------------------------------------------------------
    lengths = [4096, 8192, 16384, 32768, 65536]
    candidate_pool: list[tuple[float, np.ndarray]] = []

    print(f"[{time.time():.1f}] Starting seed generation")
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        families = [
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_hann,
            lambda nn: _distribution_beta_sym(nn, 0.2),
            lambda nn: _distribution_beta_sym(nn, 2.0),
            lambda nn: _distribution_edge_weighted(nn, 0.5),
            lambda nn: _distribution_edge_weighted(nn, 1.0),
            lambda nn: _distribution_generalized_normal(nn, 0.5),
            lambda nn: _distribution_generalized_normal(nn, 2.0),
            lambda nn: _distribution_power_law(nn, 0.5),
            lambda nn: _distribution_power_law(nn, 2.0),
            _distribution_sqrt,
        ]

        for gen in families:
            if time.time() >= deadline:
                break
            try:
                cand = gen(n)
            except Exception:
                continue
            sc = _objective(cand)
            candidate_pool.append((sc, cand.copy()))

        # random Dirichlet seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            candidate_pool.append((sc, cand.copy()))

        # sparse random seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.01)
            sc = _objective(cand)
            candidate_pool.append((sc, cand.copy()))

    # Keep the best few seeds for later refinement
    candidate_pool.sort(key=lambda x: x[0])
    top_k = 20
    top_candidates = [seq.copy() for _, seq in candidate_pool[:top_k]]

    overall_best_seq = top_candidates[0].copy()
    overall_best_score = _objective(overall_best_seq)
    print(f"[{time.time():.1f}] Initial best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Form convex mixtures of the top candidates (including uniform)
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] Mixing top candidates")
    for i in range(len(top_candidates)):
        for j in range(i + 1, len(top_candidates)):
            for lam in (0.25, 0.5, 0.75):
                mix = lam * top_candidates[i] + (1.0 - lam) * top_candidates[j]
                mix /= mix.sum()
                sc = _objective(mix)
                if sc + 1e-14 < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = mix.copy()
                    print(f"[{time.time():.1f}] mix ({i},{j}) λ={lam:.2f} new best {overall_best_score:.6f}")

    for base in top_candidates:
        for lam in (0.2, 0.5, 0.8):
            mix = _mix_with_uniform(base, lam)
            sc = _objective(mix)
            if sc + 1e-14 < overall_best_score:
                overall_best_score = sc
                overall_best_seq = mix.copy()
                print(f"[{time.time():.1f}] mix with uniform λ={lam:.2f} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Progressive up‑sampling + Adam smoothing
    # ------------------------------------------------------------------
    while time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        if cur_n >= 65536:
            break
        new_n = min(cur_n * 2, 65536)
        print(f"[{time.time():.1f}] Upsampling {cur_n} → {new_n}")
        up_seq = _upsample(overall_best_seq, new_n)
        up_score = _objective(up_seq)
        if up_score + 1e-14 < overall_best_score:
            overall_best_seq = up_seq
            overall_best_score = up_score
            print(f"[{time.time():.1f}] Upsample improved score to {overall_best_score:.6f}")

        # give a modest amount of time (up to 25 % of the remaining budget)
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        sub_deadline = time.time() + min(remaining * 0.25, 30.0)

        overall_best_seq, overall_best_score = _adam_optimize(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            tau_start=0.3,
            tau_end=5e-4,
            tau_decay=0.998,
            eval_every=150,
            step_size0=0.07,
        )
        print(f"[{time.time():.1f}] After Adam on size {new_n} best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Exact exponentiated‑gradient descent (sub‑gradient on max)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 12.0:
        sub_deadline = time.time() + remaining
        overall_best_seq, overall_best_score = _exp_gradient_descent(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.6,
            eval_every=80,
        )
        print(f"[{time.time():.1f}] After EG best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Frank‑Wolfe refinement (optional but cheap)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fw_deadline = time.time() + (deadline - time.time()) * 0.15
        overall_best_seq, overall_best_score = _frank_wolfe_descent(
            overall_best_seq.copy(),
            fw_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eval_every=100,
        )
        print(f"[{time.time():.1f}] After FW best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Final aggressive pairwise annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] Final annealing polish")
        overall_best_seq, overall_best_score = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
            max_trials=None,
            temp_factor=0.05,      # a bit hotter than the default
        )
        print(f"[{time.time():.1f}] After final anneal best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"[{time.time():.1f}] Final best score (objective=2·n·max(conv)/sum²): "
        f"{overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
