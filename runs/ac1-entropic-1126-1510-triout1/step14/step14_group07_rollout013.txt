def _fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1 << ((n+m-2).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A*B, n=size)
    return conv[:n+m-1]

def _adam_smooth_optimize(init_a, deadline, max_iters=200000,
                         lr0=0.8, alpha_start=0.5, alpha_end=80.0,
                         beta1=0.9, beta2=0.999, eps=1e-8,
                         eval_interval=250):
    n = init_a.shape[0]
    x = np.log(np.maximum(init_a, 1e-12))
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    best_seq = init_a.copy()
    best_score = _objective(init_a)
    for t in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        a = _softmax(x)
        conv = _fft_self_convolve(a)
        # compute smoothing parameter linearly ramping
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iters)
        maxc = conv.max()
        # compute weights
        shifted = conv - maxc
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        w /= w_sum
        # gradient wrt a
        w_rev = w[::-1]
        conv_full = _fft_convolve(a, w_rev)
        idx_center = len(w) - 1
        grad_a = 2.0 * conv_full[idx_center - np.arange(n)]
        # gradient wrt x
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1**t)
        v_hat = v / (1.0 - beta2**t)
        eta = lr0 / math.sqrt(t)  # decaying step size
        x -= eta * m_hat / (np.sqrt(v_hat) + eps)
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth Adam it={t:,} score={best_score:.6f} alpha={alpha:.2f}")
    # Return best found sequence (project onto simplex for safety)
    best_seq = _project_onto_simplex(best_seq, s=1.0)
    return best_seq, best_score

"""
Propose a non‑negative step‑function (probability vector) that minimises
``evaluate_sequence``.

Algorithm
---------

1️⃣  **Seed generation** –  a handful of analytically motivated families
    (uniform, sine‑weighted, centre‑exponential, left/right‑exponential,
    symmetric beta‑U‑shapes, Dirichlet with various concentrations,
    sparse random) are instantiated for several lengths.
    The 12 best seeds (lowest exact objective) are retained.

2️⃣  **Smooth Adam optimisation** –  each retained seed is refined by a
    mirror‑descent on the *smooth* max of the self‑convolution.
    The smooth approximation is
        φ(a)= (1/α)·log Σ_k exp(α·(a*a)_k) .
    Its gradient is computed analytically (via FFT) and an Adam
    optimiser is applied on the soft‑max parameters of the vector.
    The smoothing parameter α is slowly increased (continuation) so that
    the method tracks the true max while keeping a useful gradient.

3️⃣  **Long descent** –  the overall best seed from step 2 is optimised
    for the remaining time with a larger iteration budget and a larger
    final α, seeking a near‑optimal plateau.

4️⃣  **Local polishing** –  a greedy pairwise mass‑transfer step followed
    by a cheap simulated‑annealing style random walk cleans up any
    remaining irregularities.

All phases respect the global 1000 s budget.  The returned list sums to
one; ``evaluate_sequence`` will re‑normalise it anyway.

"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Simple utilities -------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


# ----------------------------------------------------------------------
# FFT based convolutions -------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using a real FFT."""
    n = len(a)
    m = len(b)
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
# Exact objective (assumes a sums to 1) ---------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector – sum = 1) -------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    w /= w.sum()
    return w


def _distribution_exponential_center(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(i - half))
    w /= w.sum()
    return w


def _distribution_exponential_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-rate * i)
    w /= w.sum()
    return w


def _distribution_exponential_right(n: int, rate: float) -> np.ndarray:
    return _distribution_exponential_left(n, rate)[::-1]


def _distribution_dirichlet(n: int, rng: np.random.Generator, conc: float) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a.astype(np.float64)


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric or asymmetric Beta distribution on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w[w < 0] = 0.0
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser ---------------------------------------------
def _adam_smooth_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    lr0: float = 0.8,
    alpha_start: float = 0.5,
    alpha_end: float = 80.0,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_interval: int = 250,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth approximation of max(conv(a,a)).
    Parameters α are increased linearly from alpha_start to alpha_end.
    Returns the best probability vector found (sum=1) and its exact score.
    """
    n = init_a.shape[0]
    # initialise log‑weights (avoid log(0))
    x = np.log(np.maximum(init_a, 1e-12))
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = init_a.copy()
    best_score = _objective(init_a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        a = _softmax(x)                         # stays on simplex
        conv = _fft_self_convolve(a)            # full self‑convolution (len=2n‑1)

        # linearly schedule α
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iters)

        # smooth‑max weights
        maxc = conv.max()
        shifted = conv - maxc                   # ≤0, stabilises exp()
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ∇_a φ = 2·(a * w) where * is the “sum‑over‑k” convolution
        # Implemented via FFT: grad_a = 2·conv_full[idx_center‑i]
        w_rev = w[::-1]
        conv_full = _fft_convolve(a, w_rev)    # length = n + (2n‑1)‑1 = 3n‑2
        idx_center = len(w) - 1                # index of zero‑lag in conv_full
        grad_a = 2.0 * conv_full[idx_center - np.arange(n)]

        # convert to gradient w.r.t. log‑weights (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update with a decaying step‑size
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = lr0 / math.sqrt(t)
        x -= eta * m_hat / (np.sqrt(v_hat) + eps)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] smooth‑Adam it={t:,}  α={alpha:.2f}  score={best_score:.6f}"
                )

    # final safeguard projection (numerical drift may happen)
    best_seq = _project_onto_simplex(best_seq, s=1.0)
    return best_seq, best_score

# Greedy pairwise refinement (same as reference) -----------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Cheap simulated‑annealing style local search -------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}"
            )
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Entry‑point used by the evaluator – returns a plain Python list."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # small safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------- 1️⃣  Seed generation ---------------------------
    lengths = [2500, 4000, 6000, 8000, 12000]
    p_grid = np.logspace(-1, 1.0, 8)          # sine‑weighted family
    lam_grid = np.linspace(0.0, 0.5, 6)       # mixing with uniform
    rate_grid = np.logspace(-4, -2, 5)       # exponential decay rates
    conc_grid = [0.05, 0.2, 1.0, 5.0]        # Dirichlet concentrations
    beta_grid = [(0.4, 0.4), (0.6, 0.6), (0.3, 0.8), (0.8, 0.3)]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic analytic families
        bases = [
            _distribution_uniform(n),
            _distribution_sine_weighted(n, p=1.0),
            _distribution_exponential_center(n, rate=0.005),
            _distribution_exponential_left(n, rate=0.003),
            _distribution_exponential_right(n, rate=0.003),
            _distribution_beta(n, 0.6, 0.6),
            _distribution_beta(n, 0.4, 0.4),
        ]
        for base in bases:
            sc = _objective(base)
            top_candidates.append((sc, base.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, base.copy()
                print(
                    f"[{time.time():.1f}] new global best (det) len={n} score={sc:.6f}"
                )

        # sine‑weighted + uniform mixing
        for p in p_grid:
            if time.time() >= deadline:
                break
            sine = _distribution_sine_weighted(n, p)
            for lam in lam_grid:
                cand = _mix_with_uniform(sine, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}] new global best sine p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                    )

        # centre‑exponential family
        for rate in rate_grid:
            if time.time() >= deadline:
                break
            cad = _distribution_exponential_center(n, rate)
            sc = _objective(cad)
            top_candidates.append((sc, cad.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cad.copy()
                print(
                    f"[{time.time():.1f}] new global best exp‑center rate={rate:.4e} score={sc:.6f}"
                )

        # Beta‑U‑shapes
        for a, b in beta_grid:
            if time.time() >= deadline:
                break
            bet = _distribution_beta(n, a, b)
            sc = _objective(bet)
            top_candidates.append((sc, bet.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, bet.copy()
                print(
                    f"[{time.time():.1f}] new global best beta a={a:.2f} b={b:.2f} score={sc:.6f}"
                )

        # Dirichlet draws
        for conc in conc_grid:
            for _ in range(3):
                if time.time() >= deadline:
                    break
                a = _distribution_dirichlet(n, rng, conc)
                sc = _objective(a)
                top_candidates.append((sc, a.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, a.copy()
                    print(
                        f"[{time.time():.1f}] new global best Dirichlet conc={conc:.3g} score={sc:.6f}"
                    )

        # Sparse random seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(
                    f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}"
                )

        # keep only the very best seeds for the next phase
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:12]

    # ------------------------------------------------------------------
    # Try to use a previously found construction, if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2️⃣  Short smooth‑Adam refinements for each retained seed
    refined: list[tuple[float, np.ndarray]] = []
    for sc, cand in top_candidates:
        if time.time() >= deadline:
            break
        slice_deadline = time.time() + min(15.0, (deadline - time.time()) * 0.05)
        best_seq, best_sc = _adam_smooth_optimize(
            cand.copy(),
            slice_deadline,
            max_iters=20_000,
            lr0=0.7,
            alpha_start=0.5,
            alpha_end=80.0,
            eval_interval=200,
        )
        refined.append((best_sc, best_seq))
        if best_sc < overall_best_score:
            overall_best_score, overall_best_seq = best_sc, best_seq.copy()
            print(
                f"[{time.time():.1f}] improved global best after short smooth‑Adam score={best_sc:.6f}"
            )

    # keep the best among the refined candidates
    if refined:
        overall_best_score, overall_best_seq = min(refined, key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 3️⃣  Long smooth‑Adam descent from the overall best seed
    if time.time() < deadline:
        remaining = deadline - time.time()
        long_deadline = time.time() + max(0.0, remaining - 30.0)
        print(
            f"[{time.time():.1f}] start long smooth‑Adam descent (≈{remaining-30:.1f}s left)"
        )
        overall_best_seq, overall_best_score = _adam_smooth_optimize(
            overall_best_seq.copy(),
            long_deadline,
            max_iters=500_000,
            lr0=0.9,
            alpha_start=1.0,
            alpha_end=200.0,
            eval_interval=250,
        )
        print(
            f"[{time.time():.1f}] after long descent best score {overall_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 4️⃣  Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 5️⃣  Final simulated‑annealing polishing
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.01,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score (exact) ≈ {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return plain Python list; evaluate_sequence will re‑normalise anyway.
    return overall_best_seq.tolist()
