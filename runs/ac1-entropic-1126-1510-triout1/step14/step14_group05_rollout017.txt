"""
Improved search for a non‑negative step function that minimizes
``evaluate_sequence``.  The pipeline is:

1️⃣  Enumerate many analytic families (uniform, arcsine, sine, cosine,
    quadratic, edge‑weighted, beta‑type with very small α, exponential‑edge,
    and also random convex combos of these bases).  Every candidate is a
    probability vector (∑a=1) and its exact objective
    ``2·n·max(convolution)`` is recorded.  The best ~30 seeds are kept.

2️⃣  For each of the best seeds run a two‑stage optimisation:

   •  **Smooth (soft‑max) descent** – a temperature‑annealed soft‑max
      approximation of the L∞‑norm.  Gradient is obtained with FFT‑based
      cross‑correlation, the update is an exponentiated‑gradient step and
      the iterate is projected onto the simplex.  τ is annealed from 0.05
      down to 1e‑5, giving a fine‑grained approximation of the true
      max‑convolution.

   •  **Exact sub‑gradient descent** – the true sub‑gradient of the
      max‑convolution is computed (only the active shifts contribute) and
      an EG update with a √t‑decaying step is applied.

   The best iterate of the two stages is kept as the new global best.

3️⃣  **Final greedy polishing** – a very cheap pairwise mass‑transfer
      hill‑climber (random source/destination, move up to 45 % of the
      source mass) is run until the time budget expires.  Every strict
      improvement updates the global optimum.

All steps respect a global 1000 s budget (a small safety margin is kept).
The returned list is a plain Python list; ``evaluate_sequence`` will
renormalise it internally.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _next_pow_two(x: int) -> int:
    """Small helper – next power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using real FFT."""
    n = a.shape[0]
    size = _next_pow_two(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_cross_correlate_valid(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (valid part):
        (w ⋆ a)[j] = Σ_i w[i+j]·a[i],  length = len(a)
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]          # = 2·n‑1
    size = _next_pow_two(m + n - 1)
    W = np.fft.rfft(w, n=size)
    A = np.fft.rfft(a, n=size)
    corr = np.fft.irfft(W * np.conj(A), n=size)
    return corr[:n]


def _fft_sliding_window_sum(w: np.ndarray, win: int) -> np.ndarray:
    """
    Compute Σ_{k=j}^{j+win‑1} w[k] for each j∈[0,win‑1].
    Returns an array of length ``win``.
    """
    m = w.shape[0]
    size = _next_pow_two(m + win - 1)
    W = np.fft.rfft(w, n=size)
    O = np.fft.rfft(np.ones(win, dtype=np.float64), n=size)
    conv = np.fft.irfft(W * O, n=size)
    start = win - 1
    return conv[start:start + win]


# ----------------------------------------------------------------------
# Analytic families – every function returns a *probability* vector
# (sum = 1, non‑negative)
# ----------------------------------------------------------------------


def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Symmetric edge‑weighted shape  w_i ∝ (i·(n‑i))^{-p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = n - i
    w = (i * j) ** (-p)
    s = w.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    w /= s
    return w


def _distribution_beta_edge(n: int, alpha: float) -> np.ndarray:
    """Beta‑type shape  w_i ∝ (i·(n‑i))^{α‑1}  (α>0)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = (i * (n - i)) ** (alpha - 1.0)
    w = np.clip(w, 0.0, None)
    s = w.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    w /= s
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse Dirichlet‑like seed with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Exact objective (the value we try to minimise)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth (soft‑max) gradient descent
# ----------------------------------------------------------------------


def _smooth_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    tau_start: float = 0.05,
    tau_end: float = 1e-5,
    lr0: float = 1.0,
    eval_every: int = 250,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on a soft‑max approximation of max‑conv."""
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    it = 0

    while time.time() < time_limit:
        it += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] smooth it={it} new best {best_score:.6f}")

        # linear annealing of τ
        frac = (time.time() - start) / (time_limit - start)
        tau = tau_start * (1.0 - frac) + tau_end * frac

        # soft‑max weights (stable formulation)
        maxc = conv.max()
        shifted = conv - maxc
        exp_vals = np.exp(shifted / tau)
        w = exp_vals / exp_vals.sum()  # length 2n‑1, sums to 1

        # gradient of the smoothed objective
        term1 = _fft_cross_correlate_valid(w, a)       # Σ w[i+j]·a[i]
        win_sum = _fft_sliding_window_sum(w, n)        # Σ_{k=j}^{j+n‑1} w[k]
        grad = term1 + a * win_sum

        # EG update (step ∝ 1/√t)
        eta = lr0 / math.sqrt(it)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if it % eval_every == 0:
            # already evaluated exact objective at each iteration
            pass

    return best_seq, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (same formulation as original)
# ----------------------------------------------------------------------


def _subgradient_descent_exact(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    t = 0

    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] exact it={t} new best {best_score:.6f}"
                )
    return best_seq, best_score

def _pairwise_swap(

    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = int(np.random.choice(src))
        j = int(np.random.randint(n - 1))
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45  # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a step‑function (plain Python list) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Scan a rich collection of analytic families on a range of lengths
    # ------------------------------------------------------------------
    lengths = [2000, 2600, 3000, 3400, 3800, 4200, 4600, 5000, 6000, 8000, 12000]

    # Parameter grids (expanded compared to the previous script)
    p_grid = np.concatenate(
        [
            np.logspace(-3, -0.1, 20),   # fine near zero
            np.linspace(0.05, 1.5, 15),  # moderate positive values
            np.linspace(-1.5, -0.05, 15),# moderate negative values
        ]
    )
    alpha_grid_edge = np.concatenate(
        [
            np.linspace(0.01, 0.08, 8),   # very edge‑heavy
            np.linspace(0.1, 0.5, 9),    # medium
        ]
    )
    alpha_grid_center = np.array([1.2, 1.5, 2.0, 3.0])  # central‑heavy
    lam_grid = np.linspace(0.0, 1.0, 21)
    c_grid = np.logspace(-4, -0.2, 12)  # exponential‑edge decay

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        base_arrays: list[np.ndarray] = []

        # 1a. Deterministic families
        for fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
        ):
            cand = fn(n)
            base_arrays.append(cand)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # 1b. Edge‑weighted families (p may be negative → central‑heavy)
        for p in p_grid:
            if time.time() >= deadline:
                break
            edge = _distribution_edge_weighted(n, float(p))
            base_arrays.append(edge)
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(edge, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(
                        f"[{time.time():.1f}] len={n} edge p={p:.4g} lam={lam:.2f} score={sc:.6f}"
                    )

        # 1c. Beta‑type (edge‑heavy) families
        for alpha in alpha_grid_edge:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_edge(n, float(alpha))
            base_arrays.append(beta)
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(beta, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(
                        f"[{time.time():.1f}] len={n} beta α={alpha:.3g} lam={lam:.2f} score={sc:.6f}"
                    )

        # 1d. Central‑heavy beta‑type families (α>1)
        for alpha in alpha_grid_center:
            if time.time() >= deadline:
                break
            cent = _distribution_beta_edge(n, float(alpha))
            base_arrays.append(cent)
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(cent, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(
                        f"[{time.time():.1f}] len={n} central β α={alpha:.2f} lam={lam:.2f} score={sc:.6f}"
                    )

        # 1e. Exponential‑edge families
        for c in c_grid:
            if time.time() >= deadline:
                break
            exp_edge = _distribution_exponential_edge(n, float(c))
            base_arrays.append(exp_edge)
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(exp_edge, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(
                        f"[{time.time():.1f}] len={n} exp c={c:.4g} lam={lam:.2f} score={sc:.6f}"
                    )

        # 1f. Random Dirichlet seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            base_arrays.append(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} score={sc:.6f}")

        # 1g. Sparse random seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(a)
            base_arrays.append(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

        # 1h. Random convex combos of the bases collected for this length
        base_arrays = list({id(arr): arr for arr in base_arrays}.values())  # dedup by id
        if len(base_arrays) >= 2:
            for _ in range(20):
                if time.time() >= deadline:
                    break
                k = int(rng.integers(2, min(5, len(base_arrays)) + 1))
                idxs = rng.choice(len(base_arrays), size=k, replace=False)
                weights = rng.dirichlet(np.ones(k))
                combo = np.zeros(n, dtype=np.float64)
                for w, ix in zip(weights, idxs):
                    combo += w * base_arrays[ix]
                # Numerical drift – re‑normalise
                combo = combo / combo.sum()
                sc = _objective(combo)
                top_candidates.append((sc, combo.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = combo.copy()
                    print(
                        f"[{time.time():.1f}] random combo len={n} score={sc:.6f}"
                    )

        # keep only the most promising seeds overall (memory / time bound)
        top_candidates.sort(key=lambda x: x[0])
        if len(top_candidates) > 80:
            top_candidates = top_candidates[:80]

    # ------------------------------------------------------------------
    # 2. Incorporate a previous construction if available
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / s
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never trigger)
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(
            f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 4. Deep optimisation on the most promising seeds
    # ------------------------------------------------------------------
    # Keep the top‑k seeds (k=10) for expensive optimisation
    top_candidates.sort(key=lambda x: x[0])
    seeds = [seq for _, seq in top_candidates[:10]]

    for idx, seed_seq in enumerate(seeds):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # allocate a slice of the remaining budget; give a bit more time to the first seeds
        per_seed = remaining / max(1, len(seeds) - idx)
        print(f"[{time.time():.1f}] optimisation on seed {idx+1}/{len(seeds)} (len={len(seed_seq)})")

        # ---- smooth descent (≈ 65 % of allocated time)
        sd_deadline = time.time() + per_seed * 0.65
        a_opt, sc_opt = _smooth_descent(
            seed_seq,
            sd_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            tau_start=0.05,
            tau_end=1e-5,
            lr0=1.0,
            eval_every=200,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(
                f"[{time.time():.1f}] after smooth seed {idx} new global best {overall_best_score:.6f}"
            )

        # ---- exact sub‑gradient descent (≈ 30 % of allocated time)
        sub_deadline = time.time() + per_seed * 0.30
        a_opt, sc_opt = _subgradient_descent_exact(
            overall_best_seq,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.9,
            eval_every=250,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(
                f"[{time.time():.1f}] after exact polish seed {idx} new best {overall_best_score:.6f}"
            )

        # ---- optional brief greedy polishing inside the per‑seed budget (if any time left)
        if time.time() < deadline:
            a_opt, sc_opt = _pairwise_swap(
                overall_best_seq.copy(),
                deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                max_trials=2000,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after quick swap seed {idx} new best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5. Final polishing on the overall best sequence
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=None,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best objective (2·n·max(conv)): {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list – evaluate_sequence will renormalise internally.
    return overall_best_seq.tolist()
