def _lp_descent(a0, deadline, best_seq, best_score, max_iter=20, t0=0.3):
    a = a0.copy()
    t = t0
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        n = a.shape[0]
        conv = _fft_self_convolve(a)
        M = conv.max()
        A = _build_conv_matrix(a)
        ub = np.full(2 * n - 1, M * (1.0 + 1e-12))
        c = -np.ones(n)
        # solve LP
        res = linprog(c, A_ub=A, b_ub=ub, bounds=(0, None), method='highs')
        if not res.success:
            print(f"[{time.time():.1f}] LP solve failed iteration {it}")
            break
        b = res.x
        if b.sum() <= 0:
            print(f"[{time.time():.1f}] LP returned zero sum iteration {it}")
            break
        g = b / b.sum()
        # mixture
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        a_new /= a_new.sum()
        cur_score = _objective(a_new)
        if cur_score + 1e-14 < best_score:
            best_score = cur_score
            best_seq = a_new.copy()
            a = a_new
            print(f"[{time.time():.1f}] LP it={it} improved to {best_score:.6f}")
            # gently increase t
            t = min(0.6, t * 1.1)
        else:
            # reduce step and possibly jitter
            t = max(0.01, t * 0.7)
            # optionally jitter
            rng = np.random.default_rng()
            jitter = rng.exponential(scale=0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0)
            a /= a.sum()
    return best_seq, best_score

"""
Hybrid optimizer for the max‑convolution objective.

The routine starts from a known good candidate (or a simple analytic
distribution).  It first runs a *linear‑programming* refinement that
maximises the total mass of a new distribution g while keeping the
convolution bound f∗g ≤ M (M = max(f∗f)).  Mixing f and g with a small
step size provably reduces the upper bound.  After a few LP steps we
smooth‑refine the result with an exponentiated‑gradient descent on a
log‑sum‑exp surrogate, and finally apply cheap pairwise swaps and a
simulated‑annealing clean‑up.  All operations respect the 1000 s budget
and return a plain list of non‑negative heights.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ------------------------------------------------------------
# optional SciPy imports (LP step)
try:
    from scipy.optimize import linprog
    from scipy.sparse import diags
    _HAS_SCIPY = True
except Exception:  # pragma: no cover
    _HAS_SCIPY = False

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b (real FFT)."""
    n = a.shape[0]
    m = b.shape[0]
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2).  a may be any non‑negative vector."""
    n = a.shape[0]
    sum_a = a.sum()
    if sum_a <= 0:
        return float("inf")
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max() / (sum_a ** 2)

# Analytic families – used for initial seeds
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    """Symmetric ‘tent’ shape – linear up then down."""
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 0:
        right = left[::-1]
    else:
        right = np.arange(half, 0, -1, dtype=np.float64)
    vals = np.concatenate([left, right])
    vals /= vals.sum()
    return vals

# Smooth surrogate gradient (exponentiated‑gradient descent)
def _smooth_gradient(a: np.ndarray, lam: float) -> tuple[np.ndarray, float]:
    """
    Returns (grad, max_conv) where
        grad = ∇_a φ_λ(a)  (φ_λ is the log‑sum‑exp surrogate)
        max_conv = max(conv(a,a)).
    a is assumed to sum to 1 (or any constant – the scaling cancels out).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)                     # length 2n‑1
    max_c = conv.max()
    z = lam * conv
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    s = exp_z / exp_z.sum()                          # softmax weights, sum to 1
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, s)               # length 3n‑2
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]          # central block, length n
    return grad, max_c


def _smooth_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    lam_start_factor: float = 0.12,
    lam_inc: float = 1.0015,
    beta: float = 0.9,
    eval_interval: int = 150,
    patience: int = 1800,
) -> tuple[np.ndarray, float]:
    """Annealed exponentiated‑gradient descent on the smooth surrogate."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    lam = lam_start_factor * n
    rng = np.random.default_rng()

    while time.time() < deadline and t < 600_000:
        t += 1
        grad, max_conv = _smooth_gradient(a, lam)

        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = lr0 / math.sqrt(t)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * momentum)
        a = np.maximum(a, 1e-30)
        a /= a.sum()

        lam *= lam_inc

        if t % eval_interval == 0:
            cur_score = 2.0 * n * max_conv / (a.sum() ** 2)
            if cur_score + 1e-14 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,}  new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

# Linear‑programming improvement (the classic f‑g iteration)
def _build_conv_matrix(f: np.ndarray):
    """
    Construct the sparse matrix A (shape (2n‑1, n)) such that
        (f * b) = A @ b
    for any non‑negative vector b.  Each column i contains the constant
    value f[i] on the diagonal offset i (exactly n non‑zero entries).
    """
    n = f.shape[0]
    # each diagonal i gets n identical entries = f[i]
    diag_data = [np.full(n, fi, dtype=np.float64) for fi in f]
    offsets = np.arange(n, dtype=int)   # 0 .. n‑1
    A = diags(diag_data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def _lp_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_iter: int = 20,
    t0: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Linear‑programming based descent.
    Repeatedly maximise Σb_j under the convolution bound and mix the
    result back into the current distribution.
    """
    if not _HAS_SCIPY:        # pragma: no cover
        return best_seq, best_score

    a = a0.copy()
    t = t0
    rng = np.random.default_rng()
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        n = a.shape[0]
        conv = _fft_self_convolve(a)
        M = conv.max()

        # build sparse convolution matrix and solve LP
        A = _build_conv_matrix(a)
        ub = np.full(2 * n - 1, M * (1.0 + 1e-12))
        c = -np.ones(n)                     # maximise Σb_j  ⇔ minimise -Σb_j
        res = linprog(c, A_ub=A, b_ub=ub, bounds=(0, None), method="highs")
        if not res.success:
            print(f"[{time.time():.1f}] LP solve failed at iter {it}")
            break
        b = np.maximum(res.x, 0.0)          # guard against tiny negatives
        if b.sum() == 0:
            print(f"[{time.time():.1f}] LP returned zero mass at iter {it}")
            break
        g = b / b.sum()                     # normalise to a simplex vector

        # convex combination
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()

        cur_score = _objective(a_new)
        if cur_score + 1e-14 < best_score:
            best_score = cur_score
            best_seq = a_new.copy()
            a = a_new
            print(f"[{time.time():.1f}] LP iter {it} improved to {best_score:.6f}")
            t = min(0.6, t * 1.1)          # gently increase step when progressing
        else:
            # no improvement – shrink step and add a tiny jitter
            t = max(0.02, t * 0.7)
            jitter = rng.exponential(scale=0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
    return best_seq, best_score

# Deterministic pairwise local improvement
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Move a small amount of mass from one index to another if it lowers the objective."""
    n = a.shape[0]

    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:

        trials += 1
        src_idx = np.where(a > 1e-12)[0]
        if src_idx.size == 0:
            break
        i = rng.choice(src_idx)

        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.2e} new best {best_score:.6f}")
    return best_seq, best_score

# Cheap simulated‑annealing clean‑up
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Simple SA that swaps a random fraction of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-15:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:

            best_score = cur
            best_seq = a.copy()

        else:
            # metropolis acceptance
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq


def _downsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a smaller length (maintains simplex property)."""
    return _upsample(seq, new_len)
