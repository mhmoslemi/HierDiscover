def _smooth_descent(a, deadline, best_seq, best_score,
                    eta0=0.4, temp0=0.01, temp_decay=0.9995, eval_interval=100):
    n = a.shape[0]
    t = 0
    best_local_seq = a.copy()
    best_local_score = _objective(a)
    temperature = temp0
    while time.time() < deadline and t < 200000:
        t += 1
        # calc conv and max
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        # compute softmax weights
        # subtract max to improve numeric stability
        w = np.exp((conv - max_c) / temperature)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # compute gradient via correlation
        # convolution of a and w via FFT
        # conv2 = np.convolve(a, w) (full)
        # use FFT
        size = 1 << ((len(a) + len(w) - 1 - 1).bit_length())
        A = rfft(a, n=size)
        W = rfft(w, n=size)
        grad_full = irfft(A * W, n=size)[:len(a)+len(w)-1]
        # gradient for i in 0..n-1 is 2*grad_full[ len(w)-1 + i ]
        grad = 2.0 * grad_full[len(w)-1 : len(w)-1 + n]

        # exponentiated gradient update
        a = a * np.exp(-eta0 / math.sqrt(t) * grad)

        # project onto simplex (just renormalize)
        total = a.sum()
        if total <= 0:
            a = np.full(n, 1.0/n)
        else:
            a = a / total

        # evaluate occasionally
        if t % eval_interval == 0:
            cur_score = _objective(a)
            if cur_score < best_local_score - 1e-14:
                best_local_score = cur_score
                best_local_seq = a.copy()
                # also update global best
                if cur_score < best_score - 1e-14:
                    best_score, best_seq = cur_score, a.copy()
                    print(f"time {time.time():.1f} it={t} global best {best_score}")
            # reduce temperature gradually
            temperature = max(temperature * temp_decay, 1e-6)

    return best_local_seq, best_local_score

"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.  The search consists of three stages:

1. **Diverse seeding** – several analytic families (uniform, arcsine,
   symmetric Beta, edge‑weighted, sine‑weighted) and random Dirichlet
   vectors are generated for a few moderate lengths (≈ 3‑6 k).  The
   best few seeds (by the exact objective 2·n·max(conv(a,a)) with
   sum(a)=1) are retained.

2. **Smoothed max descent** – for each retained seed we run a mirror‑
   descent on a *smooth* approximation of the ``max`` in the objective:
   we replace ``max(conv)`` by a temperature‑scaled log‑sum‑exp.
   This yields a dense weight vector w over all convolution indices.
   The gradient of the smooth objective is computed by a single FFT‑based
   correlation (O(N log N)).  An exponentiated‑gradient update
   ``a_i ← a_i·exp(-η·∇_i)`` keeps the vector non‑negative; a simple
   normalisation restores the simplex constraint.  The temperature and
   learning‑rate decay throughout the run.  The best point found is kept.

3. **Fine local polishing** – a cheap weighted pairwise refinement and
   a short simulated‑annealing walk are applied to the best sequence.

The whole procedure respects the given time budget (default 1000 s) and
returns a plain Python list of floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.006) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _symmetrize(seq: np.ndarray) -> np.ndarray:
    sym = (seq + seq[::-1]) * 0.5
    return sym / sym.sum()


def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq

# Smoothed max mirror descent -------------------------------------------
def _smooth_max_descent(
    a: np.ndarray,
    deadline: float,
    best_global_seq: np.ndarray,
    best_global_score: float,
    eta0: float = 0.6,
    temp0: float = 0.015,
    temp_decay: float = 0.9997,
    eval_interval: int = 120,
) -> tuple[np.ndarray, float]:
    """
    Mirror (exponentiated‑gradient) descent on the *smooth* objective
    L(a)=2·n·T·log Σ_k exp(conv[k]/T).  The gradient of the log‑sum‑exp
    is a weighted sum of the individual convolution gradients.
    """
    n = a.shape[0]
    size_fft = 1 << ((3 * n - 2 - 1).bit_length())   # enough for a*w
    best_seq = a.copy()
    best_score = _objective(a)
    t = 0
    temperature = temp0
    while time.time() < deadline and t < 200_000:
        t += 1
        # FFT of a (shared for both self‑conv and correlation)
        a_fft = rfft(a, n=size_fft)

        # --- self‑convolution (to obtain max) -----------------------
        conv_full = irfft(a_fft * a_fft, n=size_fft)[: 2 * n - 1]
        max_c = conv_full.max()

        # --- soft‑max weights ---------------------------------------
        # w_k ∝ exp((conv_k - max_c)/T)  → sum to 1
        w = np.exp((conv_full - max_c) / temperature)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --- gradient via correlation a * w -------------------------
        w_fft = rfft(w, n=size_fft)
        corr_full = irfft(a_fft * w_fft, n=size_fft)[: 3 * n - 2]
        # gradient_i = 2 * corr_full[ (len(w)-1) + i ]
        offset = len(w) - 1
        grad = 2.0 * corr_full[offset : offset + n]

        # --- exponentiated‑gradient update ---------------------------
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # re‑normalise onto simplex (keeps positivity)
        total = a.sum()
        if total <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= total

        # --- occasional evaluation ----------------------------------
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score, best_seq = cur, a.copy()
                # also update global if improved
                if cur + 1e-14 < best_global_score:
                    best_global_score, best_global_seq = cur, a.copy()
                    print(f"[{time.time():.1f}] it={t:,} new global best {cur:.6f}")
            # temperature cooling
            temperature = max(temperature * temp_decay, 1e-7)

    return best_seq, best_score

# Weighted pairwise refinement (reuse from the original code) -------
def _pairwise_weighted_refine(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer guided by the location of the maximal convolution entry."""
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        conv = _fft_self_convolve(a)
        k_max = int(conv.argmax())
        # impact L_i = a_{k_max-i}
        L = np.zeros(n, dtype=np.float64)
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        idx = np.arange(low, high + 1)
        L[idx] = a[k_max - idx]

        if L.sum() > 0:
            probs_i = L / L.sum()
            i = rng.choice(n, p=probs_i)
        else:
            positives = np.where(a > 1e-12)[0]
            if positives.size == 0:
                break
            i = rng.choice(positives)

        # destination prefers low impact
        probs_j = (L.max() - L) + 1e-12
        probs_j /= probs_j.sum()
        j = rng.choice(n, p=probs_j)
        if j == i:
            j = (j + 1) % n

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] weighted swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score

# Cheap simulated annealing (reuse) ------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Lightweight Metropolis‑type local walk."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

                best_global_score=overall_best_score,
                eta0=0.6,
                temp0=0.015,
                temp_decay=0.9997,
                eval_interval=80,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] After descent seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------

        overall_best_seq, overall_best_score = _pairwise_weighted_refine(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=200_000,
        )

    # ------------------------------------------------------------------
    # 7.  Final cheap simulated‑annealing clean‑up
    if time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] Final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
