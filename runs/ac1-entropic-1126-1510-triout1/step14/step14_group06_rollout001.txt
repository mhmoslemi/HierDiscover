def _distribution_beta(n,p,q):
    x = (np.arange(n) + 0.5) / n
    w = x**(p-1) * (1-x)**(q-1)
    w[w<0]=0
    if w.sum()==0: w = np.full(n,1.0/n)
    else: w/=w.sum()
    return w
```

def _random_local_search(a, deadline, best_seq, best_score, rng, max_trials=2000):
    n = a.shape[0]
    cur_best = best_seq.copy()
    cur_score = best_score
    while time.time() < deadline and max_trials>0:
        i = rng.integers(n)
        j = rng.integers(n)
        if i==j: continue
        delta = a[i] * rng.random() * 0.2
        if delta<=0: continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # keep non-negative
        if new_a[i] < -1e-12: continue
        # renorm (sum unchanged)
        # evaluate
        cur = _objective(new_a)
        if cur < cur_score - 1e-12:
            cur_score = cur
            cur_best = new_a.copy()
            a = new_a
        max_trials -=1
    return cur_best, cur_score

def _perturbed_seeds(base, rng, num, sigma):
    seeds=[]
    n=base.shape[0]
    for _ in range(num):
        delta = rng.normal(scale=sigma, size=n)
        new = base + delta
        new = np.maximum(new,0)
        s = new.sum()
        if s==0:
            new = base.copy()
        else:
            new/=s
        seeds.append(new)
    return seeds

"""
Propose a step‑function (probability vector) that minimizes
``evaluate_sequence``.
The search:

1.  Start from a small length (512) and generate many analytic seeds
    (uniform, arcsine, Hann, Beta, edge‑weighted, generalized‑normal,
    power‑law, Dirichlet, sparse).  Also create random perturbations of the
    current best.

2.  Keep the few best seeds and refine each of them with two complementary
    optimisers:
        * Adam on a temperature‑smoothed max‑convolution
          (τ‑log‑sum‑exp), gradually cooling τ.
        * Exponentiated‑gradient descent on the exact max‑convolution
          objective (the same objective used by ``evaluate_sequence``).

3.  After a few seconds of refinement, up‑sample the best candidate by a
    factor of two (linear interpolation) and repeat the whole process.
    The up‑sampling loop runs until the time budget is exhausted or the
    length reaches 32768.

4.  A lightweight random‑local‑search (pairwise mass transfers guided by the
    sub‑gradient of the active convolution entry) is interleaved to escape
    local minima.

5.  Finally a short pairwise annealing polish is applied.

All operations respect the total ``budget_s`` (default 1000 s) and the
returned object is a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time

import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s <= 0:
        raise ValueError("simplex radius must be positive")
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation of a (len n) with w (len m) for non‑negative shifts.
    Returns an array of length n where element i equals Σ_j a_j·w_{j+i}.
    Implemented via FFT using convolution with the reversed weight vector.
    """
    n = a.shape[0]
    m = w.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()
    rev_w = w[::-1]
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_w, n=size)
    conv = np.fft.irfft(A * B, n=size)
    start = m - 1
    return conv[start:start + n]

# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, same as evaluate_sequence after fixing sum=1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return the exact value of ``evaluate_sequence`` for an un‑normalised a."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))

# ----------------------------------------------------------------------
# Sub‑gradient of the active convolution entry (for max‑conv)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Compute the sub‑gradient of the currently maximal convolution entry.
    Returns (grad, active_index).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))
    grad = np.zeros_like(a)
    # The convolution entry k is Σ_i a_i a_{k-i} (with appropriate zero padding)
    # The derivative wrt a_j contributes a_{k-j} + a_{j-(k-n)} depending on index.
    if k < n:
        # contributions from pairs (j, k-j) with 0≤j≤k
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        # k >= n  →  pairs where one index overflows the right side
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k

# ----------------------------------------------------------------------
# Analytic families (return a probability vector, sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{-p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(-|x-½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    """Monotone decreasing power‑law (p>0)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) while preserving simplex
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new


# ----------------------------------------------------------------------
# Adam optimisation of the τ‑smoothed max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    tau_start: float = 0.2,
    tau_end: float = 0.001,
    tau_decay: float = 0.999,
    step_size0: float = 0.05,
    eval_every: int = 150,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the temperature‑smoothed objective
    2·n·τ·log‑sum‑exp(conv/τ).
    """
    n = a0.shape[0]
    # optimise in the (log‑prob) space to keep non‑negativity automatically
    x = np.log(a0 + 1e-15)  # logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    tau = tau_start

    while time.time() < deadline and tau > tau_end:
        t += 1

        # softmax probabilities
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # convolution of a with itself
        conv = _fft_self_convolve(a)          # length 2n-1
        max_conv = conv.max()

        # τ‑softmax weights over convolution entries
        exp_vals = np.exp((conv - max_conv) / tau)
        w = exp_vals / exp_vals.sum()          # same length as conv

        # correlation of a with the τ‑weights (length n)
        corr = _fft_corr(a, w)

        # gradient w.r.t. a for the smoothed objective (derived analytically)
        grad_a = 4.0 * n * corr

        # chain rule to logits (softmax Jacobian)
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = step_size0 / (np.sqrt(v_hat) + eps)
        x = x - lr * m_hat

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam iter={t} τ={tau:.6f} new best {best_score:.6f}")

        # cool temperature
        tau = max(tau * tau_decay, tau_end)

    # final projection onto simplex (might have drifted slightly)
    a_final = np.exp(x - np.max(x))
    a_final /= a_final.sum()
    final_score = _objective(a_final)
    if final_score < best_score:
        best_score = final_score
        best_seq = a_final.copy()
        print(f"[{time.time():.1f}] Adam final projection new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative mirror descent (exponentiated gradient) on the exact
    objective.  The iterates stay on the simplex.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        # re‑project onto the simplex
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new global best {best_score:.6f}")

    return best_seq, best_score

# Simple greedy mass‑transfer based on sub‑gradient (helps escape plateaus)
# ----------------------------------------------------------------------
def _gradient_transfer(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a fraction of mass from the coordinate with the largest
    sub‑gradient to the one with the smallest.  Accept every move (the step
    size is small enough to guarantee descent on average) and keep the best.
    """
    n = a.shape[0]
    cur_seq = a.copy()
    cur_score = best_score
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        grad, _ = _grad_of_max_conv(cur_seq)
        i_max = int(np.argmax(grad))
        i_min = int(np.argmin(grad))
        if cur_seq[i_max] < 1e-12:
            break
        delta = cur_seq[i_max] * rng.random() * 0.25  # up to 25 % of that bin
        cur_seq[i_max] -= delta
        cur_seq[i_min] += delta
        # projection not needed – sum unchanged, non‑negativity preserved
        cur = _objective(cur_seq)
        if cur < cur_score - 1e-14:
            cur_score = cur
            best_seq = cur_seq.copy()
            best_score = cur
            print(f"[{time.time():.1f}] grad‑transfer it={it} new best {best_score:.6f}")
        # occasional re‑normalisation to avoid floating‑point drift
        if it % 200 == 0:
            cur_seq = cur_seq / cur_seq.sum()
    return best_seq, best_score

# Pairwise annealing (simulated‑annealing style)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Randomly relocate a fraction of mass between two bins with an annealing schedule."""
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    trials = 0
    while time.time() < deadline:
        trials += 1
        # pick source (must have some mass) and destination (different)
        src = rng.choice(np.where(a > 1e-12)[0])
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1
        # transfer a random fraction of src's mass
        delta = a[src] * rng.random() * 0.4
        if delta < 1e-16:
            continue
        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta
        cur = _objective(new_a)
        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.01
        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)
        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal src={src}->dst={dst} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score

# Generate diverse seeds for a given length
# ----------------------------------------------------------------------
def _generate_seeds(n: int, rng: np.random.Generator, base: np.ndarray | None) -> list[np.ndarray]:
    seeds = []

    # deterministic analytic families
    seeds.append(_distribution_uniform(n))
    seeds.append(_distribution_arcsine(n))
    seeds.append(_distribution_hann(n))

    # randomised parametric families
    for _ in range(4):
        p = 10 ** rng.uniform(-1, 1)           # 0.1 … 10
        q = 10 ** rng.uniform(-1, 1)
        seeds.append(_distribution_beta(n, p, q))
        seeds.append(_distribution_edge_weighted(n, p))
        seeds.append(_distribution_generalized_normal(n, q=rng.uniform(0.3, 3.0)))
        seeds.append(_distribution_power_law(n, p=rng.uniform(0.2, 2.0)))
        # mixture with uniform
        lam = rng.random()
        seeds.append(_mix_with_uniform(seeds[-1], lam))

    # Dirichlet samples
    for _ in range(3):
        alpha = rng.uniform(0.01, 2.0)
        seeds.append(rng.dirichlet(np.full(n, alpha)))

    # sparse random
    for _ in range(3):
        seeds.append(_distribution_random_sparse(n, rng, density=0.02))

    # perturbations of the current best (if supplied)
    if base is not None:
        for sigma in (0.02, 0.05, 0.1):
            noise = rng.normal(scale=sigma, size=n)
            pert = base + noise
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s == 0.0:
                pert = base.copy()
            else:
                pert /= s
            seeds.append(pert)

    return seeds
