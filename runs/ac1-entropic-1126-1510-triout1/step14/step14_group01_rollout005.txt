    max_val = conv.max()

"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.

Algorithm
---------

1. **Diverse seeds** – generate many candidate distributions of modest
   length (≈2‑5 k) from a rich collection of analytic families:
   uniform, arcsine, symmetric / asymmetric Beta, edge‑weighted power‑law,
   edge‑exponential, sine‑weighted, random sparse Dirichlet, etc.  Random
   mixtures with the uniform distribution are also tried.  All candidates
   are projected onto the probability simplex and evaluated exactly
   (objective = 2·n·max(conv(a,a))).

2. **Annealed smoothed‑gradient descent** – for the top few seeds run an
   exponentiated‑gradient optimiser that minimises the log‑sum‑exp
   surrogate of the max‑convolution.  The temperature λ starts very
   smooth (≈0.1·n) and is multiplied by a factor >1 each iteration,
   gradually sharpening the surrogate.  The update uses a momentum term
   and Euclidean‑projection onto the simplex.

3. **Max‑subgradient refinement** – after the smoothed phase we switch
   to a *pure* subgradient of the exact max‑convolution.  At each step
   the shift `k*` attaining the current maximal autocorrelation is
   located, the corresponding gradient `g_i = 2·a_{k*-i}` (where defined)
   is formed, and an exponentiated‑gradient update
   `a ← a·exp(-η·g)` is applied.  This directly attacks the bottleneck
   entry and often yields a small but consistent improvement.

4. **Upsampling** – if time permits the best distribution is linearly
   interpolated to a larger length (up to ≈18 k) and refined again with
   the two‑phase gradient scheme.

5. **Deterministic pairwise sweep & cheap simulated‑annealing** – final
   local moves (mass transfers between two coordinates) polish the
   solution.

All stages respect the overall time budget (``budget_s`` seconds).  The
function returns a plain Python list; ``evaluate_sequence`` will
perform the final clipping to non‑negative values and normalise as
required.

The implementation is fully self‑contained, uses only ``numpy`` and
standard libraries, and runs on a single CPU core (well within the
2‑CPU limit).

import time

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Return the Euclidean projection of v onto {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolutions
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT) – length 2·n‑1."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b (real FFT)."""
    n = a.shape[0]
    m = b.shape[0]
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant: a must sum to 1)
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) for probability vectors."""
    b = _fft_self_convolve(a)
    return 2.0 * a.shape[0] * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric Beta(a,b)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^(-p), i = i+0.5, j = n-i-0.5."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    """Weight ∝ i^(-p)+j^(-p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_exponential(n: int, c: float) -> np.ndarray:
    """Edge‑exponential shape: exp(-c·i) + exp(-c·(n-1-i))."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.006) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, where U is the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _mix_two(a: np.ndarray, b: np.ndarray, lam: float) -> np.ndarray:
    """lam·a + (1‑lam)·b, lam∈[0,1]."""
    return lam * a + (1.0 - lam) * b


# ----------------------------------------------------------------------
# Smoothed surrogate gradient (log‑sum‑exp)
def _smooth_gradient(a: np.ndarray, lam: float) -> tuple[np.ndarray, float]:
    """Return (grad, max_conv) for φ_λ(a) = (1/λ)·log Σ exp(λ·conv)."""
    n = a.shape[0]

    conv = _fft_self_convolve(a)           # length 2n‑1
    max_c = conv.max()

    z = lam * conv
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    s = exp_z / exp_z.sum()                # softmax over shifts

    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, s)    # length 3n‑2
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]   # centre block, length n
    return grad, max_c


# ----------------------------------------------------------------------
# Pure max‑subgradient (on the exact max entry)
def _max_subgradient(a: np.ndarray) -> tuple[np.ndarray, float, int]:
    """Return (g, max_conv, k_max) where g is the sub‑gradient of max(conv(a,a))."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_c = conv.max()
    k = int(np.argmax(conv))               # first maximal shift
    i = np.arange(n)
    idx = k - i
    valid = (idx >= 0) & (idx < n)
    g = np.zeros_like(a)
    g[valid] = 2.0 * a[idx[valid]]
    return g, max_c, k

# Exponentiated‑gradient descent on the smooth surrogate
def _smooth_descent(a0: np.ndarray, deadline: float,
                    best_seq: np.ndarray, best_score: float,
                    lr0: float = 0.8, lam_start_factor: float = 0.15,
                    lam_inc: float = 1.0015, beta: float = 0.9,
                    eval_interval: int = 200, patience: int = 2500) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    lam = lam_start_factor * n
    rng = np.random.default_rng()

    while time.time() < deadline and t < 800_000:
        t += 1
        grad, max_conv = _smooth_gradient(a, lam)

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * momentum)

        # enforce positivity & simplex
        a = np.maximum(a, 1e-30)
        a /= a.sum()

        lam *= lam_inc

        if t % eval_interval == 0:
            cur_score = 2.0 * n * max_conv
            if cur_score + 1e-14 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] smooth it={t:,} new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0
    return best_seq, best_score

        a = np.maximum(a, 1e-30)
        a /= a.sum()

        if t % eval_interval == 0:
            cur_score = 2.0 * n * max_conv
            if cur_score + 1e-14 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] max‑subgrad it={t:,} new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            # occasional random jitter to escape tiny plateaus
            jitter = rng.exponential(scale=0.0015, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

# Deterministic pairwise sweep (mass transfer)
def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float,
                         max_trials: int | None = None) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Cheap simulated‑annealing clean‑up (random mass transfers)
def _random_local_search(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float,
                         init_temp: float = 0.008, decay: float = 0.9995) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsample a simplex vector via linear interpolation
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Main entry point – returns a list of non‑negative step heights.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Seed generation (moderate lengths)
    lengths = [1800, 2400, 3000, 3600, 4200, 5000]
    p_grid = np.concatenate([
        np.logspace(-4, -1, 8),   # 0.0001 … 0.1
        np.logspace(-1, 0, 9),   # 0.1 … 1
        np.logspace(0, 0.7, 8),  # 1 … 5
    ])
    lam_grid = np.linspace(0.0, 0.9, 11)
    c_grid = np.logspace(-2, 1, 8)   # for edge‑exponential

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating seeds for n={n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (base) {sc:.6f}")

        # parameterised families
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            edge2 = _distribution_edge_inversepow(n, p)
            sine = _distribution_sine_weighted(n, p)

            for base in (beta, edge, edge2, sine):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(f"[{time.time():.1f}] new best (param) n={n} p={p:.3g} lam={lam:.2f} {sc:.6f}")

        # edge‑exponential family
        for c in c_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_edge_exponential(n, c)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best (exp) n={n} c={c:.3g} {sc:.6f}")

        # asymmetric Beta
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = rng.uniform(0.001, 6.0)
            b = rng.uniform(0.001, 6.0)
            cand = _distribution_beta_asym(n, a, b)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best asym‑beta n={n} a={a:.3g} b={b:.3g} {sc:.6f}")

        # random Dirichlet
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_dirichlet(n, rng)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best Dirichlet n={n} {sc:.6f}")

        # sparse random
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best sparse n={n} {sc:.6f}")

        # keep only the elite seeds (30 best so far)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2. Inject any previously discovered construction (if present)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never happen)
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Heavy annealed descent on the elite seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:7]]
        per_seed = remaining * 0.55 / max(1, len(seeds))
        print(f"[{time.time():.1f}] smoothed descent on {len(seeds)} seeds, ~{per_seed:.1f}s each")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            a_opt, sc_opt = _smooth_descent(
                seed_seq.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.95,
                lam_start_factor=0.12,
                lam_inc=1.002,
                beta=0.9,
                eval_interval=150,
                patience=1800,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after smooth seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Max‑subgradient refinement on the current best
    remaining = deadline - time.time()
    if remaining > 20.0:
        sub_dead = time.time() + remaining * 0.35
        print(f"[{time.time():.1f}] max‑subgradient refinement (≈{remaining*0.35:.1f}s)")
        overall_best_seq, overall_best_score = _max_subgradient_descent(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.6,
            eval_interval=120,
            patience=1500,
        )

    # ------------------------------------------------------------------
    # 6. Upsampling to a larger length (if time permits)
    remaining = deadline - time.time()
    if remaining > 120.0:
        large_n = min(18000, len(overall_best_seq) * 2)
        print(f"[{time.time():.1f}] up‑sampling to {large_n}")
        up_seq = _upsample(overall_best_seq, large_n)

        sub_dead = time.time() + remaining * 0.55
        # a short smooth phase followed by a short max‑subgradient phase
        up_seq, up_sc = _smooth_descent(
            up_seq,
            sub_dead * 0.45,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.6,
            lam_start_factor=0.15,
            lam_inc=1.0018,
            beta=0.92,
            eval_interval=120,
            patience=2000,
        )
        if up_sc < overall_best_score:
            overall_best_score, overall_best_seq = up_sc, up_seq.copy()
            print(f"[{time.time():.1f}] up‑sampled smooth new best {overall_best_score:.6f}")

        # final max‑subgradient on up‑sampled vector (short)
        remaining = deadline - time.time()
        if remaining > 30.0:
            sub_dead = time.time() + remaining * 0.45
            overall_best_seq, overall_best_score = _max_subgradient_descent(
                overall_best_seq.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.45,
                eval_interval=100,
                patience=1200,
            )

    # ------------------------------------------------------------------
    # 7. Deterministic pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] deterministic pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 8. Final cheap simulated‑annealing clean‑up
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
