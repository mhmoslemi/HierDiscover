def _corr_fft(w, a):
    n = len(a)
    m = len(w)
    size = 1 << ((n + m - 1 - 1).bit_length())
    A_rev = np.fft.rfft(a[::-1], n=size)
    W = np.fft.rfft(w, n=size)
    c = np.fft.irfft(A_rev * W, n=size)  # length size
    # The convolution length = n + m - 1
    # we need slice from (n-1) to (n-1)+(n) (i.e., n elements)
    start = n - 1
    g = c[start:start + n]
    return g

The objective is  

\[
\text{score}(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2},
\qquad a_i\ge 0 .
\]

Because the score is homogeneous, we can restrict ourselves to probability vectors
(\(\sum a_i=1\)).  
For a probability vector the problem is to **minimise the peak of the self‑convolution**.  
The code below follows a multi‑phase strategy:

1. **Diversified seeding** – analytic families (uniform, arcsine, Hann, edge‑weighted,
   symmetric‑Beta, generalized‑normal, random Dirichlet, sparse) are sampled for a
   modest size (`n = 2048`). The 30 best candidates are kept.

2. **Smooth exponential‑gradient polishing** – instead of the non‑smooth sub‑gradient
   of `max(conv)`, a temperature‑smoothed version is used:
   \[
   w_i=\frac{\exp\!\big((a*a)_i/\tau\big)}{\sum_j\exp\!\big((a*a)_j/\tau\big)} .
   \]
   The gradient of the smoothed objective is a *correlation* between `w` and `a`.
   An exponentiated‑gradient update with momentum is applied, while the temperature
   `τ` is slowly annealed.  This gives a much richer search direction and escapes the
   plateaus that limited the previous implementation.

3. **Progressive up‑sampling** – the current best vector is linearly interpolated to
   double its length (up to ≈ \(6·10^4\)).  After each up‑sample a medium‑length smooth
   gradient run adapts the shape to the finer discretisation.

4. **Final pairwise annealing** – a simple simulated‑annealing that randomly transfers
   a small fraction of mass between two bins finishes the optimisation.

All phases respect the 1000 s budget (with a 2 s safety margin).  
The routine returns a plain Python list ready for `evaluate_sequence`.  

```python
"""
Propose a non‑negative step function (list of heights) that minimises

    2·n·max(conv(a,a)) / (∑a)²

The algorithm consists of four phases:

1.  Rich analytical seeding (uniform, arcsine, Hann, edge‑weighted,
    symmetric β, generalized‑normal, Dirichlet, sparse).  The 30 best
    seeds are kept.

2.  Smooth exponential‑gradient descent.  Instead of the non‑smooth
    sub‑gradient of max(conv) we use a temperature‑smoothed soft‑max
    weight vector, compute its correlation with the current a,
    and apply a mirrored (exponential) gradient update with EMA momentum.
    The temperature τ is annealed during the run.

3.  Progressive up‑sampling – the current best vector is interpolated to
    double length (up to ≈ 65536) and refined again by the smooth GD.

4.  Pairwise‑mass‑transfer simulated annealing as a final polish.

All steps stay inside the 1000 s wall‑clock budget and return a plain
list of floats suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used after additive steps
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (real → real)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (real FFT, returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Correlation w ⋆ a for “valid” shifts (produces length n)
# ----------------------------------------------------------------------
def _corr_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """Compute correlation (valid) of w (len 2·n‑1) with a (len n)."""
    n = a.shape[0]
    m = w.shape[0]               # = 2·n‑1
    size = 1 << ((n + m - 1) - 1).bit_length()
    A_rev = np.fft.rfft(a[::-1], n=size)
    W = np.fft.rfft(w, n=size)
    c = np.fft.irfft(A_rev * W, n=size)   # length = size
    start = n - 1
    return c[start:start + n]


# ----------------------------------------------------------------------
# Exact objective for a probability vector (∑a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families – each returns a *probability* vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{‑p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) sampled on cell mid‑points."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    w = np.where(
        (x == 0) | (x == 1),
        0.0,
        np.exp((alpha - 1) * np.log(x) + (alpha - 1) * np.log1p(-x))
    )
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(‑|x‑½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int,
    rng: np.random.Generator,
    density: float = 0.01,
) -> np.ndarray:
    """Sparse random vector with given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1‑λ) + uniform·λ."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smooth exponentiated‑gradient descent (entropy mirror descent)
# ----------------------------------------------------------------------
def _smooth_exp_gradient(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.5,
    beta: float = 0.9,
    tau_decay: float = 0.997,
    eval_every: int = 30,
) -> tuple[np.ndarray, float]:
    """
    Mirror descent on the temperature‑smoothed supremum.
    Uses an EMA of the gradient (momentum) and an annealing schedule
    for the temperature τ.
    """
    a = a0.copy()
    n = a.shape[0]
    # initialise temperature from the current max convolution
    conv0 = _fft_self_convolve(a)
    tau = max(conv0.max() * 0.5, 1e-12)

    m = np.zeros_like(a)          # EMA of gradients
    eta = eta0
    t = 0
    while time.time() < deadline:
        t += 1

        # 1) convolution and soft‑max weights
        conv = _fft_self_convolve(a)
        scaled = conv / tau
        max_scaled = scaled.max()
        exp_vals = np.exp(scaled - max_scaled)          # stability
        w = exp_vals / exp_vals.sum()

        # 2) gradient = 2 * correlation(w, a)
        grad = 2.0 * _corr_fft(w, a)

        # 3) EMA update
        m = beta * m + (1.0 - beta) * grad

        # 4) multiplicative (exponentiated) update
        a = a * np.exp(-eta * m)

        # 5) renormalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # occasional re‑projection to avoid drift
        if t % (eval_every * 10) == 0:
            a = _project_onto_simplex(a, s=1.0)

        # 6) temperature annealing
        tau *= tau_decay
        if tau < 1e-15:
            tau = 1e-15

        # 7) evaluate and possibly improve the global best
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                eta = min(eta0, eta * 1.02)   # reward improvement
                print(f"[{time.time():.1f}] Smooth GD it={t:,} new best {best_score:.6f}")
            else:
                eta *= 0.995                # gentle decay

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise mass‑transfer simulated annealing
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        src = rng.choice(np.where(a > 1e-14)[0])
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1
        delta = a[src] * rng.random() * 0.3
        if delta < 1e-16:
            continue
        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta
        cur = _objective(new_a)

        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.02

        accept = False
        if cur < best_score - 1e-15:
            accept = True
        elif temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature):
            accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal {src}->{dst} Δ={delta:.1e} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling while preserving total mass (=1)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new

                print(f"[{time.time():.1f}] edge p={p:.3g} λ={lam:.2f} new best {sc:.6f}")

    # symmetric Beta – much finer sweep in the “U” region

        np.linspace(0.8, 2.5, 12),              # moderate

    for alpha in beta_alphas:

            break
        cand = _distribution_beta(n0, float(alpha))
        sc = _objective(cand)
        cand_pool.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] beta α={alpha:.3g} new best {sc:.6f}")

    # generalized normal – explore a broader range
    for q in np.linspace(0.25, 5.0, 16):
        if time.time() >= deadline:
            break
        cand = _distribution_generalized_normal(n0, float(q))
        sc = _objective(cand)
        cand_pool.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] gn q={q:.2f} new best {sc:.6f}")

    # random Dirichlet seeds
    for _ in range(12):
        if time.time() >= deadline:
            break
        log_alpha = rng.uniform(-2.0, 1.0)      # α in [0.01, 10]
        alpha = 10 ** log_alpha
        cand = rng.dirichlet(np.full(n0, alpha))
        sc = _objective(cand)
        cand_pool.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] Dirichlet α={alpha:.3g} new best {sc:.6f}")

    # sparse random seeds
    for _ in range(8):
        if time.time() >= deadline:
            break
        cand = _distribution_random_sparse(n0, rng, density=0.02)
        sc = _objective(cand)
        cand_pool.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] sparse new best {sc:.6f}")

    # keep the 30 most promising seeds
    cand_pool.sort(key=lambda x: x[0])
    cand_pool = cand_pool[:30]

    # ------------------------------------------------------------------
    # 2️⃣  Incorporate a previously found construction if present
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.abs(prev)
            total = prev.sum()
            if total == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            else:
                prev = prev / total
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
            cand_pool.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3️⃣  Ensure we have a fallback solution
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        overall_best_seq = _distribution_uniform(n0)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Quick smooth‑GD on every seed (short slice)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 15.0 and cand_pool:
        per_seed = max(0.2, remaining * 0.25 / max(len(cand_pool), 1))
        print(f"[{time.time():.1f}] quick smooth‑GD on {len(cand_pool)} seeds (~{per_seed:.2f}s each)")
        for idx, (sc, a0) in enumerate(cand_pool):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            a_opt, sc_opt = _smooth_exp_gradient(
                a0.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.6,
                beta=0.9,
                eval_every=30,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after quick GD seed {idx} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Longer smooth‑GD on the current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_dead = time.time() + (deadline - time.time()) * 0.35
        a_opt, sc_opt = _smooth_exp_gradient(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.5,
            beta=0.9,
            eval_every=60,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] long smooth‑GD improved best to {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Progressive up‑sampling (doubling while time permits)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n > 65536:
            break
        print(f"[{time.time():.1f}] up‑sampling {cur_n} → {target_n}")
        up = _upsample(overall_best_seq, target_n)
        up_sc = _objective(up)
        if up_sc < overall_best_score:
            overall_best_seq, overall_best_score = up, up_sc
            print(f"[{time.time():.1f}] up‑sampled new best {overall_best_score:.6f}")

        # medium smooth‑GD on the up‑sampled vector
        sub_dead = time.time() + (deadline - time.time()) * 0.2
        a_opt, sc_opt = _smooth_exp_gradient(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.45,
            beta=0.9,
            eval_every=50,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
            print(f"[{time.time():.1f}] GD after up‑sample new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣  Final annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing phase")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
            print(f"[{time.time():.1f}] after annealing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 8️⃣  Report and return
    # ------------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)) = {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
```
