def solve_lp(f):
    n = len(f)
    # Build constraint matrix A_ub of shape (2n-1, n+1)
    rows = 2*n - 1
    A_ub = np.zeros((rows, n+1))
    # For each k, fill conv coefficients:
    for k in range(rows):
        # indices i where 0 <= i < n, j = k-i within [0,n-1]
        # Equivalent: i from max(0, k-(n-1)) to min(n-1, k)
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        # for i in range(i_start, i_end+1):
        #   j = k - i
        #   A_ub[k, i] = f[j]  # because term f_i * g_{k-i}?
        # Wait need correct: (f * g)_k = sum_i f_i * g_{k-i}.
        # So coefficient for g_j is f_{k-j}.
        # So for each j in [0,n-1], coefficient = f[k - j] if 0 <= k - j < n else 0.
        #    So we can fill using j loop:
        # For j from i_start' = max(0, k-(n-1)) to i_end' = min(n-1, k): coefficient = f[k - j].
        # That's simpler: j = index of g variable.
        for j in range(i_start, i_end+1):
            coeff = f[k - j]
            A_ub[k, j] = coeff
        A_ub[k, -1] = -1.0  # -t

    # Equality constraint sum(g)=1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Objective: minimize t => c = [0,...,0, 1]
    c = np.zeros(n+1)
    c[-1] = 1.0

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=np.zeros(rows), A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError('LP failed')
    g = res.x[:n]
    t = res.x[-1]
    return g, t

def _solve_lp_for_f(f):
    n = f.shape[0]
    rows = 2*n - 1
    K = np.arange(rows)[:, None]
    J = np.arange(n)[None, :]
    idx = K - J
    mask = (idx >= 0) & (idx < n)
    C = np.zeros((rows, n), dtype=np.float64)
    C[mask] = f[idx[mask]]
    A_ub = np.hstack([C, -np.ones((rows,1))])

"""
Hybrid optimizer for the step‑function bound.
-------------------------------------------------
1. Generate a diverse pool of seed distributions (uniform, arcsine,
   edge‑weighted, beta, Dirichlet, sparse random) at a moderate size.
2. For each seed:
   * Run a smoothed log‑sum‑exp (exponentiated‑gradient) descent.
   * Follow with a direct sub‑gradient descent on the exact max‑convolution.
   * Keep the best sequence seen.
3. Upsample the current best to a larger length and repeat the two‑phase
   optimisation.
4. Finish with deterministic pairwise swaps and a cheap simulated‑annealing
   clean‑up.
Optional LP refinement (using scipy.optimize.linprog) is applied when SciPy
is available – it solves a linear program that lowers the maximum convolution
while keeping the simplex constraints.

All variables stay non‑negative and sum to one, i.e. a probability simplex.
The returned list is a plain Python list; ``evaluate_sequence`` will
perform its final clipping.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolutions
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    m = b.shape[0]
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a simplex vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (all return a normalized probability vector)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w = np.maximum(w, 0.0)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    w = np.maximum(w, 0.0)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:

    w = np.sin(theta) ** p
    w = np.maximum(w, 0.0)

        w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.006) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Smoothed log‑sum‑exp surrogate
def _smooth_gradient(a: np.ndarray, lam: float) -> tuple[np.ndarray, float]:
    """Return (grad, max_conv) for surrogate φ_λ."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)               # length 2n‑1
    max_c = conv.max()
    z = lam * conv
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    s = exp_z / exp_z.sum()                    # softmax, length 2n‑1
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, s)        # length 3n‑2
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]   # slice of length n
    return grad, max_c


def _smooth_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    lam_start_factor: float = 0.15,
    lam_inc: float = 1.0015,
    beta: float = 0.9,
    eval_interval: int = 200,
    patience: int = 2500,
) -> tuple[np.ndarray, float]:
    """Annealed exponentiated‑gradient descent."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    lam = lam_start_factor * n
    rng = np.random.default_rng()

    while time.time() < deadline and t < 600_000:
        t += 1
        grad, max_conv = _smooth_gradient(a, lam)

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = lr0 / math.sqrt(t)                 # decreasing step
        a = a * np.exp(-eta * momentum)

        a = np.maximum(a, 1e-30)
        a /= a.sum()

        lam *= lam_inc

        if t % eval_interval == 0:
            cur_score = 2.0 * n * max_conv
            if cur_score + 1e-14 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

# Direct sub‑gradient descent on the exact max‑convolution
def _max_conv_subgradient(a: np.ndarray) -> tuple[np.ndarray, float]:
    """Sub‑gradient of max(conv(a,a)) and the max value."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_val = conv.max()
    ks = np.where(conv >= max_val - 1e-12)[0]   # tie‑breaking set
    grad = np.zeros_like(a)
    idx = np.arange(n)
    for k in ks:
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        grad[mask] += 2.0 * a[partner[mask]]
    grad /= max(1, len(ks))                    # average over ties
    return grad, max_val


def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    step0: float = 0.5,
    eval_interval: int = 100,
    patience: int = 1200,
) -> tuple[np.ndarray, float]:
    """Simple sub‑gradient method on the exact max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        t += 1
        grad, max_c = _max_conv_subgradient(a)
        eta = step0 / math.sqrt(t)
        a = a - eta * grad
        a = _project_onto_simplex(a)            # keep on simplex
        cur_score = 2.0 * n * max_c
        if cur_score + 1e-14 < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"[{time.time():.1f}] sub‑grad it={t:,} new best {best_score:.6f}")
        else:
            no_improve += 1
        if t % eval_interval == 0 and time.time() >= deadline:
            break
        if no_improve >= patience:
            # occasional jitter to escape plateaus
            jitter = rng.normal(scale=0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()
            no_improve = 0
    return best_seq, best_score

# Pairwise deterministic swaps (cheap local improvement)
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score

def _random_local_search(

    deadline: float,
    best_seq: np.ndarray,

    decay: float = 0.9995,

def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq

def _solve_lp_for_f(f: np.ndarray) -> tuple[np.ndarray, float]:
    """Solve   min t   s.t. sum(g)=1, g≥0, (f*g)_k ≤ t for all k."""
    n = f.shape[0]
    rows = 2 * n - 1

    # Build the convolution matrix C where C[k,j] = f_{k-j} if index valid.
    K = np.arange(rows)[:, None]
    J = np.arange(n)[None, :]
    idx = K - J
    mask = (idx >= 0) & (idx < n)
    C = np.zeros((rows, n), dtype=np.float64)
    C[mask] = f[idx[mask]]

    # Append the -t column
    A_ub = np.hstack([C, -np.ones((rows, 1), dtype=np.float64)])
    b_ub = np.zeros(rows, dtype=np.float64)

    # Equality: sum(g)=1
    A_eq = np.hstack([np.ones((1, n), dtype=np.float64), np.zeros((1, 1), dtype=np.float64)])
    b_eq = np.array([1.0], dtype=np.float64)

    # Objective: minimise t
    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        raise RuntimeError("LP failed: " + res.message)
    g = res.x[:n]
    t = res.x[-1]
    return g, t


def _lp_refine(a0: np.ndarray, deadline: float, max_iters: int = 3) -> tuple[np.ndarray, float]:
    """Iteratively improve a via the LP described above."""
    if not _SCIPY_AVAILABLE:
        return a0, _objective(a0)

    a = a0.copy()
    best_score = _objective(a)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        try:
            g, t = _solve_lp_for_f(a)
        except Exception as e:  # pragma: no cover
            print(f"[{time.time():.1f}] LP error: {e}")
            break
        # Rescale g to satisfy the simplex (it already does)
        new_score = _objective(g)
        if new_score + 1e-14 < best_score:
            a = g
            best_score = new_score
            print(f"[{time.time():.1f}] LP iteration {it} improved to {best_score:.6f}")
        else:
            break
    return a, best_score

            inv = _distribution_edge_inversepow(n, p)
            beta = _distribution_beta_sym(n, p)
            sine = _distribution_sine_weighted(n, p)

            for base in (edge, inv, beta, sine):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(f"[{time.time():.1f}] new best (param) n={n} p={p:.3g} lam={lam:.2f} {sc:.6f}")

        # ---- asymmetric beta ----
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = rng.uniform(0.001, 5.0)
            b = rng.uniform(0.001, 5.0)
            cand = _distribution_beta_asym(n, a, b)
            sc = _objective(cand)
            candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best asym‑beta n={n} a={a:.3g} b={b:.3g} {sc:.6f}")

        # ---- Dirichlet and sparse random ----
        for _ in range(4):
            if time.time() >= deadline:
                break
            cand = _distribution_dirichlet(n, rng)
            sc = _objective(cand)
            candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best Dirichlet n={n} {sc:.6f}")

        for _ in range(4):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.004)
            sc = _objective(cand)
            candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best sparse n={n} {sc:.6f}")

        # keep elite seeds (30 best)
        candidates.sort(key=lambda x: x[0])
        elite = [seq for _, seq in candidates[:30]]

        # ----------------------------------------------------------
        # 2. Refine elite seeds with LP + two‑phase descent
        per_seed_budget = (deadline - time.time()) * 0.55 / max(1, len(elite))
        for idx, seed_seq in enumerate(elite):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed_budget
            # LP refinement (few iterations)
            g_lp, sc_lp = _lp_refine(seed_seq, sub_dead, max_iters=2)
            # smooth descent
            g_smooth, sc_smooth = _smooth_descent(
                g_lp,
                sub_dead,
                best_seq=overall_best_seq if overall_best_seq is not None else g_lp,
                best_score=overall_best_score,
                lr0=0.9,
                lam_start_factor=0.13,
                lam_inc=1.002,
                beta=0.9,
                eval_interval=120,
                patience=1500,
            )
            # sub‑gradient descent
            g_sub, sc_sub = _subgradient_descent(
                g_smooth,
                sub_dead,
                best_seq=overall_best_seq if overall_best_seq is not None else g_smooth,
                best_score=overall_best_score,
                step0=0.6,
                eval_interval=80,
                patience=1200,
            )
            if sc_sub < overall_best_score:
                overall_best_score, overall_best_seq = sc_sub, g_sub.copy()
                print(f"[{time.time():.1f}] seed {idx} new global best {overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 3. If a previous hard‑coded construction exists, consider it
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 4. Upsample to a larger length and refine again
    if overall_best_seq is not None and time.time() < deadline - 120.0:
        large_n = min(12000, len(overall_best_seq) * 2)
        print(f"[{time.time():.1f}] up‑sampling to {large_n}")
        up_seq = _upsample(overall_best_seq, large_n)
        sub_dead = time.time() + (deadline - time.time()) * 0.65

        # smooth descent on up‑sampled sequence
        up_smooth, up_sc = _smooth_descent(
            up_seq,
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.6,
            lam_start_factor=0.15,
            lam_inc=1.0018,
            beta=0.92,
            eval_interval=100,
            patience=1800,
        )
        # sub‑gradient descent
        up_sub, up_sc2 = _subgradient_descent(
            up_smooth,
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            step0=0.5,
            eval_interval=80,
            patience=1500,
        )
        if up_sc2 < overall_best_score:
            overall_best_score, overall_best_seq = up_sc2, up_sub.copy()
            print(f"[{time.time():.1f}] up‑sampled descent new best {overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Deterministic pairwise refinement
    if overall_best_seq is not None and time.time() < deadline:
        print(f"[{time.time():.1f}] deterministic pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=400_000,
        )

    # --------------------------------------------------------------
    # 6. Final simulated‑annealing clean‑up
    if overall_best_seq is not None and time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.009,
            decay=0.9994,
        )

    # --------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
