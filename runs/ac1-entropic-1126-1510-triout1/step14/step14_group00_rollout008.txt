"""
propose_candidate
-----------------

A time‑bounded search for a non‑negative step function (probability vector)
that minimizes the evaluation function

    2·n·max(conv(a,a)) / (∑a)²

The algorithm consists of

1. **Rich seeding** – analytic families (uniform, arcsine, Hann, sine,
   β‑distribution, edge‑weighted, generalized‑normal, Dirichlet, sparse)
   are sampled for several lengths (512 … 16384). The 20 best seeds are kept.

2. **Soft‑max exponentiated‑gradient polishing** – each seed is refined by a
   mirror‑descent method that optimises a smooth approximation of the “max”
   (log‑sum‑exp) of the self‑convolution.  An EMA of the gradient (Adam‑style)
   together with an adaptive step‑size provides fast convergence.

3. **Progressive up‑sampling** – the current best vector is repeatedly
   up‑sampled (linear interpolation) while time permits (up to 65536 entries).
   After each up‑sample a medium‑length gradient run is applied.

4. **Pairwise mass‑transfer annealing** – a simple simulated‑annealing pass
   that moves a random fraction of mass between two bins finishes the optimisation.

All steps respect a 1000 s budget (2 s safety margin) and the returned list
contains non‑negative values (∑a = 1, i.e. a probability vector).  The objective
value is printed for monitoring.
"""

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used occasionally to avoid drift
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (real‑valued)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    # keep the meaningful part only
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective for a probability vector (∑a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families – all return a probability vector (Σ = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_sine(n: int) -> np.ndarray:
    """Half‑sine window (non‑negative, zero at the ends)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w[w < 0] = 0.0
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # avoid overflow/underflow for extreme α
    w = np.where(
        (x == 0) | (x == 1),
        0.0,
        np.exp((alpha - 1) * np.log(x) + (alpha - 1) * np.log1p(-x))
    )
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Random sparse vector with given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1‑λ) + uniform·λ."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) gradient of the max‑convolution term
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """Return soft‑max weights w_i ∝ exp(tau·conv_i)."""
    maxc = np.max(conv)
    exps = np.exp(tau * (conv - maxc))
    w = exps / exps.sum()
    return w


def _grad_softmax(a: np.ndarray, conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of the smooth max (soft‑max) of the convolution.
    Implements   g = 2·corr(w, a)   where   w = softmax(conv).
    """
    n = a.shape[0]
    w = _softmax_weights(conv, tau)

    # correlation w ⋆ a  = conv(w, rev(a)) shifted by (n‑1)
    a_rev = a[::-1]
    size = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    A = np.fft.rfft(a_rev, n=size)
    tmp = np.fft.irfft(W * A, n=size)
    start = n - 1
    grad = 2.0 * tmp[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated gradient) with soft‑max gradient
# ----------------------------------------------------------------------
def _softmax_exp_gradient(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.5,
    beta: float = 0.9,
    tau0: float = 0.2,
    tau_max: float = 30.0,
    tau_inc: float = 1.03,
    eval_every: int = 40,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the smoothed (soft‑max) objective.
    Uses an EMA of the gradient (mirrored Adam) and adapts the step size.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)          # EMA of gradients
    eta = eta0
    tau = tau0
    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        grad = _grad_softmax(a, conv, tau)

        # EMA update
        m = beta * m + (1.0 - beta) * grad

        # multiplicative (mirror) update
        a = a * np.exp(-eta * m)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # occasional evaluation
        if it % eval_every == 0:
            cur = float(2.0 * n * conv.max())
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                # reward improvement
                eta = min(eta0, eta * 1.02)
                last_improve = time.time()
                print(f"[{time.time():.1f}] GD it={it:,} new best {best_score:.6f}")
            else:
                # decay a little
                eta *= 0.995

        # increase smoothing (sharpen the soft‑max) gradually
        tau = min(tau * tau_inc, tau_max)

        # safety re‑projection
        if it % (eval_every * 10) == 0:
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score

# Pairwise mass‑transfer simulated annealing (same as previous version)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        src = rng.choice(np.where(a > 1e-14)[0])
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1
        delta = a[src] * rng.random() * 0.3
        if delta < 1e-16:
            continue
        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta
        cur = _objective(new_a)
        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.02
        accept = False
        if cur < best_score - 1e-15:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True
        if accept:
            a = new_a
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal {src}->{dst} Δ={delta:.1e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling whilst preserving total mass (=1)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new
