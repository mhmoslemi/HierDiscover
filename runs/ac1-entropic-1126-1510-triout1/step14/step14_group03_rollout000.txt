def _optimize_family(n, family_func, num_starts=5, max_iter=80):
    best_score = inf
    best_vec = None
    for _ in range(num_starts):
        # random initialization
        p0 = np.exp(np.random.uniform(np.log(0.01), np.log(10))) # random log-uniform
        lam0 = np.random.rand()
        x0 = np.array([p0, lam0])
        # define objective wrapper
        def fun(x):
            p, lam = x
            if p <= 0 or lam < 0 or lam > 1:
                return 1e9
            base = family_func(n, p)
            vec = _mix_with_uniform(base, lam)
            return _objective(vec)
        # minimize
        res = minimize(fun, x0, method='Nelder-Mead', options={'maxiter': max_iter, 'disp': False})
        # compute final vector
        p_opt, lam_opt = res.x
        if p_opt <= 0:
            p_opt = 1e-6
        lam_opt = np.clip(lam_opt, 0, 1)
        base = family_func(n, p_opt)
        vec = _mix_with_uniform(base, lam_opt)
        sc = _objective(vec)
        if sc < best_score:
            best_score, best_vec = sc, vec.copy()
    return best_score, best_vec

"""
Propose a step‑function (non‑negative list of heights) that minimizes the
evaluation function ``evaluate_sequence``.

Algorithm
---------
1. **Analytic seed pool** – For several lengths (3000‑8000) generate simple
   distributions: uniform, arcsine, edge‑weighted families and a
   generalized‑normal family.  Keep the best few seeds.

2. **Continuous parametric refinement** – For the two edge families
   (product‑weighted  f(i)∝(i·(n‑i))⁻ᵖ and sum‑weighted  f(i)∝i⁻ᵖ+(n‑i)⁻ᵖ)
   we treat the exponent p and a mixing coefficient λ with the uniform
   distribution as continuous parameters.  Using ``scipy.optimize.minimize``
   (Nelder‑Mead) we perform a small local search around the best grid points
   found in step 1.  The objective evaluated is the exact value used by the
   evaluator (2·n·max(conv)/sum²).  This fine‑tunes p and λ and typically
   yields a few‑per‑mille improvement.

3. **Mirror (exponentiated) gradient descent** – Starting from the best
   refined seed we run many iterations of multiplicative mirror descent.
   The sub‑gradient of the current maximal convolution entry is used;
   the step size shrinks as η₀/√t and the vector is renormalised after each
   update.

4. **Pairwise mass‑transfer simulated annealing** – A final short annealing
   phase moves a random fraction of mass between two bins and accepts the
   move if it improves the objective or with a temperature‑dependent
   probability.  This polishes the solution.

All work is bounded by the supplied budget (default 1000 s).  The function
returns a plain ``list[float]``; the evaluator will clip negative values
to 0 and enforce the sum‑normalisation internally, so we keep the vector
already normalised.

The code is fully self‑contained and uses only ``numpy`` and (optionally)
``scipy``.  If SciPy is unavailable the algorithm falls back to the original
grid‑search + mirror‑descent pipeline.
"""

import time
import math
import numpy as np

except Exception:  # pragma: no cover

# ----------------------------------------------------------------------
#  FFT‑based linear (full) convolution – real inputs only
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a*a using FFT (real input)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
#  Exact objective used by the evaluator (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a))/sum(a)² – a must be non‑negative."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
#  Analytic families – all return a *probability* vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i·(n‑i))^{-p}   with i = distance to the left edge (0‑based)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_sum_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ i^{-p} + (n‑i)^{-p}   (sum of two one‑sided tails)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (-p)) + (j ** (-p))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density proportional to exp(-|x‑½|^{q}) on the discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1‑lam) + uniform·lam."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
#  Exponentiated‑gradient (mirror) descent on the exact objective
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """Gradient of the currently maximal convolution entry."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))

    grad = np.zeros_like(a)
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


def _exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """Mirror descent with multiplicative updates."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Pairwise‑mass‑transfer simulated annealing (final polish)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins; accept improvements."""
    n = a.shape[0]
    start_time = time.time()
    total_time = max(deadline - start_time, 0.0)
    while time.time() < deadline:
        src_idx = np.where(a > 1e-12)[0]
        if src_idx.size == 0:
            break
        i = rng.choice(src_idx)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move up to 40 % of the source mass
        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Continuous refinement of the edge‑weighted families
# ----------------------------------------------------------------------
def _refine_edge_family(
    n: int,
    family_func,
    timer_deadline: float,
    starts: int = 4,
    maxiter: int = 80,
) -> tuple[float, np.ndarray]:
    """
    Optimise (p, λ) for a given edge family using Nelder‑Mead.
    Returns the best score and the corresponding normalised vector.
    """
    best_score = float("inf")
    best_vec = None
    rng = np.random.default_rng()
    for _ in range(starts):
        # Log‑uniform start for p, uniform start for λ
        p0 = np.exp(rng.uniform(np.log(0.01), np.log(30.0)))
        lam0 = rng.random()
        x0 = np.array([p0, lam0])

        def obj(x):
            p, lam = x
            # guard against invalid parameters
            if p <= 0.0 or lam < 0.0 or lam > 1.0:
                return 1e9
            base = family_func(n, p)
            vec = _mix_with_uniform(base, lam)
            return _objective(vec)

        if not _SCIPY:  # pragma: no cover
            # Without SciPy we cannot run the optimiser – fall back to the grid value.
            break

        res = minimize(
            obj,
            x0,
            method="Nelder-Mead",
            options={"maxiter": maxiter, "disp": False},
        )
        p_opt, lam_opt = res.x
        p_opt = max(p_opt, 1e-12)
        lam_opt = np.clip(lam_opt, 0.0, 1.0)

        base = family_func(n, p_opt)
        vec = _mix_with_uniform(base, lam_opt)
        sc = _objective(vec)

        if sc < best_score:
            best_score = sc
            best_vec = vec.copy()
        # time guard
        if time.time() >= timer_deadline:
            break
    return best_score, best_vec

            np.logspace(0.5, 1.5, 10, base=10),     # 3.16 … 31.6
        ))
        lam_grid = np.linspace(0.0, 1.0, 11)        # 0,0.1,…,1

        # product edge‑weighted
        for p in p_grid:
            if time.time() >= deadline:
                break
            base = _distribution_edge_weighted(n, p)
            sc = _objective(base)
            top.append((sc, base.copy()))
            if sc < best_score:
                best_score, best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new best prod p={p:.3g} score={sc:.6f}")

            for lam in lam_grid:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                top.append((sc, cand.copy()))
                if sc < best_score:
                    best_score, best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new best prod p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # sum edge‑weighted (new family)
        for p in p_grid:
            if time.time() >= deadline:
                break
            base = _distribution_edge_sum_weighted(n, p)
            sc = _objective(base)
            top.append((sc, base.copy()))
            if sc < best_score:
                best_score, best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new best sum p={p:.3g} score={sc:.6f}")

            for lam in lam_grid:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                top.append((sc, cand.copy()))
                if sc < best_score:
                    best_score, best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new best sum p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # ------------------------------------------------------------------
        # 3.   Continuous (Nelder‑Mead) refinement of the two edge families
        # ------------------------------------------------------------------
        # only launch the refined search if SciPy is available and we still have time
        if _SCIPY and time.time() < deadline:
            # product edge‑weighted refinement
            sc, vec = _refine_edge_family(
                n,
                _distribution_edge_weighted,
                deadline,
                starts=4,
                maxiter=80,
            )
            if vec is not None:
                top.append((sc, vec.copy()))
                if sc < best_score:
                    best_score, best_seq = sc, vec.copy()
                    print(f"[{time.time():.1f}] refined prod best score={sc:.6f}")

            # sum edge‑weighted refinement
            sc, vec = _refine_edge_family(
                n,
                _distribution_edge_sum_weighted,
                deadline,
                starts=4,
                maxiter=80,
            )
            if vec is not None:
                top.append((sc, vec.copy()))
                if sc < best_score:
                    best_score, best_seq = sc, vec.copy()
                    print(f"[{time.time():.1f}] refined sum best score={sc:.6f}")

        # ------------------------------------------------------------------
        # 4.   Random sparse Dirichlet seeds (few non‑zero entries)
        # ------------------------------------------------------------------
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = np.zeros(n, dtype=np.float64)
            k = max(2, int(n * 0.01))
            idx = rng.choice(n, size=k, replace=False)
            cand[idx] = rng.random(k)
            cand /= cand.sum()
            sc = _objective(cand)
            top.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best sparse score={sc:.6f}")

        # keep only the top 20 seeds after each length
        top.sort(key=lambda x: x[0])
        top = top[:20]

    # ------------------------------------------------------------------
    # 5.   Include any previously submitted construction (if present)
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            top.append((sc, prev.copy()))
            if sc < best_score:
                best_score, best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 6.   Guard against an empty result (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _distribution_uniform(5000)
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.   Mirror‑descent refinement (long exponential‑gradient run)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = deadline
        print(f"[{time.time():.1f}] EG refinement (budget {sub_deadline - time.time():.1f}s)")
        best_seq, best_score = _exp_gradient_descent(
            best_seq.copy(),
            sub_deadline,
            best_seq,
            best_score,
            eta0=0.9,
            eval_every=60,
        )
        print(f"[{time.time():.1f}] after EG best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8.   Final simulated‑annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing")
        best_seq, best_score = _pairwise_annealing(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            rng,
        )
        print(f"[{time.time():.1f}] after annealing best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9.   Report final result and return
    # ------------------------------------------------------------------
    print(
        f"Final best score (objective = 2·n·max(conv)/sum²): {best_score:.6f} "
        f"(length={len(best_seq)})"
    )
    # Ensure the returned list follows the evaluator's expectations:
    # - non‑negative entries
    # - reasonable magnitude (the evaluator caps at 1000, we are far below)
    # No explicit scaling is required because the objective is scale‑invariant.
    return best_seq.tolist()
