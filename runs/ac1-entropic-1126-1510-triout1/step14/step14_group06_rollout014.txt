"""
Simple power‑law decreasing step functions.

The objective of ``evaluate_sequence`` is
    2·n·max(conv(seq,seq)) / (sum(seq)^2).

For a distribution a_i ∝ (n‑i)^p (p ≥ 0 integer) and sum(a)=1 the two
potentially maximal convolution entries are

   * at shift 0 :  a_0² = ((p+1)/n)²                →  contribution 2·(p+1)² / n
   * at shift n‑1 :  Σ a_i a_{n‑1‑i}
                     = ((p+1)²·B(p+1,p+1))/n   →  contribution
                       2·(p+1)²·B(p+1,p+1)
                       = 2·((p+1)!)² / (2p+1)!.

Hence the exact objective is
    max( 2·(p+1)² / n ,  2·((p+1)!)² / (2p+1)! ).

For a given length n the optimal exponent p is roughly log₄(n) and the
resulting bound is on the order of (log₂ n)² / (2·n) – far below the
previous 1.5.  The search therefore enumerates a few lengths and a small
window of exponents around log₄(n) and keeps the best sequence.
A very short exponentiated‑gradient polishing step is performed if time
permits.

The whole routine runs in a few milliseconds and always returns a
non‑negative list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Analytic objective for the (n‑i)^p family (exact, no convolution)
# ----------------------------------------------------------------------
def _objective_powerlaw(n: int, p: int) -> float:
    """
    Exact objective value for the distribution a_i ∝ (n-i)^p,
    normalised so that sum(a)=1.
    """
    if p == 0:
        central = 2.0                     # uniform case
    else:
        # 2·((p+1)!)²/(2p+1)!  =  2·(p+1)² / ((2p+1)·C(2p,p))
        central = 2.0 * (p + 1) ** 2 / ((2 * p + 1) * math.comb(2 * p, p))

    edge = 2.0 * (p + 1) ** 2 / n          # from shift 0 (a₀²)
    return max(central, edge)


# ----------------------------------------------------------------------
# Build a decreasing power‑law sequence and normalise it
# ----------------------------------------------------------------------
def _distribution_power_decay(n: int, p: int) -> np.ndarray:
    """Return a probability vector a_i ∝ (n-i)^p (i = 0 … n‑1)."""
    w = np.arange(n, 0, -1, dtype=np.float64) ** p
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Small exponentiated‑gradient descent (exact objective via evaluate_sequence)
# ----------------------------------------------------------------------
def _exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.7,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative mirror descent on the true objective.
    Very few iterations – only a quick polish.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        # gradient of the currently maximal convolution entry
        conv = np.convolve(a, a)
        k = int(np.argmax(conv))
        grad = np.zeros_like(a)
        if k < n:
            grad[: k + 1] = 2.0 * a[k::-1]
        else:
            start = k - (n - 1)
            grad[start:] = 2.0 * a[start:][::-1]

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = evaluate_sequence(a.tolist())
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a monotone decreasing power‑law step function that minimises
    ``evaluate_sequence``.  The search is analytical (no heavy
    convolutions) and finishes well within the time budget.
    """
    rng = np.random.default_rng(seed)   # not strictly needed, kept for compatibility
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")
    best_n = None
    best_p = None

    # ------------------------------------------------------------------
    # 1.  Scan a few lengths and a window of exponents around log4(n)
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096, 8192, 16384]
    for n in lengths:
        if time.time() >= deadline:
            break

        # exponent that balances the two terms: p ≈ log₄(n)
        ideal = max(0, int(round(math.log(n, 4))))
        # search a small neighbourhood of that exponent
        for p in range(max(0, ideal - 4), ideal + 5):
            if time.time() >= deadline:

            # quick analytic bound – discard clearly worse candidates
            approx = _objective_powerlaw(n, p)
            if approx >= best_score:

            a = _distribution_power_decay(n, p)
            # exact evaluation (cheap – called only a handful of times)

            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                best_n, best_p = n, p

    # ------------------------------------------------------------------
    # 2.  Fallback – uniform distribution (should never trigger)

    if best_seq is None:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:

        print(f"[{time.time():.1f}] post‑EG best score={best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"[{time.time():.1f}] Final best score {best_score:.6f} "
        f"(length={best_n}, p={best_p})"
    )
    return best_seq.tolist()
