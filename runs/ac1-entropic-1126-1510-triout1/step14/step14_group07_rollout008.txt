"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.

Algorithm
---------

1️⃣ **Broad seeding** –  generate a variety of analytically‑motivated
   distributions (uniform, sine‑weighted, centre‑exponential,
   left/right‑exponential, Gaussian, Kaiser, Dirichlet, sparse random,
   power‑law).  For several lengths (≈2k–12k) we keep the 12 best
   candidates (lowest exact objective).

2️⃣ **Fast Adam refinement** –  each retained seed is short‑run
   optimised with an Adam mirror‑descent that uses the exact sub‑gradient
   of the max‑convolution (active‑set variant).  The best of these
   refined seeds becomes the global start point.

3️⃣ **Smooth‑β Adam descent** –  a long optimisation on the best seed.
   We minimise the *soft‑max* of the convolution
   `f_β(a)= (1/β)·log Σ_k exp(β·(a*a)_k)`; its gradient is a
   correlation that can be computed entirely by FFT.  β is increased
   linearly from a small value (≈0.5) to a large value (≈80) – a
   continuation method that reliably drives the exact max down.

4️⃣ **Local polishing** –  a greedy pairwise mass‑swap followed by a cheap
   simulated‑annealing style search (identical to the reference code).

All steps respect the global 1000 s budget; the returned list can be
fed directly to ``evaluate_sequence`` – the evaluator will re‑normalise
it anyway.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft
from scipy.special import i0  # Kaiser window

# Simple utilities ------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (a is assumed to sum to 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector – sum = 1) -----
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    w /= w.sum()
    return w


def _distribution_exponential_center(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(i - half))
    w /= w.sum()
    return w


def _distribution_exponential_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-rate * i)
    w /= w.sum()
    return w


def _distribution_exponential_right(n: int, rate: float) -> np.ndarray:
    return _distribution_exponential_left(n, rate)[::-1]


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - half) / sigma) ** 2)
    w /= w.sum()
    return w


def _distribution_kaiser(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    x = (i - half) / half
    # clip for numerical safety
    x = np.clip(x, -1.0, 1.0)
    w = i0(beta * np.sqrt(1.0 - x * x))
    w /= w.sum()
    return w


def _distribution_powerlaw(n: int, exponent: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    w = 1.0 / (i ** exponent)
    w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator, conc: float) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a.astype(np.float64)


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Adam with exact active‑set sub‑gradient (short runs) ---------------
def _adam_refine(
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 20_000,
    lr0: float = 0.8,
    active_frac: float = 0.025,
    max_active: int = 30,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_interval: int = 250,
    best_seq: np.ndarray | None = None,
    best_score: float = float("inf"),
) -> tuple[np.ndarray, float]:
    """Adam mirror descent using the exact sub‑gradient of max(conv)."""
    n = init_a.shape[0]
    x = np.log(np.maximum(init_a, 1e-12))
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    ind_grid = np.arange(n, dtype=np.int32)
    t = 0
    if best_seq is None:
        best_seq = init_a.copy()
        best_score = _objective(init_a)

    while time.time() < deadline and t < max_iters:
        t += 1
        a = _softmax(x)
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            top = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top]

        # sub‑gradient of the max (active set)
        grad_a = np.zeros_like(a)
        for idx_k in act:
            offset = idx_k - ind_grid
            mask = (offset >= 0) & (offset < n)
            grad_a[mask] += 2.0 * a[offset[mask]]

        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update with decaying step‑size
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = lr0 / math.sqrt(t)
        x -= eta * m_hat / (np.sqrt(v_hat) + eps)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] adam it={t:,} new best {best_score:.6f}")

    best_seq = _project_onto_simplex(best_seq, s=1.0)
    return best_seq, best_score

# Smoothed‑β gradient (used in the long descent) --------------------
def _smooth_gradient_from_conv(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smoothed max (soft‑max) objective:
    grad_a = 2 * corr(w, a) where w_i ∝ exp(beta·(conv_i - max(conv))).
    """
    max_c = conv.max()
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    n = a.shape[0]
    L = conv.shape[0]            # = 2n‑1
    size = 1 << ((L + n - 1 - 1).bit_length())   # power of two ≥ L+n‑1
    W = rfft(w, n=size)
    A = rfft(a, n=size)
    # cross‑correlation via FFT: corr = irfft(W * conj(A))
    corr_full = irfft(W * np.conj(A), n=size)
    # valid part (length n) starts at index n‑1
    start = n - 1
    grad = 2.0 * np.real(corr_full[start:start + n])
    return grad


def _adam_smooth_refine(
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    lr0: float = 0.9,
    beta0: float = 0.5,
    beta_max: float = 80.0,
    beta_schedule: str = "linear",
    beta_inc_iters: int = 1000,
    beta_factor: float = 1.05,
    beta_clip: float = 0.0,
    beta_end: float | None = None,
    beta_lin_rate: float | None = None,
    beta_exp_rate: float | None = None,
    beta_lin_end: float | None = None,
    beta_exp_end: float | None = None,
    beta_tolerance: float = 1e-12,
    beta_break: bool = False,
    beta_debug: bool = False,
    beta_debug_interval: int = 10_000,
    beta_debug_print: bool = False,
    beta_debug_label: str = "",
    beta_debug_fun: callable | None = None,
    beta_debug_opts: dict | None = None,
    beta_debug_mode: str = "print",
    beta_debug_fmt: str = "{:.4f}",
    # Adam parameters
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_interval: int = 250,
    best_seq: np.ndarray | None = None,
    best_score: float = float("inf"),
) -> tuple[np.ndarray, float]:
    """
    Long Adam optimisation using the smooth (soft‑max) gradient.
    β grows from beta0 to beta_max (linear schedule by default).
    """
    n = init_a.shape[0]
    x = np.log(np.maximum(init_a, 1e-12))
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    t = 0

    if best_seq is None:
        best_seq = init_a.copy()
        best_score = _objective(init_a)

    while time.time() < deadline and t < max_iters:
        t += 1
        a = _softmax(x)
        conv = _fft_self_convolve(a)
        max_c = conv.max()

        # β schedule – linear increase
        beta = beta0 + (beta_max - beta0) * min(1.0, t / max_iters)

        grad_a = _smooth_gradient_from_conv(a, conv, beta)

        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = lr0 / math.sqrt(t)
        x -= eta * m_hat / (np.sqrt(v_hat) + eps)

        if t % eval_interval == 0:
            cur = 2.0 * n * max_c   # exact objective
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth‑adam it={t:,} β={beta:.2f} new best {best_score:.6f}")

    best_seq = _project_onto_simplex(best_seq, s=1.0)
    return best_seq, best_score

# Greedy pairwise refinement (same as in reference) -----------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score

# Very cheap simulated‑annealing style local search -----------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Entry point – returns a plain Python list ready for ``evaluate_sequence``."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # tiny safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------- 1️⃣  seed generation -------------------------
    lengths = [2000, 3000, 4000, 6000, 8000, 12000]
    p_grid = np.logspace(-1, 1.0, 8)          # for sine‑weighted family
    lam_grid = np.linspace(0.0, 0.5, 6)       # mixing with uniform
    rate_grid = np.linspace(0.001, 0.02, 6)   # exponential decay rates
    conc_grid = [0.05, 0.2, 1.0, 5.0]        # Dirichlet concentrations
    sigma_factor_grid = np.linspace(0.05, 0.30, 6)   # Gaussian width factor
    kaiser_beta_grid = [1.0, 3.0, 5.0, 7.0, 10.0]   # Kaiser β values
    powerlaw_exp_grid = [0.5, 1.0, 1.5, 2.0]       # power‑law exponents

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic / parametric families
        families = []

        families.append(_distribution_uniform(n))
        families.append(_distribution_exponential_center(n, rate=0.005))

        # sine‑weighted + uniform mixing
        for p in p_grid:
            sine = _distribution_sine_weighted(n, p)
            for lam in lam_grid:
                families.append(_mix_with_uniform(sine, lam))

        # left / right exponentials (different rates)
        for rate in rate_grid:
            families.append(_distribution_exponential_left(n, rate))
            families.append(_distribution_exponential_right(n, rate))

        # Gaussian centre (different sigmas)
        for sf in sigma_factor_grid:
            sigma = sf * n
            families.append(_distribution_gaussian(n, sigma))

        # Kaiser windows
        for beta in kaiser_beta_grid:
            families.append(_distribution_kaiser(n, beta))

        # Power‑law decreasing
        for exp in powerlaw_exp_grid:
            families.append(_distribution_powerlaw(n, exp))

        # Dirichlet samples
        for conc in conc_grid:
            for _ in range(2):
                families.append(_distribution_dirichlet(n, rng, conc))

        # Random sparse
        for _ in range(3):
            families.append(_distribution_random_sparse(n, rng, density=0.008))

        # evaluate all candidates for this length
        for cand in families:
            if time.time() >= deadline:
                break
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # keep only the very best seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:12]

    # ------------------------------------------------------------------
    # possibly a previously found construction is available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2️⃣  Adam refinement – short runs for each retained seed
    refined: list[tuple[float, np.ndarray]] = []
    for sc, cand in top_candidates:
        if time.time() >= deadline:
            break
        # allocate a modest slice of time per short run
        remaining = deadline - time.time()
        slice_deadline = time.time() + min(12.0, remaining * 0.05)
        best_seq, best_sc = _adam_refine(
            cand.copy(),
            slice_deadline,
            max_iters=8_000,
            lr0=0.7,
            active_frac=0.03,
            max_active=20,
            beta1=0.9,
            beta2=0.999,
            eval_interval=200,
            best_seq=cand,
            best_score=sc,
        )
        refined.append((best_sc, best_seq))

    if refined:
        overall_best_score, overall_best_seq = min(refined, key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 3️⃣  Long smooth‑β Adam descent from the overall best seed
    if time.time() < deadline:
        remaining = deadline - time.time()
        # reserve a little time for final polishing
        long_deadline = time.time() + max(0.0, remaining - 30.0)
        print(f"[{time.time():.1f}] start long smooth‑Adam (≈{remaining-30:.1f}s left)")
        overall_best_seq, overall_best_score = _adam_smooth_refine(
            overall_best_seq.copy(),
            long_deadline,
            max_iters=250_000,
            lr0=0.9,
            beta0=0.5,
            beta_max=80.0,
            eval_interval=250,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] after smooth‑Adam best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=250_000,
        )

    # ------------------------------------------------------------------
    # 5️⃣  Final simulated‑annealing polishing
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.01,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(f"Final best score (exact) ≈ {overall_best_score:.6f} (length={len(overall_best_seq)})")
    #  evaluate_sequence will normalise, but we keep the vector a proper probability.
    return overall_best_seq.tolist()
