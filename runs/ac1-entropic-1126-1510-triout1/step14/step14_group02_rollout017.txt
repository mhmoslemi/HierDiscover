def _adam_smooth_minimize(a, deadline, best_seq, best_score, start_time):
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_local_seq = a.copy()
    best_local_score = _objective(a)
    while time.time() < deadline:
        t += 1
        # compute conv
        A = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(A * A, n=size)[:2 * n - 1]
        max_c = conv.max()
        # compute temperature tau based on remaining time
        remaining = deadline - time.time()
        total = deadline - start_time
        tau = max(1e-6, 0.05 * (remaining / total))
        # softmax weights
        w = np.exp((conv - max_c) / tau)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient via FFT
        a_rev = a[::-1]
        # FFT of a_rev
        B = np.fft.rfft(a_rev, n=size)
        W = np.fft.rfft(w, n=size)
        grad_full = np.fft.irfft(B * W, n=size)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        beta1 = 0.9
        beta2 = 0.999

"""
Soft‑max Adam descent for the convolution max‑objective.

The goal is to minimise
    2·n·max(conv(a,a))          (a is a non‑negative vector, sum(a)=1)

because the original `evaluate_sequence` is scale‑invariant.
The algorithm proceeds as follows:

1. Initialise a probability vector (often the previous best solution,
   up‑sampled to a larger length) and normalise it.

2. Repeatedly
   * compute the full self‑convolution `c = a*a`;
   * replace the non‑smooth `max(c)` by a smooth approximation
     `softmax_τ(c)` with temperature τ that is annealed to 0;
   * compute the gradient of the smooth objective analytically:
        ∇ = 2·∑_k w_k·∇conv_k,  where w_k = softmax weights.
     The gradient can be obtained by a single FFT‑based convolution.
   * update `a` with Adam (adaptive moments) and project back onto the
     probability simplex.

   The temperature τ and learning‑rate are decreased linearly with the
   remaining time, allowing coarse moves early and fine tuning later.

3. If time permits the current best vector is up‑sampled (linear
   interpolation) to a longer length and the descent is repeated.

4. Finally a cheap pairwise‑mass‑transfer refinement and a short simulated
   annealing phase clean up any remaining spikes.

All steps respect the global 1000 s budget.  The returned list contains
non‑negative heights and is directly usable by ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    need = x.shape[0] + y.shape[0] - 1
    size = 1 << ((need - 1).bit_length())   # next power of two
    X = rfft(x, n=size)
    Y = rfft(y, n=size)
    conv = irfft(X * Y, n=size)
    return conv[:need]


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a via FFT (length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) for a that sums to 1."""
    n = a.shape[0]
    c = _fft_self_convolve(a)
    return 2.0 * n * c.max()


# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet vector with a uniformly drawn concentration."""
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
def _adam_smooth_minimize(
    a0: np.ndarray,
    deadline: float,
    global_start: float,
    best_global_seq: np.ndarray,
    best_global_score: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth approximation of `max(conv)`.
    The temperature τ and learning‑rate decay linearly with the remaining
    time.  Returns the best local vector found.
    """
    a = a0.copy()
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())       # FFT size for length 2·n‑1
    m = np.zeros_like(a)                         # first moment
    v = np.zeros_like(a)                         # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    best_seq = a.copy()
    best_score = _objective(a)

    while time.time() < deadline:
        t += 1
        # ----- self‑convolution (objective) -----
        A = rfft(a, n=size)
        conv = irfft(A * A, n=size)[: 2 * n - 1]
        max_c = conv.max()

        # ----- temperature (linear annealing) -----
        remaining = deadline - time.time()
        total = deadline - global_start
        tau = max(1e-6, 0.05 * (remaining / total))

        # ----- soft‑max weights -----
        w = np.exp((conv - max_c) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smooth max -----
        # grad = 2 * sum_k w_k * ∂conv_k/∂a = 2 * (a ⋆ w_rev) with reversal of w
        w_rev = w[::-1]
        grad_full = _fft_convolve(a, w_rev)               # length = 3·n‑2
        # slice that corresponds to indices 2n‑2 … n‑1 (reversed order)
        slice_part = grad_full[n - 1 : 2 * n - 1]          # length n
        grad = 2.0 * slice_part[::-1]                     # reverse to match formula

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate also anneals linearly
        lr = max(0.005, 0.12 * (remaining / total))

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto the probability simplex -----
        a = _project_onto_simplex(a)

        # ----- occasional evaluation -----
        if t % 250 == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score = cur
                best_seq = a.copy()
            if cur + 1e-14 < best_global_score:
                best_global_score = cur
                best_global_seq = a.copy()
                print(
                    f"[{time.time():.1f}] iter {t:,} new global best {best_global_score:.6f}"
                )

    return best_seq, best_score

def _pairwise_weighted_refine(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer guided by the location of the maximal convolution."""
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0

    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        conv = _fft_self_convolve(a)
        k_max = int(conv.argmax())

        # Impact vector L_i = a_{k_max-i} where defined
        L = np.zeros(n, dtype=np.float64)
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        idx = np.arange(low, high + 1)
        L[idx] = a[k_max - idx]

        if L.sum() > 0.0:
            probs_i = L / L.sum()
            i = rng.choice(n, p=probs_i)
        else:
            positives = np.where(a > 1e-12)[0]
            if positives.size == 0:
                break
            i = rng.choice(positives)

        Lmax = L.max()
        probs_j = (Lmax - L) + 1e-12
        probs_j /= probs_j.sum()
        j = rng.choice(n, p=probs_j)
        if j == i:
            j = (j + 1) % n

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] weighted swap {i}->{j} δ={delta:.3e} new best {best_score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Lightweight Metropolis‑type simulated annealing."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp

    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}"
            )
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay

    return best_seq, best_score
