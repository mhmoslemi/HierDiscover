"""
search for a non‑negative step function (list of heights) that minimises
evaluate_sequence.  The algorithm builds a diverse pool of analytic seeds,
keeps the best few, and then runs a refined exponentiated‑gradient descent
that uses a *sub‑gradient of the active convolution entries* (averaged over
all indices that are within a tiny tolerance of the maximum).  Finally a
pairwise‑mass‑transfer simulated‑annealing polish is applied.

The code respects the 1000 s wall‑time limit, works on a single CPU, and
only uses the public evaluate_sequence (via an internal replica of its
objective) and NumPy.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  FFT‑based self‑convolution (full linear convolution)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
#  Exact objective used by the evaluator (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Returns 2·n·max(conv(a,a))  (the denominator sum(a)^2 is 1 because ``a``
    is normalised to a probability vector before calling this routine).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
#  Analytic families – all return a *probability* vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i·(n‑i))^{-p}   with i = distance to the left edge."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_sum_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ i^{-p} + (n‑i)^{-p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (-p)) + (j ** (-p))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp( -|x‑½|^{q} ) on the discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1‑lam) + uniform·lam."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
#  Sub‑gradient of the *active* convolution entry (average over all
#  indices that are virtually maximal).
# ----------------------------------------------------------------------
def _subgrad_active(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Return (gradient, max_conv) where the gradient is the average of the
    gradients of all convolution entries whose value is within a tiny
    tolerance of the maximum.  This is a valid sub‑gradient of the
    L∞‑objective.
    """
    conv = _fft_self_convolve(a)
    max_val = conv.max()
    # tolerance: a relative epsilon plus a tiny absolute term
    tol = max_val * 1e-12 + 1e-15
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    n = a.shape[0]
    for k in active:
        if k < n:
            # overlap length = k+1  (i runs 0..k)
            grad[: k + 1] += 2.0 * a[k::-1]
        else:
            # k >= n : overlap length = 2·n‑1‑k  (i runs k‑(n‑1)..n‑1)
            start = k - (n - 1)
            grad[start:] += 2.0 * a[start:][::-1]
    if len(active) > 0:
        grad /= len(active)
    return grad, max_val


# ----------------------------------------------------------------------
#  Exponentiated‑gradient descent with the multi‑active sub‑gradient.
# ----------------------------------------------------------------------
def _exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    eval_every: int = 120,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (entropy Bregman) on the exact objective.
    Uses the averaged sub‑gradient of all near‑max convolution entries.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, cur_max = _subgrad_active(a)
        eta = eta0 / math.sqrt(t)

        # exponentiated (multiplicative) update
        a = a * np.exp(-eta * grad)

        # safeguard against underflow
        a = np.maximum(a, 1e-18)

        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Pairwise‑mass‑transfer simulated annealing (final polish)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from one bin to another.
    Accepts every improving move and with a temperature‑scaled probability
    otherwise (linear cooling).
    """
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    while time.time() < deadline:
        # pick a source with non‑negligible mass
        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        # pick a distinct target
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move a random fraction (up to 40 %) of the source mass
        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)

        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.02

        if cur < best_score - 1e-14:
            accept = True
        else:
            # Metropolis acceptance
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Main entry point required by the judge
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Returns a plain list of non‑negative heights.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a small safety margin
    deadline = start + budget_s - 1.0

    best_seq: np.ndarray | None = None
    best_score = float("inf")
    top: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Generate a diverse pool of seeds
    # ------------------------------------------------------------------
    lengths = [4000, 8000, 12000, 20000]          # moderate to large
    p_grid = np.logspace(-2.0, 1.5, 36)          # 0.01 … 31.6
    lam_grid = np.linspace(0.0, 1.0, 21)        # 0,0.05,…,1
    q_grid = np.linspace(0.5, 3.0, 13)          # shape of generalized normal

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            top.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best ({n}) score={sc:.6f}")

        # product edge‑weighted + uniform mixtures
        for p in p_grid:
            if time.time() >= deadline:
                break
            base = _distribution_edge_weighted(n, p)
            sc = _objective(base)
            top.append((sc, base.copy()))
            if sc < best_score:
                best_score, best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new best prod p={p:.3g} score={sc:.6f}")
            for lam in lam_grid:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                top.append((sc, cand.copy()))
                if sc < best_score:
                    best_score, best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new best prod p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # sum edge‑weighted family (new)
        for p in p_grid:
            if time.time() >= deadline:
                break
            base = _distribution_edge_sum_weighted(n, p)
            sc = _objective(base)
            top.append((sc, base.copy()))
            if sc < best_score:
                best_score, best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new best sum p={p:.3g} score={sc:.6f}")
            for lam in lam_grid:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                top.append((sc, cand.copy()))
                if sc < best_score:
                    best_score, best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new best sum p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # generalized‑normal family
        for q in q_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_generalized_normal(n, q)
            sc = _objective(cand)
            top.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best gn q={q:.2f} score={sc:.6f}")

        # random sparse Dirichlet-like seeds
        for _ in range(8):
            if time.time() >= deadline:
                break
            cand = np.zeros(n, dtype=np.float64)
            k = max(2, int(n * 0.02))
            idx = rng.choice(n, size=k, replace=False)
            cand[idx] = rng.random(k)
            cand /= cand.sum()
            sc = _objective(cand)
            top.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best sparse score={sc:.6f}")

        # keep only the very best seeds (size 20)
        top.sort(key=lambda x: x[0])
        top = top[:20]

    # ------------------------------------------------------------------
    # 2.  Include the previously submitted construction if it exists
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)          # safety
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < best_score:
                best_score, best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Guard against an empty result
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _distribution_uniform(5000)
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Refine the current best seed with exponentiated gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] EG refinement (remaining {deadline - time.time():.1f}s)")
        best_seq, best_score = _exp_gradient_descent(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            eta0=0.9,
            eval_every=200,
        )
        print(f"[{time.time():.1f}] after EG best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing")
        best_seq, best_score = _pairwise_annealing(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            rng,
        )
        print(f"[{time.time():.1f}] after annealing best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Return the result
    # ------------------------------------------------------------------
    print(
        f"Final best score (objective = 2·n·max(conv)/sum²): {best_score:.6f} "
        f"(length={len(best_seq)})"
    )
    # Convert to plain Python list of floats – the evaluator will
    # perform its own clamping and normalisation.
    return best_seq.tolist()
