def _guided_mass_transfer(a0, time_limit, best_seq, best_score, transfer_frac=0.02, max_trials=None):
    a = a0.copy()
    n = a.shape[0]
    # we can compute gradient each iteration.
    trials = 0
    while time.time()<time_limit:
        if max_trials is not None and trials>= max_trials:
            break
        trials+=1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        if max_val * 0.0: pass
        max_idxs = np.where(conv == max_val)[0]
        # compute subgradient (averaged)
        w = np.zeros_like(conv)
        if len(max_idxs)>0:
            w[max_idxs] = 1.0/len(max_idxs)
        grad = 2.0 * _fft_convolve(w, a[::-1])
        # choose src with largest grad, dest with smallest grad
        src = np.argmax(grad)
        dst = np.argmin(grad)
        if src == dst:
            # random fallback
            src = np.random.randint(n)
            dst = np.random.randint(n)
        delta = a[src] * transfer_frac
        if delta < 1e-15:
            continue
        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-12:
            a = new_a
            best_score, best_seq = cur, a.copy()
            print(f"[{time.time():.1f}] guided transfer src={src}->dst={dst} Δ={delta:.2g} new best {best_score:.6f}")
        # else maybe accept with some probability? but skip for simplicity.
    return best_seq, best_score

"""
Hybrid optimiser for the non‑negative step‑function problem.

The algorithm consists of four phases:

1️⃣  **Exploratory families** – generate many analytic and random
    candidate vectors of various lengths (uniform, arcsine, Gaussian,
    beta, edge‑weighted, etc.) and keep the best few (according to the
    exact objective 2·n·max(conv)/∑a²).

2️⃣  **Smoothed‑max mirror descent** – replace the non‑smooth max by a
    log‑sum‑exp surrogate and perform Adam‑style projected gradient
    descent while gradually increasing the smoothing parameter β.
    The procedure is repeated on the top seeds.

3️⃣  **Exact sub‑gradient refinement** – an exponentiated‑gradient step
    on the true L∞‑objective (as in the previous state‑of‑the‑art).

4️⃣  **Guided mass transfer** – a hill‑climber that repeatedly moves a
    small fraction of mass from the coordinate with the largest
    sub‑gradient (most responsible for the current max) to the one
    with the smallest sub‑gradient, accepting only strict improvements.

All work respects the supplied time budget (default 1000 s).
The function returns a simplex‑normalised list that can be handed to
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  FFT‑based (real) convolutions
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a power‑of‑two FFT."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a * a."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
#  Exact objective for a simplex‑normalised vector (∑a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
#  Analytic families (all return vectors that sum to 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    """Linear rise to the centre then fall (symmetric)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    w = 1.0 - np.abs(i - mid) / (mid + 1)
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
    w /= w.sum()
    return w


def _distribution_supergaussian(n: int, p: float, c: float) -> np.ndarray:
    """w_i ∝ exp(-c·|i-mid|^p) for p>0."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    w = np.exp(-c * np.abs(i - mid) ** p)
    total = w.sum()
    if total == 0.0:
        return _distribution_uniform(n)
    w /= total
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i+0.5)^{-p}·(n-i-0.5)^{-p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    w /= w_sum
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    """w_i ∝ exp(-c·i) + exp(-c·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric (or asymmetric) beta‑type distribution."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    total = w.sum()
    if total == 0.0:
        return _distribution_uniform(n)
    w /= total
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.01) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


# ----------------------------------------------------------------------
#  Smoothed‑max mirror descent (Adam + β‑continuation)
# ----------------------------------------------------------------------
def _smooth_max_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """Projected Adam on the log‑sum‑exp surrogate of max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    # β schedule (log‑space, extended)
    beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0]

    for beta in beta_schedule:
        # give each β up to ~3000 iterations (or time‑out)
        local_iter = 0
        while time.time() < time_limit:
            t += 1
            local_iter += 1
            conv = _fft_self_convolve(a)
            max_val = conv.max()
            shifted = conv - max_val            # ≤ 0
            exp_beta = np.exp(beta * shifted)
            w = exp_beta / exp_beta.sum()       # softmax weights, Σw = 1

            a_rev = a[::-1]
            grad_conv = _fft_convolve(w, a_rev)     # length 3n‑2
            grad = 2.0 * grad_conv[n - 1 : n - 1 + n]   # size n

            grad_J = 2.0 * n * grad                # gradient of J = 2·n·smooth_max

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_J
            v = beta2 * v + (1.0 - beta2) * (grad_J ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = 0.15 / (np.sqrt(v_hat) + eps)
            a -= step * m_hat
            a = _project_onto_simplex(a, s=1.0)

            # Periodic exact evaluation
            if t % 150 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-14:
                    best_score, best_seq = cur, a.copy()
                    print(f"[{time.time():.1f}] β={beta:.1f} it={t} new best {best_score:.6f}")

            # Stop current β after a few thousand updates
            if local_iter >= 3000:
                break
        if time.time() >= time_limit:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Exact sub‑gradient descent (exponentiated gradient)
# ----------------------------------------------------------------------
def _subgradient_descent_exact(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 120,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true L∞‑objective."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        # sub‑gradient: average over all maximising shifts
        w = np.zeros_like(conv)
        if max_idxs.size > 0:
            w[max_idxs] = 1.0 / max_idxs.size
        grad = 2.0 * _fft_convolve(w, a[::-1])[n - 1 : n - 1 + n]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)          # keep positivity
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score, best_seq = cur, a.copy()
                print(f"[{time.time():.1f}] exact subgrad it={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Guided mass‑transfer hill‑climber (gradient‑guided)
# ----------------------------------------------------------------------
def _guided_mass_transfer(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    fraction: float = 0.025,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Move a small fraction of mass from the coordinate with the largest
    sub‑gradient (most responsible for the current max) to the one with
    the smallest sub‑gradient.  Accept only strict improvements.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        if max_idxs.size == 0:
            break

        w = np.zeros_like(conv)
        w[max_idxs] = 1.0 / max_idxs.size
        grad = 2.0 * _fft_convolve(w, a[::-1])[n - 1 : n - 1 + n]

        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))

        if a[i_src] <= 1e-15 or i_src == i_dst:
            # fallback to a tiny random move
            src_candidates = np.where(a > 1e-14)[0]
            if src_candidates.size == 0:
                break
            i_src = int(np.random.choice(src_candidates))
            i_dst = int(np.random.randint(n - 1))
            if i_dst >= i_src:
                i_dst += 1

        delta = fraction * a[i_src]
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i_src] -= delta
        new_a[i_dst] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score, best_seq = cur, a.copy()
            print(f"[{time.time():.1f}] guided swap i={i_src}→j={i_dst} δ={delta:.2g} new best {best_score:.6f}")
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a non‑negative step‑function (plain Python list) that
    minimises ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.5          # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, vector)

    # ------------------------------------------------------------------
    # 1️⃣  Analytic families + random seeds
    # ------------------------------------------------------------------
    lengths = [800, 1200, 2000, 3000, 4000, 6000, 8000, 12000, 16000]
    p_grid = np.concatenate(
        [
            np.logspace(-3, -0.5, 13),
            np.logspace(-0.5, 0.0, 11),
            np.linspace(0.2, 3.0, 14),
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 9)
    c_grid = np.logspace(-3, 0, 10)
    alpha_grid = np.logspace(-0.5, 0.7, 12)
    beta_grid = np.logspace(-0.5, 0.7, 12)
    sigma_grid = np.linspace(0.15, 0.45, 5)   # fractions of n
    super_p_grid = [2.0, 3.0, 4.0]
    super_c_grid = np.logspace(-2, 0, 5)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families -------------------------------------------------
        base_fns = [
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
            _distribution_triangular,
        ]
        for fn in base_fns:
            cand = fn(n)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (det) len={n} score={sc:.6f}")

        # Gaussian family -------------------------------------------------------
        for sigma_frac in sigma_grid:
            sigma = sigma_frac * n
            cand = _distribution_gaussian(n, sigma)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] gaussian σ={sigma:.1f} len={n} score={sc:.6f}")

        # Super‑Gaussian ---------------------------------------------------------
        for p in super_p_grid:
            for c in super_c_grid:
                cand = _distribution_supergaussian(n, p, c)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] super‑gauss p={p:.1f} c={c:.3g} len={n} score={sc:.6f}")

        # Beta families (asymmetric) --------------------------------------------
        for alpha in alpha_grid:
            for beta in beta_grid:
                cand = _distribution_beta(n, float(alpha), float(beta))
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] beta α={alpha:.3g} β={beta:.3g} len={n} score={sc:.6f}")

        # Edge‑weighted families -------------------------------------------------
        for p in p_grid:
            edge = _distribution_edge_weighted(n, float(p))
            # mixing with uniform
            for lam in lam_grid:
                cand = _mix_with_uniform(edge, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] edge p={p:.3g} lam={lam:.2f} len={n} score={sc:.6f}")

        # Exponential‑edge families ------------------------------------------------
        for c in c_grid:
            exp_edge = _distribution_exponential_edge(n, float(c))
            for lam in lam_grid:
                cand = _mix_with_uniform(exp_edge, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] exp c={c:.3g} lam={lam:.2f} len={n} score={sc:.6f}")

        # Dirichlet random seeds -------------------------------------------------
        for _ in range(3):
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} score={sc:.6f}")

        # Sparse random seeds ----------------------------------------------------
        for _ in range(3):
            a = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

        # keep only the best few seeds to limit memory/time
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:40]

    # ------------------------------------------------------------------
    # 2️⃣  Include previous construction if available (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3️⃣  Fallback (should never be needed)
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n0 = 2000
        overall_best_seq = _distribution_uniform(n0)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform length={n0} score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Smoothed‑max mirror descent on a few promising seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 40.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] smoothing descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_max_descent(
                seed_seq,
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after smooth seed {idx} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Exact sub‑gradient refinement (exponentiated‑gradient)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        print(f"[{time.time():.1f}] exact sub‑gradient refinement")
        a_opt, sc_opt = _subgradient_descent_exact(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.9,
            eval_every=120,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after exact subgrad new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Guided mass‑transfer polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] guided mass‑transfer polishing")
        a_opt, sc_opt = _guided_mass_transfer(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            fraction=0.025,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after guided swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best objective (2·n·max(conv)) = {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # The evaluation function rescales internally, so we can return the raw simplex‑normalised vector.
    return overall_best_seq.tolist()
