"""
Improved search for a non‑negative step‑function that minimizes
``evaluate_sequence`` (i.e. 2·n·max(conv(a,a)) / (sum a)²).

The algorithm proceeds in three phases:

1️⃣  **Parametric seed generation** – for several lengths we
    finely optimise the parameters of the edge‑weighted,
    exponential‑edge and symmetric beta families using SciPy’s
    bounded scalar optimisation.  Uniform mixing is also optimised.
    The best few seeds are kept.

2️⃣  **Continuation on a smooth surrogate** – a smoothed‑max
    (log‑sum‑exp) surrogate of the max‑convolution is minimised with
    Adam on the probability simplex.  The smoothing parameter β is
    increased from 0.5 up to 500 and many more iterations per β are
    performed than in the previous baseline.

3️⃣  **Exact polishing** – a short exponentiated‑gradient sub‑gradient
    phase followed by a greedy pairwise mass‑transfer hill‑climber
    finishes the search.

All work respects the 1000 s budget and the function returns the raw
simplex‑normalised vector (``evaluate_sequence`` rescales internally)."""

import time
import math
import numpy as np
from scipy.optimize import minimize_scalar
from scipy.special import i0

# ------------------------------------------------------------
#  Utility: projection onto the probability simplex
# ------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of `v` onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
#  FFT based convolutions (real inputs)
# ------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a * a using a real FFT."""
    return _fft_convolve(a, a)


# ------------------------------------------------------------
#  Exact objective (sum(a) = 1)
# ------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) for a with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ------------------------------------------------------------
#  Analytic families (all return a probability vector)
# ------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i+½)^{-p}·(n−i−½)^{-p}  (p>0)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    """w_i ∝ exp(-c·i) + exp(-c·(n‑1‑i))   (c>0)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric beta‑type distribution on the discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    total = w.sum()
    if total == 0.0:
        return _distribution_uniform(n)
    w /= total
    return w


def _distribution_kaiser(n: int, beta_param: float) -> np.ndarray:
    """Kaiser window (β≥0) normalised to a probability vector."""
    if beta_param < 0:
        beta_param = 0.0
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    arg = (i - alpha) / alpha
    w = i0(beta_param * np.sqrt(1 - arg * arg))
    w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ------------------------------------------------------------
#  Smoothed‑max mirror descent (Adam + β‑continuation)
# ------------------------------------------------------------
def _smooth_max_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Minimises the exact objective using a smooth surrogate (log‑sum‑exp)
    with an Adam optimiser, gradually increasing the smoothing parameter β.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    # extended β schedule (more aggressive than the baseline)
    beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0,
                     100.0, 200.0, 300.0, 400.0, 500.0]

    for beta in beta_schedule:
        # allow many more iterations per β (up to 8000) while respecting the deadline
        max_iters = 8000
        while t < max_iters and time.time() < time_limit:
            t += 1
            # full self‑convolution
            conv = _fft_self_convolve(a)               # length 2n‑1
            max_val = conv.max()
            # stable softmax weights
            shifted = conv - max_val                   # ≤ 0
            exp_beta = np.exp(beta * shifted)
            w = exp_beta / exp_beta.sum()              # Σ w = 1, w ≥ 0
            # gradient of the smoothed max:
            a_rev = a[::-1]
            grad_conv = _fft_convolve(w, a_rev)        # length 3n‑2
            grad = 2.0 * grad_conv[n - 1 : n - 1 + n]   # size n
            grad_J = 2.0 * n * grad                    # gradient of 2·n·smooth_max

            # Adam update (projected onto simplex afterwards)
            m = beta1 * m + (1.0 - beta1) * grad_J
            v = beta2 * v + (1.0 - beta2) * (grad_J ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = 0.18 / (np.sqrt(v_hat) + eps)       # slightly larger step
            a -= step * m_hat
            a = _project_onto_simplex(a, s=1.0)

            # periodic exact evaluation
            if t % 200 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-14:
                    best_score, best_seq = cur, a.copy()
                    print(
                        f"[{time.time():.1f}] β={beta:.1f} it={t} new best {best_score:.6f}"
                    )
            # early exit for this β if convergence seems reached
            if t % 4000 == 0:
                break
        if time.time() >= time_limit:
            break
    return best_seq, best_score


# ------------------------------------------------------------
#  Exact sub‑gradient descent (exponentiated gradient)
# ------------------------------------------------------------
def _subgradient_descent_exact(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 120,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true L∞‑objective."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        # exact sub‑gradient (sum of a shifted copies)
        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)          # EG keeps positivity
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score, best_seq = cur, a.copy()
                print(
                    f"[{time.time():.1f}] exact subgrad it={t} new global best {best_score:.6f}"
                )
    return best_seq, best_score


# ------------------------------------------------------------
#  Greedy pairwise mass‑transfer hill‑climber
# ------------------------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly relocate a fraction of mass; accept only strict improvements."""
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score, best_seq = cur, a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ------------------------------------------------------------
#  Parameter optimisation helpers (p, c, λ etc.)
# ------------------------------------------------------------
def _optimize_edge_weighted(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Fine‑tune p and uniform‑mix λ for the edge‑weighted family."""
    def obj_p(p):
        if time.time() >= deadline:
            return 1e9
        return _objective(_distribution_edge_weighted(n, p))

    # coarse bounded search for p
    res = minimize_scalar(
        obj_p,
        bounds=(1e-4, 5.0),
        method="bounded",
        options={"xatol": 1e-4, "maxiter": 30},
    )
    p_opt = res.x if res.success else 0.5

    # optimise λ (mix with uniform) on top of p_opt
    def obj_lam(lam):
        if time.time() >= deadline:
            return 1e9
        cand = _mix_with_uniform(_distribution_edge_weighted(n, p_opt), lam)
        return _objective(cand)

    res_lam = minimize_scalar(
        obj_lam,
        bounds=(0.0, 0.5),
        method="bounded",
        options={"xatol": 1e-3, "maxiter": 30},
    )
    lam_opt = res_lam.x if res_lam.success else 0.0
    best = _mix_with_uniform(_distribution_edge_weighted(n, p_opt), lam_opt)
    return best, _objective(best)


def _optimize_exponential_edge(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Fine‑tune c and λ for the exponential‑edge family."""
    def obj_c(c):

def _optimize_beta(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Symmetric beta family (α=β) optimisation."""
    def obj_alpha(a):
        if time.time() >= deadline:
            return 1e9
        return _objective(_distribution_beta(n, a, a))

    res = minimize_scalar(
        obj_alpha,
        bounds=(0.1, 5.0),
        method="bounded",
        options={"xatol": 1e-3, "maxiter": 30},
    )
    a_opt = res.x if res.success else 1.0
    best = _distribution_beta(n, a_opt, a_opt)
    return best, _objective(best)

#  Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a step‑function (plain Python list) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # small safety margin for the final evaluation
    deadline = start + budget_s - 1.5

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, vector)

    # ------------------------------------------------------------------
    # 1️⃣  Parametric seed generation (length sweep + fine tuning)
    # ------------------------------------------------------------------
    lengths = [1800, 2400, 3000, 3600, 4200, 4800, 5400, 6000]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] analysing length {n}")

        # deterministic families (no parameters)
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
        ):
            cand = base_fn(n)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # edge‑weighted family (p & λ optimisation)
        if time.time() < deadline:
            cand, sc = _optimize_edge_weighted(n, deadline)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] edge‑opt n={n} p/λ improved score={sc:.6f}"
                )

        # exponential‑edge family optimisation
        if time.time() < deadline:
            cand, sc = _optimize_exponential_edge(n, deadline)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] exp‑edge n={n} c/λ improved score={sc:.6f}"
                )

        # symmetric beta family optimisation
        if time.time() < deadline:
            cand, sc = _optimize_beta(n, deadline)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] beta‑opt n={n} α improved score={sc:.6f}"
                )

        # random Dirichlet seeds (diverse starting points)
        for _ in range(3):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} score={sc:.6f}")

        # keep only the top few seeds (memory / time bound)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2️⃣  Continuation on smoothed‑max (Adam) using the best seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 40.0 and top_candidates:
        # take the 5 currently best seeds
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] start smoothed‑max descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_max_descent(
                seed_seq,
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after smooth seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 3️⃣  Exact sub‑gradient refinement
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        print(f"[{time.time():.1f}] exact sub‑gradient refinement")
        a_opt, sc_opt = _subgradient_descent_exact(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.9,
            eval_every=200,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(
                f"[{time.time():.1f}] after exact subgrad new best {overall_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 4️⃣  Final greedy polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy swap polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(
                f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}"
            )

    print(
        f"Final best objective (2·n·max(conv)) = {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    # Return the raw probability vector; ``evaluate_sequence`` rescales internally.
    return overall_best_seq.tolist()
