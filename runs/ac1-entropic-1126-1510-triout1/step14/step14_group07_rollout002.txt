"""
search for a non‑negative step function minimizing `evaluate_sequence`.

The algorithm consists of

1️⃣ broad seeding – analytic families (uniform, sine‑weighted, several
   exponentials, beta, Dirichlet, sparse random) are sampled for a range
   of lengths; the best few are kept.

2️⃣ LP‑based directional improvement – for the current best vector `f`
   we solve a tiny linear programme
        max  Σ g_j
        s.t. g ≥ 0,   f * g ≤ M      (componentwise)
   where `M = max(f*f)`.  The optimal `g` is renormalised and mixed with `f`
   (mirror‑descent style).  This step often yields a noticeable drop in the
   objective.

3️⃣ Adam‑mirrored descent – each retained seed is short‑run optimised with
   an Adam optimiser on the soft‑max parametrisation, using the active set
   of near‑max convolution entries for the sub‑gradient.  The overall best
   seed then receives a long Adam run (as much time as remains).

4️⃣ local polishing – greedy pairwise mass transfers followed by a cheap
   simulated‑annealing walk clean up remaining irregularities.

All phases respect a global time budget (default 1000 s).  The returned
list is a proper probability vector; `evaluate_sequence` will renormalise
any scaling automatically.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Basic geometry utilities
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σ x = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """Exact objective  2·n·max(conv(a,a))  (a assumed to sum to 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector – sum = 1)

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    w /= w.sum()
    return w


def _distribution_exponential_center(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(i - half))
    w /= w.sum()
    return w


def _distribution_exponential_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-rate * i)
    w /= w.sum()
    return w


def _distribution_exponential_right(n: int, rate: float) -> np.ndarray:
    return _distribution_exponential_left(n, rate)[::-1]


def _distribution_beta(n: int, a: float, b: float) -> np.ndarray:
    """Beta(α,β) on the interval (0,1) – discretised on n points."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = (np.maximum(x, 1e-12) ** (a - 1.0)) * (np.maximum(1 - x, 1e-12) ** (b - 1.0))
    if w.sum() == 0.0:
        w[0] = 1.0
    w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator, conc: float) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a.astype(np.float64)


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, U uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Adam‑based mirror descent on the exact objective
def _adam_refine(
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    lr0: float = 0.9,
    active_frac: float = 0.02,
    max_active: int = 30,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_interval: int = 250,
    best_seq: np.ndarray | None = None,
    best_score: float = float("inf"),
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the soft‑max parametrisation.
    Returns the best probability vector seen up to `deadline`.
    """
    n = init_a.shape[0]
    x = np.log(np.maximum(init_a, 1e-12))      # log‑weights
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    ind_grid = np.arange(n, dtype=np.int32)
    t = 0

    if best_seq is None:
        best_seq = init_a.copy()
        best_score = _objective(init_a)

    while time.time() < deadline and t < max_iters:
        t += 1
        a = _softmax(x)                         # stays on simplex
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            # keep the `max_active` largest entries only
            top = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top]

        # sub‑gradient of the max (using the active set)
        grad_a = np.zeros_like(a)
        for idx_k in act:
            offset = idx_k - ind_grid
            mask = (offset >= 0) & (offset < n)
            grad_a[mask] += 2.0 * a[offset[mask]]

        # translate gradient w.r.t. a into gradient w.r.t. log‑weights x
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update with step size decaying like 1/√t
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = lr0 / math.sqrt(t)
        x -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] adam it={t:,} new best {best_score:.6f}")

    # final projection (soft‑max already sums to 1)
    best_seq = _project_onto_simplex(best_seq, s=1.0)
    return best_seq, best_score

# Greedy pairwise refinement (same idea as in the reference code)
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing style local search
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.01,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

# Linear‑programming directional improvement (optional, uses cvxpy)
def _lp_maximize_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """Solve   max Σ g_j   s.t. g≥0,   conv(f,g) ≤ M .
    Returns a normalised (sum = 1) vector or None on failure."""
    try:
        import cvxpy as cp
    except Exception:
        return None

    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)
    conv_fg = cp.conv(cp.Constant(f), g)                # length 2n‑1
    constraints = [conv_fg <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=4000)
    except Exception:
        return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None

    g_val = g.value
    if g_val is None:
        return None
    g_arr = np.maximum(g_val, 0.0)
    s = g_arr.sum()
    if s <= 0.0:
        return None
    return g_arr / s


def _lp_improve(a: np.ndarray, deadline: float, max_iters: int = 3) -> np.ndarray:
    """Apply a few LP‑based improvements to `a`, respecting `deadline`."""
    while max_iters > 0 and time.time() < deadline:
        M = _fft_self_convolve(a).max()
        g = _lp_maximize_sum(a, M)
        if g is None:
            break
        # Mix aggressively (tuned empirically) and re‑project.
        t = 0.3
        a_new = (1.0 - t) * a + t * g
        a_new = _project_onto_simplex(a_new)
        # Accept only if we really moved the max down.
        if _fft_self_convolve(a_new).max() < M - 1e-12:
            a = a_new
        max_iters -= 1
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Return a list of non‑negative heights (probability vector)."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []
    max_top = 20                               # keep only the very best seeds

    # ------------------------------------------------------------------
    # 1️⃣  Broad seeding (various analytic families)
    lengths = [800, 1500, 3000, 5000, 8000, 12000, 16000]
    p_grid = np.logspace(-0.5, 1.5, 9)          # sine‑weighted exponents
    lam_grid = np.linspace(0.0, 0.5, 6)         # mixing with uniform
    rate_factor_grid = [0.4, 0.8, 1.5, 3.0]    # rate = factor / n
    beta_params = [(0.5, 0.5), (1.0, 1.0), (2.0, 2.0),
                   (0.5, 2.0), (2.0, 0.5), (3.0, 1.0), (1.0, 3.0)]
    conc_grid = [0.05, 0.2, 1.0, 5.0]          # Dirichlet concentrations

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        families = [
            _distribution_uniform(n),
            _distribution_sine_weighted(n, p=1.0),
            _distribution_exponential_center(n, rate=0.005),
            _distribution_exponential_left(n, rate=0.003),
            _distribution_exponential_right(n, rate=0.003),
        ]

        for base in families:
            sc = _objective(base)
            top_candidates.append((sc, base.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new global best (det) len={n} score={sc:.6f}")

        # sine‑weighted with uniform mixing
        for p in p_grid:
            if time.time() >= deadline:
                break
            sine = _distribution_sine_weighted(n, p)
            for lam in lam_grid:
                cand = _mix_with_uniform(sine, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new global best sine p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # exponential families with rate proportional to 1/n
        for factor in rate_factor_grid:
            if time.time() >= deadline:
                break
            rate = factor / n
            left = _distribution_exponential_left(n, rate)
            right = _distribution_exponential_right(n, rate)
            cen = _distribution_exponential_center(n, rate)
            for cand in (left, right, cen):
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new global best expo factor={factor:.2f} score={sc:.6f}")

        # Beta‑distribution family
        for a_param, b_param in beta_params:
            if time.time() >= deadline:
                break
            cand = _distribution_beta(n, a_param, b_param)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best Beta α={a_param} β={b_param} score={sc:.6f}")

        # Dirichlet draws
        for conc in conc_grid:
            for _ in range(2):
                if time.time() >= deadline:
                    break
                cand = _distribution_dirichlet(n, rng, conc)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new global best Dirichlet conc={conc:.3g} score={sc:.6f}")

        # Sparse random draws
        for _ in range(3):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}")

        # keep only the very best seeds so far
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:max_top]

    # ------------------------------------------------------------------
    # include a previously found construction if it is available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev[0] = 1.0
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2️⃣  Short Adam refinements on the retained seeds
    refined: list[tuple[float, np.ndarray]] = []
    for sc, cand in top_candidates:
        if time.time() >= deadline:
            break
        slice_deadline = time.time() + min(15.0,
                                          (deadline - time.time()) * 0.04)
        best_seq, best_sc = _adam_refine(
            cand.copy(),
            slice_deadline,
            max_iters=20_000,
            lr0=0.7,
            active_frac=0.03,
            max_active=20,
            beta1=0.9,
            beta2=0.999,
            eval_interval=200,
            best_seq=cand,
            best_score=sc,
        )
        refined.append((best_sc, best_seq))

    # choose the best among the refined candidates
    if refined:
        overall_best_score, overall_best_seq = min(refined, key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 2.5  Optional LP‑based directional improvement (few cheap solves)
    if time.time() < deadline:
        lp_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.1)
        overall_best_seq = _lp_improve(overall_best_seq, lp_deadline, max_iters=3)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] after LP‑step best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Long Adam descent from the overall best seed
    if time.time() < deadline:
        remaining = deadline - time.time()
        long_deadline = time.time() + max(0.0, remaining - 30.0)   # keep ~30 s for polishing
        print(f"[{time.time():.1f}] long Adam descent (≈{remaining-30:.1f}s left)")
        overall_best_seq, overall_best_score = _adam_refine(
            overall_best_seq.copy(),
            long_deadline,
            max_iters=800_000,
            lr0=0.9,
            active_frac=0.015,
            max_active=40,
            beta1=0.9,
            beta2=0.999,
            eval_interval=250,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] after long Adam best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 5️⃣  Final simulated‑annealing polishing
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.02,
            decay=0.9995,
        )

    print(f"Final best score (exact) ≈ {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # `evaluate_sequence` will renormalise, but we keep a proper probability vector.
    return overall_best_seq.tolist()
