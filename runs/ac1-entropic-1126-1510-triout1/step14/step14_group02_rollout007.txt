def _fft_convolve(x,y):
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X*Y, n=size)
    return conv[:n]

"""
Propose a step‑function (list of non‑negative heights) that minimizes
`evaluate_sequence`.  The search proceeds in several stages:

1. **Seed generation** – a diverse collection of analytic families
   (uniform, arcsine, symmetric/asymmetric Beta, edge‑weighted,
   sine‑weighted, power‑law) and random Dirichlet vectors of moderate
   size (≈ 2‑5 k).  The ~10 best candidates are retained.

2. **Smooth‑max gradient descent** – instead of a hard active‑set
   sub‑gradient we minimise the smooth approximation
   `log‑sum‑exp(β·conv(a,a))/β`.  The gradient is obtained analytically
   via a second FFT‑based convolution, and updates are performed with
   an exponentiated‑gradient / Adam‑style rule while projecting onto the
   simplex.  β starts small and is increased gradually, allowing the
   optimizer to first explore and later focus on the true maximum.

3. **Upsampling** – the best distribution is linearly interpolated to a
   longer length (≈ 2×) and refined again with the same smooth descent.

4. **Jitter & re‑descend** – occasional exponential jitter followed by
   another short descent helps escape plateaus.

5. **Symmetrisation test** – many good solutions are (approximately)
   symmetric; the symmetrised version is evaluated and kept if it
   improves the objective.

6. **Weighted pairwise refinement** – a greedy mass‑transfer guided by
   the location of the maximal convolution entry.

7. **Final cheap simulated‑annealing** – a lightweight Metropolis search
   cleans up any remaining spikes.

All steps respect a global wall‑clock budget (default 1000 s).  The
routine always returns a plain Python list of floats; `evaluate_sequence`
will clip negatives / huge values automatically.

# Simplex utilities
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolutions
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors (length len(x)+len(y)-1)."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:n]


# ----------------------------------------------------------------------
# Objective (sum of a is assumed to be 1)
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a with sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_powerlaw_edges(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p) + (n - i + 1) ** (-p)
    w /= w.sum()
    return w


def _distribution_triangular(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = (np.minimum(i + 1, n - i)) ** p
    w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.006) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, where U is the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Force exact left‑right symmetry and re‑project onto the simplex."""
    sym = (seq + seq[::-1]) * 0.5
    return sym / sym.sum()

# Smooth‑max gradient descent (exponentiated‑gradient with Adam‑style momentum)
def _smooth_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.7,
    beta_start: float = 5.0,
    beta_end: float = 400.0,
    beta_mult: float = 1.12,
    eval_interval: int = 120,
    patience: int = 2500,
) -> tuple[np.ndarray, float]:
    """
    Minimises 2·n·max(conv(a,a)) using a smooth log‑sum‑exp
    approximation of the max.  Updates are performed with an
    exponentiated‑gradient / Adam scheme and the iterate is always
    projected onto the probability simplex.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    m = np.zeros_like(a)          # first moment estimate
    v = np.zeros_like(a)          # second moment estimate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    beta = beta_start
    rng = np.random.default_rng()

    while time.time() < deadline and t < 400_000:
        t += 1

        # 1) convolution and soft‑max weights
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        # stabilize exp
        scaled = beta * (conv - max_c)
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # 2) gradient of the smooth objective
        a_rev = a[::-1]
        h = _fft_convolve(w, a_rev)                # length = 3·n‑2
        grad = 4.0 * n * h[n - 1:n - 1 + n]         # shape (n,)

        # Adam‑style momentum on the gradient
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # 3) learning‑rate schedule and update (exponentiated)
        eta = lr0 / math.sqrt(t)
        update = -eta * m_hat / (np.sqrt(v_hat) + eps)

        # Clip extreme updates to avoid numerical overflow
        np.clip(update, -5.0, 5.0, out=update)

        a = a * np.exp(update)

        # 4) simplex projection (simple normalisation is enough)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # 5) occasional evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} new global best {best_score:.6f}")
            else:
                no_improve += 1

        # 6) patience – jitter restart
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.005, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

        # 7) slowly increase β (smoothness → true max)
        if beta < beta_end and t % 500 == 0:
            beta = min(beta * beta_mult, beta_end)

    return best_seq, best_score

# Weighted pairwise refinement (same as before, kept unchanged)
def _pairwise_weighted_refine(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0

    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        conv = _fft_self_convolve(a)
        k_max = int(conv.argmax())

        L = np.zeros(n, dtype=np.float64)
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        idx = np.arange(low, high + 1)
        L[idx] = a[k_max - idx]

        if L.sum() > 0:
            probs_i = L / L.sum()
            i = rng.choice(n, p=probs_i)
        else:
            positives = np.where(a > 1e-12)[0]
            if positives.size == 0:
                break
            i = rng.choice(positives)

        Lmax = L.max()
        probs_j = (Lmax - L) + 1e-12
        probs_j /= probs_j.sum()
        j = rng.choice(n, p=probs_j)
        if j == i:
            j = (j + 1) % n

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] weighted swap {i}->{j} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score

# Cheap simulated annealing (unchanged)
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

# Upsampling helper
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The algorithm follows the stages described
    in the module doc‑string and respects the given time budget.
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Seed generation (moderate lengths)
    lengths = [2000, 2500, 3000, 3500, 4000, 5000]
    p_grid = np.concatenate(
        [
            np.logspace(-3, -1, 9),   # 0.001 … 0.1
            np.logspace(-1, 0, 9),   # 0.1 … 1
            np.logspace(0, 0.5, 7),  # 1 … 3.16
        ]
    )
    lam_grid = np.linspace(0.0, 0.9, 10)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] sampling length {n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (uniform/arcsine) {sc:.6f}")

        # parametric families
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            edge2 = _distribution_edge_inversepow(n, p)
            sine = _distribution_sine_weighted(n, p)
            plaw = _distribution_powerlaw_edges(n, p)
            tri = _distribution_triangular(n, p)

            for base in (beta, edge, edge2, sine, plaw, tri):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # asymmetric beta
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = rng.uniform(0.001, 3.0)
            b = rng.uniform(0.001, 3.0)
            cand = _distribution_beta_asym(n, a, b)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new best asym‑beta len={n} a={a:.3g} b={b:.3g} score={sc:.6f}"
                )

        # random Dirichlet
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_dirichlet(n, rng)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best Dirichlet len={n} score={sc:.6f}")

        # sparse random
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best sparse len={n} score={sc:.6f}")

        # keep only best seeds so far
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:25]

    # ------------------------------------------------------------------
    # 2.  Incorporate any previously supplied construction (if present)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback guard
    if overall_best_seq is None:
        overall_best_seq = _distribution_uniform(3000)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Smooth‑max descent on top seeds (moderate size)
    remaining = deadline - time.time()
    if remaining > 40.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:7]]
        per_seed = remaining * 0.55 / max(1, len(seeds))
        print(
            f"[{time.time():.1f}] smooth‑max descent on {len(seeds)} seeds, ~{per_seed:.1f}s each"
        )
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_gradient_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.8,
                beta_start=5.0,
                beta_end=300.0,
                beta_mult=1.15,
                eval_interval=80,
                patience=2000,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after descent seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Upsample to larger length if time permits
    remaining = deadline - time.time()
    if remaining > 120.0:
        large_n = min(18000, len(overall_best_seq) * 2)
        print(f"[{time.time():.1f}] up‑sampling to {large_n} points for final refinement")
        up_seq = _upsample(overall_best_seq, large_n)

        sub_deadline = time.time() + remaining * 0.6
        up_seq_opt, up_score = _smooth_gradient_descent(
            up_seq,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.6,
            beta_start=10.0,
            beta_end=500.0,
            beta_mult=1.12,
            eval_interval=70,
            patience=1800,
        )
        if up_score < overall_best_score:
            overall_best_score, overall_best_seq = up_score, up_seq_opt.copy()
            print(
                f"[{time.time():.1f}] up‑sampled descent yielded new best {overall_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 6.  Jitter & re‑descend cycles (escape possible plateaus)
    while time.time() < deadline - 15.0:
        rem = deadline - time.time()
        jitter_scale = 0.018
        jitter = rng.exponential(scale=jitter_scale, size=overall_best_seq.shape)
        cand = overall_best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sub_deadline = time.time() + min(20.0, rem * 0.5)
        cand_opt, cand_score = _smooth_gradient_descent(
            cand,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.55,
            beta_start=8.0,
            beta_end=400.0,
            beta_mult=1.10,
            eval_interval=60,
            patience=1500,
        )
        if cand_score < overall_best_score:
            overall_best_score, overall_best_seq = cand_score, cand_opt.copy()
            print(f"[{time.time():.1f}] jitter‑restart improved to {overall_best_score:.6f}")
        else:
            break

    # ------------------------------------------------------------------
    # 7.  Symmetrisation test
    sym_seq = _symmetrize(overall_best_seq)
    sym_score = _objective(sym_seq)
    if sym_score + 1e-14 < overall_best_score:
        overall_best_score, overall_best_seq = sym_score, sym_seq
        print(f"[{time.time():.1f}] symmetrised version better: {sym_score:.6f}")

    # ------------------------------------------------------------------
    # 8.  Weighted pairwise deterministic refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] weighted pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_weighted_refine(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=200_000,
        )

    # ------------------------------------------------------------------
    # 9.  Final cheap simulated‑annealing clean‑up
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
