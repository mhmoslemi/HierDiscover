"""
Propose a non‑negative step function (list of heights) that minimizes
``evaluate_sequence``.

The algorithm consists of three phases:

1. **Seed generation** – a wide variety of analytically‑defined
   distributions (uniform, arcsine, symmetric/asymmetric Beta,
   edge‑weighted, sine‑weighted, random Dirichlet, sparse random) are
   created for several moderate lengths (≈ 2‑5 k).  The ~10 best
   candidates (lowest exact objective 2·n·max(conv)) are kept.

2. **Smooth‑max mirror descent** – for each seed we minimise the
   *soft‑max* of the self‑convolution:
        s_β(a) = (1/β)·log∑_k exp(β·conv_k(a))
   (β is a temperature that is increased gradually).  The gradient of
   s_β is
        ∇s_β(a) = 2·conv_valid(w, a) ,    w_k ∝ exp(β·conv_k)
   where ``conv_valid`` is a “valid’’ convolution (length = len(a)).
   An exponentiated‑gradient (entropy mirror) update
        a_i ← a_i·exp(−η·∇_i)      (η decays as 1/√t)
   is applied and the vector is projected onto the probability simplex.
   This uses the *full* weighted gradient (not just an active set) and
   therefore yields a more reliable descent.

   The schedule consists of β = [5, 20, 80, 200] – low β gives a smooth
   landscape, high β focuses on the true maximum.  The best result of
   each run is retained.

3. **Refinement** – the current best sequence is (i) up‑sampled to a
   longer length (≈ 2×) and re‑optimised with the same descent,
   (ii) optionally symmetrised, (iii) improved by a cheap weighted
   pairwise mass‑transfer, and (iv) finally polished by a brief
   simulated‑annealing local search.

All steps respect a global deadline (default 1000 s).  The returned
value is a plain ``list[float]``; the provided ``evaluate_sequence``
will clip negatives / values > 1000 automatically.
"""

# Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project non‑negative vector v onto {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Convolution utilities (FFT based)
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())               # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)[:need]
    return conv


def _conv_valid_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute the *valid* part of the convolution w * a.
    w has length 2·n‑1, a has length n, the result has length n
    (the same as ``a``).
    """
    n = a.shape[0]
    need = w.shape[0] + n - 1     # = 3·n‑2
    size = 1 << ((need - 1).bit_length())
    W = rfft(w, n=size)
    A = rfft(a, n=size)
    full = irfft(W * A, n=size)[:need]
    start = n - 1                 # = len(w) - n
    return full[start:start + n]

# Exact objective (scale‑invariant, sum(a)=1)
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a with sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (all return a probability vector, sum=1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:

    else:
        w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:

def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.006) -> np.ndarray:

def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base  (U = uniform)."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Force exact left‑right symmetry and re‑normalise."""
    sym = (seq + seq[::-1]) * 0.5
    return sym / sym.sum()

# Smooth‑max mirror descent (exponentiated‑gradient)
def _smooth_max_step(a: np.ndarray, beta: float, eta: float) -> tuple[np.ndarray, float]:
    """
    Perform one mirror‑descent step for the β‑soft‑max objective.
    Returns the updated vector (still sums to 1) and the current max
    convolution value (for monitoring).
    """
    # a has sum 1 and a_i ≥ 0
    conv = _fft_self_convolve(a)           # length 2·n‑1
    max_c = conv.max()

    # soft‑max weights
    scaled = beta * (conv - max_c)         # max becomes 0
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = 2·conv_valid(w, a)
    grad = 2.0 * _conv_valid_fft(w, a)     # shape = n, non‑negative

    # exponentiated‑gradient update on the simplex
    # work in log‑space to avoid underflow
    log_a = np.log(a + 1e-30) - eta * grad
    a_new = np.exp(log_a)
    a_new_sum = a_new.sum()
    if a_new_sum == 0.0:
        a_new = np.full_like(a_new, 1.0 / a_new.size)
    else:
        a_new = a_new / a_new_sum

    return a_new, max_c


def _smooth_max_descent(a0: np.ndarray,
                        deadline: float,
                        best_seq: np.ndarray,
                        best_score: float,
                        beta_schedule: list[float] = [5, 20, 80, 200],
                        eta0: float = 0.8,
                        eval_interval: int = 250,
                        patience: int = 5000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smooth max objective.
    ``a0`` must already be normalised (sum=1).  The function returns the
    best sequence found during the run and its exact objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    no_improve = 0
    # we keep the best from the outside as a reference
    cur_best_seq = best_seq.copy()
    cur_best_score = best_score

    for beta_idx, beta in enumerate(beta_schedule):
        # allocate roughly equal remaining time to each remaining beta
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        per_beta = remaining / (len(beta_schedule) - beta_idx)
        sub_deadline = time.time() + per_beta

        while time.time() < sub_deadline:
            t += 1
            eta = eta0 / math.sqrt(t)               # decaying stepsize
            a, max_c = _smooth_max_step(a, beta, eta)

            if t % eval_interval == 0:
                cur = _objective(a)
                if cur + 1e-14 < cur_best_score:
                    cur_best_score, cur_best_seq = cur, a.copy()
                    no_improve = 0
                    print(f"[{time.time():.1f}] beta={beta:.0f} it={t:,} new best {cur_best_score:.6f}")
                else:
                    no_improve += 1

                if no_improve >= patience:
                    # random jitter to escape a plateau
                    rng = np.random.default_rng()
                    jitter = rng.exponential(scale=0.004, size=n)
                    a = np.maximum(a + jitter, 0.0)
                    a = a / a.sum()
                    no_improve = 0

    return cur_best_seq, cur_best_score

# Weighted pairwise refinement (same as in the reference code)
def _pairwise_weighted_refine(a: np.ndarray,
                              deadline: float,
                              best_seq: np.ndarray,
                              best_score: float,
                              max_trials: int | None = None) -> tuple[np.ndarray, float]:
    """
    Greedy mass transfer guided by the location of the maximal
    convolution entry.  Small moves are accepted only when they improve
    the true objective.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0

    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        conv = _fft_self_convolve(a)
        k_max = int(conv.argmax())

        # impact L_i = a_{k_max-i} (where defined)
        L = np.zeros(n, dtype=np.float64)
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        idx = np.arange(low, high + 1)
        L[idx] = a[k_max - idx]

        if L.sum() > 0.0:
            probs_i = L / L.sum()
            i = rng.choice(n, p=probs_i)
        else:
            pos = np.where(a > 1e-12)[0]
            if pos.size == 0:
                break
            i = rng.choice(pos)

        Lmax = L.max()
        probs_j = (Lmax - L) + 1e-12
        probs_j /= probs_j.sum()
        j = rng.choice(n, p=probs_j)
        if j == i:
            j = (j + 1) % n

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] weighted swap {i}->{j} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score

# Very cheap simulated‑annealing clean‑up
def _random_local_search(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         init_temp: float = 0.008,
                         decay: float = 0.9995) -> tuple[np.ndarray, float]:
    """Metropolis‑type local search with exponential cooling."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp

    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling of a simplex vector
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq

    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # a small safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Seed generation (moderate lengths)
    lengths = [2500, 3000, 3500, 4000, 4500, 5000]
    p_grid = np.concatenate(
        [
            np.logspace(-3, -1, 9),   # 0.001 … 0.1
            np.logspace(-1, 0, 9),   # 0.1 … 1
            np.logspace(0, 0.5, 7),  # 1 … 3.16
        ]
    )
    lam_grid = np.linspace(0.0, 0.95, 10)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating seeds of length {n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (uniform/arcsine) {sc:.6f}")

        # parametric families
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            edge2 = _distribution_edge_inversepow(n, p)
            sine = _distribution_sine_weighted(n, p)

            for base in (beta, edge, edge2, sine):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # asymmetric beta
        for _ in range(8):
            if time.time() >= deadline:
                break
            a = rng.uniform(0.001, 3.0)
            b = rng.uniform(0.001, 3.0)
            cand = _distribution_beta_asym(n, a, b)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best asym‑beta len={n} a={a:.3g} b={b:.3g} score={sc:.6f}")

        # random Dirichlet
        for _ in range(8):
            if time.time() >= deadline:
                break
            cand = _distribution_dirichlet(n, rng)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best Dirichlet len={n} score={sc:.6f}")

        # sparse random
        for _ in range(8):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}")

        # keep only the best few seeds
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:25]

    # ------------------------------------------------------------------
    # 2. Incorporate a previously supplied construction, if any
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never happen)
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Smooth‑max descent on the very best seeds
    remaining = deadline - time.time()
    if remaining > 40.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:7]]
        per_seed = remaining * 0.55 / max(1, len(seeds))
        print(f"[{time.time():.1f}] smooth‑max descent on {len(seeds)} seeds, ≈{per_seed:.1f}s each")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            a_opt, sc_opt = _smooth_max_descent(
                seed_seq.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                beta_schedule=[5, 20, 80, 200],
                eta0=0.8,
                eval_interval=150,
                patience=2500,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after descent seed {idx} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Upsample to a larger length (if time permits) and re‑optimise
    remaining = deadline - time.time()
    if remaining > 120.0:
        large_n = min(18000, len(overall_best_seq) * 2)
        print(f"[{time.time():.1f}] up‑sampling to {large_n}")
        up_seq = _upsample(overall_best_seq, large_n)
        sub_dead = time.time() + remaining * 0.6
        up_opt, up_score = _smooth_max_descent(
            up_seq,
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            beta_schedule=[5, 20, 80, 200],
            eta0=0.7,
            eval_interval=120,
            patience=3000,
        )
        if up_score < overall_best_score:
            overall_best_score, overall_best_seq = up_score, up_opt.copy()
            print(f"[{time.time():.1f}] up‑sampled descent gave new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Jitter‑restart cycles (help escape plateaus)
    while time.time() < deadline - 15.0:
        jitter = rng.exponential(scale=0.018, size=overall_best_seq.shape)
        cand = overall_best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sub_dead = time.time() + min(20.0, (deadline - time.time()) * 0.5)
        cand_opt, cand_score = _smooth_max_descent(
            cand,
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            beta_schedule=[5, 20, 80, 200],
            eta0=0.6,
            eval_interval=100,
            patience=2000,
        )
        if cand_score < overall_best_score:
            overall_best_score, overall_best_seq = cand_score, cand_opt.copy()
            print(f"[{time.time():.1f}] jitter‑restart improved to {overall_best_score:.6f}")
        else:
            break

    # ------------------------------------------------------------------
    # 7. Symmetrisation test
    sym_seq = _symmetrize(overall_best_seq)
    sym_score = _objective(sym_seq)
    if sym_score + 1e-14 < overall_best_score:
        overall_best_score, overall_best_seq = sym_score, sym_seq
        print(f"[{time.time():.1f}] symmetrised version better: {sym_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Weighted pairwise deterministic refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] weighted pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_weighted_refine(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=200_000,
        )

    # ------------------------------------------------------------------
    # 9. Final cheap simulated‑annealing clean‑up
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    # 10. Final reporting
    final_score = _objective(overall_best_seq)
    print(f"Final best score (exact): {final_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
