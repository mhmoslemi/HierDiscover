"""
A fast multi‑stage search for a non‑negative step‑function that minimises
``evaluate_sequence``.

The algorithm proceeds as follows

1.  Create a very diverse pool of analytic seeds (uniform, arcsine,
    beta (symmetric and asymmetric), exponential decay, linear,
    power‑law, Hann, generalized normal, edge‑weighted, random Dirichlet,
    sparse random, piecewise‑constant).  The best few candidates are kept.

2.  Refine each top seed with a **log‑sum‑exp** smoothed objective
    (2·n·τ·log‑sum‑exp(conv/τ)) using two optimisers:
        * Adam on the logits (soft‑max parametrisation).
        * L‑BFGS‑B on the same smoothed objective (if SciPy is available).

3.  Perform an exact exponentiated‑gradient descent on the true
    ``max(conv)`` to polish the solution.

4.  Optional Frank‑Wolfe mass‑transfer moves.

5.  Upsample the best vector (doubling its length) and repeat a short
    Adam burst.

6.  A final pairwise‑annealing (simulated‑annealing) polish.

All stages respect the overall 1000 s budget and always return a plain
``list[float]`` ready for ``evaluate_sequence``.
"""

# Simple utilities

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolutions (real, no padding artefacts)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Correlation of a (len n) with w (len m) for non‑negative shifts.
    Returns length‑n array where element i = Σ a_j·w_{j+i}.
    """
    n = a.shape[0]
    m = w.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()
    rev_w = w[::-1]
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_w, n=size)
    conv = np.fft.irfft(A * B, n=size)
    start = m - 1
    return conv[start:start + n]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact value returned by ``evaluate_sequence`` for an *unnormalised*
    vector a (non‑negative).  The scale cancels out, therefore any
    normalisation (e.g. sum=1) is fine.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, sum=1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q) on the uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{-p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(-|x-½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    """Monotone decreasing power‑law (p>0)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_linear_decrease(n: int) -> np.ndarray:
    """Linear decay from left to right."""
    i = np.arange(n, dtype=np.float64)
    w = (n - i).astype(np.float64)
    w /= w.sum()
    return w


def _distribution_exponential_decay(n: int, lam: float) -> np.ndarray:
    """Geometric decay e^{-lam·i} (lam>0)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-lam * i)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_piecewise_constant(
    n: int, rng: np.random.Generator, nseg: int = 5
) -> np.ndarray:
    """Few random constant pieces (random break‑points)."""
    break_pts = np.sort(rng.choice(n - 1, size=nseg - 1, replace=False)) + 1
    seg_bounds = np.concatenate(([0], break_pts, [n]))
    vals = np.empty(n, dtype=np.float64)
    for i in range(nseg):
        start, end = seg_bounds[i], seg_bounds[i + 1]
        vals[start:end] = rng.random()
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Gradient / sub‑gradient helpers
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Sub‑gradient of the *currently maximal* convolution entry.
    Returns (grad, active_index).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))
    grad = np.zeros_like(a)
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
# Optimisers on the *exact* objective
# ----------------------------------------------------------------------
def _exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the exact (max‑conv) objective."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new global best {best_score:.6f}")
    return best_seq, best_score


def _frank_wolfe_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """Simple Frank‑Wolfe style mass‑transfer."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        i_min = int(np.argmin(grad))
        gamma = 2.0 / (t + 2.0)
        a = a * (1.0 - gamma)
        a[i_min] += gamma
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] FW it={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise simulated annealing polish
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly shift a fraction of mass between two bins."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (keeps simplex)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed (log‑sum‑exp) objective
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    tau_start: float = 0.2,
    tau_end: float = 0.001,
    tau_decay: float = 0.999,
    eval_every: int = 200,
    step_size0: float = 0.05,
) -> tuple[np.ndarray, float]:
    """Adam on 2·n·τ·log‑sum‑exp(conv/τ)."""
    n = a0.shape[0]
    x = np.log(a0 + 1e-15)  # logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    tau = tau_start
    while time.time() < deadline and tau > tau_end:
        t += 1
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        exp_vals = np.exp((conv - max_conv) / tau)
        w = exp_vals / exp_vals.sum()

        corr = _fft_corr(a, w)  # length n
        grad_a = 4.0 * n * corr
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = step_size0 / (np.sqrt(v_hat) + eps)
        x = x - lr * m_hat

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} τ={tau:.6f} new best {best_score:.6f}")

        tau = max(tau * tau_decay, tau_end)

    return best_seq, best_score


# ----------------------------------------------------------------------
# L‑BFGS‑B refinement on the smoothed objective (requires SciPy)
# ----------------------------------------------------------------------
def _lbfgs_optimize(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    tau: float = 0.001,
    max_iter: int = 200,
) -> tuple[np.ndarray, float]:
    """L‑BFGS‑B on the same log‑sum‑exp surrogate as Adam."""
    try:
        import scipy.optimize as _opt
    except Exception:
        # SciPy missing – fall back to the current best
        return best_seq, best_score

    n = a0.shape[0]
    x0 = np.log(a0 + 1e-15)

    def fun(x):
        # soft‑max probabilities
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        exp_vals = np.exp((conv - max_conv) / tau)
        w = exp_vals / exp_vals.sum()

        corr = _fft_corr(a, w)
        grad_a = 4.0 * n * corr
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # smoothed objective value
        obj = 2.0 * n * (max_conv + tau * np.log(np.sum(exp_vals)))
        return obj, grad_x.astype(np.float64)

    # honour the deadline – cut iterations if we run out of time
    options = {"maxiter": max_iter, "disp": False}
    res = _opt.minimize(fun, x0, method="L-BFGS-B", jac=True, options=options)
    if time.time() > deadline:
        return best_seq, best_score

    max_x = np.max(res.x)
    exp_x = np.exp(res.x - max_x)
    a_opt = exp_x / exp_x.sum()
    cur = _objective(a_opt)
    if cur < best_score - 1e-14:
        best_seq = a_opt.copy()
        best_score = cur
        print(f"[{time.time():.1f}] L‑BFGS new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main search entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects a 1000 s budget and
    returns a plain ``list[float]``.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Generate a very diverse pool of analytic seeds.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192, 16384]
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for base in (
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_hann(n),
            _distribution_linear_decrease(n),
        ):
            for lam in (0.0, 0.3, 0.6, 1.0):
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # beta families (symmetric)
        for p in np.logspace(-0.5, 0.5, 6):
            beta = _distribution_beta_sym(n, p)
            for lam in (0.0, 0.3, 0.6, 1.0):
                cand = _mix_with_uniform(beta, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(f"[{time.time():.1f}] new best beta p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # asymmetric beta families
        for _ in range(4):
            p = rng.uniform(0.2, 5.0)
            q = rng.uniform(0.2, 5.0)
            beta_asym = _distribution_beta_asym(n, p, q)
            for lam in (0.0, 0.4, 0.8):
                cand = _mix_with_uniform(beta_asym, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(f"[{time.time():.1f}] new best asym‑beta p={p:.2f} q={q:.2f} lam={lam:.2f} score={sc:.6f}")

        # edge‑weighted
        for p in (0.5, 1.0, 1.5):
            edge = _distribution_edge_weighted(n, p)
            for lam in (0.0, 0.5, 1.0):
                cand = _mix_with_uniform(edge, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(f"[{time.time():.1f}] new best edge p={p:.2f} lam={lam:.2f} score={sc:.6f}")

        # generalized normal
        for q in (0.5, 1.0, 1.5, 2.0):
            gn = _distribution_generalized_normal(n, q)
            for lam in (0.0, 0.5, 1.0):
                cand = _mix_with_uniform(gn, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(f"[{time.time():.1f}] new best gn q={q:.2f} lam={lam:.2f} score={sc:.6f}")

        # exponential decay
        for lam_decay in (0.001, 0.005, 0.02):
            expd = _distribution_exponential_decay(n, lam_decay)
            for lam in (0.0, 0.5, 1.0):
                cand = _mix_with_uniform(expd, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(f"[{time.time():.1f}] new best exp decay λ={lam_decay:.3f} lam={lam:.2f} score={sc:.6f}")

        # power‑law
        for pwr in (0.3, 0.6, 1.0, 1.5, 2.0):
            pl = _distribution_power_law(n, pwr)
            for lam in (0.0, 0.4, 0.8):
                cand = _mix_with_uniform(pl, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(f"[{time.time():.1f}] new best power‑law p={pwr:.2f} lam={lam:.2f} score={sc:.6f}")

        # random Dirichlet seeds
        for _ in range(4):
            alpha = rng.uniform(0.01, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(4):
            cand = _distribution_random_sparse(n, rng, density=0.01)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best sparse len={n} score={sc:.6f}")

        # piecewise‑constant random seeds
        for _ in range(2):
            cand = _distribution_piecewise_constant(n, rng, nseg=5)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best piecewise len={n} score={sc:.6f}")

        # keep only the very best for later refinements
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:15]

    # ------------------------------------------------------------------
    # 2.  Include a previously discovered construction if present.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            top_candidates.append((sc, prev.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Guard against an empty result.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 4096
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Adam optimisation on the smoothed objective (top seeds).
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 40.0 and top_candidates:
        # take the five best seeds for intensive refinement
        seed_seqs = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining * 0.4 / max(1, len(seed_seqs))
        print(f"[{time.time():.1f}] Adam smoothing on {len(seed_seqs)} seeds")
        for idx, seed_seq in enumerate(seed_seqs):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _adam_optimize(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                tau_start=0.3,
                tau_end=0.0008,
                tau_decay=0.998,
                eval_every=150,
                step_size0=0.04,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] after Adam seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  L‑BFGS refinement on the smoothed objective (if SciPy available).
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seed_seqs = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:3]]
        per_seed = remaining * 0.3 / max(1, len(seed_seqs))
        print(f"[{time.time():.1f}] L‑BFGS refinement on {len(seed_seqs)} seeds")
        for idx, seed_seq in enumerate(seed_seqs):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            overall_best_seq, overall_best_score = _lbfgs_optimize(
                seed_seq.copy(),
                sub_deadline,
                overall_best_seq,
                overall_best_score,
                tau=0.001,
                max_iter=250,
            )
            print(f"[{time.time():.1f}] after L‑BFGS {idx} best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Exact exponentiated‑gradient descent (fine‑tuning).
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 8.0:
        sub_deadline = time.time() + remaining
        overall_best_seq, overall_best_score = _exp_gradient_descent(
            overall_best_seq.copy(),
            sub_deadline,
            overall_best_seq,
            overall_best_score,
            eta0=0.7,
            eval_every=80,
        )
        print(f"[{time.time():.1f}] after EG best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Frank‑Wolfe refinement (optional).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fw_deadline = time.time() + (deadline - time.time()) * 0.15
        overall_best_seq, overall_best_score = _frank_wolfe_descent(
            overall_best_seq.copy(),
            fw_deadline,
            overall_best_seq,
            overall_best_score,
            eval_every=120,
        )
        print(f"[{time.time():.1f}] after FW best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 8.  Upsample and short Adam burst.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n <= 64000:
            print(f"[{time.time():.1f}] upsampling from {cur_n} → {target_n}")
            up_seq = _upsample(overall_best_seq, target_n)
            up_score = _objective(up_seq)
            if up_score < overall_best_score:
                overall_best_seq = up_seq
                overall_best_score = up_score
                print(f"[{time.time():.1f}] upsampled new best {overall_best_score:.6f}")

            # brief Adam burst on the up‑sampled vector
            sub_deadline = time.time() + (deadline - time.time()) * 0.12
            a_opt, sc_opt = _adam_optimize(
                overall_best_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                tau_start=0.1,
                tau_end=0.0005,
                tau_decay=0.998,
                eval_every=120,
                step_size0=0.03,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_seq = a_opt.copy()
                overall_best_score = sc_opt
                print(f"[{time.time():.1f}] Adam after upsample new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 9.  Final pairwise annealing polish.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing polish")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
            max_trials=None,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq = a_opt.copy()
            overall_best_score = sc_opt
            print(f"[{time.time():.1f}] after final anneal new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 10.  Report and return.
    # ------------------------------------------------------------------
    print(
        f"[{time.time():.1f}] Final best score (objective=2·n·max(conv)/sum²): "
        f"{overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
