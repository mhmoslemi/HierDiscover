Optimisation of a non‑negative step‑function (a list of heights) to
minimise the provided ``evaluate_sequence``.  The search proceeds in
several stages:

1. **Broad seeding** – generate analytic families (uniform, arcsine,
   symmetric‑Beta, edge‑weighted, sine‑weighted) and random Dirichlet
   vectors for several moderate lengths (≈ 2‑5 k).  For each seed we also
   try a random permutation (to break the natural ordering) and keep
   the ≈ 15 best candidates.

2. **Heavy mirror‑descent** – an exponentiated‑gradient (entropy)
   descent on the exact objective `2·n·max(conv(a,a))` (with
   ``sum(a)=1``).  The active set consists of the largest convolution
   entries; a momentum term (Adam‑like) and a decaying step‑size are
   used.  This is run on each top seed and, after a short up‑sampling,
   on a larger size (≈ 2×).

3. **Jitter / re‑descent** – moderate exponential jitter is added to
   the current best vector and the descent restarts, helping to escape
   plateaus.

4. **Permutation annealing** – the best vector is then optimised in the
   *ordering* of its entries.  Swaps of two positions are accepted
   either if they improve the objective or with a Metropolis probability
   that slowly cools.  Only swaps are performed, so the multiset of
   values stays unchanged.

5. **Symmetrisation test** – a forced left‑right symmetric version of
   the best vector is tried; if it improves the score it is kept.

6. **Weighted pairwise refinement** – greedy mass‑transfer guided by
   the index where the convolution attains its maximum.

7. **Final cheap simulated‑annealing** – a lightweight Metropolis‑type
   local search cleans up any remaining spikes.

All stages respect a global time budget (default 1000 s).  The routine
always returns a plain Python list of floats; `evaluate_sequence`
will clip negatives / huge values automatically.

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())   # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a with sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.006) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, where U is the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Force exact left‑right symmetry and re‑project onto the simplex."""
    sym = (seq + seq[::-1]) * 0.5
    return sym / sym.sum()


# ----------------------------------------------------------------------
# Momentum mirror descent -----------------------------------------------
def _subgradient_descent_momentum(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    active_frac: float = 0.04,
    max_active: int = 30,
    beta: float = 0.9,
    eval_interval: int = 120,
    patience: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (entropy mirror) descent on the exact,
    non‑smooth objective.  The active set consists of those convolution
    entries whose value lies within ``active_frac`` of the current
    maximum.  Momentum (Adam‑like) smooths the sub‑gradient.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    ind_grid = np.arange(n, dtype=np.int32)

    while time.time() < deadline and t < 500_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            top_idx = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top_idx]

        grad = np.zeros_like(a)
        for k in act:
            shift = k - ind_grid
            mask = (shift >= 0) & (shift < n)
            grad[mask] += 2.0 * a[shift[mask]]

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * momentum)

        # Normalise to the simplex (all entries stay ≥0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} new global best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            # exponential jitter to escape flat region
            jitter = rng.exponential(scale=0.004, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Weighted pairwise refinement -------------------------------------------
def _pairwise_weighted_refine(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer guided by the location of the current maximal
    convolution entry.  The impact of moving mass from i to j is
    estimated via L_i = a_{k_max-i}; we preferentially move from high
    impact to low impact indices.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0

    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        conv = _fft_self_convolve(a)
        k_max = int(conv.argmax())

        # Compute impact values L_i = a_{k_max-i} where defined
        L = np.zeros(n, dtype=np.float64)
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        idx = np.arange(low, high + 1)
        L[idx] = a[k_max - idx]

        # Sampling distributions
        if L.sum() > 0:
            probs_i = L / L.sum()                     # favour large impact
            i = rng.choice(n, p=probs_i)
        else:
            positives = np.where(a > 1e-12)[0]
            if positives.size == 0:
                break
            i = rng.choice(positives)

        Lmax = L.max()
        probs_j = (Lmax - L) + 1e-12                 # larger weight where impact is small
        probs_j /= probs_j.sum()
        j = rng.choice(n, p=probs_j)
        if j == i:
            j = (j + 1) % n

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] weighted swap {i}->{j} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Cheap simulated annealing ---------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Simple Metropolis‑type local search (continuous transfers)."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing ---------------------------------
def _permute_simulated_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.02,
    decay: float = 0.9996,
    max_swaps: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Optimise the ordering of the entries of ``a`` by swaps.
    The multiset of values stays unchanged, therefore the sum is fixed.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    cur_seq = a.copy()
    cur_score = _objective(cur_seq)
    temperature = init_temp
    swaps = 0

    while time.time() < deadline:
        if max_swaps is not None and swaps >= max_swaps:
            break
        swaps += 1

        i, j = rng.choice(n, size=2, replace=False)
        # perform swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_score = _objective(cur_seq)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.1f}] perm swap improved global best {best_score:.6f}")
        else:
            # revert swap
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        temperature = max(temperature * decay, 1e-8)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling helper -------------------------------------------------------
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimizes
    ``evaluate_sequence``.  The algorithm follows the multi‑stage
    pipeline described in the module doc‑string and respects the given
    time budget.
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Seed generation (moderate lengths) + random permutations
    lengths = [2000, 2500, 3000, 3500, 4000, 5000]
    p_grid = np.concatenate(
        [
            np.logspace(-3, -1, 9),   # 0.001 … 0.1
            np.logspace(-1, 0, 9),   # 0.1 … 1
            np.logspace(0, 0.5, 7),  # 1 … 3.16
        ]
    )
    lam_grid = np.linspace(0.0, 0.9, 10)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] sampling length {n}")

        # ----- deterministic families -------------------------------------------------
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (uniform/arcsine) {sc:.6f}")

            # random permutation of the same values
            perm_cand = cand[rng.permutation(n)]
            scp = _objective(perm_cand)
            top_candidates.append((scp, perm_cand.copy()))
            if scp < overall_best_score:
                overall_best_score, overall_best_seq = scp, perm_cand.copy()
                print(f"[{time.time():.1f}] new global best (perm) {scp:.6f}")

        # ----- parameterised families -------------------------------------------------
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            edge2 = _distribution_edge_inversepow(n, p)
            sine = _distribution_sine_weighted(n, p)

            for base in (beta, edge, edge2, sine):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}")

                    # permutation of the mixed candidate
                    perm_cand = cand[rng.permutation(n)]
                    scp = _objective(perm_cand)
                    top_candidates.append((scp, perm_cand.copy()))
                    if scp < overall_best_score:
                        overall_best_score, overall_best_seq = scp, perm_cand.copy()
                        print(f"[{time.time():.1f}] new global best (perm mix) {scp:.6f}")

        # ----- asymmetric beta --------------------------------------------------------
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = rng.uniform(0.001, 3.0)
            b = rng.uniform(0.001, 3.0)
            cand = _distribution_beta_asym(n, a, b)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best asym‑beta len={n} a={a:.3g} b={b:.3g} score={sc:.6f}")

        # ----- random Dirichlet -------------------------------------------------------
        for _ in range(12):
            if time.time() >= deadline:
                break
            cand = _distribution_dirichlet(n, rng)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best Dirichlet len={n} score={sc:.6f}")

        # ----- sparse random ---------------------------------------------------------
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}")

        # keep only the very best seeds (keep diversity limited)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:25]

    # ------------------------------------------------------------------
    # 2.  Incorporate any previously supplied construction (if present)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback guard
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Heavy mirror‑descent on a handful of top seeds (moderate size)
    remaining = deadline - time.time()
    if remaining > 40.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:7]]
        per_seed_time = remaining * 0.55 / max(1, len(seeds))
        print(
            f"[{time.time():.1f}] starting heavy descent on {len(seeds)} seeds, ≈{per_seed_time:.1f}s each"
        )
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed_time
            a_opt, sc_opt = _subgradient_descent_momentum(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=1.0,
                active_frac=0.05,
                max_active=40,
                beta=0.9,
                eval_interval=100,
                patience=1800,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after descent seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Upsample to a larger length if time permits
    remaining = deadline - time.time()
    if remaining > 120.0:
        large_n = min(18000, len(overall_best_seq) * 2)
        print(f"[{time.time():.1f}] up‑sampling to {large_n} points for final refinement")
        up_seq = _upsample(overall_best_seq, large_n)

        sub_deadline = time.time() + remaining * 0.6
        up_seq_opt, up_score = _subgradient_descent_momentum(
            up_seq,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.7,
            active_frac=0.04,
            max_active=35,
            beta=0.92,
            eval_interval=80,
            patience=2000,
        )
        if up_score < overall_best_score:
            overall_best_score, overall_best_seq = up_score, up_seq_opt.copy()
            print(
                f"[{time.time():.1f}] up‑sampled descent yielded new best {overall_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 6.  Jitter & re‑descent cycles (escape possible plateaus)
    while time.time() < deadline - 15.0:
        rem = deadline - time.time()
        jitter_scale = 0.018
        jitter = rng.exponential(scale=jitter_scale, size=overall_best_seq.shape)
        cand = overall_best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sub_deadline = time.time() + min(20.0, rem * 0.5)
        cand_opt, cand_score = _subgradient_descent_momentum(
            cand,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.6,
            active_frac=0.06,
            max_active=30,
            beta=0.9,
            eval_interval=70,
            patience=1500,
        )
        if cand_score < overall_best_score:
            overall_best_score, overall_best_seq = cand_score, cand_opt.copy()
            print(f"[{time.time():.1f}] jitter‑restart improved to {overall_best_score:.6f}")
        else:
            break

    # ------------------------------------------------------------------
    # 7.  Permutation‑based simulated annealing (new stage)
    if time.time() < deadline - 30.0:
        perm_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.3)
        overall_best_seq, overall_best_score = _permute_simulated_annealing(
            overall_best_seq.copy(),
            perm_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.02,
            decay=0.9996,
            max_swaps=20000,
        )

    # ------------------------------------------------------------------
    # 8.  Symmetrisation test
    sym_seq = _symmetrize(overall_best_seq)
    sym_score = _objective(sym_seq)
    if sym_score + 1e-14 < overall_best_score:
        overall_best_score, overall_best_seq = sym_score, sym_seq
        print(f"[{time.time():.1f}] symmetrised version better: {sym_score:.6f}")

    # ------------------------------------------------------------------
    # 9.  Weighted pairwise deterministic refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] weighted pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_weighted_refine(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=200_000,
        )

    # ------------------------------------------------------------------
    # 10.  Final cheap simulated‑annealing clean‑up
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
