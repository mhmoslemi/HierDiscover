import time
import math
import numpy as np

def _fft_self_convolve(a):
    n = a.shape[0]
    size = 1 << ((2*n-1)-1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A*A, n=size)
    return conv[:2*n-1]

def _objective(a):
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2 * n * max_b / (s * s))

propose_candidate
-----------------

Generate a non‑negative step‑function (list of heights) that tries to minimize
the opaque ``evaluate_sequence`` objective.

Algorithm outline
-----------------
1. **Exploratory seeds** – Scan a handful of analytic families
   (uniform, arcsine, edge‑weighted, sum‑edge‑weighted, generalized‑normal)
   over a modest set of lengths and parameters.  Keep the best few seeds.

2. **Evolutionary refinement** – Maintain a small population (≈30) of the
   current best vectors.  Repeatedly create offspring by convex mixing two
   parents, apply a light mutation, and run a short exponentiated‑gradient
   (mirror) descent on the offspring.  Accept the offspring if it improves
   the population; update the global best.

3. **Final polish** – A simple pairwise‑mass‑transfer simulated‑annealing
   phase finishes the search.

All operations respect the non‑negativity and sum‑normalisation constraints,
so the returned list can be fed directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  FFT‑based self‑convolution (full linear convolution)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
#  Exact objective used by the evaluator (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a))/sum(a)^2  – a must be non‑negative."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
#  Analytic families – all return a normalised (sum = 1) probability vector
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i·(n‑i))^{-p}   with i = distance to the left edge."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_sum_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ i^{-p} + (n‑i)^{-p} (sum of two one‑sided tails)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (-p)) + (j ** (-p))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp( -|x‑½|^{q} ) on the discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1‑lam) + uniform·lam."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
#  Sub‑gradient of the active convolution entry (gradient of max)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """Return (gradient, index_of_max) for the current max entry of a*a."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))

    grad = np.zeros_like(a)

    else:

        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
#  Pairwise‑mass‑transfer annealing (final polish)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins; accept improvements."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        # pick a distinct destination
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4  # up to 40 % of source mass
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score
