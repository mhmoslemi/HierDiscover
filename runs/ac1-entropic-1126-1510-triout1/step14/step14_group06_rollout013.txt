"""
A multi‑stage optimizer for the max‑autocorrelation problem.

The pipeline:
 1. Diverse analytic seeds → keep the best few.
 2. For each top seed run a sequence of L‑BFGS‑B optimisations on the
    temperature‑smoothed objective 2·n·τ·log‑sum‑exp(conv/τ) while
    gradually lowering τ (smooth → exact max).  This yields a far more
    accurate local optimum than pure Adam.
 3. Refinement with the exact objective by exponentiated‑gradient descent.
 4. Optional Frank‑Wolfe move, up‑sampling and a short annealing polish.

All phases respect the global 1000 s budget and finally return a plain
list[float] ready for ``evaluate_sequence``.
"""
import time
import math
import numpy as np
from scipy.optimize import minimize

# ---------- FFT utilities ----------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    m = w.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()
    rev_w = w[::-1]
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_w, n=size)
    conv = np.fft.irfft(A * B, n=size)
    start = m - 1
    return conv[start:start + n]

# ---------- Exact objective ----------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))

# ---------- Analytic distributions ----------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w

def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w

def _distribution_power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    w /= w.sum()
    return w

def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals

def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# ---------- Softmax ----------
def _softmax(x: np.ndarray) -> np.ndarray:
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()

# ---------- Smoothed objective + gradient ----------
def _smooth_obj_and_grad(x: np.ndarray, n: int, tau: float):
    """Return (value, gradient) of the τ‑smoothed objective."""
    # probabilities
    a = _softmax(x)

    # convolution a*a
    conv = _fft_self_convolve(a)

    # smooth max via log‑sum‑exp
    max_c = conv.max()
    exp_vals = np.exp((conv - max_c) / tau)
    w = exp_vals / exp_vals.sum()          # softmax weights over shifts

    # gradient w.r.t a (see Adam implementation)
    corr = _fft_corr(a, w)                 # length n
    grad_a = 4.0 * n * corr                # ∂/∂a of 2·n·τ·log‑sum‑exp

    # chain rule through the softmax
    avg = np.dot(a, grad_a)
    grad_x = a * (grad_a - avg)

    # smoothed max value
    smooth_max = max_c + tau * np.log(exp_vals.sum())
    value = 2.0 * n * smooth_max

    return value, grad_x

# ---------- L‑BFGS smoothing ----------
def _l_bfgs_smoothing(
    a0: np.ndarray,
    tau_schedule: list[float],
    deadline: float,
    maxiter_per_tau: int = 80,
) -> tuple[np.ndarray, float]:
    """Repeated L‑BFGS optimisation for decreasing τ.

    Returns the best probability vector (sum = 1) found and its exact score.
    """
    n = a0.shape[0]
    # initialise logits; add tiny epsilon to avoid log(0)
    x = np.log(a0 + 1e-15)

    best_a = a0.copy()
    best_score = _objective(best_a)

    for tau in tau_schedule:
        if time.time() >= deadline:
            break
        # SciPy's minimize will call the function many times; we limit by maxiter
        res = minimize(
            fun=_smooth_obj_and_grad,
            x0=x,
            args=(n, tau),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': maxiter_per_tau, 'ftol': 1e-12, 'disp': False}
        )
        # update logits
        x = res.x

        # evaluate exact objective
        a = _softmax(x)
        cur_score = _objective(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] L‑BFGS τ={tau:.6f} new best {best_score:.6f}")

    return best_a, best_score

# ---------- Gradient‑based refinements (reuse from previous code) ----------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))
    grad = np.zeros_like(a)
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k

def _exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new best {best_score:.6f}")
    return best_seq, best_score

def _frank_wolfe_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        i_min = int(np.argmin(grad))
        gamma = 2.0 / (t + 2.0)
        a = a * (1.0 - gamma)
        a[i_min] += gamma
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] FW it={t} new best {best_score:.6f}")
    return best_seq, best_score

def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01
        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)
        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score

def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= a_new.sum()
    return a_new

"""
A multi‑stage optimizer for the max‑autocorrelation problem.

The pipeline:
 1. Generate a diverse set of analytic seeds (uniform, arcsine, beta,
    edge‑weighted, Hann, generalized‑normal, power‑law, Dirichlet,
    sparse).  Keep the best few.
 2. For each top seed run several L‑BFGS‑B optimisations on the
    temperature‑smoothed objective
        2·n·τ·log‑sum‑exp(conv/τ)
    while gradually cooling τ (τ → 0).  This gives a more accurate
    local optimum than pure Adam.
 3. Refine the current best with exponentiated‑gradient descent on the
    exact objective.
 4. Optional Frank‑Wolfe move, up‑sampling and a short L‑BFGS burst.
 5. Final pair‑wise annealing polish.

All stages respect a global 1000 s budget and finally return a plain
``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import operator
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation of a (len n) with w (len m) for non‑negative shifts.
    Returns an array of length n where element i equals Σ_j a_j·w_{j+i}.
    """
    n = a.shape[0]
    m = w.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()
    rev_w = w[::-1]
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_w, n=size)
    conv = np.fft.irfft(A * B, n=size)
    start = m - 1
    return conv[start:start + n]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a (unnormalised)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic probability families (always sum to 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Softmax helper
# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# ----------------------------------------------------------------------
# τ‑smoothed objective + gradient (for L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(x: np.ndarray, n: int, tau: float):
    """
    Return (value, gradient) of the τ‑smoothed objective
        2·n·τ·log‑sum‑exp(conv/τ)
    where conv = a*a and a = softmax(x).
    """
    a = _softmax(x)

    conv = _fft_self_convolve(a)

    max_c = conv.max()
    exp_vals = np.exp((conv - max_c) / tau)
    w = exp_vals / exp_vals.sum()                           # softmax over shifts

    corr = _fft_corr(a, w)                                 # length n
    grad_a = 4.0 * n * corr                                 # ∂/∂a of the smoothed term

    avg = np.dot(a, grad_a)
    grad_x = a * (grad_a - avg)                             # chain rule through softmax

    smooth_max = max_c + tau * np.log(exp_vals.sum())
    value = 2.0 * n * smooth_max

    return value, grad_x


# ----------------------------------------------------------------------
# L‑BFGS optimisation with a cooling τ schedule
# ----------------------------------------------------------------------
def _l_bfgs_smoothing(
    a0: np.ndarray,
    tau_schedule: list[float],
    deadline: float,
    maxiter_per_tau: int = 80,
) -> tuple[np.ndarray, float]:
    """Repeated L‑BFGS optimisation for decreasing τ.

    Returns the best probability vector (sum = 1) and its exact score.
    """
    n = a0.shape[0]
    x = np.log(a0 + 1e-15)               # initialise logits safely

    best_a = a0.copy()
    best_score = _objective(best_a)

    for tau in tau_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=_smooth_obj_and_grad,
            x0=x,
            args=(n, tau),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_tau, "ftol": 1e-12, "disp": False},
        )
        x = res.x
        a = _softmax(x)
        cur_score = _objective(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] L‑BFGS τ={tau:.6f} new best {best_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient of the active convolution entry (used by EG & FW)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))
    grad = np.zeros_like(a)
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple Frank‑Wolfe style refinement
# ----------------------------------------------------------------------
def _frank_wolfe_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        i_min = int(np.argmin(grad))
        gamma = 2.0 / (t + 2.0)
        a = a * (1.0 - gamma)
        a[i_min] += gamma
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] FW it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise‑annealing polishing
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01
        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)
        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling of a probability vector
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the total 1000 s budget
    and returns a plain list ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Populate a diverse pool of seeds.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n), _distribution_hann(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # beta and edge‑weighted families
        for p in np.logspace(-0.5, 0.5, 6):
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, float(p))
            edge = _distribution_edge_weighted(n, float(p))
            for base in (beta, edge):
                for lam in (0.0, 0.3, 0.6, 1.0):
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(f"[{time.time():.1f}] new best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # generalized normal shapes
        for q in (0.5, 1.0, 1.5, 2.0):
            if time.time() >= deadline:
                break
            cand = _distribution_generalized_normal(n, q)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best gn q={q:.2f} len={n} score={sc:.6f}")

        # power‑law families
        for pwr in (0.3, 0.6, 1.0, 1.5, 2.0):
            if time.time() >= deadline:
                break
            cand = _distribution_power_law(n, pwr)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best powerlaw p={pwr:.2f} len={n} score={sc:.6f}")

        # Dirichlet random seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.01)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best sparse len={n} score={sc:.6f}")

        # keep only the best ten overall so far
        top_candidates.sort(key=operator.itemgetter(0))
        top_candidates = top_candidates[:10]

    # ------------------------------------------------------------------
    # 2. Include a previously discovered construction if available.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            top_candidates.append((sc, prev.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Guard against empty result.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 4096
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. L‑BFGS smoothing on the top candidates.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seed_seqs = [seq for _, seq in sorted(top_candidates, key=operator.itemgetter(0))[:5]]
        tau_schedule = [0.3, 0.15, 0.07, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001]
        per_seed_budget = remaining * 0.5 / max(1, len(seed_seqs))
        print(f"[{time.time():.1f}] L‑BFGS smoothing on {len(seed_seqs)} seeds")
        for idx, seed_seq in enumerate(seed_seqs):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed_budget
            a_opt, sc_opt = _l_bfgs_smoothing(
                seed_seq.copy(),
                tau_schedule,
                sub_deadline,
                maxiter_per_tau=80,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] after L‑BFGS seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Exact exponentiated‑gradient descent (fine tuning).
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 8.0:
        sub_deadline = time.time() + remaining
        best_seq, best_score = _exp_gradient_descent(
            overall_best_seq.copy(),
            sub_deadline,
            overall_best_seq,
            overall_best_score,
            eta0=0.7,
            eval_every=80,
        )
        overall_best_seq, overall_best_score = best_seq, best_score

    # ------------------------------------------------------------------
    # 6. Frank‑Wolfe refinement (optional).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fw_deadline = time.time() + (deadline - time.time()) * 0.15
        overall_best_seq, overall_best_score = _frank_wolfe_descent(
            overall_best_seq.copy(),
            fw_deadline,
            overall_best_seq,
            overall_best_score,
            eval_every=120,
        )
        print(f"[{time.time():.1f}] after FW best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Upsample and a short L‑BFGS burst (if time permits).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n <= 64000:
            print(f"[{time.time():.1f}] upsampling from {cur_n} → {target_n}")
            up_seq = _upsample(overall_best_seq, target_n)
            up_score = _objective(up_seq)
            if up_score < overall_best_score:
                overall_best_seq = up_seq
                overall_best_score = up_score
                print(f"[{time.time():.1f}] upsampled new best {overall_best_score:.6f}")

            sub_deadline = time.time() + (deadline - time.time()) * 0.2
            tau_sched = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
            a_opt, sc_opt = _l_bfgs_smoothing(
                overall_best_seq.copy(),
                tau_sched,
                sub_deadline,
                maxiter_per_tau=60,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_seq = a_opt.copy()
                overall_best_score = sc_opt
                print(f"[{time.time():.1f}] L‑BFGS after upsample new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final pairwise annealing polish.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing polish")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            overall_best_seq,
            overall_best_score,
            rng,
            max_trials=None,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq = a_opt.copy()
            overall_best_score = sc_opt
            print(f"[{time.time():.1f}] after final anneal new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"[{time.time():.1f}] Final best score (objective=2·n·max(conv)/sum²): "
        f"{overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
