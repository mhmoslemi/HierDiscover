"""
Optimisation of the step‑function bound 2·n·max(conv) on the probability
simplex.

The algorithm proceeds in three stages within the time budget:

1️⃣  **Broad exploration** – a large collection of analytic families (uniform,
    arcsine, sine/cosine, quadratic, beta‑edge, edge‑weighted, exponential‑edge,
    inverse‑power‑edge) is sampled on several lengths (up to 20 000).  For each
    family a fine grid of parameters (including convex mixing with the uniform
    distribution) is evaluated.  Random sparse Dirichlet draws and random
    convex mixtures of a few families are also generated.  Only the best ≈80
    candidates are retained.

2️⃣  **Refinement** – the best seeds are optimised with a smooth (soft‑max)
    approximation of the L∞ objective.  The gradient is obtained via FFT‑based
    convolution, and an Adam‑style exponentiated‑gradient update is applied in
    the log‑probability domain, keeping the iterates on the simplex.  After a
    short soft‑max phase the exact sub‑gradient descent is run to polish the
    solution.

3️⃣  **Final polishing** – a greedy pairwise mass‑transfer hill‑climber
    (with optional simulated‑annealing style acceptance) removes any remaining
    slack.

All helper functions are top‑level; there are no closures or lambda
expressions.  The routine returns a plain Python list, which the external
``evaluate_sequence`` call will re‑normalise automatically.
"""

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_cross_correlate_valid(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (valid part)  (w ⋆ a)[j] = Σ_i w[i+j]·a[i],
    returning an array of length len(a).  Implemented with FFT for speed.
    """
    n = a.shape[0]
    m = w.shape[0]                # = 2·n‑1
    size = 1 << ((m + n - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A = np.fft.rfft(a, n=size)
    corr = np.fft.irfft(W * np.conj(A), n=size)
    return corr[:n]               # valid region


def _fft_sliding_window_sum(w: np.ndarray, win: int) -> np.ndarray:
    """
    For each j compute Σ_{i=j}^{j+win‑1} w[i] (window length = win).
    Returns an array of length win (i.e. len(a)).  Implemented via FFT.
    """
    m = w.shape[0]
    size = 1 << ((m + win - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    ones = np.ones(win, dtype=np.float64)
    O = np.fft.rfft(ones, n=size)
    conv = np.fft.irfft(W * O, n=size)
    start = win - 1
    return conv[start:start + win]


# ----------------------------------------------------------------------
# Analytic probability families (all sum to 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_edge(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta‑type density w_i ∝ (i·(n‑i))^{α‑1} (α>0)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = (i * (n - i)) ** (alpha - 1.0)
    vals = np.clip(vals, 0.0, None)
    s = vals.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    vals /= s
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i·(n‑i))^{‑p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = (i * (n - i)) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    w /= w_sum
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_inverse_power_edge(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i+1)^{‑p} + (n‑i)^{‑p}."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 1) ** (-p) + (n - i) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    w /= w_sum
    return w


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.005
) -> np.ndarray:
    """Sparse Dirichlet‑like seed with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base (λ∈[0,1])."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _random_mixture(
    n: int, rng: np.random.Generator, families, n_families: int = 3
) -> np.ndarray:
    """Random convex mixture of *n_families* members from *families*."""
    chosen = rng.choice(families, size=n_families, replace=False)
    weights = rng.random(n_families)
    weights /= weights.sum()
    mix = np.zeros(n, dtype=np.float64)
    for w, fn in zip(weights, chosen):
        # draw random parameters appropriate for the family
        if fn.__name__ == "_distribution_beta_edge":
            alpha = rng.uniform(0.001, 0.3)
            mix += w * _distribution_beta_edge(n, alpha)
        elif fn.__name__ == "_distribution_edge_weighted":
            p = 10 ** rng.uniform(-4, 1)  # log‑uniform 1e‑4 .. 10
            mix += w * _distribution_edge_weighted(n, p)
        elif fn.__name__ == "_distribution_exponential_edge":
            c = 10 ** rng.uniform(-4, 0)  # log‑uniform 1e‑4 .. 1
            mix += w * _distribution_exponential_edge(n, c)
        elif fn.__name__ == "_distribution_inverse_power_edge":
            p = 10 ** rng.uniform(-4, 1)
            mix += w * _distribution_inverse_power_edge(n, p)
        elif fn.__name__ == "_distribution_arcsine":
            mix += w * _distribution_arcsine(n)
        elif fn.__name__ == "_distribution_sine":
            mix += w * _distribution_sine(n)
        elif fn.__name__ == "_distribution_cosine":
            mix += w * _distribution_cosine(n)
        elif fn.__name__ == "_distribution_quadratic":
            mix += w * _distribution_quadratic(n)
        else:
            mix += w * _distribution_uniform(n)
    # ensure exact normalisation (tiny numerical drift may happen)
    mix = np.maximum(mix, 0.0)
    s = mix.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    mix /= s
    return mix

# Objective
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) for a probability vector (sum=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth (soft‑max) descent with Adam‑style updates
# ----------------------------------------------------------------------
def _smooth_descent_adam(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    tau_start: float = 0.02,
    tau_end: float = 0.001,
    lr: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (Adam) descent on a smoothed version of the L∞
    objective (soft‑max temperature τ annealed linearly from ``tau_start`` to
    ``tau_end``).  The iterate is kept on the simplex via an explicit
    normalisation step after each update.
    """
    a = a0.copy()
    n = a.shape[0]
    # work in log‑space to avoid under‑flow for near‑zero entries
    loga = np.log(np.maximum(a, 1e-30))
    m = np.zeros_like(loga)
    v = np.zeros_like(loga)
    start = time.time()
    it = 0
    while time.time() < time_limit:
        it += 1
        # --------- evaluate current exact objective ----------
        a = np.exp(loga)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
            loga = np.log(a)
            a_sum = 1.0
        a /= a_sum

        cur_score = _objective(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] smooth (Adam) it={it} new best {best_score:.6f}")

        # --------- soft‑max temperature ----------
        frac = (time.time() - start) / (time_limit - start)
        tau = tau_start * (1.0 - frac) + tau_end * frac

        # --------- compute smoothed gradient ----------
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        shifted = conv - maxc
        # stable softmax
        exp_vals = np.exp(shifted / tau)
        w = exp_vals / exp_vals.sum()                     # length 2n‑1, sums to 1
        term1 = _fft_cross_correlate_valid(w, a)          # length n
        win_sum = _fft_sliding_window_sum(w, n)           # length n
        grad = term1 + a * win_sum                       # shape (n,)

        # Gradient w.r.t. log‑probabilities:
        grad_log = grad * a                               # ∂L/∂log a_i = ∂L/∂a_i * a_i

        # Adam update in log‑space
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        loga -= step

        # No explicit projection needed – the later normalisation restores the simplex.
        # Prevent numerical overflow/underflow:
        if np.any(np.isnan(loga)) or np.any(np.isinf(loga)):
            loga = np.log(np.maximum(a, 1e-30))

    return best_seq, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (same as original)
# ----------------------------------------------------------------------
def _subgradient_descent_exact(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent directly on the L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] exact subgrad it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise hill‑climber (with optional annealing)
# ----------------------------------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
    temperature: float = 0.0,
) -> tuple[np.ndarray, float]:
    """
    Random pairwise mass transfers.  With ``temperature>0`` a move that
    worsens the objective can be accepted with probability exp(−Δ/temperature).
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = int(rng.choice(src))
        j = int(rng.integers(0, n - 1))
        if j == i:
            continue

        delta = a[i] * rng.random() * 0.45  # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
        elif temperature > 0.0:
            # accept with probability exp(-(cur-best_score)/temperature)
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                a = new_a
    return best_seq, best_score

        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            # allocate a major chunk for the soft‑max phase (≈65 % of per‑seed budget)
            smooth_dead = time.time() + per_seed * 0.65
            a_opt, sc_opt = _smooth_descent_adam(
                seed_seq,
                smooth_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                tau_start=0.02,
                tau_end=0.001,
                lr=0.5,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after smooth seed {idx} new global best {overall_best_score:.6f}"
                )
            # remaining time for exact sub‑gradient polishing (≈35 %)
            exact_dead = time.time() + per_seed * 0.35
            a_opt, sc_opt = _subgradient_descent_exact(
                overall_best_seq,
                exact_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                eval_every=500,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after exact polish seed {idx} new best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5️⃣ Final greedy pairwise polishing (with a tiny temperature for occasional uphill moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy swap polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=None,
            temperature=1e-4,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best objective (2·n·max(conv)): {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return plain Python list – external ``evaluate_sequence`` will re‑normalise.
    return overall_best_seq.tolist()
