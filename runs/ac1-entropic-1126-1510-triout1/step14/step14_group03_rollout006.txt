"""
Hybrid optimizer for the step‑function problem.

The algorithm proceeds in four stages:

1. **Seed generation** – deterministic families (uniform, arcsine,
   sum‑edge‑weighted, generalized normal) are sampled for several
   lengths.  The 20 best seeds (lowest exact objective) are retained.

2. **Exponentiated‑gradient refinement** – a multiplicative mirror
   descent on the true objective (max‑convolution) quickly improves the
   seeds.

3. **Smooth‑max gradient descent** – the non‑smooth max is replaced by a
   soft‑max (log‑sum‑exp) with a temperature β that is increased during
   the run.  The gradient of the smoothed objective is computed using
   FFT‑based convolutions and a multiplicative update (still keeping the
   vector on the probability simplex).  This step is able to escape the
   plateaus where the pure max‑gradient gets stuck.

4. **Pairwise mass‑transfer annealing** – a cheap simulated‑annealing
   polish that randomly moves a fraction of mass between two bins and
   accepts improvements (or occasional uphill moves).

All steps respect the 1000 s wall‑time limit (2 s safety margin).  The
returned list contains non‑negative heights; the sum is normalised to 1,
which does not affect the evaluation because the objective is scale‑invariant.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  FFT utilities (real, full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Convolution a * a (full, real) using FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = _next_pow2(n)
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:n]


# ----------------------------------------------------------------------
#  Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) / (sum(a)²).  a may contain zeros, must be non‑negative."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
#  Analytic families – return probability vectors (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_sum_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ i^{‑p} + (n‑i)^{‑p} (sum of two one‑sided tails)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (-p)) + (j ** (-p))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Discrete density ∝ exp( -|x‑½|^{q} )."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1‑lam) + uniform·lam."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
#  Sub‑gradient of max‑convolution (used in the EG stage)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """Return (gradient, index_of_max) for the active entry of a*a."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))
    grad = np.zeros_like(a)

    if k < n:
        # overlap length = k+1   (i runs 0..k)
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
#  Exponentiated‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """Multiplicative mirror descent on the true max‑convolution objective."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Smooth‑max gradient descent (soft‑max approximation of the max)
# ----------------------------------------------------------------------
def _smooth_exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.5,
    beta_start: float = 1.0,
    beta_end: float = 120.0,
    eval_every: int = 30,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative update using the gradient of a log‑sum‑exp
    approximation of max(conv).  The temperature β is increased
    gradually (beta_start → beta_end) to tighten the approximation.
    """
    n = a.shape[0]
    start = time.time()
    total_time = deadline - start
    t = 0
    while time.time() < deadline:
        t += 1
        # linear schedule for β on the interval [beta_start, beta_end]
        fraction = (time.time() - start) / total_time
        beta = beta_start + fraction * (beta_end - beta_start)

        # ----- smooth max (soft‑max) weights -----
        conv = _fft_self_convolve(a)                     # size 2n‑1
        # stability shift
        shifted = beta * conv
        max_shifted = shifted.max()
        exp_shifted = np.exp(shifted - max_shifted)
        w = exp_shifted / exp_shifted.sum()               # probabilities, length 2n‑1

        # ----- gradient of the smoothed objective -----
        # grad_i = Σ_j a_j w[i+j] + Σ_j a_j w[i‑j]
        conv_aw = _fft_convolve(a, w)                     # length 3n‑2
        conv_rev_aw = _fft_convolve(a[::-1], w)           # length 3n‑2
        grad = conv_aw[:n] + conv_rev_aw[n - 1 : 2 * n - 1]

        # ----- multiplicative mirror update -----
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # ----- projection onto the simplex (renormalise) -----
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)

        # ----- periodic evaluation -----
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth GD step={t} β={beta:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
#  Pairwise‑mass‑transfer annealing (final polish)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins; accept improvements."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4  # move at most 40 % of source mass
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Main entry point required by the challenge
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function (list of heights) that
    minimises ``evaluate_sequence``.  The algorithm follows the four
    stages described in the module docstring.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0       # safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")
    top: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Generate a diverse pool of seeds
    # ------------------------------------------------------------------
    lengths = [3000, 5000, 8000, 12000, 16000]   # moderate to large sizes
    p_grid = np.concatenate((
        np.logspace(-2.0, -0.5, 12, base=10),   # 0.01 … 0.316
        np.logspace(-0.5, 0.5, 12, base=10),    # 0.316 … 3.16
        np.logspace(0.5, 1.5, 10, base=10),     # 3.16 … 31.6
    ))
    lam_grid = np.linspace(0.0, 1.0, 11)        # convex mixes with uniform
    q_grid = np.linspace(0.5, 3.0, 9)           # generalized‑normal exponents

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # ----------------------------------------------------------------
        # deterministic families (uniform, arcsine)
        # ----------------------------------------------------------------
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            top.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best ({n}) score={sc:.6f}")

        # ----------------------------------------------------------------
        # sum‑edge‑weighted family + mixing with uniform
        # ----------------------------------------------------------------
        for p in p_grid:
            if time.time() >= deadline:
                break
            base = _distribution_edge_sum_weighted(n, p)
            sc = _objective(base)
            top.append((sc, base.copy()))
            if sc < best_score:
                best_score, best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new best sum‑edge p={p:.3g} score={sc:.6f}")

            for lam in lam_grid:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                top.append((sc, cand.copy()))
                if sc < best_score:
                    best_score, best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new best sum‑edge p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # ----------------------------------------------------------------
        # generalized‑normal family
        # ----------------------------------------------------------------
        for q in q_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_generalized_normal(n, q)
            sc = _objective(cand)
            top.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best gn q={q:.2f} score={sc:.6f}")

        # keep only the elite seeds for the next stage
        top.sort(key=lambda x: x[0])
        top = top[:20]

    # ------------------------------------------------------------------
    # 2. Incorporate a previously submitted construction if available
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < best_score:
                best_score, best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Guard against an empty result
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _distribution_uniform(5000)
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Exponentiated‑gradient refinement (quick improvement)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + (deadline - time.time()) * 0.30  # ~30 % of remaining time
        print(f"[{time.time():.1f}] EG refinement (budget {sub_deadline - time.time():.1f}s)")
        best_seq, best_score = _exp_gradient_descent(
            best_seq.copy(),
            sub_deadline,
            best_seq,
            best_score,
            eta0=0.9,
            eval_every=60,
        )
        print(f"[{time.time():.1f}] after EG best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Smooth‑max gradient descent (continuation on β)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + (deadline - time.time()) * 0.55  # ~55 % of remaining time
        print(f"[{time.time():.1f}] smooth‑max GD (budget {sub_deadline - time.time():.1f}s)")
        best_seq, best_score = _smooth_exp_gradient_descent(
            best_seq.copy(),
            sub_deadline,
            best_seq,
            best_score,
            eta0=0.5,
            beta_start=1.0,
            beta_end=120.0,
            eval_every=30,
        )
        print(f"[{time.time():.1f}] after smooth GD best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing")
        best_seq, best_score = _pairwise_annealing(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            rng,
        )
        print(f"[{time.time():.1f}] after annealing best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Return the best candidate found
    # ------------------------------------------------------------------
    print(
        f"Final best score (objective = 2·n·max(conv)/sum²): {best_score:.6f} "
        f"(length={len(best_seq)})"
    )
    # ensure a plain Python list of floats
    return best_seq.tolist()
