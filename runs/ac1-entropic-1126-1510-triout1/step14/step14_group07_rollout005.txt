"""
Improved step‑function search.

The algorithm explores a rich family of U‑shaped and sinusoidal
distributions, keeps the best seeds (by the exact objective
`2·n·max(conv)`) and refines them with an Adam‑based mirror descent that
uses a true sub‑gradient of the max‑convolution.  After a long
fine‑grained Adam run on the overall best seed we apply greedy
pairwise mass transfers and a simulated‑annealing polish.

All work respects a global 1000 s budget.  The returned list is a
probability vector (sum≈1) – the evaluator will re‑normalise it anyway.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (a assumed to sum to 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector – sum = 1)

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    w /= w.sum()
    return w


def _distribution_exponential_center(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(i - half))
    w /= w.sum()
    return w.astype(np.float64)


def _distribution_exponential_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-rate * i)
    w /= w.sum()
    return w.astype(np.float64)


def _distribution_exponential_right(n: int, rate: float) -> np.ndarray:
    return _distribution_exponential_left(n, rate)[::-1]


def _distribution_exponential_ends(n: int, rate: float) -> np.ndarray:
    """U‑shaped exponential: high at both ends, low in the middle."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-rate * np.minimum(i, n - 1 - i))
    w /= w.sum()
    return w.astype(np.float64)


def _distribution_beta_U(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) with α<1 – heavy tails, low centre."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n          # shift to avoid exact 0/1
    w = (x * (1.0 - x)) ** (alpha - 1.0)
    # guard against overflow / NaN for extremely small x
    w = np.where(np.isfinite(w), w, 0.0)
    w[w < 0] = 0.0
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full(n, 1.0 / n)
    else:
        w /= w_sum
    return w.astype(np.float64)


def _distribution_dirichlet(n: int, rng: np.random.Generator, conc: float) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a.astype(np.float64)


def _distribution_random_sparse(
    n: int,
    rng: np.random.Generator,
    density: float = 0.006,
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, U uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
def _adam_refine(
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    lr0: float = 0.8,
    active_frac: float = 0.02,
    max_active: int = 30,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_interval: int = 250,
    best_seq: np.ndarray | None = None,
    best_score: float = float("inf"),
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent Adam on the exact objective.
    Parameters are chosen for a robust, slowly‑decaying schedule.
    """
    n = init_a.shape[0]
    x = np.log(np.maximum(init_a, 1e-12))
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    ind_grid = np.arange(n, dtype=np.int32)
    t = 0

    if best_seq is None:
        best_seq = init_a.copy()
        best_score = _objective(init_a)

    while time.time() < deadline and t < max_iters:
        t += 1
        a = _softmax(x)                     # always a simplex point
        conv = _fft_self_convolve(a)
        max_c = conv.max()

        # active set: values close to the current maximum
        thresh = max_c * (1.0 - active_frac)
        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            top = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top]

        # sub‑gradient of max (using the active set)
        grad_a = np.zeros_like(a)
        for idx_k in act:
            offset = idx_k - ind_grid
            mask = (offset >= 0) & (offset < n)
            grad_a[mask] += 2.0 * a[offset[mask]]

        # convert grad wrt a into grad wrt logits x (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = lr0 / math.sqrt(t)            # decaying step‑size
        x -= eta * m_hat / (np.sqrt(v_hat) + eps)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] Adam it={t:,} new best {best_score:.6f}"
                )
    # final safety projection
    best_seq = _project_onto_simplex(best_seq, s=1.0)
    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Greedy transfer of a random fraction of mass from a non‑zero entry
    to another random index.
    """
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score

def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.01,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """
    Very cheap simulated‑annealing style local search.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}"
            )
        else:
            # accept with probability exp(-(Δ)/T)
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score
