"""
Optimisation of a non‑negative step function for `evaluate_sequence`.

The algorithm works in three phases:

1️⃣  **Randomised family sampling** – we draw a large number of
    candidate probability vectors from a rich catalogue of analytic
    families (uniform, arcsine, beta‑edge, edge‑weighted, exponential‑edge,
    sine, cosine, quadratic) on several lengths.  Parameter ranges have
    been dramatically widened (e.g. beta‑edge exponent α ∈ [0.001,0.99])
    compared to the previous version.  The best ~10‑15 candidates are kept.

2️⃣  **Two‑stage continuous optimisation** – each kept seed is refined
    first by an exponentiated‑gradient descent on a *soft‑max* surrogate
    of the L∞ objective (temperature annealed from 2·10⁻² to 10⁻³) and
    then by an exact sub‑gradient EG descent on the true max‑convolution.
    The two stages receive 40 % / 50 % of the time budget per seed;
    learning‑rates are slightly larger than before.

3️⃣  **Polishing** – a greedy pairwise mass‑transfer hill‑climber is
    applied to the globally best vector for the remaining time.

All vectors are kept on the probability simplex (sum = 1) – the
objective is scale‑invariant, so this does not affect the final score.
The implementation is fully vectorised, uses real FFTs for O(n log n)
convolution and respects the 1000 s overall wall‑clock budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_cross_correlate_valid(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (valid part)  (w ⋆ a)[j] = Σ_i w[i+j]·a[i],
    returning an array of length len(a).  Implemented with FFT for speed.
    """
    n = a.shape[0]
    m = w.shape[0]                # = 2·n‑1
    size = 1 << ((m + n - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A = np.fft.rfft(a, n=size)
    corr = np.fft.irfft(W * np.conj(A), n=size)
    return corr[:n]               # valid region


def _fft_sliding_window_sum(w: np.ndarray, win: int) -> np.ndarray:
    """
    For each j compute Σ_{i=j}^{j+win‑1} w[i] (window length = win).
    Returns an array of length win (i.e. len(a)).  Implemented via FFT.
    """
    m = w.shape[0]
    size = 1 << ((m + win - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    ones = np.ones(win, dtype=np.float64)
    O = np.fft.rfft(ones, n=size)
    conv = np.fft.irfft(W * O, n=size)
    start = win - 1
    return conv[start:start + win]


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------


def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_edge(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta‑type density w_i ∝ (i·(n‑i))^{α‑1} (α>0)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = (i * (n - i)) ** (alpha - 1.0)
    s = vals.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    vals /= s
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i·(n‑i))^{-p} (p≥0)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = (i * (n - i)) ** (-p)
    s = w.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    w /= s
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse Dirichlet‑like seed with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Exact objective (sum = 1 ⇒ 2·n·max(conv))
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth (soft‑max) gradient descent
# ----------------------------------------------------------------------


def _smooth_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    tau_start: float = 0.02,
    tau_end: float = 0.001,
    lr0: float = 1.0,
    eval_every: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a smoothed version of the L∞ objective.
    The soft‑max temperature τ is linearly annealed from ``tau_start`` to
    ``tau_end``.  The (exact) sub‑gradient of the smoothed objective is
    computed via FFT‑based cross‑correlation.
    """
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    it = 0

    while time.time() < time_limit:
        it += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] smooth it={it} new global best {best_score:.6f}")

        # anneal τ
        frac = (time.time() - start) / (time_limit - start)
        tau = tau_start * (1.0 - frac) + tau_end * frac

        # soft‑max weights (stable)
        shifted = conv - max_val
        exp_vals = np.exp(shifted / tau)
        w = exp_vals / exp_vals.sum()            # length 2n‑1, sums to 1

        # gradient of smoothed objective
        term1 = _fft_cross_correlate_valid(w, a)   # Σ w[i+j]·a[i]
        win_sum = _fft_sliding_window_sum(w, n)   # Σ_{k=j}^{j+n‑1} w[k]
        grad = term1 + a * win_sum

        # EG update (step‑size ∝ 1/√t)
        eta = lr0 / math.sqrt(it)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if it % eval_every == 0:
            pass

    return best_seq, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (exponentiated‑gradient)
# ----------------------------------------------------------------------


def _subgradient_descent_exact(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 1.0,
    eval_every: int = 200,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] exact subgrad it={t} new global best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise hill‑climber
# ----------------------------------------------------------------------


def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a step‑function (plain Python list) that minimises
    ``evaluate_sequence`` within the given time budget.

    The search consists of a quick random family sweep, followed by
    intensive smooth + exact EG descent on the few most promising seeds
    and a final greedy polishing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the external environment
    deadline = start_time + budget_s - 2.0

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, seq)

    # ------------------------------------------------------------------
    # Phase 1 – Random family sampling (roughly 15 % of budget)
    # ------------------------------------------------------------------
    length_options = [2000, 4000, 6000, 8000, 12000, 16000]
    families = [
        ("uniform", lambda n: _distribution_uniform(n)),
        ("arcsine", lambda n: _distribution_arcsine(n)),
        ("quadratic", lambda n: _distribution_quadratic(n)),
        ("sine", lambda n: _distribution_sine(n)),
        ("cosine", lambda n: _distribution_cosine(n)),
        ("beta_edge", None),            # parametrised later
        ("edge_weighted", None),
        ("exp_edge", None),
        ("dirichlet", None),
        ("sparse", None),
    ]

    # decide how much time to spend on this phase
    phase1_deadline = min(deadline, start_time + 0.15 * (budget_s - 2.0))
    while time.time() < phase1_deadline:
        n = int(rng.choice(length_options))
        fam_name, base_fn = rng.choice(families)

        # generate candidate distribution
        if fam_name == "beta_edge":
            alpha = rng.uniform(0.001, 0.99)            # extended range
            lam = rng.random()
            base = _distribution_beta_edge(n, alpha)
            cand = _mix_with_uniform(base, lam)
        elif fam_name == "edge_weighted":
            p = 10 ** rng.uniform(-3, 0.7)              # up to ~5
            lam = rng.random()
            base = _distribution_edge_weighted(n, p)
            cand = _mix_with_uniform(base, lam)
        elif fam_name == "exp_edge":
            c = 10 ** rng.uniform(-4, 0)                # very slow decay to fast
            lam = rng.random()
            base = _distribution_exponential_edge(n, c)
            cand = _mix_with_uniform(base, lam)
        elif fam_name == "dirichlet":
            concentration = rng.uniform(0.01, 2.0)
            cand = rng.dirichlet(np.full(n, concentration))
        elif fam_name == "sparse":
            density = rng.uniform(0.004, 0.02)
            cand = _distribution_random_sparse(n, rng, density=density)
        else:
            # deterministic families (uniform, arcsine, quadratic, sine, cosine)
            cand = base_fn(n)

        # ensure simplex (should already hold)
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            cand = _distribution_uniform(n)
        else:
            cand /= cand_sum

        sc = _objective(cand)
        # keep a short list of the best so far
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] new global best (phase1) len={n} score={sc:.6f}")

        top_candidates.append((sc, cand.copy()))
        # keep only the top 25 seeds to bound memory/time
        if len(top_candidates) > 25:
            top_candidates.sort(key=lambda x: x[0])
            top_candidates = top_candidates[:25]

    # ------------------------------------------------------------------
    # Phase 2 – Continuous refinement of the best seeds
    # ------------------------------------------------------------------
    # ensure we have at least one seed
    if not top_candidates:
        # fallback to a simple arcsine seed
        n = 4000
        overall_best_seq = _distribution_arcsine(n)
        overall_best_score = _objective(overall_best_seq)
        top_candidates = [(overall_best_score, overall_best_seq.copy())]

    # keep the five most promising seeds for deep optimisation

    seeds = [seq for _, seq in top_candidates[:5]]
    remaining = deadline - time.time()
    if remaining <= 0.0:

    per_seed = max(remaining / len(seeds), 0.0)

    for idx, seed_seq in enumerate(seeds):

        # smooth descent – 40 % of this seed's budget

        a_opt, sc_opt = _smooth_descent(
            seed_seq,

            overall_best_seq.copy(),

            best_seq=overall_best_seq,

    # ------------------------------------------------------------------
    # Phase 3 – Final global polishing (use any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:

            deadline,
            best_seq=overall_best_seq,

        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = a_opt.copy(), sc_opt

    )
    # Return plain Python list – evaluate_sequence will re‑normalise internally.
    return overall_best_seq.tolist()
