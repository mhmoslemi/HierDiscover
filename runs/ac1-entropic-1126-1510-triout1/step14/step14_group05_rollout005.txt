"""
Improved search for a non‑negative step function that minimizes
``evaluate_sequence``.

The algorithm runs in four stages within the supplied time budget:

1️⃣  **Broad seeding** – a rich set of analytic families (uniform,
    arcsine, sine, cosine, quadratic, beta‑edge, edge‑weighted,
    exponential‑edge) over several lengths plus random Dirichlet and
    sparse seeds.  The best few (≈30) candidates are retained.

2️⃣  **Differential evolution** – a simple DE optimiser (exponential
    mutation, binomial crossover) works on the simplex.  The population
    is initialised from the retained seeds and runs until the time
    budget leaves ∼30 % for the final phases.

3️⃣  **Soft‑max gradient descent** – a smooth approximation of the
    ``max`` with an annealed temperature (τ∈[5·10⁻²,5·10⁻⁴]) and an
    exponentiated‑gradient (EG) update.  The best vector from DE is
    refined; the exact sub‑gradient descent (EG on the true L∞
    objective) follows for polishing.

4️⃣  **Greedy pairwise hill‑climber** – small random mass transfers
    between coordinates are accepted only if they strictly improve the
    exact objective.  This final polish often yields a few extra
    10⁻⁴‑level improvements.

All stages respect the overall 1000 s budget, perform frequent
time‑checks and keep the globally best sequence found.  The returned
list is a plain Python list; ``evaluate_sequence`` will re‑normalise it
internally.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project vector v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = v.astype(np.float64, copy=False)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT utilities for convolutions (real inputs)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_cross_correlate_valid(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (valid part):
    (w ⋆ a)[j] = Σ_i w[i+j]·a[i], length = len(a).
    """
    n = a.shape[0]
    m = w.shape[0]                # = 2·n‑1
    size = 1 << ((m + n - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A = np.fft.rfft(a, n=size)
    corr = np.fft.irfft(W * np.conj(A), n=size)
    return corr[:n]


def _fft_sliding_window_sum(w: np.ndarray, win: int) -> np.ndarray:
    """
    Sliding‑window sums of length ``win`` over w.
    Returns an array of length ``win`` (i.e. len(a)).
    """
    m = w.shape[0]
    size = 1 << ((m + win - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    ones = np.fft.rfft(np.ones(win, dtype=np.float64), n=size)
    conv = np.fft.irfft(W * ones, n=size)
    start = win - 1
    return conv[start:start + win]


# ----------------------------------------------------------------------
# Analytic families – all return a *probability* vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_beta_edge(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta‑type density w_i ∝ (i·(n‑i))^{α‑1} (α>0)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = (i * (n - i)) ** (alpha - 1.0)
    vals = np.clip(vals, 0.0, None)
    s = vals.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    vals /= s
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    w /= w_sum
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Sparse Dirichlet‑like seed with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Exact objective (sum = 1 ⇒ 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth (soft‑max) gradient descent on a smoothed max‑convolution
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    time_limit: float,
                    best_seq: np.ndarray,
                    best_score: float,
                    tau_start: float = 0.05,
                    tau_end: float = 0.0005,
                    lr0: float = 0.8,
                    eval_every: int = 200) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on a smoothed max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    it = 0

    while time.time() < time_limit:
        it += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] smooth it={it} new best {best_score:.6f}")

        # linear annealing of τ
        frac = (time.time() - start) / (time_limit - start)
        tau = tau_start * (1.0 - frac) + tau_end * frac

        # soft‑max weights (stable version)
        maxc = conv.max()
        shifted = conv - maxc
        exp_vals = np.exp(shifted / tau)
        w = exp_vals / exp_vals.sum()          # length 2n‑1, sums to 1

        # gradient of smoothed objective
        term1 = _fft_cross_correlate_valid(w, a)      # Σ w[i+j]·a[i]
        win_sum = _fft_sliding_window_sum(w, n)       # Σ_{k=j}^{j+n‑1} w[k]
        grad = term1 + a * win_sum

        # EG update
        eta = lr0 / math.sqrt(it)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = _distribution_uniform(n)
        else:
            a /= s

        if it % eval_every == 0:
            pass

    return best_seq, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (exponentiated gradient on true L∞)
# ----------------------------------------------------------------------
def _subgradient_descent_exact(a0: np.ndarray,
                               time_limit: float,
                               best_seq: np.ndarray,
                               best_score: float,
                               lr0: float = 0.9,
                               eval_every: int = 100) -> tuple[np.ndarray, float]:
    """EG descent directly on 2·n·max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    while time.time() < time_limit:
        it += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(it)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = _distribution_uniform(n)
        else:
            a /= s

        if it % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] exact it={it} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple differential evolution on the simplex
# ----------------------------------------------------------------------
def _differential_evolution(pop: list[np.ndarray],
                            time_limit: float,
                            best_seq: np.ndarray,
                            best_score: float,
                            rng: np.random.Generator,
                            F: float = 0.8,
                            CR: float = 0.9) -> tuple[np.ndarray, float]:
    """Classic DE (DE/rand/1/bin) where every individual lives on the simplex."""
    pop = [np.copy(v) for v in pop]
    popsize = len(pop)
    n = pop[0].shape[0]
    scores = np.array([_objective(v) for v in pop], dtype=np.float64)

    # keep global best up‑to‑date
    best_idx = int(np.argmin(scores))
    if scores[best_idx] < best_score:
        best_score = float(scores[best_idx])
        best_seq = pop[best_idx].copy()

    while time.time() < time_limit:
        for i in range(popsize):
            if time.time() >= time_limit:
                break

            # choose three distinct indices a,b,c ≠ i
            idxs = list(range(popsize))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)

            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = np.maximum(mutant, 0.0)
            s = mutant.sum()
            if s == 0.0:
                mutant = _distribution_uniform(n)
            else:
                mutant /= s

            # binomial crossover
            cross = rng.random(n) < CR
            if not cross.any():
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, pop[i])

            trial_score = _objective(trial)
            if trial_score < scores[i] - 1e-14:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score - 1e-14:
                    best_score = trial_score
                    best_seq = trial.copy()
                    print(f"[{time.time():.1f}] DE improved to {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise hill‑climber (mass transfer)
# ----------------------------------------------------------------------
def _pairwise_swap(a0: np.ndarray,
                   time_limit: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator,
                   max_trials: int | None = None) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.45  # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Produce a step‑function (list of non‑negative heights) that aims to
    minimise ``evaluate_sequence`` within the allocated time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    overall_best_seq = None          # np.ndarray, sum = 1
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, seq)

    # ------------------------------------------------------------------
    # Helper to insert a new candidate into the elite pool
    # ------------------------------------------------------------------
    def _register(seq: np.ndarray):
        nonlocal overall_best_seq, overall_best_score, top_candidates
        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s == 0.0:
            return
        seq = seq / s
        sc = _objective(seq)
        top_candidates.append((sc, seq.copy()))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = seq.copy()
            print(f"[{time.time():.1f}] new global best len={len(seq)} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 0. Bring in a previous construction if it exists
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            _register(prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Broad seeding with analytic families and random draws
    # ------------------------------------------------------------------
    lengths = [4000, 8000, 12000, 16000, 20000]   # moderate set of lengths
    lam_grid = np.linspace(0.0, 1.0, 6)          # 0,0.2,...,1.0
    p_grid = np.array([0.02, 0.05, 0.1, 0.2, 0.5])
    alpha_grid = np.array([0.1, 0.2, 0.3, 0.4])
    c_grid = np.array([1e-3, 5e-3, 1e-2, 5e-2])

    max_elite = 30   # keep only the best 30 candidates overall

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families (no mixing)
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
        ):
            _register(base_fn(n))

        # beta‑edge family + mixing
        for alpha in alpha_grid:
            base = _distribution_beta_edge(n, float(alpha))
            for lam in lam_grid:
                cand = _mix_with_uniform(base, float(lam))
                _register(cand)

        # edge‑weighted family + mixing
        for p in p_grid:
            base = _distribution_edge_weighted(n, float(p))
            for lam in lam_grid:
                cand = _mix_with_uniform(base, float(lam))
                _register(cand)

        # exponential‑edge family + mixing
        for c in c_grid:
            base = _distribution_exponential_edge(n, float(c))
            for lam in lam_grid:
                cand = _mix_with_uniform(base, float(lam))
                _register(cand)

        # random Dirichlet seeds (various concentration)
        for alpha in [0.01, 0.05, 0.1, 0.5, 1.0, 2.0]:
            a = rng.dirichlet(np.full(n, float(alpha)))
            _register(a)

        # sparse random seeds
        for _ in range(6):
            a = _distribution_random_sparse(n, rng, density=0.005)
            _register(a)

        # keep elite list bounded
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:max_elite]

    # If nothing was found (should not happen) fall back to uniform
    if overall_best_seq is None:
        n = 8000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback to uniform length={n} score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Differential Evolution on the best‑length seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        best_len = overall_best_seq.shape[0]
        # collect elite candidates of the best length
        elite_of_best_len = [seq for sc, seq in top_candidates if seq.shape[0] == best_len]
        # ensure we have enough individuals
        popsize = max(12, min(30, len(elite_of_best_len)))
        init_pop = elite_of_best_len[:popsize]

        # augment with random Dirichlet if needed
        while len(init_pop) < popsize:
            a = rng.dirichlet(np.full(best_len, 0.1))
            init_pop.append(a)

        de_deadline = time.time() + remaining * 0.55   # allocate ≈55 % to DE
        print(f"[{time.time():.1f}] start DE on length={best_len} popsize={popsize}")
        overall_best_seq, overall_best_score = _differential_evolution(
            init_pop,
            de_deadline,
            overall_best_seq,
            overall_best_score,
            rng,
            F=0.8,
            CR=0.9,
        )
    else:
        print(f"[{time.time():.1f}] skipping DE (insufficient time)")

    # ------------------------------------------------------------------
    # 3. Smooth‑max gradient descent + exact sub‑gradient polish
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        smooth_deadline = time.time() + remaining * 0.60
        print(f"[{time.time():.1f}] smooth descent (τ:{0.05}->{0.0005})")
        overall_best_seq, overall_best_score = _smooth_descent(
            overall_best_seq,
            smooth_deadline,
            overall_best_seq,
            overall_best_score,
            tau_start=0.05,
            tau_end=0.0005,
            lr0=0.8,
            eval_every=200,
        )
        # Exact sub‑gradient polishing (remaining time)
        exact_deadline = time.time() + (deadline - time.time())
        print(f"[{time.time():.1f}] exact sub‑gradient polish")
        overall_best_seq, overall_best_score = _subgradient_descent_exact(
            overall_best_seq,
            exact_deadline,
            overall_best_seq,
            overall_best_score,
            lr0=0.9,
            eval_every=200,
        )
    else:
        print(f"[{time.time():.1f}] skipping gradient phases (time low)")

    # ------------------------------------------------------------------
    # 4. Final greedy pairwise hill‑climber
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy polishing")
        overall_best_seq, overall_best_score = _pairwise_swap(
            overall_best_seq,
            deadline,
            overall_best_seq,
            overall_best_score,
            rng,
            max_trials=None,
        )

    # ------------------------------------------------------------------
    # Reporting
    # ------------------------------------------------------------------
    print(
        f"Final best objective (2·n·max(conv)) = {overall_best_score:.6f} "
        f"with length = {len(overall_best_seq)}"
    )
    # Return as plain list – the caller will re‑normalise it.
    return overall_best_seq.tolist()
