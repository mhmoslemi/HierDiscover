"""
Improved search for a step‑function minimizing
    J(a) = 2·n·max(conv(a,a))   with a_i≥0, Σa_i=1.

The algorithm consists of three stages:

1. **Diverse seeding** – generate many candidates from analytic families
   (uniform, arcsine, beta, edge‑weighted, sine‑weighted, random sparse,
   Dirichlet) on several lengths.  Keep the best few.

2. **Hard exponentiated‑gradient descent** – minimise the exact
   max‑convolution (no smooth surrogate).  The sub‑gradient of
   `max(conv(a,a))` is obtained by a single FFT‑based convolution with a
   one‑hot (or almost‑one‑hot) mask of the active lag(s).  An
   exponentiated‑gradient update with momentum and occasional random
   jitter forces monotone progress while staying on the probability
   simplex.

3. **Refinement** – up‑sample the current best vector, continue the hard
   descent, then perform cheap pairwise‑swap and simulated‑annealing
   sweeps.

All operations respect the 1000 s budget.  The final return value is a
plain Python list of non‑negative heights that can be fed directly to
`evaluate_sequence`.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b (real FFT)."""
    n = a.shape[0]
    m = b.shape[0]
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a simplex vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric Beta(a,b)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    logw -= np.max(logw)           # stability
    w = np.exp(logw)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Hard (exact) exponentiated‑gradient descent ---------------------------
def _hard_expo_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    beta: float = 0.9,
    eval_interval: int = 200,
    patience: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Uses the sub‑gradient of max(conv(a,a)).
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()

    while time.time() < deadline and t < 600_000:
        t += 1

        # --- exact convolution and active set ---
        conv = _fft_self_convolve(a)
        M = conv.max()
        tol = max(1e-15, M * 1e-12)
        mask = (conv >= M - tol).astype(np.float64)
        if mask.sum() == 0:
            mask[conv.argmax()] = 1.0
        s = mask / mask.sum()                     # sum(s)=1, support ⊂ active lags

        # --- sub‑gradient of the max (linear combination of active gradients) ---
        a_rev = a[::-1]
        grad = 2.0 * _fft_convolve(a_rev, s)[n - 1 : n - 1 + n]

        # --- momentum & step size ---
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = lr0 / math.sqrt(t)

        # --- exponentiated‑gradient update, stay on the simplex ---
        a = a * np.exp(-eta * momentum)
        a = np.maximum(a, 1e-30)          # keep strictly positive
        a /= a.sum()

        # --- occasional evaluation & best‑tracking ---
        if t % eval_interval == 0:
            cur_score = 2.0 * n * M
            if cur_score + 1e-14 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] hard it={t:,} new best {best_score:.6f}")
            else:
                no_improve += 1

        # --- jitter on plateaus ---
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Deterministic pairwise refinement ------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Random pairwise transfer sweep."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e}  new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Cheap simulated‑annealing clean‑up -----------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Stochastic local moves with Metropolis acceptance."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e}  best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Main entry point.  Returns a list of non‑negative heights that
    (hopefully) improve on the previous best bound 1.509938.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Diverse seed generation (moderate lengths)
    lengths = [2000, 2500, 3000, 3500, 4000, 5000, 6000]
    p_grid = np.array(
        [
            0.001,
            0.005,
            0.01,
            0.02,
            0.05,
            0.1,
            0.2,
            0.5,
            1.0,
            2.0,
            5.0,
            10.0,
        ]
    )
    lam_grid = np.linspace(0.0, 0.9, 7)  # mixing with uniform

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating seeds for n={n}")

        # deterministic families
        for base in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(base)
            top_candidates.append((sc, base.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new global best (base) {sc:.6f}")

        # parameterised families
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            edge2 = _distribution_edge_inversepow(n, p)
            sine = _distribution_sine_weighted(n, p)

            for base in (beta, edge, edge2, sine):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new best (param) n={n} p={p:.3g} lam={lam:.2f} {sc:.6f}"
                        )

        # asymmetric beta
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = rng.uniform(0.001, 5.0)
            b = rng.uniform(0.001, 5.0)
            cand = _distribution_beta_asym(n, a, b)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best asym‑beta n={n} a={a:.3g} b={b:.3g} {sc:.6f}")

        # random sparse
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.004)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best sparse n={n} {sc:.6f}")

        # Dirichlet
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_dirichlet(n, rng)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best Dirichlet n={n} {sc:.6f}")

        # keep only elite seeds
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2. Incorporate previously discovered construction, if any
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback in the (unlikely) event that no seed was generated
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Hard exponentiated‑gradient descent on elite seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining * 0.6 / max(1, len(seeds))
        print(
            f"[{time.time():.1f}] hard descent on {len(seeds)} seeds, ~{per_seed:.1f}s each"
        )
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            best_seq, best_score = _hard_expo_descent(
                seed_seq.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.8,
                beta=0.9,
                eval_interval=200,
                patience=2000,
            )
            if best_score < overall_best_score:
                overall_best_score, overall_best_seq = best_score, best_seq.copy()
                print(
                    f"[{time.time():.1f}] after hard descent seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5. Upsample & continue optimisation (if time permits)
    remaining = deadline - time.time()
    if remaining > 120.0:
        target_len = min(18000, len(overall_best_seq) * 3)
        print(f"[{time.time():.1f}] up‑sampling to {target_len}")
        up_seq = _upsample(overall_best_seq, target_len)

        sub_dead = time.time() + remaining * 0.65
        up_opt, up_sc = _hard_expo_descent(
            up_seq,
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.6,
            beta=0.92,
            eval_interval=150,
            patience=2500,
        )
        if up_sc < overall_best_score:
            overall_best_score, overall_best_seq = up_sc, up_opt.copy()
            print(
                f"[{time.time():.1f}] up‑sampled hard descent new best {overall_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 6. Deterministic pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] deterministic pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 7. Final cheap simulated‑annealing clean‑up
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.01,
            decay=0.999,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list – evaluate_sequence will finish any clipping
    return overall_best_seq.tolist()
