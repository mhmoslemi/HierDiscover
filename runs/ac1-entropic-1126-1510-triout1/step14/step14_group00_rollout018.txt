def _lp_max_mass(f, M):
    import cvxpy as cp
    n = len(f)
    g = cp.Variable(n)
    conv = cp.conv(f, g)  # shape (2n-1,)
    constraints = [conv <= M, g >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
    if g.value is None:
        # fallback uniform
        return np.full(n, 1.0 / n)
    g_val = np.array(g.value).flatten()
    # ensure nonnegative
    g_val = np.maximum(g_val, 0.0)
    # Normalize to sum=1
    s = g_val.sum()
    if s == 0:
        g_val = np.full(n, 1.0 / n)
    else:
        g_val = g_val / s
    return g_val

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:

def _distribution_cauchy_like(n, scale):
    i = np.arange(n, dtype=np.float64) + 0.5
    x = (i / n) - 0.5
    w = 1.0 / (1.0 + (x/scale)**2)
    w /= w.sum()
    return w

"""
Propose a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (∑a)²

by combining several complementary ideas:

1.  A rich pool of analytic seeds (uniform, arcsine, Hann, edge‑weighted,
    Beta, Generalized‑Normal, Cauchy‑like, random Dirichlet and sparse).
2.  Fast exponentiated‑gradient polishing (mirror descent on the exact
    objective with momentum and an adaptive step‑size).
3.  A convex‑programming “mass‑augmentation” step:
    given the current vector f, solve the LP
        maximise   ∑ g_i
        subject to  f∗g ≤ M   (M = max(f∗f)) ,   g ≥ 0 .
    The optimal g is mixed back into f with a small step; this
    provably cannot increase the maximal convolution value.
4.  Progressive up‑sampling (linear interpolation) while repeatedly
    polishing with gradient descent and the LP step.
5.  A final pairwise‑mass‑transfer simulated annealing for fine‑tuning.

The routine respects a 1000 s wall‑time budget, prints progress,
and finally returns a plain Python list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (real inputs)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective for a probability vector (Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max( a ∗ a ), assumed Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families – each returns a *probability* vector (Σ = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{‑p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(‑|x‑½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) sampled on the cell mid‑points."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # safe evaluation in log‑space
    w = np.where(
        (x == 0) | (x == 1),
        0.0,
        np.exp((alpha - 1) * np.log(x) + (alpha - 1) * np.log1p(-x))
    )
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_cauchy_like(n: int, scale: float) -> np.ndarray:
    """Heavy‑tailed, symmetric Cauchy‑shaped density."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = 1.0 / (1.0 + (x / scale) ** 2)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Very sparse random vector (few non‑zeros)."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex blend with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (used by EG)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Returns (grad, k) where k is the index attaining max conv.
    grad_i = 2·a_{k‑i} for i that participates in that term,
    otherwise 0.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))

    grad = np.zeros_like(a)
    if k < n:
        # overlap indices 0 … k
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        # overlap indices k‑(n‑1) … n‑1
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
# Adaptive exponentiated‑gradient descent (mirrored gradient)
# ----------------------------------------------------------------------
def _adaptive_exp_gradient(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    beta: float = 0.9,
    eval_every: int = 50,
) -> tuple[np.ndarray, float]:
    """
    Mirror descent on the exact objective.
    Uses EMA of the sub‑gradient, an adaptive stepsize (grow on success,
    decay otherwise) and periodic evaluation.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)          # EMA of gradients
    eta = eta0
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        # EMA update
        m = beta * m + (1.0 - beta) * grad

        # multiplicative update on the simplex (mirrored step)
        a = a * np.exp(-eta * m)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                eta = min(eta0, eta * 1.02)          # reward
                print(f"[{time.time():.1f}] GD it={t:,} new best {best_score:.6f}")
            else:
                eta *= 0.995                         # decay

        # occasional re‑projection to avoid drift
        if t % (eval_every * 10) == 0:
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise mass‑transfer simulated annealing (final polish)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        src = rng.choice(np.where(a > 1e-14)[0])
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        delta = a[src] * rng.random() * 0.3
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        cur = _objective(new_a)

        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.02

        accept = False
        if cur < best_score - 1e-15:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal {src}->{dst} Δ={delta:.1e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserving total mass = 1)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new


# ----------------------------------------------------------------------
# LP step – maximize total mass of g under f∗g ≤ M
# ----------------------------------------------------------------------
def _lp_max_mass(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   maximise Σ g_i   subject to   f∗g ≤ M , g ≥ 0.
    Returns g normalised to sum = 1 (or uniform fallback).
    """
    try:
        import cvxpy as cp
    except Exception:          # cvxpy not available – fall back to uniform
        n = f.shape[0]
        return np.full(n, 1.0 / n, dtype=np.float64)

    n = f.shape[0]
    g = cp.Variable(n)
    conv_fg = cp.conv(f, g)                     # length 2n‑1
    constraints = [conv_fg <= M, g >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    # fast interior‑point; warm start helpful for repeated calls
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)

    if g.value is None:
        # Solver failed – use uniform vector
        return np.full(n, 1.0 / n, dtype=np.float64)

    g_val = np.maximum(np.asarray(g.value).flatten(), 0.0)
    s = g_val.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return g_val / s


def _lp_mixed_step(f: np.ndarray, step: float = 0.2) -> tuple[np.ndarray, float]:
    """
    One LP‑based improvement step:
        * compute M = max(f∗f)
        * solve LP for g
        * mix f←(1‑step)·f + step·g   (re‑project onto simplex)
    Returns the new vector and its objective value.
    """
    M = _fft_self_convolve(f).max()
    g = _lp_max_mass(f, M)                     # already normalised
    new_f = (1.0 - step) * f + step * g
    new_f = _project_onto_simplex(new_f, s=1.0)
    return new_f, _objective(new_f)
