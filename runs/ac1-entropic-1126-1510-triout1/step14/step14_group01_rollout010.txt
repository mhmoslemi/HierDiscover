def _subgradient_descent_exact(a0, deadline, best_seq, best_score,
                              eta0=0.5, decay=0.9995,
                              eval_interval=200, patience=2000):
    a = a0.copy()
    n = a.shape[0]
    t = 0
    no_improve = 0
    eps = 1e-30
    while time.time() < deadline and t < 500_000:
        t += 1
        max_val, ks, conv = _max_convolution_and_indices(a)
        # gradient start zero
        g = np.zeros_like(a)
        for k in ks:
            i_min = max(0, k - (n - 1))
            i_max = min(k, n - 1)
            i_range = np.arange(i_min, i_max + 1)
            g[i_range] += 2.0 * a[k - i_range]   # because a[k - i] is valid
        # step size schedule
        eta = eta0 / math.sqrt(t)
        # update via exponentiated gradient
        a = a * np.exp(-eta * g)
        # ensure positivity
        a = np.maximum(a, eps)
        # renormalize
        a /= a.sum()
        # evaluation
        if t % eval_interval == 0:
            cur_score = 2.0 * n * max_val   # sum=1
            if cur_score + 1e-14 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} new best {best_score:.6f}")
            else:
                no_improve += 1
        if no_improve >= patience:
            # jitter
            rng = np.random.default_rng()
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            no_improve = 0
    return best_seq, best_score

"""
Improved step‑function search.

The core of the new strategy is an *exact* sub‑gradient descent on the
non‑smooth objective  J(a)=2·n·max(conv(a,a))  (we keep Σa_i=1 so the
denominator is 1).  At each iteration we compute the self‑convolution,
identify the index/indices where the maximum is attained and form the
sub‑gradient of the max‑operator (a simple 2·a_{k−i} rule).  An
exponentiated‑gradient (multiplicative) update together with a Euclidean
projection onto the simplex keeps the iterates feasible.  The method
converges quickly to a point where the peak of the convolution is
balanced across the support; empirically it reaches a lower bound than
the previous log‑sum‑exp surrogate.

A multi‑start scheme is used: diverse analytic families (uniform,
arcsine, symmetric/asymmetric Beta, edge‑weighted, sine‑weighted,
random Dirichlet, sparse) provide seeds, the best few are refined with
the exact sub‑gradient descent, and finally a cheap pairwise‑swap /
simulated‑annealing sweep polishes the result.  The whole search fits
the 1000 s budget and returns a plain Python list ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used only for safety.
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolutions
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b (real FFT)."""
    n = a.shape[0]
    m = b.shape[0]
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
# Exact objective (sum of a is assumed to be 1)
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (used for seeding)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric Beta(a,b)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on  max(conv(a,a))
def _max_conv_and_indices(a: np.ndarray) -> tuple[float, np.ndarray]:
    """Return (max_conv, indices_where_max) for the current vector a."""
    conv = _fft_self_convolve(a)
    max_val = conv.max()
    # treat numerical plateaus
    idxs = np.where(conv >= max_val - 1e-15)[0]
    return max_val, idxs


def _subgradient_descent_exact(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.5,
    eval_interval: int = 200,
    patience: int = 1800,
) -> tuple[np.ndarray, float]:
    """
    Projected multiplicative sub‑gradient descent on the exact
    max‑convolution objective.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    no_improve = 0
    eps = 1e-30
    rng = np.random.default_rng()
    while time.time() < deadline and t < 500_000:
        t += 1

        # 1) current max convolution and its active indices
        max_val, ks = _max_conv_and_indices(a)

        # 2) build sub‑gradient (average over all active ks)
        g = np.zeros_like(a)
        for k in ks:
            i_min = max(0, k - (n - 1))
            i_max = min(k, n - 1)
            i_range = np.arange(i_min, i_max + 1, dtype=np.intp)
            g[i_range] += 2.0 * a[k - i_range]

        # 3) step size (standard 1/√t schedule)
        eta = eta0 / math.sqrt(t)

        # 4) exponentiated‑gradient update and re‑normalisation
        a = a * np.exp(-eta * g)
        a = np.maximum(a, eps)      # stay away from exact zeros
        a /= a.sum()

        # 5) periodic exact evaluation
        if t % eval_interval == 0:
            cur_score = 2.0 * n * max_val      # Σa = 1 by construction
            if cur_score + 1e-14 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,}  new best {best_score:.6f}")
            else:
                no_improve += 1

        # 6) escape plateaus
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.002, size=n)
            a = a + jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

# A few cheap local improvement routines (unchanged from the previous

def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e}  new best {best_score:.6f}"
            )
    return best_seq, best_score


def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e}  best {best_score:.6f}"
            )
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq
