def _smooth_max_and_grad(a, eps):
    # Compute convolution of a with itself
    conv = _fft_self_convolve(a)  # length 2n-1
    # compute softmax weights
    # Subtract max for numerical stability.
    maxc = conv.max()
    # Use parameter epsilon (positive). compute exp((conv-maxc)/eps)
    scaled = np.exp((conv - maxc) / eps)
    w = scaled / scaled.sum()
    # gradient of smooth max wrt a
    # compute grad_a = 2 * Σ_j a_j * w_{i + j}
    w_rev = w[::-1]
    # conv of a and w_rev
    # pad sizes to next power-of-two for speed
    # We'll use np.convolve (naive O(n^2) could be heavy) so use fft.

def _adam_smooth_opt(init_a, deadline, max_iters=200000, lr=0.5):

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1=0.9, beta2=0.999, eps=1e-8
    best_seq=init_a.copy()
    best_score=_objective(init_a)
    n=len(init_a)
    for t in range(1, max_iters+1):
        if time.time() > deadline: break
        a = _softmax(x)
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # eps schedule
        eps_smooth = max(maxc * 0.1, 1e-8) * (0.99**t)  # or decaying epsilon

        eps_smooth = max(maxc * 0.05, 1e-8)

        w = exp_scaled / exp_scaled.sum()
        # compute grad_a via convolution with w_rev

        # conv of a and w_rev via FFT (size as before)
        need = 3*n - 2
        size = 1 << ((need - 1).bit_length())

        Wr = np.fft.rfft(w_rev, n=size)
        conv_aw = np.fft.irfft(Af * Wr, n=size)[:need]

    conv = _fft_self_convolve(a)  # length 2n-1
    maxc = conv.max()

    scaled = np.exp((conv - maxc) / eps)   # stable: maxc subtracted

def _adam_smooth(init_a, deadline, max_iters=200000, lr0=0.5, eps_factor=0.05, eps_min=1e-8, beta1=0.9, beta2=0.999, eps=1e-8, eval_interval=500):
    n = init_a.shape[0]
    x = np.log(np.maximum(init_a, 1e-12))
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    best_seq = init_a.copy()
    best_score = _objective(init_a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        a = _softmax(x)
        # compute max and gradient
        # compute epsilon schedule: eps = max(eps_factor * maxc, eps_min)
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        eps = max(eps_factor * maxc, eps_min)
        # compute smooth weights
        scaled = np.exp((conv - maxc) / eps)
        w = scaled / scaled.sum()
        # gradient w.r.t a
        w_rev = w[::-1]
        need = 3 * n - 2
        size = 1 << ((need - 1).bit_length())
        Af = np.fft.rfft(a, n=size)
        Wf = np.fft.rfft(w_rev, n=size)
        conv_aw = np.fft.irfft(Af * Wf, n=size)[:need]
        grad_a = 2.0 * conv_aw[n-1:2*n-1][::-1]
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        eta = lr0 / math.sqrt(t)
        x -= eta * m_hat / (np.sqrt(v_hat) + eps)
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-12:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth adam it={t:,} new best {best_score:.6f}")
    best_seq = _project_onto_simplex(best_seq, 1.0)
    return best_seq, best_score

"""
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]

A three‑phase optimizer for the step‑function problem.

Phase 1 – **Broad seeding**  
    Several analytically‑motivated families (uniform, sine‑weighted,
    exponential, power‑law, beta‑shaped, Dirichlet, sparse random) are
    instantiated for a handful of lengths.  The 15 best seeds (lowest
    exact objective) are retained.

Phase 2 – **Fast mirror‑descent**  
    Each retained seed receives a short Adam run that uses the exact
    sub‑gradient of the max‑convolution (active‑set method).  The best
    result of these runs becomes the starting point for the next phase.

Phase 3 – **Smooth‑max Adam refinement**  
    The current best distribution is optimised for the remaining time
    with a smooth approximation of the max (log‑sum‑exp).  The gradient
    is computed via two FFT‑based convolutions, giving a true dense
    sub‑gradient while keeping O(n log n) cost.  A decreasing smoothing
    parameter makes the optimisation start with a gentle landscape and
    finish at the exact max.

Finally a greedy pairwise mass‑transfer and a cheap simulated‑annealing
polish are applied.  All vectors are kept on the probability simplex
so the evaluator’s normalisation is irrelevant.

The routine respects the global 1000 s budget and returns a Python list
of non‑negative heights.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Basic utilities
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (a assumed to sum to 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    w /= w.sum()
    return w


def _distribution_exponential_center(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(i - half))
    w /= w.sum()
    return w


def _distribution_exponential_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-rate * i)
    w /= w.sum()
    return w


def _distribution_exponential_right(n: int, rate: float) -> np.ndarray:
    return _distribution_exponential_left(n, rate)[::-1]


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) shape on a regular grid."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    # avoid exact 0/1 when exponent < 1
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    if w.sum() == 0.0:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator, conc: float) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a.astype(np.float64)


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

    lr0: float = 0.8,
    active_frac: float = 0.025,
    max_active: int = 30,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_interval: int = 250,
    best_seq: np.ndarray | None = None,
    best_score: float = float("inf"),
) -> tuple[np.ndarray, float]:
    """Mirror‑descent Adam using the exact max‑convolution sub‑gradient."""
    n = init_a.shape[0]
    x = np.log(np.maximum(init_a, 1e-12))
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    ind_grid = np.arange(n, dtype=np.int32)

    if best_seq is None:
        best_seq = init_a.copy()
        best_score = _objective(init_a)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        a = _softmax(x)
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            top = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top]

        grad_a = np.zeros_like(a)
        for idx_k in act:
            offset = idx_k - ind_grid
            mask = (offset >= 0) & (offset < n)
            grad_a[mask] += 2.0 * a[offset[mask]]

        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = lr0 / math.sqrt(t)
        x -= eta * m_hat / (np.sqrt(v_hat) + eps)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] adam it={t:,} new best {best_score:.6f}")

    best_seq = _project_onto_simplex(best_seq, s=1.0)
    return best_seq, best_score

# Adam with smooth max (log‑sum‑exp) – dense gradient
def _adam_smooth(
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    lr0: float = 0.7,
    eps_factor: float = 0.05,
    eps_min: float = 1e-8,
    beta1: float = 0.9,
    beta2: float = 0.999,
    adam_eps: float = 1e-8,
    eval_interval: int = 400,
    best_seq: np.ndarray | None = None,
    best_score: float = float("inf"),
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent Adam on the smooth approximation
        f_ε(a) = ε·log Σ_k exp(conv_k/ε).
    The gradient of f_ε is a dense combination of all convolution entries.
    """
    n = init_a.shape[0]
    x = np.log(np.maximum(init_a, 1e-12))
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    if best_seq is None:
        best_seq = init_a.copy()
        best_score = _objective(init_a)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        a = _softmax(x)

        # -------------------------- convolution & smooth weights ----------
        conv = _fft_self_convolve(a)                     # length 2n‑1
        maxc = conv.max()
        eps = max(eps_factor * maxc, eps_min)           # smoothing parameter
        # stable soft‑max over the convolution values
        scaled = np.exp((conv - maxc) / eps)
        w = scaled / scaled.sum()                       # shape (2n‑1,)

        # -------------------------- gradient w.r.t a -----------------------
        # grad_a = 2 * Σ_j a_j w_{i+j}  (cross‑correlation)
        w_rev = w[::-1]
        need = 3 * n - 2
        size = 1 << ((need - 1).bit_length())
        Af = rfft(a, n=size)
        Wf = rfft(w_rev, n=size)
        conv_aw = irfft(Af * Wf, n=size)[:need]
        grad_a = 2.0 * conv_aw[n - 1 : 2 * n - 1][::-1]

        # ------ transform gradient to log‑weights (mirror descent) ------
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # ----------------------------- Adam step -------------------------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = lr0 / math.sqrt(t)
        x -= eta * m_hat / (np.sqrt(v_hat) + adam_eps)

        # ---------------------- occasional exact evaluation ---------------
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-12:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] smooth‑adam it={t:,} new best {best_score:.6f}"
                )

    best_seq = _project_onto_simplex(best_seq, s=1.0)
    return best_seq, best_score

# Greedy pairwise refinement (same as reference)
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Cheap simulated‑annealing style local search
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Main entry point: returns a list of non‑negative heights."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5.0  # small safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1️⃣  Broad seeding – diverse families, several lengths
    lengths = [2000, 3000, 5000, 8000, 12000]
    rate_grid = np.linspace(0.0005, 0.02, 5)  # exponential rates
    power_grid = np.logspace(-0.5, 1.0, 5)   # power‑law exponents
    beta_params = [(0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0), (5.0, 5.0)]
    conc_grid = [0.05, 0.2, 1.0, 5.0]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        bases = [
            _distribution_uniform(n),
            _distribution_sine_weighted(n, p=1.0),
            _distribution_exponential_center(n, rate=0.005),
            _distribution_exponential_left(n, rate=0.003),
            _distribution_exponential_right(n, rate=0.003),
        ]
        for base in bases:
            sc = _objective(base)
            top_candidates.append((sc, base.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new global best (det) len={n} score={sc:.6f}")

        # power‑law and beta shapes
        for p in power_grid:
            cand = _distribution_power_law(n, p)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best power‑law p={p:.3g} score={sc:.6f}")

        for alpha, beta in beta_params:
            cand = _distribution_beta_shape(n, alpha, beta)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best beta α={alpha} β={beta} score={sc:.6f}")

        # sine‑weighted with mixing
        for p in [0.5, 1.0, 2.0]:
            sine = _distribution_sine_weighted(n, p)
            for lam in np.linspace(0.0, 0.5, 4):
                cand = _mix_with_uniform(sine, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}] new best mix sine p={p} lam={lam:.2f} score={sc:.6f}"
                    )

        # exponentials
        for rate in rate_grid:
            for dist in (
                _distribution_exponential_center,
                _distribution_exponential_left,
                _distribution_exponential_right,
            ):
                cand = dist(n, rate)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new best exp rate={rate:.5f} score={sc:.6f}")

        # Dirichlet
        for conc in conc_grid:
            for _ in range(2):
                cand = _distribution_dirichlet(n, rng, conc)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new best Dirichlet conc={conc} score={sc:.6f}")

        # sparse random
        for _ in range(2):
            cand = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best sparse len={n} score={sc:.6f}")

        # keep only the very best seeds (global across lengths)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:15]

    # ------------------------------------------------------------------
    # Include a previously found construction if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            top_candidates.append((sc, prev.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.sort(key=lambda x: x[0])
            top_candidates = top_candidates[:15]
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2️⃣  Short Adam runs (active‑set) for each retained seed
    refined: list[tuple[float, np.ndarray]] = []
    for sc, cand in top_candidates:
        if time.time() >= deadline:
            break
        slice_deadline = time.time() + min(12.0, (deadline - time.time()) * 0.04)
        best_seq, best_sc = _adam_refine(
            cand.copy(),
            slice_deadline,
            max_iters=8000,
            lr0=0.7,
            active_frac=0.03,
            max_active=20,
            eval_interval=120,
            best_seq=cand,
            best_score=sc,
        )
        refined.append((best_sc, best_seq))

    if refined:
        overall_best_score, overall_best_seq = min(refined, key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 3️⃣  Long smooth‑max Adam descent from the current best
    if time.time() < deadline:
        remaining = deadline - time.time()
        if remaining > 40.0:
            long_deadline = time.time() + remaining - 20.0
            print(f"[{time.time():.1f}] starting long smooth‑Adam ({remaining-20:.1f}s left)")
            overall_best_seq, overall_best_score = _adam_smooth(
                overall_best_seq.copy(),
                long_deadline,
                max_iters=400_000,
                lr0=0.9,
                eps_factor=0.04,
                eps_min=1e-8,
                eval_interval=300,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
            )
            print(f"[{time.time():.1f}] after smooth‑Adam best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 5️⃣  Final simulated‑annealing polishing
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing polishing")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.015,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best objective ≈ {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # The evaluator re‑normalises, but we keep the vector on the simplex.
    return overall_best_seq.tolist()
