"""
Proposes a non‑negative step function that (hopefully) improves the
objective 2·n·max(conv(a,a))/sum(a)^2.

The search proceeds in several stages:

1. **Broad seed generation** – a large, diverse pool of analytic
   “window” families (uniform, Hann, Hamming, Blackman, Kaiser,
   arcsine, beta, edge‑weighted, power‑law, generalized‑normal,
   sparse Dirichlet, random combs, mixtures).  All seeds are normalised
   to a probability vector and their exact objective is evaluated.
   Only the best few (≈20) are retained.

2. **Temperature‑smoothed Adam optimisation** – each retained seed is
   refined by minimising the smooth surrogate
   `2·n·τ·log‑sum‑exp(conv/τ)`.  The temperature τ is annealed from 0.2
   down to 1e‑3.  The best result of all runs becomes the current global
   best.

3. **Projected sub‑gradient descent on the true max‑convex** – starting
   from the best candidate we directly minimise the exact objective
   using sub‑gradients of the active convolution entry and projection
   onto the simplex.  A diminishing stepsize ηₜ=η₀/√t is used.

4. **Iterated up‑sampling** – the current best vector is repeatedly
   interpolated to double its length (as long as ≤ 64000).  After each
   up‑sample a short Adam burst is performed, and the result is kept
   if it improves the objective.

5. **Final polishing** – a short exponentiated‑gradient descent followed
   by a stochastic pairwise‑annealing move finishes the optimisation.

All stages respect the global 1000 s budget and periodically report the
current best objective.  The returned list is ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# --------------------------------------------------------------
# Simplex projection (used after sub‑gradient steps)
# --------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# FFT based convolution utilities
# --------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Cross‑correlation of a with w for non‑negative shifts (length n)."""
    n = a.shape[0]
    m = w.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()
    rev_w = w[::-1]
    A = rfft(a, n=size)
    B = rfft(rev_w, n=size)
    conv = irfft(A * B, n=size)
    start = m - 1
    return conv[start:start + n]


# --------------------------------------------------------------
# Exact objective used by ``evaluate_sequence`` (scale‑invariant)
# --------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a (unnormalised)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# --------------------------------------------------------------
# Analytic families – always return a probability vector (sum = 1)
# --------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w /= w.sum()
    return w


def _distribution_hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    w /= w.sum()
    return w


def _distribution_blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0
         - a1 * np.cos(2.0 * np.pi * i / (n - 1))
         + a2 * np.cos(4.0 * np.pi * i / (n - 1)))
    w /= w.sum()
    return w


def _distribution_kaiser(n: int, beta: float) -> np.ndarray:
    # fallback if scipy is unavailable
    try:
        from scipy.special import i0
    except Exception:  # pragma: no cover
        # simple approximation: fall back to Hann
        return _distribution_hann(n)
    i = np.arange(n, dtype=np.float64) + 0.5
    x = (i / n) - 0.5
    w = i0(beta * np.sqrt(1 - (2 * x) ** 2))
    w /= w.sum()
    return w


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w /= w.sum()
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_random_dirichlet(n: int, rng: np.random.Generator,
                                  alpha: float) -> np.ndarray:
    v = rng.dirichlet(np.full(n, alpha))
    return v


def _distribution_comb(n: int, m: int) -> np.ndarray:
    """Place equal mass on m equally‑spaced bins (a simple comb)."""
    if m <= 0:
        m = 1
    idx = np.linspace(0, n - 1, m, dtype=int)
    v = np.zeros(n, dtype=np.float64)
    v[idx] = 1.0 / m
    return v


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex mix of a distribution with the uniform one."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# --------------------------------------------------------------
# Sub‑gradient of the active convolution entry (used by EG & FW)
# --------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Compute the sub‑gradient of the currently maximal convolution entry.
    Returns (grad, active_index).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))
    grad = np.zeros_like(a)
    if k < n:
        # i ranges 0..k
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        # tail part: i ranges k-(n-1) .. n-1
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# --------------------------------------------------------------
# Adam optimisation of a temperature‑smoothed version of the objective
# --------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float,
                   best_seq: np.ndarray, best_score: float,
                   tau_start: float = 0.2, tau_end: float = 0.001,
                   tau_decay: float = 0.999,
                   eval_every: int = 200,
                   step_size0: float = 0.05) -> tuple[np.ndarray, float]:
    """Adam on the smoothed objective ``2·n·τ·log‑sum‑exp(conv/τ)``."""
    n = a0.shape[0]
    x = np.log(a0 + 1e-15)               # logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    tau = tau_start
    while time.time() < deadline and tau > tau_end:
        t += 1
        # softmax probabilities
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        exp_vals = np.exp((conv - max_conv) / tau)
        w = exp_vals / exp_vals.sum()           # soft‑max over convolution entries

        # gradient w.r.t. a (4·n·corr)
        corr = _fft_corr(a, w)                  # length n
        grad_a = 4.0 * n * corr

        # chain rule to logits (mirror descent)
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = step_size0 / (np.sqrt(v_hat) + eps)
        x = x - lr * m_hat

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} τ={tau:.6f} new best {best_score:.6f}")

        tau = max(tau * tau_decay, tau_end)

    return best_seq, best_score


# --------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convex objective
# --------------------------------------------------------------
def _subgradient_descent(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float,
                         eta0: float = 0.5,
                         eval_every: int = 80) -> tuple[np.ndarray, float]:
    """Plain sub‑gradient descent on the true max‑conv (projected onto simplex)."""
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a - eta * grad
        # projection onto simplex (non‑negative + unit sum)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / a.shape[0])
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Sub‑grad it={t} new best {best_score:.6f}")
    return best_seq, best_score


# --------------------------------------------------------------
# Exponentiated‑gradient descent (multiplicative mirror descent)
# --------------------------------------------------------------
def _exp_gradient_descent(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float,
                         eta0: float = 0.8,
                         eval_every: int = 80) -> tuple[np.ndarray, float]:
    """Exact exponentiated‑gradient descent on the objective."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new best {best_score:.6f}")
    return best_seq, best_score


# --------------------------------------------------------------
# Pairwise‑annealing polishing (simulated annealing style)
# --------------------------------------------------------------
def _pairwise_annealing(a: np.ndarray, deadline: float,
                        best_seq: np.ndarray, best_score: float,
                        rng: np.random.Generator,
                        max_trials: int | None = None) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# --------------------------------------------------------------
# Linear up‑sampling while preserving the simplex
# --------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the total budget and returns a plain list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Diverse seed generation
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192, 16384]
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] seeding length {n}")

        # deterministic families
        families = [
            _distribution_uniform,
            _distribution_hann,
            _distribution_hamming,
            _distribution_blackman,
            _distribution_arcsine,
        ]

        for func in families:
            cand = func(n)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best (det) len={n} score={sc:.6f}")

        # Kaiser (if scipy available)
        for beta in (0.5, 2.0, 5.0, 8.0):
            cand = _distribution_kaiser(n, beta)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best Kaiser β={beta:.1f} len={n} score={sc:.6f}")

        # beta and edge‑weighted families
        for p in np.logspace(-0.6, 0.6, 8):
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            for base in (beta, edge):
                for lam in (0.0, 0.2, 0.5, 0.8, 1.0):
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(f"[{time.time():.1f}] new best p={p:.3g} lam={lam:.2f} len={n} score={sc:.6f}")

        # generalized normal
        for q in (0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
            cand = _distribution_generalized_normal(n, q)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best gn q={q:.2f} len={n} score={sc:.6f}")

        # power‑law
        for pwr in (0.3, 0.6, 1.0, 1.5, 2.0):
            cand = _distribution_power_law(n, pwr)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best power‑law p={pwr:.2f} len={n} score={sc:.6f}")

        # random Dirichlet
        for _ in range(8):
            alpha = rng.uniform(0.01, 3.0)
            cand = _distribution_random_dirichlet(n, rng, alpha)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best Dirichlet α={alpha:.3f} len={n} score={sc:.6f}")

        # sparse random
        for density in (0.01, 0.03, 0.05):
            for _ in range(4):
                cand = _distribution_random_sparse(n, rng, density)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(f"[{time.time():.1f}] new best sparse dens={density:.2%} len={n} score={sc:.6f}")

        # simple combs of various sizes
        for m in (max(2, n // 8), max(2, n // 5), max(2, n // 3)):
            cand = _distribution_comb(n, m)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best comb m={m} len={n} score={sc:.6f}")

        # prune to keep the most promising (global) candidates
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:20]

    # ------------------------------------------------------------------
    # 2. Include a previously discovered construction if present
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            top_candidates.append((sc, prev.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Guard against empty result
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 4096
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Adam smoothing on top seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # keep the 8 best seeds for the expensive Adam stage
        seed_seqs = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:8]]
        per_seed = remaining * 0.5 / max(1, len(seed_seqs))
        print(f"[{time.time():.1f}] Adam smoothing on {len(seed_seqs)} seeds")
        for idx, seed_seq in enumerate(seed_seqs):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _adam_optimize(
                seed_seq.copy(),
                sub_deadline,
                overall_best_seq,
                overall_best_score,
                tau_start=0.2,
                tau_end=0.001,
                tau_decay=0.999,
                eval_every=200,
                step_size0=0.05,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] after Adam seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Projected sub‑gradient descent on the exact objective
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 12.0:
        sub_deadline = time.time() + remaining * 0.4
        overall_best_seq, overall_best_score = _subgradient_descent(
            overall_best_seq.copy(),
            sub_deadline,
            overall_best_seq,
            overall_best_score,
            eta0=0.5,
            eval_every=80,
        )
        print(f"[{time.time():.1f}] after sub‑grad best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Exact exponentiated‑gradient descent (fine‑tune)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 8.0:
        sub_deadline = time.time() + remaining * 0.3
        overall_best_seq, overall_best_score = _exp_gradient_descent(
            overall_best_seq.copy(),
            sub_deadline,
            overall_best_seq,
            overall_best_score,
            eta0=0.8,
            eval_every=80,
        )
        print(f"[{time.time():.1f}] after EG best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Iterated up‑sampling + short Adam bursts
    # ------------------------------------------------------------------
    while time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n > 64000:
            break
        # allocate a modest slice of remaining time for each up‑sample step
        slice_time = min(0.12 * (deadline - time.time()), 20.0)
        if slice_time <= 1.0:
            break
        up_seq = _upsample(overall_best_seq, target_n)
        up_score = _objective(up_seq)
        if up_score < overall_best_score:
            overall_best_seq = up_seq
            overall_best_score = up_score
            print(f"[{time.time():.1f}] upsample improved to {overall_best_score:.6f}")

        sub_deadline = time.time() + slice_time
        overall_best_seq, overall_best_score = _adam_optimize(
            overall_best_seq.copy(),
            sub_deadline,
            overall_best_seq,
            overall_best_score,
            tau_start=0.1,
            tau_end=0.001,
            tau_decay=0.998,
            eval_every=120,
            step_size0=0.04,
        )
        print(f"[{time.time():.1f}] post‑upsample Adam best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final pairwise annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing polish")
        overall_best_seq, overall_best_score = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            overall_best_seq,
            overall_best_score,
            rng,
            max_trials=None,
        )
        print(f"[{time.time():.1f}] after final anneal best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Report and return
    # ------------------------------------------------------------------
    print(
        f"[{time.time():.1f}] Final best score: {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
