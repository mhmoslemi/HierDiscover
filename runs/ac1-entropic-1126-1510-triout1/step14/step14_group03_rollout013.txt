def _project_onto_simplex(v):
    # Ensure sum = 1 and nonnegative
    # Using algorithm sorted v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _smooth_descent(a, deadline, best_seq, best_score, init_beta=1.0, max_beta=50.0, beta_mult=1.02, eta0=0.1, eval_every=30):
    t = 0
    beta = init_beta
    while time.time() < deadline:
        t += 1
        # compute convolution
        conv = _fft_self_convolve(a)
        M = conv.max()
        # compute softmax weights focusing on near max
        # shift by M to avoid overflow; exponent factor beta
        exp_vals = np.exp(beta * (conv - M))
        w = exp_vals / exp_vals.sum()
        # compute gradient
        grad = _grad_weighted(a, w)
        # step size schedule
        eta = eta0 / math.sqrt(t)
        # gradient descent update (project onto simplex)
        a_new = a - eta * grad
        # project onto simplex
        a_new = _project_onto_simplex(a_new)
        a = a_new

        # optionally increase beta
        if beta < max_beta:
            beta = min(max_beta, beta * beta_mult)

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-12:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth it={t} beta={beta:.2f} new best {best_score:.6f}")
    return best_seq, best_score

"""
Propose a step function (list of non‑negative heights) that minimises
``evaluate_sequence``.  The search consists of three stages

1.  Generate a diverse pool of seeds (uniform, arcsine, edge‑weighted,
    sum‑edge‑weighted, generalised‑normal and random Dirichlet vectors).
2.  Refine the current best seed with a *smoothed* exponentiated‑gradient
    descent.  Instead of using the sub‑gradient of only the maximal
    convolution entry we form a soft‑max weighting of all entries;
    this yields a smoother descent direction that can escape shallow
    local minima.
3.  A short pair‑wise mass‑transfer simulated‑annealing polish.

All work respects the 1000 s wall‑time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  FFT‑based self‑convolution (full linear convolution)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
#  Exact objective used by the evaluator (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a))/sum(a)^2 , a is a raw (non‑negative) vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
#  Analytic families – all return a *probability* vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i·(n‑i))^{-p}   with i = distance to the left edge."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_sum_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ i^{-p} + (n‑i)^{-p} (sum of two one‑sided tails)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (-p)) + (j ** (-p))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1‑lam) + uniform·lam."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
#  Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Gradient of a *soft‑max* weighted convolution entry
# ----------------------------------------------------------------------
def _grad_weighted_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a  Σ_k w_k (a*a)_k   using FFT.
    a  – vector of length n
    w  – vector of length 2n‑1 (weights, non‑negative, sum = 1)
    Returns gradient of length n.
    """
    n = a.shape[0]
    # Pad to length sufficient for a (n)  *  w_rev (2n‑1)
    size = 1 << ((3 * n - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    w_rev = w[::-1]
    W = np.fft.rfft(w_rev, n=size)
    conv_rev = np.fft.irfft(A * W, n=size)
    # slice indices n‑1 … 2n‑2 (inclusive) -> length n
    slice_part = conv_rev[n - 1 : 2 * n - 1]
    grad = 2.0 * slice_part[::-1]
    return grad


# ----------------------------------------------------------------------
#  Smoothed exponentiated‑gradient descent
# ----------------------------------------------------------------------
def _smooth_exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_beta: float = 1.0,
    max_beta: float = 30.0,
    beta_mult: float = 1.02,
    eta0: float = 0.08,
    eval_every: int = 60,
) -> tuple[np.ndarray, float]:
    """
    Perform exponentiated‑gradient descent on a *soft‑max* approximation
    of the max‑convolution objective.
    """
    n = a.shape[0]
    t = 0
    beta = init_beta

    while time.time() < deadline:
        t += 1

        # --- compute full convolution -------------------------------------------------
        size_conv = 1 << ((2 * n - 1) - 1).bit_length()
        A = np.fft.rfft(a, n=size_conv)
        conv = np.fft.irfft(A * A, n=size_conv)[: 2 * n - 1]

        # --- soft‑max weights ---------------------------------------------------------
        M = conv.max()
        # shift by M to avoid overflow, clamp exponent for safety
        exp_vals = np.exp(beta * (conv - M))
        w = exp_vals / exp_vals.sum()

        # --- gradient of the weighted sum --------------------------------------------
        grad = _grad_weighted_fft(a, w)

        # --- exponentiated update ----------------------------------------------------
        eta = eta0 / math.sqrt(t)
        # clip to keep the exponent well‑behaved
        grad_clip = np.clip(grad, -50.0, 50.0)
        a = a * np.exp(-eta * grad_clip)

        # normalise back to a probability vector
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # --- increase the temperature (beta) slowly ---------------------------------
        if beta < max_beta:
            beta = min(max_beta, beta * beta_mult)

        # --- occasional exact evaluation ---------------------------------------------
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth‑EG it={t} beta={beta:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
#  Pairwise‑mass‑transfer annealing (final polish)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins; accept improvements."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.4  # move at most 40 % of source mass
        if delta < 1e-16:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01
        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)
        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Returns a plain ``list[float]`` ready for the evaluator.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1.  Generate a pool of seeds (moderate length, power‑of‑two for FFT)
    # ------------------------------------------------------------------
    n = 8192                         # power of two → fast FFTs
    seeds = []

    # deterministic families
    seeds.append(_distribution_uniform(n))
    seeds.append(_distribution_arcsine(n))

    # edge‑weighted family (product of distances)
    for p in np.logspace(-2, 0.5, 20):   # focus on the most promising range
        base = _distribution_edge_weighted(n, float(p))
        seeds.append(base)
        for lam in np.linspace(0.0, 0.5, 6):
            seeds.append(_mix_with_uniform(base, lam))

    # sum‑edge‑weighted family (new ingredient)
    for p in np.logspace(-2, 0.5, 20):
        base = _distribution_edge_sum_weighted(n, float(p))
        seeds.append(base)
        for lam in np.linspace(0.0, 0.5, 6):
            seeds.append(_mix_with_uniform(base, lam))

    # generalized normal
    for q in np.linspace(0.5, 2.5, 10):
        seeds.append(_distribution_generalized_normal(n, float(q)))

    # random sparse Dirichlet seeds (few non‑zero entries)
    for _ in range(30):
        alpha = rng.uniform(0.02, 0.2, size=n)
        vec = rng.gamma(alpha, 1.0)
        if vec.sum() == 0.0:
            continue
        vec /= vec.sum()
        seeds.append(vec)

    # ------------------------------------------------------------------
    # 2.  Evaluate them and keep the best few
    # ------------------------------------------------------------------
    scored = [( _objective(s), s.copy() ) for s in seeds]
    scored.sort(key=lambda x: x[0])
    top = [vec for (_, vec) in scored[:12]]
    best_seq = top[0].copy()
    best_score = scored[0][0]
    print(f"[{time.time():.1f}] initial best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Include the previously submitted construction if it exists
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < best_score:
                best_score, best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best improved to {best_score:.6f}")
            top.append(prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4.  Smoothed exponentiated‑gradient descent (main optimisation)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = deadline * 0.6 + time.time()   # about 60 % of the remaining budget
        print(f"[{time.time():.1f}] smooth‑EG phase (budget {sub_deadline - time.time():.1f}s)")
        best_seq, best_score = _smooth_exp_gradient_descent(
            best_seq.copy(),
            sub_deadline,
            best_seq,
            best_score,
            init_beta=1.0,
            max_beta=30.0,
            beta_mult=1.02,
            eta0=0.08,
            eval_every=70,
        )
        print(f"[{time.time():.1f}] after smooth‑EG best {best_score:.6f}")

        # a second run from the second‑best seed if time permits
        if time.time() < deadline and len(top) > 1:
            second = top[1].copy()
            sub_deadline2 = deadline - 5.0   # leave a few seconds for the final anneal
            best_seq2, best_score2 = _smooth_exp_gradient_descent(
                second,
                sub_deadline2,
                best_seq,
                best_score,
                init_beta=1.0,
                max_beta=30.0,
                beta_mult=1.02,
                eta0=0.08,
                eval_every=70,
            )
            if best_score2 < best_score:
                best_seq, best_score = best_seq2, best_score2
                print(f"[{time.time():.1f}] second seed gave improvement {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final pairwise annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing polish")
        best_seq, best_score = _pairwise_annealing(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            rng,
        )
        print(f"[{time.time():.1f}] after annealing best {best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective = 2·n·max(conv)/sum²): {best_score:.6f} "
        f"(length={len(best_seq)})"
    )
    return best_seq.tolist()
