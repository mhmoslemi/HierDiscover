"""
Propose a candidate step sequence using a diversified set of analytic families,
annealed exponentiated‑gradient descent on a smooth log‑sum‑exp surrogate,
progressive up‑sampling and aggressive deterministic/annealed local refinements.
The algorithm is tuned to push the bound below the previous 1.509938 target.

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolutions (full linear convolution)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Return full linear convolution a*a using real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return full linear convolution a*b using real FFT."""
    n = a.shape[0]
    m = b.shape[0]
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
# Exact objective used for bookkeeping (sum of a is 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv(a,a)) for a simplex vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families of candidate densities (all return a probability vector)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric Beta(a,b)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    logw -= np.max(logw)          # numerical stabilisation
    w = np.exp(logw)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential_edges(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential decay from each endpoint."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_piecewise_linear(n: int, rng: np.random.Generator,
                                   segments: int = 4) -> np.ndarray:
    """Random piecewise‑linear density with a given number of linear pieces."""
    xs = np.sort(rng.uniform(0.0, 1.0, size=segments - 1))
    xs = np.concatenate(([0.0], xs, [1.0]))
    heights = rng.random(segments)
    # build linear interpolation on the grid
    grid = (np.arange(n, dtype=np.float64) + 0.5) / n
    vals = np.empty_like(grid)
    for k in range(segments):
        mask = (grid >= xs[k]) & (grid <= xs[k + 1])
        if xs[k + 1] - xs[k] > 0:
            slope = (heights[k + 1] - heights[k]) / (xs[k + 1] - xs[k])
            vals[mask] = heights[k] + slope * (grid[mask] - xs[k])
        else:
            vals[mask] = heights[k]
    vals = np.maximum(vals, 0.0)
    if vals.sum() == 0.0:
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.006) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Return lam·U + (1‑lam)·base where U is the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smooth log‑sum‑exp surrogate and its gradient
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, lam: float) -> tuple[np.ndarray, float]:
    """
    Returns (grad, max_conv) where
        grad = ∇_a φ_λ(a)   (exact gradient of the smooth surrogate)
        max_conv = max(conv(a,a))   (exact max for monitoring)
    The vector `a` is assumed to be a probability vector (sum=1).
    """
    n = a.shape[0]

    # 1) self‑convolution
    conv = _fft_self_convolve(a)               # length 2n‑1
    max_c = conv.max()

    # 2) softmax over λ·conv  (log‑sum‑exp trick)
    z = lam * conv
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    s = exp_z / exp_z.sum()                    # probability vector on length 2n‑1

    # 3) gradient = 2 * (a_rev * s) restricted to centre block
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, s)        # length 3n‑2
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]   # slice of length n
    return grad, max_c


# ----------------------------------------------------------------------
# Annealed exponentiated‑gradient descent on the smooth surrogate
# ----------------------------------------------------------------------
def _smooth_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    lam_start_factor: float = 0.08,
    lam_inc: float = 1.0018,
    beta: float = 0.9,
    eval_interval: int = 150,
    patience: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Perform exponentiated‑gradient descent on the smooth surrogate.
    Returns the best sequence and its exact score discovered during the run.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    lam = lam_start_factor * n          # initial λ
    rng = np.random.default_rng()

    # loop until deadline or a very large iteration count
    while time.time() < deadline and t < 2_000_000:
        t += 1
        grad, max_conv = _smooth_gradient(a, lam)

        # momentum (exponential moving average)
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size schedule (diminishing)
        eta = lr0 / math.sqrt(t)

        # multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-eta * momentum)

        # keep numerics healthy
        a = np.maximum(a, 1e-30)
        a /= a.sum()

        # anneal λ (sharpen surrogate)
        lam *= lam_inc

        # periodic exact evaluation
        if t % eval_interval == 0:
            cur_score = 2.0 * n * max_conv
            if cur_score + 1e-14 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,}  new best {best_score:.6f}")
            else:
                no_improve += 1

        # escape plateaus with a tiny jitter
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.0025, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Deterministic pairwise mass‑transfer refinement
# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Systematically move a small amount of mass from one index to another."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35   # up to 35 % of the source weight
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e}  new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Cheap simulated‑annealing clean‑up (additive moves)
# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.009,
    decay: float = 0.9993,
) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing based hill‑climber."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e}  best {best_score:.6f}")
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling of a simplex vector
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq

    # 6.   Final simulated‑annealing clean‑up
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9994,
        )

    # ------------------------------------------------------------------
    # 7.   Optional convex combination of top candidates (convexity can help)
    # ------------------------------------------------------------------
    if len(top_candidates) >= 3 and time.time() < deadline:
        # Take the three best distinct candidates and average them
        best_three = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:3]]
        avg_seq = sum(best_three) / len(best_three)
        avg_seq = np.maximum(avg_seq, 0.0)
        avg_seq /= avg_seq.sum()
        avg_score = _objective(avg_seq)
        if avg_score < overall_best_score:
            overall_best_score, overall_best_seq = avg_score, avg_seq.copy()
            print(f"[{time.time():.1f}] convex‑average improvement {overall_best_score:.6f}")

        # one last short descent from the averaged point
        sub_dead = time.time() + min(30.0, deadline - time.time())
        overall_best_seq, overall_best_score = _smooth_descent(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.45,
            lam_start_factor=0.15,
            lam_inc=1.0012,
            beta=0.92,
            eval_interval=80,
            patience=1200,
        )

    # ------------------------------------------------------------------
    # 8.   Wrap‑up
    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Convert to plain Python list – evaluate_sequence will clip if needed
    return overall_best_seq.tolist()
