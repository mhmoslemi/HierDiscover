def _adam_smooth_descent(a0, time_limit, best_seq, best_score):
    a = a0.copy()
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1  # base learning rate
    start = time.time()
    it = 0
    while time.time() < time_limit:
        it += 1
        # compute conv and smooth max weights
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
        # anneal tau linearly like before
        frac = (time.time() - start) / max(time_limit - start, 1e-9)
        tau = (0.02 * (1-frac) + 0.001 * frac)
        # compute softmax weights
        shifted = conv - max_val
        exp_vals = np.exp(shifted / tau)
        w = exp_vals / exp_vals.sum()
        # gradient
        term1 = _fft_cross_correlate_valid(w, a)
        win_sum = _fft_sliding_window_sum(w, n)
        grad = term1 + a * win_sum
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # ensure nonnegative, reproject onto simplex
        a = np.maximum(a, 0.0)
        sum_a = a.sum()
        if sum_a == 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / sum_a
    return best_seq, best_score

    2·n·max(conv(a,a))

where conv is the full linear convolution.  The algorithm runs under a
1000 s budget and consists of:

1️⃣  Exhaustive scan of a few analytic families (uniform, arcsine,

    performed (soft‑max temperature τ annealed from 2·10⁻² → 10⁻³).  The
    exact gradient of the smoothed objective is obtained via FFT‑based
    cross‑correlation.  After the smooth phase a short exact
    sub‑gradient EG polish is applied.

3️⃣  A final simulated‑annealing hill‑climber moves small amounts of
    mass between random coordinates; worse moves are occasionally
    accepted with a decreasing temperature, allowing the search to
    escape shallow basins.

All steps respect the overall time budget and the best sequence found

import time
import math

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (real FFT)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_cross_correlate_valid(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (valid part):
    (w ⋆ a)[j] = Σ_i w[i+j]·a[i]   (0 ≤ j < len(a)).
    """
    n = a.shape[0]
    m = w.shape[0]  # = 2·n‑1
    size = 1 << ((m + n - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A = np.fft.rfft(a, n=size)
    corr = np.fft.irfft(W * np.conj(A), n=size)
    return corr[:n]


def _fft_sliding_window_sum(w: np.ndarray, win: int) -> np.ndarray:
    """
    For each position j compute Σ_{k=j}^{j+win‑1} w[k] (window length = win).
    Returns an array of length win (the same length as a).
    """
    m = w.shape[0]
    size = 1 << ((m + win - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    ones = np.ones(win, dtype=np.float64)
    O = np.fft.rfft(ones, n=size)
    conv = np.fft.irfft(W * O, n=size)
    start = win - 1
    return conv[start:start + win]


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_edge(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta‑type density w_i ∝ (i·(n‑i))^{α‑1}, α>0."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = (i * (n - i)) ** (alpha - 1.0)
    vals = np.clip(vals, 0.0, None)
    s = vals.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    vals /= s
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    w /= w_sum
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    """Symmetric triangle, peak in the middle."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    w = 1.0 - np.abs(i - mid) / mid
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_inverse_triangular(n: int) -> np.ndarray:
    """U‑shaped density – large at the edges, small in the centre."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    w = np.abs(i - mid) / mid
    w = np.clip(w, 1e-12, None)   # avoid exact zero
    w /= w.sum()
    return w


def _distribution_two_peaks(n: int, frac: float) -> np.ndarray:
    """
    Two symmetric flat peaks of relative width ``frac``·n,
    each carrying half of the total mass.
    """
    w = np.zeros(n, dtype=np.float64)
    peak_len = max(2, int(frac * n))
    if peak_len * 2 > n:
        peak_len = n // 2
    w[:peak_len] = 0.5 / peak_len
    w[-peak_len:] = 0.5 / peak_len
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse Dirichlet‑like seed with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Exact objective (sum = 1 ⇒ 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smoothed‑max Adam descent
# ----------------------------------------------------------------------
def _adam_smooth_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    tau_start: float = 0.02,
    tau_end: float = 0.001,
    lr0: float = 0.15,
    beta1: float = 0.9,
    beta2: float = 0.999,
) -> tuple[np.ndarray, float]:
    """Adam descent on the soft‑max approximation of the L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)         # first moment
    v = np.zeros_like(a)         # second moment
    eps = 1e-8
    start = time.time()
    it = 0
    while time.time() < time_limit:
        it += 1
        # compute current exact objective – keep track of global best
        conv = _fft_self_convolve(a)
        cur_val = conv.max()
        cur_score = 2.0 * n * cur_val
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time():.1f}] adam it={it} new global best {best_score:.6f}")

        # anneal τ linearly
        frac = (time.time() - start) / max(1e-12, time_limit - start)
        tau = tau_start * (1.0 - frac) + tau_end * frac

        # soft‑max weights (stable)
        maxc = conv.max()
        shifted = conv - maxc                 # ≤ 0
        exp_vals = np.exp(shifted / tau)     # ≤ 1
        w = exp_vals / exp_vals.sum()        # length 2n‑1, sums to 1

        # gradient of the smooth objective
        term1 = _fft_cross_correlate_valid(w, a)          # Σ w[i+j]·a[i]
        win_sum = _fft_sliding_window_sum(w, n)          # Σ_{k=j}^{j+n‑1} w[k]
        grad = term1 + a * win_sum                        # shape (n,)

        # Adam update (descent → subtract step)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = lr0 / (np.sqrt(v_hat) + eps)
        a = a - eta * m_hat

        # project onto the probability simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
    return best_seq, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient EG descent (kept from previous implementation)
# ----------------------------------------------------------------------
def _subgradient_descent_exact(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] exact subgrad it={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climber
# ----------------------------------------------------------------------
def _hill_climb_simulated(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Random mass‑transfer moves with occasional uphill acceptance."""
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    # temperature schedule (simple exponential decay)
    start = time.time()
    while time.time() < time_limit:
        # pick source (must have some mass) and destination
        src_indices = np.where(a > 1e-14)[0]
        if src_indices.size == 0:
            break
        i = rng.choice(src_indices)
        j = rng.integers(0, n)
        if j == i:
            continue

        # transfer a random fraction (up to 30 %) of source mass
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-15:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = _objective(a_new)

        # accept if better, otherwise with probability exp(-Δ/T)
        elapsed = time.time() - start
        frac = elapsed / max(1e-12, time_limit - start)
        T = (0.5 * (1.0 - frac) + 0.001 * frac)   # temperature decays to near‑zero
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score, best_seq = new_score, a.copy()
                print(
                    f"[{time.time():.1f}] hill‑climb accept i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a step‑function (plain Python list) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # keep the overall best solution
    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, seq)

    # ------------------------------------------------------------------
    # 1. Scan analytic families on a handful of lengths
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 6000, 8000, 12000]

    # grids for parameters
    p_grid = np.concatenate(
        [
            np.logspace(-3, -0.5, 12),
            np.logspace(-0.5, 0.0, 8),
            np.linspace(0.1, 3.0, 8),
        ]
    )
    alpha_grid = np.linspace(0.07, 0.45, 8)
    lam_grid = np.linspace(0.0, 1.0, 15)
    c_grid = np.logspace(-3, 0, 10)
    peak_frac_grid = [0.05, 0.10, 0.20]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families (no extra parameters)
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
            _distribution_triangular,
            _distribution_inverse_triangular,
        ):
            cand = base_fn(n)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # two‑peak family
        for frac in peak_frac_grid:
            cand = _distribution_two_peaks(n, frac)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] len={n} two‑peak frac={frac:.2f} score={sc:.6f}"
                )

        # beta‑edge (α) families with uniform mixing
        for alpha in alpha_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_edge(n, float(alpha))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(beta, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}] len={n} β α={alpha:.3g} lam={lam:.2f} score={sc:.6f}"
                    )

        # edge‑weighted families (p) with uniform mix
        for p in p_grid:
            if time.time() >= deadline:
                break
            edge = _distribution_edge_weighted(n, float(p))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(edge, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}] len={n} edge p={p:.4g} lam={lam:.2f} score={sc:.6f}"
                    )

        # exponential‑edge families
        for c in c_grid:
            if time.time() >= deadline:
                break
            exp_edge = _distribution_exponential_edge(n, float(c))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(exp_edge, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}] len={n} exp c={c:.4g} lam={lam:.2f} score={sc:.6f}"
                    )

        # random Dirichlet seeds (few)
        for _ in range(6):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

        # keep a trimmed list of the most promising
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:80]

    # ------------------------------------------------------------------
    # 2. Try to incorporate a previously found construction (if any)
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / s
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never happen)
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(
            f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 4. Adam‑based smooth descent on the best seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # select a handful of seeds (5 best) for deep optimisation
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] smooth Adam descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            # apply Adam smooth descent – allocate most of the time slice
            sg_deadline = time.time() + per_seed * 0.65
            a_opt, sc_opt = _adam_smooth_descent(
                seed_seq,
                sg_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                tau_start=0.02,
                tau_end=0.001,
                lr0=0.15,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after Adam seed {idx} new global best {overall_best_score:.6f}"
                )
            # short exact sub‑gradient polish
            sg_deadline = time.time() + per_seed * 0.35
            a_opt, sc_opt = _subgradient_descent_exact(
                overall_best_seq,
                sg_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                eval_every=400,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after exact polish seed {idx} new best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5. Final simulated‑annealing hill‑climber
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final hill‑climb phase")
        a_opt, sc_opt = _hill_climb_simulated(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(
                f"[{time.time():.1f}] after hill‑climb new best {overall_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    print(
        f"Final best objective (2·n·max(conv)) = {overall_best_score:.6f}, length = {len(overall_best_seq)}"
    )
    return overall_best_seq.tolist()
