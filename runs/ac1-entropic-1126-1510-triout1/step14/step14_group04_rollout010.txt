Hybrid optimiser for the step‑function attack.

The algorithm follows four stages:

1️⃣  Analytic and random seed generation.
    – a wide range of parameterised families (uniform, arcsine, sine,
      cosine, quadratic, edge‑weighted, exponential‑edge, beta,

    – an Adam‑style optimiser with Euclidean projection onto the simplex

      bounded by the global time limit).

    – For the current best vector `a`, compute `M = max(conv(a,a))`.
    – Solve a linear programme
          maximise   Σ g_i
          subject to  a * g ≤ M   (convolution, linear in g)
                      g_i ≥ 0 .
      The solution `g` is normalised to a probability vector.
    – If `g` improves the exact objective it replaces `a`,
      otherwise we mix the two vectors.
    – The step is repeated a few times (the LP is built with a sparse
      Toeplitz matrix, solved by ``scipy.optimize.linprog``).

4️⃣  Final greedy polishing.
    – a simple hill‑climber that moves a random fraction of mass
      between two positions and keeps the move only if the exact
      objective strictly decreases.

All stages respect the 1000 s wall‑clock budget; the function finally
returns the best simplex‑normalised sequence as a plain ``list[float]``.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
#  Geometry helpers
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the probability simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
#  FFT based convolution (real inputs)
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
#  Exact objective (assumes Σ a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a on the simplex (Σ a = 1)."""
    n = a.shape[0]
    return float(2.0 * n * _fft_self_convolve(a).max())


# ----------------------------------------------------------------------
#  Analytic families (all return a probability vector)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    w /= w_sum
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    total = w.sum()
    if total == 0.0:
        return _distribution_uniform(n)
    w /= total
    return w


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
#  Random seed factories
# ----------------------------------------------------------------------
def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.01) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


# ----------------------------------------------------------------------
#  Smoothed‑max mirror descent (Adam + β‑continuation)
# ----------------------------------------------------------------------
def _smooth_max_descent(a0: np.ndarray,
                        time_limit: float,
                        best_seq: np.ndarray,
                        best_score: float) -> tuple[np.ndarray, float]:
    """Mirror descent on the log‑sum‑exp surrogate of max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    # β schedule (logarithmic up‑to a fairly large value)
    beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0]

    for beta in beta_schedule:
        # Run as many iterations as time permits for this β
        while time.time() < time_limit:
            t += 1
            conv = _fft_self_convolve(a)          # length 2n‑1
            max_val = conv.max()
            # softmax weights (stable)
            shifted = conv - max_val
            exp_beta = np.exp(beta * shifted)
            w = exp_beta / exp_beta.sum()         # Σ w = 1
            # gradient of the smooth max term
            a_rev = a[::-1]
            grad_conv = _fft_convolve(w, a_rev)   # length 3n‑2
            grad = 2.0 * grad_conv[n - 1:n - 1 + n]   # size n
            grad_J = 2.0 * n * grad               # gradient of J = 2·n·smooth_max

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_J
            v = beta2 * v + (1.0 - beta2) * (grad_J ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = 0.2 / (np.sqrt(v_hat) + eps)   # a slightly larger base step
            a -= step * m_hat
            a = _project_onto_simplex(a, s=1.0)

            # periodic exact evaluation
            if t % 200 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-14:
                    best_score, best_seq = cur, a.copy()
                    print(f"[{time.time():.1f}] β={beta:.1f} it={t} new best {best_score:.6f}")

            # Safety stop for the current β – we give the algorithm
            # enough room but also limit the overall runtime.
            if t % 4000 == 0:
                break
        if time.time() >= time_limit:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Exact sub‑gradient descent (exponentiated gradient)
# ----------------------------------------------------------------------
def _subgradient_descent_exact(a0: np.ndarray,
                               time_limit: float,
                               best_seq: np.ndarray,
                               best_score: float,
                               lr0: float = 0.9,
                               eval_every: int = 100) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true L∞‑objective."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        # exact sub‑gradient
        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score, best_seq = cur, a.copy()
                print(f"[{time.time():.1f}] exact subgrad it={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
#  LP‑based refinement (alternating linear programme)
# ----------------------------------------------------------------------
def _build_toeplitz_matrix(f: np.ndarray) -> sp.csr_matrix:
    """Return the (2n‑1)×n matrix A with (Af)_k = (f * g)_k (linear in g)."""
    n = f.shape[0]
    I = np.arange(n, dtype=np.int64)
    rows = (I[:, None] + I[None, :]).ravel()
    cols = np.tile(I, n)
    data = np.repeat(f, n)
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_refine(a0: np.ndarray,
               time_limit: float,
               best_seq: np.ndarray,
               best_score: float,
               max_iters: int = 10,
               alpha: float = 0.5) -> tuple[np.ndarray, float]:
    """Alternate linear programme (max Σg) with mixing."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    while time.time() < time_limit and it < max_iters:
        conv = _fft_self_convolve(a)
        M = conv.max()

        # Build constraint matrix for the linear programme a * g ≤ M
        A = _build_toeplitz_matrix(a)
        b = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)                      # maximise Σg   ⇔ minimise –Σg
        bounds = [(0, None)] * n

        # Solve with HiGHS (fast interior‑point)
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            print(f"[{time.time():.1f}] LP failed (status={res.message}) – aborting refinement")
            break

        g = res.x
        g_sum = g.sum()
        if g_sum <= 0.0:
            break
        g_norm = g / g_sum

        # Evaluate the pure LP solution
        obj_g = _objective(g_norm)
        if obj_g < best_score - 1e-14:
            a = g_norm
            best_score, best_seq = obj_g, a.copy()
            print(f"[{time.time():.1f}] LP iter {it} pure solution improves to {best_score:.6f}")
        else:
            # Otherwise mix the two vectors
            a = (1.0 - alpha) * a + alpha * g_norm
            a = _project_onto_simplex(a, s=1.0)
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score, best_seq = cur, a.copy()
                print(f"[{time.time():.1f}] LP iter {it} mixed solution improves to {best_score:.6f}")

        it += 1
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Greedy pairwise mass‑transfer hill‑climber
# ----------------------------------------------------------------------
def _pairwise_swap(a0: np.ndarray,
                   time_limit: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   max_trials: int | None = None) -> tuple[np.ndarray, float]:
    """Randomly relocate a fraction of mass; accept only strict improvements."""
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score, best_seq = cur, a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a step‑function (plain list) that minimises ``evaluate_sequence``.
    The routine follows the four‑stage hybrid scheme described in the
    module docstring.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a tiny safety margin for the host environment
    deadline = start + budget_s - 1.0

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, vector)

    # ------------------------------------------------------------------
    # 1️⃣ Analytic families + permutations + random seeds
    # ------------------------------------------------------------------
    lengths = [1200, 1500, 2000, 2500, 3000, 3500, 4000]
    # parameter grids
    p_grid = np.concatenate([
        np.logspace(-3, -0.5, 13),
        np.logspace(-0.5, 0.0, 11),
        np.linspace(0.2, 3.0, 14)
    ])
    lam_grid = np.linspace(0.0, 1.0, 11)
    c_grid = np.logspace(-3, 0, 10)
    alpha_grid = np.logspace(-0.5, 0.7, 12)
    sigma_grid = np.linspace(0.1, 0.5, 9) * np.mean([1200, 4000])  # absolute sigma for Gaussian

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # Deterministic families
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
        ):
            cand = base_fn(n)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

            # also try a few random permutations of the same shape
            for _ in range(3):
                perm = rng.permutation(n)
                cand_perm = cand[perm]
                scp = _objective(cand_perm)
                top_candidates.append((scp, cand_perm.copy()))
                if scp < overall_best_score:
                    overall_best_score, overall_best_seq = scp, cand_perm.copy()
                    print(f"[{time.time():.1f}] permuted best len={n} score={scp:.6f}")

        # Beta‑type families
        for alpha in alpha_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_beta(n, float(alpha), float(alpha))
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] beta α={alpha:.4g} len={n} score={sc:.6f}")

        # Edge‑weighted + uniform mixes
        for p in p_grid:
            if time.time() >= deadline:
                break
            edge = _distribution_edge_weighted(n, float(p))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(edge, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] edge p={p:.4g} lam={lam:.2f} len={n} score={sc:.6f}")

        # Exponential‑edge family
        for c in c_grid:
            if time.time() >= deadline:
                break
            exp_edge = _distribution_exponential_edge(n, float(c))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(exp_edge, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] exp c={c:.4g} lam={lam:.2f} len={n} score={sc:.6f}")

        # Gaussian family (σ scaled with n)
        for sigma_frac in np.linspace(0.05, 0.30, 6):
            sigma = sigma_frac * n
            cand = _distribution_gaussian(n, sigma)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] gaussian σ={sigma:.1f} len={n} score={sc:.6f}")

        # Dirichlet random seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 3.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} score={sc:.6f}")

        # Sparse random seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

        # keep only a bounded number of best seeds (memory & time)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:40]

    # ------------------------------------------------------------------
    #  2️⃣ Smoothed‑max mirror descent on a few promising seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seeds = [vec for _, vec in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] smoothed‑max descent on {len(seeds)} seeds, {per_seed:.1f}s each")
        for idx, seed_vec in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_max_descent(
                seed_vec,
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after smooth seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    #  3️⃣ LP‑based alternating refinement (if time permits)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0 and overall_best_seq is not None:
        print(f"[{time.time():.1f}] LP‑based refinement (≈{int(remaining)} s left)")
        a_opt, sc_opt = _lp_refine(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_iters=8,
            alpha=0.4,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after LP refinement new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    #  4️⃣ Exact sub‑gradient refinement (optional, budget permitting)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 15.0:
        print(f"[{time.time():.1f}] exact sub‑gradient refinement")
        a_opt, sc_opt = _subgradient_descent_exact(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.9,
            eval_every=120,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after exact subgrad new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    #  5️⃣ Final greedy polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy pairwise polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    print(f"FINAL best = {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # ``evaluate_sequence`` rescales internally, so returning the raw simplex‑normalised
    # vector is fine.
    return overall_best_seq.tolist()
