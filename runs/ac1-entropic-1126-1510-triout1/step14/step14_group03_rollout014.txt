def _fft_convolve(a,b):
    size = 1 << ((len(a)+len(b)-1)-1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A*B, n=size)
    return conv[:len(a)+len(b)-1]

def _lbfgs_refine(a0, deadline):
    n = a0.shape[0]
    # start with a0 already normalized sum=1.
    # Use softmax var: we can set z0 = np.log(a0 + epsilon)
    epsilon = 1e-12
    z0 = np.log(a0 + epsilon)
    # define function f(z) returns (obj, grad_z) where obj = smoothed max (maybe scaled) also compute actual objective.
    def f(z):
        a = np.exp(z - np.max(z))
        a_sum = a.sum()
        a = a / a_sum  # though sum should be 1 anyway but stable.
        # compute smoothed max and gradient.
        S, grad_a = _smoothed_max_and_grad(a, alpha_current)  # uses current alpha.
        # compute gradient wrt z:
        grad_z = _grad_z_from_a_grad(a, grad_a)
        # For objective: using actual max conv (not smoothed):
        conv = _fft_self_convolve(a)
        actual_obj = 2.0 * n * np.max(conv)
        # Return value as actual_obj? but gradient corresponds to S (smoothed) not actual; L-BFGS expects gradient of the function we return.
        # We could minimize S directly but we want to minimize actual max. However gradient of actual max is subgradient at max location; L-BFGS may not converge properly. Use smoothed S, which approximates actual. So we ignore actual_obj for gradient.
        return S, grad_z

    # We'll iterate across multiple alpha values gradually increasing.
    # For each alpha value we run minimize with some max_iter.
    best_a = a0.copy()
    best_obj = 2.0 * n * np.max(_fft_self_convolve(best_a))

    # schedule alphas: [1.0, 3.0, 10.0, 30.0, 100.0] maybe.
    alphas = [1.0, 3.0, 10.0, 30.0, 100.0]
    for alpha_current in alphas:
        if time.time() >= deadline:
            break
        # call minimize
        res = minimize(fun=lambda z: f(z), x0=z0, method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 80, 'ftol': 1e-12, 'gtol': 1e-8})
        z_opt = res.x
        a_opt = np.exp(z_opt - np.max(z_opt))
        a_opt /= a_opt.sum()
        # compute actual objective
        cur_obj = 2.0 * n * np.max(_fft_self_convolve(a_opt))
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a_opt.copy()
            print(f"[{time.time():.1f}] L-BFGS alpha={alpha_current} new best {best_obj:.6f}")
        # Set starting point for next stage:
        z0 = z_opt

    return best_a, best_obj

def _lbfgs_refine(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    n = a0.shape[0]
    # Setup z0 = np.log(a0 + epsilon)
    epsilon = 1e-12
    z = np.log(a0 + epsilon)
    best_seq = a0.copy()
    best_score = 2.0 * n * np.max(_fft_self_convolve(a0))
    # schedule alphas
    alphas = [0.5, 1.0, 3.0, 10.0, 30.0, 100.0]
    for alpha in alphas:
        if time.time() >= deadline:
            break
        # define closure capturing alpha & n
        def func_and_grad(z):
            a = np.exp(z - np.max(z))
            a_sum = a.sum()
            if a_sum == 0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / a_sum
            # smoothed max and grad
            conv = _fft_self_convolve(a)  # length 2n-1
            # compute smoothed max
            x = alpha * conv
            max_x = np.max(x)
            exp_x = np.exp(x - max_x)
            Z = exp_x.sum()
            S = (max_x + np.log(Z)) / alpha
            w = exp_x / Z  # length 2n-1
            # gradient wrt a:
            # compute w_rev conv
            w_rev = w[::-1]
            conv_rev = _fft_convolve(a, w_rev)  # length 3n-2
            grad_a = 2.0 * conv_rev[(2*n-2) - np.arange(n)]
            # gradient wrt z:
            mu = np.dot(a, grad_a)
            grad_z = a * (grad_a - mu)
            return S, grad_z

        # optimize using L-BFGS-B
        # Use options to limit function evaluations per alpha
        try:
            res = minimize(fun=func_and_grad, x0=z, method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 100, 'ftol': 1e-12})
        except Exception as e:
            print(f"LBFGS failed with {e}")
            break
        z = res.x
        a = np.exp(z - np.max(z))
        a /= a.sum()
        # compute actual objective
        cur_score = 2.0 * n * np.max(_fft_self_convolve(a))
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] LBFGS alpha={alpha} new best {best_score:.6f}")

    return best_seq, best_score

def _lbfgs_obj(z, n, alpha):
    # softmax
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)
    a = exp_z / exp_z.sum()
    # compute convolution of a with itself
    conv = _fft_self_convolve(a)  # length 2n-1
    # smoothed max:
    x = alpha * conv
    max_x = np.max(x)
    exp_x = np.exp(x - max_x)
    Z = exp_x.sum()
    S = (max_x + np.log(Z)) / alpha
    # weights:
    w = exp_x / Z  # shape 2n-1
    # gradient wrt a:
    w_rev = w[::-1]
    conv_rev = _fft_convolve(a, w_rev)  # length 3n-2
    grad_a = 2.0 * conv_rev[(2*n-2) - np.arange(n)]
    # gradient wrt z:
    mu = np.dot(a, grad_a)
    grad_z = a * (grad_a - mu)
    return S, grad_z

This implementation searches for a non‑negative vector `a` that minimises
``evaluate_sequence``.  The workflow is:

1. **Seed generation** – a modest pool (≈ 30) of analytic candidates is
   built: uniform, arcsine, edge‑weighted, edge‑sum‑weighted and several
   mixtures with the uniform distribution.  Sparse Dirichlet‑type

   minimising a *smooth* surrogate of the non‑smooth term
   `max(conv(a,a))`.  Let `α` be a temperature; the surrogate is
   `S(a)= (1/α)·log∑exp(α·(a*a))`.  `S` is differentiable and its
   gradient is obtained analytically using FFT‑based convolutions.
   The variables are re‑parameterised as `a = softmax(z)` so that
   non‑negativity and `∑a=1` are satisfied automatically.
   A schedule of increasing `α` (0.5 → 100) drives the optimiser towards
   the true maximum.  The optimisation is performed with L‑BFGS‑B
   (SciPy) which needs only a few dozen iterations and thus fits easily
   into the 1000 s budget.

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution of a real vector using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.shape[0] + b.shape[0] - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.shape[0] + b.shape[0] - 1]

#  Exact objective used by the evaluator (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a))/sum(a)², a is a raw (non‑negative) vector."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (s * s)

def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i·(n‑i))^{‑p}   (i = distance to left edge)."""

def _distribution_edge_sum_weighted(n: int, p: float) -> np.ndarray:

    i = np.arange(n, dtype=np.float64) + 0.5

    """Convex combination  (1‑lam)·base  +  lam·uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
#  Smoothed‑max objective (log‑sum‑exp) and its gradient in z‑space
# ----------------------------------------------------------------------
def _lbfgs_obj(z: np.ndarray, n: int, alpha: float):
    """
    Returns (value, gradient) for the smoothed objective

        S(z) = 1/α * log Σ_i exp( α·conv_i(a) ),   a = softmax(z).

    The gradient w.r.t. z is obtained via the chain rule.
    """
    # ----- a = softmax(z) --------------------------------------------
    mx = np.max(z)
    exp_z = np.exp(z - mx)
    a = exp_z / exp_z.sum()               # Σ a = 1, a_i ≥ 0

    # ----- convolution a*a -------------------------------------------
    conv = _fft_self_convolve(a)           # length 2n‑1

    # ----- smoothed max (log‑sum‑exp) -------------------------------
    x = alpha * conv
    mx_x = np.max(x)
    exp_x = np.exp(x - mx_x)
    Z = exp_x.sum()
    S = (mx_x + np.log(Z)) / alpha        # scalar

    # ----- weights w_i = ∂S/∂conv_i ---------------------------------
    w = exp_x / Z                         # shape (2n‑1,)

    # ----- grad_a = ∂S/∂a  (using convolution) ----------------------
    #   ∂S/∂a_j = Σ_i w_i (a_{i‑j} + a_{j‑i}) = 2 Σ_i w_i a_{i‑j}
    w_rev = w[::-1]
    conv_rev = _fft_convolve(a, w_rev)    # length 3n‑2
    grad_a = 2.0 * conv_rev[(2 * n - 2) - np.arange(n)]

    # ----- grad_z = ∂S/∂z using ∂a/∂z = a (I – 1·aᵀ) --------------------
    mu = np.dot(a, grad_a)
    grad_z = a * (grad_a - mu)            # shape (n,)

    return S, grad_z


# ----------------------------------------------------------------------
#  L‑BFGS refinement with increasing smoothing parameter α
# ----------------------------------------------------------------------
def _lbfgs_refine(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Starts from a normalised probability vector `a0` and runs a short
    L‑BFGS‑B optimisation of the smoothed‑max surrogate.  The schedule
    of α values moves from very smooth (α≈0.5) to nearly the true max
    (α≈100).  Returns the best vector found and its exact objective.
    """
    n = a0.shape[0]
    eps = 1e-12
    z = np.log(a0 + eps)                  # initialise z = log‑softmax
    best_seq = a0.copy()
    best_score = _objective(best_seq)

    # increasing temperatures – the last value approximates the true max
    alphas = [0.5, 1.0, 3.0, 10.0, 30.0, 100.0]

    for alpha in alphas:
        if time.time() >= deadline:
            break
        res = minimize(fun=_lbfgs_obj,
                       x0=z,
                       args=(n, alpha),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 80, 'ftol': 1e-12})
        z = res.x
        a = np.exp(z - np.max(z))
        a /= a.sum()
        cur_score = _objective(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] L‑BFGS α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
#  Pairwise‑mass‑transfer annealing (final polish)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins; accept if
    it improves or with a temperature‑dependent probability."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4      # move up to 40 % of source mass
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function (list of heights) that minimises
    ``evaluate_sequence``.  Returns a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")
    global_top: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Build a diverse set of seeds (moderate sizes to stay fast)
    # ------------------------------------------------------------------
    lengths = [3000, 5000, 8000, 12000]
    # exponents for edge families
    p_grid = np.array([0.05, 0.1, 0.2, 0.4, 0.8, 1.2, 2.0, 3.0])
    lam_grid = np.array([0.0, 0.25, 0.5, 0.75, 1.0])

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        cand = _distribution_uniform(n)
        sc = _objective(cand)
        global_top.append((sc, cand.copy()))
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] new global best uniform ({n}) {sc:.6f}")

        cand = _distribution_arcsine(n)
        sc = _objective(cand)
        global_top.append((sc, cand.copy()))
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] new global best arcsine ({n}) {sc:.6f}")

        # edge‑weighted families + mixes
        for p in p_grid:
            if time.time() >= deadline:
                break

            base = _distribution_edge_weighted(n, float(p))
            sc = _objective(base)
            global_top.append((sc, base.copy()))
            if sc < best_score:
                best_score, best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new best prod p={p:.3g} {sc:.6f}")

            for lam in lam_grid:
                cand = _mix_with_uniform(base, float(lam))
                sc = _objective(cand)
                global_top.append((sc, cand.copy()))
                if sc < best_score:
                    best_score, best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new best prod p={p:.3g} lam={lam:.2f} {sc:.6f}")

            # sum‑edge weighted family (new ingredient)
            base = _distribution_edge_sum_weighted(n, float(p))
            sc = _objective(base)
            global_top.append((sc, base.copy()))
            if sc < best_score:
                best_score, best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new best sum p={p:.3g} {sc:.6f}")

            for lam in lam_grid:
                cand = _mix_with_uniform(base, float(lam))
                sc = _objective(cand)
                global_top.append((sc, cand.copy()))
                if sc < best_score:
                    best_score, best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new best sum p={p:.3g} lam={lam:.2f} {sc:.6f}")

        # a few random sparse Dirichlet seeds
        for _ in range(5):
            if time.time() >= deadline:
                break
            k = max(2, int(n * 0.01))
            idx = rng.choice(n, size=k, replace=False)
            vals = rng.random(k)
            cand = np.zeros(n, dtype=np.float64)
            cand[idx] = vals
            cand /= cand.sum()
            sc = _objective(cand)
            global_top.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best sparse {sc:.6f}")

        # keep only the very best seeds (size 30) to limit memory/time
        global_top.sort(key=lambda x: x[0])
        global_top = global_top[:30]

    # ------------------------------------------------------------------
    # 2.  Include the previously submitted construction if present
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, 0.0)
            if prev.sum() > 0.0:
                prev = prev / prev.sum()
                sc = _objective(prev)
                global_top.append((sc, prev.copy()))
                if sc < best_score:
                    best_score, best_seq = sc, prev.copy()
                    print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Guard against an empty result
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _distribution_uniform(5000)
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  L‑BFGS refinement on the (up to) 10 best seeds
    # ------------------------------------------------------------------
    top_k = min(10, len(global_top))
    for idx in range(top_k):
        if time.time() >= deadline:
            break
        cand = global_top[idx][1]
        # normalise again for safety
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        refined, refined_score = _lbfgs_refine(cand, deadline)
        if refined_score < best_score - 1e-14:
            best_score, best_seq = refined_score, refined.copy()
            print(f"[{time.time():.1f}] after L‑BFGS seed {idx} new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final simulated‑annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing")
        best_seq, best_score = _pairwise_annealing(
            best_seq.copy(),
            deadline,
            best_seq.copy(),
            best_score,
            rng,
        )
        print(f"[{time.time():.1f}] after annealing best {best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective = 2·n·max(conv)/sum²): {best_score:.6f} "
        f"(length={len(best_seq)})"
    )
    return best_seq.tolist()
