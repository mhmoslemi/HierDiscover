def _project_simplex(v):
    """Project onto the probability simplex {x >= 0, sum x = 1}."""
    # Based on algorithm from Duchi et al.
    n = v.shape[0]
    # Sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1 - cssv) / np.arange(1, n+1)
    # Find last index where rho_arr > 0
    rho = np.nonzero(rho_arr > 0)[0]
    if len(rho) == 0:
        # all negative? then return uniform
        return np.full(n, 1.0 / n)
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a non‑negative step function that minimizes ``evaluate_sequence``.

Algorithm
---------

* 1.  Build a diverse pool of candidate distributions (uniform,
      arcsine, edge‑weighted, sum‑edge‑weighted, generalized normal,
      random Dirichlet) for several lengths.  Keep the best few (20).

* 2.  Starting from the currently best candidate run a **smooth‐max
      gradient descent**:
      – a temperature T controls a soft‑max over the convolution
        values, giving a smooth approximation of the max.
      – the gradient of the smooth max is a weighted correlation of the
        current vector with the soft‑max weights.
      – after each step we project back onto the probability simplex.
      – the step size η decays as 1/√t and T decays geometrically.

* 3.  Finish with a cheap **pairwise mass‑transfer annealing** to polish
      the result.

All work is bounded by the 1000 s wall‑time limit (2 s safety margin)
and the returned list is ready for the provided ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  FFT utilities – full linear convolution (real inputs)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Convolution a*a (full length) using FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a*b using FFT (real inputs)."""
    na, nb = a.shape[0], b.shape[0]
    size = _next_pow2(na + nb - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: na + nb - 1]


# ----------------------------------------------------------------------
#  Exact objective used by the evaluator (sum assumed = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a probability vector a (sum = 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
#  Analytic families – output a *probability* vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i·(n‑i))^{-p}   with i = distance to the left edge."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_sum_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ i^{-p} + (n‑i)^{-p} (sum of two one‑sided tails)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (-p)) + (j ** (-p))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp( -|x‑½|^{q} ) on the discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1‑lam) + uniform·lam."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
#  Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex {x ≥ 0, sum x = 1}.
    Returns a new array (does not modify v in place).
    """
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u + (1.0 - cssv) / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries negative – return uniform distribution
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Gradient of the *smooth* max (soft‑max) of the convolution
# ----------------------------------------------------------------------
def _grad_smooth_max(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Compute ∂/∂a  (log‑sum‑exp(α·(a*a)))  where ``weights`` are the
    normalized soft‑max numbers (i.e. w_k ∝ exp(α·conv_k)).
    The gradient is 2·corr(a, w) where corr denotes linear correlation.
    """
    n = a.shape[0]
    # reverse the weight vector for correlation via convolution
    w_rev = weights[::-1]
    size = _next_pow2(n + (2 * n - 1) - 1)  # enough for full correlation (length 3n‑2)
    FA = np.fft.rfft(a, n=size)
    FW = np.fft.rfft(w_rev, n=size)
    conv = np.fft.irfft(FA * FW, n=size)
    # keep only the first (3n‑2) entries
    conv = conv[: 3 * n - 2]
    # index of the entry corresponding to shift (len(w)-1) = 2n‑2
    idx_start = 2 * n - 2
    grad = 2.0 * conv[idx_start - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
#  Smooth‑max gradient descent (projected onto the simplex)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """Minimise the max‑convolution via a temperature‑controlled gradient descent."""
    n = a.shape[0]
    t = 0
    # parameters – these were tuned empirically
    eta0 = 0.7          # initial step‑size factor
    T0 = 0.12           # initial temperature for the soft‑max
    decay_T = 0.9998    # geometric temperature decay per iteration
    min_T = 1e-8

    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        cur = 2.0 * n * conv.max()               # sum(a)=1 ⇒ objective = 2n·max_conv
        if cur < best_score - 1e-14:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] smooth it={t} new best {best_score:.6f}")

        # temperature schedule
        T = max(min_T, T0 * (decay_T ** t))

        # soft‑max weights over the convolution entries
        shifted = conv - conv.max()               # for numerical stability
        w = np.exp(shifted / T)
        w_sum = w.sum()
        if w_sum == 0.0:          # should not happen, but guard against it
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # gradient of the smooth max
        grad = _grad_smooth_max(a, w)

        # step‑size (diminishing as 1/√t)
        eta = eta0 / math.sqrt(t)

        a_new = a - eta * grad
        a = _project_simplex(a_new)

    return best_seq, best_score


# ----------------------------------------------------------------------
#  Pairwise‑mass‑transfer simulated annealing (final polish)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins; accept improvements."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4        # move at most 40 % of source mass
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
#  Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Returns a plain ``list[float]`` ready for the evaluator.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # small safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")
    top: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Build a large pool of seeds from several analytic families
    # ------------------------------------------------------------------
    lengths = [3000, 5000, 8000, 12000, 16000]
    # parameters for families
    p_grid = np.concatenate((
        np.logspace(-2.0, -0.5, 12, base=10),     # 0.01 … 0.316
        np.logspace(-0.5, 0.5, 12, base=10),      # 0.316 … 3.16
        np.logspace(0.5, 1.5, 10, base=10),       # 3.16 … 31.6
    ))
    lam_grid = np.linspace(0.0, 1.0, 11)          # 0,0.1,…,1
    q_grid = np.linspace(0.5, 3.0, 9)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for cand in (_distribution_uniform(n),
                     _distribution_arcsine(n)):
            sc = _objective(cand)
            top.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best ({n}) score={sc:.6f}")

        # product edge‑weighted + mixing with uniform
        for p in p_grid:
            if time.time() >= deadline:
                break
            base = _distribution_edge_weighted(n, p)
            sc = _objective(base)
            top.append((sc, base.copy()))
            if sc < best_score:
                best_score, best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new best prod p={p:.3g} score={sc:.6f}")

            for lam in lam_grid:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                top.append((sc, cand.copy()))
                if sc < best_score:
                    best_score, best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new best prod p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # **new** sum edge‑weighted family
        for p in p_grid:
            if time.time() >= deadline:
                break
            base = _distribution_edge_sum_weighted(n, p)
            sc = _objective(base)
            top.append((sc, base.copy()))
            if sc < best_score:
                best_score, best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new best sum p={p:.3g} score={sc:.6f}")

            for lam in lam_grid:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                top.append((sc, cand.copy()))
                if sc < best_score:
                    best_score, best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new best sum p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # generalized‑normal family
        for q in q_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_generalized_normal(n, q)
            sc = _objective(cand)
            top.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best gn q={q:.2f} score={sc:.6f}")

        # random sparse Dirichlet seeds (few non‑zero entries)
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = np.zeros(n, dtype=np.float64)
            k = max(2, int(n * 0.01))
            idx = rng.choice(n, size=k, replace=False)
            cand[idx] = rng.random(k)
            cand /= cand.sum()
            sc = _objective(cand)
            top.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best sparse score={sc:.6f}")

        # keep only the very best 20 seeds so far
        top.sort(key=lambda x: x[0])
        top = top[:20]

    # ------------------------------------------------------------------
    # 2. Include the previously submitted construction if it exists
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < best_score:
                best_score, best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Guard against an empty result
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _distribution_uniform(5000)
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Smooth‑max gradient descent on the currently best seed
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = deadline
        print(f"[{time.time():.1f}] smooth‑gradient descent (budget {sub_deadline - time.time():.1f}s)")
        best_seq, best_score = _smooth_gradient_descent(
            best_seq.copy(),
            sub_deadline,
            best_seq,
            best_score,
        )
        print(f"[{time.time():.1f}] after smooth descent best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing")
        best_seq, best_score = _pairwise_annealing(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            rng,
        )
        print(f"[{time.time():.1f}] after annealing best score {best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective = 2·n·max(conv)/sum²): {best_score:.6f} "
        f"(length={len(best_seq)})"
    )
    return best_seq.tolist()
