def _lp_update(f, M, t, active_frac, deadline):
    # compute conv_f
    conv_f = _fft_self_convolve(f)
    thresh = M * (1 - active_frac)
    active_idxs = np.where(conv_f >= thresh)[0]
    # Ensure at least one constraint
    if active_idxs.size == 0:
        active_idxs = np.array([np.argmax(conv_f)])
        M = conv_f[active_idxs[0]]
    # Build A_ub matrix
    rows = []
    for k in active_idxs:
        idx = k - np.arange(len(f))
        mask = (idx >= 0) & (idx < len(f))
        row = np.zeros_like(f)
        row[mask] = f[idx[mask]]
        rows.append(row)
    A = np.vstack(rows)  # shape m x n

def _lp_step(f, M, t, active_frac, deadline):
    n = len(f)
    conv_f = _fft_self_convolve(f)  # length 2n-1
    thresh = M * (1 - active_frac)
    active_idxs = np.where(conv_f >= thresh)[0]
    if active_idxs.size == 0:
        active_idxs = np.array([np.argmax(conv_f)], dtype=np.int64)
        M = conv_f[active_idxs[0]]
    # Build constraints matrix
    A = _build_constraints(f, active_idxs)
    b = np.full(A.shape[0], M, dtype=np.float64)
    # objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')

"""
step‑function optimizer (LP + Adam)

The algorithm proceeds in three stages:

1️⃣  **Seeding** –   a variety of analytic families (uniform, sinusoidal,
    centre‑exponential, left/right‑exponential, Beta, Dirichlet,
    sparse random) are instantiated for several lengths.  The 8 best
    candidates (lowest exact objective) are retained.

2️⃣  **Linear‑programming refinement** –   each retained candidate is
    iteratively improved by solving the LP described in the statement:
    maximise Σ g_j  subject to  g≥0  and  (f * g)_k ≤ M,
    where M = max(f * f).  The LP is solved only on the active
    convolution indices (those within a fraction of the current maximum),
    which makes each step cheap (≈10 constraints).  After obtaining g we
    mix it with the current f (t≈0.2, shrinking when the mixture would
    increase the maximal autocorrelation).  The active‑set tolerance is
    exponentially decreased so that the optimisation gradually focuses
    on the true peak.

3️⃣  **Polishing** –   the best LP‑refined vector is fed to a short Adam
    mirror‑descent run (soft‑max parametrisation) together with a final
    greedy pairwise mass‑transfer and a cheap simulated‑annealing sweep.

All stages respect the global 1000 s budget.  The returned list is a
proper probability vector (sum≈1) and can be passed directly to
``evaluate_sequence``.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# basic utilities -------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))  (a is assumed to sum to 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# analytic families (all return a probability vector – sum = 1) -----
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    w /= w.sum()
    return w


def _distribution_exponential_center(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(i - half))
    w /= w.sum()
    return w


def _distribution_exponential_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-rate * i)
    w /= w.sum()
    return w


def _distribution_exponential_right(n: int, rate: float) -> np.ndarray:
    return _distribution_exponential_left(n, rate)[::-1]


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.power(x, max(alpha - 1.0, 0.0)) * np.power(1.0 - x, max(beta - 1.0, 0.0))
    w[w < 0] = 0.0
    s = w.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    w /= s
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator, conc: float) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a.astype(np.float64)


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Adam‑based mirror descent (exact objective) -------------------------
def _adam_refine(
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    lr0: float = 0.8,
    active_frac: float = 0.025,
    max_active: int = 30,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_interval: int = 250,
    best_seq: np.ndarray | None = None,
    best_score: float = float("inf"),
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact objective using the soft‑max
    parametrisation (log‑weights → probability simplex).
    """
    n = init_a.shape[0]
    x = np.log(np.maximum(init_a, 1e-12))
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    ind_grid = np.arange(n, dtype=np.int32)
    t = 0
    if best_seq is None:
        best_seq = init_a.copy()
        best_score = _objective(init_a)

    while time.time() < deadline and t < max_iters:
        t += 1
        a = _softmax(x)
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            top = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top]

        # sub‑gradient of the max (using the active set)
        grad_a = np.zeros_like(a)
        for idx_k in act:
            offset = idx_k - ind_grid
            mask = (offset >= 0) & (offset < n)
            grad_a[mask] += 2.0 * a[offset[mask]]

        # convert gradient w.r.t. a into gradient w.r.t. log‑weights x
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update with a decaying step‑size
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = lr0 / math.sqrt(t)
        x -= eta * m_hat / (np.sqrt(v_hat) + eps)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t:,} new best {best_score:.6f}")

    best_seq = _project_onto_simplex(best_seq, s=1.0)
    return best_seq, best_score

# Linear‑programming based refinement -----------------------------------
def _build_constraint_matrix(f: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    """Construct A (rows = active convolution indices) for A·g ≤ M."""
    n = f.shape[0]
    m = len(active_idxs)
    A = np.zeros((m, n), dtype=np.float64)
    for r, k in enumerate(active_idxs):
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        cols = slice(start, end + 1)
        f_idx = k - np.arange(start, end + 1)
        A[r, cols] = f[f_idx]
    return A


def _lp_step(
    f: np.ndarray,
    M: float,
    t_mix: float,
    active_frac: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    One LP‑based update:
      * build active constraints (conv ≥ (1−active_frac)·M)
      * solve maximise Σg  s.t.  f∗g ≤ M
      * scale g to sum‑1 and mix with f (step size t_mix)
    Returns the new f and its maximal convolution.
    """
    n = f.shape[0]
    conv_f = _fft_self_convolve(f)
    thresh = M * (1.0 - active_frac)
    active_idxs = np.where(conv_f >= thresh)[0]
    if active_idxs.size == 0:
        active_idxs = np.array([np.argmax(conv_f)], dtype=np.int64)
        M = conv_f[active_idxs[0]]

    A = _build_constraint_matrix(f, active_idxs)
    b = np.full(A.shape[0], M, dtype=np.float64)

    # maximise Σg  → minimise -Σg
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        # fall back to the original vector if LP fails
        return f, M

    g = res.x
    sum_g = g.sum()
    if sum_g <= 1e-12:
        return f, M
    g = g / sum_g

    # try to mix; if the mixture raises the peak, halve the step size
    cur_t = t_mix
    while cur_t > 1e-6:
        new_f = (1.0 - cur_t) * f + cur_t * g
        new_M = _fft_self_convolve(new_f).max()
        if new_M <= M:
            return new_f, new_M
        cur_t *= 0.5

    return f, M  # no improvement


def _lp_optimize(
    f: np.ndarray,
    deadline: float,
    max_iters: int = 2000,
    t_mix: float = 0.2,
    active_start: float = 0.15,
    active_end: float = 0.005,
) -> tuple[np.ndarray, float]:
    """
    Repeated LP updates with a decreasing active fraction.
    Returns the best vector found and its exact objective.
    """
    best_f = f.copy()
    best_M = _fft_self_convolve(best_f).max()
    best_score = _objective(best_f)
    it = 0
    while time.time() < deadline and it < max_iters:
        # exponential schedule for the active fraction
        frac = it / max_iters if max_iters else 0.0
        active_frac = active_start * ((active_end / active_start) ** frac)

        new_f, new_M = _lp_step(best_f, best_M, t_mix, active_frac, deadline)
        new_score = _objective(new_f)

        if new_score < best_score - 1e-14:
            best_f, best_M, best_score = new_f, new_M, new_score
            print(f"[{time.time():.1f}] LP iter={it:,} new best {best_score:.6f}")

        # slowly decay the mixing weight to keep progress stable
        if it % 200 == 0 and it > 0:
            t_mix = max(t_mix * 0.9, 0.01)

        it += 1
    return best_f, best_score

def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:

        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:

        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:

        new_a = a.copy()

            best_score = cur
            best_seq = a.copy()

            )
    return best_seq, best_score

# Very cheap simulated‑annealing style local search
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,

    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:

        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        if cur < best_score - 1e-14:

        else:

                a = new_a

        remaining = deadline - time.time()
        long_deadline = time.time() + max(0.0, remaining - 30.0)
        print(f"[{time.time():.1f}] start long Adam descent (≈{remaining-30:.1f}s left)")
        overall_best_seq, overall_best_score = _adam_refine(
            overall_best_seq.copy(),
            long_deadline,
            max_iters=400_000,
            lr0=0.9,
            active_frac=0.02,
            max_active=30,
            beta1=0.9,
            beta2=0.999,
            eval_interval=250,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] after Adam best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=200_000,
        )

    # ------------------------------------------------------------------
    # 5️⃣  Final simulated‑annealing polishing
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.01,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best exact score ≈ {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # ``evaluate_sequence`` will re‑normalise; we keep a clean probability vector.
    return overall_best_seq.tolist()
