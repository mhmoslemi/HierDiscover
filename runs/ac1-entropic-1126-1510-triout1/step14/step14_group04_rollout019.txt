"""
A compact search for a non‑negative step‑function that minimizes

    evaluate_sequence(a) = 2·n·max(conv(a, a)) / (sum a)²

The algorithm is pure‑Python + NumPy and works within the 1000 s budget.

Strategy
--------
1. Scan a few lengths (up to 32000).  For each length evaluate
   several analytically‑motivated families (edge‑weighted, exponential‑edge,
   symmetric beta) and a dense grid of shape parameters.
2. For every base‑shape apply a mild linear “tilt” (asymmetry) and mix it
   with the uniform distribution.  The three parameters (shape, tilt,
   mixing weight) are explored on fine grids; the best candidate is kept.
3. A small number of random Dirichlet seeds are also tried.
4. Finally a short exponentiated‑gradient (mirror‑descent) refinement
   runs on the current best vector while the time budget permits.

All operations are performed on the simplex (the vector is always
re‑normalised), so the scaling invariance of the objective is respected.
The routine returns a plain Python list; `evaluate_sequence` will rescale
it internally.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  FFT based self‑convolution (real inputs)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a power‑of‑two real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())        # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
#  Exact objective for a vector normalised to Σa = 1
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a, a)) for a with sum 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
#  Analytic families (all return a probability vector – sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i+½)^{‑p}·(n‑i‑½)^{‑p} – symmetric, edge‑heavy for larger p."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    total = w.sum()
    if total == 0.0:
        return _distribution_uniform(n)
    w /= total
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    """w_i ∝ exp(‑c·i) + exp(‑c·(n‑1‑i))  (c>0)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    total = w.sum()
    if total == 0.0:
        return _distribution_uniform(n)
    w /= total
    return w


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric Beta‑type law on the discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    total = w.sum()
    if total == 0.0:
        return _distribution_uniform(n)
    w /= total
    return w


# ----------------------------------------------------------------------
#  Optional linear tilt (asymmetry) – still re‑normalised to a probability vector
# ----------------------------------------------------------------------
def _apply_tilt(a: np.ndarray, t: float) -> np.ndarray:
    """Multiply a_i by 1 + t·(i‑mid)/mid, clip negatives and renormalise."""
    if abs(t) < 1e-12:
        return a
    n = a.shape[0]
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    factor = 1.0 + t * ((idx - mid) / mid)
    b = a * factor
    b = np.clip(b, 0.0, None)
    s = b.sum()
    if s <= 0.0:               # fallback – should not happen
        return a
    b /= s
    return b


# ----------------------------------------------------------------------
#  Main search entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a step‑function (plain Python list) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0                # a small safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣  Parameter grids
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 16000, 32000]

    p_vals = np.linspace(0.2, 1.5, 27)                # edge‑weighted exponent
    c_vals = np.concatenate(
        (np.linspace(0.001, 0.5, 20), np.linspace(0.5, 2.0, 20))
    )                                                # exponential decay
    alpha_vals = np.linspace(0.2, 2.0, 25)            # symmetric beta

    tilt_vals = np.linspace(-0.4, 0.4, 17)            # linear asymmetry
    lam_vals = np.linspace(0.0, 0.9, 19)              # mixing with uniform

    # ------------------------------------------------------------------
    # 2️⃣  Exhaustive sweep over the families
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        uniform = np.full(n, 1.0 / n, dtype=np.float64)

        # ---- edge‑weighted -------------------------------------------------
        for p in p_vals:
            if time.time() >= deadline:
                break
            base = _distribution_edge_weighted(n, float(p))
            for t in tilt_vals:
                tilted = _apply_tilt(base, t)
                for lam in lam_vals:
                    a = lam * uniform + (1.0 - lam) * tilted
                    a /= a.sum()
                    sc = _objective(a)
                    if sc < best_score:
                        best_score, best_seq = sc, a.copy()
                        print(
                            f"[{time.time():.1f}] n={n} edge p={p:.4g} tilt={t:.3f} lam={lam:.2f} "
                            f"new best {best_score:.6f}"
                        )

        # ---- exponential‑edge ---------------------------------------------
        for c in c_vals:
            if time.time() >= deadline:
                break
            base = _distribution_exponential_edge(n, float(c))
            for t in tilt_vals:
                tilted = _apply_tilt(base, t)
                for lam in lam_vals:
                    a = lam * uniform + (1.0 - lam) * tilted
                    a /= a.sum()
                    sc = _objective(a)
                    if sc < best_score:
                        best_score, best_seq = sc, a.copy()
                        print(
                            f"[{time.time():.1f}] n={n} exp c={c:.4g} tilt={t:.3f} lam={lam:.2f} "
                            f"new best {best_score:.6f}"
                        )

        # ---- symmetric beta -----------------------------------------------
        for alpha in alpha_vals:
            if time.time() >= deadline:
                break
            base = _distribution_beta(n, float(alpha), float(alpha))
            for t in tilt_vals:
                tilted = _apply_tilt(base, t)
                for lam in lam_vals:
                    a = lam * uniform + (1.0 - lam) * tilted
                    a /= a.sum()
                    sc = _objective(a)
                    if sc < best_score:
                        best_score, best_seq = sc, a.copy()
                        print(
                            f"[{time.time():.1f}] n={n} beta α={alpha:.4g} tilt={t:.3f} lam={lam:.2f} "
                            f"new best {best_score:.6f}"
                        )

        # ---- a few random Dirichlet seeds (asymmetric candidates) ---------
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = rng.dirichlet(np.full(n, 0.3))
            sc = _objective(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(
                    f"[{time.time():.1f}] n={n} random Dirichlet new best {best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 3️⃣  Short exponentiated‑gradient refinement (if time remains)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        it = 0
        while time.time() < deadline:
            it += 1
            conv = _fft_self_convolve(a)
            max_val = conv.max()
            max_idxs = np.where(conv == max_val)[0]

            # exact sub‑gradient of the L∞‑objective
            grad = np.zeros_like(a)
            for idx in max_idxs:
                i_start = max(0, idx - (n - 1))
                i_end = min(n - 1, idx)
                i = np.arange(i_start, i_end + 1, dtype=np.int64)
                j = idx - i
                grad[i] += 2.0 * a[j]

            eta = 0.8 / math.sqrt(it)          # simple decaying step‑size
            a = a * np.exp(-eta * grad)        # EG keeps positivity
            s = a.sum()
            if s > 0.0:
                a /= s

            if it % 200 == 0:
                cur = 2.0 * n * max_val
                if cur < best_score:
                    best_score, best_seq = cur, a.copy()
                    print(
                        f"[{time.time():.1f}] refinement it={it} new best {best_score:.6f}"
                    )

    # ------------------------------------------------------------------
    # 4️⃣  Fallback (should never be triggered)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        best_seq = _distribution_uniform(n)
        best_score = _objective(best_seq)

    print(
        f"Final best objective (2·n·max(conv)) = {best_score:.6f} "
        f"(length={len(best_seq)})"
    )
    return best_seq.tolist()
