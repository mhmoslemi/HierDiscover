def _fft_convolve(a,b):
    n = len(a)
    m = len(b)
    size = 1 << ((n + m - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)

"""
Propose a step‑function (list of non‑negative heights) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------
1. Build a modest pool of candidate distributions using a few analytic
   families (uniform, arcsine, sum‑edge‑weighted, sparse Dirichlet).  All
   candidates are normalised to sum = 1 and the 10 best (lowest objective)
   are kept.

2. For each of the best seeds run a *smoothed* exponentiated‑gradient
   descent.
   •  The true objective is 2·n·max(conv(a,a)) (the sum‑normalisation makes it
      scale‑invariant).  Direct sub‑gradients of the max are very sparse and
      tend to get stuck.
   •  We replace the hard max by a soft‑max:
          Sα(a) = (1/α)·log Σ exp(α·conv(a,a))
      which is smooth for moderate α and converges to the true max as α→∞.
   •  The gradient of Sα can be written in closed form:
          ∂Sα/∂a = 2·corr(a, p) where
          p_i ∝ exp(α·conv_i)   (soft‑max weights)
          and corr denotes the (full) cross‑correlation.
      The correlation is evaluated efficiently by a single FFT‑based
      convolution.
   •  An exponential‑gradient (mirror‑descent) update
          a ← a·exp(−η·∇Sα) ,   a ← a / Σa
      keeps a on the probability simplex.
   •  The smoothing parameter α is increased slowly (from 0.5 up to 300)
      so the optimiser first moves in a smoother landscape and later refines
      the true max.

3. After the gradient phase a short pairwise‑mass‑transfer simulated‑annealing
   polish (identical to the one used in the previous state‑of‑the‑art) is run
   to capture any remaining local improvements.

All work respects the 1000 s wall‑time limit; if the budget expires early,
the best sequence found so far is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  1.  Convolution utilities (real FFT, full linear convolution)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (real FFT)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    size = 1 << ((n + m - 1 - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[: n + m - 1]


# ----------------------------------------------------------------------
#  2.  Exact objective (scale‑invariant, sum is assumed = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) for a non‑negative vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
#  3.  Analytic families (return a probability vector)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_sum_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ i^{-p} + (n‑i)^{-p} (edges summed)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (-p)) + (j ** (-p))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_sparse_dirichlet(n: int, rng: np.random.Generator,
                                  sparsity: float = 0.02) -> np.ndarray:
    """Random sparse vector: keep ~sparsity·n entries, Dirichlet‑like masses."""
    k = max(2, int(n * sparsity))
    idx = rng.choice(n, size=k, replace=False)
    vec = np.zeros(n, dtype=np.float64)
    vec[idx] = rng.random(k) + 1e-6
    vec /= vec.sum()
    return vec


# ----------------------------------------------------------------------
#  4.  Smoothed max and its gradient (sub‑gradient of the soft‑max)
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """p_i ∝ exp(alpha * conv_i)  (numerically stable)."""
    m = conv.max()
    exp_alpha = np.exp(alpha * (conv - m))
    sum_exp = exp_alpha.sum()
    if sum_exp == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp_alpha / sum_exp


def _smooth_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute ∂/∂a  (soft‑max of the auto‑convolution) .
    Result is a dense gradient (same shape as a).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)                     # length 2n‑1
    p = _softmax_weights(conv, alpha)               # same length
    # gradient = 2 * correlation(a, p)
    rev_p = p[::-1]
    corr = _fft_convolve(a, rev_p)                  # length 3n‑2
    # extract the n elements corr[n‑1 … 2n‑2] and reverse
    grad = 2.0 * corr[n - 1: 2 * n - 1][::-1]
    return grad


# ----------------------------------------------------------------------
#  5.  Exponentiated‑gradient descent on the smoothed objective
# ----------------------------------------------------------------------
def _smooth_exp_gradient_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 eta0: float = 1.0,
                                 eval_every: int = 120) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with exponential update on the smoothed max.
    The smoothing parameter α grows from 0.5 up to 300.
    """
    n = a.shape[0]
    t = 0
    alpha = 0.5
    alpha_target = 300.0
    alpha_growth = 1.02                     # multiplicative growth per iteration
    while time.time() < deadline:
        t += 1
        # gradient of the smoothed max
        grad = _smooth_gradient(a, alpha)

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)        # exponential update
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # increase smoothing
        if alpha < alpha_target:
            alpha = min(alpha_target, alpha * alpha_growth)

        # occasional exact evaluation
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} α={alpha:.1f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
#  6.  Pairwise‑mass‑transfer simulated annealing (final polish)
# ----------------------------------------------------------------------
def _pairwise_annealing(a: np.ndarray,
                        deadline: float,
                        best_seq: np.ndarray,
                        best_score: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins, accepting occasional uphill moves."""
    n = a.shape[0]
    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01
        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)
        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
#  7.  Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a step‑function that minimises ``evaluate_sequence``."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    best_seq = None
    best_score = float("inf")
    seed_pool: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    #  A.  Assemble a pool of diverse seeds (moderate lengths)
    # ------------------------------------------------------------------
    lengths = [3500, 5000, 7000, 9000]
    p_vals = [0.05, 0.1, 0.2, 0.4, 0.8]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating seeds for n={n}")

        # uniform
        cand = _distribution_uniform(n)
        sc = _objective(cand)
        seed_pool.append((sc, cand.copy()))
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] new global best (uniform) score={sc:.6f}")

        # arcsine
        cand = _distribution_arcsine(n)
        sc = _objective(cand)
        seed_pool.append((sc, cand.copy()))
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] new global best (arcsine) score={sc:.6f}")

        # sum‑edge‑weighted family
        for p in p_vals:
            if time.time() >= deadline:
                break
            cand = _distribution_edge_sum_weighted(n, p)
            sc = _objective(cand)
            seed_pool.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (sum‑edge p={p}) score={sc:.6f}")

        # random sparse Dirichlet seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            cand = _distribution_sparse_dirichlet(n, rng, sparsity=0.02)
            sc = _objective(cand)
            seed_pool.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (sparse) score={sc:.6f}")

    # keep only the top K seeds
    seed_pool.sort(key=lambda x: x[0])
    top_seeds = seed_pool[:10]

    # ------------------------------------------------------------------
    # B.  Include the previously submitted construction if available
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.ndim == 1:
                prev = np.abs(prev)
                if prev.sum() == 0.0:
                    prev = _distribution_uniform(prev.size)
                else:
                    prev = prev / prev.sum()
                sc = _objective(prev)
                top_seeds.append((sc, prev.copy()))
                if sc < best_score:
                    best_score, best_seq = sc, prev.copy()
                    print(f"[{time.time():.1f}] previous best score={sc:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # C.  Refine each top seed with smoothed exponentiated‑gradient descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0 and top_seeds:
        per_seed_time = remaining / len(top_seeds)
        for idx, (seed_score, seq) in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed_time
            print(f"[{time.time():.1f}] Seed {idx+1}/{len(top_seeds)} n={seq.shape[0]} start {seed_score:.6f}")
            seq, sc = _smooth_exp_gradient_descent(seq.copy(),
                                                   sub_deadline,
                                                   best_seq.copy(),
                                                   best_score)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time():.1f}] new global best after seed {idx+1} {best_score:.6f}")

    # ------------------------------------------------------------------
    # D.  Final annealing polish (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing")
        best_seq, best_score = _pairwise_annealing(best_seq.copy(),
                                                   deadline,
                                                   best_seq,
                                                   best_score,
                                                   rng)

    print(f"[{time.time():.1f}] final best score {best_score:.6f} (length={len(best_seq)})")
    # Convert to plain Python list (evaluate_sequence will re‑normalise anyway)
    return best_seq.tolist()
