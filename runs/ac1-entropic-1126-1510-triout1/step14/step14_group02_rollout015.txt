"""
An improved search for a step‑function that minimises ``evaluate_sequence``.
The algorithm augments the previous multi‑stage approach with

* a much larger hyper‑parameter sweep (lengths up to 8000, p up to 8,
  finer λ‑grid);
* two complementary descent methods (entropy mirror descent and a plain
  projected gradient descent) that are run sequentially;
* an LP‑based “Frank‑Wolfe‑style’’ refinement on a reduced problem
  (n≈600) that injects a direction obtained from solving a linear program;
* a longer, finer up‑sampling stage (up to 18000 points) and a more
  aggressive pair‑wise and simulated‑annealing cleanup.

All steps respect the global time budget (default 1000 s) and the routine
always returns a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# optional, used only for the LP refinement
import scipy.optimize
import scipy.sparse


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())   # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.004) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, where U is the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Force exact left‑right symmetry and renormalise onto the simplex."""
    sym = (seq + seq[::-1]) * 0.5
    return _project_onto_simplex(sym)

# Mirror‑descent (entropy) --------------------------------------------
def _subgradient_descent_mirror(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    active_frac: float = 0.04,
    max_active: int = 30,
    beta: float = 0.9,
    eval_interval: int = 120,
    patience: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the exact non‑smooth objective.
    ``a`` is assumed to be on the simplex.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    ind_grid = np.arange(n, dtype=np.int32)

    while time.time() < deadline and t < 2_000_000:
        t += 1
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            # keep the largest entries only
            top_idx = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top_idx]

        # compute (sub)gradient
        grad = np.zeros_like(a)
        for k in act:
            shift = k - ind_grid
            mask = (shift >= 0) & (shift < n)
            grad[mask] += 2.0 * a[shift[mask]]

        # momentum (Adam‑style)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * momentum)

        # renormalise (simplex projection is just normalisation here)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] mirror it={t:,} new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            jitter = rng.exponential(scale=0.004, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

# Projected gradient descent (plain) ------------------------------------
def _subgradient_descent_proj(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.5,
    active_frac: float = 0.02,
    max_active: int = 50,
    eval_interval: int = 80,
    patience: int = 1500,
) -> tuple[np.ndarray, float]:
    """
    Simple projected gradient descent. The step size decays as 1/t.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    ind_grid = np.arange(n, dtype=np.int32)

    while time.time() < deadline and t < 2_000_000:
        t += 1
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            top_idx = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top_idx]

        grad = np.zeros_like(a)
        for k in act:
            shift = k - ind_grid
            mask = (shift >= 0) & (shift < n)
            grad[mask] += 2.0 * a[shift[mask]]

        eta = lr0 / (1.0 + t * 0.0005)   # slower decay than 1/sqrt
        a = a - eta * grad
        a = np.maximum(a, 0.0)
        a = _project_onto_simplex(a)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] proj it={t:,} new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            # small random jitter to escape possible plateaux
            jitter = rng.exponential(scale=0.002, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a)
            no_improve = 0

    return best_seq, best_score

# Linear‑programming refinement (small scale) -------------------------
def _lp_refine_small(f_small: np.ndarray) -> np.ndarray | None:
    """
    Solve   max sum(g)   s.t.   (f_small * g)_k ≤ M   for all k,
    where M = max(f_small * f_small).  Returns the normalised g
    (sum(g)=1) or ``None`` if LinProg fails.
    """
    n = f_small.shape[0]
    M = _fft_self_convolve(f_small).max()
    # Build sparse A_ub where A_ub[k,i] = f_small[k-i] when defined
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # j = k - i  runs over the overlapping indices of f_small
        # coefficient equals f_small[j]
        for i in range(i_start, i_end + 1):
            j = k - i
            rows.append(k)
            cols.append(i)
            data.append(f_small[j])
    A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)   # maximise sum(g)
    res = scipy.optimize.linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None
    g = res.x
    if g.sum() == 0.0:
        return None
    g /= g.sum()
    return g

# Weighted pairwise refinement -----------------------------------------
def _pairwise_weighted_refine(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0

    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        conv = _fft_self_convolve(a)
        k_max = int(conv.argmax())

        # impact values L_i = a_{k_max-i} where defined
        L = np.zeros(n, dtype=np.float64)
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        idx = np.arange(low, high + 1)
        L[idx] = a[k_max - idx]

        # source i (high impact)
        if L.sum() > 0:
            probs_i = L / L.sum()
            i = rng.choice(n, p=probs_i)
        else:
            positives = np.where(a > 1e-12)[0]
            if positives.size == 0:
                break
            i = rng.choice(positives)

        # destination j (low impact)
        Lmax = L.max()
        probs_j = (Lmax - L) + 1e-12
        probs_j /= probs_j.sum()
        j = rng.choice(n, p=probs_j)
        if j == i:
            j = (j + 1) % n

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] weighted swap {i}->{j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score

# Cheap simulated annealing -------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

# Upsample / downsample helpers ---------------------------------------
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq = _project_onto_simplex(new_seq)
    return new_seq


def _downsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Simple averaging down‑sample (preserves the simplex)."""
    old_len = seq.shape[0]
    if new_len >= old_len:
        return seq.copy()
    factor = old_len / new_len
    out = np.zeros(new_len, dtype=np.float64)
    for i in range(new_len):
        start = int(round(i * factor))
        end = int(round((i + 1) * factor))
        out[i] = seq[start:end].sum()
    return _project_onto_simplex(out)

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine follows an extended multi‑stage pipeline and always returns a
    plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Seed generation – wider hyper‑parameter sweep and larger lengths
    lengths = [2000, 2500, 3000, 3500, 4000, 5000, 6000, 7000, 8000]
    # p values covering a much larger range (including very sharp edges)
    p_grid = np.concatenate(
        [
            np.logspace(-3, -1, 9),      # 0.001 … 0.1
            np.logspace(-1, 0, 9),      # 0.1 … 1
            np.logspace(0, 0.8, 9),     # 1 … 6.3
            np.logspace(0.8, 1.2, 5),   # 6.3 … 16
        ]
    )
    lam_grid = np.linspace(0.0, 0.9, 13)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] sampling length {n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (base) {sc:.6f}")

        # parametric families
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            edge2 = _distribution_edge_inversepow(n, p)
            sine = _distribution_sine_weighted(n, p)

            for base in (beta, edge, edge2, sine):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # asymmetric beta
        for _ in range(8):
            if time.time() >= deadline:
                break
            a = rng.uniform(0.001, 4.0)
            b = rng.uniform(0.001, 4.0)
            cand = _distribution_beta_asym(n, a, b)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best asym‑beta len={n} a={a:.3g} b={b:.3g} score={sc:.6f}"
                )

        # random Dirichlet
        for _ in range(12):
            if time.time() >= deadline:
                break
            cand = _distribution_dirichlet(n, rng)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best Dirichlet len={n} score={sc:.6f}")

        # sparse random vectors
        for _ in range(8):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}")

        # keep only the very best seeds
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2.  Include previously supplied construction if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev = _project_onto_simplex(prev)
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Guard against empty result
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Mirror‑descent on a handful of top seeds (moderate size)
    remaining = deadline - time.time()
    if remaining > 40.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:9]]
        per_seed = remaining * 0.55 / max(1, len(seeds))
        print(
            f"[{time.time():.1f}] heavy mirror‑descent on {len(seeds)} seeds, ≈{per_seed:.1f}s each"
        )
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _subgradient_descent_mirror(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                active_frac=0.06,
                max_active=40,
                beta=0.92,
                eval_interval=80,
                patience=2500,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after mirror descent seed {idx} new best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Projected‑gradient fine‑tuning (same seeds)
    remaining = deadline - time.time()
    if remaining > 30.0 and overall_best_seq is not None:
        per_seed = remaining * 0.30 / 3
        print(
            f"[{time.time():.1f}] projected‑gradient fine‑tuning, ≈{per_seed:.1f}s each"
        )
        for idx in range(3):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _subgradient_descent_proj(
                overall_best_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.6,
                active_frac=0.02,
                max_active=50,
                eval_interval=60,
                patience=1800,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] proj‑tune pass {idx} new best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 6.  LP‑based refinement on a reduced problem
    if time.time() < deadline - 15.0:
        # downsample to a manageable size for the LP (≈600)
        n_lp = min(800, max(400, overall_best_seq.shape[0] // 6))
        f_small = _downsample(overall_best_seq, n_lp)
        print(f"[{time.time():.1f}] LP refinement on reduced size {n_lp}")
        g_small = _lp_refine_small(f_small)
        if g_small is not None:
            # upsample the LP result back to the original length
            g_big = _upsample(g_small, overall_best_seq.shape[0])
            # blend a little to keep feasibility
            lam = 0.25
            candidate = (1.0 - lam) * overall_best_seq + lam * g_big
            candidate = _project_onto_simplex(candidate)
            cand_score = _objective(candidate)
            if cand_score + 1e-14 < overall_best_score:
                overall_best_score, overall_best_seq = cand_score, candidate.copy()
                print(
                    f"[{time.time():.1f}] LP‑blend improved to {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 7.  Upsample to a larger length (if time permits) and re‑optimize
    remaining = deadline - time.time()
    if remaining > 120.0:
        large_n = min(18000, overall_best_seq.shape[0] * 2)
        print(f"[{time.time():.1f}] up‑sampling to {large_n} for final refinement")
        up_seq = _upsample(overall_best_seq, large_n)

        # a short mirror‑descent on the enlarged vector
        sub_deadline = time.time() + remaining * 0.55
        up_seq_opt, up_score = _subgradient_descent_mirror(
            up_seq,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.7,
            active_frac=0.04,
            max_active=35,
            beta=0.93,
            eval_interval=70,
            patience=2500,
        )
        if up_score < overall_best_score:
            overall_best_score, overall_best_seq = up_score, up_seq_opt.copy()
            print(
                f"[{time.time():.1f}] up‑sampled descent yielded new best {overall_best_score:.6f}"
            )
        else:
            # if mirror descent did not help, try a short projected‑gradient pass
            sub_deadline = time.time() + remaining * 0.30
            up_seq_opt2, up_score2 = _subgradient_descent_proj(
                up_seq,
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.45,
                active_frac=0.015,
                max_active=45,
                eval_interval=50,
                patience=1500,
            )
            if up_score2 < overall_best_score:
                overall_best_score, overall_best_seq = up_score2, up_seq_opt2.copy()
                print(
                    f"[{time.time():.1f}] up‑sampled proj‑refine gave new best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 8.  Jitter & re‑descend cycles (escape plateaus)
    while time.time() < deadline - 12.0:
        jitter = rng.exponential(scale=0.015, size=overall_best_seq.shape)
        cand = overall_best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_onto_simplex(cand)
        sub_deadline = time.time() + min(20.0, (deadline - time.time()) * 0.5)
        cand_opt, cand_score = _subgradient_descent_mirror(
            cand,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.55,
            active_frac=0.05,
            max_active=30,
            beta=0.9,
            eval_interval=60,
            patience=2000,
        )
        if cand_score < overall_best_score:
            overall_best_score, overall_best_seq = cand_score, cand_opt.copy()
            print(f"[{time.time():.1f}] jitter‑restart improved to {overall_best_score:.6f}")
        else:
            break

    # ------------------------------------------------------------------
    # 9.  Symmetrisation test (exact symmetry often helps)
    sym_seq = _symmetrize(overall_best_seq)
    sym_score = _objective(sym_seq)
    if sym_score + 1e-14 < overall_best_score:
        overall_best_score, overall_best_seq = sym_score, sym_seq
        print(f"[{time.time():.1f}] symmetrised version better: {sym_score:.6f}")

    # ------------------------------------------------------------------
    # 10. Weighted pairwise deterministic refinement (more trials)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] weighted pairwise refinement (extensive)")
        overall_best_seq, overall_best_score = _pairwise_weighted_refine(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 11. Final cheap simulated‑annealing clean‑up
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.009,
            decay=0.9994,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
