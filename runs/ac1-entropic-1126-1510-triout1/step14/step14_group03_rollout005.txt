"""
A fast, multi‑stage optimiser for the self‑convolution bound.

Stage 1 – generate a set of analytic seeds (uniform, arcsine and a
parameterised “sum‑edge” family).  The latter is
    w_i ∝ (i+½+s)^{-p} + (n‑i‑½+s)^{-p},
with parameters p>0, s≥0 and an optional mixing with the uniform distribution.
A coarse grid over (p,λ) is scanned and the best few candidates are kept.

Stage 2 – locally refine each of these top candidates by a tiny hill‑climbing
search on the three parameters (p, s, λ).  The refinement stops when the
deadline is reached or when the step sizes become negligible.

Stage 3 – run a long exponentiated‑gradient (mirror) descent on the best
candidate found so far.  The gradient is the sub‑gradient of the active
convolution entry (the entry attaining the maximum).

Stage 4 – finish with a cheap pairwise mass‑transfer simulated‑annealing
polish.

All work respects the 1000 s wall‑time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  FFT‑based self‑convolution (full linear convolution)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
#  Exact objective used by the evaluator (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a))/sum(a)² – a must be non‑negative."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
#  Gradient of the *active* convolution entry (sub‑gradient of max)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """Return (gradient, index_of_max) for the entry of a*a attaining max."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))

    grad = np.zeros_like(a)
    if k < n:
        # overlap length = k+1   (i runs 0..k)
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        # k ≥ n : overlap length = 2·n‑1‑k   (i runs k‑(n‑1)..n‑1)
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
#  Analytic families – all return a *probability* vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_sum_weighted_param(
    n: int, p: float, s: float = 0.0
) -> np.ndarray:
    """w_i ∝ (i+½+s)^{-p} + (n‑i‑½+s)^{-p}  (i = 0,…,n‑1)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i + s) ** (-p) + (j + s) ** (-p)
    # protect against infinities for extreme p
    w = np.where(np.isfinite(w), w, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1‑lam) + uniform·lam (both sum to 1)."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
#  Exponentiated‑gradient descent (multiplicative mirror descent)
# ----------------------------------------------------------------------
def _exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """Run EG descent on the exact objective until ``deadline``."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Pairwise‑mass‑transfer annealing (final polish)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins; accept improvements."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4        # move at most 40 % of source mass
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        accept = cur < best_score - 1e-14 or (
            temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)
        )

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Local hill‑climbing refinement for the sum‑edge family
# ----------------------------------------------------------------------
def _refine_edge_sum(
    n: int,
    p0: float,
    s0: float,
    lam0: float,
    score0: float,
    seq0: np.ndarray,
    deadline: float,
) -> tuple[np.ndarray, float, float, float, float]:
    """Tiny deterministic hill‑climb on (p, s, λ) using the exact objective."""
    p, s, lam = p0, s0, lam0
    best_seq, best_score = seq0.copy(), score0

    # initial step sizes – they shrink automatically when no improvement occurs
    step_p = max(p * 0.2, 0.01)
    step_s = max(s * 0.5, 0.5) if s > 0 else 0.5
    step_lam = 0.2

    while time.time() < deadline:
        improved = False

        # ---- p variations -------------------------------------------------
        for factor in (0.7, 0.85, 0.95, 1.05, 1.15, 1.3):
            p_new = max(1e-6, p * factor)
            base = _distribution_edge_sum_weighted_param(n, p_new, s)
            cand = _mix_with_uniform(base, lam)
            sc = _objective(cand)
            if sc + 1e-12 < best_score:
                p, best_score, best_seq = p_new, sc, cand.copy()
                improved = True
                print(f"[{time.time():.1f}] hill p={p:.4g} s={s:.3f} λ={lam:.3f} score={best_score:.6f}")

        # ---- s variations -------------------------------------------------
        for ds in (-step_s, step_s):
            s_new = max(0.0, s + ds)
            base = _distribution_edge_sum_weighted_param(n, p, s_new)
            cand = _mix_with_uniform(base, lam)
            sc = _objective(cand)
            if sc + 1e-12 < best_score:
                s, best_score, best_seq = s_new, sc, cand.copy()
                improved = True
                print(f"[{time.time():.1f}] hill p={p:.4g} s={s:.3f} λ={lam:.3f} score={best_score:.6f}")

        # ---- λ variations -------------------------------------------------
        for dlam in (-step_lam, step_lam):
            lam_new = lam + dlam
            lam_new = min(1.0, max(0.0, lam_new))
            base = _distribution_edge_sum_weighted_param(n, p, s)
            cand = _mix_with_uniform(base, lam_new)
            sc = _objective(cand)
            if sc + 1e-12 < best_score:
                lam, best_score, best_seq = lam_new, sc, cand.copy()
                improved = True
                print(f"[{time.time():.1f}] hill p={p:.4g} s={s:.3f} λ={lam:.3f} score={best_score:.6f}")

        if not improved:
            # shrink steps – stop when they become tiny
            step_p *= 0.5
            step_s *= 0.5
            step_lam *= 0.5
            if step_p < 1e-6 and step_s < 1e-6 and step_lam < 1e-4:
                break
    return best_seq, best_score, p, s, lam


# ----------------------------------------------------------------------
#  Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The method combines a dense parametric sweep, a tiny hill‑climb,
    exponentiated‑gradient descent and a final annealing polish.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray, int, float, float, float]] = []

    # ------------------------------------------------------------------
    # 1.  Coarse grid over lengths, p, λ
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000]        # modest sizes – very cheap to evaluate
    p_grid = np.concatenate((
        np.logspace(-2.0, -0.5, 12),          # 0.01 … 0.316
        np.logspace(-0.5, 0.5, 12),           # 0.316 … 3.16
        np.logspace(0.5, 1.0, 12),            # 3.16 … 10
    ))
    lam_grid = np.linspace(0.0, 1.0, 11)       # 0 … 1 in steps of 0.1

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families (uniform / arcsine) – keep for reference
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (uniform) score={sc:.6f}")
            top_candidates.append((sc, cand.copy(), n, 0.0, 0.0, 0.0))

        # sum‑edge weighted family with mixing
        for p in p_grid:
            if time.time() >= deadline:
                break
            base = _distribution_edge_sum_weighted_param(n, p, 0.0)
            sc_base = _objective(base)
            top_candidates.append((sc_base, base.copy(), n, p, 0.0, 0.0))
            if sc_base < best_score:
                best_score, best_seq = sc_base, base.copy()
                print(f"[{time.time():.1f}] new best sum p={p:.4g} score={sc_base:.6f}")

            for lam in lam_grid:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy(), n, p, 0.0, lam))
                if sc < best_score:
                    best_score, best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new best sum p={p:.4g} λ={lam:.2f} score={sc:.6f}")

        # keep only a small pool for the costly hill‑climb
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]      # 30 best seeds overall so far

    # ------------------------------------------------------------------
    # 2.  Local hill‑climbing on the sum‑edge family
    # ------------------------------------------------------------------
    refined: list[tuple[float, np.ndarray]] = []
    for sc, seq, n, p, s, lam in top_candidates:
        if time.time() >= deadline:
            break
        # Only attempt refinement on candidates that actually come from the
        # sum‑edge family (lam∈[0,1] and p>0).  Others are kept as‑is.
        if p <= 0:
            refined.append((sc, seq))
            continue
        # Reserve a slice of the remaining time for this candidate
        cand_deadline = min(deadline, time.time() + 12.0)   # at most ~12 s each
        seq_ref, sc_ref, p_ref, s_ref, lam_ref = _refine_edge_sum(
            n, p, s, lam, sc, seq, cand_deadline
        )
        refined.append((sc_ref, seq_ref))

        if sc_ref < best_score:
            best_score, best_seq = sc_ref, seq_ref.copy()
            print(f"[{time.time():.1f}] post‑refine best score={best_score:.6f}")

    # If we have no candidate (should not happen), fall back to uniform
    if best_seq is None:
        best_seq = _distribution_uniform(5000)
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Exponentiated‑gradient descent refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] EG refinement (remaining {deadline - time.time():.1f}s)")
        best_seq, best_score = _exp_gradient_descent(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            eta0=0.9,
            eval_every=40,
        )
        print(f"[{time.time():.1f}] after EG best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Final annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing")
        best_seq, best_score = _pairwise_annealing(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            rng,
        )
        print(f"[{time.time():.1f}] after annealing best score {best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective = 2·n·max(conv)/sum²): {best_score:.6f} "
        f"(length={len(best_seq)})"
    )
    # The evaluator will clip negatives / out‑of‑range values itself
    return best_seq.tolist()
