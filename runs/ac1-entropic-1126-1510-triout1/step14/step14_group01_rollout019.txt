"""
LP‑based iterative refinement for minimizing max(conv(a,a))

The algorithm:
1. Start from a diverse set of simplex seeds (uniform, beta, edge‑weighted, random dirichlet).
2. For each seed run a few iterations of a linear‑programming sub‑step:
   – Compute M = max(conv(f,f)).
   – Solve the LP  maximize  1ᵀ g  subject to  f * g ≤ M,  g ≥ 0 .
   – If the LP returns a total mass Σg > 1, blend the normalized solution
     ĝ = g/Σg with the current f:   f ← (1−τ)·f + τ·ĝ .
   – Project back onto the probability simplex.
   The LP reduces the bound M and drives the sequence towards the optimal
   trade‑off between a_i² (large spikes) and the cross‑terms f_i f_{n‑1‑i}.
3. After a fixed number of LP iterations the best seed is up‑sampled
   linearly to a larger length and the LP refinement is repeated.
4. A cheap pairwise‑swap / simulated‑annealing phase finishes the search.

The whole routine respects the 1000 s budget and returns a Python list that
can be fed directly to ``evaluate_sequence``.
"""

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]

    """Exact objective 2·n·max(conv(a,a)) for a simplex vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

def _linear_program_step(f: np.ndarray, M: float, deadline: float) -> np.ndarray | None:
    """
    Solve  max 1ᵀ g   subject to   f * g ≤ M ,  g ≥ 0 .
    Returns the optimal g or None on failure / timeout.
    """
    n = f.shape[0]
    # Build the convolution matrix F (size (2n‑1) × n) using sparse diagonals.
    # Row k, column j carries f[k‑j] if 0≤k‑j<n.
    # Equivalently each diagonal with offset = i (i = 0..n‑1) is filled by f[i].
    diags = [np.full(n, f[i]) for i in range(n)]
    offsets = np.arange(n)               # 0,1,…,n‑1
    F = sparse.diags(diags, offsets, shape=(2 * n - 1, n), format="csc")
    # Upper‑bound vector
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # Linear program: minimize -∑g   →   maximize ∑g
    c = -np.ones(n, dtype=np.float64)

    res = linprog(
        c,
        A_ub=F,
        b_ub=b,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True, "time_limit": max(0.0, deadline - time.time())},
    )
    if not res.success:
        return None
    return res.x


def _lp_iterate(
    f0: np.ndarray,
    deadline: float,
    max_iter: int = 30,
    tau_start: float = 0.6,
    tau_decay: float = 0.94,
) -> np.ndarray:
    """
    Repeated LP‑based refinement of f.  The routine stops early if time runs out.
    """
    f = f0.copy()
    n = f.shape[0]
    tau = tau_start
    it = 0
    while it < max_iter and time.time() < deadline:
        it += 1
        conv_f = _fft_self_convolve(f)
        M = conv_f.max()
        g = _linear_program_step(f, M, deadline)
        if g is None:
            break
        sum_g = g.sum()
        # If the LP cannot increase the total mass we are at a stationary point.
        if sum_g <= 1.0 + 1e-12:
            break
        g_norm = g / sum_g
        # convex combination and simplex projection
        f = (1.0 - tau) * f + tau * g_norm
        f = _project_onto_simplex(f, s=1.0)
        tau = max(0.05, tau * tau_decay)   # slowly shrink the step size
    return f


def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq

        # evaluate each seed and keep the best few for the LP phase
        scored: list[tuple[float, np.ndarray]] = []
        for s in seeds:
            sc = _objective(s)
            scored.append((sc, s))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, s.copy()
                print(f"[{time.time():.1f}] new global best (seed) n={n} {sc:.6f}")

        # keep only the top‑5 seeds for this size
        scored.sort(key=lambda x: x[0])
        top_seeds = [seq for _, seq in scored[:5]]

        # ------------------------------------------------------------------
        # 2. Apply the LP refinement to each top seed
        per_seed_time = max(0.0, (deadline - time.time()) * 0.4 / max(1, len(top_seeds)))
        print(f"[{time.time():.1f}] LP refinement on {len(top_seeds)} seeds, ~{per_seed_time:.1f}s each")
        for idx, seed_seq in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed_time
            opt_seq = _lp_iterate(seed_seq, sub_dead, max_iter=25, tau_start=0.7)
            sc = _objective(opt_seq)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, opt_seq.copy()
                print(f"[{time.time():.1f}] after LP seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Upsample the current best (if time permits) and polish again
    if overall_best_seq is not None and time.time() < deadline - 5.0:
        new_len = min(15000, overall_best_seq.shape[0] * 2)
        print(f"[{time.time():.1f}] up‑sampling best solution to {new_len}")
        up_seq = _upsample(overall_best_seq, new_len)
        up_seq = _lp_iterate(up_seq, deadline, max_iter=30, tau_start=0.6)
        sc = _objective(up_seq)
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, up_seq.copy()
            print(f"[{time.time():.1f}] after up‑sample new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final cheap local improvements (pairwise swap + simulated annealing)
    if overall_best_seq is not None and time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=200_000,
        )
    if overall_best_seq is not None and time.time() < deadline:
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.012,
            decay=0.9996,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best objective: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return plain Python list – evaluate_sequence will do final clipping
    return overall_best_seq.tolist()

    offsets = np.arange(n)                     # 0 … n‑1
    F = sparse.diags(diags, offsets, shape=(2 * n - 1, n), format="csc")

    b = np.full(2 * n - 1, M, dtype=np.float64)   # RHS of the inequality
    c = -np.ones(n, dtype=np.float64)            # minimise –∑g  ⇔ maximise ∑g

    res = linprog(c, A_ub=F, b_ub=b,
                  bounds=[(0, None)] * n,
                  method="highs")
    if not res.success:
        return None
    return res.x


def _lp_iterate(
    f0: np.ndarray,
    deadline: float,
    max_iter: int = 30,
    tau_start: float = 0.7,
    tau_decay: float = 0.94,
) -> np.ndarray:
    """
    Repeated LP‑based refinement of a simplex vector f.
    Stops early if time runs out or if the LP cannot increase the total mass.
    """
    f = f0.copy()
    tau = tau_start
    it = 0
    while it < max_iter and time.time() < deadline:
        it += 1
        M = _fft_self_convolve(f).max()
        g = _linear_program_step(f, M)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-12:      # stationary point – nothing to improve
            break
        g_norm = g / sum_g
        f = (1.0 - tau) * f + tau * g_norm
        f = _project_onto_simplex(f, s=1.0)
        tau = max(0.05, tau * tau_decay)   # slowly shrink the step size
    return f


# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
# Deterministic pairwise refinement
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Simple deterministic mass‑transfer sweep."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e}  new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Cheap simulated‑annealing clean‑up
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Random jitter / simulated‑annealing style local improvement."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e}  best {best_score:.6f}")
        else:
            # occasional uphill move
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Multi‑start LP‑based optimiser for the step‑function evaluation.
    Returns a Python list of heights (non‑negative, sum normalised).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")

    # --------------------------------------------------------------
    # 1. generate a pool of seeds (moderate lengths)
    lengths = [1200, 1600, 2000, 2400, 3000]
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating seeds for n={n}")

        seeds: list[np.ndarray] = []

        # deterministic families
        seeds.append(_distribution_uniform(n))
        seeds.append(_distribution_arcsine(n))

        # symmetric beta with a few exponents
        for p in (0.2, 0.5, 1.0, 2.0, 4.0):
            seeds.append(_distribution_beta_sym(n, p))

        # edge‑weighted families
        for p in (0.2, 0.5, 1.0, 2.0):
            seeds.append(_distribution_edge_weighted(n, p))
            seeds.append(_distribution_edge_inversepow(n, p))

        # sine‑weighted
        for p in (1.0, 2.0, 4.0):
            seeds.append(_distribution_sine_weighted(n, p))

        # random Dirichlet / sparse
        for _ in range(3):
            seeds.append(_distribution_dirichlet(n, rng))
        for _ in range(2):
            seeds.append(_distribution_random_sparse(n, rng, density=0.015))

        # ----------------------------------------------------------
        # 2. keep the best few seeds and apply the LP refinement
        scored: list[tuple[float, np.ndarray]] = [( _objective(s), s ) for s in seeds]
        scored.sort(key=lambda x: x[0])
        top_seeds = [seq for _, seq in scored[:5]]

        per_seed = max(0.0, (deadline - time.time()) * 0.4 / max(1, len(top_seeds)))
        print(f"[{time.time():.1f}] LP refinement on {len(top_seeds)} seeds, ~{per_seed:.1f}s each")
        for idx, seed_seq in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            opt_seq = _lp_iterate(seed_seq, sub_dead, max_iter=25, tau_start=0.7)
            sc = _objective(opt_seq)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, opt_seq.copy()
                print(f"[{time.time():.1f}] seed {idx} new global best {overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Upsample the current best (if time permits) and polish again
    if overall_best_seq is not None and time.time() < deadline - 5.0:
        new_len = min(15000, overall_best_seq.shape[0] * 2)
        print(f"[{time.time():.1f}] up‑sampling best solution to {new_len}")
        up_seq = _upsample(overall_best_seq, new_len)
        up_seq = _lp_iterate(up_seq, deadline, max_iter=30, tau_start=0.6)
        sc = _objective(up_seq)
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, up_seq.copy()
            print(f"[{time.time():.1f}] after up‑sample new best {overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Final cheap local refinements
    if overall_best_seq is not None and time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=200_000,
        )
    if overall_best_seq is not None and time.time() < deadline:
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.012,
            decay=0.9996,
        )

    # --------------------------------------------------------------
    print(
        f"Final best objective: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list – evaluate_sequence will apply the final clipping.
    return overall_best_seq.tolist()
