def gradient_smooth_max(a, beta):
    conv = self_convolve(a)  # full conv length 2n-1
    # Compute weights softmax over conv: weight_k = exp(beta * conv_k) / sum exp.
    # For numeric stability, subtract max conv.
    max_conv = conv.max()
    exps = np.exp(beta * (conv - max_conv))  # shape 2n-1
    sum_exps = exps.sum()
    # p_k = exps / sum_exps  # probabilities
    p = exps / sum_exps

    # Compute gradient 2 * cross-correlation of a and p.
    # Compute p_rev = p[::-1]
    p_rev = p[::-1]
    # Compute convolution via FFT
    size = 1 << ((n + (2*n-1) - 1) - 1).bit_length()
    # Actually convolution length = n + (2n-1) - 1 = 3n-2.
    size = 1 << ((3*n-2) - 1).bit_length()
    # Or use np.fft.rfft for real values.
    A = np.fft.rfft(a, n=size)
    P_rev = np.fft.rfft(p_rev, n=size)
    conv_ap = np.fft.irfft(A * P_rev, n=size)
    conv_ap = conv_ap[: 3*n-2]  # only needed part
    # Then r_j = conv_ap[(2*n-2) - j] for j in [0,n-1]
    start = 2*n-2
    r = conv_ap[start - np.arange(n)]
    grad = 2.0 * r
    return grad, conv, p

def _smoothmax_gradient(a, beta):
    n = a.shape[0]
    conv = _fft_self_convolve(a)   # shape 2n-1
    max_c = conv.max()
    # stability
    exps = np.exp(beta * (conv - max_c))
    sum_exps = exps.sum()
    p = exps / sum_exps  # shape 2n-1
    # Compute r = cross-correlation between a and p
    # Use convolution a * p_rev
    p_rev = p[::-1]
    size = 1 << ((n + (2*n-1) - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    P_rev = np.fft.rfft(p_rev, n=size)
    conv_ap = np.fft.irfft(A * P_rev, n=size)

    conv_len_needed = n + (2*n-1) - 1    # = 3n-2
    conv_ap = conv_ap[:conv_len_needed]
    idx = (2*n - 2)   # len(p)-1
    r = conv_ap[idx - np.arange(n)]
    grad = 2.0 * r
    return grad, conv.max()

"""
Propose a non‑negative step function that minimises the
``evaluate_sequence`` objective.

Approach
--------
1.  Generate a large and diverse pool of candidate distributions.
    Families include uniform, arcsine, edge‑product, edge‑sum, exponential‑edge,
    generalized‑normal and random Dirichlet seeds (both dense and sparse).
2.  Keep the best few (≈20) candidates.
3.  Refine each of them with a *smooth* exponentiated‑gradient descent.
    The nonsmooth max of the self‑convolution is approximated by a
    soft‑max (temperature β).  The gradient of the soft‑max is obtained
    efficiently using FFT‑based cross‑correlation.
4.  Finish with a pairwise‑mass‑transfer simulated annealing polish.
5.  Return the best distribution found (as a plain ``list[float]``).

The algorithm respects the 1000 s wall‑time limit and works with any
reasonable length (up to a few‑tens of thousands).  It is fully
self‑contained and requires only ``numpy`` and the already imported
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  Utility
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
#  Convolution helpers (real, full linear convolution)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using FFT (real inputs)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _cross_correlate(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute r_j = Σ_i a_i * w_{i+j}   (cross‑correlation).
    a has length n, w has length 2n‑1.
    Returns an array of length n.
    """
    n = a.shape[0]
    # reverse w for convolution
    w_rev = w[::-1]
    size = _next_pow2(n + w_rev.shape[0] - 1)      # ≥ 3n‑2
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w_rev, n=size)
    conv = np.fft.irfft(A * W, n=size)
    # needed length is n + (2n‑1) – 1 = 3n‑2
    conv = conv[: 3 * n - 2]
    # index (len(w)‑1) = 2n‑2 ; r_j = conv[2n‑2‑j]
    idx = 2 * n - 2
    return conv[idx - np.arange(n)]


# ----------------------------------------------------------------------
#  Exact objective used by the evaluator (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a))/sum(a)² for a non‑negative vector."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 0.01:                     # protect against division by zero
        return float("inf")
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
#  Analytic families – all return a *probability* vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i·(n‑i))^{‑p}   with i = distance to the left edge (0‑based)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_sum_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ i^{‑p} + (n‑i)^{‑p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (-p)) + (j ** (-p))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    """w_i ∝ exp(‑c·dist_to_nearest_edge)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    d = np.minimum(i, j)               # distance to the closest edge
    w = np.exp(-c * d)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(‑|x‑½|^{q}) on the discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1‑lam) + uniform·lam."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
#  Smooth‑max gradient (soft‑max approximation of max(conv))
# ----------------------------------------------------------------------
def _smoothmax_grad(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Return (gradient, current_max) where the gradient is that of
    the soft‑max approximation  (1/β)·log Σ exp(β·conv_k).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)               # length 2n‑1
    max_c = conv.max()
    # soft‑max weights (stable)
    exps = np.exp(beta * (conv - max_c))
    sum_exps = exps.sum()
    p = exps / sum_exps                         # shape 2n‑1, sums to 1

    # gradient = 2 * cross‑correlation(a, p)
    grad = 2.0 * _cross_correlate(a, p)
    return grad, max_c


# ----------------------------------------------------------------------
#  Exponentiated‑gradient descent using the smooth‑max sub‑gradient
# ----------------------------------------------------------------------
def _exp_gradient_descent_smooth(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    beta: float = 10.0,
    eta0: float = 0.5,
    eval_every: int = 120,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative mirror descent on the smooth‑max objective.
    ``beta`` controls the temperature (large → tighter max approximation).
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _smoothmax_grad(a, beta)
        eta = eta0 / math.sqrt(t)
        # multiplicative update (keeps positivity)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth‑EG it={t} β={beta:.1f} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Pairwise‑mass‑transfer annealing (final polish)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins, accept occasional uphill moves."""
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        # source index with positive mass
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        # destination (different index)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # how much to move
        delta = a[i] * rng.random() * 0.4          # ≤40 % of source mass
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
#  Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Returns a plain ``list[float]`` ready for the evaluator.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")
    top: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Generate a wide pool of seeds
    # ------------------------------------------------------------------
    lengths = [3000, 5000, 8000, 12000, 16000]   # affordable moderate sizes
    # Parameter grids
    p_grid = np.concatenate((
        np.logspace(-3.0, -0.5, 12, base=10),   # 0.001 … 0.316
        np.logspace(-0.5, 0.5, 12, base=10),    # 0.316 … 3.16
        np.logspace(0.5, 2.0, 12, base=10),    # 3.16 … 100
    ))
    lam_grid = np.linspace(0.0, 0.5, 6)          # 0,0.1,…,0.5
    q_grid = np.linspace(0.5, 6.0, 12)
    c_grid = np.logspace(-1.0, 1.0, 10)         # exponential‑edge decay

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families (uniform, arcsine)
        for cand in (_distribution_uniform(n),
                     _distribution_arcsine(n)):
            sc = _objective(cand)
            top.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best ({n}) score={sc:.6f}")

        # edge‑product weighted + mixing
        for p in p_grid:
            if time.time() >= deadline:
                break
            base = _distribution_edge_weighted(n, p)
            sc = _objective(base)
            top.append((sc, base.copy()))
            if sc < best_score:
                best_score, best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new best prod p={p:.3g} score={sc:.6f}")

            for lam in lam_grid:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                top.append((sc, cand.copy()))
                if sc < best_score:
                    best_score, best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new best prod p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # edge‑sum weighted (+ mixing)
        for p in p_grid:
            if time.time() >= deadline:
                break
            base = _distribution_edge_sum_weighted(n, p)
            sc = _objective(base)
            top.append((sc, base.copy()))
            if sc < best_score:
                best_score, best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new best sum p={p:.3g} score={sc:.6f}")

            for lam in lam_grid:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                top.append((sc, cand.copy()))
                if sc < best_score:
                    best_score, best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new best sum p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # exponential‑edge family (+ mixing)
        for c in c_grid:
            if time.time() >= deadline:
                break
            base = _distribution_exponential_edge(n, c)
            sc = _objective(base)
            top.append((sc, base.copy()))
            if sc < best_score:
                best_score, best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new best exp‑edge c={c:.3g} score={sc:.6f}")

            for lam in lam_grid:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                top.append((sc, cand.copy()))
                if sc < best_score:
                    best_score, best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new best exp‑edge c={c:.3g} lam={lam:.2f} score={sc:.6f}")

        # generalized‑normal family
        for q in q_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_generalized_normal(n, q)
            sc = _objective(cand)
            top.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best gn q={q:.2f} score={sc:.6f}")

        # random Dirichlet seeds (dense)
        for _ in range(30):
            if time.time() >= deadline:
                break
            cand = rng.random(n)
            cand /= cand.sum()
            sc = _objective(cand)
            top.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best dense rand score={sc:.6f}")

        # sparse random seeds (≈1 % non‑zero)
        for _ in range(30):
            if time.time() >= deadline:
                break
            cand = np.zeros(n, dtype=np.float64)
            k = max(2, int(n * 0.01))
            idx = rng.choice(n, size=k, replace=False)
            cand[idx] = rng.random(k)
            cand /= cand.sum()
            sc = _objective(cand)
            top.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best sparse rand score={sc:.6f}")

        # keep only the very best seeds (size 20)
        top.sort(key=lambda x: x[0])
        top = top[:20]

    # ------------------------------------------------------------------
    # 2. Include previous construction if available
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < best_score:
                best_score, best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Guard against empty result
    # ------------------------------------------------------------------
    if best_seq is None:

        best_score = _objective(best_seq)

    # ------------------------------------------------------------------
    # 4. Smooth‑gradient descent on the top candidates
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # allocate remaining time roughly equally among the top few seeds
        remaining = deadline - time.time()
        num_opt = min(5, len(top))                 # run a few independent refinements
        sub_budget = remaining / (num_opt + 1)    # keep a slice for final annealing
        for idx, (_, cand) in enumerate(top[:num_opt]):
            if time.time() >= deadline:
                break
            print(f"[{time.time():.1f}] optimisation pass {idx+1}/{num_opt}")
            # smooth‑EG schedule with increasing β
            seq = cand.copy()
            score = _objective(seq)
            betas = [0.5, 2.0, 5.0, 10.0, 20.0]

                    seq,
                    sub_deadline,
                    seq,
                    score,
                    beta=beta,
                    eta0=0.8,
                    eval_every=80,
                )
                if time.time() >= deadline:
                    break
            if score < best_score - 1e-14:
                best_score, best_seq = score, seq.copy()
                print(f"[{time.time():.1f}] new global best after smooth‑EG score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final annealing polish (uses any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing")
        best_seq, best_score = _pairwise_annealing(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            rng,
        )
        print(f"[{time.time():.1f}] after annealing best score {best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective = 2·n·max(conv)/sum²): {best_score:.6f} "
        f"(length={len(best_seq)})"
    )
    return best_seq.tolist()
