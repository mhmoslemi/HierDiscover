"""
Search for a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (∑a)²

The algorithm (≈1000 s budget) consists of:

1. **Rich analytic seeding** – a handful of families (uniform,
   arcsine, Hann, edge‑weighted, β‑distribution) are sampled for a
   few modest lengths.  The best few seeds are kept.

2. **Adam‑style exponentiated‑gradient polishing** – the objective
   2·n·max(conv) is differentiable w.r.t. the *max* index; a mirror
   descent with bias‑corrected first/second moments (Adam) is applied.
   The update is multiplicative (a ← a·exp(‑η·m̂/√v̂)) and after
   each step the vector is projected back onto the probability simplex.
   The step‑size η is increased on improvement and decayed otherwise.

3. **Progressive up‑sampling** – the current best vector is linearly
   interpolated to a larger size (doubling while ≤ 65536).  After each
   up‑sample a short Adam run is performed; even if the up‑sampled
   objective does not improve immediately the new (larger) vector is
   kept and further refined – this overcomes the stagnation of the
   previous implementation which repeatedly re‑tried the same size.

4. **Pairwise‑mass‑transfer annealing** – a light simulated‑annealing
   that moves a random fraction of mass between two bins, accepting
   downhill moves always and uphill moves with a linear cooling schedule.

All steps respect the 1000 s budget (2 s safety margin).  The function
returns a plain Python list ready for ``evaluate_sequence`` (the list is
already non‑negative and normalised to sum = 1, which does not affect the
objective because the evaluator is homogeneous).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """ Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}. """
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (real vectors)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective for a probability vector (∑a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Returns 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families – each returns a *probability* vector (sum=1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{-p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) sampled on cell mid‑points."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    w = np.where(
        (x == 0) | (x == 1),
        0.0,
        np.exp((alpha - 1) * np.log(x) + (alpha - 1) * np.log1p(-x))
    )
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1‑λ) + uniform·λ."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Gradient (sub‑gradient) of the maximal convolution entry
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Returns (grad, k) where k is the index attaining max conv.
    grad_i = 2·a_{k‑i} for i that participates in the maximal term,
    otherwise 0.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))

    grad = np.zeros_like(a)
    if k < n:
        # overlap indices 0 … k
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        # overlap indices k‑(n‑1) … n‑1
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
# Adam‑style exponentiated‑gradient minimiser (mirrored updates)
# ----------------------------------------------------------------------
def _adam_exp_gradient(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """
    Mirror descent with Adam‑style adaptive moments.
    Multiplicative update: a ← a·exp(‑η·m̂/√v̂)  (ensures non‑negativity).
    After each update a is projected onto the probability simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)          # EMA of gradients (first moment)
    v = np.zeros_like(a)          # EMA of squared gradients (second moment)
    eta = eta0
    t = 0
    eps = 1e-12

    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)

        # Adam moments update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # multiplicative mirror step
        factor = np.exp(-eta * m_hat / (np.sqrt(v_hat) + eps))
        a = a * factor

        # projection onto the simplex (keeps sum = 1)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-15:
                best_score, best_seq = cur, a.copy()
                eta = min(eta0, eta * 1.02)          # reward
                print(f"[{time.time():.1f}] Adam it={t:,} new best {best_score:.6f}")
            else:
                eta *= 0.995                         # decay

        # occasional re‑projection (numerical drift)
        if t % (eval_every * 10) == 0:
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise mass‑transfer simulated annealing
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        src = rng.choice(src_candidates)

        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        delta = a[src] * rng.random() * 0.25   # up to 25 % of src mass
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        cur = _objective(new_a)

        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.02

        accept = False
        if cur < best_score - 1e-15:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-15:
                best_score, best_seq = cur, a.copy()
                print(
                    f"[{time.time():.1f}] anneal {src}->{dst} Δ={delta:.1e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling while preserving total mass (=1)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new
