"""
Propose a non‑negative step function (probability vector) that minimises
``evaluate_sequence``.  The search consists of:

1.  A broad set of analytic seeds (uniform, arcsine, Hann, beta,
    edge‑weighted, generalized‑normal, power‑law, cosine, sinc,
    exponential, Dirichlet, sparse).  The best ~15 are retained.

2.  A temperature‑smoothed Adam optimiser (large initial τ, slow decay)
    applied to each top seed.

3.  Exponentiated‑gradient descent on the exact max‑convolution objective
    (sub‑gradient of the active entry).

4.  Frank‑Wolfe with a cheap line‑search (move towards the coordinate
    with smallest gradient).

5.  Gradient‑biased pairwise annealing (mass transfer from high‑gradient
    to low‑gradient indices) for the remaining time.

All stages respect the 1000 s budget and always return a plain
``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation of a (len n) with w (len m) for non‑negative shifts.
    Returns an array of length n where element i equals Σ_j a_j·w_{j+i}.
    Implemented via FFT using convolution with the reversed weight vector.
    """
    n = a.shape[0]
    m = w.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()
    rev_w = w[::-1]
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_w, n=size)
    conv = np.fft.irfft(A * B, n=size)
    start = m - 1
    return conv[start:start + n]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return the exact value of ``evaluate_sequence`` for a (unnormalised)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Sub‑gradient of the active convolution entry
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """Sub‑gradient of the currently maximal self‑convolution entry."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))
    grad = np.zeros_like(a)
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
# Analytic families (return probability vectors, i.e. sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(-|x-½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.cos(np.pi * (i - (n - 1) / 2) / (n - 1))
    w = np.maximum(w, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_sinc(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = (i - n / 2) / n
    # sinc(x) = sin(pi*x)/(pi*x), safe at 0
    with np.errstate(divide='ignore', invalid='ignore'):
        w = np.sinc(x)
    w = np.maximum(w, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_exponential(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    w = np.exp(-lam * np.abs(x - 0.5))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Adam optimisation of a temperature‑smoothed max‑conv objective
# ----------------------------------------------------------------------
def _smooth_adam(a0: np.ndarray,
                 deadline: float,
                 best_seq: np.ndarray,
                 best_score: float,
                 tau_start: float = 0.5,
                 tau_end: float = 0.0005,
                 tau_decay: float = 0.9995,
                 step_size0: float = 0.1,
                 eval_every: int = 150) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed objective
        2·n·τ·log‑sum‑exp(conv/τ)
    The temperature τ is geometrically decayed.
    """
    n = a0.shape[0]
    x = np.log(a0 + 1e-15)                     # logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    tau = tau_start

    while time.time() < deadline and tau > tau_end:
        t += 1
        # softmax probabilities
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # smoothed max‑conv
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        exp_vals = np.exp((conv - max_conv) / tau)
        w = exp_vals / exp_vals.sum()          # softmax over convolution entries

        # gradient wrt a (4·n·correlation)
        corr = _fft_corr(a, w)                 # length n
        grad_a = 4.0 * n * corr

        # chain rule to logits
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = step_size0 / (np.sqrt(v_hat) + eps)
        x = x - lr * m_hat

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam τ={tau:.5f} new best {best_score:.6f}")

        # decay temperature
        tau = max(tau * tau_decay, tau_end)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the exact max‑conv objective
# ----------------------------------------------------------------------
def _exp_gradient(a: np.ndarray,
                  deadline: float,
                  best_seq: np.ndarray,
                  best_score: float,
                  eta0: float = 1.0,
                  eval_every: int = 80) -> tuple[np.ndarray, float]:
    """
    Multiplicative updates a ← a·exp(−η·∇maxconv) projected onto the simplex.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Frank‑Wolfe with a cheap line‑search
# ----------------------------------------------------------------------
def _frank_wolfe_line_search(a: np.ndarray,
                             deadline: float,
                             best_seq: np.ndarray,
                             best_score: float,
                             eval_every: int = 100) -> tuple[np.ndarray, float]:
    """
    Classic FW: direction = e_i – a where i = argmin grad.
    The step size γ is chosen from a small discrete set to minimise the exact objective.
    """
    n = a.shape[0]
    t = 0
    gamma_grid = np.array([0.0, 0.005, 0.02, 0.07, 0.15, 0.3, 0.6, 1.0])
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        i_min = int(np.argmin(grad))
        # candidate solutions for each γ
        best_local = best_score
        best_a = a
        for gamma in gamma_grid[1:]:  # skip γ=0 (no move)
            a_new = a * (1.0 - gamma)
            a_new[i_min] += gamma
            cur = _objective(a_new)
            if cur < best_local:
                best_local = cur
                best_a = a_new
        if best_local < best_score - 1e-14:
            best_score = best_local
            best_seq = best_a.copy()
            print(f"[{time.time():.1f}] FW it={t} new best {best_score:.6f}")
        a = best_a  # continue from the best local point

        if t % eval_every == 0:
            # occasional sanity check (optional)
            pass

    return best_seq, best_score


# ----------------------------------------------------------------------
# Gradient‑biased pairwise annealing
# ----------------------------------------------------------------------
def _biased_pairwise_anneal(a: np.ndarray,
                            deadline: float,
                            best_seq: np.ndarray,
                            best_score: float,
                            rng: np.random.Generator,
                            eval_every: int = 100) -> tuple[np.ndarray, float]:
    """
    Choose source index proportionally to its positive gradient (mass that
    hurts the max‑conv) and destination uniformly among the rest.
    Accept if improves, otherwise with a temperature schedule.
    """
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    it = 0
    while time.time() < deadline:
        it += 1
        grad, _ = _grad_of_max_conv(a)
        # source distribution: weight by grad (add small constant to avoid zeros)
        src_weights = grad + 1e-12
        src_weights_sum = src_weights.sum()
        if src_weights_sum == 0.0:
            src_idx = rng.integers(n)
        else:
            src_idx = rng.choice(n, p=src_weights / src_weights_sum)

        # choose a different destination uniformly
        dst_idx = rng.integers(n - 1)
        if dst_idx >= src_idx:
            dst_idx += 1

        # mass to transfer (up to 40% of source mass)
        delta = a[src_idx] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        a_new = a.copy()
        a_new[src_idx] -= delta
        a_new[dst_idx] += delta

        cur = _objective(a_new)
        # temperature linearly decays to zero
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = a_new
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal it={it} src={src_idx}->dst={dst_idx} Δ={delta:.3e} new best {best_score:.6f}")

        if it % eval_every == 0:
            # occasional status
            pass

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (piecewise‑constant replication)
# ----------------------------------------------------------------------
def _upsample_piecewise(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Upsample by splitting each entry into (new_n // old_n) copies.
    Remaining slots (if any) are filled by linear interpolation of the tail.
    The result stays on the simplex.
    """
    old_n = a.shape[0]
    factor = new_n // old_n
    remainder = new_n % old_n
    # replicate each entry `factor` times
    a_rep = np.repeat(a, factor)
    if remainder > 0:
        # linear interpolation for the missing tail
        tail = np.linspace(a[-1], a[-1], remainder)  # simply repeat the last value
        a_rep = np.concatenate([a_rep, tail])
    # renormalise (numerical errors)
    if a_rep.sum() == 0.0:
        a_rep[:] = 1.0 / new_n
    else:
        a_rep /= a_rep.sum()
    return a_rep


# ----------------------------------------------------------------------
# Candidate generation (diverse analytic families)
# ----------------------------------------------------------------------
def _generate_candidates(rng: np.random.Generator, lengths: list[int]) -> list[tuple[float, np.ndarray]]:
    """Return a list of (score, vector) for a wide variety of seeds."""
    candidates = []
    for n in lengths:
        # simple deterministic families
        for gen in (_distribution_uniform,
                    _distribution_arcsine,
                    _distribution_hann,
                    _distribution_cosine,
                    _distribution_sinc):
            vec = gen(n)
            sc = _objective(vec)
            candidates.append((sc, vec))

        # beta & edge‑weighted families, mixing with uniform
        for p in np.logspace(-0.5, 0.5, 6):
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            for base in (beta, edge):
                for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
                    vec = _mix_with_uniform(base, lam)
                    sc = _objective(vec)
                    candidates.append((sc, vec))

        # generalized normal shapes
        for q in (0.5, 0.9, 1.2, 1.6, 2.0):
            vec = _distribution_generalized_normal(n, q)
            sc = _objective(vec)
            candidates.append((sc, vec))

        # power‑law
        for pwr in (0.2, 0.5, 0.8, 1.1, 1.5, 2.0):
            vec = _distribution_power_law(n, pwr)
            sc = _objective(vec)
            candidates.append((sc, vec))

        # exponential shape
        for lam in (1.0, 2.0, 4.0, 8.0):
            vec = _distribution_exponential(n, lam)
            sc = _objective(vec)
            candidates.append((sc, vec))

        # random Dirichlet
        for _ in range(3):
            alpha = rng.uniform(0.01, 2.0)
            vec = rng.dirichlet(np.full(n, alpha))
            sc = _objective(vec)
            candidates.append((sc, vec))

        # random sparse
        for _ in range(3):
            vec = _distribution_random_sparse(n, rng, density=0.01)
            sc = _objective(vec)
            candidates.append((sc, vec))

    # keep only the best few
    candidates.sort(key=lambda x: x[0])
    return candidates[:15]   # return at most 15 top seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that (approximately) minimises
    ``evaluate_sequence``.  The routine respects a 1000 s budget and
    returns a plain list ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # a small safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. Generate a diverse pool of analytic seeds
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192, 16384]
    top_candidates = _generate_candidates(rng, lengths)

    print(f"[{time.time():.1f}] Seed phase: {len(top_candidates)} candidates kept")
    for sc, vec in top_candidates:
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = vec.copy()
            print(f"[{time.time():.1f}] New global best (seed) score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. If a previous construction exists, inject it
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            top_candidates.append((sc, prev.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] Previous best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Guard against empty result
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 4096
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] Fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Adam smoothing on the elite seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        elite = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining * 0.45 / max(1, len(elite))
        print(f"[{time.time():.1f}] Adam smoothing on {len(elite)} elite seeds")
        for idx, seed_seq in enumerate(elite):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_adam(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                tau_start=0.5,
                tau_end=0.0005,
                tau_decay=0.9995,
                step_size0=0.12,
                eval_every=200,
            )
            if sc_opt < overall_best_score - 1e-13:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] After Adam seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Exact exponentiated‑gradient descent on the current best
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 8.0:
        sub_deadline = time.time() + remaining * 0.25
        best_seq, best_score = _exp_gradient(
            overall_best_seq.copy(),
            sub_deadline,
            overall_best_seq,
            overall_best_score,
            eta0=1.0,
            eval_every=60,
        )
        overall_best_seq, overall_best_score = best_seq, best_score
        print(f"[{time.time():.1f}] After EG best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Frank‑Wolfe with line search (few iterations)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fw_deadline = time.time() + (deadline - time.time()) * 0.15
        overall_best_seq, overall_best_score = _frank_wolfe_line_search(
            overall_best_seq.copy(),
            fw_deadline,
            overall_best_seq,
            overall_best_score,
            eval_every=80,
        )
        print(f"[{time.time():.1f}] After FW best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Upsample (piecewise‑constant) and brief Adam refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n <= 65536:
            print(f"[{time.time():.1f}] Upsampling {cur_n}→{target_n}")
            up_seq = _upsample_piecewise(overall_best_seq, target_n)
            up_score = _objective(up_seq)
            if up_score < overall_best_score:
                overall_best_seq = up_seq
                overall_best_score = up_score
                print(f"[{time.time():.1f}] Upsampled new best {overall_best_score:.6f}")

            # short Adam on upsampled vector
            sub_deadline = time.time() + (deadline - time.time()) * 0.10
            a_opt, sc_opt = _smooth_adam(
                overall_best_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                tau_start=0.3,
                tau_end=0.0003,
                tau_decay=0.999,
                step_size0=0.09,
                eval_every=150,
            )
            if sc_opt < overall_best_score - 1e-13:
                overall_best_seq = a_opt.copy()
                overall_best_score = sc_opt
                print(f"[{time.time():.1f}] Adam post‑upsample new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final gradient‑biased pairwise annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] Final biased annealing")
        a_opt, sc_opt = _biased_pairwise_anneal(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
            eval_every=120,
        )
        if sc_opt < overall_best_score - 1e-13:
            overall_best_seq = a_opt.copy()
            overall_best_score = sc_opt
            print(f"[{time.time():.1f}] After final anneal new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Report final result
    # ------------------------------------------------------------------
    print(
        f"[{time.time():.1f}] FINAL best objective = {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
