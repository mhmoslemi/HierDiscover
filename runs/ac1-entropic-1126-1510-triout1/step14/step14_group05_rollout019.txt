def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]
```

def _hill_climb(a0, time_limit, best_seq, best_score):
    a = a0.copy()
    n = len(a)
    while time.time() < time_limit:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        if a[i] < 1e-12:
            continue
        # propose move
        delta = a[i] * rng.random() * 0.5
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective_scaled(new_a)
        if cur < best_score - 1e-12:
            best_score = cur
            best_seq = new_a.copy()
            a = new_a
            print(f"[...] hillclimb improved to {best_score:.6f}")
    return best_seq, best_score

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of *v* onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _objective_exact(a: np.ndarray) -> float:
    """Exact evaluate_sequence score for a (may be used as a cheap proxy)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, i.e. Σ = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_beta_edge(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta‑type density w_i ∝ (i·(n‑i))^{α‑1} (α>0)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = (i * (n - i)) ** (alpha - 1.0)
    vals = np.clip(vals, 0.0, None)
    s = vals.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    vals /= s
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i·(n‑i))^{‑p}  (p>0)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = (i * (n - i)) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    w /= w_sum
    return w


def _distribution_power_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ i^{‑p} + (n‑i)^{‑p}  (p>0)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = (i ** (-p)) + ((n - i) ** (-p))
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    w /= w_sum
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_log_weighted(n: int) -> np.ndarray:
    """w_i ∝ log( n / i ) + log( n / (n‑i) ).  Edges get the most weight."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.log(n / i) + np.log(n / (n - i))
    w = np.clip(w, 0.0, None)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    w /= w_sum
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.007) -> np.ndarray:
    """Sparse Dirichlet‑like seed with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smooth (soft‑max) gradient descent – works on the exact L∞ objective
# ----------------------------------------------------------------------
def _smooth_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    tau_start: float = 0.05,
    tau_end: float = 1e-5,
    lr0: float = 0.7,
    eval_every: int = 250,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    it = 0
    while time.time() < time_limit:
        it += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        cur_score = float(2.0 * n * max_val / (a.sum() ** 2))
        if cur_score < best_score - 1e-14:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time():.1f}] smooth it={it} new global best {best_score:.6f}")

        # linear temperature annealing
        frac = (time.time() - start) / (time_limit - start)
        tau = tau_start * (1.0 - frac) + tau_end * frac

        # soft‑max over convolution entries
        maxc = conv.max()
        shifted = conv - maxc
        exp_vals = np.exp(shifted / tau)          # numerically stable
        w = exp_vals / exp_vals.sum()             # length 2n‑1, sums to 1

        # gradient of the smoothed objective
        # term1 = Σ w[k+i]·a[i]  (cross‑correlation)
        term1 = np.fft.irfft(np.fft.rfft(w) * np.conj(np.fft.rfft(a, n=w.shape[0])), n=w.shape[0])
        term1 = term1[:n]                         # valid region
        # term2 = a * Σ_{window} w
        win_sum = np.convolve(w, np.ones(n, dtype=np.float64), mode='valid')
        grad = term1 + a * win_sum

        # EG update
        eta = lr0 / math.sqrt(it)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if it % eval_every == 0 and time.time() >= time_limit:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (used for polishing)
# ----------------------------------------------------------------------
def _subgradient_descent_exact(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = float(2.0 * n * max_val / (a.sum() ** 2))
            if cur < best_score - 1e-14:
                best_score, best_seq = cur, a.copy()
                print(f"[{time.time():.1f}] exact it={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Randomised pairwise hill‑climber (final polishing)
# ----------------------------------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    while time.time() < time_limit:
        src = np.where(a > 1e-15)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue
        delta = a[i] * rng.random() * 0.45   # up‑to 45 % of source mass
        if delta < 1e-15:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective_exact(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score, best_seq = cur, a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a step‑function (plain Python list) that minimises
    ``evaluate_sequence`` within the allotted time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, seq)

    # ------------------------------------------------------------------
    # 1. Broad scan of analytic families on several lengths
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 12000, 15000]

    # parameter grids (dense where it matters)
    p_grid = np.concatenate(
        [
            np.logspace(-4, -0.5, 20),   # tiny → moderate
            np.logspace(-0.5, 0.5, 15),  # up to 3
            np.linspace(0.1, 4.0, 12),   # linear region
        ]
    )
    alpha_grid = np.concatenate(
        [
            np.logspace(np.log10(0.01), np.log10(0.2), 12),
            np.linspace(0.2, 0.5, 8),
        ]
    )
    c_grid = np.logspace(-4, 0, 14)   # exponential‑edge decay
    lam_grid = np.linspace(0.0, 1.0, 21)  # mixing with uniform

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families (no parameters)
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
            _distribution_log_weighted,
        ):
            cand = base_fn(n)
            sc = _objective_exact(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # beta‑type families
        for alpha in alpha_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_edge(n, float(alpha))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(beta, lam)
                sc = _objective_exact(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}] len={n} β α={alpha:.4g} lam={lam:.2f} score={sc:.6f}"
                    )

        # edge‑weighted product families (i·(n‑i))^{-p}
        for p in p_grid:
            if time.time() >= deadline:
                break
            edge = _distribution_edge_weighted(n, float(p))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(edge, lam)
                sc = _objective_exact(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}] len={n} edge p={p:.4g} lam={lam:.2f} score={sc:.6f}"
                    )

        # power‑law sum families (i^{-p} + (n‑i)^{-p})
        for p in p_grid:
            if time.time() >= deadline:
                break
            pw = _distribution_power_weighted(n, float(p))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(pw, lam)
                sc = _objective_exact(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}] len={n} power p={p:.4g} lam={lam:.2f} score={sc:.6f}"
                    )

        # exponential‑edge families
        for c in c_grid:
            if time.time() >= deadline:
                break
            exp_edge = _distribution_exponential_edge(n, float(c))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(exp_edge, lam)
                sc = _objective_exact(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}] len={n} exp c={c:.4g} lam={lam:.2f} score={sc:.6f}"
                    )

        # random Dirichlet seeds (varying concentration)
        for _ in range(12):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 1.5)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective_exact(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(
                    f"[{time.time():.1f}] Dirichlet len={n} α={alpha:.4g} score={sc:.6f}"
                )

        # sparse random seeds
        for _ in range(8):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective_exact(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(
                    f"[{time.time():.1f}] sparse len={n} score={sc:.6f}"
                )

        # keep only the most promising candidates (memory bound)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:80]

    # ------------------------------------------------------------------
    # 2. Incorporate a user‑provided previous construction if available
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            sc = _objective_exact(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. If nothing found, fallback to a simple uniform vector
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective_exact(overall_best_seq)
        print(
            f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 4. Continuous optimisation from the best few seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 40.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] smooth & sub‑gradient on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            # smooth descent – 60 % of allocated time
            sg_deadline = time.time() + per_seed * 0.6
            a_opt, sc_opt = _smooth_descent(
                seed_seq,
                sg_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                tau_start=0.05,
                tau_end=1e-5,
                lr0=0.7,
                eval_every=250,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after smooth seed {idx} new best {overall_best_score:.6f}"
                )
            # exact sub‑gradient polishing – remaining 40 %
            sg_deadline = time.time() + per_seed * 0.4
            a_opt, sc_opt = _subgradient_descent_exact(
                overall_best_seq,
                sg_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                eval_every=200,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after exact seed {idx} new best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5. Final stochastic hill‑climber (pairwise swaps)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final pairwise hill‑climber")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best (2·n·max(conv)/sum²) = {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list – ``evaluate_sequence`` will handle any scaling.
    return overall_best_seq.tolist()
