"""
Simulated‑annealing optimizer for the step‑function objective.

The algorithm:
1. Load the previously discovered best sequence (``height_sequence_1``) if
   available, otherwise start from a random Dirichlet seed.
2. Optionally generate a few extra random Dirichlet seeds and keep the best.
3. Perform a long simulated‑annealing (SA) run that moves a random
   fraction of mass between two positions.  The SA temperature is
   linearly annealed from `T0≈0.02` to a tiny final value.
   Two move types are used:
   * a simple random source → destination transfer,
   * a “targeted” move that preferentially removes mass from indices
     contributing to the current maximum of the convolution.
4. When the time budget is almost exhausted we apply a short
   smoothed‑max mirror‑descent (Adam on the log‑sum‑exp surrogate) to
   polish the best candidate.
5. The best normalised vector is returned – ``evaluate_sequence`` will
   handle the final scaling internally.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
#  FFT based convolution (real inputs)
# ------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a * a using a real FFT."""
    return _fft_convolve(a, a)


# ------------------------------------------------------------
#  Exact objective (sum(a) = 1)
# ------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) for a with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ------------------------------------------------------------
#  Projection onto the probability simplex (used by the smooth‑max step)
# ------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of `v` onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
#  Smoothed‑max mirror descent (Adam + β‑continuation)
# ------------------------------------------------------------
def _smooth_max_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """Minimise the exact objective using a smooth log‑sum‑exp surrogate."""
    a = a0.copy()
    n = a.shape[0]
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    # β schedule (log‑space)
    beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0]

    for beta in beta_schedule:
        while time.time() < time_limit:
            t += 1
            conv = _fft_self_convolve(a)               # full self‑convolution
            max_val = conv.max()
            shifted = conv - max_val                   # ≤ 0
            exp_beta = np.exp(beta * shifted)
            w = exp_beta / exp_beta.sum()              # Σw = 1, w ≥ 0

            # gradient of the smooth max: 2·(w ∗ a_rev)
            a_rev = a[::-1]
            grad_conv = _fft_convolve(w, a_rev)        # length 3n‑2
            grad = 2.0 * grad_conv[n - 1 : n - 1 + n]   # size n

            grad_J = 2.0 * n * grad                     # ∇J

            # Adam update, then project onto the simplex
            m = beta1 * m + (1.0 - beta1) * grad_J
            v = beta2 * v + (1.0 - beta2) * (grad_J ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = 0.15 / (np.sqrt(v_hat) + eps)
            a -= step * m_hat
            a = _project_onto_simplex(a, s=1.0)

            # occasional exact evaluation
            if t % 120 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-14:
                    best_score, best_seq = cur, a.copy()
                    print(
                        f"[{time.time():.1f}] Smooth β={beta:.1f} it={t} new best {best_score:.6f}"
                    )
            # keep each β for a limited number of steps
            if t % 2500 == 0:
                break
        if time.time() >= time_limit:
            break
    return best_seq, best_score
