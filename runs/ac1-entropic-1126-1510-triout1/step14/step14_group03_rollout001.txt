```python
def _distribution_beta(n, alpha):
    i = np.arange(1, n+1, dtype=np.float64) # indices 1..n
    x = i / (n+1)  # open interval (0,1)
    vals = (x ** (alpha - 1)) * ((1 - x) ** (alpha - 1))
    vals_sum = vals.sum()
    if vals_sum == 0:
        return np.full(n, 1.0 / n)
    else:
        return vals / vals_sum
```

"""
Optimise a non‑negative step function for the objective

    2·n·max(conv(a,a)) / (sum(a)^2)

using a large seed pool (analytic families + random Dirichlet) and a
two‑phase local search:
  * Exponentiated‑gradient (mirror) descent on the simplex.
  * Simulated‑annealing polish with both pairwise mass transfers and
    multi‑bin redistributions.

Additional families (Beta distributions) and mixing of the arcsine family
with uniform are introduced to enlarge the seed pool.
The algorithm respects a wall‑time budget (default 1000 s) and prints
progress information.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  FFT‑based self‑convolution (real inputs, full linear convolution)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()        # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
#  Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) / (sum(a)^2)  –  a is assumed non‑negative."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
#  Analytic families – all return a *probability* vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i·(n‑i))^{‑p}   with i = distance to the left edge."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_sum_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ i^{‑p} + (n‑i)^{‑p} (sum of two one‑sided tails)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (-p)) + (j ** (-p))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp( -|x‑½|^{q} ) on the discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta(n: int, alpha: float) -> np.ndarray:
    """Beta(α,α) density on {1,…,n} – includes arcsine for α=0.5."""
    i = np.arange(1, n + 1, dtype=np.float64)
    x = i / (n + 1)                     # open interval (0,1)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (alpha - 1.0))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1‑lam) + uniform·lam."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
#  Gradient of the *active* convolution entry (sub‑gradient of max)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """Return (gradient, index_of_max) for the current max entry of a*a."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))

    grad = np.zeros_like(a)
    if k < n:
        # overlap length = k+1   (i runs 0..k)
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        # k ≥ n : overlap length = 2·n‑1‑k   (i runs k‑(n‑1)..n‑1)
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
#  Exponentiated‑gradient descent (multiplicative mirror descent)
# ----------------------------------------------------------------------
def _exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """Run EG descent on the exact objective until ``deadline``."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Pairwise‑mass‑transfer simulated annealing
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins; accept improvements."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4        # move at most 40 % of source mass
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Multi‑bin mass‑transfer annealing (larger moves)
# ----------------------------------------------------------------------
def _multi_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    factor: float = 0.12,
) -> tuple[np.ndarray, float]:
    """Redistribute mass among ≥2 randomly chosen bins; Metropolis acceptance."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    while time.time() < deadline:
        k = rng.integers(2, 7)                # 2 … 6 bins
        idx = rng.choice(n, size=k, replace=False)
        mass = a[idx].sum()
        if mass == 0.0:
            continue

        # propose a new allocation via a symmetric Dirichlet (α=1)
        proposal = rng.random(k)
        proposal = proposal / proposal.sum() * mass

        new_a = a.copy()
        new_a[idx] = proposal

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * factor

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] multi‑anneal k={k} new best {best_score:.6f}")
    return best_seq, best_score

        # mixing arcsine with uniform
        for lam in lam_grid:
            cand = _mix_with_uniform(arcsine, lam)
            sc = _objective(cand)
            top.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best arcsine+uniform lam={lam:.2f} score={sc:.6f}")

        # ---- Edge‑weighted families ---------------------------------
        for p in p_grid:
            if time.time() >= deadline:
                break

            # product edge‑weighted
            base = _distribution_edge_weighted(n, p)
            sc = _objective(base)
            top.append((sc, base.copy()))
            if sc < best_score:
                best_score, best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new best edge p={p:.3g} score={sc:.6f}")

            for lam in lam_grid:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                top.append((sc, cand.copy()))
                if sc < best_score:
                    best_score, best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new best edge p={p:.3g} lam={lam:.2f} score={sc:.6f}")

            # sum‑edge‑weighted (new family)
            base2 = _distribution_edge_sum_weighted(n, p)
            sc = _objective(base2)
            top.append((sc, base2.copy()))
            if sc < best_score:
                best_score, best_seq = sc, base2.copy()
                print(f"[{time.time():.1f}] new best sum‑edge p={p:.3g} score={sc:.6f}")

            for lam in lam_grid:
                cand = _mix_with_uniform(base2, lam)
                sc = _objective(cand)
                top.append((sc, cand.copy()))
                if sc < best_score:
                    best_score, best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new best sum‑edge p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # ---- Generalised normal -------------------------------------
        for q in q_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_generalized_normal(n, q)
            sc = _objective(cand)
            top.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best gen‑norm q={q:.2f} score={sc:.6f}")

        # ---- Beta family --------------------------------------------
        for alpha in alpha_grid:
            if time.time() >= deadline:
                break
            base = _distribution_beta(n, alpha)
            sc = _objective(base)
            top.append((sc, base.copy()))
            if sc < best_score:
                best_score, best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new best beta α={alpha:.3g} score={sc:.6f}")

            for lam in lam_grid:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                top.append((sc, cand.copy()))
                if sc < best_score:
                    best_score, best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new best beta α={alpha:.3g} lam={lam:.2f} score={sc:.6f}")

        # ---- Random sparse Dirichlet seeds -------------------------
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = np.zeros(n, dtype=np.float64)
            k = max(2, int(n * 0.01))                     # ~1 % non‑zero
            idx = rng.choice(n, size=k, replace=False)
            cand[idx] = rng.random(k)
            cand /= cand.sum()
            sc = _objective(cand)
            top.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best sparse score={sc:.6f}")

        # keep only the very best seeds (size 20)
        top.sort(key=lambda x: x[0])
        top = top[:20]

    # ------------------------------------------------------------------
    # 2.  Include a previously submitted construction if present
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < best_score:
                best_score, best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Guard against an empty result
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _distribution_uniform(5000)
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Exponentiated‑gradient refinement (multiple η₀ values)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate three equal slots for different η₀ values
        eta_candidates = [0.6, 0.9, 1.2]
        slot = remaining / (len(eta_candidates) + 1)   # leave a slot for annealing
        cur_seq = best_seq.copy()
        cur_score = best_score
        for eta0 in eta_candidates:
            sub_deadline = time.time() + slot
            print(f"[{time.time():.1f}] EG refinement η₀={eta0:.2f} (budget {slot:.1f}s)")
            cur_seq, cur_score = _exp_gradient_descent(
                cur_seq, sub_deadline, cur_seq, cur_score, eta0=eta0, eval_every=60
            )
            print(f"[{time.time():.1f}] after η₀={eta0:.2f} best score {cur_score:.6f}")
        best_seq, best_score = cur_seq, cur_score

    # ------------------------------------------------------------------
    # 5.  Pairwise annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] pairwise annealing")
        best_seq, best_score = _pairwise_annealing(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            rng,
        )
        print(f"[{time.time():.1f}] after pairwise best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Multi‑bin annealing polish (larger moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] multi‑bin annealing")
        best_seq, best_score = _multi_annealing(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            rng,
            factor=0.12,               # a bit hotter than pairwise
        )
        print(f"[{time.time():.1f}] after multi‑bin best score {best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective = 2·n·max(conv)/sum²): {best_score:.6f} "
        f"(length={len(best_seq)})"
    )
    return best_seq.tolist()
