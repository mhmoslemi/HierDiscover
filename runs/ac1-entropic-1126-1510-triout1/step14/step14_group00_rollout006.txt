from scipy.signal import correlate

def _smooth_gradient(a: np.ndarray, tau: float) -> np.ndarray:
    conv = _fft_self_convolve(a)  # length 2n-1
    # compute weights
    # to avoid overflow, subtract max before exp
    exp = np.exp(tau * conv - np.max(tau * conv))
    w = exp / exp.sum()
    # compute gradient
    grad = 2 * correlate(w, a, mode='valid')
    return grad

def _smooth_exp_gradient(a0, deadline, best_seq, best_score, tau0=0.5, tau_max=80, tau_inc=1.2, eta0=0.7, beta=0.9, eval_every=20):
    a = a0.copy()
    n = len(a)
    m = np.zeros_like(a) # EMA of gradient
    eta = eta0
    tau = tau0
    t = 0
    while time.time() < deadline:
        t+=1

        # maybe increase tau gradually
        if t % 200 == 0:
            tau = min(tau_max, tau * tau_inc)

        grad = _smooth_gradient(a, tau)
        m = beta * m + (1-beta) * grad

        a = a * np.exp(-eta * m)
        # renormalize
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= total

        if t % eval_every == 0:
            cur_score = _objective(a)
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_seq = a.copy()
                # increase eta a bit
                eta = min(eta0, eta * 1.02)
            else:
                eta *= 0.995

        # occasionally project onto simplex to correct drift
        if t % (eval_every * 10) == 0:
            a = _project_onto_simplex(a, s=1.0)
    return best_seq, best_score

"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (∑a)²

using a multi‑phase optimisation:
    1. Diverse analytic seeds (uniform, arcsine, Hann, edge‑weighted,
       Beta, Generalised‑Normal, exponential, random Dirichlet,
       sparse random).
    2. Smooth exponentiated‑gradient descent.  The max in the objective
       is approximated by a soft‑max (τ‑scaled log‑sum‑exp) which yields
       a dense sub‑gradient and helps escape local minima.  τ is raised
       gradually (continuation method) while an EMA of the gradient and
       an adaptive step size are used.
    3. Once a good vector is found, it is up‑sampled (linear interpolation)
       and the smooth optimiser is run again on the higher resolution.
    4. A final pair‑wise mass‑transfer simulated annealing polishing.
    5. Small components are pruned (they only enlarge n without improving
       the convolution peak).

All steps respect the 1000 s budget (2 s safety margin).  The returned
list is ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used occasionally to avoid drift.
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, len(v) + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based self‑convolution (real inputs)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective for a probability vector (Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families – each returns a *probability* vector (Σ=1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # avoid overflow/underflow for extreme α
    w = np.where(
        (x == 0) | (x == 1),
        0.0,
        np.exp((alpha - 1) * np.log(x) + (alpha - 1) * np.log1p(-x))
    )
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_exponential(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i) + 0.5
    w = np.exp(-lam * dist)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.02) -> np.ndarray:
    """Sparse random vector with given density (relative to n)."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1‑λ) + uniform·λ."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

    A_rev = np.fft.rfft(a[::-1], n=size)
    conv_full = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    return conv_full[start:start + n]


# ----------------------------------------------------------------------
# Smooth (soft‑max) gradient of the objective
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, tau: float) -> np.ndarray:
    """Return ∇  ( soft‑max_τ  (a*a) ).  Gradient = 2·corr(w,a)."""
    conv = _fft_self_convolve(a)                               # length 2n‑1
    # soft‑max weights – numerically stable
    shifted = tau * conv
    max_shift = shifted.max()
    exp_shift = np.exp(shifted - max_shift)
    w = exp_shift / exp_shift.sum()                            # still length 2n‑1
    # correlation w ⋆ a (valid mode)
    grad = 2.0 * _valid_correlation_fft(w, a)
    return grad


# ----------------------------------------------------------------------
# Adaptive exponentiated‑gradient on the *smooth* objective
# ----------------------------------------------------------------------
def _smooth_exp_gradient(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    tau0: float = 0.8,
    tau_max: float = 40.0,
    tau_mult: float = 1.5,
    eta0: float = 0.7,
    beta: float = 0.9,
    eval_every: int = 30,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the soft‑max approximation.
    τ is increased geometrically (continuation) up to tau_max.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)          # EMA of gradients
    eta = eta0
    tau = tau0
    t = 0
    last_improve = time.time()
    while time.time() < deadline:
        t += 1
        # gradient of smoothed objective
        grad = _smooth_gradient(a, tau)

        # EMA of gradients
        m = beta * m + (1.0 - beta) * grad

        # multiplicative update (exponentiated gradient)
        a = a * np.exp(-eta * m)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # periodically evaluate the *true* objective
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                eta = min(eta0, eta * 1.02)           # reward
                last_improve = time.time()
                print(f"[{time.time():.1f}] smooth EG it={t:,} new best {best_score:.6f}")
            else:
                eta *= 0.995                           # decay

        # increase τ every few hundred iterations (or after stagnation)
        if t % (eval_every * 10) == 0:
            if tau < tau_max:
                tau = min(tau_max, tau * tau_mult)

        # occasional re‑projection to combat numerical drift
        if t % (eval_every * 20) == 0:
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise mass‑transfer simulated annealing (final polishing)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        # pick source with non‑negligible mass
        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        src = rng.choice(src_candidates)
        # pick a different destination uniformly
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # move a random fraction (up to 30 %) of source mass
        delta = a[src] * rng.random() * 0.3
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        cur = _objective(new_a)

        elapsed = time.time() - start
        # simple linear cooling
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.02

        accept = False
        if cur < best_score - 1e-15:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal {src}->{dst} Δ={delta:.1e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserving total mass = 1)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new


# ----------------------------------------------------------------------
# Prune very small entries (they only make n larger).
# ----------------------------------------------------------------------
def _prune_small(a: np.ndarray, tol: float = 1e-5) -> np.ndarray:
    mask = a > tol
    if mask.sum() == 0:
        return a.copy()
    a_pruned = a[mask]
    a_pruned /= a_pruned.sum()
    return a_pruned


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    cand_pool: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1️⃣ Diverse seeds (multiple lengths)
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096, 8192]
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating seeds, n={n}")

        # deterministic families
        for base in (
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_hann(n),
        ):
            sc = _objective(base)
            cand_pool.append((sc, base.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new global best (det) {sc:.6f}")

        # edge‑weighted family with a wide grid of exponents
        p_vals = np.concatenate((
            np.logspace(-3.0, -0.5, 12, base=10),   # 0.001 → 0.316
            np.logspace(-0.5, 0.0, 12, base=10),    # 0.316 → 1
            np.linspace(0.1, 3.0, 15),              # linear region
            np.logspace(0.0, 0.8, 10, base=10)      # 1 → ~6.3
        ))
        lam_vals = np.linspace(0.0, 1.0, 9)
        for p in p_vals:
            if time.time() >= deadline:
                break
            base = _distribution_edge_weighted(n, float(p))
            for lam in lam_vals:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(base, float(lam))
                sc = _objective(cand)
                cand_pool.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] edge p={p:.3g} λ={lam:.2f} new best {sc:.6f}")

        # symmetric Beta family
        for alpha in np.concatenate((
            np.logspace(-2.5, -0.2, 14, base=10),   # heavy tails
            np.linspace(0.2, 2.5, 12),              # moderate
            np.linspace(2.5, 6.0, 9)                # central
        )):
            if time.time() >= deadline:
                break
            cand = _distribution_beta(n, float(alpha))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] beta α={alpha:.3g} new best {sc:.6f}")

        # generalized normal family
        for q in np.linspace(0.3, 5.0, 14):
            if time.time() >= deadline:
                break
            cand = _distribution_generalized_normal(n, float(q))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] gn q={q:.2f} new best {sc:.6f}")

        # exponential family (edge‑decay)
        for lam in np.linspace(0.01, 2.0, 12):
            if time.time() >= deadline:
                break
            cand = _distribution_exponential(n, float(lam))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] exp λ={lam:.2f} new best {sc:.6f}")

        # random Dirichlet seeds
        for _ in range(12):
            if time.time() >= deadline:
                break
            log_alpha = rng.uniform(-2.5, 1.0)   # α in [~0.003,10]
            alpha = 10 ** log_alpha
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] Dirichlet α={alpha:.3g} new best {sc:.6f}")

        # sparse random seeds
        for _ in range(8):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.015)
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] sparse new best {sc:.6f}")

        # keep only top‑30 seeds overall to bound memory
        cand_pool.sort(key=lambda x: x[0])
        cand_pool = cand_pool[:30]

    # ------------------------------------------------------------------
    # 2️⃣ Incorporate a possibly existing construction
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.abs(prev)
            s = prev.sum()
            if s == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            else:
                prev = prev / s
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
            cand_pool.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3️⃣ Fallback if nothing was generated
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 2048
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Short smooth EG on every seed (quick polishing)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0 and cand_pool:
        per_seed = max(0.3, remaining * 0.25 / max(len(cand_pool), 1))
        print(f"[{time.time():.1f}] quick smooth EG on {len(cand_pool)} seeds (~{per_seed:.2f}s each)")
        for idx, (sc, a0) in enumerate(cand_pool):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            a_opt, sc_opt = _smooth_exp_gradient(
                a0.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                tau0=0.8,
                tau_max=25.0,
                tau_mult=1.4,
                eta0=0.6,
                beta=0.9,
                eval_every=25,
            )
            if sc_opt < overall_best_score - 1e-13:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] quick EG seed {idx} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ Longer smooth EG on the current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_dead = time.time() + (deadline - time.time()) * 0.35
        a_opt, sc_opt = _smooth_exp_gradient(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            tau0=0.8,
            tau_max=35.0,
            tau_mult=1.3,
            eta0=0.5,
            beta=0.9,
            eval_every=30,
        )
        if sc_opt < overall_best_score - 1e-13:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] long smooth EG improved best to {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣ Progressive up‑sampling (double size while time permits)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n > 65536:
            break
        print(f"[{time.time():.1f}] up‑sampling {cur_n} → {target_n}")
        up = _upsample(overall_best_seq, target_n)
        up_sc = _objective(up)
        if up_sc < overall_best_score:
            overall_best_seq, overall_best_score = up, up_sc
            print(f"[{time.time():.1f}] up‑sampled new best {overall_best_score:.6f}")

        # medium smooth EG on the up‑sampled vector
        sub_dead = time.time() + (deadline - time.time()) * 0.20
        a_opt, sc_opt = _smooth_exp_gradient(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            tau0=1.0,
            tau_max=40.0,
            tau_mult=1.25,
            eta0=0.45,
            beta=0.9,
            eval_every=40,
        )
        if sc_opt < overall_best_score - 1e-13:
            overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
            print(f"[{time.time():.1f}] post‑upsample EG new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣ Final simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing phase")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        if sc_opt < overall_best_score - 1e-13:
            overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
            print(f"[{time.time():.1f}] after annealing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 8️⃣ Prune near‑zero entries (can only improve the n‑factor)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        for tol in [1e-6, 5e-6, 1e-5, 5e-5]:
            pruned = _prune_small(overall_best_seq, tol=tol)
            if pruned.shape[0] == overall_best_seq.shape[0]:
                continue
            sc = _objective(pruned)
            if sc < overall_best_score - 1e-13:
                overall_best_seq, overall_best_score = pruned, sc
                print(f"[{time.time():.1f}] prune tol={tol:.1e} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)) = {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    # ``evaluate_sequence`` expects a plain Python list
    return overall_best_seq.tolist()
