def _smooth_max_descent(a0, time_limit, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps=1e-8
    t=0
    beta_schedule = [...]
    for beta in beta_schedule:
        while time.time() < time_limit:
            t+=1
            conv = _fft_self_convolve(a)
            max_val = conv.max()
            # compute softmax weights
            shifted = conv - max_val
            exp_beta = np.exp(beta*shifted)
            w = exp_beta / exp_beta.sum()
            # gradient
            grad_conv = _fft_convolve(w, a[::-1])
            grad = 2.0*grad_conv[n-1:n-1+n]
            grad_J = 2.0*n*grad
            m = beta1*m + (1-beta1)*grad_J
            v = beta2*v + (1-beta2)*(grad_J**2)
            m_hat = m/(1-beta1**t)
            v_hat = v/(1-beta2**t)
            step = 0.2/(np.sqrt(v_hat)+eps)
            a -= step*m_hat

"""
Hybrid local‑search for a non‑negative step‑function that minimizes

    2·n·max(conv(a,a)) / (sum a)²

The algorithm consists of five stages:

1️⃣  **Broad seeding** – generate a diverse pool of candidates from
    analytic families (uniform, arcsine, sine, cosine, quadratic,
    edge‑weighted, exponential‑edge, symmetric & asymmetric beta)
    on several lengths, plus random Dirichlet and sparse seeds.
    The best few are retained.

2️⃣  **Smooth‑max mirror descent** – minimise a log‑sum‑exp surrogate
    of the max‐convolution using Adam on the simplex.
    The smoothing parameter β is increased gradually up to 2000,
    allowing the optimiser to escape local plateaus.

3️⃣  **Projected sub‑gradient descent** – using the exact sub‑gradient
    of the L∞‑objective (the “active‑set” gradient) and a simple
    projected‑gradient (additive) update with a diminishing stepsize.

4️⃣  **Exponentiated‑gradient refinement** – a short EG pass as a
    final “momentum‑free’’ cleanup.

5️⃣  **Greedy pairwise mass transfer** – many random mass‑moves
    (i→j) are tried; a move is kept only if it strictly lowers the
    exact objective.  This hill‑climber often yields the last few
    hundredths of improvement.

All phases stop as soon as the global time limit (`budget_s`)
is reached.  The returned list is already normalised (sum = 1);
`evaluate_sequence` will rescale it internally.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Projection of vector v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    rho = rho[-1] if rho.size else -1
    theta = 0.0 if rho == -1 else cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# ----------------------------------------------------------------------
#  FFT based convolutions (real)
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    return _fft_convolve(a, a)

# ----------------------------------------------------------------------
#  Exact objective (sum(a)=1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# ----------------------------------------------------------------------
#  Analytic families
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    w /= w.sum()
    return w


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    if w.sum() == 0.0:
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.008) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


# ----------------------------------------------------------------------
#  Smoothed‑max mirror descent (Adam + β continuation)
# ----------------------------------------------------------------------
def _smooth_max_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp surrogate, β grows to 2000."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    # extended β schedule
    beta_schedule = [
        0.5, 1.0, 2.0, 5.0, 10.0, 20.0,
        50.0, 100.0, 200.0, 500.0, 1000.0, 2000.0
    ]

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        # allow up to ~5000 internal steps per β (or until time runs out)
        while time.time() < deadline:
            t += 1
            conv = _fft_self_convolve(a)                # length 2n‑1
            max_val = conv.max()
            shifted = conv - max_val                    # ≤0
            exp_beta = np.exp(beta * shifted)
            w = exp_beta / exp_beta.sum()               # softmax weights, Σw=1

            # gradient of the smoothed max (see paper)
            grad_conv = _fft_convolve(w, a[::-1])       # length 3n‑2
            grad = 2.0 * grad_conv[n - 1 : n - 1 + n]   # size n

            grad_J = 2.0 * n * grad                     # gradient of J = 2·n·smooth_max

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_J
            v = beta2 * v + (1.0 - beta2) * (grad_J ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = 0.18 / (np.sqrt(v_hat) + eps)        # tuned base step‑size
            a -= step * m_hat
            a = _project_onto_simplex(a, s=1.0)

            # occasional exact evaluation
            if t % 120 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-15:
                    best_score, best_seq = cur, a.copy()
                    print(f"[{time.time():.1f}] β={beta:.1f} it={t} new best {best_score:.6f}")

            # give up on this β after a few thousand iters to keep schedule alive
            if t % 5000 == 0:
                break
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Projected additive sub‑gradient descent on the exact L∞‑objective
# ----------------------------------------------------------------------
def _additive_subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eval_every: int = 150,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = 0.9 / math.sqrt(t)          # diminishing stepsize
        a_new = a - eta * grad
        a_new = _project_onto_simplex(a_new, s=1.0)

        if t % eval_every == 0:
            cur = _objective(a_new)
            if cur < best_score - 1e-15:
                best_score, best_seq = cur, a_new.copy()
                print(f"[{time.time():.1f}] add‑subgrad it={t} new best {best_score:.6f}")

        a = a_new
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Exponentiated‑gradient refinement (same as before, but shorter)
# ----------------------------------------------------------------------
def _exp_subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eval_every: int = 120,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = 0.8 / math.sqrt(t)
        a *= np.exp(-eta * grad)          # multiplicative EG step
        s = a.sum()
        if s <= 0.0:
            a = _distribution_uniform(n)
        else:
            a /= s

        if t % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-15:
                best_score, best_seq = cur, a.copy()
                print(f"[{time.time():.1f}] exp‑subgrad it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
#  Greedy pairwise mass‑transfer hill‑climber
# ----------------------------------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.43
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-15:
            a = new_a
            best_score, best_seq = cur, a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
#  Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a step‑function (plain Python list) that minimises
    ``evaluate_sequence`` within the allotted time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # small safety margin
    deadline = start + budget_s - 2.0

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, vector)

    # ------------------------------------------------------------------
    # 1️⃣  Broad seeding
    # ------------------------------------------------------------------
    lengths = [1800, 2500, 3200, 4000, 6000, 8000, 12000]
    # parameter grids – extended compared with previous code
    p_grid = np.concatenate([
        np.logspace(-2, -0.5, 12),          # more edge‑heavy
        np.logspace(-0.5, 0.0, 12),         # moderate
        np.linspace(0.2, 4.0, 16),          # broader linear sweep
    ])
    lam_grid = np.linspace(0.0, 1.0, 13)    # finer mixing
    c_grid = np.logspace(-3, 0.2, 12)      # more exponential‑edge options
    alpha_grid = np.logspace(-1.5, 0.7, 14)   # includes α < 0.3
    beta_grid = np.logspace(-1.5, 0.7, 14)    # for asymmetric beta

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families -------------------------------------------------
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
        ):
            cand = base_fn(n)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # symmetric beta families -------------------------------------------------
        for alpha in alpha_grid:
            cand = _distribution_beta(n, float(alpha), float(alpha))
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] beta α={alpha:.4g} len={n} score={sc:.6f}")

        # asymmetric beta families (α ≠ β) ---------------------------------------
        for _ in range(6):
            a = 10 ** rng.uniform(-1.5, 0.7)
            b = 10 ** rng.uniform(-1.5, 0.7)
            cand = _distribution_beta(n, float(a), float(b))
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] asymmetric β α={a:.4g} β={b:.4g} len={n} score={sc:.6f}")

        # edge‑weighted + uniform mixes -----------------------------------------
        for p in p_grid:
            edge = _distribution_edge_weighted(n, float(p))
            for lam in lam_grid:
                cand = _mix_with_uniform(edge, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] edge p={p:.4g} lam={lam:.2f} len={n} score={sc:.6f}")

        # exponential‑edge family -------------------------------------------------
        for c in c_grid:
            exp_edge = _distribution_exponential_edge(n, float(c))
            for lam in lam_grid:
                cand = _mix_with_uniform(exp_edge, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] exp c={c:.4g} lam={lam:.2f} len={n} score={sc:.6f}")

        # random Dirichlet seeds -------------------------------------------------
        for _ in range(5):
            alpha = rng.uniform(0.01, 2.5)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] Dirichlet α={alpha:.3f} len={n} score={sc:.6f}")

        # sparse random seeds ----------------------------------------------------
        for _ in range(5):
            a = _distribution_random_sparse(n, rng, density=0.005)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

        # keep only the best 40 seeds overall (memory & time)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:40]

    # ------------------------------------------------------------------
    # 2️⃣  Include previously discovered best (if present)
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3️⃣  Fallback (should never trigger)
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform len={n} score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Smoothed‑max mirror descent on a handful of promising seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 60.0 and top_candidates:
        seed_vecs = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seed_vecs))
        print(f"[{time.time():.1f}] smoothed‑max descent on {len(seed_vecs)} seeds")
        for idx, seed_seq in enumerate(seed_vecs):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_max_descent(
                seed_seq,
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after smooth seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Projected additive sub‑gradient descent (exact objective)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 70.0:
        print(f"[{time.time():.1f}] additive sub‑gradient refinement")
        a_opt, sc_opt = _additive_subgradient_descent(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eval_every=150,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after additive subgrad new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Exponentiated‑gradient fine‑tuning
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 40.0:
        print(f"[{time.time():.1f}] exponentiated‑gradient fine‑tuning")
        a_opt, sc_opt = _exp_subgradient_descent(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eval_every=120,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after exp‑subgrad new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣  Final greedy pairwise polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy pairwise polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=None,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best objective (2·n·max(conv)) = {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return the raw probability vector (evaluate_sequence rescales internally)
    return overall_best_seq.tolist()
