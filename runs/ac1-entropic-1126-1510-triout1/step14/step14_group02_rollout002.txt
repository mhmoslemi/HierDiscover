"""
Propose a non‑negative step function (list of heights) that minimizes
``evaluate_sequence``.

Strategy
--------
1. **Initial seeds** – generate a variety of analytic families
   (uniform, arcsine, symmetric Beta, edge‑weighted, sine‑weighted)
   for moderate lengths (≈ 2‑4 k) and keep the best one.

2. **Low‑dimensional global search** – optimise a piece‑wise linear
   shape with a small number of control points using SciPy’s
   differential evolution (DE).  The shape is interpolated to a
   moderate length, symmetrised and normalised; DE explores the
   non‑convex landscape more globally than the gradient‑descent
   stages.

3. **Heavy mirror descent** – starting from the best seed (or the DE
   result) run an exponentiated‑gradient (entropy‑mirror) descent on
   the exact objective ``2·n·max(conv(a,a))`` (with ``sum(a)=1``).  The
   active set consists of the largest convolution entries; a momentum
   term (Adam‑like) and a decaying stepsize are used.

4. **Upsampling + refinement** – linearly up‑sample the vector to a
   larger size (≈ 2×) and repeat the mirror descent to exploit the
   finer discretisation.

5. **Final polish** – a weighted pairwise greedy refinement followed
   by a cheap simulated‑annealing local search.

All stages respect a total time budget (default 1000 s).  The function
returns a plain Python list of floats; the supplied ``evaluate_sequence``
will clip negatives / huge values automatically.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a with sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.006) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, where U is the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Force exact left‑right symmetry and re‑project onto the simplex."""
    sym = (seq + seq[::-1]) * 0.5
    total = sym.sum()
    if total == 0.0:
        return np.full_like(sym, 1.0 / len(sym))
    return sym / total

# Momentum mirror descent -----------------------------------------------
def _subgradient_descent_momentum(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    active_frac: float = 0.04,
    max_active: int = 30,
    beta: float = 0.9,
    eval_interval: int = 120,
    patience: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (entropy mirror) descent on the exact,
    non‑smooth objective.  The active set consists of those convolution
    entries whose value lies within ``active_frac`` of the current
    maximum.  Momentum (Adam‑like) smooths the sub‑gradient.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    ind_grid = np.arange(n, dtype=np.int32)

    while time.time() < deadline and t < 500_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            top_idx = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top_idx]

        grad = np.zeros_like(a)
        for k in act:
            shift = k - ind_grid
            mask = (shift >= 0) & (shift < n)
            grad[mask] += 2.0 * a[shift[mask]]

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * momentum)

        # Normalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} new global best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            # mild jitter to escape plateaus
            jitter = rng.exponential(scale=0.004, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

# Weighted pairwise refinement -------------------------------------------
def _pairwise_weighted_refine(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer guided by the location of the maximal convolution entry."""
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0

    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        conv = _fft_self_convolve(a)
        k_max = int(conv.argmax())

        # impact values L_i = a_{k_max-i}
        L = np.zeros(n, dtype=np.float64)
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        idx = np.arange(low, high + 1)
        L[idx] = a[k_max - idx]

        # pick source (high impact) and destination (low impact)
        if L.sum() > 0:
            probs_i = L / L.sum()
            i = rng.choice(n, p=probs_i)
        else:
            positives = np.where(a > 1e-12)[0]
            if positives.size == 0:
                break
            i = rng.choice(positives)

        Lmax = L.max()
        probs_j = (Lmax - L) + 1e-12
        probs_j /= probs_j.sum()
        j = rng.choice(n, p=probs_j)
        if j == i:
            j = (j + 1) % n

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] weighted swap {i}->{j} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score

# Cheap simulated annealing ---------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Simple Metropolis‑type local search."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

# Upsampling helper -------------------------------------------------------
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq

            if sc < best_score:
                best_seq, best_score = cand.copy(), sc
                print(f"[{time.time():.1f}] new best (uniform/arcsine) {sc:.6f}")

        # symmetric beta family with uniform mixing
        for p in p_vals:
            if time.time() >= deadline:
                break
            base = _distribution_beta_sym(n, p)
            for lam in lam_grid:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                if sc < best_score:
                    best_seq, best_score = cand.copy(), sc
                    print(f"[{time.time():.1f}] new best beta p={p:.4g} lam={lam:.2f} {sc:.6f}")

        # edge‑weighted family
        for p in p_vals:
            if time.time() >= deadline:
                break
            edge = _distribution_edge_weighted(n, p)
            for lam in lam_grid:
                cand = _mix_with_uniform(edge, lam)
                sc = _objective(cand)
                if sc < best_score:
                    best_seq, best_score = cand.copy(), sc
                    print(f"[{time.time():.1f}] new best edge p={p:.4g} lam={lam:.2f} {sc:.6f}")

        # sine‑weighted family
        for p in p_vals:
            if time.time() >= deadline:
                break
            sine = _distribution_sine_weighted(n, p)
            for lam in lam_grid:
                cand = _mix_with_uniform(sine, lam)
                sc = _objective(cand)
                if sc < best_score:
                    best_seq, best_score = cand.copy(), sc
                    print(f"[{time.time():.1f}] new best sine p={p:.4g} lam={lam:.2f} {sc:.6f}")

        # random Dirichlet vectors
        for _ in range(8):
            if time.time() >= deadline:
                break
            cand = _distribution_dirichlet(n, rng)
            sc = _objective(cand)
            if sc < best_score:
                best_seq, best_score = cand.copy(), sc
                print(f"[{time.time():.1f}] new best Dirichlet {sc:.6f}")

        # sparse random
        for _ in range(8):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(cand)
            if sc < best_score:
                best_seq, best_score = cand.copy(), sc
                print(f"[{time.time():.1f}] new best sparse {sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  Low‑dimensional global optimisation (DE on piece‑wise linear shape)
    try:
        from scipy.optimize import differential_evolution
    except Exception as e:
        print(f"[{time.time():.1f}] scipy not available: {e}")
        differential_evolution = None

    if differential_evolution is not None:
        # Use a moderate length for DE – fast evaluations
        n_de = 2000
        m_ctrl = 8                     # number of control points
        bounds = [(0.0, 10.0) for _ in range(m_ctrl)]

        def de_objective(params):
            shape = np.abs(params) + 1e-12   # ensure positivity
            # linear interpolation
            seq = np.interp(
                np.linspace(0, 1, n_de),
                np.linspace(0, 1, m_ctrl),
                shape,
            )
            seq = _symmetrize(seq)
            seq = seq / seq.sum()
            return _objective(seq)

        remaining = deadline - time.time()
        # allocate at most 60 s to DE
        if remaining > 70:
            de_deadline = time.time() + min(60.0, remaining * 0.4)
            print(f"[{time.time():.1f}] running DE (n={n_de}, m={m_ctrl})")
            result = differential_evolution(
                de_objective,
                bounds,
                maxiter=30,
                popsize=15,
                tol=1e-6,
                updating='deferred',
                polish=False,
                seed=rng,
                workers=1,
            )
            seq = np.interp(
                np.linspace(0, 1, n_de),
                np.linspace(0, 1, m_ctrl),
                np.abs(result.x),
            )
            seq = _symmetrize(seq)
            seq = seq / seq.sum()
            sc = _objective(seq)
            if sc < best_score:
                best_seq, best_score = seq.copy(), sc
                print(f"[{time.time():.1f}] DE improvement to {sc:.6f}")
    else:
        print(f"[{time.time():.1f}] Skipping DE (SciPy unavailable)")

    # ------------------------------------------------------------------
    # 3.  Heavy mirror‑descent on the current best candidate
    if best_seq is None:
        best_seq = _distribution_uniform(3000)
        best_score = _objective(best_seq)

    # Guarantee sum=1
    best_seq = best_seq / best_seq.sum()
    # Run the descent for the remaining time budget
    a_opt, sc_opt = _subgradient_descent_momentum(
        best_seq.copy(),
        deadline,
        best_seq.copy(),
        best_score,
        lr0=0.9,
        active_frac=0.04,
        max_active=40,
        beta=0.9,
        eval_interval=150,
        patience=2500,
    )
    if sc_opt < best_score:
        best_seq, best_score = a_opt.copy(), sc_opt
        print(f"[{time.time():.1f}] post‑descent improvement to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Upsample and refine further if time permits
    remaining = deadline - time.time()
    if remaining > 80:
        large_n = min(18000, len(best_seq) * 2)
        print(f"[{time.time():.1f}] upsampling to {large_n}")
        up_seq = _upsample(best_seq, large_n)
        sub_deadline = time.time() + remaining * 0.6
        up_opt, up_score = _subgradient_descent_momentum(
            up_seq,
            sub_deadline,
            best_seq,
            best_score,
            lr0=0.7,
            active_frac=0.04,
            max_active=35,
            beta=0.92,
            eval_interval=80,
            patience=2000,
        )
        if up_score < best_score:
            best_seq, best_score = up_opt.copy(), up_score
            print(f"[{time.time():.1f}] up‑sampled descent yields {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Jitter‑restart cycles (escape possible plateaus)
    while time.time() < deadline - 20.0:
        jitter = rng.exponential(scale=0.018, size=best_seq.shape)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sub_deadline = time.time() + min(15.0, (deadline - time.time()) * 0.5)
        cand_opt, cand_score = _subgradient_descent_momentum(
            cand,
            sub_deadline,
            best_seq,
            best_score,
            lr0=0.6,
            active_frac=0.06,
            max_active=30,
            beta=0.9,
            eval_interval=70,
            patience=1500,
        )
        if cand_score < best_score:
            best_seq, best_score = cand_opt.copy(), cand_score
            print(f"[{time.time():.1f}] jitter‑restart improved to {best_score:.6f}")
        else:
            break

    # ------------------------------------------------------------------
    # 6.  Symmetrisation test
    sym_seq = _symmetrize(best_seq)
    sym_score = _objective(sym_seq)
    if sym_score + 1e-14 < best_score:
        best_seq, best_score = sym_seq, sym_score
        print(f"[{time.time():.1f}] symmetrised version better: {sym_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Weighted pairwise refinement
    if time.time() < deadline - 10.0:
        print(f"[{time.time():.1f}] weighted pairwise refinement")
        best_seq, best_score = _pairwise_weighted_refine(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            max_trials=200_000,
        )

    # ------------------------------------------------------------------
    # 8.  Final cheap simulated‑annealing clean‑up
    if time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final simulated‑annealing clean‑up")
        best_seq, best_score = _random_local_search(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            init_temp=0.008,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    # Return a plain list of Python floats
    return best_seq.tolist()
