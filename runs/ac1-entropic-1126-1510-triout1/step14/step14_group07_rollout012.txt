def _distribution_beta(n, alpha, beta):
    i = np.arange(n, dtype=np.float64)
    # map i from 0..n-1 to (i+0.5)/n maybe to avoid zeros at boundaries
    x = (i + 0.5) / n
    w = (x**(alpha-1)) * ((1 - x)**(beta-1))
    # Oops if alpha/beta < 1, x**negative may blow up near edges; use np.clip to avoid 0.
    w[np.isnan(w)] = 0.0
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full(n, 1.0/n)
    else:
        w = w / w_sum
    return w

"""
Optimisation of a non‑negative step‑function (probability vector) for
``evaluate_sequence``.

The algorithm has four stages:

1️⃣  **Diversified seeding** – a collection of analytic families
    (uniform, sine weighted, beta‑shaped, centre‑exponential,
    left/right exponential, Dirichlet, sparse random) is sampled for
    several lengths.  A small mixture with the uniform vector is also
    considered.  The best ≈15 seeds (by the exact objective
    ``2·n·max(conv)``) are retained.

2️⃣  **L2 (squared‑error) Adam refinement** – each retained seed is
    short‑run optimised with Adam on the smooth loss
    ``‖a*a−u‖₂²`` where ``u`` is the uniform target convolution.
    The gradient is computed analytically with FFTs and the parameters
    are kept on the probability simplex using a soft‑max
    parametrisation.

3️⃣  **Long Adam descent on the true max‑convolution objective** –
    the best seed from step 2 is further refined with the original
    sub‑gradient (active‑set) method for the max‑norm.

4️⃣  **Local polishing** – a greedy pairwise mass‑transfer step and a
    cheap simulated‑annealing walk finish the search.

All stages respect the global 1000 s budget (a 2 s safety margin is
kept).  The returned list is a proper probability vector; the external
evaluator will re‑normalise it anyway.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Simplex utilities ----------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


# ----------------------------------------------------------------------
# Convolution helpers --------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT of appropriate size."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())
    X = rfft(x, n=size)
    Y = rfft(y, n=size)
    conv = irfft(X * Y, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    w /= w.sum()
    return w


def _distribution_exponential_center(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(i - half))
    w /= w.sum()
    return w


def _distribution_exponential_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-rate * i)
    w /= w.sum()
    return w


def _distribution_exponential_right(n: int, rate: float) -> np.ndarray:
    return _distribution_exponential_left(n, rate)[::-1]


def _distribution_dirichlet(n: int, rng: np.random.Generator, conc: float) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a.astype(np.float64)


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.006) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) pdf sampled at centred points (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # avoid 0^negative and 1^negative by clipping the interior
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    s = w.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    w /= s
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Objective -----------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (a is assumed to sum to 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Gradient for the L2 (squared‑error) loss ----------------------------
def _grad_l2(a: np.ndarray) -> np.ndarray:
    """
    Gradient of  ‖a*a – u‖₂²  w.r.t. a, where u is the uniform
    vector of length 2·n‑1.  The formula is
        g = 4 * ( (a_rev) ⋆ (a*a - u) )
    where ⋆ is linear convolution and a_rev is a reversed.
    """
    n = a.shape[0]
    # a*a
    conv = _fft_self_convolve(a)                     # length 2n‑1
    target = 1.0 / (2 * n - 1)
    d = conv - target                                 # residual
    a_rev = a[::-1]
    full = _fft_convolve(d, a_rev)                   # length 3n‑2
    start = n - 1
    grad = 4.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimisation (max‑norm) ----------------------------------------
def _adam_refine(
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    lr0: float = 0.8,
    active_frac: float = 0.025,
    max_active: int = 30,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_interval: int = 250,
    best_seq: np.ndarray | None = None,
    best_score: float = float("inf"),
) -> tuple[np.ndarray, float]:
    """Adam on the true ``max‑convolution`` objective (sub‑gradient)."""
    n = init_a.shape[0]
    x = np.log(np.maximum(init_a, 1e-12))
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    ind_grid = np.arange(n, dtype=np.int32)

    if best_seq is None:
        best_seq = init_a.copy()
        best_score = _objective(init_a)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        a = _softmax(x)                     # stays on the simplex
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            top = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top]

        # sub‑gradient of the max (active‑set)
        grad_a = np.zeros_like(a)
        for idx_k in act:
            offset = idx_k - ind_grid
            mask = (offset >= 0) & (offset < n)
            grad_a[mask] += 2.0 * a[offset[mask]]

        # convert gradient w.r.t a into gradient w.r.t log‑weights x
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = lr0 / math.sqrt(t)
        x -= eta * m_hat / (np.sqrt(v_hat) + eps)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] max‑Adam it={t:,} new best {best_score:.6f}")

    best_seq = _project_onto_simplex(best_seq, s=1.0)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Adam optimisation (L2 loss) -----------------------------------------
def _adam_refine_l2(
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 100_000,
    lr0: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_interval: int = 200,
    best_seq: np.ndarray | None = None,
    best_score: float = float("inf"),
) -> tuple[np.ndarray, float]:
    """Adam on the smooth L2 loss (‖a*a−u‖₂²)."""
    n = init_a.shape[0]
    x = np.log(np.maximum(init_a, 1e-12))
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    if best_seq is None:
        best_seq = init_a.copy()
        best_score = _objective(init_a)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        a = _softmax(x)

        # gradient of L2 loss
        grad_a = _grad_l2(a)

        # convert to gradient wrt logits
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = lr0 / math.sqrt(t)
        x -= eta * m_hat / (np.sqrt(v_hat) + eps)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] L2‑Adam it={t:,} new best {best_score:.6f}")

    best_seq = _project_onto_simplex(best_seq, s=1.0)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise refinement -------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing local search -------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

# Main entry point -----------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Return a list of non‑negative heights (probability vector)."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # a tiny safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1️⃣  Seed generation
    lengths = [2000, 3000, 4000, 6000, 8000]   # try to keep them modest
    lam_grid = [0.0, 0.2, 0.4, 0.6]           # mix weight with uniform
    p_grid = np.logspace(-1, 1.0, 6)          # sine‑weights
    rate_grid = np.linspace(0.001, 0.02, 5)   # exponentials
    beta_grid = [
        (0.3, 0.3), (0.5, 0.5), (0.7, 0.7), (1.0, 1.0),
        (2.0, 2.0), (5.0, 5.0), (0.5, 2.0), (2.0, 0.5)
    ]
    conc_grid = [0.05, 0.2, 1.0]              # Dirichlet concentration

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # ---- deterministic analytic families ---------------------------------
        bases = [
            _distribution_uniform(n),
            _distribution_exponential_center(n, rate=0.005),
            _distribution_exponential_left(n, rate=0.003),
            _distribution_exponential_right(n, rate=0.003),
        ]
        for base in bases:
            sc = _objective(base)
            top_candidates.append((sc, base.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new global best (det) len={n} score={sc:.6f}")

        # ---- sine‑weighted family --------------------------------------------
        for p in p_grid:
            if time.time() >= deadline:
                break
            sine = _distribution_sine_weighted(n, p)
            for lam in lam_grid:
                cand = _mix_with_uniform(sine, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new best sine p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # ---- beta‑shaped families ---------------------------------------------
        for alpha, beta in beta_grid:
            if time.time() >= deadline:
                break
            beta_vec = _distribution_beta(n, alpha, beta)
            sc = _objective(beta_vec)
            top_candidates.append((sc, beta_vec.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, beta_vec.copy()
                print(f"[{time.time():.1f}] new best beta ({alpha},{beta}) score={sc:.6f}")
            for lam in lam_grid:
                cand = _mix_with_uniform(beta_vec, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new best beta mix ({alpha},{beta}) lam={lam:.2f} score={sc:.6f}")

        # ---- exponential families ---------------------------------------------
        for rate in rate_grid:
            if time.time() >= deadline:
                break
            cad = _distribution_exponential_center(n, rate)
            sc = _objective(cad)
            top_candidates.append((sc, cad.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cad.copy()
                print(f"[{time.time():.1f}] new best exp‑center rate={rate:.4f} score={sc:.6f}")

        # ---- Dirichlet random seeds -------------------------------------------
        for conc in conc_grid:
            for _ in range(2):
                if time.time() >= deadline:
                    break
                a = _distribution_dirichlet(n, rng, conc)
                sc = _objective(a)
                top_candidates.append((sc, a.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, a.copy()
                    print(f"[{time.time():.1f}] new best Dirichlet conc={conc:.3g} score={sc:.6f}")

        # ---- sparse random seeds ----------------------------------------------
        for _ in range(2):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] new best sparse len={n} score={sc:.6f}")

        # keep only a handful of the very best seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:15]

    # ------------------------------------------------------------------
    # include a previously stored construction if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2️⃣  Short L2‑Adam refinements for each retained seed
    refined: list[tuple[float, np.ndarray]] = []
    for sc, cand in top_candidates:
        if time.time() >= deadline:
            break
        slice_deadline = time.time() + min(20.0, (deadline - time.time()) * 0.05)
        best_seq, best_sc = _adam_refine_l2(
            cand.copy(),
            slice_deadline,
            max_iters=15_000,
            lr0=0.6,
            eval_interval=150,
            best_seq=cand,
            best_score=sc,
        )
        refined.append((best_sc, best_seq))

    if refined:
        overall_best_score, overall_best_seq = min(refined, key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 3️⃣  Long Adam descent on the true max‑norm objective
    if time.time() < deadline:
        remaining = deadline - time.time()
        long_deadline = time.time() + max(0.0, remaining - 30.0)
        print(f"[{time.time():.1f}] start long max‑Adam descent (≈{remaining-30:.1f}s left)")
        overall_best_seq, overall_best_score = _adam_refine(
            overall_best_seq.copy(),
            long_deadline,
            max_iters=400_000,
            lr0=0.9,
            active_frac=0.02,
            max_active=40,
            eval_interval=300,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] after long descent best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 5️⃣  Final simulated‑annealing polishing
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.01,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    # Normalise once more (just in case) and return as a plain list.
    overall_best_seq = np.array(overall_best_seq, dtype=np.float64)
    overall_best_seq = overall_best_seq / max(overall_best_seq.sum(), 1e-12)
    overall_best_seq = np.clip(overall_best_seq, 0.0, 1000.0)

    print(f"Final best score (exact) ≈ {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
