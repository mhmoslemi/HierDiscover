def _additive_smooth_descent(a0, time_limit, best_seq, best_score, lr0=0.02, tau_start=0.02, tau_end=0.001):
    a = a0.copy()
    n = a.shape[0]
    it = 0
    start = time.time()
    while time.time() < time_limit:
        it += 1
        # compute conv and exact objective
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] add-smooth it={it} new best {best_score:.6f}")

        # anneal tau
        frac = (time.time() - start) / (time_limit - start)
        tau = tau_start * (1-frac) + tau_end * frac

        # compute softmax weights
        maxc = conv.max()
        shifted = conv - maxc
        # compute exp(shifted/tau) safely
        exp_vals = np.exp(shifted / tau)
        w = exp_vals / exp_vals.sum()

        # gradient of smoothed objective:
        term1 = _fft_cross_correlate_valid(w, a)
        win_sum = _fft_sliding_window_sum(w, n)
        grad = term1 + a * win_sum

        # step size
        eta = lr0 / math.sqrt(it)
        # additive update, then project onto simplex
        a = a - eta * grad
        # ensure non-negative
        a = np.maximum(a, 0.0)
        # project onto simplex (sum to 1)
        a = _project_onto_simplex(a, s=1.0)
    return best_seq, best_score

"""
Improved optimiser for the “step‑function” objective.

The programme follows three stages

1️⃣  A quick sweep of a handful of analytic families (uniform,
    arcsine, beta‑edges, edge‑weighted, exponential‑edge) on several
    lengths.  All candidates are normalised to a probability vector
    (sum = 1) and their exact objective 2·n·max(conv) is evaluated.
    The 𝑘 best seeds are retained.

2️⃣  Each retained seed is refined with a *Adam* optimiser on a
    smoothed version of the max‑convolution (τ‑soft‑max).  This is a
    differentiable surrogate   Lτ(a)=τ·log Σ exp(conv/τ).  The exact
    sub‑gradient of Lτ is computed via FFT‑based cross‑correlation.
    After the Adam phase a short run of the original exponentiated‑
    gradient sub‑gradient method (exact L∞) is performed for polishing.

3️⃣  A final greedy pairwise mass‑transfer hill‑climber tries random
    moves that strictly improve the exact objective.

All steps respect the overall 1000 s budget and the best sequence is
returned as a plain list – the caller will apply the normalisation
inside ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# FFT helpers for convolution / cross‑correlation
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()           # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_cross_correlate_valid(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (valid part) (w ⋆ a)[j] = Σ_i w[i+j]·a[i],
    returning an array of length len(a).
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2·n‑1
    size = 1 << ((m + n - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A = np.fft.rfft(a, n=size)
    corr = np.fft.irfft(W * np.conj(A), n=size)
    return corr[:n]


def _fft_sliding_window_sum(w: np.ndarray, win: int) -> np.ndarray:
    """
    For each j compute Σ_{i=j}^{j+win‑1} w[i] (window length = win).
    Returns an array of length win, i.e. len(a).
    """
    m = w.shape[0]
    size = 1 << ((m + win - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    O = np.fft.rfft(np.ones(win, dtype=np.float64), n=size)
    conv = np.fft.irfft(W * O, n=size)
    start = win - 1
    return conv[start:start + win]


# ----------------------------------------------------------------------
# Analytic families (return probability vectors, Σ=1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_edge(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta‑type: w_i ∝ (i·(n‑i))^{α‑1} , α>0."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = (i * (n - i)) ** (alpha - 1.0)
    vals = np.clip(vals, 0.0, None)
    s = vals.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    vals /= s
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Power‑law edge weighting w_i ∝ (i·(n‑i))^{-p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    w /= w_sum
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse Dirichlet‑like seed with prescribed non‑zero density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Exact objective (for a probability vector Σa=1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Adam optimiser on the τ‑softmax surrogate
# ----------------------------------------------------------------------
def _adam_smooth_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr: float = 0.05,
    tau_start: float = 0.02,
    tau_end: float = 0.001,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam on Lτ(a)=τ·log Σ exp(conv/τ).  The gradient is obtained via FFT.
    The iterate stays on the simplex by an Euclidean projection after
    each additive update.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    start = time.time()

    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] adam iter={t} new best {best_score:.6f}")

        # linear τ‑annealing
        frac = (time.time() - start) / (time_limit - start)
        tau = tau_start * (1.0 - frac) + tau_end * frac

        # soft‑max weights
        maxc = conv.max()
        shifted = conv - maxc                     # for numerical stability
        exp_vals = np.exp(shifted / tau)
        w = exp_vals / exp_vals.sum()

        # gradient of the smoothed objective
        term1 = _fft_cross_correlate_valid(w, a)   # Σ_k w[k+i]·a[k]
        win_sum = _fft_sliding_window_sum(w, n)   # Σ_{k=i}^{i+n‑1} w[k]
        grad = term1 + a * win_sum

        # Adam update (additive)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1**t)
        v_hat = v / (1.0 - beta2**t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # enforce feasibility (simplex)
        a = np.maximum(a, 0.0)
        a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (exponentiated‑gradient version, unchanged)
# ----------------------------------------------------------------------
def _subgradient_descent_exact(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] exact subgrad iter={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise hill‑climber (unchanged)
# ----------------------------------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.45   # up to 45 % of the source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:

            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after exact polish seed {idx} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Final greedy pairwise polishing (as much time as remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy pairwise polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=None,
        )
        if sc_opt < overall_best_score:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Report and return
    # ------------------------------------------------------------------
    print(
        f"Final best objective (2·n·max(conv)) = {overall_best_score:.6f} "
        f"with length = {len(overall_best_seq)}"
    )
    # evaluate_sequence will re‑normalise internally, so we simply return the list
    return overall_best_seq.tolist()
