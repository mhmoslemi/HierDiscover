def propose_candidate(seed=42, budget_s=1000):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    best_seq = None
    best_score = float('inf')
    top = []  # list of (score, seq)

    # lengths list
    lengths = [3000, 4000, 5000, 8000, 12000, 16000, 20000]
    # p grid: logspace
    p_grid = np.concatenate((
        np.logspace(-2, -0.5, 8),
        np.logspace(-0.5, 0.5, 8),
        np.logspace(0.5, 1.5, 6)
    ))
    lam_grid = np.linspace(0.0, 1.0, 9)  # more fine
    c_grid = np.logspace(-3, 1, 8)  # for exponential decay

    for n in lengths:
        if time.time() >= deadline: break
        print(f"[{time.time():.1f}] scanning length {n}")

        # uniform
        cand = _distribution_uniform(n)
        sc = _objective(cand)
        top.append((sc,cand.copy()))
        if sc < best_score: best_score,best_seq = sc,cand.copy()
        # arcsine
        cand = _distribution_arcsine(n)
        sc = _objective(cand)
        top.append((sc,cand.copy()))
        if sc < best_score: best_score,best_seq = sc,cand.copy()
        # exponential decay
        for c in c_grid:
            cand = _distribution_exponential_decay(n,c)
            sc=_objective(cand)
            top.append((sc,cand.copy()))
            if sc<best_score: best_score,best_seq = sc,cand.copy()
        # product edge weighted
        for p in p_grid:
            if time.time()>=deadline: break
            base = _distribution_edge_weighted(n,p)
            sc=_objective(base)
            top.append((sc,base.copy()))
            if sc<best_score: best_score,best_seq= sc,base.copy()
            # mixture with uniform
            for lam in lam_grid:
                cand = _mix_with_uniform(base, lam)
                sc=_objective(cand)
                top.append((sc,cand.copy()))
                if sc<best_score: best_score,best_seq = sc,cand.copy()
        # sum edge weighted
        for p in p_grid:
            if time.time()>=deadline: break
            base = _distribution_edge_sum_weighted(n,p)
            sc=_objective(base)
            top.append((sc,base.copy()))
            if sc<best_score: best_score,best_seq = sc,base.copy()
            for lam in lam_grid:
                cand = _mix_with_uniform(base, lam)
                sc=_objective(cand)
                top.append((sc,cand.copy()))
                if sc<best_score: best_score,best_seq = sc,cand.copy()
        # generalized normal
        for q in np.linspace(0.5, 3.0, 7):
            if time.time()>=deadline: break
            cand = _distribution_generalized_normal(n,q)
            sc=_objective(cand)
            top.append((sc,cand.copy()))
            if sc<best_score: best_score,best_seq = sc,cand.copy()
        # random mixtures of three families
        for _ in range(10):
            if time.time()>=deadline: break
            # pick random p's and lam
            p1 = 10**rng.uniform(-2,1.5)
            p2 = 10**rng.uniform(-2,1.5)
            lam1 = rng.random()  # weight for edge weighted 0..1
            lam2 = rng.random()*(1-lam1)
            # Compose mixture
            edge = _distribution_edge_weighted(n,p1)
            sumedge = _distribution_edge_sum_weighted(n,p2)
            uniform = _distribution_uniform(n)
            cand = lam1*edge + lam2*sumedge + (1-lam1-lam2)*uniform
            cand = cand / cand.sum()
            sc=_objective(cand)
            top.append((sc,cand.copy()))
            if sc<best_score: best_score,best_seq = sc,cand.copy()
        # random sparse Dirichlet seeds
        for _ in range(6):
            if time.time()>=deadline: break
            cand = np.zeros(n, dtype=np.float64)
            k = max(2, int(n * 0.008))

assistantfinal**Solution Explanation**

The objective that `evaluate_sequence` computes is  

\[
\text{obj}(a)=\frac{2\,n\;\max_k (a*a)[k]}{(\sum_i a_i)^2},
\qquad a_i\ge 0 .
\]

Because the formula is scale‑invariant we may restrict ourselves to *probability*
vectors (`∑a_i = 1`).  
For a probability vector the objective simplifies to  

\[
\text{obj}(a)=2\,n\;\max_k (a*a)[k].
\]

So we have to minimise the largest entry of the *aperiodic* self‑convolution.
The problem is non‑convex, but the following ingredients work very well in practice:

* **Large, diverse seed pool** – analytic families (uniform, arcsine,
  product‑edge, sum‑edge, generalised‑normal, exponential‑decay) and random
  mixtures of them.
* **Fast convolution** – FFT based self‑convolution (`_fft_self_convolve`).
* **Mirror / exponentiated‑gradient descent** – a sub‑gradient of the active
  convolution entry is cheap (`_grad_of_max_conv`).  Multiplicative updates
  stay on the simplex automatically.
* **Multiple restarts** – each of the best seeds is refined for a short
  time; the best result among all runs is kept.
* **Pairwise mass‑transfer annealing** – a cheap simulated‑annealing polish
  that moves a fraction of mass between two bins.

All work is limited to the wall‑time budget (default 1000 s).  The algorithm
is completely self‑contained, uses only NumPy and the standard library, and
returns a plain `list[float]` that can be fed to `evaluate_sequence`.

```python
"""
Propose a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.

Algorithm
---------
1. Build a large pool of candidate distributions from several analytic families
   (uniform, arcsine, product‑edge, sum‑edge, exponential‑decay,
   generalised‑normal) and from random mixtures of them.
2. Keep the 30 best candidates (lowest exact objective).
3. Run a short exponentiated‑gradient (multiplicative mirror) descent on every
   kept candidate (time is shared among the candidates).  The sub‑gradient of
   the current maximal convolution entry is used.
4. Finish with a pairwise‑mass‑transfer simulated‑annealing polish.
5. Return the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  FFT based self‑convolution (real inputs, linear convolution)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
#  Exact objective used by the evaluator (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a))/sum(a)^2 ; a is non‑negative raw vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
#  Analytic families – return a probability vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i·(n‑i))^{‑p}   with i = distance to the left edge (0‑based)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = n - np.arange(n, dtype=np.float64) - 0.5
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_sum_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ i^{‑p} + (n‑i)^{‑p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = n - np.arange(n, dtype=np.float64) - 0.5
    w = (i ** (-p)) + (j ** (-p))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(‑|x‑½|^{q}) on the discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_exponential_decay(n: int, c: float) -> np.ndarray:
    """Symmetric exponential decay: w_i ∝ e^{‑c·i} + e^{‑c·(n‑1‑i)}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1‑lam) + uniform·lam."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
#  Gradient of the *active* convolution entry (sub‑gradient of max)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """Return (gradient, index_of_max) of the currently maximal entry of a*a."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))

    grad = np.zeros_like(a)
    if k < n:
        # overlap length = k+1   (i = 0 … k)
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        # k ≥ n : overlap length = 2·n‑1‑k   (i = k‑(n‑1) … n‑1)
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
#  Exponentiated‑gradient descent (multiplicative mirror descent)
# ----------------------------------------------------------------------
def _exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.9,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """
    Run EG descent on the exact objective until ``deadline``.
    The update is a multiplicative mirror step on the simplex.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Pairwise‑mass‑transfer annealing (final polish)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins; accept improvements."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4        # move at most 40 % of source mass
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score

        f"(length={len(best_seq)})"
    )
    return best_seq.tolist()
```
