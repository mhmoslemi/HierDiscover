"""
Propose a low‑value step‑function for ``evaluate_sequence``.

The routine follows a multi‑stage strategy that significantly improves upon
the previous 1.509935 bound.

1. **Rich seeding** – many analytic families (uniform, arcsine, Hann,
   power‑law at the edges, symmetric Beta, generalized normal, logistic,
   random Dirichlet, sparse) are sampled for several lengths.  The 30 best
   seeds are retained.

2. **Smooth‑max exponentiated‑gradient polishing** – each seed is refined
   with a continuation method: a smooth approximation of the “max”
   (soft‑max) is used as the loss, the gradient is obtained via FFT
   convolution and an exponentiated‑gradient update is performed.
   The “temperature’’ (β) of the soft‑max is increased slowly so the method
   first moves in a well‑conditioned direction and later focuses on the
   true maximum.

3. **Hard‑max mirror descent** – after the smooth phase a few iterations of
   the original sub‑gradient method (max‑only) are applied to finish the
   polishing.

4. **Progressive up‑sampling** – the best vector is linearly interpolated
   (doubling the length each time) while time permits.  After every
   up‑sampling a medium‑length smooth‑gradient run is performed.

5. **Final pairwise mass‑transfer annealing** – a simple simulated annealing
   that moves a random fraction of mass between two bins is run until the
   budget expires.

All steps respect the 1000 s budget (2 s safety margin).  The returned
list is a plain Python list ready for ``evaluate_sequence`` (non‑negative,
automatically normalised inside the objective).
"""

# ----------------------------------------------------------------------
# Utility: projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolutions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution (len = 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _full_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b (no padding tricks)."""
    size = _next_pow2(len(a) + len(b) - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: len(a) + len(b) - 1]


# ----------------------------------------------------------------------
# Exact objective for a probability vector (∑a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^(-p) where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w /= w.sum()
    return w


def _distribution_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) sampled on cell centres."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    w = np.where(
        (x == 0) | (x == 1),
        0.0,
        np.exp((alpha - 1) * np.log(x) + (alpha - 1) * np.log1p(-x)),
    )
    w /= w.sum()
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w /= w.sum()
    return w


def _distribution_logistic(n: int, steep: float) -> np.ndarray:
    """Symmetric logistic distribution centred at 0.5."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = 1.0 / (1.0 + np.exp(-steep * x))
    w = w - w.min()
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.02
) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1‑λ) + uniform·λ."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smooth‑max mirrored (exponentiated) gradient descent
# ----------------------------------------------------------------------
def _smooth_exp_gradient(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    beta0: float = 0.5,
    beta_factor: float = 1.03,
    eval_every: int = 40,
) -> tuple[np.ndarray, float]:
    """
    Mirror descent on a smooth approximation of the max‑convolution.
    Starts with a small β (smooth) and raises it gradually.
    """
    a = a0.copy()
    n = a.shape[0]
    eta = eta0
    beta = beta0
    t = 0
    while time.time() < deadline:
        t += 1
        # full convolution a*a
        conv = _fft_self_convolve(a)               # shape (2n‑1)
        max_conv = conv.max()

        # soft‑max weights (numerically stable)
        w = np.exp(beta * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of Σ w_k * conv_k  (see analysis)
        w_rev = w[::-1]
        grad_full = _full_convolve(a, w_rev)       # length 3n‑2
        # extract the needed slice
        grad = 2.0 * grad_full[2 * n - 2 - np.arange(n)]

        # exponentiated update on the simplex (mirror descent)
        a = a * np.exp(-eta * grad)
        a = _project_onto_simplex(a, s=1.0)

        # occasional evaluation
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                eta = min(eta0, eta * 1.02)          # reward improvement
                beta = min(100.0, beta * beta_factor)  # increase smoothness
                print(f"[{time.time():.1f}] smooth‑GD it={t:,} new best {best_score:.6f}")
            else:
                eta *= 0.995                           # gentle decay

        # safety re‑projection once in a while
        if t % (eval_every * 10) == 0:
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Hard‑max exponentiated gradient (identical to the original code)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """(sub‑gradient, index of maximal convolution entry)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))

    grad = np.zeros_like(a)
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]          # overlap 0 … k
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]    # overlap start … n‑1
    return grad, k


def _hard_exp_gradient(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    beta: float = 0.9,
    eval_every: int = 50,
) -> tuple[np.ndarray, float]:
    """Original max‑only mirror descent (kept for final polishing)."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)          # EMA of gradients
    eta = eta0
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        m = beta * m + (1.0 - beta) * grad
        a = a * np.exp(-eta * m)
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                eta = min(eta0, eta * 1.02)
                print(f"[{time.time():.1f}] hard‑GD it={t:,} new best {best_score:.6f}")
            else:
                eta *= 0.995

        if t % (eval_every * 10) == 0:
            a = _project_onto_simplex(a, s=1.0)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise mass‑transfer simulated annealing
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        src = rng.choice(np.where(a > 1e-14)[0])
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1
        delta = a[src] * rng.random() * 0.3
        if delta < 1e-16:
            continue
        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta
        cur = _objective(new_a)

        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.02

        accept = False
        if cur < best_score - 1e-15:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal {src}->{dst} Δ={delta:.1e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling while preserving total mass (=1)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    a_new = _project_onto_simplex(a_new, s=1.0)
    return a_new

                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, a.copy()
                    print(f"[{time.time():.1f}] edge p={p:.3g} λ={lam:.2f} new best {sc:.6f}")

        # symmetric Beta family
        alphas = np.concatenate((
            np.logspace(-2.0, -0.2, 10, base=10),
            np.linspace(0.2, 3.0, 12),
            np.linspace(3.0, 6.0, 6),
        ))
        for alpha in alphas:
            a = _distribution_beta(n, float(alpha))
            sc = _objective(a)
            cand_pool.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] beta α={alpha:.3g} new best {sc:.6f}")

        # generalized normal
        for q in np.linspace(0.2, 5.0, 12):
            a = _distribution_generalized_normal(n, float(q))
            sc = _objective(a)
            cand_pool.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] gn q={q:.2f} new best {sc:.6f}")

        # logistic family (steepness)
        for steep in np.linspace(0.5, 6.0, 10):
            a = _distribution_logistic(n, float(steep))
            sc = _objective(a)
            cand_pool.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] logistic s={steep:.2f} new best {sc:.6f}")

        # random Dirichlet seeds
        for _ in range(12):
            log_alpha = rng.uniform(-2.0, 1.0)          # α ∈ [0.01, 10]
            alpha = 10 ** log_alpha
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            cand_pool.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] Dirichlet α={alpha:.3g} new best {sc:.6f}")

        # sparse random seeds
        for _ in range(8):
            a = _distribution_random_sparse(n, rng, density=0.03)
            sc = _objective(a)
            cand_pool.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] sparse new best {sc:.6f}")

        # keep only the top‑30 seeds overall
        cand_pool.sort(key=lambda x: x[0])
        cand_pool = cand_pool[:30]

    # ------------------------------------------------------------------
    # 2️⃣  Incorporate a possibly‑existing construction
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.abs(prev)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
            cand_pool.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3️⃣  Ensure we have something to return
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 2048
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Smooth‑gradient polishing of each seed (short runs)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0 and cand_pool:
        per_seed = max(0.2, remaining * 0.25 / max(len(cand_pool), 1))
        print(f"[{time.time():.1f}] smooth‑GD on {len(cand_pool)} seeds (~{per_seed:.2f}s each)")
        for idx, (sc, a0) in enumerate(cand_pool):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            a_opt, sc_opt = _smooth_exp_gradient(
                a0.copy(),
                sub_dead,
                overall_best_seq,
                overall_best_score,
                eta0=0.7,
                beta0=0.4,
                eval_every=25,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] seed {idx} smooth‑GD improved best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Hard‑max gradient on the current global best (medium run)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_dead = time.time() + (deadline - time.time()) * 0.30
        a_opt, sc_opt = _hard_exp_gradient(
            overall_best_seq.copy(),
            sub_dead,
            overall_best_seq,
            overall_best_score,
            eta0=0.5,
            beta=0.9,
            eval_every=60,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] hard‑GD improved best to {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Progressive up‑sampling (doubling while time permits)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n > 65536:
            break
        print(f"[{time.time():.1f}] up‑sampling {cur_n} → {target_n}")
        up = _upsample(overall_best_seq, target_n)
        up_sc = _objective(up)
        if up_sc < overall_best_score:
            overall_best_seq, overall_best_score = up, up_sc
            print(f"[{time.time():.1f}] up‑sampled new best {overall_best_score:.6f}")

        # medium smooth‑gradient run on the up‑sampled vector
        sub_dead = time.time() + (deadline - time.time()) * 0.18
        a_opt, sc_opt = _smooth_exp_gradient(
            overall_best_seq.copy(),
            sub_dead,
            overall_best_seq,
            overall_best_score,
            eta0=0.45,
            beta0=0.5,
            eval_every=40,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
            print(f"[{time.time():.1f}] after up‑sample smooth‑GD new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣  Final annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing phase")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            overall_best_seq,
            overall_best_score,
            rng,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
            print(f"[{time.time():.1f}] post‑anneal new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)) = {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
