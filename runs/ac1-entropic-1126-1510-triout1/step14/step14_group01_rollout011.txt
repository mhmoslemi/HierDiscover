def _exact_descent(a0, deadline, best_seq, best_score, lr0=0.6, beta=0.9, eval_interval=100, patience=2000):
    a = a0.copy()
    n = a.shape[0]
    t = 0
    no_improve=0
    momentum = np.zeros_like(a)
    while time.time() < deadline and t < 500_000:
        t += 1
        grad, max_c = _subgradient(a)
        # momentum
        momentum = beta * momentum + (1 - beta) * grad
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * momentum)
        a = np.maximum(a, 1e-30)
        a /= a.sum()
        # Evaluate occasional
        if t % eval_interval == 0:
            cur_score = 2.0 * n * max_c
            if cur_score + 1e-14 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] exact it={t:,} new best {best_score:.6f}")
            else:
                no_improve += 1
        if no_improve >= patience:
            # jitter to escape plateau
            rng = np.random.default_rng()
            jitter = rng.exponential(scale=0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0
    return best_seq, best_score

"""
Multi‑stage optimisation for the step‑function bound.

1. Generate a diverse pool of seed distributions (uniform, arcsine,
   triangular, Gaussian, beta families, edge‑weighted, Dirichlet,
   sparse random).  All seeds are normalised to a probability simplex.

2. Keep the elite few seeds and improve them with an annealed
   exponentiated‑gradient descent on the smooth log‑sum‑exp surrogate
   of the max‑convolution (smooth stage).

3. Switch to an *exact* sub‑gradient of the true objective
   J(a)=max(conv(a,a)) (max‑stage).  The sub‑gradient is obtained by
   convolving the current vector with the indicator of the active
   constraints and an exponentiated‑gradient update is performed.
   A small momentum term and occasional jitter help to escape plateaus.

4. If time permits, linearly up‑sample the current best sequence to a
   larger length and repeat the exact max‑stage refinement.

5. Finish with deterministic pairwise mass‑transfer sweeps and a cheap
   simulated‑annealing clean‑up.

All updates keep the vector non‑negative and re‑normalise it to sum = 1,
so the scale‑invariance of ``evaluate_sequence`` is respected.
The routine respects the 1000 s budget and returns the best list
found.
"""

import time

from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using real FFT."""
    n = a.shape[0]
    m = b.shape[0]
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a simplex vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    w = np.minimum(i, n + 1 - i)
    w /= w.sum()
    return w


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    if sigma <= 0:
        return _distribution_uniform(n)
    i = np.arange(n, dtype=np.float64)
    cen = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - cen) / sigma) ** 2)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric Beta(a,b)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Smoothed surrogate gradient -------------------------------------------
def _smooth_gradient(a: np.ndarray, lam: float) -> tuple[np.ndarray, float]:
    """
    Returns (grad, max_conv) where grad = ∇_a φ_λ(a) and max_conv = max(conv(a,a)).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)          # length 2n‑1
    max_c = conv.max()
    # softmax weights
    z = lam * conv
    zmax = np.max(z)
    exp_z = np.exp(z - zmax)
    s = exp_z / exp_z.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, s)   # length 3n‑2
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, max_c


# ----------------------------------------------------------------------
# Exact sub‑gradient (max‑stage) -----------------------------------------
def _subgradient(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient of J(a)=max(conv(a,a)).
    Returns (grad, max_conv) where grad_i = 2 * Σ_j a_j * 1_{i+j∈S},
    S = {k | conv[k] is maximal}.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_c = conv.max()
    # mask of active constraints (tolerance for floating point)
    eps = 1e-12
    mask = conv >= max_c - eps
    count = mask.sum()
    if count == 0:
        s = mask.astype(float)   # should not happen
    else:
        s = mask.astype(float) / count
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, s)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, max_c

def _exponential_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    grad_func,
    lr0: float = 0.8,
    beta: float = 0.9,
    eval_interval: int = 150,
    patience: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent using `grad_func` which returns
    (gradient, max_conv).  The exact objective value is 2·n·max_conv.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)

    while time.time() < deadline and t < 500_000:
        t += 1
        grad, max_c = grad_func(a)

        # momentum (exponential averaging)
        momentum = beta * momentum + (1.0 - beta) * grad

        # step‑size schedule
        eta = lr0 / math.sqrt(t)

        # EG update, keep positivity
        a = a * np.exp(-eta * momentum)
        a = np.maximum(a, 1e-30)
        a /= a.sum()

        # periodic exact evaluation
        if t % eval_interval == 0:
            cur_score = 2.0 * n * max_c
            if cur_score + 1e-14 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} new best {best_score:.6f}")
            else:
                no_improve += 1

        # jitter on plateaus
        if no_improve >= patience:
            rng = np.random.default_rng()
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

def _smooth_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    lam_start_factor: float = 0.15,
    lam_inc: float = 1.0015,
    beta: float = 0.9,
    eval_interval: int = 200,
    patience: int = 2500,
) -> tuple[np.ndarray, float]:
    """Annealed EG descent on the smoothed log‑sum‑exp surrogate."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    lam = lam_start_factor * n
    rng = np.random.default_rng()

    while time.time() < deadline and t < 600_000:
        t += 1
        grad, max_c = _smooth_gradient(a, lam)

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * momentum)
        a = np.maximum(a, 1e-30)
        a /= a.sum()

        lam *= lam_inc

        if t % eval_interval == 0:
            cur_score = 2.0 * n * max_c
            if cur_score + 1e-14 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] smooth it={t:,} new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Deterministic pairwise mass‑transfer sweep."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Cheap simulated‑annealing clean‑up."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq

    # 8. final cheap simulated‑annealing clean‑up
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.006,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return plain Python list – evaluate_sequence will perform any final clipping
    return overall_best_seq.tolist()
