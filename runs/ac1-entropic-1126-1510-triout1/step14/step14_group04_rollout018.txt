def _l_bfgs_refine(a0, time_limit, beta, best_seq, best_score):
    n = len(a0)
    # transform a0 -> z0: log out? Use softmax: a = softmax(z) => z = log(a) - offset
    # Can set z0 = np.log(a0 + 1e-12)
    eps = 1e-12
    z0 = np.log(a0 + eps)
    # define objective
    def fun(z):
        a = np.exp(z)
        a = a / a.sum()
        # compute conv
        conv = _fft_self_convolve(a)
        # compute smooth max with logsumexp
        # Note: we want smooth_max = (1/beta) * logsumexp(beta*conv)
        # Must use stable logsumexp
        from scipy.special import logsumexp
        smooth_max = logsumexp(beta * conv) / beta
        J = 2.0 * n * smooth_max
        # gradient w.r.t a: grad_a = 2 * n * (conv gradient)
        # compute w = exp(beta * conv) / sum(exp(beta*conv))
        exp_beta = np.exp(beta * (conv - conv.max()))  # stable
        w = exp_beta / exp_beta.sum()
        # gradient of smooth_max wrt a is 2 * (w * a_rev) convolve? Actually gradient of smooth_max = 2 * (w * a_rev) convolve as before.
        a_rev = a[::-1]
        grad_conv = _fft_convolve(w, a_rev)
        grad_a = 2.0 * n * grad_conv[n-1:n-1+n]   # this is gradient dJ/da
        # gradient wrt z: grad_z = a * (grad_a - np.dot(a, grad_a))
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)
        return J, grad_z
    # call scipy.optimize.minimize
    res = scipy.optimize.minimize(fun, z0, method='L-BFGS-B', jac=True,
                                  options={'maxiter': 200, 'disp': False})
    a_opt = np.exp(res.x)
    a_opt /= a_opt.sum()
    sc_opt = _objective(a_opt)
    if sc_opt < best_score:
        best_seq, best_score = a_opt.copy(), sc_opt
    return best_seq, best_score

def _smooth_max_descent(a0, time_deadline, best_seq, best_score):
    a = a0.copy()
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 400.0, 800.0]
    for beta in beta_schedule:
        # limit number iteration per beta based on time
        while time.time() < time_deadline:
            t += 1
            conv = _fft_self_convolve(a)
            max_val = conv.max()
            shifted = conv - max_val
            exp_beta = np.exp(beta * shifted)
            w = exp_beta / exp_beta.sum()
            # compute grad of smooth max
            a_rev = a[::-1]
            grad_conv = _fft_convolve(w, a_rev)
            grad = 2.0 * grad_conv[n-1:n-1+n]   # gradient of smooth max times 1 (???)
            grad_J = 2.0 * n * grad
            # Adam update
            m = beta1 * m + (1-beta1) * grad_J
            v = beta2 * v + (1-beta2) * (grad_J**2)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            step = 0.2 / (np.sqrt(v_hat) + eps)  # maybe larger step
            a -= step * m_hat
            # project onto simplex
            a = _project_onto_simplex(a, s=1.0)

            # periodic exact evaluation
            if t % 200 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-14:
                    best_score, best_seq = cur, a.copy()
                    print(f'[{time.time():.1f}] beta={beta} it={t} new best {best_score:.6f}')
            # break condition per beta after certain iterations
            if t % 3000 == 0:
                # move to next beta
                break
            if time.time() >= time_deadline:
                break
    return best_seq, best_score

def _l_bfgs_refine(a0, time_deadline, beta, best_seq, best_score):
    if not SCIPY_AVAILABLE:
        return best_seq, best_score
    # compute

    import numpy as np
    from scipy.optimize import minimize
    from scipy.special import logsumexp

    n = len(a0)
    eps = 1e-12
    z0 = np.log(a0 + eps)

    def fun(z):
        a = np.exp(z)
        a_sum = a.sum()
        a = a / a_sum
        conv = _fft_self_convolve(a)
        # compute log sum exp
        # stable: shift by max
        scaled = beta * conv
        max_scaled = scaled.max()
        # logsumexp
        lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
        smooth_max = lse / beta
        J = 2.0 * n * smooth_max
        # compute w (softmax)
        exp_beta = np.exp(beta * (conv - conv.max()))  # using conv.max
        w = exp_beta / exp_beta.sum()
        # gradient w.r.t a
        a_rev = a[::-1]
        grad_conv = _fft_convolve(w, a_rev)  # length 3n-2
        grad_a = 2.0 * n * grad_conv[n-1:n-1+n]   # shape n
        # gradient w.r.t z (softmax derivative)
        # grad_z = a * (grad_a - np.dot(a, grad_a))
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)
        return J, grad_z

    res = minimize(fun, z0, method='L-BFGS-B', jac=True, options={'maxiter': 200, 'disp': False})
    a_opt = np.exp(res.x)
    a_opt /= a_opt.sum()
    sc_opt = _objective(a_opt)
    if sc_opt < best_score:
        best_score, best_seq = sc_opt, a_opt.copy()
        print(f"[{time.time():.1f}] L-BFGS refined new best {best_score:.6f}")
    return best_seq, best_score

def _subgradient_descent_exact(a0, time_deadline, best_seq, best_score, lr0=0.9, eval_every=200):
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]
        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]
        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s
        if t % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score, best_seq = cur, a.copy()
                print(f"[{time.time():.1f}] subgrad it={t} new best {best_score:.6f}")
    return best_seq, best_score

def _pairwise_swap(a0, time_deadline, best_seq, best_score, max_trials=None):
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        # choose source with non-negligible mass
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * np.random.rand() * 0.5  # up to 50% of source
        if delta <= 0:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score, best_seq = cur, a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score

def _smooth_max_descent(a0, time_deadline, best_seq, best_score, beta_schedule=None):
    if beta_schedule is None:
        beta_schedule = [0.5,1.0,2.0,5.0,10.0,20.0,50.0,100.0,200.0,400.0,800.0]
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    for beta in beta_schedule:
        # each beta step we allow many iterations until time out or certain count
        while time.time() < time_deadline:
            t += 1
            conv = _fft_self_convolve(a)
            max_val = conv.max()
            # compute softmax weights
            shifted = conv - max_val
            # safe exponentiation
            exp_beta = np.exp(beta * shifted)
            w = exp_beta / exp_beta.sum()
            # gradient of smooth max
            a_rev = a[::-1]
            grad_conv = _fft_convolve(w, a_rev)
            grad = 2.0 * grad_conv[n-1:n-1+n]  # gradient of smooth max
            grad_J = 2.0 * n * grad
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_J
            v = beta2 * v + (1 - beta2) * (grad_J**2)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            step = 0.2 / (np.sqrt(v_hat) + eps)
            a -= step * m_hat
            # projection onto simplex
            a = _project_onto_simplex(a, s=1.0)

            if t % 250 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-14:
                    best_score, best_seq = cur, a.copy()
                    print(f"[{time.time():.1f}] β={beta:.1f} it={t} new best {best_score:.6f}")
            # limit iterations per beta to maybe 5000 to avoid too many
            if t % 5000 == 0:
                break
            # if time almost up, break
            if time.time() >= time_deadline:
                break
        if time.time() >= time_deadline:
            break
    return best_seq, best_score

Hybrid optimizer for the step‑function problem.

The algorithm works in several phases:

1️⃣  Start from the previously discovered best sequence (``height_sequence_1``)

    optimises a log‑sum‑exp surrogate of the ‑∞‑convolution norm.
    The best iterate is kept.

3️⃣  If SciPy is available a high‑β (β=800) L‑BFGS‑B refinement is performed.
    The optimisation variable is the soft‑max of unconstrained logits, so the
    simplex constraint is satisfied exactly.

4️⃣  Exact sub‑gradient descent using an exponentiated‑gradient update
    (mirror descent w.r.t. KL) further reduces the true max‑convolution value.

5️⃣  A final greedy pairwise mass‑transfer hill‑climber moves a random fraction
    of mass between two indices whenever it strictly improves the exact objective.

All steps respect the 1000 s time budget and return the best sequence found.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Optional SciPy import (used for the L‑BFGS refinement)
# ------------------------------------------------------------
try:
    from scipy.optimize import minimize  # type: ignore
    from scipy.special import logsumexp  # type: ignore
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False

# ------------------------------------------------------------
# Utility: Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# FFT‑based convolutions (real inputs)
# ------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a * a."""
    return _fft_convolve(a, a)

# ------------------------------------------------------------
# Exact objective (sum(a)=1)
# ------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) for a with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# ------------------------------------------------------------
# Analytic families (used for fallback / occasional seeds)
# ------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    total = w.sum()
    if total == 0.0:
        return _distribution_uniform(n)
    w /= total
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    """w_i ∝ exp(-c·i) + exp(-c·(n‑1‑i))   (c>0)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    total = w.sum()
    if total == 0.0:
        return _distribution_uniform(n)
    w /= total
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.random(k)
    w = np.zeros(n, dtype=np.float64)
    w[idx] = vals
    total = w.sum()
    if total == 0.0:
        w[0] = 1.0
        total = 1.0
    w /= total
    return w

# ------------------------------------------------------------
# Smoothed‑max mirror descent (Adam) – the core continuous optimiser
# ------------------------------------------------------------
def _smooth_max_descent(
    a0: np.ndarray,
    time_deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    beta_schedule=None,
) -> tuple[np.ndarray, float]:
    """Adam minimisation of the log‑sum‑exp surrogate of max(conv)."""
    if beta_schedule is None:
        beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0,
                         50.0, 100.0, 200.0, 400.0, 800.0]
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    for beta in beta_schedule:
        # Run many Adam steps for the current smoothing parameter.
        while time.time() < time_deadline:
            t += 1
            conv = _fft_self_convolve(a)                    # length 2n‑1
            max_val = conv.max()
            shifted = conv - max_val                       # ≤ 0
            # softmax weights (stable)
            exp_beta = np.exp(beta * shifted)
            w = exp_beta / exp_beta.sum()
            # gradient of the smooth max: 2·(w * a_rev) convolved
            a_rev = a[::-1]
            grad_conv = _fft_convolve(w, a_rev)            # length 3n‑2
            grad = 2.0 * grad_conv[n - 1 : n - 1 + n]       # ∂smooth_max/∂a
            grad_J = 2.0 * n * grad                         # ∂J/∂a  (J = 2·n·smooth_max)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_J
            v = beta2 * v + (1.0 - beta2) * (grad_J ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = 0.15 / (np.sqrt(v_hat) + eps)
            a -= step * m_hat
            a = _project_onto_simplex(a, s=1.0)

            # Periodic exact evaluation
            if t % 250 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-14:
                    best_score, best_seq = cur, a.copy()
                    print(
                        f"[{time.time():.1f}] β={beta:.1f} it={t} new best {best_score:.6f}"
                    )
            # Give up on this β after a few thousand iterations
            if t % 5000 == 0:
                break
            if time.time() >= time_deadline:
                break
        if time.time() >= time_deadline:
            break
    return best_seq, best_score

# ------------------------------------------------------------
# L‑BFGS‑B refinement (if SciPy is present)
# ------------------------------------------------------------
def _l_bfgs_refine(
    a0: np.ndarray,
    time_deadline: float,
    beta: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """High‑β L‑BFGS‑B on the smooth surrogate (softmax parametrisation)."""
    if not _SCIPY_AVAILABLE:
        return best_seq, best_score

    n = a0.shape[0]
    eps = 1e-12
    z0 = np.log(a0 + eps)  # initialise logits so that softmax(z0)=a0

    def fun(z):
        a = np.exp(z)
        a_sum = a.sum()
        a = a / a_sum
        conv = _fft_self_convolve(a)                     # length 2n‑1
        # smooth max = (1/β)·log‑sum‑exp
        scaled = beta * conv
        max_scaled = scaled.max()
        lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
        smooth_max = lse / beta
        J = 2.0 * n * smooth_max

        # softmax weights for the gradient
        exp_beta = np.exp(beta * (conv - conv.max()))   # stable
        w = exp_beta / exp_beta.sum()

        # ∂J/∂a = 4·n·conv(w, a_rev) (central part)
        a_rev = a[::-1]
        grad_conv = _fft_convolve(w, a_rev)
        grad_a = 4.0 * n * grad_conv[n - 1 : n - 1 + n]

        # Chain rule through the softmax: ∂a/∂z = diag(a) – a aᵀ
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)
        return J, grad_z

    res = minimize(
        fun,
        z0,
        method="L-BFGS-B",
        jac=True,
        options={"maxiter": 300, "disp": False},
    )
    a_opt = np.exp(res.x)
    a_opt /= a_opt.sum()
    sc_opt = _objective(a_opt)
    if sc_opt < best_score - 1e-14:
        best_score, best_seq = sc_opt, a_opt.copy()
        print(f"[{time.time():.1f}] L‑BFGS refined new best {best_score:.6f}")
    return best_seq, best_score

# ------------------------------------------------------------
# Exact sub‑gradient descent (exponentiated gradient)
# ------------------------------------------------------------
def _subgradient_descent_exact(
    a0: np.ndarray,
    time_deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 200,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true L∞‑objective."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        # Sub‑gradient: sum of shifted copies for each active index
        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)          # EG keeps positivity
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score, best_seq = cur, a.copy()
                print(f"[{time.time():.1f}] subgrad it={t} new best {best_score:.6f}")
    return best_seq, best_score

# ------------------------------------------------------------
# Greedy pairwise mass‑transfer hill‑climber
# ------------------------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly relocate a fraction of mass; accept only strict improvements."""
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45  # move up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score, best_seq = cur, a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a step‑function (plain Python list) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # small safety margin at the end
    deadline = start + budget_s - 0.5

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    #  Use the previously discovered sequence if it is available
    # ------------------------------------------------------------------
    try:
        seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        seq = np.maximum(seq, 0.0)
        tot = seq.sum()
        if tot == 0.0:
            seq = _distribution_uniform(len(seq))
        else:
            seq = seq / tot
        sc = _objective(seq)
        overall_best_seq = seq.copy()
        overall_best_score = sc
        print(f"[{time.time():.1f}] start from height_sequence_1 len={len(seq)} score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    #  Fallback seed (if no previous sequence)
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 4000
        seq = _distribution_edge_weighted(n, 2.0)
        seq = seq / seq.sum()
        sc = _objective(seq)
        overall_best_seq = seq.copy()
        overall_best_score = sc
        print(f"[{time.time():.1f}] fallback seed len={n} score={sc:.6f}")

    # ------------------------------------------------------------------
    #  Build a small pool of seeds (perturbations + a few random Dirichlet draws)
    # ------------------------------------------------------------------
    seeds: list[np.ndarray] = [overall_best_seq.copy()]
    # perturb the current best a few times
    for _ in range(3):
        if time.time() >= deadline:
            break
        pert = overall_best_seq.copy()
        noise = rng.normal(scale=0.02, size=pert.shape)
        pert = pert + noise
        pert = np.maximum(pert, 0.0)
        tot = pert.sum()
        if tot == 0.0:
            pert = _distribution_uniform(len(pert))
        else:
            pert = pert / tot
        seeds.append(pert)

    # add a couple of Dirichlet random seeds
    for _ in range(2):
        if time.time() >= deadline:
            break
        n = overall_best_seq.shape[0]
        a = rng.dirichlet(np.full(n, 0.5))
        seeds.append(a)

    # ------------------------------------------------------------------
    #  Stage 1 – Smoothed‑max mirror descent on each seed
    # ------------------------------------------------------------------
    beta_sched = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0,
                  50.0, 100.0, 200.0, 400.0, 800.0]
    for idx, seed_seq in enumerate(seeds):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] smoothed descent on seed {idx}")
        best_seq, best_score = _smooth_max_descent(
            seed_seq,
            time_deadline=deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            beta_schedule=beta_sched,
        )
        if best_score < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = best_seq.copy(), best_score
            print(
                f"[{time.time():.1f}] after smooth seed {idx} new global best {overall_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    #  Stage 2 – L‑BFGS‑B refinement (high β)
    # ------------------------------------------------------------------
    if time.time() < deadline and _SCIPY_AVAILABLE:
        print(f"[{time.time():.1f}] L‑BFGS refinement")
        overall_best_seq, overall_best_score = _l_bfgs_refine(
            overall_best_seq,
            time_deadline=deadline,
            beta=800.0,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    #  Stage 3 – Exact sub‑gradient (exponentiated‑gradient) descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] exact sub‑gradient refinement")
        overall_best_seq, overall_best_score = _subgradient_descent_exact(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.9,
            eval_every=150,
        )
        print(
            f"[{time.time():.1f}] after sub‑gradient best {overall_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    #  Stage 4 – Final greedy pairwise hill‑climber
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy swapping")
        overall_best_seq, overall_best_score = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    #  Return the best sequence (simple list, unscaled – evaluate_sequence rescales)
    # ------------------------------------------------------------------
    print(
        f"Final best objective (2·n·max(conv)) = {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
