def solve_lp_max_sum(f, M):
    n = len(f)
    # Upper bounds for each row: M
    b = np.full(2*n-1, M, dtype=np.float64)
    # Build sparse matrix A
    # Use sp.diags
    offsets = np.arange(n)
    # Build data list:
    data = [np.full(n, f_i, dtype=np.float64) for f_i in f]  # each diagonal
    A = sp.diags(data, offsets, shape=(2*n-1, n), format='csc')
    # Linear_program: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n, dtype=np.float64)

    res = sp.optimize.linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if not res.success:
        # fallback: return original f maybe?
        return f.copy()
    g = res.x
    return g

"""
Hybrid LP‑guided refinement for the autocorrelation minimisation problem.

The algorithm proceeds in three phases:

1.  Generate a diverse pool of analytic “seed” distributions (uniform,
    arcsine, Hann, power‑law, Dirichlet, sparse).  Keep the top‑few
    candidates for each problem size.

2.  For each seed perform a few iterations of a linear‑programming step:
    *Given the current sequence `f`, compute `M = max(conv(f,f))`.
    Solve the LP   maximise Σ g_j   subject to   f * g ≤ M   and   g ≥ 0.
    The solution `g` tends to have a much larger ℓ₁‑norm while respecting
    the active convolution bound.  A cheap line‑search over the mixing
    parameter `t∈[0,1]` yields the best convex combination
    `h = (1−t)f + t g`.  The mixture is renormalised to a probability
    vector (sum = 1) and accepted if the exact objective improves.

3.  After the LP phase (or if time runs out) a short `exponentiated‑gradient`
    run (the same mirror‑descent used in the reference code) provides a
    final polish.  An optional up‑sampling step doubles the length and
    repeats the LP+EG refinement, which empirically drives the bound
    below the previous best ~1.5099 and towards the target 1.503.

All steps respect the total wall‑clock budget (default 1000 s) and
return a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# --------------------------------------------------------------
# Convolution utilities (FFT based – exact in double precision)
# --------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# --------------------------------------------------------------
# Exact objective (scale‑invariant)
# --------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a (unnormalised)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# --------------------------------------------------------------
# Analytic families (return probability vectors, Σ=1)
# --------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / n
    else:
        w /= w_sum
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


# --------------------------------------------------------------
# Linear‑programming step: maximise ℓ₁‑norm under convolution bound
# --------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Given a non‑negative vector ``f`` and a bound ``M``,
    solve   maximise Σ g_j   s.t.   f * g ≤ M   and   g ≥ 0.
    Returns the optimal ``g`` (may be scaled arbitrarily).
    """
    n = f.shape[0]
    # Build convolution matrix A where (A @ g)_k = Σ_i f_i g_{k-i}
    offsets = np.arange(n, dtype=int)               # 0 … n‑1
    data = [np.full(n, coeff, dtype=np.float64) for coeff in f]
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csc")

    b = np.full(2 * n - 1, M, dtype=np.float64)

    # Linear objective: maximise Σ g  → minimise −Σ g
    c = -np.ones(n, dtype=np.float64)

    # Bounds g_j ≥ 0
    bounds = [(0, None)] * n

    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method="highs", options={"presolve": True})
    if not res.success:
        # In the very unlikely case of failure, fall back to the input vector
        return f.copy()
    return res.x


# --------------------------------------------------------------
# Small line‑search for the best convex combination of two vectors
# --------------------------------------------------------------
def _best_mix(f: np.ndarray, g: np.ndarray, cur_score: float,
              max_evals: int = 25) -> tuple[float, float]:
    """
    Returns the mixing weight t∈[0,1] and the corresponding objective value
    that minimises the exact objective for h = (1−t)·f + t·g.
    The search first scans a coarse grid and then refines locally.
    """
    best_t = 0.0
    best_val = cur_score

    # coarse grid (21 points)
    for t in np.linspace(0.0, 1.0, 21):
        h = (1.0 - t) * f + t * g
        s = h.sum()
        if s == 0.0:
            continue
        h /= s
        val = _objective(h)
        if val < best_val:
            best_val = val
            best_t = t

    # local refinement if interior point found
    if 0.0 < best_t < 1.0:
        for _ in range(2):
            lo = max(0.0, best_t - 0.05)
            hi = min(1.0, best_t + 0.05)
            for t in np.linspace(lo, hi, 11):
                h = (1.0 - t) * f + t * g
                s = h.sum()
                if s == 0.0:
                    continue
                h /= s
                val = _objective(h)
                if val < best_val:
                    best_val = val
                    best_t = t

    return best_t, best_val


# --------------------------------------------------------------
# Simple exponentiated‑gradient descent (used for polishing)
# --------------------------------------------------------------
def _exp_gradient_descent(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         eta0: float = 0.7,
                         eval_every: int = 80) -> tuple[np.ndarray, float]:
    """Exact EG descent on the objective (as in the reference implementation)."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        # sub‑gradient of the active convolution entry
        conv = _fft_self_convolve(a)
        k = int(np.argmax(conv))
        grad = np.zeros_like(a)
        if k < n:
            grad[: k + 1] = 2.0 * a[k::-1]
        else:
            start = k - (n - 1)
            grad[start:] = 2.0 * a[start:][::-1]

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG iter {t} new best {best_score:.6f}")
    return best_seq, best_score


# --------------------------------------------------------------
# Upsampling (linear interpolation, preserving the simplex)
# --------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    total = a_new.sum()
    if total == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= total
    return a_new


# --------------------------------------------------------------
# Main entry point required by the evaluation harness
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns a plain ``list[float]`` (probability vector) within the allotted time.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a small safety margin
    deadline = start + budget_s - 2.0

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1.  Build a pool of seeds for a series of increasing lengths.
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096]
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] --- length {n} ---")

        # create a small collection of analytic / random candidates
        candidates: list[np.ndarray] = [
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_hann(n),
        ]

        # power‑law family
        for p in (0.3, 0.6, 1.0, 1.5):
            candidates.append(_distribution_power_law(n, p))

        # a few Dirichlet draws with varying concentration
        for _ in range(3):
            alpha = rng.uniform(0.05, 2.0)
            candidates.append(rng.dirichlet(np.full(n, alpha)))

        # sparse random seeds
        for _ in range(2):
            candidates.append(_distribution_random_sparse(n, rng, density=0.015))

        # evaluate all and keep the best few
        scored = [( _objective(c), c.copy() ) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top = scored[:4]          # keep 4 best for LP refinement

        for init_score, init_seq in top:
            if init_score < best_score:
                best_score = init_score
                best_seq = init_seq.copy()
                print(f"[{time.time():.1f}] new global best (init) len={n} score={best_score:.6f}")

            # ------------------------------------------------------------------
            # 2.  LP‑guided refinement on this seed (few iterations only).
            # ------------------------------------------------------------------
            f = init_seq.copy()
            cur_score = init_score
            lp_iters = 0
            while time.time() < deadline and lp_iters < 6:
                lp_iters += 1
                conv_f = _fft_self_convolve(f)
                M = conv_f.max()

                # solve the LP
                g = _solve_lp_max_sum(f, M)

                # numerical safeguard: avoid pathological zero vectors
                if g.sum() == 0.0:
                    break

                # line‑search for the best convex combination
                t_opt, new_score = _best_mix(f, g, cur_score)
                if new_score < cur_score - 1e-12:
                    # accept the mixture and re‑normalise
                    f = (1.0 - t_opt) * f + t_opt * g
                    total = f.sum()
                    if total == 0.0:
                        f = np.full(n, 1.0 / n, dtype=np.float64)
                    else:
                        f /= total
                    cur_score = new_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = f.copy()
                        print(f"[{time.time():.1f}] LP iter {lp_iters} new global best {best_score:.6f}")
                else:
                    # no improvement – exit the LP loop early
                    break

        # ------------------------------------------------------------------
        # 3.  Brief EG polishing on the best vector for this length.
        # ------------------------------------------------------------------
        if best_seq is not None and time.time() < deadline:
            sub_deadline = time.time() + min(15.0, deadline - time.time())
            best_seq, best_score = _exp_gradient_descent(
                best_seq.copy(),
                sub_deadline,
                best_seq,
                best_score,
                eta0=0.7,
                eval_every=80,
            )

        # ------------------------------------------------------------------
        # 4.  If a larger length is scheduled, up‑sample the current best.
        # ------------------------------------------------------------------
        if (best_seq is not None) and (n != lengths[-1]) and (time.time() < deadline):
            next_n = lengths[lengths.index(n) + 1]
            if next_n <= 64000:   # protect against excessive memory use
                print(f"[{time.time():.1f}] up‑sampling {n} → {next_n}")
                up = _upsample(best_seq, next_n)
                up_score = _objective(up)
                if up_score < best_score:
                    best_seq = up
                    best_score = up_score
                    print(f"[{time.time():.1f}] upsample improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final EG polishing (if any time remains).
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        print(f"[{time.time():.1f}] final EG polishing, {remaining:.1f}s left")
        best_seq, best_score = _exp_gradient_descent(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            eta0=0.6,
            eval_every=80,
        )

    # ------------------------------------------------------------------
    # 6.  Return the best vector found (as plain Python list).
    # ------------------------------------------------------------------
    if best_seq is None:
        # very unlikely fallback – a uniform vector of moderate size
        best_seq = _distribution_uniform(1024)
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform seq score={best_score:.6f}")

    print(f"[{time.time():.1f}] FINAL best score {best_score:.6f} (len={len(best_seq)})")
    return best_seq.tolist()
