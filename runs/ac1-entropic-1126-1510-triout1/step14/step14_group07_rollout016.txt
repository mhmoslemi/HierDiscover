"""
Propose a non‑negative step‑function (list of floats) that minimises

    evaluate_sequence(seq) = 2·len(seq)·max(conv(seq,seq)) / (sum(seq)**2)

The algorithm combines:

1️⃣ Analytic families (uniform, sine‑weighted, centre‑/side‑exponential,
   Gaussian, raised‑cosine, Dirichlet, sparse random) to obtain a diverse
   set of seeds for several lengths.

2️⃣ A long mirror‑descent optimisation on the exact objective.
   The optimiser works on the log‑weights (soft‑max parametrisation) and
   uses a sub‑gradient built from *all* convolution entries whose value
   lies within a small fraction of the current maximum.  A small amount
   of Gaussian noise is injected periodically to escape shallow minima.
   The learning‑rate decays as 0.8/√t, giving a fine‑grained final search.

3️⃣ Two‑stage local polishing:

   – a cheap simulated‑annealing style random walk with a slow cooling
     schedule.

All phases respect the overall 1000 s budget; the final vector is

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = v.astype(np.float64, copy=False)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a (real) FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """Exact objective  2·n·max(conv(a,a))  (assumes sum(a)=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (all return a probability vector – sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    w /= w.sum()
    return w


def _distribution_exponential_center(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(i - half))
    w /= w.sum()
    return w


def _distribution_exponential_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-rate * i)
    w /= w.sum()
    return w


def _distribution_exponential_right(n: int, rate: float) -> np.ndarray:
    return _distribution_exponential_left(n, rate)[::-1]


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    w = np.exp(-0.5 * (i / sigma) ** 2)
    w /= w.sum()
    return w


def _distribution_raised_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    w = np.cos(np.pi * i / (n + 1)) ** 2
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator, conc: float) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a.astype(np.float64)


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
def _adam_refine(
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    lr0: float = 0.8,
    active_frac: float = 0.015,
    max_active: int | None = None,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_interval: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using Adam and a sub‑gradient of the
    exact max‑convolution objective.  All convolution entries whose value
    is within ``active_frac`` of the current maximum are taken into the
    active set (no hard cap unless ``max_active`` is supplied).  Every
    20 000 iterations a tiny Gaussian perturbation is added to the logits
    to help escape flat regions.
    """
    n = init_a.shape[0]
    x = np.log(np.maximum(init_a, 1e-12))                 # log‑weights
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    ind_grid = np.arange(n, dtype=np.int32)

    best_seq = init_a.copy()
    best_score = _objective(init_a)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        a = _softmax(x)                               # stays on simplex

        # --- convolution and active set ---------------------------------
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)
        act = np.where(conv >= thresh)[0]

        if max_active is not None and act.size > max_active:
            # keep only the largest contributions
            top = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top]

        # --- sub‑gradient (sum of shifted copies) -----------------------
        grad_a = np.zeros_like(a)
        for k in act:
            offset = k - ind_grid
            mask = (offset >= 0) & (offset < n)
            grad_a[mask] += 2.0 * a[offset[mask]]

        # gradient w.r.t. log‑weights
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # --- Adam update ------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = lr0 / math.sqrt(t)                      # decaying step‑size
        x -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional random jitter to avoid stagnation
        if t % 20_000 == 0:
            x += 0.02 * np.random.randn(n)

        # --- keep the best seen ----------------------------------------
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] adam it={t:,} new best {best_score:.6f}")

    # final safety projection
    best_seq = _project_onto_simplex(best_seq, s=1.0)
    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Greedy deterministic mass‑transfer between two coordinates."""
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)               # donor
        j = rng.integers(n - 1)           # receiver
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.01,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Very cheap simulated‑annealing style random walk."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Main entry‑point used by the evaluator – returns a plain Python list."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0    # tiny safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1️⃣  Seed generation (broad and diverse)
    lengths = [2000, 3000, 4000, 6000, 8000, 12000]
    p_grid = np.logspace(-1, 1.0, 10)       # sine‑weighted exponents
    lam_grid = np.linspace(0.0, 0.5, 7)    # mixing with uniform
    rate_grid = np.linspace(0.001, 0.025, 8)
    conc_grid = [0.05, 0.2, 1.0, 5.0]
    sigma_grid = np.linspace(0.2 * max(lengths), 2.0 * max(lengths), 6)  # for Gaussian

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        bases = [
            _distribution_uniform(n),
            _distribution_sine_weighted(n, p=1.0),
            _distribution_exponential_center(n, rate=0.005),
            _distribution_exponential_left(n, rate=0.003),
            _distribution_exponential_right(n, rate=0.003),
            _distribution_raised_cosine(n),
        ]
        for base in bases:
            sc = _objective(base)
            top_candidates.append((sc, base.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new global best (det) len={n} score={sc:.6f}")

        # sine‑weighted + uniform mix
        for p in p_grid:
            if time.time() >= deadline:
                break
            sine = _distribution_sine_weighted(n, p)
            for lam in lam_grid:
                cand = _mix_with_uniform(sine, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new global best sine p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # centre‑exponential sweep
        for rate in rate_grid:
            if time.time() >= deadline:
                break
            cad = _distribution_exponential_center(n, rate)
            sc = _objective(cad)
            top_candidates.append((sc, cad.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cad.copy()
                print(f"[{time.time():.1f}] new global best exp‑center rate={rate:.4f} score={sc:.6f}")

        # Gaussian sweep (very broad shapes)
        for sigma in sigma_grid:
            if time.time() >= deadline:
                break
            gauss = _distribution_gaussian(n, sigma)
            sc = _objective(gauss)
            top_candidates.append((sc, gauss.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, gauss.copy()
                print(f"[{time.time():.1f}] new global best Gaussian sigma={sigma:.1f} score={sc:.6f}")

        # Dirichlet / sparse random seeds
        for conc in conc_grid:
            for _ in range(3):
                if time.time() >= deadline:
                    break
                a = _distribution_dirichlet(n, rng, conc)
                sc = _objective(a)
                top_candidates.append((sc, a.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, a.copy()
                    print(f"[{time.time():.1f}] new global best Dirichlet conc={conc:.3g} score={sc:.6f}")

        for _ in range(4):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}")

        # keep only the top‑12 seeds so far
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:12]

    # ------------------------------------------------------------------
    # try to inject a previously found good construction if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2️⃣  Short Adam refinement for each retained seed
    refined: list[tuple[float, np.ndarray]] = []
    for sc, cand in top_candidates:
        if time.time() >= deadline:
            break
        slice_deadline = time.time() + min(20.0, (deadline - time.time()) * 0.04)
        best_seq, best_sc = _adam_refine(
            cand.copy(),
            slice_deadline,
            max_iters=15_000,
            lr0=0.7,
            active_frac=0.025,
            max_active=30,
            beta1=0.9,
            beta2=0.999,
            eval_interval=250,
            best_seq=cand,
            best_score=sc,
        )
        refined.append((best_sc, best_seq))

    if refined:
        overall_best_score, overall_best_seq = min(refined, key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 3️⃣  Long Adam descent from the overall best seed
    if time.time() < deadline:
        remaining = deadline - time.time()
        long_deadline = time.time() + max(0.0, remaining - 30.0)
        print(f"[{time.time():.1f}] start long Adam descent (≈{remaining-30:.1f}s left)")
        overall_best_seq, overall_best_score = _adam_refine(
            overall_best_seq.copy(),
            long_deadline,
            max_iters=700_000,
            lr0=0.9,
            active_frac=0.02,
            max_active=None,                 # use *all* near‑max entries
            beta1=0.9,
            beta2=0.999,
            eval_interval=300,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] after long descent best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Greedy pairwise refinement (moderate number of trials)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=400_000,
        )

    # ------------------------------------------------------------------
    # 5️⃣  Simulated‑annealing polishing
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.015,
            decay=0.9993,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score (exact) ≈ {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # ``evaluate_sequence`` will re‑normalise; we keep a proper probability vector.
    return overall_best_seq.tolist()
