"""
Search for a non‑negative step‑function (vector) that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm consists of:

1️⃣  **Broad seeding** – generate a diverse collection of analytic families
    (uniform, sine‑weighted, centre‑exponential, left/right‑exponential,
    Dirichlet, sparse random) for several lengths.  The 15 best seeds are kept.

2️⃣  **Exact‑gradient Adam** – each retained seed is short‑run optimised with a
    custom Adam optimiser that uses the *exact* sub‑gradient of the
    L∞‑objective (only the true maximal convolution entry(s) contribute).
    The best short‑run candidate becomes the starting point for a long
    descent that consumes the remaining budget.

3️⃣  **Fine‑grained local moves** – greedy pairwise mass transfer followed by a
    cheap simulated‑annealing style search further polishes the solution.

4️⃣  **Symmetrisation & smoothing** – a final deterministic pass tries to
    improve the best vector by enforcing symmetry and applying a tiny smoothing
    kernel.  All steps respect the global 1000 s time budget.

The code is self‑contained; the evaluator will re‑normalise the returned list,
so the optimisation works on probability vectors (sum = 1).
"""

# Helper utilities ------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – always returns a probability vector."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """Exact objective  2·n·max(conv(a,a))  (a is assumed to sum to 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families – return probability vectors (sum = 1) ----------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    w /= w.sum()
    return w


def _distribution_exponential_center(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(i - half))
    w /= w.sum()
    return w


def _distribution_exponential_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-rate * i)
    w /= w.sum()
    return w


def _distribution_exponential_right(n: int, rate: float) -> np.ndarray:
    return _distribution_exponential_left(n, rate)[::-1]


def _distribution_dirichlet(n: int, rng: np.random.Generator, conc: float) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a.astype(np.float64)


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Adam optimiser with *exact* sub‑gradient of the max‐convolution -----
def _adam_refine_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    lr0: float = 0.8,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_interval: int = 200,
    best_seq: np.ndarray | None = None,
    best_score: float = float("inf"),
) -> tuple[np.ndarray, float]:
    """
    Adam on the soft‑max parameters using the *exact* sub‑gradient of
    the L∞‑objective (gradient taken only from the truly maximal convolution
    entry/entries).  The returned vector is a probability vector (sum = 1).
    """
    n = init_a.shape[0]
    x = np.log(np.maximum(init_a, 1e-12))                # log‑weights
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    ind_grid = np.arange(n, dtype=np.int32)

    if best_seq is None:
        best_seq = init_a.copy()
        best_score = _objective(init_a)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        a = _softmax(x)               # stays on the simplex
        conv = _fft_self_convolve(a)
        max_c = conv.max()

        # exact active set – indices whose value equals the maximum (within tiny tol)
        # Using a relative tolerance guards against floating‑point noise.
        tol = max_c * 1e-12
        act = np.where(conv >= max_c - tol)[0]

        # gradient of the max (sum of per‑active‑entry contributions)
        grad_a = np.zeros_like(a)
        for k in act:
            offset = k - ind_grid
            mask = (offset >= 0) & (offset < n)
            grad_a[mask] += 2.0 * a[offset[mask]]

        # convert to gradient w.r.t. the log‑weights (mirror‑descent on simplex)
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update (learning‑rate decays like 1/√t)
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = lr0 / math.sqrt(t)
        x -= eta * m_hat / (np.sqrt(v_hat) + eps)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] adam it={t:,} new best {best_score:.6f}")

    # final safety projection (numerical drifts may appear)
    best_seq = _project_onto_simplex(best_seq, s=1.0)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise mass transfer -----------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing style local search --------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
# Deterministic polishing – symmetry + tiny smoothing -----------------
def _symmetrize_and_smooth(a: np.ndarray, deadline: float) -> np.ndarray:
    """Iteratively enforce symmetry and apply a 3‑point smoothing kernel."""
    while time.time() < deadline:
        improved = False

        # 1) symmetrisation
        a_sym = (a + a[::-1]) * 0.5
        if _objective(a_sym) + 1e-12 < _objective(a):
            a = a_sym
            improved = True
            continue

        # 2) smoothing with a tiny Gaussian‑like kernel
        kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
        a_smooth = np.convolve(a, kernel, mode="same")
        a_smooth = np.maximum(a_smooth, 0.0)
        a_smooth /= a_smooth.sum()
        if _objective(a_smooth) + 1e-12 < _objective(a):
            a = a_smooth
            improved = True
            continue

        if not improved:
            break
    return a

# Main entry point -------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Return a list of non‑negative heights that (hopefully) beats the current record."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # a tiny safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1️⃣  Seed generation – a deliberately rich collection
    lengths = [2000, 3000, 4000, 6000, 8000]
    p_grid = np.logspace(-1, 1.0, 9)          # sine‑weighted exponents
    lam_grid = np.linspace(0.0, 0.5, 6)       # mixing with uniform
    rate_grid = np.linspace(0.001, 0.02, 7)   # exponential rates
    conc_grid = [0.02, 0.05, 0.2, 1.0, 5.0]  # Dirichlet concentrations

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic analytic families
        bases = [
            _distribution_uniform(n),
            _distribution_sine_weighted(n, p=1.0),
            _distribution_exponential_center(n, rate=0.005),
            _distribution_exponential_left(n, rate=0.003),
            _distribution_exponential_right(n, rate=0.003),
        ]
        for base in bases:
            sc = _objective(base)
            top_candidates.append((sc, base.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new global best (det) len={n} score={sc:.6f}")

        # sine‑weighted family with uniform mixing
        for p in p_grid:
            if time.time() >= deadline:
                break
            sine = _distribution_sine_weighted(n, p)
            for lam in lam_grid:
                cand = _mix_with_uniform(sine, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new global best sine p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # exponential families
        for rate in rate_grid:
            if time.time() >= deadline:
                break
            cad = _distribution_exponential_center(n, rate)
            sc = _objective(cad)
            top_candidates.append((sc, cad.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cad.copy()
                print(f"[{time.time():.1f}] new global best exp‑center rate={rate:.4f} score={sc:.6f}")

        # Dirichlet draws – a handful of randomised seeds
        for conc in conc_grid:
            for _ in range(4):
                if time.time() >= deadline:
                    break
                a = _distribution_dirichlet(n, rng, conc)
                sc = _objective(a)
                top_candidates.append((sc, a.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, a.copy()
                    print(f"[{time.time():.1f}] new global best Dirichlet conc={conc:.3g} score={sc:.6f}")

        # sparse random vectors
        for _ in range(5):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}")

        # keep only a modest but diverse set of elite seeds
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:15]

    # ------------------------------------------------------------------
    # If a previous solution is available, add it to the pool
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2️⃣  Short Adam runs for every elite seed
    refined: list[tuple[float, np.ndarray]] = []
    for sc, cand in top_candidates:
        if time.time() >= deadline:
            break
        # give each seed a tiny slice of the remaining budget
        slice_deadline = time.time() + min(15.0, (deadline - time.time()) * 0.04)
        best_seq, best_sc = _adam_refine_exact(
            cand.copy(),
            slice_deadline,
            max_iters=15_000,
            lr0=0.9,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            eval_interval=150,
            best_seq=cand,
            best_score=sc,
        )
        refined.append((best_sc, best_seq))

    # keep the best refined candidate
    if refined:
        overall_best_score, overall_best_seq = min(refined, key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 3️⃣  Long Adam descent on the overall champion
    if time.time() < deadline:
        remaining = deadline - time.time()
        long_deadline = time.time() + max(0.0, remaining - 30.0)   # leave 30 s for polishing
        print(f"[{time.time():.1f}] long Adam descent (≈{remaining-30:.1f}s left)")
        overall_best_seq, overall_best_score = _adam_refine_exact(
            overall_best_seq.copy(),
            long_deadline,
            max_iters=500_000,
            lr0=1.0,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            eval_interval=300,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] after long descent best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )
        print(f"[{time.time():.1f}] after pairwise best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Cheap simulated‑annealing polishing
    if time.time() < deadline:
        print(f"[{time.time():.1f}] simulated‑annealing search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.012,
            decay=0.9995,
        )
        print(f"[{time.time():.1f}] after SA best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Deterministic symmetry & smoothing pass
    if time.time() < deadline:
        print(f"[{time.time():.1f}] symmetry & smoothing polishing")
        overall_best_seq = _symmetrize_and_smooth(overall_best_seq.copy(), deadline)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] final polished score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best score (exact) ≈ {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # The evaluator will re‑normalise; we keep a proper probability vector.
    return overall_best_seq.tolist()
