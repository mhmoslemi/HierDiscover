"""
Propose a low‑score step‑function for ``evaluate_sequence``.

The algorithm is a multi‑phase search that improves on the previous
implementation by

* adding several new analytic families (exponential, asymmetric Beta,
  Gaussian‑mixture) and a dense random‑permutation search,
* using a larger, adaptive pool of seeds (top‑50 instead of 30),
* performing a longer exponentiated‑gradient polishing on the global best,
* up‑sampling *more aggressively* (up to 65536 points) and polishing each
  up‑sampled version, and
* finishing with a combined pairwise‑mass‑transfer + random‑swap annealing.

All phases respect the 1000 s wall‑time budget.  The objective used
internally is

    f(a) = 2·n·max(conv(a,a))          (∑a = 1)

which is exactly the value returned by ``evaluate_sequence`` after the
normalisation steps inside that function.  The final result is a plain
Python list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project a vector onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast full self‑convolution via FFT (real‑valued)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective for a probability vector (∑a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming a sums to 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families (all return a *probability* vector)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{-p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(‑|x‑½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(α,β) sampled on cell mid‑points."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # Avoid log of 0
    with np.errstate(divide='ignore', invalid='ignore'):
        logp = (alpha - 1) * np.log(x) + (beta - 1) * np.log1p(-x)
        w = np.exp(logp)
        w[(x == 0) | (x == 1)] = 0.0
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_exponential(n: int, lam: float) -> np.ndarray:
    """Decreasing exponential (lam≥0). lam=0 gives uniform."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-lam * i)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_gaussian_mixture(
    n: int, mu1: float, sigma1: float, mu2: float, sigma2: float, w1: float
) -> np.ndarray:
    """Two Gaussian bumps on [0,1] (weights w1 and 1‑w1)."""
    xs = (np.arange(n, dtype=np.float64) + 0.5) / n
    g1 = np.exp(-0.5 * ((xs - mu1) / sigma1) ** 2)
    g2 = np.exp(-0.5 * ((xs - mu2) / sigma2) ** 2)
    w = w1 * g1 + (1.0 - w1) * g2
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.02
) -> np.ndarray:
    """Sparse random vector with given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination ``(1‑λ)·base + λ·uniform``."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution entry (mirrored gradient)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """Return (grad, k) where k gives the index of the maximal convolution entry."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))

    grad = np.zeros_like(a)
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
# Adaptive exponentiated‑gradient descent (mirrored gradient)
# ----------------------------------------------------------------------
def _adaptive_exp_gradient(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    beta: float = 0.9,
    eval_every: int = 50,
) -> tuple[np.ndarray, float]:
    """Run a simple exponentiated‑gradient / mirror‑descent loop."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)  # exponential moving average of gradients
    eta = eta0
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        m = beta * m + (1.0 - beta) * grad

        # multiplicative (mirror) update
        a = a * np.exp(-eta * m)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                eta = min(eta0, eta * 1.02)   # reward improvement
                print(f"[{time.time():.1f}] GD it={t:,} new best {best_score:.6f}")
            else:
                eta *= 0.995               # decay on no‑improve

        # safety projection to avoid numerical drift
        if t % (eval_every * 10) == 0:
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise mass‑transfer + random‑swap annealing
# ----------------------------------------------------------------------
def _pairwise_and_swap_annealing(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Simulated annealing that moves a random fraction and also swaps two entries."""
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        # ----- mass transfer -----
        src = rng.choice(np.where(a > 1e-14)[0])
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1
        delta = a[src] * rng.random() * 0.3
        if delta < 1e-16:
            continue
        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        # ----- random swap (occasionally) -----
        if rng.random() < 0.15 and n > 2:
            i, j = rng.choice(n, size=2, replace=False)
            new_a[i], new_a[j] = new_a[j], new_a[i]

        cur = _objective(new_a)
        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.02

        accept = False
        if cur < best_score - 1e-15:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling while preserving total mass (=1)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:

    # ------------------------------------------------------------------
    if time.time() < deadline:

            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,

    # ------------------------------------------------------------------
    while time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n > 65536:
            break
        print(f"[{time.time():.1f}] up‑sampling {cur_n} → {target_n}")
        up = _upsample(overall_best_seq, target_n)
        up_sc = _objective(up)
        if up_sc < overall_best_score:
            overall_best_seq, overall_best_score = up, up_sc
            print(f"[{time.time():.1f}] up‑sampled new best {overall_best_score:.6f}")

        # medium‑length GD on the up‑sampled vector
        sub_dead = time.time() + (deadline - time.time()) * 0.18
        a_opt, sc_opt = _adaptive_exp_gradient(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.45,
            beta=0.9,
            eval_every=50,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
            print(f"[{time.time():.1f}] GD after up‑sample new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣  Final annealing (mass‑transfer + random swaps)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing phase")
        a_opt, sc_opt = _pairwise_and_swap_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
            print(f"[{time.time():.1f}] after annealing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 8️⃣  Report and return
    # ------------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)) = {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    # convert to a plain Python list for the external evaluator
    return overall_best_seq.tolist()
