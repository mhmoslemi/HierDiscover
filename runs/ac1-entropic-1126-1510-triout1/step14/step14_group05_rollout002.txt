def evolution_strategy(n, time_limit, popsize=30, offspring=200):
    # Init population random Dirichlet
    rng = np.random.default_rng()
    pop = rng.dirichlet(np.ones(n), size=popsize) # each row sums to 1
    scores = np.array([_objective(p) for p in pop])
    best_idx = np.argmin(scores)
    best_seq = pop[best_idx].copy()
    best_score = scores[best_idx]
    # schedule sigma decreasing
    start = time.time()
    generation = 0
    while time.time() < time_limit:
        generation += 1
        # compute sigma: maybe sigma0 * (0.99)**generation
        sigma = max(0.01, 0.2 / (generation**0.5))
        # generate offspring
        children = []
        for _ in range(offspring):
            parent_idx = rng.integers(popsize)
            parent = pop[parent_idx]
            # mutate in log-space
            log_parent = np.log(parent + 1e-12)
            log_child = log_parent + sigma * rng.normal(size=n)
            child = np.exp(log_child)
            child_sum = child.sum()
            if child_sum == 0:
                child = _distribution_uniform(n)
            else:
                child /= child_sum
            children.append(child)
        # evaluate
        child_scores = np.array([_objective(c) for c in children])
        # combine parents + children
        combined = np.vstack((pop, children))
        combined_scores = np.concatenate((scores, child_scores))
        # select best popsize individuals
        idx = np.argpartition(combined_scores, popsize)[:popsize]
        pop = combined[idx]
        scores = combined_scores[idx]
        # update best
        cur_best_idx = np.argmin(scores)
        cur_best_score = scores[cur_best_idx]
        if cur_best_score < best_score:
            best_score = cur_best_score
            best_seq = pop[cur_best_idx].copy()
            print(f"[{time.time():.1f}] ES gen={generation} new best {best_score:.6f}")
    return best_seq, best_score
```

"""
Optimisation of a non‑negative step function for the objective

    f ↦ 2·n·max(conv(f,f))   (where conv is the linear self‑convolution)

The algorithm consists of four stages and works under a 1000 s time budget.

1️⃣  **Analytic families** – a diverse collection of parametric probability
    vectors (uniform, arcsine, beta‑edge, edge‑weighted, exponential‑edge,
    sine, cosine, quadratic, logistic) is scanned on several lengths.
    The best few seeds are kept.

2️⃣  **Evolution Strategy (μ/λ‑ES)** – a simple population based optimiser
    works in log‑space (to keep positivity) and uses exponential‑gradient
    updates with a decreasing mutation strength.  This stage explores the
    high‑dimensional simplex far beyond the analytic families.

3️⃣  **Soft‑max (temperature) gradient descent** – the L∞ objective is
    smoothed with an annealed soft‑max (τ≈2·10⁻² … 10⁻³).  The exact
    gradient of the smoothed objective is obtained by FFT‑based cross‑
    correlation; an exponentiated‑gradient (mirror descent) step is taken
    and the vector is projected back onto the probability simplex.

4️⃣  **Greedy pairwise mass‑transfer polishing** – small random transfers
    of probability mass between two coordinates are accepted only if they
    strictly improve the exact objective.

The best sequence found (as a plain ``list[float]``) is returned; the
calling environment will re‑normalise it inside ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    # next power of two ≥ 2·n‑1
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective (sum = 1 ⇒ 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_edge(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta‑type density w_i ∝ (i·(n‑i))^{α‑1}, α>0."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = (i * (n - i)) ** (alpha - 1.0)
    vals = np.clip(vals, 0.0, None)
    s = vals.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    vals /= s
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weights ∝ (i·(n‑i))^{-p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = (i * (n - i)) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    w /= w_sum
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    """w_i ∝ exp(-c·i) + exp(-c·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_logistic(n: int, scale: float) -> np.ndarray:
    """Symmetric logistic bell: w_i ∝ 1/(1+exp(|i-(n-1)/2|/scale))."""
    i = np.arange(n, dtype=np.float64) + 0.5
    centre = (n - 1) / 2.0
    d = np.abs(i - centre) / scale
    w = 1.0 / (1.0 + np.exp(d))
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse Dirichlet‑like seed with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _mix_two(a: np.ndarray, b: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·a + (1‑λ)·b."""
    return lam * a + (1.0 - lam) * b


# ----------------------------------------------------------------------
# Smooth (soft‑max) gradient descent
# ----------------------------------------------------------------------
def _softmax_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    tau_start: float = 0.02,
    tau_end: float = 0.001,
    lr0: float = 0.6,
    eval_every: int = 250,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a smoothed version of the L∞ objective.
    The temperature τ is linearly annealed from ``tau_start`` to ``tau_end``.
    """
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    it = 0

    while time.time() < time_limit:
        it += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] softmax it={it} new best {best_score:.6f}")

        # anneal τ linearly
        frac = (time.time() - start) / (time_limit - start)
        tau = tau_start * (1.0 - frac) + tau_end * frac

        # soft‑max over all 2n‑1 convolution entries
        maxc = conv.max()
        shifted = conv - maxc  # for numerical stability
        exp_vals = np.exp(shifted / tau)      # ≤ 1
        w = exp_vals / exp_vals.sum()         # length 2n‑1, sums to 1

        # gradient of the smoothed objective:
        #   ∂/∂a  ( Σ w[k]·(a*a)[k] ) = 2·(w ⋆ a)   where (⋆) is cross‑correlation
        # Use FFT cross‑correlation for efficiency.
        # cross‑correlation (valid) of w with a:
        #   (w ⋆ a)[j] = Σ_i w[i+j]·a[i]   (j = 0..n‑1)
        # Implement via FFT:
        m = w.shape[0]
        size = 1 << ((m + n - 1) - 1).bit_length()
        W = np.fft.rfft(w, n=size)
        A = np.fft.rfft(a, n=size)
        corr = np.fft.irfft(W * np.conj(A), n=size)[:n]

        grad = 2.0 * corr

        # exponentiated‑gradient update
        eta = lr0 / math.sqrt(it)
        a *= np.exp(-eta * grad)
        a = _project_onto_simplex(a)   # guarantee Σa = 1, a ≥ 0

        if it % eval_every == 0:
            # already evaluated above
            pass

    return best_seq, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (same style as before)
# ----------------------------------------------------------------------
def _exact_subgradient_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 150,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    while time.time() < time_limit:
        it += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(it)
        a *= np.exp(-eta * grad)
        a = _project_onto_simplex(a)

        if it % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] exact it={it} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise mass‑transfer hill‑climber
# ----------------------------------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        # pick a source index with non‑trivial mass
        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)

        # pick a different target index uniformly
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # move a random fraction of the source mass
        delta = a[i] * np.random.rand() * 0.45  # ≤ 45 %
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Evolution Strategy (μ/λ‑ES) on the simplex
# ----------------------------------------------------------------------
def _evolution_strategy(
    n: int,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    popsize: int = 30,
    offspring: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Simple (μ,λ)‑ES that works in log‑space, projects back onto the simplex,
    and uses an adaptive Gaussian mutation strength.
    """
    # initialise a population from a Dirichlet distribution (concentration = 1)
    pop = rng.dirichlet(np.ones(n), size=popsize)
    scores = np.array([_objective(p) for p in pop])

    # keep track of current best
    cur_best_idx = np.argmin(scores)
    best_seq = pop[cur_best_idx].copy()
    best_score = scores[cur_best_idx]
    print(f"[{time.time():.1f}] ES init best {best_score:.6f}")

    start = time.time()
    generation = 0
    while time.time() < time_limit:
        generation += 1
        # simple schedule for mutation strength σ
        sigma = max(0.005, 0.25 / math.sqrt(generation))

        # produce offspring
        children = np.empty((offspring, n), dtype=np.float64)
        for k in range(offspring):
            parent = pop[rng.integers(popsize)]
            # log‑space mutation
            log_parent = np.log(parent + 1e-18)  # avoid log(0)
            log_child = log_parent + sigma * rng.standard_normal(n)
            child = np.exp(log_child)
            child_sum = child.sum()
            if child_sum == 0.0 or np.isnan(child_sum):
                child = _distribution_uniform(n)
            else:
                child /= child_sum
            children[k] = child

        child_scores = np.array([_objective(c) for c in children])

        # combine and select top‑popsize individuals
        combined = np.vstack((pop, children))
        combined_scores = np.concatenate((scores, child_scores))
        idx = np.argpartition(combined_scores, popsize)[:popsize]
        pop = combined[idx]
        scores = combined_scores[idx]

        # update global best
        cur_best_idx = np.argmin(scores)
        cur_best_score = scores[cur_best_idx]
        if cur_best_score < best_score - 1e-12:
            best_score = cur_best_score
            best_seq = pop[cur_best_idx].copy()
            print(f"[{time.time():.1f}] ES gen={generation} new best {best_score:.6f}")

    return best_seq, best_score

        # allocate 60 % to smooth descent, 40 % to exact polish
        smooth_deadline = time.time() + remaining * 0.6
        overall_best_seq, overall_best_score = _softmax_descent(
            a0=overall_best_seq,
            time_limit=smooth_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            tau_start=0.025,
            tau_end=0.0008,
            lr0=0.55,
            eval_every=200,
        )
        print(f"[{time.time():.1f}] after smooth descent best {overall_best_score:.6f}")

        exact_deadline = time.time() + remaining * 0.4
        overall_best_seq, overall_best_score = _exact_subgradient_descent(
            a0=overall_best_seq,
            time_limit=exact_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.85,
            eval_every=300,
        )
        print(f"[{time.time():.1f}] after exact subgrad best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Final greedy pairwise polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy swapping")
        overall_best_seq, overall_best_score = _pairwise_swap(
            a0=overall_best_seq,
            time_limit=deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] final best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Return the result
    # ------------------------------------------------------------------
    print(
        f"Finished – best objective (2·n·max(conv)) = {overall_best_score:.6f} "
        f"with length = {len(overall_best_seq)}"
    )
    # evaluate_sequence will renormalise internally; we keep sum≈1 for safety.
    return overall_best_seq.tolist()
