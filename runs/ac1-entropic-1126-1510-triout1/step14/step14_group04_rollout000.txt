"""
Hybrid evolutionary+gradient search for a non‑negative step‑function that
minimises ``evaluate_sequence`` (i.e. 2·n·max(conv(a,a)) when Σa=1).

Main ideas
---------
1️⃣  Scan a richer collection of analytic families (including asymmetric β‑,
    power‑law edge, and mixed uniform families) on several lengths.
2️⃣  Keep the best few seeds and run an enhanced smoothed‑max mirror descent
    (Adam) with a finer β‑continuation (up to β≈500) and more iterations.
3️⃣  Refine the current best by exponentiated‑gradient sub‑gradient descent.
4️⃣  Apply an evolutionary step: repeatedly perturb the best vector by mixing
    with a random Dirichlet and run a short smooth‑max descent on the offspring.
5️⃣  Finish with a greedy pairwise‑mass‑transfer hill‑climber.

All steps respect the global time budget (default 1000 s).  The returned
list is a simplex‑normalised probability vector; ``evaluate_sequence`` will
handle the final scaling automatically.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  Utility – Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project vector v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
#  FFT‑based convolution (real inputs)
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    return _fft_convolve(a, a)

# ----------------------------------------------------------------------
#  Exact objective for Σa = 1  (returns 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# ----------------------------------------------------------------------
#  Analytic families (all return a probability vector)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w

def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w

def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w

def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    s = w.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    w /= s
    return w

def _distribution_powerlaw_edge(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i+1)^(-p) + (n-i)^(-p)   (p>0)."""
    i = np.arange(n, dtype=np.float64) + 1.0
    w = i ** (-p) + (n + 1 - i) ** (-p)
    s = w.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    w /= s
    return w

def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w

def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric Beta‑type (α may equal β)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    s = w.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    w /= s
    return w

def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta‑type (α≠β)."""
    return _distribution_beta(n, alpha, beta)

def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# ----------------------------------------------------------------------
#  Smoothed‑max mirror descent (Adam) – more iterations & larger β grid
# ----------------------------------------------------------------------
def _smooth_max_descent(a0: np.ndarray,
                        deadline: float,
                        best_seq: np.ndarray,
                        best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-9
    t = 0
    # β schedule (more points up to 500)
    beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 350.0, 500.0]

    for beta in beta_schedule:
        # we allow up to 4000 inner iterations for each β (stopping early if out of time)
        while time.time() < deadline:
            t += 1
            conv = _fft_self_convolve(a)
            max_val = conv.max()
            # soft‑max weighting
            shifted = conv - max_val               # ≤0
            exp_beta = np.exp(beta * shifted)
            w = exp_beta / exp_beta.sum()          # Σw = 1
            # gradient of smooth max (convolution of w with reversed a)
            a_rev = a[::-1]
            grad_conv = _fft_convolve(w, a_rev)    # length 3n‑2
            grad = 2.0 * grad_conv[n - 1 : n - 1 + n]   # size n
            grad_J = 2.0 * n * grad

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_J
            v = beta2 * v + (1.0 - beta2) * (grad_J ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = 0.12 / (np.sqrt(v_hat) + eps)   # tuned step size
            a -= step * m_hat
            a = _project_onto_simplex(a, s=1.0)

            if t % 80 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-14:
                    best_score, best_seq = cur, a.copy()
                    print(f"[{time.time():.1f}] β={beta:.1f} it={t} new best {best_score:.6f}")

            # stop after ~3000 iters per β to keep schedule reasonably short
            if t % 3000 == 0:
                break
        if time.time() >= deadline:
            break
    return best_seq, best_score

# ----------------------------------------------------------------------
#  Exact sub‑gradient descent using exponentiated gradient
# ----------------------------------------------------------------------
def _subgradient_descent_exact(a0: np.ndarray,
                               deadline: float,
                               best_seq: np.ndarray,
                               best_score: float,
                               lr0: float = 0.9,
                               eval_every: int = 80) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        # exact sub‑gradient (sum of shifted copies)
        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)   # EG – stays positive
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score, best_seq = cur, a.copy()
                print(f"[{time.time():.1f}] exact subgrad it={t} new best {best_score:.6f}")
    return best_seq, best_score

# ----------------------------------------------------------------------
#  Greedy pairwise mass‑transfer hill‑climber
# ----------------------------------------------------------------------
def _pairwise_swap(a0: np.ndarray,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   max_trials: int | None = None) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * np.random.rand() * 0.45
        if delta < 1e-15:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score, best_seq = cur, a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")
    return best_seq, best_score

# ----------------------------------------------------------------------
#  Helper: sparse random vector
# ----------------------------------------------------------------------
def _distribution_random_sparse(n: int,
                                rng: np.random.Generator,
                                density: float = 0.006) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals

# ----------------------------------------------------------------------
#  Main entry point
# ----------------------------------------------------------------------

    print(f"Final best objective (2·n·max(conv)) = {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # The external ``evaluate_sequence`` rescales internally; returning the simplex‑normalised vector suffices.
    return overall_best_seq.tolist()
