"""
Improved 1000‑second search for a non‑negative step function that minimises
``evaluate_sequence``.  The algorithm proceeds in three phases:

1️⃣  **Rich seed generation** – for several lengths (up to 20000) we enumerate
    a wide variety of analytic families (uniform, arcsine, beta‑edge,
    edge‑weighted, exponential‑edge, double‑exponential, sine, cosine,
    quadratic) on fine grids, plus many random mixtures of them.  Each
    candidate is normalised to sum = 1 and the best few dozen are retained.

2️⃣  **Two‑stage continuous optimisation** – the retained seeds are refined
    with a *soft‑max* (temperature‑annealed) exponentiated‑gradient descent,
    then with an *exact* sub‑gradient EG step on the true L∞ objective.
    Both stages respect the simplex constraint and run for a fixed share of
    the remaining budget.

3️⃣  **Polishing & restarts** – a greedy pairwise mass‑transfer hill‑climber
    is applied, followed by a few random‑jitter restarts (small Gaussian
    perturbations of the current best) each followed by a short smooth+exact
    descent.  The best sequence found at any time is returned as a plain
    Python list (the caller re‑normalises it inside ``evaluate_sequence``).

All heavy work is performed with FFT‑based convolutions (O(n log n) per
iteration).  The code uses only NumPy and the standard library, runs on a
single core but respects the 1000‑second wall‑clock limit.
"""

# Utility helpers

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())  # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_cross_correlate_valid(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (valid part)  (w ⋆ a)[j] = Σ_i w[i+j]·a[i],
    returning an array of length len(a). Implemented with FFT for speed.
    """
    n = a.shape[0]
    m = w.shape[0]                # = 2·n‑1
    size = 1 << ((m + n - 2).bit_length())
    W = np.fft.rfft(w, n=size)
    A = np.fft.rfft(a, n=size)
    corr = np.fft.irfft(W * np.conj(A), n=size)
    return corr[:n]


def _fft_sliding_window_sum(w: np.ndarray, win: int) -> np.ndarray:
    """
    For each j compute Σ_{i=j}^{j+win‑1} w[i] (window length = win).
    Returns an array of length win. Implemented via FFT.
    """
    m = w.shape[0]
    size = 1 << ((m + win - 2).bit_length())
    W = np.fft.rfft(w, n=size)
    ones = np.fft.rfft(np.ones(win, dtype=np.float64), n=size)
    conv = np.fft.irfft(W * ones, n=size)
    start = win - 1
    return conv[start:start + win]


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_edge(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta‑type density w_i ∝ (i·(n‑i))^{α‑1} (α>0)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = (i * (n - i)) ** (alpha - 1.0)
    vals = np.clip(vals, 0.0, None)
    s = vals.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    vals /= s
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i·(n‑i))^{‑p}  (p≥0)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    w /= w_sum
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_double_exponential(n: int, c: float) -> np.ndarray:
    """Symmetric double‑exponential decreasing from the centre."""
    d = np.minimum(np.arange(n, dtype=np.float64),
                   (n - 1 - np.arange(n, dtype=np.float64)))
    w = np.exp(-c * d)
    w /= w.sum()
    return w


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse Dirichlet‑like seed with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Exact objective for a *probability* vector a (sum = 1):
#   2·n·max(conv(a,a))
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth (soft‑max) exponentiated‑gradient descent
# ----------------------------------------------------------------------
def _smooth_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    tau_start: float = 0.02,
    tau_end: float = 0.001,
    lr0: float = 0.6,
    eval_every: int = 200,
) -> tuple[np.ndarray, float]:
    """
    EG descent on a soft‑max approximation of max(conv).  Temperature τ is
    linearly annealed from tau_start to tau_end.  The step‑size decays like
    1/√t.  Updates stay on the probability simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    it = 0

    while time.time() < time_limit:
        it += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] smooth it={it} new best {best_score:.6f}")

        # temperature schedule
        frac = (time.time() - start) / (time_limit - start)
        tau = tau_start * (1.0 - frac) + tau_end * frac

        # soft‑max weights (stable log‑sum‑exp trick)
        shifted = conv - max_val
        exp_vals = np.exp(shifted / tau)
        w = exp_vals / exp_vals.sum()                 # length 2n‑1, sums to 1

        # gradient of the smoothed objective
        term1 = _fft_cross_correlate_valid(w, a)      # shape (n,)
        win_sum = _fft_sliding_window_sum(w, n)       # shape (n,)
        grad = term1 + a * win_sum

        eta = lr0 / math.sqrt(it)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if eval_every and (it % eval_every == 0):
            pass

    return best_seq, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (exponentiated‑gradient on the true L∞)
# ----------------------------------------------------------------------
def _subgradient_descent_exact(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        # exact sub‑gradient: for each maximal index collect contributions
        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if eval_every and (t % eval_every == 0):
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] exact it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise hill‑climber
# ----------------------------------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")
    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a step‑function (plain Python list) that (approximately) minimizes
    ``evaluate_sequence`` within the allotted time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, seq)

    # ------------------------------------------------------------------
    # 1️⃣  Generate a diverse collection of seeds
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000, 16000, 20000]

    # parameter grids (more extensive than before)
    p_grid = np.concatenate([
        np.logspace(-3, -0.5, 12),          # 0.001 … 0.316
        np.logspace(-0.5, 0.0, 8),          # 0.316 … 1
        np.linspace(1.0, 8.0, 15),          # 1 … 8
    ])
    alpha_grid = np.linspace(0.05, 5.0, 12)   # beta‑edge exponent α
    lam_grid = np.linspace(0.0, 1.0, 21)     # mixing with uniform
    c_grid = np.logspace(-3, 0, 12)         # exponential‑edge decay
    double_c_grid = np.logspace(-3, 0, 12)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families (no extra parameters)
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
        ):
            cand = base_fn(n)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # beta‑edge families (α) with mixing
        for alpha in alpha_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_edge(n, float(alpha))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(beta, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] len={n} β α={alpha:.3g} lam={lam:.2f} score={sc:.6f}")

        # edge‑weighted families (p) with mixing
        for p in p_grid:
            if time.time() >= deadline:
                break
            edge = _distribution_edge_weighted(n, float(p))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(edge, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] len={n} edge p={p:.4g} lam={lam:.2f} score={sc:.6f}")

        # exponential‑edge families
        for c in c_grid:
            if time.time() >= deadline:
                break
            exp_edge = _distribution_exponential_edge(n, float(c))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(exp_edge, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] len={n} exp c={c:.4g} lam={lam:.2f} score={sc:.6f}")

        # double‑exponential families
        for c in double_c_grid:
            if time.time() >= deadline:
                break
            dbl = _distribution_double_exponential(n, float(c))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(dbl, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] len={n} dbl‑exp c={c:.4g} lam={lam:.2f} score={sc:.6f}")

        # random Dirichlet seeds (moderate concentration)
        for _ in range(6):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.005)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

        # random mixtures of 2–3 analytic families
        base_fns = [
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
            lambda nn: _distribution_beta_edge(nn, rng.uniform(0.05, 5.0)),
            lambda nn: _distribution_edge_weighted(nn, rng.uniform(0.001, 8.0)),
            lambda nn: _distribution_exponential_edge(nn, rng.uniform(1e-3, 1.0)),
            lambda nn: _distribution_double_exponential(nn, rng.uniform(1e-3, 1.0)),
        ]
        for _ in range(8):
            if time.time() >= deadline:
                break
            k = rng.integers(2, 4)                # mix 2 or 3 families
            chosen = rng.choice(len(base_fns), size=k, replace=False)
            weights = rng.random(k)
            weights /= weights.sum()
            mixed = np.zeros(n, dtype=np.float64)
            for idx, w in zip(chosen, weights):
                mixed += w * base_fns[idx](n)
            mixed = np.maximum(mixed, 0.0)
            s = mixed.sum()
            if s == 0.0:
                mixed = _distribution_uniform(n)
            else:
                mixed /= s
            sc = _objective(mixed)
            top_candidates.append((sc, mixed.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, mixed.copy()
                print(f"[{time.time():.1f}] random mix len={n} score={sc:.6f}")

        # keep only the most promising seeds (memory bound)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:120]   # larger pool than before

    # ------------------------------------------------------------------
    # 2️⃣  Incorporate a previous construction if available
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3️⃣  Fallback (should never fire)
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 8000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Continuous refinement on the best seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # use the ten most promising seeds
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:10]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] start refinement on {len(seeds)} seeds (≈{per_seed:.1f}s each)")

        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            # smooth descent (≈60 % of per‑seed budget)
            smooth_deadline = time.time() + per_seed * 0.6
            a_opt, sc_opt = _smooth_descent(
                seed_seq,
                smooth_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                tau_start=0.02,
                tau_end=0.0008,
                lr0=0.6,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after smooth seed {idx} new global best {overall_best_score:.6f}")

            # exact sub‑gradient polish (≈40 % of per‑seed budget)
            exact_deadline = time.time() + per_seed * 0.4
            a_opt, sc_opt = _subgradient_descent_exact(
                overall_best_seq,
                exact_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after exact polish seed {idx} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Greedy pairwise hill‑climber
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy swap polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Random‑jitter restarts (if time permits)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        n = overall_best_seq.shape[0]
        # add a small Gaussian jitter (≈2 % of each coordinate)
        jitter = rng.normal(scale=0.02, size=n)
        a_jit = overall_best_seq + jitter
        a_jit = np.maximum(a_jit, 0.0)
        s = a_jit.sum()
        if s == 0.0:
            continue
        a_jit /= s

        # short smooth + exact descent on the perturbed point
        short_budget = min(30.0, deadline - time.time())
        mid = time.time() + short_budget * 0.55
        a_opt, sc_opt = _smooth_descent(
            a_jit,
            mid,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            tau_start=0.02,
            tau_end=0.001,
            lr0=0.5,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] jitter smooth new best {overall_best_score:.6f}")

        fin = time.time() + short_budget * 0.45
        a_opt, sc_opt = _subgradient_descent_exact(
            overall_best_seq,
            fin,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.7,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] jitter exact new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best objective (2·n·max(conv)) = {overall_best_score:.6f} (length = {len(overall_best_seq)})"
    )
    # Return plain Python list; evaluate_sequence will re‑normalise.
    return overall_best_seq.tolist()
