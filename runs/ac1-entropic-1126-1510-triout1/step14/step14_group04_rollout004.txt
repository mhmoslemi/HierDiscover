"""
Hybrid optimizer for the step‑function evaluation problem.

The algorithm searches a large discrete simplex using a mixture of:
  * systematic scan of three analytic families (edge‑weighted,
    exponential‑edge, symmetric beta) on a relatively long sequence,
    plus a few random Dirichlet seeds;
  * a smooth‑max mirror‑descent (Adam) with a long β‑continuation and
    many more iterations than the previous state‑of‑the‑art;
  * exponentiated‑gradient sub‑gradient descent on the exact L∞‑objective;
  * a greedy pairwise mass‑transfer hill‑climber to polish the result.

All steps respect the 1000 s time budget and keep the sequence on the
probability simplex (non‑negative, Σa = 1).  The returned list can be fed
directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
#  Fast (real) FFT based convolutions
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
#  Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
#  Analytic families (all return a normalized distribution)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ ((i+0.5)*(n‑i‑0.5))^{‑p} (symmetric, edges heavy for p>0)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = n - i
    w = (i * j) ** (-p)
    s = w.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    w /= s
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    """w_i ∝ exp(-c·i) + exp(-c·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    s = w.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    w /= s
    return w


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric beta‑type law on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n  # (0,1)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    s = w.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    w /= s
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
#  Smoothed‑max mirror descent (Adam) – longer schedule & more iters
# ----------------------------------------------------------------------
def _smooth_max_descent_extended(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp surrogate with an extended β‑schedule."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    # extended schedule – we push β up to 1000 for a very sharp max
    beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0]
    max_iters_per_beta = 6000   # fairly generous

    for beta in beta_schedule:
        iters = 0
        while time.time() < time_limit and iters < max_iters_per_beta:
            t += 1
            iters += 1

            conv = _fft_self_convolve(a)                     # length 2n‑1
            max_val = conv.max()
            # log‑sum‑exp smooth max
            shifted = conv - max_val                        # ≤0
            exp_beta = np.exp(beta * shifted)
            w = exp_beta / exp_beta.sum()                   # Σw = 1

            # gradient of smooth max (see derivation in the paper)
            a_rev = a[::-1]
            grad_conv = _fft_convolve(w, a_rev)             # length 3n‑2
            grad = 2.0 * grad_conv[n - 1 : n - 1 + n]       # size n
            grad_J = 2.0 * n * grad

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_J
            v = beta2 * v + (1.0 - beta2) * (grad_J ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = 0.2 / (np.sqrt(v_hat) + eps)

            a -= step * m_hat
            a = _project_onto_simplex(a, s=1.0)

            # occasional exact evaluation
            if t % 120 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-14:
                    best_score, best_seq = cur, a.copy()
                    print(
                        f"[{time.time():.1f}] β={beta:.0f} it={t} new best {best_score:.6f}"
                    )
        if time.time() >= time_limit:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Exact sub‑gradient descent (exponentiated gradient)
# ----------------------------------------------------------------------
def _subgradient_descent_exact(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.7,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true L∞‑objective."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        # exact sub‑gradient (sum of shifted copies of a)
        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)          # EG keeps positivity
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score, best_seq = cur, a.copy()
                print(
                    f"[{time.time():.1f}] exact subgrad it={t} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Greedy pairwise mass‑transfer hill‑climber
# ----------------------------------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly relocate a fraction of mass; keep only strict improvements."""
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score, best_seq = cur, a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    Returns a plain Python list (Σa = 1, a_i ≥ 0) ready for
    ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.5      # tiny safety margin

    # ------------------------------------------------------------------
    #  Choose a length that balances quality and speed.
    #  Empirically n≈12000 gives a good trade‑off.
    # ------------------------------------------------------------------
    n = 12000
    uniform = _distribution_uniform(n)

    # ------------------------------------------------------------------
    #  1️⃣  Scan three analytic families + a handful of random Dirichlet seeds.
    # ------------------------------------------------------------------
    best_candidates: list[tuple[float, np.ndarray]] = []

    # grids
    p_grid = np.concatenate(
        [
            np.logspace(-3, -0.5, 13),          # extreme edge‑heavy
            np.logspace(-0.5, 0.0, 13),         # moderate
            np.linspace(0.2, 3.0, 16),          # broader range
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 11)       # mixing with uniform
    c_grid = np.logspace(-3, 0, 12)           # exponential‑edge decay
    alpha_grid = np.concatenate(
        [np.logspace(-0.5, 0.0, 12), np.logspace(0.0, 0.8, 12)]
    )                                         # symmetric beta

    # ---- edge‑weighted ------------------------------------------------
    for p in p_grid:
        if time.time() >= deadline:
            break
        base = _distribution_edge_weighted(n, float(p))
        for lam in lam_grid:
            if time.time() >= deadline:
                break
            cand = _mix_with_uniform(base, float(lam))
            sc = _objective(cand)
            best_candidates.append((sc, cand.copy()))

    # ---- exponential‑edge --------------------------------------------
    for c in c_grid:
        if time.time() >= deadline:
            break
        base = _distribution_exponential_edge(n, float(c))
        for lam in lam_grid:
            if time.time() >= deadline:
                break
            cand = _mix_with_uniform(base, float(lam))
            sc = _objective(cand)
            best_candidates.append((sc, cand.copy()))

    # ---- symmetric beta ------------------------------------------------
    for a in alpha_grid:
        if time.time() >= deadline:
            break
        cand = _distribution_beta(n, float(a), float(a))
        sc = _objective(cand)
        best_candidates.append((sc, cand.copy()))

    # ---- few Dirichlet random draws ------------------------------------
    for _ in range(8):
        if time.time() >= deadline:
            break
        alpha = rng.uniform(0.01, 2.0)
        cand = rng.dirichlet(np.full(n, alpha))
        sc = _objective(cand)
        best_candidates.append((sc, cand.copy()))

    # ---- consider the previously known construction if available -----
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            if prev.size != n:
                # simple linear interpolation to the new length
                x_old = np.linspace(0.0, 1.0, prev.size)
                x_new = np.linspace(0.0, 1.0, n)
                cand = np.interp(x_new, x_old, prev)
            else:
                cand = prev.copy()
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            cand = uniform if s == 0.0 else cand / s
            sc = _objective(cand)
            best_candidates.append((sc, cand.copy()))
    except Exception:
        pass

    # keep only the most promising seeds
    best_candidates.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in best_candidates[:10]]
    overall_best_seq = best_candidates[0][1].copy()
    overall_best_score = best_candidates[0][0]

    print(f"[{time.time():.1f}] initial best {overall_best_score:.6f} (len={n})")

    # ------------------------------------------------------------------
    #  2️⃣  Smoothed‑max mirror descent on the top seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 40.0 and top_seeds:
        per_seed = remaining / len(top_seeds)
        for idx, seed_seq in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_max_descent_extended(
                seed_seq,
                sub_deadline,
                overall_best_seq,
                overall_best_score,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after smooth seed {idx} new best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    #  3️⃣  Exact sub‑gradient refinement (exponentiated‑gradient)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        a_opt, sc_opt = _subgradient_descent_exact(
            overall_best_seq.copy(),
            deadline,
            overall_best_seq,
            overall_best_score,
            lr0=0.7,
            eval_every=80,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(
                f"[{time.time():.1f}] after exact subgrad new best {overall_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    #  4️⃣  Final greedy pairwise polishing (lots of trials)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            overall_best_seq,
            overall_best_score,
            max_trials=1_500_000,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(
                f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    print(
        f"Final best objective (2·n·max(conv)) = {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    # The internal evaluator rescales anyway; we return the raw simplex vector.
    return overall_best_seq.tolist()
