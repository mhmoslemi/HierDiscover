"""
Improved search for a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (∑a)²

The algorithm is a hybrid of:

1. **Rich seeding** – analytic families (uniform, arcsine, Hann,
   edge‑weighted with a much wider exponent range, double‑exponential,
   Beta) are sampled for several lengths.  The 30 best seeds are kept.

2. **Smooth exponentiated‑gradient descent** – the loss “max(conv)’’ is
   approximated by a smooth‐max (log‑sum‑exp) with a temperature that
   is slowly increased.  Gradient steps are performed in the mirror
   (multiplicative) domain and the iterate is projected onto the
   probability simplex.  Both short runs on every seed and a long run
   on the current best are executed.

3. **Progressive up‑sampling** – the best vector is repeatedly doubled
   (linear interpolation) while time permits (up to 65536 entries).  After
   each up‑sample a medium‑length smooth‑gradient run refines the shape.

4. **Final annealing** – a pairwise mass‑transfer simulated annealing
   with a higher temperature budget finishes the optimisation.

All steps respect the 1000 s budget (2 s safety margin).  The returned
list is ready for ``evaluate_sequence`` (the values are non‑negative and
automatically normalised to sum = 1 inside the objective).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (full length 2n‑1)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective for a probability vector (∑a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families – each returns a *probability* vector (sum=1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{-p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_double_exponential(n: int, lam: float) -> np.ndarray:
    """Symmetric sum of two exponentials decaying from the edges."""
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-lam * (i + 0.5))
    right = np.exp(-lam * (n - i - 0.5))
    w = left + right
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density sampled on the cell mid‑points."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # avoid overflow/underflow for extreme α
    w = np.where(
        (x == 0) | (x == 1),
        0.0,
        np.exp((alpha - 1) * np.log(x) + (alpha - 1) * np.log1p(-x))
    )
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1‑λ) + uniform·λ."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smoothed max‑gradient (log‑sum‑exp) utilities
# ----------------------------------------------------------------------
def _grad_smooth_conv(a: np.ndarray, c: float) -> np.ndarray:
    """
    Gradient of the smooth‑max of the self‑convolution.
    Uses w_k ∝ exp(c·conv_k) as the softmax weights.
    Returns a dense gradient vector of length n.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)          # length 2n‑1
    max_c = conv.max()
    # soft‑max weights (numerically stable)
    w = np.exp(c * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient_i = 2·∑_k w_k·a_{k‑i}   (indices out of range give 0)
    # This is a cross‑correlation of w with a.
    a_rev = a[::-1]
    size = 1 << ((len(w) + len(a_rev) - 1 - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    full = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Adaptive exponentiated‑gradient using the (smoothed) max loss
# ----------------------------------------------------------------------
def _adaptive_exp_gradient_smooth(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    beta: float = 0.9,
    eval_every: int = 50,
    c0: float = 5.0,
    c_inc: float = 1.02,
) -> tuple[np.ndarray, float]:
    """
    Mirrors gradient descent on the smooth approximation of the objective.
    Multiplicative updates are performed in the probability simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)          # EMA of gradients
    eta = eta0
    c = c0
    it = 0
    while time.time() < deadline:
        it += 1
        grad = _grad_smooth_conv(a, c)

        # EMA of the gradient
        m = beta * m + (1.0 - beta) * grad

        # multiplicative (exponentiated) update
        a = a * np.exp(-eta * m)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if it % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                eta = min(eta0, eta * 1.02)   # reward improvement
                c = min(200.0, c * c_inc)     # sharpen the smooth‑max
                print(f"[{time.time():.1f}] smooth GD it={it:,} new best {best_score:.6f}")
            else:
                eta *= 0.995                 # gentle decay

        # occasional re‑projection to avoid numerical drift
        if it % (eval_every * 10) == 0:
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise mass‑transfer simulated annealing (slightly hotter)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        src = rng.choice(np.where(a > 1e-14)[0])
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        delta = a[src] * rng.random() * 0.3
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        cur = _objective(new_a)

        elapsed = time.time() - start
        # more aggressive linear cooling
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.08

        accept = False
        if cur < best_score - 1e-15:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal {src}->{dst} Δ={delta:.1e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling while preserving total mass (=1)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the 1000 s budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    cand_pool: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1️⃣  Rich seeding (multiple lengths)
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096, 8192, 16384]
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] seeding n={n}")

        # deterministic families
        for base in (_distribution_uniform(n),
                     _distribution_arcsine(n),
                     _distribution_hann(n)):
            sc = _objective(base)
            cand_pool.append((sc, base.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new global best (det) {sc:.6f}")

        # edge‑weighted – extend exponent range considerably
        p_vals = np.unique(np.concatenate((
            np.logspace(-3.0, -0.5, 12),          # 0.001 → 0.316
            np.logspace(-0.5, 0.0, 12),           # 0.316 → 1
            np.linspace(0.1, 3.0, 15),            # linear region
            np.logspace(0.0, 0.9, 12)             # 1 → ≈8
        )))
        lam_vals = np.linspace(0.0, 1.0, 7)
        for p in p_vals:
            if time.time() >= deadline:
                break
            base = _distribution_edge_weighted(n, float(p))
            for lam in lam_vals:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(base, float(lam))
                sc = _objective(cand)
                cand_pool.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] edge p={p:.3g} λ={lam:.2f} new best {sc:.6f}")

        # double‑exponential family
        lam_vals_de = np.logspace(-2.0, 1.0, 12)   # 0.01 → 10
        for lam in lam_vals_de:
            if time.time() >= deadline:
                break
            cand = _distribution_double_exponential(n, float(lam))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] de λ={lam:.3g} new best {sc:.6f}")

        # symmetric Beta family
        for alpha in np.concatenate((
            np.logspace(-2.0, -0.2, 8),   # heavy tails
            np.linspace(0.2, 2.0, 10),    # moderate
            np.linspace(2.0, 5.0, 6)      # central
        )):
            if time.time() >= deadline:
                break
            cand = _distribution_beta(n, float(alpha))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] beta α={alpha:.3g} new best {sc:.6f}")

        # random Dirichlet seeds
        for _ in range(10):
            if time.time() >= deadline:
                break
            log_alpha = rng.uniform(-2.0, 1.0)          # α in [0.01,10]
            alpha = 10 ** log_alpha
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] Dirichlet α={alpha:.3g} new best {sc:.6f}")

        # sparse random seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.02)
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] sparse new best {sc:.6f}")

        # keep only top‑30 seeds overall
        cand_pool.sort(key=lambda x: x[0])
        cand_pool = cand_pool[:30]

    # ------------------------------------------------------------------
    # 2️⃣  Incorporate a possibly existing construction
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.abs(prev)
            s = prev.sum()
            if s == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            else:
                prev = prev / s
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
            cand_pool.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3️⃣  Ensure we have something to return
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 2048
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Quick smooth GD on every seed (short time slice)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 12.0 and cand_pool:
        per_seed = max(0.2, remaining * 0.25 / max(len(cand_pool), 1))
        print(f"[{time.time():.1f}] quick smooth GD on {len(cand_pool)} seeds (~{per_seed:.2f}s each)")
        for idx, (sc, a0) in enumerate(cand_pool):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            a_opt, sc_opt = _adaptive_exp_gradient_smooth(
                a0.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.7,
                beta=0.9,
                eval_every=30,
                c0=5.0,
                c_inc=1.03,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after quick GD seed {idx} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Longer smooth GD on the current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_dead = time.time() + (deadline - time.time()) * 0.35
        a_opt, sc_opt = _adaptive_exp_gradient_smooth(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.5,
            beta=0.9,
            eval_every=60,
            c0=8.0,
            c_inc=1.02,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] long smooth GD improved best to {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Progressive up‑sampling (doubling while time permits)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n > 65536:
            break
        print(f"[{time.time():.1f}] up‑sampling {cur_n} → {target_n}")
        up = _upsample(overall_best_seq, target_n)
        up_sc = _objective(up)
        if up_sc < overall_best_score:
            overall_best_seq, overall_best_score = up, up_sc
            print(f"[{time.time():.1f}] up‑sampled new best {overall_best_score:.6f}")

        # medium‑length smooth GD on the up‑sampled vector
        sub_dead = time.time() + (deadline - time.time()) * 0.2
        a_opt, sc_opt = _adaptive_exp_gradient_smooth(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.45,
            beta=0.9,
            eval_every=50,
            c0=10.0,
            c_inc=1.015,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
            print(f"[{time.time():.1f}] GD after up‑sample new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣  Final simulated‑annealing polishing (hotter schedule)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing phase")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
            print(f"[{time.time():.1f}] after annealing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)) = {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    # ``evaluate_sequence`` expects a plain Python list
    return overall_best_seq.tolist()
