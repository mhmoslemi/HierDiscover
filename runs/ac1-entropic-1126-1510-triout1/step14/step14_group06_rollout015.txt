def _fft_self_convolve(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A*A, n=size)
    return conv[:2*n - 1]
```

Define cross-correlation for gradient:

def _fft_corr(a, w):
    n = a.shape[0]; m = w.shape[0]
    size = 1 << ((n+m-1)-1).bit_length()
    rev_w = w[::-1]
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_w, n=size)
    conv = np.fft.irfft(A*B, n=size)
    start = m-1
    return conv[start:start+n]
```

def _objective(a):
    n = a.shape[0]

    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a*sum_a))

def _distribution_beta_sym(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5)/n
    w = np.exp((p - 1) * np.log(x) + (p - 1)* np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0/n)
    else:
        w /= w.sum()
    return w
```

def _distribution_edge_weighted(n, p):
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0:
        w = np.full_like(w, 1.0/n)
    else:
        w /= w.sum()
    return w

def _distribution_generalized_normal(n, q):
    i = np.arange(n, dtype=np.float64) + 0.5

def _distribution_random_sparse(n, rng, density=0.01):
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)

def _grad_of_max_conv(a):
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(conv.argmax())
    grad = np.zeros_like(a)
    if k < n:
        # indices 0..k
        grad[:k+1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        # indices start..n-1
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k
```

def _exp_gradient_descent(a, deadline, best_seq, best_score, eta0=0.8, eval_every=80):
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full(n, 1.0/n)
        else:
            a = a / s
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f'[{time.time():.1f}] EG it={t} new best {best_score:.6f}')
    return best_seq, best_score

def _adam_optimize(a0, deadline, best_seq, best_score,
                   tau_start=0.2, tau_end=1e-4, tau_decay=0.998,
                   eval_every=200, step_size0=0.05):
    n = a0.shape[0]
    x = np.log(a0 + 1e-15)  # logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    tau = tau_start
    while time.time() < deadline and tau > tau_end:
        t += 1
        # softmax probabilities
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()
        # compute conv and weights
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        exp_vals = np.exp((conv - max_conv) / tau)
        w = exp_vals / exp_vals.sum()
        # gradient w.r.t a
        corr = _fft_corr(a, w)   # length n
        grad_a = 4.0 * n * corr
        # chain rule to logits
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr = step_size0 / (np.sqrt(v_hat) + eps)
        x = x - lr * m_hat
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f'[{time.time():.1f}] Adam it={t} τ={tau:.6f} new best {best_score:.6f}')
        # decay tau
        tau = max(tau * tau_decay, tau_end)
    return best_seq, best_score
```

def _pairwise_annealing(a, deadline, best_seq, best_score, rng, max_trials=None):
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed/total_time) * best_score * 0.01
        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score)/temperature)
        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f'[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}')
    return best_seq, best_score

def _upsample(a, new_n):
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= a_new.sum()
    return a_new

    Search for a non‑negative step function (probability vector) that minimizes
    the objective 2·n·max(conv)/sum².  The method proceeds in three phases:

    1.  Exhaustively generate a diverse pool of analytic seed vectors
        (uniform, arcsine, Hann, beta, edge‑weighted, generalized‑normal,
        power‑law, random Dirichlet and sparse).  Parameters are swept finely
        for each family and the best few seeds are retained.

    2.  Each retained seed is refined by a temperature‑annealed Adam optimiser
        on the smooth surrogate 2·n·τ·log‑sum‑exp(conv/τ).  The temperature is
        driven down to 1e‑4, yielding a close approximation of the true max.

    3.  The current best vector undergoes a short exponentiated‑gradient
        descent on the exact max‑convolution, followed by a pair‑wise
        annealing polish.  If time permits the vector is up‑sampled (doubling

"""
Proposes a non‑negative step‑function (probability vector) that seeks to
minimise the evaluation function ``evaluate_sequence``.  The search proceeds
through three phases:

1.  Exhaustively generate a diverse pool of analytic seeds (uniform,
    arcsine, Hann, symmetric beta, edge‑weighted, generalized‑normal,
    power‑law, Dirichlet and sparse).  Parameters are swept finely and the
    best few seeds are retained.

2.  Each retained seed is refined by an Adam optimiser on the
    temperature‑smoothed surrogate
    ``2·n·τ·log‑sum‑exp(conv/τ)``.  The temperature is driven down
    to ≈10⁻⁴, yielding a close approximation of the true maximum
    convolution.

3.  The best vector is polished by an exponentiated‑gradient descent on the
    exact objective, an optional up‑sampling (doubling the length) and a
    final pairwise‑annealing (simulated‑annealing‑style) pass.

All steps respect a total 1000 s budget and always return a plain
``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation of a (len n) with w (len m) for non‑negative shifts.
    Returns an array of length n where element i equals Σ_j a_j·w_{j+i}.
    Implemented via FFT using convolution with the reversed weight vector.
    """
    n = a.shape[0]
    m = w.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()
    rev_w = w[::-1]
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_w, n=size)
    conv = np.fft.irfft(A * B, n=size)
    start = m - 1
    return conv[start:start + n]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a (unnormalised)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic families – always return a probability vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{-p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(-|x-½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    """Monotone decreasing power‑law (p>0)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Sub‑gradient of the active convolution entry (used by EG and FW)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Compute the sub‑gradient of the currently maximal convolution entry.
    Returns (grad, active_index).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))
    grad = np.zeros_like(a)
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent (multiplicative mirror descent)
# ----------------------------------------------------------------------
def _exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """Perform exponentiated‑gradient descent on the exact objective."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Adam optimiser of the temperature‑smoothed objective
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    tau_start: float = 0.2,
    tau_end: float = 0.001,
    tau_decay: float = 0.999,
    eval_every: int = 200,
    step_size0: float = 0.05,
) -> tuple[np.ndarray, float]:
    """Adam on the smoothed objective ``2·n·τ·log‑sum‑exp(conv/τ)``."""
    n = a0.shape[0]
    x = np.log(a0 + 1e-15)               # initialise logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    tau = tau_start
    while time.time() < deadline and tau > tau_end:
        t += 1
        # softmax probabilities
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # convolution and τ‑softmax weights
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        exp_vals = np.exp((conv - max_conv) / tau)
        w = exp_vals / exp_vals.sum()

        # gradient w.r.t. a (4·n·correlation)
        corr = _fft_corr(a, w)                 # length n
        grad_a = 4.0 * n * corr

        # chain rule to logits
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = step_size0 / (np.sqrt(v_hat) + eps)
        x = x - lr * m_hat

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} τ={tau:.6f} new best {best_score:.6f}")

        # cool the temperature
        tau = max(tau * tau_decay, tau_end)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Frank‑Wolfe style refinement (pairwise mass transfer guided by the gradient)
# ----------------------------------------------------------------------
def _frank_wolfe_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """Simple Frank‑Wolfe iteration: move mass towards the coordinate with the
    smallest gradient (the least harmful direction)."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        i_min = int(np.argmin(grad))
        gamma = 2.0 / (t + 2.0)          # classic diminishing step size
        a = a * (1.0 - gamma)
        a[i_min] += gamma
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] FW it={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise‑annealing polishing (simulated annealing style)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins; accept improving
    moves or, early on, worsening moves with a temperature‑dependent probability."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling while preserving the simplex (used after a good candidate)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the total 1000 s budget and returns a plain list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Populate a diverse pool of seeds.
    # ------------------------------------------------------------------
    lengths = [2048, 4096, 8192]   # a compromise between resolution and speed
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for cand in (_distribution_uniform(n),
                     _distribution_arcsine(n),
                     _distribution_hann(n)):
            sc = _objective(cand)
            candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # symmetric beta – dense sweep
        for p in np.logspace(-0.8, 0.8, 15):
            if time.time() >= deadline: break
            beta = _distribution_beta_sym(n, p)
            sc = _objective(beta)
            candidates.append((sc, beta.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, beta.copy()
                print(f"[{time.time():.1f}] new best beta p={p:.3g} len={n} score={sc:.6f}")

        # edge‑weighted – dense sweep
        for p in np.logspace(-0.8, 0.8, 15):
            if time.time() >= deadline: break
            edge = _distribution_edge_weighted(n, p)
            sc = _objective(edge)
            candidates.append((sc, edge.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, edge.copy()
                print(f"[{time.time():.1f}] new best edge p={p:.3g} len={n} score={sc:.6f}")

        # generalized normal – varied shape
        for q in np.linspace(0.3, 3.0, 12):
            if time.time() >= deadline: break
            gn = _distribution_generalized_normal(n, q)
            sc = _objective(gn)
            candidates.append((sc, gn.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, gn.copy()
                print(f"[{time.time():.1f}] new best gn q={q:.2f} len={n} score={sc:.6f}")

        # power‑law
        for pwr in np.linspace(0.2, 2.5, 12):
            if time.time() >= deadline: break
            pl = _distribution_power_law(n, pwr)
            sc = _objective(pl)
            candidates.append((sc, pl.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, pl.copy()
                print(f"[{time.time():.1f}] new best power p={pwr:.2f} len={n} score={sc:.6f}")

        # random Dirichlet seeds
        for _ in range(6):
            if time.time() >= deadline: break
            alpha = rng.uniform(0.05, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(4):
            if time.time() >= deadline: break
            cand = _distribution_random_sparse(n, rng, density=0.005)
            sc = _objective(cand)
            candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best sparse len={n} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  Include a previously discovered construction if available.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            candidates.append((sc, prev.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Guard against empty result.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 4096
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # Keep only the very best seeds for intensive refinement.
    candidates.sort(key=lambda x: x[0])
    top_candidates = [seq for _, seq in candidates[:20]]
    print(f"[{time.time():.1f}] retained {len(top_candidates)} top seeds for refinement")

    # ------------------------------------------------------------------
    # 4.  Adam‑based smoothing optimisation on the top seeds.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        per_seed = remaining * 0.5 / max(1, len(top_candidates))
        print(f"[{time.time():.1f}] Adam smoothing on {len(top_candidates)} seeds (≈{per_seed:.2f}s each)")
        for idx, seed_seq in enumerate(top_candidates):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _adam_optimize(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                tau_start=0.4,
                tau_end=5e-5,
                tau_decay=0.998,
                eval_every=120,
                step_size0=0.06,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after Adam seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Exact exponentiated‑gradient descent on the current best.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 8.0:
        sub_deadline = time.time() + remaining
        print(f"[{time.time():.1f}] EG descent on best (budget {remaining:.1f}s)")
        best_seq, best_score = _exp_gradient_descent(
            overall_best_seq.copy(),
            sub_deadline,
            overall_best_seq,
            overall_best_score,
            eta0=0.6,
            eval_every=60,
        )
        overall_best_seq, overall_best_score = best_seq, best_score

    # ------------------------------------------------------------------
    # 6.  Optional up‑sampling and second‑round refinement.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n <= 65536:
            print(f"[{time.time():.1f}] upsampling from {cur_n} → {target_n}")
            up_seq = _upsample(overall_best_seq, target_n)
            up_score = _objective(up_seq)
            if up_score < overall_best_score:
                overall_best_seq, overall_best_score = up_seq, up_score
                print(f"[{time.time():.1f}] upsampled new best {overall_best_score:.6f}")

            # brief Adam burst on the upsampled vector
            sub_deadline = time.time() + min(15.0, deadline - time.time())
            a_opt, sc_opt = _adam_optimize(
                overall_best_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                tau_start=0.2,
                tau_end=1e-4,
                tau_decay=0.997,
                eval_every=80,
                step_size0=0.05,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
                print(f"[{time.time():.1f}] after Adam on upsample new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Final pairwise annealing polish.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing polish")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
            max_trials=None,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
            print(f"[{time.time():.1f}] after final anneal new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"[{time.time():.1f}] Final best score (objective=2·n·max(conv)/sum²): "
        f"{overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
