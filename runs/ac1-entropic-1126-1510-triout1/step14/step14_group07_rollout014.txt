"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.

Algorithm overview
------------------

1️⃣ **Broad seeding** –  Several analytically motivated families of
   probability vectors (uniform, sine‑weighted, double‑ended exponentials,
   power‑law ends, arcsine‑like, Dirichlet, sparse random) are sampled for
   a handful of lengths (up to 20 000).  The best 30 candidates (by the exact
   objective) are kept.

2️⃣ **Smooth‑max Adam refinement** –  Each retained candidate is short‑run
   optimised with Adam on a *smooth* approximation of the max‑convolution:
   ```
   smooth_max(a) = (1/β)·log∑_k exp(β·(a*a)_k)
   ```
   The gradient of this quantity is obtained in O(N log N) via FFT
   correlation (no explicit active set).  A small β (≈20) is used initially

   refined seed is kept.

   Adam run (up to the remaining budget) with a larger β (≈150) finishes
   the optimisation.

4️⃣ **Local polishing** –  A greedy pairwise mass‑transfer step and a cheap

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = _next_pow2(need)
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))  (a assumed to sum to 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    w /= w.sum()
    return w


def _distribution_exponential_center(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(i - half))
    w /= w.sum()
    return w


def _distribution_exponential_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-rate * i)
    w /= w.sum()
    return w


def _distribution_exponential_right(n: int, rate: float) -> np.ndarray:
    return _distribution_exponential_left(n, rate)[::-1]


def _distribution_double_exponential_ends(n: int, rate: float) -> np.ndarray:
    """Sum of a left‑and right‑exponential, high mass at both ends."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_powerlaw_ends(n: int, alpha: float) -> np.ndarray:
    """Power‑law weights concentrated at the two ends."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 1) ** alpha + (n - i) ** alpha
    w /= w.sum()
    return w


def _distribution_arcsine_like(n: int) -> np.ndarray:
    """Approximation to the arcsine law (1/√(x(1‑x))) on a discrete grid."""
    i = np.arange(n, dtype=np.float64)
    # shift by 0.5 to avoid division by zero at the edges
    w = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator, conc: float) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a.astype(np.float64)


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    """Pick a few random coordinates and assign them random mass."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base,  U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

    deadline : float
        Unix timestamp after which the optimisation must stop.
    max_iters : int
        Hard cap on iteration count.
    lr0 : float
        Initial learning rate (decays as 1/√t).
    beta_start, beta_end : float
        Smoothing parameter: we minimise
        f_β(a) = (1/β)·log∑_k exp(β·(a*a)_k).
        ``beta`` is increased during the run (continuation method).
    beta_inc_frac : float
        Fraction of total iterations after which ``beta`` is increased.
    beta_steps : int
        Number of increments from ``beta_start`` to ``beta_end``.
    eval_interval : int
        How often (in iterations) to evaluate the *exact* objective.
    best_seq, best_score : optional warm‑start.
    """
    n = init_a.shape[0]
    # initialise log‑weights, avoid log(0)
    x = np.log(np.maximum(init_a, 1e-12))
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    # pre‑compute FFT size needed for both convolution (2n‑1) and correlation (3n‑2)
    conv_len = 2 * n - 1
    corr_len = n + conv_len - 1          # = 3n‑2
    N = _next_pow2(corr_len)

    # schedule for beta
    beta_vals = np.linspace(beta_start, beta_end, beta_steps)

    t = 0
    if best_seq is None:
        best_seq = init_a.copy()
        best_score = _objective(init_a)

    while time.time() < deadline and t < max_iters:
        t += 1
        # optional beta continuation
        beta = beta_vals[min(t // int(beta_inc_frac * max_iters), beta_steps - 1)]

        a = _softmax(x)                     # stays on the simplex
        # FFT of a (zero‑padded to length N)
        a_fft = rfft(a, n=N)

        # linear self‑convolution a*a (full length conv_len)
        conv_full = irfft(a_fft * a_fft, n=N)[:conv_len]

        # smooth max via log‑sum‑exp
        max_c = conv_full.max()
        # stable exponentiation
        exp_scaled = np.exp(beta * (conv_full - max_c))
        sum_exp = exp_scaled.sum()
        smooth_max = max_c + (1.0 / beta) * np.log(sum_exp)

        # normalized weights over all convolution entries
        w = exp_scaled / sum_exp

        # correlation w * a (linear, length at least n)
        w_fft = rfft(w, n=N)
        corr_full = irfft(w_fft * np.conj(a_fft), n=N)[:n]
        grad_a = 2.0 * corr_full          # ∂/∂a of the *smooth* max

        # chain rule through the soft‑max parametrisation
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update with decaying step‑size
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = lr0 / math.sqrt(t)
        x -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional evaluation of the *exact* (hard‑max) objective
        if t % eval_interval == 0:
            cur_exact = _objective(a)
            if cur_exact + 1e-14 < best_score:
                best_score = cur_exact
                best_seq = a.copy()
                print(f"[{time.time():.1f}] adam it={t:,} β={beta:.1f} exact={best_score:.6f}")

    # final projection for safety
    best_seq = _project_onto_simplex(best_seq, s=1.0)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise refinement (identical to the reference implementation)
# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing style local search (identical to reference)
# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.01,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point – called by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Return a step‑function (list of heights) that the evaluator will accept."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # small safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1️⃣  Seed generation – a richer collection of families & lengths
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 15000, 20000]
    # parameters for families
    sine_ps = [0.1, 0.3, 1.0, 2.0, 5.0, 10.0]
    rates = [0.001, 0.003, 0.006, 0.012]
    alphas = [0.5, 1.0, 2.0]
    concs = [0.1, 0.5, 1.0, 2.0, 5.0]
    lam_grid = [0.0, 0.1, 0.2, 0.35]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic analytic families
        base_families = [
            _distribution_uniform(n),
            _distribution_exponential_center(n, rate=0.005),
            _distribution_exponential_left(n, rate=0.003),
            _distribution_exponential_right(n, rate=0.003),
            _distribution_arcsine_like(n),
        ]
        for base in base_families:
            sc = _objective(base)
            top_candidates.append((sc, base.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new global best (det) len={n} score={sc:.6f}")

        # sine‑weighted with mixing
        for p in sine_ps:
            if time.time() >= deadline:
                break
            sine = _distribution_sine_weighted(n, p)
            for lam in lam_grid:
                cand = _mix_with_uniform(sine, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new global best sine p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # double‑ended exponentials
        for rate in rates:
            if time.time() >= deadline:
                break
            de = _distribution_double_exponential_ends(n, rate)
            sc = _objective(de)
            top_candidates.append((sc, de.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, de.copy()
                print(f"[{time.time():.1f}] new global best double‑exp rate={rate:.4f} score={sc:.6f}")

        # power‑law ends
        for alpha in alphas:
            if time.time() >= deadline:
                break
            pl = _distribution_powerlaw_ends(n, alpha)
            sc = _objective(pl)
            top_candidates.append((sc, pl.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, pl.copy()
                print(f"[{time.time():.1f}] new global best power‑law α={alpha:.2f} score={sc:.6f}")

        # Dirichlet samples
        for conc in concs:
            for _ in range(2):
                if time.time() >= deadline:
                    break
                a = _distribution_dirichlet(n, rng, conc)
                sc = _objective(a)
                top_candidates.append((sc, a.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, a.copy()
                    print(f"[{time.time():.1f}] new global best Dirichlet conc={conc:.2g} score={sc:.6f}")

        # Sparse random
        for _ in range(2):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.004)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}")

        # keep only the very best seeds seen so far
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # Incorporate a previously found construction if available
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2️⃣  Short Adam runs on the retained seeds (smooth‑max version)
    # ------------------------------------------------------------------
    refined: list[tuple[float, np.ndarray]] = []
    for sc, cand in top_candidates:
        if time.time() >= deadline:
            break
        slice_deadline = time.time() + min(8.0, (deadline - time.time()) * 0.07)
        best_seq, best_sc = _adam_refine_smooth(
            cand.copy(),
            slice_deadline,
            max_iters=8_000,
            lr0=0.8,
            beta_start=15.0,
            beta_end=80.0,
            beta_steps=6,
            eval_interval=200,
            best_seq=cand,
            best_score=sc,
        )
        refined.append((best_sc, best_seq))

    # keep the best among the refined candidates
    if refined:
        overall_best_score, overall_best_seq = min(refined, key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 3️⃣  Long Adam descent from the overall best seed
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        long_deadline = time.time() + max(0.0, remaining - 30.0)
        print(f"[{time.time():.1f}] start long Adam descent (≈{remaining-30:.1f}s left)")
        overall_best_seq, overall_best_score = _adam_refine_smooth(
            overall_best_seq.copy(),
            long_deadline,
            max_iters=300_000,
            lr0=0.9,
            beta_start=30.0,
            beta_end=150.0,
            beta_steps=10,
            eval_interval=250,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] after long descent best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Greedy pairwise refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 5️⃣  Final simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.02,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(f"Final best score (exact) ≈ {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # Return a plain Python list – the evaluator will re‑normalise if needed.
    return overall_best_seq.tolist()
