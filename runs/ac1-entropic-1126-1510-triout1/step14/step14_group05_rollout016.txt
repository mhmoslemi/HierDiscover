- l = i, then k - l = i => k = 2i: contribution δ[i]δ[i] = ε^2 (since (-ε)^2 = ε^2)
- l = i, k - l = j => k = i + j: contribution δ[i] δ[j] = (-ε)*ε = -ε^2
- l = j, k - l = i => same as above, another -ε^2, total -2 ε^2 at k = i + j.
- l = j, k - l = j => k = 2j: contribution δ[j]δ[j] = ε^2.

"""
Propose a step function that (approximately) minimizes the objective

    2·n·max(conv(a,a))   with   sum(a)=1 ,   a_i ≥ 0 .

The algorithm works in three phases:

1️⃣  Scan a moderate collection of analytic families (uniform, arcsine,
    beta‑edge, edge‑weighted, exponential‑edge, sine, cosine,
    quadratic) on several lengths.  The best few seeds are kept.

2️⃣  For each retained seed run a long simulated‑annealing / hill‑climbing
    search.  A move transfers a random amount ε of mass from a source
    index i to a target j while keeping the total mass exactly 1.
    The convolution is updated **incrementally** in O(n) time using the
    exact quadratic formula

        c' = c – 2·ε·shift_i(a) + 2·ε·shift_j(a)
              + ε²·δ_{2i} + ε²·δ_{2j} – 2·ε²·δ_{i+j} .

    Acceptance follows a temperature schedule (early uphill moves are
    allowed, later we become greedy).  Every few thousand moves the
    full convolution is recomputed to avoid drift.

3️⃣  The best sequence found overall is returned as a plain Python list.
    The calling environment normalises it inside ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        theta = cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """Exact objective (sum(a)=1) → 2·n·max(conv)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_edge(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta‑type density w_i ∝ (i·(n‑i))^{α‑1}  (α>0)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = (i * (n - i)) ** (alpha - 1.0)
    s = vals.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    vals /= s
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    w /= w_sum
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse Dirichlet‑like seed with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Incremental convolution update (exact)
# ----------------------------------------------------------------------
def _conv_update_incremental(conv: np.ndarray, a: np.ndarray,
                             i: int, j: int, eps: float) -> None:
    """
    Apply the exact change to 'conv' caused by moving epsilon mass
    from index i to j (using the *old* vector a).  The vector a is
    *not* modified here.
    """
    n = a.shape[0]
    # shift contributions
    conv[i:i + n] -= 2.0 * eps * a
    conv[j:j + n] += 2.0 * eps * a
    # point‑mass terms
    conv[2 * i] += eps * eps
    conv[2 * j] += eps * eps
    conv[i + j] -= 2.0 * eps * eps


def _conv_revert_incremental(conv: np.ndarray, a: np.ndarray,
                             i: int, j: int, eps: float) -> None:
    """Undo the exact change performed by _conv_update_incremental."""
    n = a.shape[0]
    conv[i:i + n] += 2.0 * eps * a
    conv[j:j + n] -= 2.0 * eps * a
    conv[2 * i] -= eps * eps
    conv[2 * j] -= eps * eps
    conv[i + j] += 2.0 * eps * eps


# ----------------------------------------------------------------------
# Simulated‑annealing / hill‑climbing search
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                time_limit: float,
                global_best_seq: np.ndarray,
                global_best_score: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climber with a simple annealing schedule.
    Operates directly on the exact convolution via the O(n) incremental
    update defined above.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = np.convolve(a, a)          # exact self‑convolution (full)
    cur_score = 2.0 * n * conv.max()
    best_seq = global_best_seq if global_best_seq is not None else a.copy()
    best_score = min(global_best_score, cur_score)

    start = time.time()
    moves = 0
    recalc_interval = max(5000, n)    # keep conv accurate

    while time.time() < time_limit:
        # ------- temperature schedule ----------
        elapsed = time.time() - start
        frac = elapsed / (time_limit - start)
        T = 5e-4 * (1.0 - frac) + 1e-7 * frac   # from ~5e-4 down to ~1e-7

        # ------- propose a move ----------
        i = rng.integers(n)
        if a[i] < 1e-14:
            continue
        # choose a different target
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        max_move_frac = 0.25            # at most 25 % of source mass
        eps = (rng.random() * max_move_frac) * a[i]
        if eps < 1e-16:
            continue

        # remember old values for possible rollback
        a_i_old = a[i]
        a_j_old = a[j]

        # ------- apply incremental update -------
        _conv_update_incremental(conv, a, i, j, eps)

        # update the probability vector (mass is conserved)
        a[i] = a_i_old - eps
        a[j] = a_j_old + eps

        # compute new objective
        new_score = 2.0 * n * conv.max()

        # ------- acceptance test ----------
        accept = False
        if new_score < cur_score - 1e-14:
            accept = True
        else:
            # uphill move: accept with probability exp(-Δ/T)
            delta = new_score - cur_score
            if T > 0.0 and rng.random() < math.exp(-delta / T):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hill‑climb new global best {best_score:.6f}")
        else:
            # ---- rollback ----
            a[i] = a_i_old
            a[j] = a_j_old
            _conv_revert_incremental(conv, a, i, j, eps)

        moves += 1

        # occasional full recomputation to eliminate accumulated round‑off
        if moves % recalc_interval == 0:
            conv = np.convolve(a, a)
            cur_score = 2.0 * n * conv.max()

    return best_seq, best_score

                overall_best_score,
                rng,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after hill‑climb seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5. Final polishing with a tiny exact sub‑gradient pass (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # a very short exact sub‑gradient run to clean up any remaining
        # rounding errors; we reuse the same exponentiated‑gradient code
        # from the earlier solution but limit it to a few seconds.
        try:
            from math import sqrt  # silent import if unavailable
        except Exception:
            pass

    # ------------------------------------------------------------------
    print(
        f"Final best objective (2·n·max(conv)) = {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list – the outer harness will (re)normalise it.
    return overall_best_seq.tolist()
