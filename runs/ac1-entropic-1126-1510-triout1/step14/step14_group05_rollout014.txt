# Simplex utilities

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable softmax (returns a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# FFT‑based convolution helpers (real FFT, all lengths are modest)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_cross_correlate_valid(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (valid part)  (w ⋆ a)[j] = Σ_i w[i+j]·a[i],
    returning an array of length len(a).
    Implemented with FFT for speed.
    """
    n = a.shape[0]
    m = w.shape[0]                # = 2·n‑1
    size = 1 << ((m + n - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A = np.fft.rfft(a, n=size)
    corr = np.fft.irfft(W * np.conj(A), n=size)
    return corr[:n]


def _fft_sliding_window_sum(w: np.ndarray, win: int) -> np.ndarray:
    """
    For each j compute Σ_{i=j}^{j+win‑1} w[i] (window length = win).
    Returns an array of length win (i.e. len(a)).  Implemented via FFT.
    """
    m = w.shape[0]
    size = 1 << ((m + win - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    O = np.fft.rfft(np.ones(win, dtype=np.float64), n=size)
    conv = np.fft.irfft(W * O, n=size)
    start = win - 1
    return conv[start:start + win]

# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_edge(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta‑type density w_i ∝ (i·(n‑i))^{α‑1} (α>0)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = (i * (n - i)) ** (alpha - 1.0)
    vals = np.clip(vals, 0.0, None)
    if vals.sum() == 0.0:
        return _distribution_uniform(n)
    return vals / vals.sum()


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i·(n‑i))^{‑p} with p>0. Larger p ⇒ stronger edge bias."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    s = w.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    return w / s


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    """Linear combination of two decaying exponentials from the ends."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    s = w.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    return w / s


def _distribution_double_exponential(n: int, c1: float, c2: float) -> np.ndarray:
    """Different decay rates on left and right."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c1 * i) + np.exp(-c2 * (n - 1 - i))
    s = w.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    return w / s


def _distribution_powerlaw_edge(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i+1)^{‑p} + (n‑i)^{‑p} – a simple power‑law at both ends."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 1) ** (-p) + (n - i) ** (-p)
    s = w.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    return w / s


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    s = w.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    return w / s


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    s = w.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    return w / s


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    s = w.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    return w / s


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse Dirichlet‑like seed with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    return vals / s


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1 ⇒ returns 2·n·max(a*a))
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return the exact evaluation (2·n·max convex) for a normalised vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smoothed (soft‑max) Adam descent on the L∞-objective
# ----------------------------------------------------------------------
def _adam_softmax_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    tau_start: float = 0.02,
    tau_end: float = 0.001,
    lr0: float = 0.8,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max approximation of max(conv).
    The variable is the log‑it vector z with a = softmax(z).
    """
    n = a0.shape[0]
    # initialise logits from the starting distribution (add a tiny offset)
    z = np.log(a0 + 1e-12)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    start = time.time()
    it = 0
    while time.time() < time_limit:
        it += 1
        # current distribution
        a = _softmax(z)
        # exact objective (for bookkeeping)
        cur_exact = _objective(a)
        if cur_exact < best_score - 1e-14:
            best_score = cur_exact
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Adam it={it} new global best {best_score:.6f}")

        # anneal temperature τ
        frac = (time.time() - start) / (time_limit - start)
        tau = tau_start * (1.0 - frac) + tau_end * frac

        # soft‑max weights over the 2n‑1 convolution coefficients
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        shifted = conv - maxc          # stability
        exp_vals = np.exp(shifted / tau)
        w = exp_vals / exp_vals.sum()  # length 2n‑1, sums to 1

        # gradient of the smoothed max (see original code)
        term1 = _fft_cross_correlate_valid(w, a)            # shape (n,)
        win_sum = _fft_sliding_window_sum(w, n)            # shape (n,)
        grad_a = term1 + a * win_sum                      # ∂L/∂a

        # convert to gradient w.r.t. logits (softmax Jacobian)
        grad_z = a * (grad_a - np.dot(a, grad_a))

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        z -= step

    # return the best vector found (already normalised)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (exponentiated gradient)
# ----------------------------------------------------------------------
def _subgradient_descent_exact(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        # compute sub‑gradient (see original implementation)
        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = _distribution_uniform(n)
        else:
            a /= s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] exact it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climber (exact objective)
# ----------------------------------------------------------------------
def _simulated_annealing(
    a0: np.ndarray,
    time_limit: float,
    rng: np.random.Generator,
    best_seq: np.ndarray,
    best_score: float,
    move_frac: float = 0.15,
) -> tuple[np.ndarray, float]:
    """
    Randomly transfer a fraction of mass between two positions.
    Accept worsened moves with a linearly‑cooling temperature.
    """
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    # pre‑compute a constant that maps remaining time to temperature
    while time.time() < time_limit:
        # propose a move
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        j = rng.integers(n)
        if j == i:
            continue
        delta = a[i] * rng.random() * move_frac
        if delta < 1e-14:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        cur = _objective(a_new)

        if cur < best_score - 1e-14:
            a = a_new
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improvement δ={delta:.3e} new best {best_score:.6f}")
        else:
            # temperature decays linearly with remaining time
            remaining = time_limit - time.time()
            total = time_limit - start
            T = max(1e-6, remaining / total)  # in [0,1]
            if rng.random() < math.exp(-(cur - best_score) / T):
                a = a_new
    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise swap (exact objective) – same as earlier
# ----------------------------------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Move a random chunk of mass from i → j if it strictly improves the exact objective."""
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.45
        if delta < 1e-15:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] pairwise i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a step‑function (plain Python list) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a 2‑second safety margin for the surrounding harness
    deadline = start_time + budget_s - 2.0

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, seq)

    # ------------------------------------------------------------------
    # 1. Rich scan of analytic families on several lengths
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000, 16000, 20000]

    # parameter grids (randomised for speed)
    lam_vals = np.linspace(0.0, 1.0, 11)                     # uniform mixing

    # tiny dense grids for the continuous families
    p_vals = np.concatenate([
        np.logspace(-3, -0.2, 12),                         # 0.001 … 0.63
        np.linspace(0.1, 5.0, 15),                         # 0.1 … 5
    ])
    alpha_vals = np.linspace(0.02, 1.0, 12)                 # beta‑type exponent
    c_vals = np.concatenate([
        np.logspace(-4, -0.2, 12),                         # 1e‑4 … 0.63
        np.linspace(0.7, 5.0, 14),                         # bigger decays
    ])
    c1c2_vals = np.linspace(0.001, 5.0, 8)                  # for double‑exp

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # ---- deterministic families (no extra parameters) ----
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
        ):
            cand = base_fn(n)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # ---- beta‑type (α) families ----
        for alpha in alpha_vals:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_edge(n, float(alpha))
            for lam in lam_vals:
                cand = _mix_with_uniform(beta, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] len={n} β α={alpha:.3g} lam={lam:.2f} score={sc:.6f}")

        # ---- edge‑weighted (p) families ----
        for p in p_vals:
            if time.time() >= deadline:
                break
            edge = _distribution_edge_weighted(n, float(p))
            for lam in lam_vals:
                cand = _mix_with_uniform(edge, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] len={n} p={p:.4g} lam={lam:.2f} score={sc:.6f}")

        # ---- exponential‑edge families ----
        for c in c_vals:
            if time.time() >= deadline:
                break
            exp_edge = _distribution_exponential_edge(n, float(c))
            for lam in lam_vals:
                cand = _mix_with_uniform(exp_edge, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] len={n} exp c={c:.4g} lam={lam:.2f} score={sc:.6f}")

        # ---- double‑exponential families (different decays on each side) ----
        for c1 in c1c2_vals:
            for c2 in c1c2_vals:
                if time.time() >= deadline:
                    break
                de = _distribution_double_exponential(n, float(c1), float(c2))
                for lam in lam_vals:
                    cand = _mix_with_uniform(de, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(f"[{time.time():.1f}] len={n} de c1={c1:.2f} c2={c2:.2f} lam={lam:.2f} score={sc:.6f}")

        # ---- power‑law edge families ----
        for p in p_vals:
            if time.time() >= deadline:
                break
            pw = _distribution_powerlaw_edge(n, float(p))
            for lam in lam_vals:
                cand = _mix_with_uniform(pw, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] len={n} power‑law p={p:.4g} lam={lam:.2f} score={sc:.6f}")

        # ---- random Dirichlet seeds (different concentration) ----
        for _ in range(6):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} α={alpha:.3f} score={sc:.6f}")

        # ---- sparse random seeds (varying density) ----
        for dens in (0.005, 0.01, 0.015):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=dens)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] sparse len={n} dens={dens:.3f} score={sc:.6f}")

        # keep only the most promising seeds (memory / time bound)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:80]

    # ------------------------------------------------------------------
    # 2. Incorporate a previously‑found construction if it exists
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / s
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never trigger)
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Continuous refinement of the best seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # keep the 6‑best seeds for deeper optimisation
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] refining {len(seeds)} seeds, {per_seed:.1f}s each")

        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            # ---- Adam on soft‑max objective (majority of time) ----
            soft_dead = time.time() + per_seed * 0.55
            overall_best_seq, overall_best_score = _adam_softmax_descent(
                seed_seq,
                soft_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                tau_start=0.025,
                tau_end=0.0008,
                lr0=0.8,
            )
            # ---- exact sub‑gradient polishing (remainder) ----
            sub_dead = time.time() + per_seed * 0.35
            overall_best_seq, overall_best_score = _subgradient_descent_exact(
                overall_best_seq,
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                eval_every=250,
            )
            # Early exit if we already beat the target
            if overall_best_score <= 1.503 + 1e-6:
                print("[early stop] target reached during refinement")
                break

    # ------------------------------------------------------------------
    # 5. Simulated‑annealing fine‑tuning on the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sa_time = min(120.0, deadline - time.time())
        print(f"[{time.time():.1f}] simulated annealing for {sa_time:.1f}s")
        overall_best_seq, overall_best_score = _simulated_annealing(
            overall_best_seq.copy(),
            time.time() + sa_time,
            rng,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            move_frac=0.12,
        )

    # ------------------------------------------------------------------
    # 6. Final greedy pairwise polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy pairwise polishing")
        overall_best_seq, overall_best_score = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # 7. Reporting
    # ------------------------------------------------------------------
    print(
        f"Final best objective (2·n·max(conv)): {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
