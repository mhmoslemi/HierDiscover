def _lp_step(f: np.ndarray, M: float) -> np.ndarray:
    n = len(f)
    rows = 2*n - 1
    # Build dense matrix? Use sparse.
    from scipy.sparse import csr_matrix
    data = np.empty(n*n, dtype=np.float64)
    row_ind = np.empty(n*n, dtype=np.int32)
    col_ind = np.empty(n*n, dtype=np.int32)
    idx = 0
    for j in range(n):
        # rows j ... j+n-1
        rows_range = np.arange(j, j+n, dtype=np.int32)
        data[idx:idx+n] = f
        row_ind[idx:idx+n] = rows_range
        col_ind[idx:idx+n] = j
        idx += n
    A = csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))
    # bounds for variables >=0, no upper bound
    # c = -1s to maximize sum
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(rows, M, dtype=np.float64)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        # fallback: return f (no improvement)
        return f
    b = res.x
    # Normalize to sum 1
    s = b.sum()
    if s == 0:
        return f
    else:
        return b / s
```

        idx += n

"""
Improved 1000‑second search for a non‑negative step function minimizing

    2·n·max(conv(a,a)) / (∑a)²

Algorithm
---------

1. **Rich seeding** – families (uniform, arcsine, Hann, edge‑weighted,
   β‑distribution, generalized‑normal, sparse) are sampled for several
   lengths (512 … 4096).  The best 15 seeds are retained.

2. **Linear‑program refinement** – for a current distribution f we solve
   a linear program that maximises the total mass of a new vector b
   under the *fixed* convolution bound
        (f * b)(k) ≤ M = max(f * f) .
   The LP has simple linear constraints because f is fixed.
   After normalising b we move towards it with a step size t
   (mirrored‑gradient style).  This step is repeated while time permits.

3. **Progressive up‑sampling** – the best vector is linearly up‑sampled
   (doubling its length) and the LP refinement is applied again.
   Larger n reduces the objective because the factor 2·n grows while
   the maximal convolution typically shrinks.

4. **Final annealing** – a cheap pairwise‑mass‑transfer simulated
   annealing cleans up any remaining small irregularities.

All steps respect the 1000 s budget (2 s safety margin).  The returned
list is a plain Python list of non‑negative numbers (already normalised
to sum = 1), ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – occasional re‑projection to avoid drift
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (real‑valued, fast)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    F = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(F * F, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective for a probability vector (∑a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families – each returns a *probability* vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = 1.0 / np.sqrt(i * (n - i))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """weights ∝ ( (i+½)·(n‑i‑½) )^{-p}  (i = distance to left edge)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = n - np.arange(n, dtype=np.float64) - 0.5
    w = (i * j) ** (-p)
    w /= w.sum()
    return w


def _distribution_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) sampled on cell mid‑points."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # safe evaluation of log‑density
    logw = (alpha - 1) * (np.log(x) + np.log1p(-x))
    w = np.exp(logw, dtype=np.float64)
    w[np.isnan(w)] = 0.0
    w /= w.sum()
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(‑|x‑½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.02
) -> np.ndarray:
    """Sparse random vector with given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1‑λ) + uniform·λ."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Linear‑program step: given fixed f, compute a feasible b that maximises Σb
# under the convolution bound (f * b) ≤ M.
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float) -> np.ndarray:
    n = f.shape[0]
    rows = 2 * n - 1

    # Build the sparse constraint matrix A  (shape rows × n)
    # Row k contains coefficients f_{k‑j} for column j.
    data = np.empty(n * n, dtype=np.float64)
    row_ind = np.empty(n * n, dtype=np.int32)
    col_ind = np.empty(n * n, dtype=np.int32)

    idx = 0
    for j in range(n):
        # rows j … j+n‑1 receive f
        row_ind[idx:idx + n] = np.arange(j, j + n, dtype=np.int32)
        col_ind[idx:idx + n] = j
        data[idx:idx + n] = f
        idx += n

    A = csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))

    # Linear programme: maximise Σ b_i  →  minimise −Σ b_i
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(rows, M, dtype=np.float64)

    res = linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True},
    )

    if not res.success:
        # fallback: keep the current vector unchanged
        return f.copy()

    b = res.x
    s = b.sum()
    if s == 0.0:
        return f.copy()
    return b / s   # return a normalised probability vector


# ----------------------------------------------------------------------
# Iterative LP–based refinement (mirrored‑gradient style)
# ----------------------------------------------------------------------
def _lp_refine(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    t0: float = 0.5,
    min_step: float = 0.05,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    t = t0
    while time.time() < deadline and t > min_step:
        M = _fft_self_convolve(a).max()
        b = _lp_step(a, float(M))

        # convex combination (mirrored step)
        a_new = (1.0 - t) * a + t * b
        a_new = _project_onto_simplex(a_new, s=1.0)

        cur = _objective(a_new)
        if cur < best_score - 1e-15:
            best_score = cur
            best_seq = a_new.copy()
            print(f"[{time.time():.1f}] LP refinement it t={t:.3f} new best {best_score:.6f}")
            # a successful step -> enlarge step a little
            t = min(t0, t * 1.05)
        else:
            # no improvement -> shrink step size
            t *= 0.90

        a = a_new
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling while preserving total mass (=1)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new


# ----------------------------------------------------------------------
# Pairwise mass‑transfer simulated annealing (final polish)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        # source index with non‑negligible mass
        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        src = rng.choice(src_candidates)

        # destination index (different from src)
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # transfer a random fraction (up to 30 %)
        delta = a[src] * rng.random() * 0.3
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta
        cur = _objective(new_a)

        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.02

        accept = False
        if cur < best_score - 1e-15:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal {src}->{dst} Δ={delta:.1e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the 1000 s budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # small safety margin

    # --------------------------------------------------------------
    # 1️⃣  Diverse seed generation (multiple lengths)
    # --------------------------------------------------------------
    seed_pool: list[tuple[float, np.ndarray]] = []
    lengths = [512, 1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating seeds, n={n}")

        # Deterministic families
        for base in (
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_hann(n),
        ):
            sc = _objective(base)
            seed_pool.append((sc, base.copy()))

        # Edge‑weighted family with a modest grid of exponents
        for p in (0.1, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
            if time.time() >= deadline:
                break
            base = _distribution_edge_weighted(n, float(p))
            for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(base, float(lam))
                sc = _objective(cand)
                seed_pool.append((sc, cand.copy()))

        # Symmetric Beta family
        for alpha in (0.2, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 5.0):
            if time.time() >= deadline:
                break
            cand = _distribution_beta(n, float(alpha))
            sc = _objective(cand)
            seed_pool.append((sc, cand.copy()))

        # Generalised‑normal family
        for q in np.linspace(0.3, 4.0, 12):
            if time.time() >= deadline:
                break
            cand = _distribution_generalized_normal(n, float(q))
            sc = _objective(cand)
            seed_pool.append((sc, cand.copy()))

        # Random Dirichlet seeds
        for _ in range(5):
            if time.time() >= deadline:
                break
            log_alpha = rng.uniform(-2.0, 1.0)          # α ∈ [0.01, 10]
            alpha = 10 ** log_alpha
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            seed_pool.append((sc, cand.copy()))

        # Sparse random seeds
        for _ in range(3):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.02)
            sc = _objective(cand)
            seed_pool.append((sc, cand.copy()))

        # keep only the top‑15 seeds overall (memory bounded)
        seed_pool.sort(key=lambda x: x[0])
        seed_pool = seed_pool[:15]

    # --------------------------------------------------------------
    # 2️⃣  Initialise with the best seed (or a previous construction)
    # --------------------------------------------------------------
    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")

    if seed_pool:
        overall_best_score, overall_best_seq = seed_pool[0]
        overall_best_seq = overall_best_seq.copy()

    # Use a possibly‑existing construction from a former run
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.abs(prev)
            s = prev.sum()
            if s == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            else:
                prev = prev / s
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous construction improved best to {sc:.6f}")
    except Exception:
        pass

    if overall_best_seq is None:
        # sanity fallback – uniform of moderate size
        overall_best_seq = _distribution_uniform(2048)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score = {overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Main LP‑based polishing loop (with optional up‑sampling)
    # --------------------------------------------------------------
    while time.time() < deadline:
        # ----- 3a – LP refinement on the current best vector -----
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        sub_dead = time.time() + remaining * 0.6
        best_seq, best_score = _lp_refine(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            t0=0.5,
            min_step=0.04,
        )
        if best_score < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = best_seq.copy(), best_score
            print(f"[{time.time():.1f}] after LP refine global best {overall_best_score:.6f}")

        # ----- 3b – progressive up‑sampling when time permits -----
        cur_n = overall_best_seq.shape[0]
        if cur_n >= 65536:
            break

        # need a decent chunk of time for the next pass
        if deadline - time.time() < 30.0:
            break

        target_n = min(cur_n * 2, 65536)
        up = _upsample(overall_best_seq, target_n)
        up_sc = _objective(up)
        if up_sc < overall_best_score:
            overall_best_seq, overall_best_score = up, up_sc
            print(f"[{time.time():.1f}] up‑sampled new best {overall_best_score:.6f}")

        # polish the up‑sampled vector
        sub_dead = time.time() + (deadline - time.time()) * 0.5
        best_seq, best_score = _lp_refine(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            t0=0.45,
            min_step=0.04,
        )
        if best_score < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = best_seq.copy(), best_score
            print(f"[{time.time():.1f}] after LP on up‑sampled best {overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final simulated‑annealing polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing phase")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
            print(f"[{time.time():.1f}] after annealing new best {overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Report and return
    # --------------------------------------------------------------
    print(
        f"Final best score = {overall_best_score:.6f} (length = {len(overall_best_seq)})"
    )
    # evaluate_sequence expects a plain Python list
    return overall_best_seq.tolist()
