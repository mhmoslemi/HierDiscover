def _adam_optimize_mse(a0, deadline, best_seq, best_score,
                       eval_every=200, step_size0=0.05):
    n = a0.shape[0]
    x = np.log(a0 + 1e-15)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    t=0
    while time.time() < deadline:
        t+=1
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()
        conv = _fft_self_convolve(a)
        s = a.sum()
        avg = s*s/(2*n - 1)
        w = conv - avg
        # loss = np.sum(w*w) (maybe not needed)
        grad = 4 * _fft_corr(a, w)   # shape n
        # chain rule for logits
        avg_grad = np.dot(a, grad)
        grad_x = a * (grad - avg_grad)

        m = beta1*m + (1-beta1)*grad_x
        v = beta2*v + (1-beta2)*(grad_x*grad_x)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        lr = step_size0/(np.sqrt(v_hat)+eps)
        x = x - lr * m_hat

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] MSE ADAM it={t} new best {best_score:.6f}")
    return best_seq, best_score

The search proceeds in four stages:

4. **Fine tuning** – exponentiated‑gradient descent on the exact

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation of a (len n) with w (len m) for non‑negative shifts.
    Returns an array of length n where element i equals Σ_j a_j·w_{j+i}.
    """
    n = a.shape[0]
    m = w.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()
    rev_w = w[::-1]
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_w, n=size)
    conv = np.fft.irfft(A * B, n=size)
    start = m - 1
    return conv[start:start + n]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for an (unnormalised) vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic families – always return a *probability* vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Sparse random vector with a given density (fraction of non‑zeros)."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Gradient of L2 flattening loss (∑(conv‑avg)²)
# ----------------------------------------------------------------------
def _grad_mse(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Returns (grad, loss) for the MSE loss  L = Σ (a*a - avg)²,
    where avg = (∑a)²/(2n‑1).  Gradient w.r.t. a is 4·corr(a, conv‑avg).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    s = a.sum()
    avg = s * s / (2 * n - 1)
    w = conv - avg            # length 2n‑1
    loss = np.dot(w, w)      # scalar
    grad = 4.0 * _fft_corr(a, w)
    return grad, loss
