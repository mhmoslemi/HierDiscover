"""
Propose a step function that minimises
    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum seq)² .

The method consists of four stages:

1️⃣  **Wide analytic sweep** – generate many candidate distributions
    (uniform, arcsine, sine, cosine, quadratic, symmetric & asymmetric
    beta, edge‑weighted power‑law, exponential‑edge, mixtures with the
    uniform distribution and a few random Dirichlet/sparse draws).  The
    best few are kept.

2️⃣  **Smooth‑max mirror descent** – replace the non‑smooth max by a
    log‑sum‑exp surrogate.  An Adam optimiser with a schedule of
    smoothing parameters β (from 0.5 up to 5000) is run for many
    iterations, always projecting back onto the probability simplex.

3️⃣  **Additive sub‑gradient descent** – on the exact L∞ objective we
    perform projected sub‑gradient steps with a diminishing step‑size
    (ηₜ ∝ 1/√t).  The gradient for a given maximal convolution index k* is
    gᵢ = 2·aⱼ for all i+j = k* (j = k*−i).  After each step we project onto
    the simplex.

4️⃣  **Greedy pairwise mass transfer** – repeatedly move a random
    fraction of mass from one position to another and keep the move only
    if it strictly reduces the exact objective.  This final polishing
    often extracts the last few percent of improvement.

All stages respect the global time budget (default 1000 s).  The
returned list is the raw simplex‑normalised vector; ``evaluate_sequence``
will internally re‑scale it.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and math.isclose(v.sum(), s, rel_tol=1e-12, abs_tol=1e-12):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
#  FFT based convolutions (real inputs)
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using real FFT."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a * a using a real FFT."""
    return _fft_convolve(a, a)

# ----------------------------------------------------------------------
#  Exact objective for a simplex‑normalised vector
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
#  Analytic families (all returned already normalised)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_beta_general(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric beta‑type distribution, but α and β may differ."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    total = w.sum()
    if total == 0.0:
        return _distribution_uniform(n)
    w /= total
    return w


def _distribution_edge_weighted_asym(n: int, p: float, q: float) -> np.ndarray:
    """w_i ∝ (i+½)^(-p)·(n‑i‑½)^(-q).  Allows asymmetric edge weighting."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (-p)) * (j ** (-q))
    total = w.sum()
    if total == 0.0:
        return _distribution_uniform(n)
    w /= total
    return w


def _distribution_exponential_edge_asym(n: int, c: float, d: float) -> np.ndarray:
    """w_i ∝ exp(-c·i) + exp(-d·(n‑1‑i)).  Asymmetric exponential edge."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-d * (n - 1 - i))
    total = w.sum()
    if total == 0.0:
        return _distribution_uniform(n)
    w /= total
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _distribution_dirichlet(n: int, rng: np.random.Generator, alpha: float) -> np.ndarray:
    """Draw a Dirichlet vector with concentration α."""
    a = rng.dirichlet(np.full(n, alpha))
    return a


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.008) -> np.ndarray:
    """Sparse random vector with given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals

# ----------------------------------------------------------------------
#  Smoothed‑max mirror descent (Adam with β‑continuation)
# ----------------------------------------------------------------------
def _smooth_max_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    # β schedule – geometric growth, ending at 5000
    beta_vals = np.geomspace(0.5, 5000.0, num=12)

    for beta in beta_vals:
        # bail out early if we cannot finish even a few iterations
        if time.time() >= deadline - 1.0:
            break
        max_iters = 3500
        while t < max_iters and time.time() < deadline:
            t += 1

            # convolution and smoothed‑max weights
            conv = _fft_self_convolve(a)
            max_val = conv.max()
            shifted = conv - max_val          # ≤ 0
            # safe exponentiation
            exp_beta = np.exp(beta * shifted)
            w = exp_beta / exp_beta.sum()     # Σ w = 1

            # gradient of the smooth max (2 * (w * a_rev) convolved)
            a_rev = a[::-1]
            grad_conv = _fft_convolve(w, a_rev)          # length 3n‑2
            grad = 2.0 * grad_conv[n - 1 : n - 1 + n]    # size n

            # gradient of the full objective J = 2·n·smooth_max
            grad_J = 2.0 * n * grad

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_J
            v = beta2 * v + (1.0 - beta2) * (grad_J ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = 0.14 / (np.sqrt(v_hat) + eps)   # a modest step size
            a -= step * m_hat

            # projection onto simplex
            a = _project_onto_simplex(a, s=1.0)

            # occasional exact evaluation
            if t % 200 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-14:
                    best_score, best_seq = cur, a.copy()
                    print(
                        f"[{time.time():.1f}] β={beta:.1f} it={t} new best {best_score:.6f}"
                    )
        # end while over iterations for current β
        if time.time() >= deadline - 1.0:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Additive sub‑gradient descent on the exact L∞ objective
# ----------------------------------------------------------------------
def _subgradient_descent_additive(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < deadline - 0.5:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        idx = int(np.argmax(conv))

        # exact sub‑gradient for the chosen maximal index
        grad = np.zeros_like(a)
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_range = idx - i_range
        grad[i_range] = 2.0 * a[j_range]

        # step size decreasing like 1/√t
        eta = eta0 / math.sqrt(t)
        a_new = a - eta * (2.0 * n * grad)   # gradient of 2·n·max(conv)
        # projection onto simplex (maintains non‑negativity)
        a = _project_onto_simplex(np.maximum(a_new, 0.0), s=1.0)

        if t % 150 == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score, best_seq = cur, a.copy()
                print(
                    f"[{time.time():.1f}] subgrad it={t} new global best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Greedy pairwise mass transfer
# ----------------------------------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move a random fraction (up to 45 %) from i to j
        delta = a[i] * rng.random() * 0.45
        if delta < 1e-15:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score, best_seq = cur, a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` within the given time budget (seconds).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0   # tiny safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, vector)

    # ------------------------------------------------------------------
    # 1️⃣ Analytic families sweep (≈ 30 % of the budget)
    # ------------------------------------------------------------------
    lengths = [1200, 2000, 4000, 8000, 12000, 16000]
    # parameter grids
    p_grid = np.geomspace(0.05, 5.0, num=12)
    q_grid = np.geomspace(0.05, 5.0, num=12)
    lam_grid = np.linspace(0.0, 1.0, 9)
    alpha_grid = np.geomspace(0.2, 3.0, num=10)
    beta_grid = np.geomspace(0.2, 3.0, num=10)
    c_grid = np.geomspace(0.01, 2.0, num=10)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning analytic families, n={n}")

        # deterministic families
        for func in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
        ):
            cand = func(n)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (det) n={n} score={sc:.6f}")

        # asymmetric beta families (α ≠ β)
        for al in alpha_grid:
            for be in beta_grid:
                cand = _distribution_beta_general(n, float(al), float(be))
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}] beta α={al:.3g} β={be:.3g} n={n} score={sc:.6f}"
                    )
                # early‑exit if we are spending too much time
                if time.time() >= deadline:
                    break
            if time.time() >= deadline:
                break

        # edge‑weighted asymmetric families + uniform mixes
        for p in p_grid:
            for q in q_grid:
                base = _distribution_edge_weighted_asym(n, float(p), float(q))
                for lam in lam_grid:
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] edge p={p:.3g} q={q:.3g} lam={lam:.2f} n={n} score={sc:.6f}"
                        )
                if time.time() >= deadline:
                    break
            if time.time() >= deadline:
                break

        # exponential‑edge asymmetric families + uniform mixes
        for c in c_grid:
            for d in c_grid:
                base = _distribution_exponential_edge_asym(n, float(c), float(d))
                for lam in lam_grid:
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] exp c={c:.3g} d={d:.3g} lam={lam:.2f} n={n} score={sc:.6f}"
                        )
                if time.time() >= deadline:
                    break
            if time.time() >= deadline:
                break

        # random Dirichlet draws
        for _ in range(5):
            conc = rng.uniform(0.01, 2.5)
            cand = _distribution_dirichlet(n, rng, conc)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] Dirichlet conc={conc:.3g} n={n} score={sc:.6f}")
            if time.time() >= deadline:
                break

        # random sparse seeds
        for _ in range(3):
            cand = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] sparse n={n} score={sc:.6f}")
            if time.time() >= deadline:
                break

        # keep only the most promising 40 candidates (memory & speed)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:40]

    # ------------------------------------------------------------------
    # 2️⃣  Use the previously found construction if present
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(
                    f"[{time.time():.1f}] imported previous sequence, score={sc:.6f}"
                )
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3️⃣  Fallback (should never happen)
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(
            f"[{time.time():.1f}] fallback uniform, n={n}, score={overall_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 4️⃣  Smoothed‑max mirror descent on the best seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 40.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]]
        per_seed = remaining / max(1, len(seeds))
        print(
            f"[{time.time():.1f}] starting smoothed‑max descent on {len(seeds)} seeds"
        )
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_max_descent(
                seed_seq,
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after smooth seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5️⃣  Additive sub‑gradient descent (exact objective)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        print(f"[{time.time():.1f}] exact sub‑gradient descent (additive)")
        a_opt, sc_opt = _subgradient_descent_additive(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.9,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(
                f"[{time.time():.1f}] after additive subgrad new best {overall_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 6️⃣  Final greedy pairwise polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy pairwise polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(
                f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    print(
        f"Finished – best objective 2·n·max(conv) = {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return the raw simplex‑normalised list; ``evaluate_sequence`` will
    # rescale internally.
    return overall_best_seq.tolist()
