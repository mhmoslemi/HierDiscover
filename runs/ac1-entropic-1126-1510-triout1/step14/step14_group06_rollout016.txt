"""
Improved search for a non‑negative step‑function (probability vector) that
minimises ``evaluate_sequence``.

The algorithm builds a pool of analytic “seed’’ distributions, keeps a few
best candidates and then performs a multi‑scale optimisation:

1.  Adam optimisation of a temperature‑smoothed version of the true objective.
2.  Exponentiated‑gradient descent on the exact (non‑smooth) objective.
3.  Gradient‑guided pairwise mass transfers (move mass from the most harmful
    to the least harmful index).
4.  Repeated up‑sampling (doubling the length) followed by a short Adam burst.
5.  Final pairwise‑annealing polish.

All steps respect the global 1000 s budget and always return a plain
``list[float]`` suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation of a (len n) with w (len m) for non‑negative shifts.
    Returns an array of length n where element i equals Σ_j a_j·w_{j+i}.
    Implemented via FFT using convolution with the reversed weight vector.
    """
    n = a.shape[0]
    m = w.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()
    rev_w = w[::-1]
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_w, n=size)
    conv = np.fft.irfft(A * B, n=size)
    start = m - 1
    return conv[start:start + n]


# ----------------------------------------------------------------------
# Exact objective (identical to evaluate_sequence, but without the safety checks)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return the exact value of ``evaluate_sequence`` for a (unnormalised)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic families – they all return a *probability* vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{-p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(-|x-½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    """Monotone decreasing power‑law (p>0)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Sub‑gradient of the entry of the convolution that attains the maximum
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Compute the sub‑gradient of the currently maximal convolution entry.
    Returns (grad, active_index).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))
    grad = np.zeros_like(a)
    # The contribution of a_i to conv_k is a_{k-i} (if inside range)
    # Mirrored term gives factor 2 except when i == k-i
    if k < n:
        # indices i = 0..k
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        # indices i = k-(n-1) .. n-1
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
# Exponentiated‑gradient (multiplicative mirror) descent on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """Perform exponentiated‑gradient descent on the exact objective."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Frank‑Wolfe style refinement (mass transfer towards the most promising index)
# ----------------------------------------------------------------------
def _frank_wolfe_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """Simple Frank‑Wolfe iteration: move mass towards the coordinate with the
    smallest gradient (the least harmful direction)."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        i_min = int(np.argmin(grad))
        gamma = 2.0 / (t + 2.0)          # classic diminishing step size
        a = a * (1.0 - gamma)
        a[i_min] += gamma
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] FW it={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise‑annealing polishing (simulated‑annealing style)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins; accept improving
    moves or, early on, worsening moves with a temperature‑dependent probability."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling while preserving the simplex
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Adam optimisation of a temperature‑smoothed version of the objective
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    tau_start: float = 0.2,
    tau_end: float = 0.001,
    tau_decay: float = 0.999,
    eval_every: int = 200,
    step_size0: float = 0.05,
) -> tuple[np.ndarray, float]:
    """Adam on the smoothed objective ``2·n·τ·log‑sum‑exp(conv/τ)``."""
    n = a0.shape[0]
    x = np.log(a0 + 1e-15)               # initialise logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    tau = tau_start
    while time.time() < deadline and tau > tau_end:
        t += 1
        # softmax probabilities
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # convolution and τ‑softmax weights
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        exp_vals = np.exp((conv - max_conv) / tau)
        w = exp_vals / exp_vals.sum()

        # gradient w.r.t. a (4·n·correlation)
        corr = _fft_corr(a, w)                 # length n
        grad_a = 4.0 * n * corr

        # chain rule to logits
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = step_size0 / (np.sqrt(v_hat) + eps)
        x = x - lr * m_hat

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} τ={tau:.6f} new best {best_score:.6f}")

        # cool the temperature
        tau = max(tau * tau_decay, tau_end)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Gradient‑guided pairwise mass transfers (deterministic version of annealing)
# ----------------------------------------------------------------------
def _gradient_pairwise_opt(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly move mass from the index that most *increases* the current maximal
    convolution entry to the index that most *decreases* it, accepting any
    improvement.
    """
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        grad, _ = _grad_of_max_conv(a)
        i_max = int(np.argmax(grad))
        i_min = int(np.argmin(grad))

        if a[i_max] <= 1e-12:
            break          # nothing left to move

        # try a few fractions of the available mass
        improved = False
        for frac in (0.5, 0.3, 0.1, 0.05):
            delta = a[i_max] * frac
            if delta < 1e-15:
                continue
            new_a = a.copy()
            new_a[i_max] -= delta
            new_a[i_min] += delta
            cur = _objective(new_a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = new_a.copy()
                a = new_a            # continue from the improved point
                improved = True
                print(f"[{time.time():.1f}] grad‑move {i_max}->{i_min} δ={delta:.3e} new best {best_score:.6f}")
                break
        if not improved:
            # occasionally try a random move to escape tiny plateaus
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            delta = a[i] * rng.random() * 0.2
            if delta < 1e-15:
                continue
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta
            cur = _objective(new_a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = new_a.copy()
                a = new_a
                print(f"[{time.time():.1f}] random‑move {i}->{j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the total 1000 s budget and returns a plain ``list[float]``.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Populate a diverse pool of seeds (short lengths for speed)
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048]
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n), _distribution_hann(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # beta and edge‑weighted families
        for p in np.logspace(-0.5, 0.5, 6):
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            for base in (beta, edge):
                for lam in (0.0, 0.3, 0.6, 1.0):
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(f"[{time.time():.1f}] new best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # generalized normal shapes
        for q in (0.5, 1.0, 1.5, 2.0):
            if time.time() >= deadline:
                break
            cand = _distribution_generalized_normal(n, q)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best gn q={q:.2f} len={n} score={sc:.6f}")

        # power‑law families
        for pwr in (0.3, 0.6, 1.0, 1.5, 2.0):
            if time.time() >= deadline:
                break
            cand = _distribution_power_law(n, pwr)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best powerlaw p={pwr:.2f} len={n} score={sc:.6f}")

        # random Dirichlet seeds
        for _ in range(3):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(3):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.01)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best sparse len={n} score={sc:.6f}")

        # keep only the very best few for later refinement
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:10]

    # ------------------------------------------------------------------
    # 2.  Include a previously discovered construction if available.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        if prev.sum() > 0.0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            top_candidates.append((sc, prev.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Guard against the unlikely case of an empty result.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 1024
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Adam‑based smoothing optimisation on the top seeds.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seed_seqs = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining * 0.4 / max(1, len(seed_seqs))
        print(f"[{time.time():.1f}] Adam smoothing on {len(seed_seqs)} seeds")
        for idx, seed_seq in enumerate(seed_seqs):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _adam_optimize(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                tau_start=0.3,
                tau_end=0.0005,
                tau_decay=0.998,
                eval_every=150,
                step_size0=0.07,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] after Adam seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Multi‑scale optimisation loop (up‑sample, optimise, repeat)
    # ------------------------------------------------------------------
    max_len = 65536
    while time.time() < deadline and overall_best_seq.shape[0] < max_len:
        # give the current length a few dedicated optimisation steps
        cur_len = overall_best_seq.shape[0]
        print(f"[{time.time():.1f}] multi‑scale stage, length={cur_len}")

        # a) short Adam burst
        sub_deadline = time.time() + min(12.0, (deadline - time.time()) * 0.2)
        overall_best_seq, overall_best_score = _adam_optimize(
            overall_best_seq.copy(),
            sub_deadline,
            overall_best_seq,
            overall_best_score,
            tau_start=0.2,
            tau_end=0.0008,
            tau_decay=0.999,
            eval_every=120,
            step_size0=0.06,
        )

        # b) exponentiated‑gradient descent
        sub_deadline = time.time() + min(8.0, (deadline - time.time()) * 0.15)
        overall_best_seq, overall_best_score = _exp_gradient_descent(
            overall_best_seq.copy(),
            sub_deadline,
            overall_best_seq,
            overall_best_score,
            eta0=0.7,
            eval_every=60,
        )

        # c) deterministic gradient‑guided pairwise transfers
        sub_deadline = time.time() + min(6.0, (deadline - time.time()) * 0.1)
        overall_best_seq, overall_best_score = _gradient_pairwise_opt(
            overall_best_seq.copy(),
            sub_deadline,
            overall_best_seq,
            overall_best_score,
            rng,
            max_trials=None,
        )

        # d) up‑sample (if we still have budget)
        new_len = cur_len * 2
        if new_len > max_len:
            break
        overall_best_seq = _upsample(overall_best_seq, new_len)
        overall_best_seq = overall_best_seq / overall_best_seq.sum()
        cur = _objective(overall_best_seq)
        if cur < overall_best_score:
            overall_best_score = cur
            print(f"[{time.time():.1f}] after upsample len={new_len} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Final polishing: Frank‑Wolfe + annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fw_deadline = time.time() + (deadline - time.time()) * 0.15
        overall_best_seq, overall_best_score = _frank_wolfe_descent(
            overall_best_seq.copy(),
            fw_deadline,
            overall_best_seq,
            overall_best_score,
            eval_every=80,
        )
        print(f"[{time.time():.1f}] after FW best score {overall_best_score:.6f}")

    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing polish")
        overall_best_seq, overall_best_score = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            overall_best_seq,
            overall_best_score,
            rng=rng,
            max_trials=None,
        )
        print(f"[{time.time():.1f}] after final anneal best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    final_seq = overall_best_seq.tolist()
    print(
        f"[{time.time():.1f}] Final best score (objective=2·n·max(conv)/sum²): "
        f"{overall_best_score:.6f} (length={len(final_seq)})"
    )
    return final_seq
