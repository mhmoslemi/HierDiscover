"""
Propose a non‑negative step function (probability vector) that minimizes

    2·n·max(conv(a, a)) / (∑ a)²

The algorithm consists of three stages:

1️⃣  **Diverse seeding** – a large collection of analytic families
    (uniform, arcsine, Hann, triangular, edge‑weighted,
    β‑distribution, generalized‑normal, random Fourier, sparse Dirichlet)
    is sampled at several lengths.  The 15‑best seeds are kept.

2️⃣  **Smooth exponentiated‑gradient polish** – each seed is refined with a
    mirror‑descent update that uses a *soft‑max* (log‑sum‑exp) approximation
    of the max‑convolution.  The gradient is obtained by an FFT‑based
    convolution, the step‑size and the smoothing parameter ε are
    adapted automatically, and the iterate is projected onto the simplex
    after each epoch.

3️⃣  **Progressive up‑sampling + annealing** – the best vector is
    repeatedly up‑sampled (linear interpolation) while time permits.
    After each up‑sample a short gradient run is performed.
    Finally a simple pairwise‑mass‑transfer simulated annealing finishes
    the optimisation.

All operations respect the 1000 s budget (2 s safety margin).  The
returned list is ready for ``evaluate_sequence`` (non‑negative, sum>0).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helper (Euclidean projection onto the probability simplex)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based self‑convolution (full length = 2·n‑1)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective for a probability vector (∑a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families – each returns a *probability* vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_triangle(n: int) -> np.ndarray:
    # symmetric triangle, peak at the centre
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = 1.0 - np.abs(i - centre) / centre
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{-p} where i,j are distances to the two edges (1‑based)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w /= w.sum()
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(‑|x‑½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w /= w.sum()
    return w


def _distribution_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) sampled on cell mid‑points."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # avoid overflow/underflow for extreme α
    w = np.where(
        (x == 0) | (x == 1),
        0.0,
        np.exp((alpha - 1) * np.log(x) + (alpha - 1) * np.log1p(-x))
    )
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.02) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_random_dirichlet(n: int, rng: np.random.Generator, conc: float = 1.0) -> np.ndarray:
    """Draw a Dirichlet vector with concentration parameter `conc`."""
    alpha = np.full(n, conc)
    vec = rng.dirichlet(alpha)
    if vec.sum() == 0.0:
        vec = _distribution_uniform(n)
    return vec


def _distribution_random_fourier(n: int, rng: np.random.Generator, modes: int = 5) -> np.ndarray:
    """Sum of a few random sinusoidal modes, shifted to be non‑negative."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    coeff = rng.normal(size=modes)
    y = np.zeros_like(x)
    for k in range(1, modes + 1):
        y += coeff[k - 1] * np.sin(2.0 * np.pi * k * x)
    # make it non‑negative
    y -= y.min()
    s = y.sum()
    if s == 0.0:
        y = np.full_like(y, 1.0 / n)
    else:
        y /= s
    return y


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1‑λ) + uniform·λ."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) gradient of max‑convolution
# ----------------------------------------------------------------------
def _smooth_grad(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smoothed objective
        F_eps(a) = eps * log( Σ_k exp(conv_k / eps) )
    where conv = a * a (full convolution).  Returns a vector of length n.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)                     # length 2n‑1
    max_c = conv.max()
    # stable soft‑max weights
    w = np.exp((conv - max_c) / eps)
    W = w.sum()
    # reverse a for the cross‑correlation needed in the gradient
    a_rev = a[::-1]

    size = 1 << ((len(w) + len(a_rev) - 1 - 1).bit_length())
    W_fft = np.fft.rfft(w, n=size)
    A_rev_fft = np.fft.rfft(a_rev, n=size)
    conv_wa_rev = np.fft.irfft(W_fft * A_rev_fft, n=size)
    conv_wa_rev = conv_wa_rev[: len(w) + len(a_rev) - 1]      # keep needed part

    start = n - 1
    grad = 2.0 * conv_wa_rev[start:start + n] / W
    return grad


# ----------------------------------------------------------------------
# Adaptive exponentiated‑gradient descent using the smooth gradient
# ----------------------------------------------------------------------
def _adaptive_exp_smooth(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    beta: float = 0.9,
    eps0: float = 0.2,
    eps_decay: float = 0.999,
    eval_every: int = 30,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smoothed objective.
    Uses EMA of the gradient and an adaptive stepsize η.
    """
    a = a0.copy()
    m = np.zeros_like(a)          # EMA of the gradient
    eta = eta0
    eps = eps0
    t = 0
    while time.time() < deadline:
        t += 1
        grad = _smooth_grad(a, eps)
        m = beta * m + (1.0 - beta) * grad
        a = a * np.exp(-eta * m)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / len(a))
        else:
            a /= s
        # gradually shrink smoothing parameter
        eps = max(eps * eps_decay, 0.005)

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                eta = min(eta0, eta * 1.02)          # reward improvement
                print(f"[{time.time():.1f}] smooth GD it={t:,} new best {best_score:.6f}")
            else:
                eta *= 0.995                         # slight decay

        if t % (eval_every * 10) == 0:
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score

# Linear up‑sampling while preserving total mass (=1)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new


# ----------------------------------------------------------------------
# Pairwise mass‑transfer simulated annealing (final polishing)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        # choose a source with non‑negligible mass
        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        src = rng.choice(src_candidates)
        # choose a different destination uniformly
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # transfer a random fraction of src's mass (up to 30 %)
        delta = a[src] * rng.random() * 0.3
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        cur = _objective(new_a)

        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.02

        accept = False
        if cur < best_score - 1e-15:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal {src}->{dst} Δ={delta:.1e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the 1000 s budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # small safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    cand_pool: list[tuple[float, np.ndarray]] = []

    # --------------------------------------------------------------
    # 1️⃣  Seeding – generate a rich set of analytic / random vectors
    # --------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384]
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] seeding length={n}")

        # deterministic families
        for base in (
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_hann(n),
            _distribution_triangle(n),
        ):
            sc = _objective(base)
            cand_pool.append((sc, base.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new global best (det) {sc:.6f}")

        # edge‑weighted family (dense grid)
        p_vals = np.concatenate((
            np.logspace(-3.0, -0.5, 8, base=10),          # 0.001 → 0.316
            np.logspace(-0.5, 0.0, 8, base=10),           # 0.316 → 1
            np.linspace(0.2, 2.0, 10),                    # linear region
            np.logspace(0.0, 0.7, 6, base=10)            # 1 → 5
        ))
        lam_vals = np.linspace(0.0, 1.0, 5)
        for p in p_vals:
            if time.time() >= deadline:
                break
            base = _distribution_edge_weighted(n, float(p))
            for lam in lam_vals:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(base, float(lam))
                sc = _objective(cand)
                cand_pool.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] edge p={p:.3g} λ={lam:.2f} new best {sc:.6f}")

        # symmetric Beta family
        for alpha in np.concatenate((
            np.logspace(-2.0, -0.2, 8, base=10),   # heavy tails
            np.linspace(0.2, 2.0, 8),               # moderate
            np.linspace(2.0, 5.0, 5)               # central
        )):
            if time.time() >= deadline:
                break
            cand = _distribution_beta(n, float(alpha))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] beta α={alpha:.3g} new best {sc:.6f}")

        # generalized normal family
        for q in np.linspace(0.3, 4.0, 12):
            if time.time() >= deadline:
                break
            cand = _distribution_generalized_normal(n, float(q))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] gn q={q:.2f} new best {sc:.6f}")

        # random Fourier‑style seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_random_fourier(n, rng, modes=7)
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] rand fourier new best {sc:.6f}")

        # Dirichlet seeds (different concentration)
        for _ in range(8):
            if time.time() >= deadline:
                break
            conc = 10 ** rng.uniform(-2.0, 1.0)   # α ∈ [0.01, 10]
            cand = _distribution_random_dirichlet(n, rng, conc=conc)
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] Dirichlet α={conc:.3g} new best {sc:.6f}")

        # sparse random seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.015)
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] sparse new best {sc:.6f}")

        # keep only the best 15 seeds overall (memory bounded)
        cand_pool.sort(key=lambda x: x[0])
        cand_pool = cand_pool[:15]

    # --------------------------------------------------------------
    # 2️⃣  Try to incorporate a previously found construction (if any)
    # --------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.abs(prev)                 # safety: non‑negative
            s = prev.sum()
            if s == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            else:
                prev = prev / s
            sc = _objective(prev)
            cand_pool.append((sc, prev.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous construction improved to {sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Ensure we have something to return
    # --------------------------------------------------------------
    if overall_best_seq is None:
        n = 2048
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Quick smooth GD on each seed (short time slice)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 12.0 and cand_pool:
        per_seed = max(0.3, remaining * 0.25 / max(len(cand_pool), 1))
        print(f"[{time.time():.1f}] quick smooth GD on {len(cand_pool)} seeds (~{per_seed:.2f}s each)")
        for idx, (sc, a0) in enumerate(cand_pool):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            a_opt, sc_opt = _adaptive_exp_smooth(
                a0.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.7,
                beta=0.9,
                eps0=0.25,
                eps_decay=0.998,
                eval_every=25,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after quick GD seed {idx} new best {overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Longer smooth GD on the current global best
    # --------------------------------------------------------------
    if time.time() < deadline:
        sub_dead = time.time() + (deadline - time.time()) * 0.35
        a_opt, sc_opt = _adaptive_exp_smooth(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.5,
            beta=0.9,
            eps0=0.2,
            eps_decay=0.999,
            eval_every=50,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] long GD improved best to {overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Progressive up‑sampling (doubling while beneficial)
    # --------------------------------------------------------------
    no_improve_up = 0
    while time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        if cur_n >= 65536:
            break
        target_n = cur_n * 2
        print(f"[{time.time():.1f}] attempting up‑sample {cur_n} → {target_n}")
        up = _upsample(overall_best_seq, target_n)
        up_sc = _objective(up)
        improved = False
        if up_sc < overall_best_score:
            overall_best_seq, overall_best_score = up, up_sc
            print(f"[{time.time():.1f}] up‑sampled best now {overall_best_score:.6f}")
            improved = True
            no_improve_up = 0
        else:
            no_improve_up += 1
        # run a modest smooth‑GD on the up‑sampled vector (or on the original if not improved)
        base_vec = up if improved else overall_best_seq
        sub_dead = time.time() + (deadline - time.time()) * 0.15
        a_opt, sc_opt = _adaptive_exp_smooth(
            base_vec.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.45,
            beta=0.9,
            eps0=0.18,
            eps_decay=0.999,
            eval_every=40,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
            print(f"[{time.time():.1f}] GD after up‑sample new best {overall_best_score:.6f}")
            no_improve_up = 0
        # stop up‑sampling if we fail to improve after a few attempts
        if no_improve_up >= 3:
            print(f"[{time.time():.1f}] no improvement after {no_improve_up} successive up‑samples – stop up‑sampling")
            break

    # --------------------------------------------------------------
    # 6️⃣  Final annealing polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing phase (remaining {deadline - time.time():.1f}s)")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
            print(f"[{time.time():.1f}] after annealing new best {overall_best_score:.6f}")

    # --------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)) = {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
