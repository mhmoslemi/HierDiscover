"""
Propose a step function that (approximately) minimizes

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

because the objective is scale‑invariant we keep the sum normalized to 1.
The algorithm is a three‑phase search:

1️⃣  Exhaustive scan of several analytic families (uniform, arcsine,
    sine, cosine, quadratic, beta‑edge, edge‑weighted, exponential‑edge)
    on a handful of lengths.  For every family a fine grid of parameters
    and a dense mixing grid with the uniform distribution is tried.
    The top 120 candidates (by exact objective) are retained.

2️⃣  Each retained candidate is refined by a two‑stage gradient method:
    – a soft‑max (smoothed max) gradient descent with an aggressively
      annealed temperature (τ from 2·10⁻² down to 5·10⁻⁵);
    – an exact sub‑gradient descent on the true L∞ objective.
    The best result of the refinement becomes the global incumbent.

3️⃣  Starting from the best incumbent we perform a simulated‑annealing
    hill‑climber that accepts occasional uphill moves.  Finally a greedy
    pairwise‑swap optimiser (exact evaluation) polishes the solution.

All stages respect a global 1000 s budget.  The function returns a plain
list of non‑negative floats; the caller will re‑normalise it inside
``evaluate_sequence``.
"""

# Simple utilities

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_cross_correlate_valid(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (valid part):
    (w ⋆ a)[j] = Σ_i w[i+j]·a[i], length = len(a).
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2·n‑1
    size = 1 << ((m + n - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A = np.fft.rfft(a, n=size)
    corr = np.fft.irfft(W * np.conj(A), n=size)
    return corr[:n]


def _fft_sliding_window_sum(w: np.ndarray, win: int) -> np.ndarray:
    """
    Compute Σ_{i=j}^{j+win‑1} w[i] for each j (length = win).
    Implemented via FFT.
    """
    m = w.shape[0]
    size = 1 << ((m + win - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    ones = np.fft.rfft(np.ones(win, dtype=np.float64), n=size)
    conv = np.fft.irfft(W * ones, n=size)
    start = win - 1
    return conv[start:start + win]


# ----------------------------------------------------------------------
# Analytic families  (each returns a probability vector, Σ=1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_edge(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta‑type density  w_i ∝ (i·(n‑i))^{α‑1}  (α>0)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = (i * (n - i)) ** (alpha - 1.0)
    vals = np.clip(vals, 0.0, None)
    s = vals.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    vals /= s
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i·(n‑i))^{‑p}  (p can be negative)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)                 # works for negative p as well
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    w /= w_sum
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse Dirichlet‑like seed with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Exact objective  (input already sums to 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a).  a is assumed non‑negative and Σa=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth (soft‑max) gradient descent
# ----------------------------------------------------------------------
def _smooth_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    tau_start: float = 0.02,
    tau_end: float = 5e-5,
    lr0: float = 0.6,
    eval_every: int = 200,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the soft‑max surrogate of max."""
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    it = 0
    while time.time() < time_limit:
        it += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] smooth it={it} new best {best_score:.6f}")

        # anneal τ linearly
        frac = (time.time() - start) / (time_limit - start)
        tau = tau_start * (1.0 - frac) + tau_end * frac

        # soft‑max weights
        shifted = conv - max_val
        exp_vals = np.exp(shifted / tau)          # ≤ 1
        w = exp_vals / exp_vals.sum()             # length 2n‑1, sums to 1

        # gradient of the smooth objective
        term1 = _fft_cross_correlate_valid(w, a)   # Σ w[i+j]·a[i]
        win_sum = _fft_sliding_window_sum(w, n)   # Σ_{k=j}^{j+n‑1} w[k]
        grad = term1 + a * win_sum

        # EG update
        eta = lr0 / math.sqrt(it)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if it % eval_every == 0:
            pass
    return best_seq, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (true L∞ objective)
# ----------------------------------------------------------------------
def _subgradient_descent_exact(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the exact max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] exact it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise hill‑climber (exact objective)
# ----------------------------------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")
    return best_seq, best_score

    # 5. Simulated‑annealing hill‑climber starting from the incumbent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] simulated annealing from incumbent")
        cur_seq = overall_best_seq.copy()
        cur_score = overall_best_score
        best_seq = cur_seq.copy()
        best_score = cur_score

        # initial temperature – a modest fraction of the current objective
        T_start = max(1e-4, (cur_score - 1.2) * 0.2)
        T_end = 1e-7
        while time.time() < deadline:
            elapsed = time.time() - start_time
            frac = elapsed / (deadline - start_time)
            T = T_start * (1.0 - frac) + T_end * frac

            src_idx = np.where(cur_seq > 1e-12)[0]
            if src_idx.size == 0:
                break
            i = rng.choice(src_idx)
            j = rng.integers(len(cur_seq))
            if j == i:
                continue
            delta = cur_seq[i] * rng.random() * 0.5   # up to 50 % of source mass
            new_seq = cur_seq.copy()
            new_seq[i] -= delta
            new_seq[j] += delta

            new_score = _objective(new_seq)
            accept = False
            if new_score < cur_score:
                accept = True
            else:
                prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
                if rng.random() < prob:
                    accept = True
            if accept:
                cur_seq, cur_score = new_seq, new_score
                if cur_score < best_score - 1e-14:
                    best_score, best_seq = cur_score, cur_seq.copy()
                    print(f"[{time.time():.1f}] SA new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final greedy pairwise polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy swap polishing")
        a_opt, sc_opt = _pairwise_swap(
            best_seq.copy(),
            deadline,
            best_seq=best_seq,
            best_score=best_score,
        )
        if sc_opt < best_score - 1e-12:
            best_score, best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps best {best_score:.6f}")

    print(f"[{time.time():.1f}] final objective {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
