def _solve_lp_max_sum(f, M, deadline):
    n = len(f)
    m = 2*n - 1
    # Build matrix A (m,n) dense
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = f  # fill slice rows j..j+n-1 for column j
    b = np.full(m, M, dtype=np.float64)
    c = -np.ones(n)  # maximize sum => minimize -sum
    bounds = [(0, None)] * n
    # Use method='highs' (default??). Set options/time limit maybe using max_iter?
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None, False
    g = res.x
    return g, True

"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.

Algorithm
---------

1. **Generate a few analytic seeds**  (uniform, left‑exponential,
   Dirichlet, sparse random).  All seeds are normalised to sum = 1.

2. **Iterative LP refinement** – given a current vector *f*, compute
   M = max(conv(f,f)).  Solve the linear program

        maximise   Σ g_i
        subject to conv(f,g) ≤ M      (one inequality per shift)
                  g_i ≥ 0

   The LP pushes mass into positions where *f* is small while keeping the
   convolution bound.  The solution *g* is re‑normalised to sum = 1 and
   mixed with *f*:   f←(1‑t)·f + t·g  (t starts at 0.5 and decays).

   A few LP‑mix iterations are performed; each LP is built with a dense
   Toeplitz matrix and solved with SciPy’s HiGHS solver.

3. **Adam‑based mirror descent** – the best vector from step 2 is then
   polished by Adam optimisation on the soft‑max parametrisation.
   The sub‑gradient uses *all* active convolution entries (those equal to
   the maximum up to a tiny tolerance).

4. **Local search** – a short greedy pairwise mass‑transfer followed by
   a cheap simulated‑annealing style search removes any remaining spikes.

All phases respect the global time budget (default 1000 s).  The returned
list is a valid non‑negative sequence; the evaluator will re‑normalise it
anyway.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Basic convolution / objective utilities
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) – a must sum to 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families (return probability vectors, sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_exponential_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-rate * i)
    w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator, conc: float) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a.astype(np.float64)


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, U = uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Linear programme: maximise total mass of g under the convolution bound
def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve
        maximise   Σ g_i
        subject to conv(f,g) ≤ M   (one inequality per shift)
                  g_i ≥ 0
    Returns the optimal g (not normalised).  Raises RuntimeError if LP fails.
    """
    n = f.shape[0]
    m = 2 * n - 1                     # number of convolution entries
    # Build the Toeplitz‐style constraint matrix A (shape m×n)
    A = np.empty((m, n), dtype=np.float64)
    A[:] = 0.0
    for j in range(n):
        rows = slice(j, j + n)        # rows j … j+n‑1  (exclusive upper bound)
        A[rows, j] = f                # column j gets a copy of f
    b = np.full(m, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)     # maximise Σg_i  ⇔ minimise –Σg_i
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


# ----------------------------------------------------------------------
# Adam mirror‑descent on the exact objective
def _adam_refine(
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    lr0: float = 0.6,
    active_frac: float = 0.001,
    max_active: int | None = None,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_interval: int = 250,
    best_seq: np.ndarray | None = None,
    best_score: float = float("inf"),
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max parametrisation.
    The gradient is the exact sub‑gradient obtained from **all**
    maximal convolution entries (within a tiny tolerance).
    """
    n = init_a.shape[0]
    x = np.log(np.maximum(init_a, 1e-12))
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    ind_grid = np.arange(n, dtype=np.int32)
    t = 0
    if best_seq is None:
        best_seq = init_a.copy()
        best_score = _objective(init_a)

    while time.time() < deadline and t < max_iters:
        t += 1
        a = np.exp(x - np.max(x))
        a /= a.sum()

        conv = _fft_self_convolve(a)
        max_c = conv.max()
        # active set = all indices whose value is within 1e‑12 absolute of the max
        act = np.where(np.abs(conv - max_c) <= 1e-12)[0]
        if max_active is not None and act.size > max_active:
            # keep the largest ones only – rarely triggered
            top = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top]

        # sub‑gradient of the max (sum of gradients of the active entries)
        grad_a = np.zeros_like(a)
        for k in act:
            offset = k - ind_grid
            mask = (offset >= 0) & (offset < n)
            grad_a[mask] += 2.0 * a[offset[mask]]

        # convert ∂L/∂a to ∂L/∂x (mirror‑descent on simplex)
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update with a decaying step‑size η_t = lr0 / √t
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = lr0 / math.sqrt(t)
        x -= eta * m_hat / (np.sqrt(v_hat) + eps)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] adam it={t:,} new best {best_score:.6f}"
                )
    best_seq = best_seq / best_seq.sum()   # safety projection onto simplex
    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise refinement
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score

# Very cheap simulated‑annealing style local search
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}"
            )
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score
