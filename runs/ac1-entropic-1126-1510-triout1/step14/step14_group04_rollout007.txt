"""
Hybrid optimizer for the step‑function objective

Goal: minimise  evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum a)^2
under a_i ≥ 0.  The global optimum is invariant to scaling, so we
optimise on the probability simplex (Σa=1) and return the raw vector.

Algorithm
---------
1.  Scan a modest set of lengths (≈2–4.5k) with a variety of analytic
    families (uniform, arcsine, sine, cosine, quadratic, beta,
    edge‑weighted, exponential‑edge).  A dense grid on the
    family‑specific parameters (p, c, α) and a mixing weight λ with the
    uniform distribution is explored.  The best few candidates are
    kept.

2.  For each length, a local continuous optimisation (SciPy L‑BFGS‑B)
    fine‑tunes the two‑parameter families
        – edge‑weighted  w_i ∝ (i·(n‑i))^{-p}
        – exponential‑edge  w_i ∝ e^{-c·i}+e^{-c·(n‑1‑i)}
        – symmetric beta  w_i ∝ x^{α-1}(1-x)^{α-1}
    (both mixed with the uniform distribution).  The optimisation
    respects 10^{x}∈[10^{-4},10^{2}] and λ∈[0,1].

3.  A smoothed‑max mirror‑descent (Adam) is run from the best
    candidates.  The non‑smooth max is approximated by a log‑sum‑exp
    with a schedule of smoothing parameters β growing up to 500.
    More iterations per β are allowed than in the previous code.

4.  Exact sub‑gradient descent (exponentiated‑gradient) refines the
    best solution on the true L∞‑objective.

5.  A final greedy pairwise mass‑transfer hill‑climber moves a random
    fraction of mass between two indices whenever the exact objective
    strictly improves.

The whole pipeline respects a user‑supplied time budget (default
1000 s) and always returns the best sequence found so far.
"""

# ------------------------------------------------------------
#  Utility: projection onto the probability simplex
# ------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of `v` onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
#  FFT based convolutions (real inputs)
# ------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a * a using a real FFT."""
    return _fft_convolve(a, a)


# ------------------------------------------------------------
#  Exact objective on the simplex (Σa = 1)
# ------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

#  Analytic families (all already normalised to sum‑1)
# ------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i·(n‑i))^{-p}   (i,j are 0‑based half‑offsets)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    w /= w_sum
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    """w_i ∝ exp(-c·i) + exp(-c·(n‑1‑i))   (c>0)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric beta‑type distribution on the discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    total = w.sum()
    if total == 0.0:
        return _distribution_uniform(n)
    w /= total
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base (both already sum‑1)."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


# ------------------------------------------------------------
#  Mixed‑parameter objective helpers (used by SciPy)
# ------------------------------------------------------------
def _edge_mix_obj(params: np.ndarray, n: int) -> float:
    x, lam = params
    p = 10.0 ** x
    a = _mix_with_uniform(_distribution_edge_weighted(n, p), float(lam))
    return _objective(a)


def _edge_mix_vec(params: np.ndarray, n: int) -> np.ndarray:
    x, lam = params
    p = 10.0 ** x
    a = _mix_with_uniform(_distribution_edge_weighted(n, p), float(lam))
    # normalisation is a property of the mixing operation, but guard against round‑off
    a_sum = a.sum()
    if a_sum == 0.0:
        return _distribution_uniform(n)
    return a / a_sum


def _exp_mix_obj(params: np.ndarray, n: int) -> float:
    x, lam = params
    c = 10.0 ** x
    a = _mix_with_uniform(_distribution_exponential_edge(n, c), float(lam))
    return _objective(a)


def _exp_mix_vec(params: np.ndarray, n: int) -> np.ndarray:
    x, lam = params
    c = 10.0 ** x
    a = _mix_with_uniform(_distribution_exponential_edge(n, c), float(lam))
    a_sum = a.sum()
    if a_sum == 0.0:
        return _distribution_uniform(n)
    return a / a_sum


def _beta_mix_obj(params: np.ndarray, n: int) -> float:
    x, lam = params
    alpha = 10.0 ** x
    a = _mix_with_uniform(_distribution_beta(n, alpha, alpha), float(lam))
    return _objective(a)


def _beta_mix_vec(params: np.ndarray, n: int) -> np.ndarray:
    x, lam = params
    alpha = 10.0 ** x
    a = _mix_with_uniform(_distribution_beta(n, alpha, alpha), float(lam))
    a_sum = a.sum()
    if a_sum == 0.0:
        return _distribution_uniform(n)
    return a / a_sum


def _local_optimize(obj_fn, x0, n, deadline):
    """Run a quick L‑BFGS‑B optimisation (max 80 iters) but abort if out of time."""
    if not _HAS_SCIPY:
        return None, None
    if time.time() >= deadline:
        return None, None
    bounds = Bounds([-4.0, 0.0], [2.0, 1.0])
    res = minimize(
        lambda p: obj_fn(p, n),
        x0=np.array(x0, dtype=np.float64),
        method="L-BFGS-B",
        bounds=bounds,
        options={"maxiter": 80, "ftol": 1e-12, "disp": False},
    )
    if not res.success:
        return None, None
    return res.x, res.fun


# ------------------------------------------------------------
#  Smoothed‑max mirror descent (Adam + β‑continuation)
# ------------------------------------------------------------
def _smooth_max_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp surrogate, β growing up to 500."""
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-9
    t = 0

    beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0]
    max_iter_beta = 12000

    for beta in beta_schedule:
        it_this_beta = 0
        while time.time() < time_limit and it_this_beta < max_iter_beta:
            t += 1
            it_this_beta += 1

            conv = _fft_self_convolve(a)                 # length 2n‑1
            max_val = conv.max()
            # log‑sum‑exp (smooth max)
            shifted = conv - max_val                    # ≤0
            exp_beta = np.exp(beta * shifted)
            w = exp_beta / exp_beta.sum()               # Σw = 1

            # gradient of smooth max: 2·(w * a_rev) convolved
            a_rev = a[::-1]
            grad_conv = _fft_convolve(w, a_rev)         # length 3n‑2
            grad = 2.0 * grad_conv[n - 1 : n - 1 + n]    # size n

            grad_J = 2.0 * n * grad                     # gradient of J = 2n·smooth_max

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_J
            v = beta2 * v + (1.0 - beta2) * (grad_J ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = 0.12 / (np.sqrt(v_hat) + eps)

            a -= step * m_hat
            a = _project_onto_simplex(a, s=1.0)

            # Periodic exact evaluation
            if t % 250 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-14:
                    best_score, best_seq = cur, a.copy()
                    print(f"[{time.time():.1f}] β={beta:.1f} it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ------------------------------------------------------------
#  Exact sub‑gradient descent (exponentiated gradient)
# ------------------------------------------------------------
def _subgradient_descent_exact(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true L∞‑objective."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        # sub‑gradient (sum of shifted copies)
        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)          # stay non‑negative
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score, best_seq = cur, a.copy()
                print(f"[{time.time():.1f}] exact subgrad it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ------------------------------------------------------------
#  Greedy pairwise mass‑transfer hill‑climber
# ------------------------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly relocate a fraction of mass; accept only strict improvements."""
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score, best_seq = cur, a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")

    return best_seq, best_score


# ------------------------------------------------------------
#  Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a step‑function (plain Python list) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # small safety margin
    deadline = start + budget_s - 1.0

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, vector)

    # ---------------------------------------------------------
    # 1️⃣  Analytic‑family scan (moderate lengths)
    # ---------------------------------------------------------
    lengths = [2000, 2600, 3200, 3800, 4400]

    # Parameter grids
    p_grid = np.concatenate(
        [
            np.logspace(-3, -0.5, 13),          # extreme edge‑heavy
            np.logspace(-0.5, 0.0, 11),         # moderate
            np.linspace(0.2, 3.0, 14),          # broader range
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 11)       # mixing weight with uniform
    c_grid = np.logspace(-3, 0, 10)           # exponential‑edge decay
    alpha_grid = np.logspace(-0.5, 0.7, 12)   # beta‑distribution parameters (α=β)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families -------------------------------------------------
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
        ):
            a = base_fn(n)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"   new global best (det) len={n} score={sc:.6f}")

        # beta family ---------------------------------------------------------
        for alpha in alpha_grid:
            if time.time() >= deadline:
                break
            a = _distribution_beta(n, float(alpha), float(alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"   new global best beta α={alpha:.4g} len={n} score={sc:.6f}")

        # edge‑weighted + uniform mixes ----------------------------------------
        for p in p_grid:
            if time.time() >= deadline:
                break
            base = _distribution_edge_weighted(n, float(p))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                a = _mix_with_uniform(base, lam)
                sc = _objective(a)
                top_candidates.append((sc, a.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, a.copy()
                    print(f"   new global best edge p={p:.4g} lam={lam:.2f} len={n} score={sc:.6f}")

        # exponential‑edge family ----------------------------------------------
        for c in c_grid:
            if time.time() >= deadline:
                break
            base = _distribution_exponential_edge(n, float(c))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                a = _mix_with_uniform(base, lam)
                sc = _objective(a)
                top_candidates.append((sc, a.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, a.copy()
                    print(f"   new global best exp c={c:.4g} lam={lam:.2f} len={n} score={sc:.6f}")

        # Dirichlet random seeds ------------------------------------------------
        for _ in range(4):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.05, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"   new global best Dirichlet seed len={n} score={sc:.6f}")

        # Sparse random seeds ---------------------------------------------------
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"   new global best sparse seed len={n} score={sc:.6f}")

    # Keep a manageable number of top seeds
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:30]

    # ---------------------------------------------------------
    # 2️⃣  Local continuous optimisation (SciPy) if available
    # ---------------------------------------------------------
    if _HAS_SCIPY:
        for n in lengths:
            if time.time() >= deadline:
                break
            # ---- edge‑weighted -------------------------------------------------
            x0 = [np.log10(float(p_grid[len(p_grid) // 2])), 0.5]
            res_x, res_f = _local_optimize(_edge_mix_obj, x0, n, deadline)
            if res_x is not None and res_f is not None and res_f < overall_best_score:
                a_opt = _edge_mix_vec(res_x, n)
                overall_best_score, overall_best_seq = res_f, a_opt.copy()
                print(f"[{time.time():.1f}] edge‑opt n={n} score={res_f:.6f}")
                top_candidates.append((res_f, a_opt.copy()))

            # ---- exponential‑edge --------------------------------------------
            x0 = [np.log10(float(c_grid[len(c_grid) // 2])), 0.5]
            res_x, res_f = _local_optimize(_exp_mix_obj, x0, n, deadline)
            if res_x is not None and res_f is not None and res_f < overall_best_score:
                a_opt = _exp_mix_vec(res_x, n)
                overall_best_score, overall_best_seq = res_f, a_opt.copy()
                print(f"[{time.time():.1f}] exp‑opt n={n} score={res_f:.6f}")
                top_candidates.append((res_f, a_opt.copy()))

            # ---- beta ---------------------------------------------------------
            x0 = [np.log10(float(alpha_grid[len(alpha_grid) // 2])), 0.5]
            res_x, res_f = _local_optimize(_beta_mix_obj, x0, n, deadline)
            if res_x is not None and res_f is not None and res_f < overall_best_score:
                a_opt = _beta_mix_vec(res_x, n)
                overall_best_score, overall_best_seq = res_f, a_opt.copy()
                print(f"[{time.time():.1f}] beta‑opt n={n} score={res_f:.6f}")
                top_candidates.append((res_f, a_opt.copy()))

        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:20]

    # ---------------------------------------------------------
    # 3️⃣  Smoothed‑max mirror descent on a few promising seeds
    # ---------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 40.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] smoothed‑max descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_max_descent(
                seed_seq,
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after smooth seed {idx} new best {overall_best_score:.6f}")

    # ---------------------------------------------------------
    # 4️⃣  Exact sub‑gradient refinement (exponentiated‑gradient)
    # ---------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        print(f"[{time.time():.1f}] exact sub‑gradient refinement")
        a_opt, sc_opt = _subgradient_descent_exact(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.9,
            eval_every=120,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after exact subgrad new best {overall_best_score:.6f}")

    # ---------------------------------------------------------
    # 5️⃣  Final greedy pairwise polishing
    # ---------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy swap polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ---------------------------------------------------------
    # 6️⃣  Return result
    # ---------------------------------------------------------
    print(f"Final best objective (2·n·max(conv)) = {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
