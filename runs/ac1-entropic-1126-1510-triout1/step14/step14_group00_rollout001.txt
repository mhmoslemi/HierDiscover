"""
Improved search for a non‑negative step function minimizing

    2·n·max(conv(a,a)) / (∑a)²

The algorithm expands on the previous pipeline:

1. **Very diverse seeding** – families: uniform, Hann, sine‑window,
   Gaussian, exponential‑taper, symmetric Beta, generalized normal and
   mixtures with the uniform distribution.  Seeds are generated for a
   schedule of lengths up to 65536 (the practical FFT limit) but only a
   handful of parameter combinations per length are evaluated.

2. **Fast mirror‑descent polishing** – an adaptive exponentiated‑gradient
   (mirrored sub‑gradient) with momentum (EMA) and step‑size control.
   All seeds receive a short run; the current global best receives a
   longer run.

3. **Progressive up‑sampling** – whenever time permits the best vector
   is linearly interpolated to twice its size, immediately followed by a
   medium‑length gradient run.

4. **Targeted annealing** – a specialised simulated‑annealing phase.
   The move chooser focuses on the indices that contribute most to the
   current maximal convolution entry, transferring a random fraction of
   their mass to low‑mass locations.  This aggressively flattens the
   autocorrelation and often yields a final improvement beyond the
   generic pairwise‑annealing used previously.

All steps respect the 1000 s budget (2 s safety margin).  The returned
list is ready for ``evaluate_sequence`` (the values are non‑negative).

"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast full self‑convolution using FFT (real‑valued input)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective for a probability vector (∑a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families – each returns a *probability* vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w /= w.sum()
    return w


def _distribution_sine(n: int) -> np.ndarray:
    """Window w_i ∝ sin(π·(i+0.5)/n) – zero at the edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    """Gaussian‑like: exp(−(x−½)² / (2σ²))."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(-0.5 * (x / sigma) ** 2)
    w /= w.sum()
    return w


def _distribution_exponential_taper(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential taper exp(−λ·|x−½|)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(-lam * np.abs(x))
    w /= w.sum()
    return w


def _distribution_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the cell mid‑points."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # avoid 0/1 edge cases
    eps = np.finfo(float).eps
    x = np.clip(x, eps, 1 - eps)
    w = np.exp((alpha - 1) * np.log(x) + (alpha - 1) * np.log1p(-x))
    w /= w.sum()
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(−|x−½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1‑λ) + uniform·λ."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution entry (mirrored gradient)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """Return (grad, k) where k = argmax conv and grad_i = 2·a_{k‑i} on the
    overlap, otherwise 0."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))
    grad = np.zeros_like(a)
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
# Adaptive exponentiated‑gradient (mirror descent) on the simplex
# ----------------------------------------------------------------------
def _adaptive_exp_gradient(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    beta: float = 0.9,
    eval_every: int = 50,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent style optimisation with EMA of sub‑gradients."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)  # EMA of gradients
    eta = eta0
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        m = beta * m + (1.0 - beta) * grad
        a = a * np.exp(-eta * m)
        s = a.sum()
        if s == 0.0:
            a.fill(1.0 / n)
        else:
            a /= s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                eta = min(eta0, eta * 1.02)          # reward
                print(f"[{time.time():.1f}] GD it={t:,} new best {best_score:.6f}")
            else:
                eta *= 0.995                         # decay
        # occasional re‑projection for numerical safety
        if t % (eval_every * 10) == 0:
            a = _project_onto_simplex(a, s=1.0)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Targeted simulated annealing – moves mass from heavy contributors
# ----------------------------------------------------------------------
def _targeted_annealing(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Mass‑transfer moves that focus on the indices contributing most to the
    active (max) convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        # compute current maximal convolution entry
        conv = _fft_self_convolve(a)
        k = int(np.argmax(conv))

        # overlapping region [start_i, end_i] (inclusive)
        start_i = max(0, k - (n - 1))
        end_i = min(k, n - 1)
        idxs = np.arange(start_i, end_i + 1)

        # contribution of each i to the max entry
        contrib = a[idxs] * a[k - idxs]
        total_contrib = contrib.sum()
        if total_contrib == 0.0:
            src = rng.integers(start_i, end_i + 1)
        else:
            # weighted random selection
            r = rng.random() * total_contrib
            src = idxs[np.searchsorted(np.cumsum(contrib), r)]

        # pick destination among indices with relatively small mass
        # (helps to flatten the distribution)
        low_mask = a < np.median(a)
        low_idxs = np.flatnonzero(low_mask)
        if low_idxs.size == 0:
            dst = rng.integers(n - 1)
            if dst >= src:
                dst += 1
        else:
            dst = int(rng.choice(low_idxs))
            if dst == src:
                # ensure distinct
                dst = (src + rng.integers(1, n)) % n

        # transfer a random fraction (up to 30 %) of the source mass
        delta = a[src] * rng.random() * 0.30
        if delta < 1e-18:
            continue

        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        cur = _objective(new_a)

        # linear cooling: temperature decreases from a small fraction of best_score
        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.015

        accept = False
        if cur < best_score - 1e-15:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] targeted anneal {src}->{dst} Δ={delta:.1e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling while preserving total mass (=1)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new

    # --------------------------------------------------------------
    if time.time() < deadline:
        sub_dead = time.time() + (deadline - time.time()) * 0.35
        a_opt, sc_opt = _adaptive_exp_gradient(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.55,
            beta=0.9,
            eval_every=60,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] long GD improved best to {overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 6️⃣  Progressive up‑sampling (doubling) while time permits
    # --------------------------------------------------------------
    while time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n > 65536:
            break
        print(f"[{time.time():.1f}] up‑sampling {cur_n} → {target_n}")
        up = _upsample(overall_best_seq, target_n)
        up_sc = _objective(up)
        if up_sc < overall_best_score:
            overall_best_seq, overall_best_score = up, up_sc
            print(f"[{time.time():.1f}] up‑sampled new best {overall_best_score:.6f}")

        # medium‑length GD on the up‑sampled vector
        sub_dead = time.time() + (deadline - time.time()) * 0.2
        a_opt, sc_opt = _adaptive_exp_gradient(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.45,
            beta=0.9,
            eval_every=50,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
            print(f"[{time.time():.1f}] GD after up‑sample new best {overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 7️⃣  Final targeted annealing phase
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final targeted annealing")
        a_opt, sc_opt = _targeted_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
            print(f"[{time.time():.1f}] after annealing new best {overall_best_score:.6f}")

    # --------------------------------------------------------------
    # Finish
    # --------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)) = {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    # ``evaluate_sequence`` expects a plain Python list
    return overall_best_seq.tolist()
