"""
Hybrid optimization for the convolution‑maximisation problem.

The evaluator computes
    2·n·max(conv(a,a)) / (sum(a)^2)

which is invariant to a global scaling, so we optimise a probability vector
(a ≥ 0, sum a = 1).  The algorithm consists of

1️⃣  Analytic seeding – a few parametric families for a handful of lengths.
2️⃣  Fast Adam mirror‑descent on the exact objective (soft‑max parametrisation).
3️⃣  One “LP boost’’ – given the current f, maximise Σ g under the
    (f * g) ≤ max(f * f) constraints (linear programme, SciPy HiGHS).  The
    resulting g is rescaled to a probability vector and convex‑combined with f.
4️⃣  A long Adam run from the best candidate.
5️⃣  Greedy pairwise mass transfers and a cheap simulated‑annealing polishing.

All steps stop automatically before the global deadline (default 1000 s).  The
returned list can be fed directly to ``evaluate_sequence`` – it will be
re‑normalised there.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft
from scipy.optimize import linprog
from scipy import sparse

# ----------------------------------------------------------------------
# Utility building blocks
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(x: np.ndarray) -> np.ndarray:
    """Stable softmax (probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """
    Exact objective  2·n·max(conv(a,a)) .
    The argument must already be a probability vector (Σa = 1).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families – all return a probability vector (Σ=1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    w /= w.sum()
    return w


def _distribution_exponential_center(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(i - half))
    w /= w.sum()
    return w


def _distribution_exponential_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-rate * i)
    w /= w.sum()
    return w


def _distribution_exponential_right(n: int, rate: float) -> np.ndarray:
    return _distribution_exponential_left(n, rate)[::-1]


def _distribution_dirichlet(n: int, rng: np.random.Generator, conc: float) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a.astype(np.float64)


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Adam‑based mirror descent on the exact objective
def _adam_refine(
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    lr0: float = 0.9,
    active_frac: float = 0.02,
    max_active: int = 30,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_interval: int = 250,
    best_seq: np.ndarray | None = None,
    best_score: float = float("inf"),
) -> tuple[np.ndarray, float]:
    """Adam on the (soft‑max) probability simplex."""
    n = init_a.shape[0]
    x = np.log(np.maximum(init_a, 1e-12))   # log‑weights
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    idx_grid = np.arange(n, dtype=np.int32)
    t = 0
    if best_seq is None:
        best_seq = init_a.copy()
        best_score = _objective(init_a)

    while time.time() < deadline and t < max_iters:
        t += 1
        a = _softmax(x)                     # stays on simplex
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            # keep only the largest max_active entries
            top = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top]

        # sub‑gradient from the active set
        grad_a = np.zeros_like(a)
        for idx_k in act:
            offset = idx_k - idx_grid
            mask = (offset >= 0) & (offset < n)
            grad_a[mask] += 2.0 * a[offset[mask]]

        # convert to gradient w.r.t. log‑weights (mirror‑descent rule)
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update (step‑size decays like 1/√t)
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = lr0 / math.sqrt(t)
        x -= eta * m_hat / (np.sqrt(v_hat) + eps)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t:,} new best {best_score:.6f}")

    best_seq = _project_onto_simplex(best_seq, s=1.0)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear‑programming boost (maximise Σ g under f * g ≤ M)
def _lp_maximise_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max Σ g_j  s.t.   (f * g)_k ≤ M  (∀k),   g ≥ 0.
    Returns the optimal g (not normalised) or None on failure.
    """
    n = f.shape[0]
    # Objective: minimise -Σ g  (i.e. maximise Σ g)
    c = -np.ones(n)

    # Build the (2n‑1) × n constraint matrix for the linear convolution f * g.
    # Row k contains f_{k‑j} for all feasible j.
    i = np.arange(n, dtype=np.int32)
    j = np.arange(n, dtype=np.int32)

    # rows = i + j (broadcast)
    rows = (j[:, None] + i[None, :]).ravel()
    cols = np.repeat(j, n)
    data = np.tile(f, n)

    A = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    b = np.full(2 * n - 1, M)

    bounds = [(0, None)] * n

    # HiGHS (default) works well with sparse matrices.
    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if res.success:
        return np.maximum(res.x, 0.0)   # safety strip
    else:
        print(f"[{time.time():.1f}] LP failed: {res.message}")
        return None


def _lp_boost(f: np.ndarray, deadline: float) -> np.ndarray:
    """
    Perform a single LP‑boost step on f.
    Returns a possibly improved probability vector (same sum) or f unchanged.
    """
    n = f.shape[0]
    conv_f = _fft_self_convolve(f)
    M = conv_f.max()
    # solve LP
    g_raw = _lp_maximise_sum(f, M)
    if g_raw is None:
        return f
    sum_g_raw = g_raw.sum()
    if sum_g_raw <= 1.0 + 1e-12:
        # no improvement possible
        return f

    # rescale to unit sum
    g = g_raw / sum_g_raw

    # convex combine – a modest step size keeps the bound safe
    t = 0.4
    f_new = (1.0 - t) * f + t * g
    f_new = _project_onto_simplex(f_new, s=1.0)
    new_score = _objective(f_new)
    old_score = _objective(f)
    if new_score < old_score - 1e-12:
        print(f"[{time.time():.1f}] LP‑boost improved {old_score:.6f} → {new_score:.6f}")
        return f_new
    return f

# Greedy pairwise refinement (same as before)
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Try to move a small fraction of mass from one index to another."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score

# Very cheap simulated‑annealing style local search
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Entry point – returns a plain Python list of non‑negative heights."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ==================== 1️⃣ SEEDING ====================
    lengths = [1500, 2500, 3500, 4500, 6000]
    p_grid = np.logspace(-1, 1.0, 6)          # sine‑weighted exponents
    lam_grid = np.linspace(0.0, 0.5, 5)       # mixing with uniform
    rate_grid = np.linspace(0.001, 0.02, 5)   # exponential rates
    conc_grid = [0.05, 0.2, 1.0, 5.0]        # Dirichlet concentrations

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        bases = [
            _distribution_uniform(n),
            _distribution_sine_weighted(n, p=1.0),
            _distribution_exponential_center(n, rate=0.005),
            _distribution_exponential_left(n, rate=0.003),
            _distribution_exponential_right(n, rate=0.003),
        ]
        for base in bases:
            sc = _objective(base)
            top_candidates.append((sc, base.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new global best (det) len={n} score={sc:.6f}")

        # sine‑weighted + uniform mixing
        for p in p_grid:
            sine = _distribution_sine_weighted(n, p)
            for lam in lam_grid:
                cand = _mix_with_uniform(sine, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new global best sine p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # centre‑exponential family
        for rate in rate_grid:
            cad = _distribution_exponential_center(n, rate)
            sc = _objective(cad)
            top_candidates.append((sc, cad.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cad.copy()
                print(f"[{time.time():.1f}] new global best exp‑center rate={rate:.4f} score={sc:.6f}")

        # Dirichlet draws
        for conc in conc_grid:
            for _ in range(2):
                a = _distribution_dirichlet(n, rng, conc)
                sc = _objective(a)
                top_candidates.append((sc, a.copy()))
                if sc < overall_best_score:

                    print(f"[{time.time():.1f}] new global best Dirichlet conc={conc:.3g} score={sc:.6f}")

        # sparse random
        for _ in range(3):
            a = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}")

        # keep only a handful of promising seeds
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:12]

    # ------------------------------------------------------------------
    # include a previously‑found construction if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2️⃣ short Adam runs on the retained seeds

            overall_best_seq.copy(),

            beta1=0.9,
            beta2=0.999,
            eval_interval=250,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] after long descent best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------

    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=200_000,
        )

    # ------------------------------------------------------------------
    # 6️⃣ final simulated‑annealing polishing
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.01,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(f"Final best score (exact) ≈ {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
