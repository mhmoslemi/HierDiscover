    return float(2.0 * n * maxb / (s*s))

def _grad_of_max_conv(a):
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))
    grad = np.zeros_like(a)
    if k < n:
        grad[:k+1] = 2.0 * a[k::-1]
    else:
        start = k - (n-1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k

def _project_simplex(v):
    # classic algorithm
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv/ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _project_simplex(v):
    # v is a ndarray of floats (may contain negative)
    # Projects onto probability simplex {x >=0, sum=1}
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    # find rho
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # all values are zero after projection
        return np.full(n, 1.0/n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _distribution_beta(n, alpha, beta):

All stages respect the 1000 s wall‑time limit and keep the current
best candidate.  The functions are written at top level and avoid
closures as required.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  FFT‑based linear convolution (real inputs, full convolution)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
#  Exact objective used by the evaluator (cost = 2·n·max(conv)/sum²)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a))/sum(a)², a must be non‑negative."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
#  Sub‑gradient of the current max convolution entry
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """Return (gradient, index_of_max) for the max entry of a*a."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))

    grad = np.zeros_like(a)
    if k < n:
        # overlap length = k+1   (i runs 0..k)
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        # k ≥ n : overlap length = 2·n‑1‑k   (i runs k‑(n‑1)..n‑1)
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
#  Simplex projection (Duchi et al.)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Analytic families – all return a *probability* vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i·(n‑i))^{-p}, i = distance to the left edge (½‑offset)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_sum_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ i^{-p} + (n‑i)^{-p} (sum of two one‑sided tails)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (-p)) + (j ** (-p))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Discrete pdf ∝ exp( -|x‑½|^{q} ) on the grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑family: w_i ∝ (i+½)^{α‑1}·(n‑i‑½)^{β‑1}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (alpha - 1.0)) * (j ** (beta - 1.0))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1‑lam) + uniform·lam."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
#  Exponentiated‑gradient descent (multiplicative mirror descent)
# ----------------------------------------------------------------------
def _exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """Run EG descent on the exact objective until ``deadline``."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Projected Adam optimizer on the simplex
# ----------------------------------------------------------------------
def _adam_optimize(
    a: np.ndarray,
    deadline: float,
    best_score: float,
    best_seq: np.ndarray,
    eta: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_every: int = 30,
) -> tuple[np.ndarray, float]:
    """Additive Adam with projection onto the simplex."""
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    while time.time() < deadline:
        grad, _ = _grad_of_max_conv(a)
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = _project_simplex(a_new)

        cur = _objective(a_new)
        if cur < best_score - 1e-14:
            best_score = cur
            best_seq = a_new.copy()
            print(f"[{time.time():.1f}] Adam it={t} new best {best_score:.6f}")

        a = a_new
        if t % eval_every == 0 and time.time() >= deadline:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Greedy pairwise mass transfer directed at the active convolution entry
# ----------------------------------------------------------------------
def _pairwise_greedy_opt(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Deterministic moves from high‑gradient to low‑mass bins."""
    n = a.shape[0]
    no_improve_streak = 0
    max_no_improve = 2000  # stop after many unsuccessful attempts

    while time.time() < deadline and no_improve_streak < max_no_improve:
        grad, _ = _grad_of_max_conv(a)
        i_src = int(np.argmax(grad))
        # target: the index with smallest mass (break ties randomly)
        i_tgt = int(np.argmin(a))

        if i_src == i_tgt:
            # select a random distinct target
            i_tgt = rng.integers(n - 1)
            if i_tgt >= i_src:
                i_tgt += 1

        if a[i_src] < 1e-12:
            break

        max_delta = a[i_src]
        improved = False

        # try a few fractions, largest first
        for frac in (0.5, 0.2, 0.1, 0.05, 0.02, 0.01):
            delta = max_delta * frac
            if delta < 1e-14:
                continue
            new_a = a.copy()
            new_a[i_src] -= delta
            new_a[i_tgt] += delta
            cur = _objective(new_a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = new_a.copy()
                a = new_a
                improved = True
                no_improve_streak = 0
                print(f"[{time.time():.1f}] greedy src={i_src} tgt={i_tgt} δ={delta:.3e} new best {best_score:.6f}")
                break

        if not improved:
            # try a tiny move toward a random location – may help escape plateaus
            for _ in range(2):
                i_rand = rng.integers(n)
                if i_rand == i_src:
                    continue
                delta = max_delta * 0.005
                new_a = a.copy()
                new_a[i_src] -= delta
                new_a[i_rand] += delta
                cur = _objective(new_a)
                if cur < best_score - 1e-14:
                    best_score = cur
                    best_seq = new_a.copy()
                    a = new_a
                    improved = True
                    no_improve_streak = 0
                    print(f"[{time.time():.1f}] tiny greedy src={i_src} tgt={i_rand} δ={delta:.3e} new best {best_score:.6f}")
                    break

        if not improved:
            no_improve_streak += 1

    return best_seq, best_score


# ----------------------------------------------------------------------
#  Pairwise‑mass‑transfer simulated‑annealing polish (existing implementation)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins; accept improvements."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4        # move at most 40 % of source mass
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Returns a plain list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0      # safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")
    top: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Generate a diverse pool of seeds
    # ------------------------------------------------------------------
    lengths = [2500, 4000, 6000, 9000, 13000]       # varied moderate sizes
    # parameters for edge families
    p_grid = np.concatenate((
        np.logspace(-2.0, -0.5, 10, base=10),      # 0.01 … 0.316
        np.logspace(-0.5, 0.5, 12, base=10),       # 0.316 … 3.16
        np.logspace(0.5, 1.5, 8, base=10),         # 3.16 … 31.6
    ))
    lam_grid = np.linspace(0.0, 1.0, 11)           # mixing with uniform
    q_grid = np.linspace(0.5, 3.0, 9)              # generalized normal
    beta_alphas = [0.2, 0.3, 0.4, 0.5, 0.7, 1.0]   # beta‑family exponents

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for base in (_distribution_uniform(n),
                     _distribution_arcsine(n)):
            sc = _objective(base)
            top.append((sc, base.copy()))
            if sc < best_score:
                best_score, best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new global best (uniform/arcsine) score={sc:.6f}")

        # product edge‑weighted family (+ mixing)
        for p in p_grid:
            if time.time() >= deadline:
                break
            base = _distribution_edge_weighted(n, p)
            sc = _objective(base)
            top.append((sc, base.copy()))
            if sc < best_score:
                best_score, best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new best prod p={p:.3g} score={sc:.6f}")

            for lam in lam_grid:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                top.append((sc, cand.copy()))
                if sc < best_score:
                    best_score, best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new best prod p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # sum edge‑weighted family (+ mixing)
        for p in p_grid:
            if time.time() >= deadline:
                break
            base = _distribution_edge_sum_weighted(n, p)
            sc = _objective(base)
            top.append((sc, base.copy()))
            if sc < best_score:
                best_score, best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new best sum p={p:.3g} score={sc:.6f}")

            for lam in lam_grid:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                top.append((sc, cand.copy()))
                if sc < best_score:
                    best_score, best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new best sum p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # generalized‑normal family
        for q in q_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_generalized_normal(n, q)
            sc = _objective(cand)
            top.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best gn q={q:.2f} score={sc:.6f}")

        # beta‑family (symmetric)
        for alpha in beta_alphas:
            for beta_param in beta_alphas:
                base = _distribution_beta(n, alpha, beta_param)
                sc = _objective(base)
                top.append((sc, base.copy()))
                if sc < best_score:
                    best_score, best_seq = sc, base.copy()
                    print(f"[{time.time():.1f}] new best beta α={alpha:.2f} β={beta_param:.2f} score={sc:.6f}")

                for lam in lam_grid:
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top.append((sc, cand.copy()))
                    if sc < best_score:
                        best_score, best_seq = sc, cand.copy()
                        print(f"[{time.time():.1f}] new best beta mix α={alpha:.2f} β={beta_param:.2f} lam={lam:.2f} score={sc:.6f}")

        # random sparse Dirichlet seeds (few non‑zero entries)
        for _ in range(8):
            if time.time() >= deadline:
                break
            cand = np.zeros(n, dtype=np.float64)
            k = max(2, int(n * 0.015))
            idx = rng.choice(n, size=k, replace=False)
            cand[idx] = rng.random(k)
            cand /= cand.sum()
            sc = _objective(cand)
            top.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best sparse score={sc:.6f}")

        # keep only the very best seeds (size 20)
        top.sort(key=lambda x: x[0])
        top = top[:20]

    # ------------------------------------------------------------------
    # 2.  Include the previously submitted construction if it exists
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev_sum = prev.sum()
            if prev_sum > 0:
                prev = prev / prev_sum
                sc = _objective(prev)
                if sc < best_score:
                    best_score, best_seq = sc, prev.copy()
                    print(f"[{time.time():.1f}] previous best score={sc:.6f}")
                top.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Guard against an empty result
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _distribution_uniform(5000)
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Exponentiated‑gradient refinement (fast, many cheap steps)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] EG refinement phase")
        best_seq, best_score = _exp_gradient_descent(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            eta0=0.9,
            eval_every=40,
        )
        print(f"[{time.time():.1f}] after EG best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Projected Adam refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] Adam refinement phase")
        best_seq, best_score = _adam_optimize(
            best_seq.copy(),
            deadline,
            best_score,
            best_seq,
            eta=0.5,
        )
        print(f"[{time.time():.1f}] after Adam best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Greedy pairwise mass‑transfer directed by the active constraint
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] Greedy pairwise refinement")
        best_seq, best_score = _pairwise_greedy_opt(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            rng,
        )
        print(f"[{time.time():.1f}] after greedy best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Final annealing polish (as in the original pipeline)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing")
        best_seq, best_score = _pairwise_annealing(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            rng,
        )
        print(f"[{time.time():.1f}] after annealing best score {best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective = 2·n·max(conv)/sum²): {best_score:.6f} "
        f"(length={len(best_seq)})"
    )
    return best_seq.tolist()
