def _lp_refine(a0, deadline, best_score, best_seq, max_iter=10, t_initial=0.2):
    a = a0.copy()
    n = len(a)
    best_local_seq = a.copy()
    best_local_score = _objective(a)
    t = t_initial
    it=0
    while it < max_iter and time.time() < deadline:
        M = _fft_self_convolve(a).max()
        # Build Toeplitz matrix
        c = np.concatenate([a, np.zeros(n-1, dtype=np.float64)])
        r = np.concatenate([a[:1], np.zeros(n-1, dtype=np.float64)])
        A = toeplitz(c, r)  # shape (2n-1,n)
        # Solve LP
        # We'll use the "highs" method default.
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M), bounds=(0, None), method='highs')
        if not res.success:
            break
        b0 = res.x
        sum_b0 = b0.sum()
        if sum_b0 <= 0:
            break
        # Scale to sum 1
        b = b0 / sum_b0
        # Mix
        a = (1 - t) * a + t * b
        a /= a.sum()  # renorm
        # Evaluate
        cur_score = _objective(a)
        if cur_score < best_local_score - 1e-12:
            best_local_score = cur_score
            best_local_seq = a.copy()
            # maybe increase t slightly?
            t = min(0.5, t*1.1)
            print(f"[{time.time():.1f}] LP refine it={it} new best {cur_score:.6f}")
        else:
            t *= 0.9
        it += 1
    return best_local_seq, best_local_score

"""
Propose a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (∑a)²

The search consists of three stages:

1. **Rich analytic seeding** – families (uniform, arcsine, Hann,
   edge‑weighted, Beta, generalized‑normal, Dirichlet, sparse) are
   sampled for several lengths (512 … 2048).  The 5 best seeds are kept.

2. **Linear‑programming refinement** – for each seed we solve the LP
   “maximise ‖b‖₁ subject to a⋆b ≤ max(a⋆a)”.  The solution b is mixed
   with the current a (convex combination) and the process is iterated
   a few times (mirrors the classic alternating‑LP construction).  This
   dramatically flattens the auto‑convolution.

3. **Progressive up‑sampling + mirror‑gradient polishing** – the best
   vector is repeatedly doubled in length (linear interpolation) while
   time permits; after each up‑sample a short exponentiated‑gradient
   run (mirrored gradient on the exact objective) is applied.  A final
   simulated‑annealing pass moves small mass packets between bins.

All steps respect the 1000 s budget (2 s safety margin).  The routine
returns a plain Python list ready for ``evaluate_sequence`` (the values
are non‑negative and automatically normalised to sum = 1 inside the
objective).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (used for the LP refinement)
# ----------------------------------------------------------------------
try:

    from scipy.optimize import linprog
    _HAS_SCIPY = True
except Exception:  # pragma: no cover – SciPy might be unavailable
    _HAS_SCIPY = False

# Utility functions

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (real‑valued)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families (all return a *probability* vector, i.e. sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    w = np.where(
        (x == 0) | (x == 1),
        0.0,
        np.exp((alpha - 1) * np.log(x) + (alpha - 1) * np.log1p(-x))
    )
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.01) -> np.ndarray:
    """Sparse random vector with given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1‑λ) + uniform·λ."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Gradient of the maximal convolution entry (sub‑gradient)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))
    grad = np.zeros_like(a)
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
# Adaptive exponentiated‑gradient descent (mirrored gradient)
# ----------------------------------------------------------------------
def _adaptive_exp_gradient(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    beta: float = 0.9,
    eval_every: int = 50,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)          # EMA of gradients
    eta = eta0
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        m = beta * m + (1.0 - beta) * grad
        a = a * np.exp(-eta * m)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                eta = min(eta0, eta * 1.02)   # reward
            else:
                eta *= 0.995
        if t % (eval_every * 10) == 0:
            a = _project_onto_simplex(a, s=1.0)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise mass‑transfer simulated annealing (final polishing)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        src = rng.choice(np.where(a > 1e-14)[0])
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1
        delta = a[src] * rng.random() * 0.30
        if delta < 1e-16:
            continue
        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta
        cur = _objective(new_a)
        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.02
        accept = False
        if cur < best_score - 1e-15:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True
        if accept:
            a = new_a
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling while preserving total mass (=1)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new


# ----------------------------------------------------------------------
# Linear‑programming refinement (mirrors the classic alternating‑LP step)
# ----------------------------------------------------------------------
def _lp_refine(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_iter: int = 8,
    mix_alpha: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly:
        * solve   max 1ᵀb   s.t.   a⋆b ≤ max(a⋆a), b ≥ 0
        * scale b to unit‑sum
        * mix a ← (1‑α)a + αb   (α slowly adapted)
    Returns the best vector seen during the loop.
    """
    if not _HAS_SCIPY:
        # SciPy unavailable – fall back to the current best without LP.
        return a0, _objective(a0)

    a = a0.copy()
    n = a.shape[0]
    best_local_seq = a.copy()
    best_local_score = _objective(a)
    alpha = mix_alpha

    it = 0
    while it < max_iter and time.time() < deadline:
        # 1. current bound M
        M = _fft_self_convolve(a).max()

        # 2. build Toeplitz matrix for the convolution constraints
        col = np.concatenate([a, np.zeros(n - 1, dtype=np.float64)])
        row = np.concatenate([a[:1], np.zeros(n - 1, dtype=np.float64)])
        A = toeplitz(col, row)                 # shape (2n‑1, n)

        # 3. linear program – maximise 1ᵀb  ⇔ minimise –1ᵀb
        lp_res = linprog(
            c=-np.ones(n),
            A_ub=A,
            b_ub=np.full(2 * n - 1, M),
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": max(deadline - time.time(), 0.0)},
        )
        if not lp_res.success:
            # abort refinement – LP failed (often due to numerical issues)
            break

        b0 = lp_res.x
        s0 = b0.sum()
        if s0 <= 0.0:
            break
        b = b0 / s0                # scale to unit‑sum

        # 4. convex combination (mirrored update)
        a = (1.0 - alpha) * a + alpha * b
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _objective(a)
        if cur_score < best_local_score - 1e-12:
            best_local_score = cur_score
            best_local_seq = a.copy()
            alpha = min(0.9, alpha * 1.10)       # slowly increase step size on success
        else:
            alpha *= 0.90                        # decay on failure
        it += 1
    # Return the *globally* best version seen (may be older than final a)
    return best_local_seq, best_local_score
