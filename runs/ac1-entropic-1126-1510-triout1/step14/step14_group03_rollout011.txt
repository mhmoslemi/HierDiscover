                                eta0=0.8, gamma0=0.01, gamma_final=20.0,

        t += 1
        # schedule gamma

        gamma = gamma0 * (1 + 0.001 * t)  # e.g., increase linearly
        if gamma > gamma_final:

        # compute gradient via softmax weighting
        conv = _fft_self_convolve(a)
        # compute softmax
        maxc = conv.max()

        w = exp_vals / exp_vals.sum()
        # compute gradient via correlation
        # compute convolution of reversed a with w

        a_rev = a[::-1]

        A_rev = np.fft.rfft(a_rev, n=size)
        W = np.fft.rfft(w, n=size)
        conv_rev = np.fft.irfft(A_rev * W, n=size)
        start = len(a) - 1

        # update a with EG

"""
Propose a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.  

The search consists of three phases

1. **Diverse seed generation** – a pool of analytic families (uniform,
   arcsine, edge‑weighted with both positive and negative exponents,
   sum‑edge weighted, generalized normal, random sparse Dirichlet) is
   built for several lengths.  The 20 best seeds (lowest objective) are
   retained.

2. **Smoothed exponentiated‑gradient descent** – instead of using the
   sub‑gradient of the *single* maximal convolution entry we use a
   weighted combination of the gradients of the *M* largest convolution
   values (soft‑max weighting).  This yields a smoother descent direction
   and helps escape the plateau observed with the pure max‑gradient.
   The learning rate decays as 1/√t and a schedule gradually raises the
   soft‑max temperature γ.

3. **Pairwise mass‑transfer annealing** – a final inexpensive Monte‑Carlo
   polish that moves a fraction of mass between two bins and accepts
   uphill moves according to a linear cooling schedule.

All operations respect the constraints enforced by ``evaluate_sequence``
(non‑negative, clipping at 0…1000, sum > 0.01).  The whole routine stops
short of the allotted wall‑time (2 s safety margin).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  FFT‑based full linear convolution (real inputs)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
#  Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a))/sum(a)^2   (a must be non‑negative)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
#  Analytic families (return *probability* vectors – sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i·(n‑i))^{-p}   with i = distance to the left edge."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_sum_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ i^{-p} + (n‑i)^{-p} (sum of two one‑sided tails)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (-p)) + (j ** (-p))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp( -|x‑½|^{q} ) on the discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1‑lam) + uniform·lam."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
#  Gradient of a weighted combination of the top‑M convolution entries
# ----------------------------------------------------------------------
def _grad_top_softmax(a: np.ndarray, conv: np.ndarray,
                     gamma: float, M: int = 5) -> np.ndarray:
    """
    Compute a smooth sub‑gradient:
      g = Σ_{k∈topM} w_k * ∂conv_k/∂a,
      w_k ∝ exp(gamma * conv_k) (soft‑max over the top M entries).

    Complexity O(M·n).  For n up to ~20000 and M≈5 this is negligible.
    """
    n = a.shape[0]
    m = conv.shape[0]                      # = 2n‑1
    # pick the indices of the M largest convolution values
    if M >= m:
        top_idx = np.argsort(-conv)
    else:
        # argpartition gives unsorted top‑M; we sort them afterwards
        top_idx = np.argpartition(-conv, M - 1)[:M]
        top_idx = top_idx[np.argsort(-conv[top_idx])]
    top_vals = conv[top_idx]
    max_val = top_vals[0]                  # largest value (since sorted)
    # soft‑max weights, shifted for numerical stability
    exps = np.exp(gamma * (top_vals - max_val))
    w = exps / exps.sum()

    grad = np.zeros_like(a)
    for k, weight in zip(top_idx, w):
        if k < n:
            # overlap length = k+1, indices 0..k
            grad[: k + 1] += weight * (2.0 * a[k::-1])
        else:
            # k ≥ n : overlap length = 2·n‑1‑k
            start = k - (n - 1)
            grad[start:] += weight * (2.0 * a[start:][::-1])
    return grad


# ----------------------------------------------------------------------
#  Smoothed exponentiated‑gradient descent (multiplicative mirror)
# ----------------------------------------------------------------------
def _smooth_exp_gradient_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 eta0: float = 0.8,
                                 gamma0: float = 0.2,
                                 gamma_max: float = 10.0,
                                 M: int = 5,
                                 eval_every: int = 80) -> tuple[np.ndarray, float]:
    """
    Run EG descent where the gradient is a soft‑max combination of the top‑M
    convolution entries.  γ grows linearly from ``gamma0`` up to
    ``gamma_max``, making the descent gradually focus on the true maximum.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        # current convolution (full)
        conv = _fft_self_convolve(a)

        # temperature schedule for the soft‑max
        gamma = min(gamma_max, gamma0 * (1.0 + t * 0.0005))

        # smooth gradient
        grad = _grad_top_softmax(a, conv, gamma, M=M)

        # step size (standard EG schedule)
        eta = eta0 / math.sqrt(t)

        # multiplicative update – clamp exponent to avoid overflow
        update = -eta * grad
        np.clip(update, -30.0, 30.0, out=update)   # safeguards exp()
        a = a * np.exp(update)

        # re‑normalise to the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # occasional evaluation of the exact objective
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth‑EG it={t:,} γ={gamma:.2f} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Pairwise‑mass‑transfer annealing (final polish)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing style mass transfer between two bins."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move a random fraction (≤40 %) of the source mass
        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score

            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < best_score:
                best_score, best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Guard against an empty result
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _distribution_uniform(4096)
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Smoothed exponentiated‑gradient descent on the current best seed
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] start smoothed EG descent (budget {deadline - time.time():.1f}s)")
        best_seq, best_score = _smooth_exp_gradient_descent(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            eta0=0.9,
            gamma0=0.2,
            gamma_max=12.0,
            M=5,
            eval_every=60,
        )
        print(f"[{time.time():.1f}] after smoothed EG best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing")
        best_seq, best_score = _pairwise_annealing(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            rng,
        )
        print(f"[{time.time():.1f}] after annealing best score {best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective = 2·n·max(conv)/sum²): {best_score:.6f} "
        f"(length={len(best_seq)})"
    )
    # convert to plain Python list as required by the evaluator
    return best_seq.tolist()
