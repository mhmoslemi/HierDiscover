                         lr0=0.8, beta=0.9, eval_interval=200,

"""
Mirrored exponential‑gradient and sub‑gradient optimizer.

The algorithm proceeds in three stages:

1. **Diverse seeding** – a collection of analytic families (uniform,
   arcsine, beta, edge‑weighted, exponential, Gaussian, random Dirichlet,
   sparse) is sampled for several lengths (≈2 k … 8 k).  The best few
   candidates are retained.

2. **Warm‑up** – a smoothed log‑sum‑exp surrogate (as in the reference
   code) is minimised with an annealed exponentiated‑gradient (EG) method.
   This quickly drives the objective close to a good local basin.

3. **Refinement** – the exact objective `max(conv(a,a))` is tackled by a
   sub‑gradient EG method that uses the true maximising convolution index.
   After a (optional) up‑sampling the same sub‑gradient routine is run
   again.  Finally a deterministic pairwise‑swap sweep and a cheap
   simulated‑annealing clean‑up are executed.

All updates stay on the probability simplex (non‑negative, sum = 1) by
projecting with the Euclidean simplex projector.
The function returns a plain Python list; ``evaluate_sequence`` will take
care of the final clipping.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (real FFT, length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b (real FFT)."""
    n = a.shape[0]
    m = b.shape[0]
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a simplex vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_exponential(n: int, p: float) -> np.ndarray:
    """Symmetric exponential decay |i-center| with rate p."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    w = np.exp(-p * np.abs(i - center))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at the middle, variance sigma**2."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, where U is the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) ---------------------------------------
def _smooth_gradient(a: np.ndarray, lam: float) -> tuple[np.ndarray, float]:
    """
    Returns (grad, max_conv) where grad = ∇_a φ_λ(a) and
    max_conv = max(conv(a,a)).  a is assumed to sum to 1.
    """
    n = a.shape[0]

    # 1) self‑convolution
    conv = _fft_self_convolve(a)               # length 2n‑1
    max_c = conv.max()

    # 2) softmax weights (stable)
    z = lam * conv
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    s = exp_z / exp_z.sum()                    # length 2n‑1, sums to 1

    # 3) gradient = 2 * (a_rev * s) restricted to centre block
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, s)        # length 3n‑2
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]   # slice of length n

    return grad, max_c


# ----------------------------------------------------------------------
# Exact sub‑gradient for the true max ------------------------------------
def _subgradient_maxconv(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient of max(conv(a,a)) and the current max value.
    The sub‑gradient corresponds to the (average) gradient of all
    maximising positions.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_val = conv.max()
    max_idx = np.where(conv == max_val)[0]

    # weight vector for the active set (average)
    w = np.zeros_like(conv, dtype=np.float64)
    w[max_idx] = 1.0 / len(max_idx)

    a_rev = a[::-1]
    grad_full = _fft_convolve(a_rev, w)       # length 3n‑2
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]  # centre block
    return grad, max_val

# Annealed exponentiated‑gradient descent (smooth surrogate) -----------
def _smooth_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    lam_start_factor: float = 0.15,
    lam_inc: float = 1.0015,
    beta: float = 0.9,
    eval_interval: int = 200,
    patience: int = 2500,
) -> tuple[np.ndarray, float]:
    """Annealed EG descent on the log‑sum‑exp surrogate."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    lam = lam_start_factor * n
    rng = np.random.default_rng()

    while time.time() < deadline and t < 600_000:
        t += 1
        grad, max_conv = _smooth_gradient(a, lam)

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * momentum)
        a = np.maximum(a, 1e-30)
        a /= a.sum()

        lam *= lam_inc

        if t % eval_interval == 0:
            cur_score = 2.0 * n * max_conv
            if cur_score + 1e-14 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] smooth it={t:,}  new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

                best_score = cur_score
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] subgrad it={t:,}  new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            jitter = rng.exponential(scale=0.002, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Deterministic pairwise mass transfer sweep."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e}  new best {best_score:.6f}")

    return best_seq, best_score


def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Cheap simulated‑annealing clean‑up."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e}  best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Return a step‑function (list of non‑negative heights) with a low score."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Seed generation (moderate lengths, many analytic families)
    lengths = [2000, 3000, 4000, 5000, 6000, 8000]
    p_grid = np.concatenate(
        [
            np.logspace(-3, -1, 9),   # 0.001 … 0.1
            np.logspace(-1, 0, 9),    # 0.1 … 1
            np.logspace(0, 0.6, 7),   # 1 … 4
        ]
    )
    lam_grid = np.linspace(0.0, 0.9, 9)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating seeds for n={n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (base) {sc:.6f}")

        # parametrised families
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            edge2 = _distribution_edge_inversepow(n, p)
            sine = _distribution_sine_weighted(n, p)
            expo = _distribution_exponential(n, p)
            gauss = _distribution_gaussian(n, max(0.1, n * p * 0.02))  # rough sigma

            for base in (beta, edge, edge2, sine, expo, gauss):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(f"[{time.time():.1f}] new best (param) n={n} p={p:.3g} lam={lam:.2f} {sc:.6f}")

        # asymmetric beta
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = rng.uniform(0.001, 5.0)
            b = rng.uniform(0.001, 5.0)
            cand = _distribution_beta_asym(n, a, b)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best asym‑beta n={n} a={a:.3g} b={b:.3g} {sc:.6f}")

        # random Dirichlet
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_dirichlet(n, rng)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best Dirichlet n={n} {sc:.6f}")

        # sparse random
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best sparse n={n} {sc:.6f}")

        # keep only the elite seeds
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2. Include any previously discovered construction, if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never happen)
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Warm‑up smooth descent on the strongest seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]]
        per_seed = remaining * 0.5 / max(1, len(seeds))
        print(f"[{time.time():.1f}] warm‑up smooth descent on {len(seeds)} seeds, ~{per_seed:.1f}s each")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            a_opt, sc_opt = _smooth_descent(
                seed_seq.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                lam_start_factor=0.12,
                lam_inc=1.002,
                beta=0.9,
                eval_interval=150,
                patience=1800,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after warm‑up seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Exact sub‑gradient refinement (the crucial improvement stage)
    remaining = deadline - time.time()
    if remaining > 30.0:
        sub_dead = time.time() + remaining * 0.65
        print(f"[{time.time():.1f}] sub‑gradient refinement for {sub_dead - time.time():.1f}s")
        overall_best_seq, overall_best_score = _subgradient_descent(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.7,
            beta=0.9,
            eval_interval=100,
            patience=1500,
        )
        print(f"[{time.time():.1f}] after sub‑gradient new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Upsample to a larger length and refine again (if time permits)
    remaining = deadline - time.time()
    if remaining > 120.0:
        large_n = min(20000, len(overall_best_seq) * 2)
        print(f"[{time.time():.1f}] up‑sampling to {large_n}")
        up_seq = _upsample(overall_best_seq, large_n)

        sub_dead = time.time() + remaining * 0.6
        up_opt, up_sc = _subgradient_descent(
            up_seq,
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.6,
            beta=0.92,
            eval_interval=120,
            patience=2000,
        )
        if up_sc < overall_best_score:
            overall_best_score, overall_best_seq = up_sc, up_opt.copy()
            print(f"[{time.time():.1f}] up‑sampled refinement new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Deterministic pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] deterministic pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=500_000,
        )

    # ------------------------------------------------------------------
    # 8. Final cheap simulated‑annealing clean‑up
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
