def _optimize_smooth_max(a0, beta_schedule, time_limit, best_seq, best_score):
    x = np.log(a0 + 1e-12)  # logits
    for beta in beta_schedule:
        if time.time() >= time_limit:
            break
        # define objective functions closure capturing beta:
        def fun_obj(x_vec):
            return _smooth_obj_and_grad(x_vec, beta)  # returns (obj, grad)
        # Use SciPy's minimize
        # Use method='L-BFGS-B', jac=True
        # set options: maxiter maybe 100, ftol small.
        # initialization from previous x
        import scipy.optimize as opt
        res = opt.minimize(fun_obj, x, method='L-BFGS-B', jac=True,
                           options={'maxiter': 150, 'ftol': 1e-12})
        # update x = res.x
        x = res.x
        # compute a = softmax(x)
        a = np.exp(x)
        a = a / a.sum()
        # evaluate exact objective
        cur = _objective(a)
        if cur < best_score:
            best_score = cur
            best_seq = a.copy()
        # possibly print progress
    return best_seq, best_score

"""
Propose a step‑function (non‑negative simplex vector) that minimizes
evaluate_sequence (i.e. 2·n·max(conv(a,a))).

Algorithm
---------

1️⃣  **Fast seeding** – generate a modest collection of candidate
    distributions from analytic families (uniform, arcsine, sine,
    cosine, quadratic, beta, edge‑weighted, exponential‑edge).  Keep
    the few best candidates.

2️⃣  **Smooth‑max L‑BFGS optimisation** – for each promising seed,
    minimise a smooth surrogate of the ‑∞‑norm:
        M_β(a) = (1/β)·log∑_k exp(β·(a*a)_k) .
    The variable is the soft‑max (logits) of a so that the simplex
    constraint is satisfied automatically.  A schedule of β
    (0.5 → 200) is used; at each β an L‑BFGS‑B optimisation (SciPy)
    runs for a few hundred gradient evaluations.  Gradients are
    obtained analytically using FFT‑based convolutions.

3️⃣  **Exact sub‑gradient (exponentiated‑gradient) polishing** – a short
    run of the true L∞‑objective improves the solution further.

4️⃣  **Greedy pairwise mass transfer** – random moves of a tiny
    fraction of mass are accepted only if they strictly improve the
    exact objective.

All stages respect a global time budget (default 1000 s).  The best
vector found is returned as a plain list; it is already normalised
to sum = 1, so ``evaluate_sequence`` will apply its own scaling.
"""

import time
import math
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
#  FFT based convolution utilities (real inputs)
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a * a (real FFT)."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
#  Exact objective (sum(a)=1)  →  2·n·max(conv)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) for a with Σa = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
#  Analytic families of distributions (return probability vectors)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i+0.5)^{-p}·(n-i-0.5)^{-p}   (p can be positive or negative)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    """w_i ∝ exp(-c·i) + exp(-c·(n‑1‑i))  (c>0)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric beta‑type distribution on the discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n                                 # (0,1)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    total = w.sum()
    if total == 0.0:
        return _distribution_uniform(n)
    w /= total
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
#  Smooth‑max surrogate (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(logits: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """
    Returns (objective, gradient) for the smooth surrogate
    J_β(logits) = 2·n·M_β(a) ,  a = softmax(logits),
    where M_β(a) = (1/β)·log∑_k exp(β·(a*a)_k).
    Gradient is w.r.t. the logits (so the simplex constraint is
    automatically satisfied).
    """
    # softmax → probability vector on the simplex
    max_log = np.max(logits)
    exp_shifted = np.exp(logits - max_log)          # avoid overflow
    a = exp_shifted / exp_shifted.sum()             # Σa = 1, a≥0

    n = a.shape[0]

    # self‑convolution of a
    conv = _fft_self_convolve(a)                    # length 2n‑1
    max_c = conv.max()

    # smooth max via log‑sum‑exp (stable)
    shifted = conv - max_c                         # ≤0
    exp_beta = np.exp(beta * shifted)               # max term becomes 1
    sum_exp = exp_beta.sum()
    smooth_max = max_c + np.log(sum_exp) / beta

    obj = 2.0 * n * smooth_max

    # softmax weights w_k = exp(beta·shifted_k) / sum_exp
    w = exp_beta / sum_exp                         # length 2n‑1, Σw = 1

    # gradient of smooth max w.r.t. a:
    #   ∂M_β/∂a = 2·n·conv(w, rev(a))    (central part of convolution)
    a_rev = a[::-1]
    grad_conv = _fft_convolve(w, a_rev)           # length 3n‑2
    grad_a = 2.0 * n * grad_conv[n - 1:n - 1 + n]  # size n

    # chain rule for logits (softmax)
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)               # shape (n,)

    return obj, grad_logits


# ----------------------------------------------------------------------
#  Exact sub‑gradient descent (exponentiated‑gradient) on the true L∞ objective
# ----------------------------------------------------------------------
def _subgradient_descent_exact(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the exact L∞‐objective.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        # sub‑gradient (sum of shifted copies of a)
        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)          # EG keeps positivity
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score, best_seq = cur, a.copy()
                print(f"[{time.time():.1f}] exact subgrad it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
#  Greedy pairwise mass‑transfer hill‑climber
# ----------------------------------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly relocate a fraction of mass; accept only strict improvements."""
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score, best_seq = cur, a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")

    return best_seq, best_score

                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(exp_edge, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] exp c={c:.4g} lam={lam:.2f} len={n} score={sc:.6f}")

        # Dirichlet random seeds (small alphas promote edge mass) ---------------
        for _ in range(6):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 0.5)           # favour edge‑heavy draws
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} score={sc:.6f}")

        # keep only the best few seeds (memory & time)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2️⃣  Include previous construction if available (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / s
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3️⃣  Fallback (should never be needed)
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Smooth‑max L‑BFGS refinement on a handful of top seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # we will use at most 5 seeds for the expensive L‑BFGS stage
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0]
        print(f"[{time.time():.1f}] L‑BFGS refinement on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            # initialise logits from the seed (log‑softmax)
            logits = np.log(seed_seq + 1e-32)          # avoid log(0)
            # run through the β‑schedule
            for beta in beta_schedule:
                if time.time() >= deadline:
                    break
                # L‑BFGS‑B – a few hundred maxiterations are plenty for each β
                opt_res = _opt.minimize(
                    _smooth_obj_and_grad,
                    logits,
                    args=(beta,),
                    method='L-BFGS-B',
                    jac=True,
                    options={
                        'maxiter': 250,
                        'ftol': 1e-12,
                        'gtol': 1e-12,
                    },
                )
                logits = opt_res.x                     # use the result as start for next β

                # evaluate exact objective at the current point
                a_curr = np.exp(logits - np.max(logits))
                a_curr /= a_curr.sum()
                cur = _objective(a_curr)
                if cur < overall_best_score - 1e-14:
                    overall_best_score, overall_best_seq = cur, a_curr.copy()
                    print(f"[{time.time():.1f}] seed {idx}, β={beta:.1f} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Exact sub‑gradient (exponentiated‑gradient) refinement
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 25.0:
        print(f"[{time.time():.1f}] exact sub‑gradient refinement")
        overall_best_seq, overall_best_score = _subgradient_descent_exact(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.9,
            eval_every=60,
        )
        print(f"[{time.time():.1f}] after subgrad best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Final greedy pairwise polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy swap polishing")
        overall_best_seq, overall_best_score = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] after swaps best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best objective (2·n·max(conv)) = {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    # ``evaluate_sequence`` will re‑scale internally, so we can return the raw simplex‑normalised vector.
    return overall_best_seq.tolist()
