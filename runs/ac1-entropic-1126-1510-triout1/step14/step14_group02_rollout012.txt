``evaluate_sequence``.  The algorithm consists of the following stages:

1. **Diverse seed generation** – analytic families (uniform, arcsine,

   run a simplex‑preserving exponentiated‑gradient descent on the exact
   objective ``2·n·max(conv(a,a))`` (with ``sum(a)=1``).  The gradient is the
   sub‑gradient of the currently maximal convolution entries; a small
   momentum term (Adam‑like) smoothes the updates.  After each step we
   project onto the simplex and optionally enforce exact left‑right
   symmetry.

3. **Upsampling** – the current global best is linearly interpolated to the
   maximal allowed length (``18000``) and the descent is continued there
   for the remaining time budget.

   jitter, renormalise and restart the descent; this helps escape
   plateaus.

6. **Weighted pairwise refinement** – using the index of the maximal
   convolution entry we repeatedly move a small amount of mass from a
   “high‑impact’’ index to a “low‑impact’’ one whenever the objective
   improves.

7. **Cheap simulated‑annealing clean‑up** – a lightweight Metropolis‑type
   local search makes final adjustments.

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())   # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a with sum(a)=1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.006) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, where U is the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Force exact left‑right symmetry and re‑project onto the simplex."""
    sym = (seq + seq[::-1]) * 0.5
    return sym / sym.sum()

def _subgradient_descent_expg(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    active_frac: float = 0.02,
    max_active: int = 20,
    beta: float = 0.9,
    eval_interval: int = 50,
    patience: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient ascent on the exact (non‑smooth) objective.
    Active set = indices of convolution values within ``active_frac`` of
    the current maximum.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    ind_grid = np.arange(n, dtype=np.int32)

    while time.time() < deadline and t < 1_000_000:
        t += 1
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            # keep only the top max_active entries
            top_idx = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top_idx]

        grad = np.zeros_like(a)
        for k in act:
            shift = k - ind_grid
            mask = (shift >= 0) & (shift < n)
            # derivative of max(conv_k) w.r.t. a_i is 2 * a_{k-i}
            grad[mask] += 2.0 * a[shift[mask]]

        # momentum (Adam‑like, no bias‑correction for speed)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * momentum)

        # simplex projection (just renormalise, all entries stay ≥0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # enforce symmetry – many good solutions are symmetric
        a = _symmetrize(a)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} new global best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            # jitter restart
            jitter = rng.exponential(scale=0.005, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            a = _symmetrize(a)
            no_improve = 0

    return best_seq, best_score

def _pairwise_weighted_refine(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer guided by the location of the maximal convolution entry."""
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0

    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        conv = _fft_self_convolve(a)
        k_max = int(conv.argmax())

        # Impact values L_i = a_{k_max-i} where defined
        L = np.zeros(n, dtype=np.float64)
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        idx = np.arange(low, high + 1)
        L[idx] = a[k_max - idx]

        # source: large impact, target: small impact
        if L.sum() > 0:
            probs_i = L / L.sum()
            i = rng.choice(n, p=probs_i)
        else:
            positives = np.where(a > 1e-12)[0]
            if positives.size == 0:
                break
            i = rng.choice(positives)

        Lmax = L.max()
        probs_j = (Lmax - L) + 1e-12
        probs_j /= probs_j.sum()
        j = rng.choice(n, p=probs_j)
        if j == i:
            j = (j + 1) % n

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_a = _symmetrize(new_a)

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] weighted swap {i}->{j} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score

def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Lightweight Metropolis–type annealing."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_a = _symmetrize(new_a)

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq

        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
