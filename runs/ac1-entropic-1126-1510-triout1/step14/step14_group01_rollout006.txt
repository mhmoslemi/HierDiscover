def _lp_refine(a, deadline, best_seq, best_score):
    n = a.shape[0]
    while time.time() < deadline - 2:
        conv = _fft_self_convolve(a)
        M = conv.max()
        # Build constraint matrix
        rows = np.add.outer(np.arange(n), np.arange(n)).ravel()
        cols = np.tile(np.arange(n), n)
        data = np.repeat(a, n)
        A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        # Solve
        res = opt.linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
        if not res.success:
            print("LP failed:", res.message)
            break
        g0 = res.x
        sumg = g0.sum()
        if sumg <= 0:
            break
        g = g0 / sumg
        # line search for step size t
        improved = False
        for t in (0.1, 0.25, 0.5, 0.75, 1.0):
            cand = (1.0 - t) * a + t * g
            cand = np.maximum(cand, 0)
            cand /= cand.sum()
            cand_score = _objective(cand)
            if cand_score + 1e-12 < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand.copy()
                improved = True
                print(f"LP refine t={t:.2f} score {best_score:.6f}")
                break
        if not improved:
            # no improvement for any t, exit loop
            break
    return best_seq, best_score

"""
Multi‑stage optimizer for the step‑function bound.

1. Generate a diverse pool of analytic and random seeds (uniform,
   arcsine, symmetric/asymmetric Beta, edge‑weighted, sine‑weighted,
   Dirichlet, sparse).
2. Refine each seed with an annealed exponentiated‑gradient descent
   on the smooth log‑sum‑exp surrogate of the max‑convolution.
3. Starting from the current best sequence, run a *linear‑program
   refinement*: maximise the total mass of a new vector `g` while
   keeping the convolution `a * g` bounded by the current maximum.
   The LP solution is rescaled to unit sum and mixed with the current
   iterate (line‑search on the mixing weight).  This step typically
   pushes the sequence towards an extreme point of the feasible set
   and reduces the objective further.
4. If time permits, up‑sample the best sequence, run another round of
   gradient descent and a final LP refinement.
5. Finish with a deterministic pairwise‑mass‑transfer sweep and a cheap
   simulated‑annealing clean‑up.

The algorithm respects the 1000 s budget and returns a plain Python
list of non‑negative heights suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b (real FFT)."""
    n = a.shape[0]
    m = b.shape[0]
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a simplex vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric Beta(a,b)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp gradient -----------------------------------------
def _smooth_gradient(a: np.ndarray, lam: float) -> tuple[np.ndarray, float]:
    """
    Returns (grad, max_conv) where grad = ∇_a φ_λ(a) and
    max_conv = max(conv(a,a)).  a is assumed to sum to 1.
    """
    n = a.shape[0]

    # 1) self‑convolution
    conv = _fft_self_convolve(a)               # length 2n‑1
    max_c = conv.max()

    # 2) softmax weights (stable)
    z = lam * conv
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    s = exp_z / exp_z.sum()                    # length 2n‑1, sums to 1

    # 3) gradient = 2 * (a_rev * s) restricted to centre block
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, s)        # length 3n‑2
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]   # slice of length n

    return grad, max_c


def _smooth_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    lam_start_factor: float = 0.15,
    lam_inc: float = 1.0015,
    beta: float = 0.9,
    eval_interval: int = 200,
    patience: int = 2500,
) -> tuple[np.ndarray, float]:
    """
    Annealed exponentiated‑gradient descent using the smooth surrogate.
    Returns the best sequence and its (exact) score seen during the run.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    lam = lam_start_factor * n  # initial λ (smoothness)
    rng = np.random.default_rng()

    while time.time() < deadline and t < 600_000:
        t += 1
        grad, max_conv = _smooth_gradient(a, lam)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step‑size schedule
        eta = lr0 / math.sqrt(t)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * momentum)

        # keep inside simplex
        a = np.maximum(a, 1e-30)
        a /= a.sum()

        # anneal λ (sharpen the surrogate)
        lam *= lam_inc

        # periodic exact evaluation
        if t % eval_interval == 0:
            cur_score = 2.0 * n * max_conv
            if cur_score + 1e-14 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,}  new best {best_score:.6f}")
            else:
                no_improve += 1

        # plateaus → jitter
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise deterministic sweep -----------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Deterministic pairwise mass‑transfer sweep."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e}  new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Cheap simulated‑annealing clean‑up ------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Cheap simulated‑annealing clean‑up."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e}  best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear‑program refinement ---------------------------------------------
def _lp_refine(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    One‑step LP refinement:
    maximise Σ g_j  subject to  g ≥ 0  and  a * g ≤ M   (component‑wise),
    where M = max(conv(a,a)).  Rescale g to unit sum and mix with a.
    Repeats while improvements are observed.
    """
    n = a.shape[0]
    while time.time() < deadline - 2.0:
        # current bound
        M = _fft_self_convolve(a).max()

        # build the constraint matrix A (size (2n‑1) × n)
        rows = np.add.outer(np.arange(n), np.arange(n)).ravel()
        cols = np.tile(np.arange(n), n)
        data = np.repeat(a, n)
        A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

        # linear programme: maximise 1ᵀg  ⇔  minimise –1ᵀg
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        res = opt.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            print(f"[{time.time():.1f}] LP failed: {res.message}")
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break
        g = g0 / sum_g0  # unit‑sum feasible direction

        # try a few mixing ratios
        improved = False
        for t in (0.1, 0.25, 0.5, 0.75, 1.0):
            cand = (1.0 - t) * a + t * g
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            cand_score = _objective(cand)
            if cand_score + 1e-14 < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand.copy()
                improved = True
                print(f"[{time.time():.1f}] LP mix t={t:.2f}  new best {best_score:.6f}")
                break
        if not improved:
            # no progress → stop the LP loop
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling ------------------------------------------------------------
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine follows the multi‑stage plan described in the module docstring.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5.0   # keep a safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Build a modest pool of diverse seeds
    top_candidates: list[tuple[float, np.ndarray]] = []
    lengths = [1800, 2200, 2600, 3000]

    for n in lengths:
        if time.time() >= deadline:
            break
        # deterministic families
        for base in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(base)
            top_candidates.append((sc, base.copy()))

        # symmetric beta and edge‑weighted families (few p values)
        for p in (0.1, 0.3, 0.5, 0.8, 1.2, 1.6, 2.0, 2.5, 3.0):
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            sine = _distribution_sine_weighted(n, p)
            for cand in (beta, edge, sine):
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))

        # random seeds
        for _ in range(5):
            cand = _distribution_dirichlet(n, rng)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            cand = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))

    # keep elite seeds
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:12]

    # ------------------------------------------------------------------
    # 2️⃣  Include the previously discovered construction if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() < 1e-12:
                prev = np.full_like(prev, 1.0 / prev.size)
            prev = prev / prev.sum()
            sc = _objective(prev)
            top_candidates.append((sc, prev.copy()))
            if sc < top_candidates[0][0]:
                top_candidates[0] = (sc, prev.copy())
                top_candidates.sort(key=lambda x: x[0])
    except Exception:
        pass

    overall_best_seq = top_candidates[0][1].copy()
    overall_best_score = top_candidates[0][0]
    print(f"[{time.time():.1f}] initial best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Annealed exponentiated‑gradient refinement of elite seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        per_seed = remaining * 0.5 / max(1, len(top_candidates))
        for idx, (_, seed_seq) in enumerate(top_candidates):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            a_opt, sc_opt = _smooth_descent(
                seed_seq.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                lam_start_factor=0.12,
                lam_inc=1.002,
                beta=0.9,
                eval_interval=120,
                patience=1800,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after EG seed {idx} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Linear‑program refinement loop (core of the new improvement)
    while time.time() < deadline - 15.0:
        overall_best_seq, overall_best_score = _lp_refine(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        # break early if LP loop reports no more progress
        # (the function itself stops when stagnation occurs)

    # ------------------------------------------------------------------
    # 5️⃣  Optional up‑sampling and another round of refinement
    remaining = deadline - time.time()
    if remaining > 120.0:
        large_n = min(6000, overall_best_seq.shape[0] * 2)
        print(f"[{time.time():.1f}] up‑sampling to {large_n}")
        up_seq = _upsample(overall_best_seq, large_n)

        sub_dead = time.time() + remaining * 0.6
        up_opt, up_sc = _smooth_descent(
            up_seq,
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.6,
            lam_start_factor=0.15,
            lam_inc=1.0015,
            beta=0.92,
            eval_interval=120,
            patience=2000,
        )
        if up_sc < overall_best_score:
            overall_best_score, overall_best_seq = up_sc, up_opt.copy()
            print(f"[{time.time():.1f}] up‑sampled new best {overall_best_score:.6f}")

        # one more LP refinement at larger size (if time permits)
        if time.time() < deadline - 5.0:
            overall_best_seq, overall_best_score = _lp_refine(
                overall_best_seq.copy(),
                deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
            )

    # ------------------------------------------------------------------
    # 6️⃣  Deterministic pairwise sweep
    if time.time() < deadline:
        print(f"[{time.time():.1f}] deterministic pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 7️⃣  Final cheap simulated‑annealing clean‑up
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
