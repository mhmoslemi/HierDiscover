Improved search for a non‑negative step function minimizing
``evaluate_sequence``.

Strategy
--------
1. **Rich seed generation** – a large collection of analytic families
   (uniform, arcsine, symmetric / asymmetric Beta, edge‑weighted, inverse‑power,
   sine‑weighted, geometric, power‑law) together with random Dirichlet
   and sparse vectors.  Parameter grids are extended beyond the previous
   implementation (e.g. edge‑weight exponent up to 10, geometric ratio
   across a fine log‑grid).  The best ~15 seeds are retained.

2. **Two‑phase mirror descent** – each seed is first refined by a *smooth*
   exponential‑gradient descent that minimises a soft‑max approximation of the
   convolution peak (softmax temperature μ is annealed from 2 to 80).  The
   update uses the exact gradient of the smoothed objective, computed via
   FFT‑based cross‑correlation.  Afterwards a classic sub‑gradient descent
   (active‑set version) finishes the optimisation.

3. **Upsampling** – if time permits the current best sequence is linearly
   interpolated to twice the length and the two‑phase descent is repeated.
   This often discovers a lower constant factor while keeping the scale
  ‑invariant nature of the objective.

4. **Local polishing** – weighted pairwise mass transfers, a symmetric
   projection test, and a cheap simulated‑annealing sweep clean any remaining
   spikes.

All stages respect the global budget (default 1000 s) and report intermediate
best scores.  The returned list is a plain Python list of floats ready for
``evaluate_sequence``.

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())  # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a with sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·(n‑i))^{‑p} (symmetric heavy‑edge shape)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    """Weight ∝ i^{‑p}+ (n‑i)^{‑p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_geometric(n: int, r: float) -> np.ndarray:
    """Right‑hand geometric: a_i ∝ r**i, normalised."""
    i = np.arange(n, dtype=np.float64)
    w = r ** i
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_geometric_sym(n: int, r: float) -> np.ndarray:
    """Symmetric mixture of left‑ and right‑hand geometrics."""
    left = _distribution_geometric(n, r)
    right = left[::-1]
    w = (left + right) * 0.5
    w /= w.sum()
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+0.5)^{‑p} on the left side, mirrored to the right."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = i ** (-p)
    w = (w + w[::-1]) * 0.5
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.006) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, where U is the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Exact left‑right symmetry and re‑normalisation."""
    sym = (seq + seq[::-1]) * 0.5
    return sym / sym.sum()

# Smooth‑max gradient utilities -----------------------------------------
def _smooth_weights(conv: np.ndarray, mu: float) -> np.ndarray:
    """Soft‑max weights for a given temperature μ."""
    maxc = conv.max()
    # shift by maxc for numerical stability
    exps = np.exp(mu * (conv - maxc))
    w = exps / exps.sum()
    return w


def _grad_smoothmax(a: np.ndarray, mu: float) -> np.ndarray:
    """
    Gradient of the smooth‑max approximation of max(conv(a,a))
    with respect to a (assuming sum(a)=1).  Uses FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    w = _smooth_weights(conv, mu)               # length 2n‑1
    w_rev = w[::-1]

    # Full convolution a * w_rev (length 3n‑2)
    need = n + (2 * n - 1) - 1          # 3n‑2
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    B = rfft(w_rev, n=size)
    full = irfft(A * B, n=size)[:need]

    # gradient_j = 2 * full[(len(w)-1) - j]
    offset = len(w) - 1                # = 2n‑2
    grad = 2.0 * full[offset - np.arange(n)]
    return grad

# Mirror‑descent with smooth‑max ----------------------------------------
def _smooth_exponential_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    mu_start: float = 2.0,
    mu_end: float = 80.0,
    lr0: float = 0.8,
    eval_interval: int = 120,
    patience: int = 2500,
) -> tuple[np.ndarray, float]:
    """
    Two‑phase exponential‑gradient descent.
    * Early iterations use low μ (smooth approximation) to obtain a good
      direction; μ is increased linearly to μ_end so that the method
      gradually focuses on the true max.
    * Updates are a_i ← a_i·exp(‑η·grad) with η decreasing as 1/√t,
      followed by a simplex projection.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    beta = 0.9
    rng = np.random.default_rng()
    mu_step = (mu_end - mu_start) / max(1, (deadline - time.time()) * 0.5)  # crude schedule
    mu = mu_start

    while time.time() < deadline and t < 400_000:
        t += 1
        # Update temperature slowly (linear schedule)
        mu = min(mu_end, mu + mu_step)

        # Compute smooth gradient
        grad = _grad_smoothmax(a, mu)

        # Momentum (Adam‑like)
        momentum = beta * momentum + (1.0 - beta) * grad

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * momentum)

        # Simplex correction (non‑negative + normalisation)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} smooth‑descent new best {best_score:.6f} μ={mu:.2f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            # Jitter to escape a possible plateau
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

# Active‑set sub‑gradient descent (as in the previous version) -------
def _subgradient_descent_momentum(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    active_frac: float = 0.04,
    max_active: int = 30,
    beta: float = 0.9,
    eval_interval: int = 120,
    patience: int = 2000,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the exact max‑convolution objective."""
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    ind_grid = np.arange(n, dtype=np.int32)

    while time.time() < deadline and t < 500_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            top_idx = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top_idx]

        grad = np.zeros_like(a)
        for k in act:
            shift = k - ind_grid
            mask = (shift >= 0) & (shift < n)
            grad[mask] += 2.0 * a[shift[mask]]

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * momentum)

        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} sub‑grad new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            jitter = rng.exponential(scale=0.004, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

# Weighted pairwise refinement -------------------------------------------
def _pairwise_weighted_refine(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer guided by the location of the maximal convolution entry."""
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0

    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        conv = _fft_self_convolve(a)
        k_max = int(conv.argmax())

        # impact L_i = a_{k_max‑i} where defined
        L = np.zeros(n, dtype=np.float64)
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        idx = np.arange(low, high + 1)
        L[idx] = a[k_max - idx]

        # sample source i (high impact) and destination j (low impact)
        if L.sum() > 0:
            probs_i = L / L.sum()
            i = rng.choice(n, p=probs_i)
        else:
            positives = np.where(a > 1e-12)[0]
            if positives.size == 0:
                break
            i = rng.choice(positives)

        Lmax = L.max()
        probs_j = (Lmax - L) + 1e-12
        probs_j /= probs_j.sum()
        j = rng.choice(n, p=probs_j)
        if j == i:
            j = (j + 1) % n

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] weighted swap {i}->{j} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score

# Cheap simulated annealing ---------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Metropolis‑type local search."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

# Upsampling helper -------------------------------------------------------
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine follows the multi‑stage scheme described in the module doc‑string
    and respects the supplied time budget.
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Seed generation (moderate lengths)
    lengths = [2500, 3000, 3500, 4000, 5000, 6500]
    # extended parameter grids
    p_grid = np.concatenate(
        [
            np.logspace(-3, -1, 12),   # 0.001 … 0.1
            np.logspace(-1, 0, 12),   # 0.1 … 1
            np.logspace(0, 0.7, 15),  # 1 … 5.0
        ]
    )
    lam_grid = np.linspace(0.0, 0.95, 12)
    r_grid = np.logspace(-2, 0, 12)   # geometric ratios 0.01 … 1
    pow_grid = np.logspace(-2, 1, 12)  # power‑law exponent 0.01 … 10

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] sampling length {n}")

        # deterministic families (no parameters)
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (uniform/arcsine) {sc:.6f}")

        # parametric families ------------------------------------------------
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            edge2 = _distribution_edge_inversepow(n, p)
            sine = _distribution_sine_weighted(n, p)
            powlaw = _distribution_power_law(n, p)

            for base in (beta, edge, edge2, sine, powlaw):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # geometric families -------------------------------------------------
        for r in r_grid:
            if time.time() >= deadline:
                break
            geom = _distribution_geometric(n, r)
            geom_sym = _distribution_geometric_sym(n, r)
            for base in (geom, geom_sym):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new best geom len={n} r={r:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # asymmetric beta ----------------------------------------------------
        for _ in range(8):
            if time.time() >= deadline:
                break
            a = rng.uniform(0.001, 6.0)
            b = rng.uniform(0.001, 6.0)
            cand = _distribution_beta_asym(n, a, b)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new best asym‑beta len={n} a={a:.3g} b={b:.3g} score={sc:.6f}"
                )

        # random Dirichlet ---------------------------------------------------
        for _ in range(8):
            if time.time() >= deadline:
                break
            cand = _distribution_dirichlet(n, rng)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best Dirichlet len={n} score={sc:.6f}")

        # sparse random -------------------------------------------------------
        for _ in range(8):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.004)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best sparse len={n} score={sc:.6f}")

        # retain only the top seeds found so far
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:25]

    # ------------------------------------------------------------------
    # 2.  Incorporate any previously supplied construction (if available)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Guard against empty result
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Two‑phase descent on a selection of top seeds
    remaining = deadline - time.time()
    if remaining > 45.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:9]]
        per_seed_time = remaining * 0.55 / max(1, len(seeds))
        print(
            f"[{time.time():.1f}] heavy descent on {len(seeds)} seeds, ≈{per_seed_time:.1f}s each"
        )
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed_time * 0.45
            # smooth‑max phase
            a_mid, sc_mid = _smooth_exponential_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                mu_start=2.0,
                mu_end=70.0,
                lr0=0.9,
                eval_interval=80,
                patience=1500,
            )
            # exact max phase
            sub_deadline2 = time.time() + per_seed_time * 0.55
            a_opt, sc_opt = _subgradient_descent_momentum(
                a_mid,
                sub_deadline2,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.7,
                active_frac=0.06,
                max_active=35,
                beta=0.92,
                eval_interval=70,
                patience=1800,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] seed {idx} yielded new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Upsample and repeat refinement (if time permits)
    remaining = deadline - time.time()
    if remaining > 130.0:
        large_n = min(24000, len(overall_best_seq) * 2)
        print(f"[{time.time():.1f}] up‑sampling to {large_n} for final polishing")
        up_seq = _upsample(overall_best_seq, large_n)

        sub_deadline = time.time() + remaining * 0.55
        up_seq_mid, up_score_mid = _smooth_exponential_descent(
            up_seq,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            mu_start=2.0,
            mu_end=80.0,
            lr0=0.75,
            eval_interval=70,
            patience=1800,
        )
        sub_deadline2 = time.time() + remaining * 0.35
        up_seq_opt, up_score_opt = _subgradient_descent_momentum(
            up_seq_mid,
            sub_deadline2,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.6,
            active_frac=0.05,
            max_active=30,
            beta=0.9,
            eval_interval=60,
            patience=2000,
        )
        if up_score_opt < overall_best_score:
            overall_best_score, overall_best_seq = up_score_opt, up_seq_opt.copy()
            print(f"[{time.time():.1f}] up‑sampled descent new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Jitter‑restart loop (escape any plateaus)
    while time.time() < deadline - 15.0:
        jitter = rng.exponential(scale=0.02, size=overall_best_seq.shape)
        cand = overall_best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sub_deadline = time.time() + min(20.0, (deadline - time.time()) * 0.4)
        cand_mid, cand_mid_score = _smooth_exponential_descent(
            cand,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            mu_start=2.0,
            mu_end=70.0,
            lr0=0.6,
            eval_interval=70,
            patience=1500,
        )
        # a short exact‑max refinement
        sub_deadline2 = time.time() + min(12.0, (deadline - time.time()) * 0.3)
        cand_opt, cand_opt_score = _subgradient_descent_momentum(
            cand_mid,
            sub_deadline2,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.5,
            active_frac=0.07,
            max_active=30,
            beta=0.9,
            eval_interval=50,
            patience=1200,
        )
        if cand_opt_score < overall_best_score:
            overall_best_score, overall_best_seq = cand_opt_score, cand_opt.copy()
            print(f"[{time.time():.1f}] jitter‑restart improved to {overall_best_score:.6f}")
        else:
            # stop after a few unsuccessful attempts
            break

    # ------------------------------------------------------------------
    # 7.  Symmetrisation test
    sym_seq = _symmetrize(overall_best_seq)
    sym_score = _objective(sym_seq)
    if sym_score + 1e-14 < overall_best_score:
        overall_best_score, overall_best_seq = sym_score, sym_seq
        print(f"[{time.time():.1f}] symmetrised version better: {sym_score:.6f}")

    # ------------------------------------------------------------------
    # 8.  Weighted pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] weighted pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_weighted_refine(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=250_000,
        )

    # ------------------------------------------------------------------
    # 9.  Final cheap simulated‑annealing clean‑up
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.009,
            decay=0.9996,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
