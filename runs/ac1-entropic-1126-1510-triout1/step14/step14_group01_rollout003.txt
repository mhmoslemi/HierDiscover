def _subgradient_descent(a0, deadline, best_seq, best_score, lr0=0.5, beta=0.9, eval_interval=200, patience=2000):
    a = a0.copy()
    n = a.shape[0]
    t = 0
    momentum = np.zeros_like(a)
    while time.time() < deadline:
        # compute conv and max index
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idx = np.flatnonzero(conv == max_val)
        # pick random max index if multiple
        idx = max_idx[np.random.randint(len(max_idx))]
        # compute subgradient: conv_rev = a_rev * e_idx
        e = np.zeros(2*n-1)
        e[idx] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], e)[n-1:n-1+n]
        # momentum
        momentum = beta * momentum + (1-beta) * grad
        # step size decays
        lr = lr0 / math.sqrt(t+1)
        # update via exponentiated gradient (mirrored) or additive?
        # Use exponentiated to maintain positivity: a = a * np.exp(-lr*momentum)
        a = a * np.exp(-lr * momentum)
        a = np.maximum(a, 1e-30)
        a /= a.sum()
        t += 1
        if t % eval_interval == 0:
            cur_score = 2 * n * max_val
            if cur_score + 1e-14 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad it={t} new best {best_score:.6f}")
    return best_seq, best_score
```

"""
Propose a low‑value step‑function sequence.

The algorithm consists of three stages:

1. **Diverse seeding** – generate a large collection of candidate
   probability vectors from analytic families (uniform, arcsine,
   beta, edge‑weighted, sine‑weighted, sparse Dirichlet, …) for several
   lengths (2000–8000).  The exact objective  
   ``J(a)=2·n·max(conv(a,a))``  is evaluated (sum is normalised to 1)
   and the best few seeds are retained.

2. **Evolutionary refinement** – a small population (≈15) of the elite
   seeds is evolved by random convex combinations (crossover) and a
   tiny Gaussian mutation.  New offspring replace the worst individuals
   whenever they improve the exact objective.  This cheap global search
   quickly explores the space of mixtures.

3. **Local optimisation** – the current best candidate is refined by
   a *sub‑gradient* descent that works directly on the non‑smooth
   objective (no log‑sum‑exp surrogate).  The sub‑gradient of ``max(conv)``
   is obtained by averaging the gradients of all maximising lag indices.
   An exponentiated‑gradient update with Euclidean projection onto the
   simplex is used.  After the sub‑gradient phase a deterministic
   pairwise mass‑transfer sweep followed by a short simulated‑annealing
   clean‑up are performed.

All phases respect the supplied time budget (default 1000 s).  The
returned list is a probability vector (non‑negative, sums to 1); the
`evaluate_sequence` function will apply its own clipping before scoring.
"""

import time

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σ x = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using real FFT."""
    n = a.shape[0]
    m = b.shape[0]
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective J(a)=2·n·max(conv(a,a)) for a simplex vector."""
    n = a.shape[0]
    return 2.0 * n * _fft_self_convolve(a).max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.6,
    beta: float = 0.9,
    eval_interval: int = 200,
    patience: int = 2500,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact objective.
    Uses an exponentiated‑gradient step and Euclidean projection onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)
    t = 0
    no_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline and t < 800_000:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idx = np.flatnonzero(conv == max_val)

        # average sub‑gradient over all maximising indices
        e = np.zeros(2 * n - 1, dtype=np.float64)
        e[max_idx] = 1.0 / len(max_idx)
        grad = 2.0 * _fft_convolve(a[::-1], e)[n - 1 : n - 1 + n]

        # momentum (optional)
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size schedule
        eta = lr0 / math.sqrt(t)

        # exponentiated‑gradient update (mirror descent w.r.t. KL)
        a = a * np.exp(-eta * momentum)

        # safety and re‑normalisation
        a = np.maximum(a, 1e-30)
        a /= a.sum()

        # periodic exact evaluation
        if t % eval_interval == 0:
            cur_score = 2.0 * n * max_val
            if cur_score + 1e-14 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] subgrad it={t:,}  new best {best_score:.6f}")
            else:
                no_improve += 1

        # random jitter on long plateaus
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.004, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Cheap simulated‑annealing clean‑up."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e}  best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq

                    candidate_pool.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(f"[{time.time():.1f}] new best (param) n={n} p={p:.3g} lam={lam:.2f} {sc:.6f}")

        # asymmetric beta draws
        for _ in range(8):
            if time.time() >= deadline:
                break
            a = rng.uniform(0.001, 5.0)
            b = rng.uniform(0.001, 5.0)
            cand = _distribution_beta_asym(n, a, b)
            sc = _objective(cand)
            candidate_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best asym‑beta n={n} a={a:.3g} b={b:.3g} {sc:.6f}")

        # Dirichlet draws
        for _ in range(8):
            if time.time() >= deadline:
                break
            cand = _distribution_dirichlet(n, rng)
            sc = _objective(cand)
            candidate_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best Dirichlet n={n} {sc:.6f}")

        # Sparse random draws
        for _ in range(8):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.005)
            sc = _objective(cand)
            candidate_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best sparse n={n} {sc:.6f}")

    # keep only the elite seeds
    candidate_pool.sort(key=lambda x: x[0])
    candidate_pool = candidate_pool[:40]

    # ------------------------------------------------------------------
    # Stage 2 – evolutionary refinement (convex combinations)
    pop_size = min(15, len(candidate_pool))
    population = candidate_pool[:pop_size]                 # list of (score, vector)
    pop_vectors = [vec for _, vec in population]

    # evolutionary loop
    while time.time() < deadline - 30.0:                   # leave ~30 s for later phases
        # select two parents (biased towards better half)
        idx1 = rng.integers(pop_size // 2)
        idx2 = rng.integers(pop_size // 2)
        if idx1 == idx2:
            idx2 = (idx2 + 1) % (pop_size // 2)

        w = rng.random()
        child = w * pop_vectors[idx1] + (1.0 - w) * pop_vectors[idx2]

        # Gaussian mutation (tiny)
        child += rng.normal(scale=0.0015, size=child.shape)
        child = np.maximum(child, 0.0)
        child /= child.sum()

        child_score = _objective(child)

        # replace worst if better
        if child_score < population[-1][0]:
            population[-1] = (child_score, child.copy())
            pop_vectors[-1] = child.copy()
            # keep population sorted
            population.sort(key=lambda x: x[0])
            pop_vectors = [vec for _, vec in population]
            if child_score < overall_best_score:
                overall_best_score, overall_best_seq = child_score, child.copy()
                print(f"[{time.time():.1f}] EA new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – local sub‑gradient descent on the current best
    if overall_best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 15.0:
            sub_dead = time.time() + remaining - 5.0
            print(f"[{time.time():.1f}] sub‑gradient refinement (n={len(overall_best_seq)})")
            overall_best_seq, overall_best_score = _subgradient_descent(
                overall_best_seq.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.6,
                beta=0.9,
                eval_interval=120,
                patience=2000,
            )

    # ------------------------------------------------------------------
    # Optional up‑sampling if we still have slack
    if time.time() < deadline - 10.0 and overall_best_seq is not None:
        n_cur = len(overall_best_seq)
        new_len = min(18000, n_cur * 2)
        print(f"[{time.time():.1f}] up‑sampling to {new_len}")
        up_seq = _upsample(overall_best_seq, new_len)
        sub_dead = time.time() + (deadline - time.time()) * 0.6
        up_seq, up_score = _subgradient_descent(
            up_seq,
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.5,
            beta=0.9,
            eval_interval=100,
            patience=1800,
        )
        if up_score < overall_best_score:
            overall_best_score, overall_best_seq = up_score, up_seq.copy()
            print(f"[{time.time():.1f}] up‑sampled improvement {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # Deterministic pairwise refinement (always cheap)
    if overall_best_seq is not None and time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=250_000,
        )

    # Final simulated‑annealing clean‑up
    if overall_best_seq is not None and time.time() < deadline:
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return plain Python list; evaluate_sequence will perform final clipping.
    return overall_best_seq.tolist()
