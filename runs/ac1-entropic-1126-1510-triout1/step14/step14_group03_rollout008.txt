"""
Propose a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.

Algorithm outline
-----------------
1. **Seed generation** – a large, diverse pool of probability vectors is
   created from several analytic families (uniform, arcsine, edge‑weighted,
   sum‑edge‑weighted, generalized‑normal, triangular, cosine) together
   with random Dirichlet draws.  All seeds are normalised to sum = 1.

2. **Keep the best seeds** – after evaluating the objective, only the
   top‐K (default 30) are retained.

3. **Independent mirror‑descent refinements** – each of the top seeds is
   improved with a cheap sub‑gradient mirror‑descent (exponentiated‑gradient
   style) on the exact non‑smooth objective.  The best result among all
   refinements is kept as the global candidate.

4. **Final annealing polish** – a fast pairwise mass‑transfer simulated
   annealing is run on the global best for the remaining time.

All work respects the 1000 s wall‑time limit (2 s safety margin).  The
implementation is written with only the standard library and NumPy; no
closures or lambdas are used, and the function `propose_candidate` is the
entry point expected by the grader.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  FFT‑based self‑convolution (real inputs, linear convolution)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a*a using an FFT of length power‑of‑2."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
#  Exact objective used by the evaluator (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a))/sum(a)², a is a non‑negative vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:                     # identical to the security check
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
#  Analytic families – each returns a *probability* vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i·(n‑i))^{‑p}  with i measured from the left edge."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_sum_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ i^{‑p} + (n‑i)^{‑p}  (two one‑sided tails added)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (-p)) + (j ** (-p))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(‑|x‑½|^{q}) on the discretised interval."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    """Symmetric triangular shape, peak in the middle."""
    i = np.arange(n, dtype=np.float64) + 0.5
    center = n / 2.0
    w = 1.0 - np.abs(i - center) / center
    w = np.clip(w, 0, None)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    """Cosine bell: w_i ∝ cos(π·(i+0.5)/n) shifted to be non‑negative."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.cos(np.pi * (i / n - 0.5))
    w = np.clip(w, 0, None)
    w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination: (1‑lam)·base + lam·uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
#  Simple projection onto the probability simplex Σx_i = 1, x_i ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    # implementation of the algorithm from Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u - (cssv - 1) / (np.arange(len(v)) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Sub‑gradient of the *active* convolution entry (gradient of max)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """Return (gradient, index_of_max) for the current max entry of a*a."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))

    grad = np.zeros_like(a)
    if k < n:
        # overlap length = k+1   (indices 0 … k)
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        # k ≥ n : overlap length = 2·n‑1‑k   (indices start … n‑1)
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
#  Mirror‑descent (exponentiated‑gradient) on the exact objective
# ----------------------------------------------------------------------
def _mirror_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    eval_every: int = 30,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient mirror descent with KL‑divergence (multiplicative updates).
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)

        # multiplicative update, keeping positivity automatically
        a = a * np.exp(-eta * grad)

        # normalise back to the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] MD it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
#  Pairwise mass‑transfer annealing (final polish)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins; accept improvements."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    while time.time() < deadline:
        # pick a non‑zero source
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        # pick a distinct destination
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move at most 30 % of the source mass
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)

        # simple linear cooling schedule
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            # Metropolis criterion
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
#  Main entry point required by the grader
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Returns a plain ``list[float]`` ready for the evaluator.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # tiny safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")
    top: list[tuple[float, np.ndarray]] = []   # (score, vector)

    # ------------------------------------------------------------------
    # 1.  Generate a large and diverse pool of seeds
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 12000, 16000, 20000]
    # parameter grids
    p_grid = np.concatenate((
        np.logspace(-2.0, -0.5, 10, base=10),   # 0.01 … 0.316
        np.logspace(-0.5, 0.5, 10, base=10),    # 0.316 … 3.16
        np.logspace(0.5, 1.5, 8, base=10),      # 3.16 … 31.6
    ))
    lam_grid = np.linspace(0.0, 1.0, 9)          # 0, 0.125, …, 1
    q_grid = np.linspace(0.5, 3.0, 9)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] === scanning length {n}")

        # 1a) deterministic families
        for cand in (
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_triangular(n),
            _distribution_cosine(n),
        ):
            sc = _objective(cand)
            top.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (det) score={sc:.6f}")

        # 1b) edge‑weighted families + mixing with uniform
        for p in p_grid:
            if time.time() >= deadline:
                break
            base = _distribution_edge_weighted(n, p)
            sc = _objective(base)
            top.append((sc, base.copy()))
            if sc < best_score:
                best_score, best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new best edge p={p:.3g} score={sc:.6f}")

            base = _distribution_edge_sum_weighted(n, p)
            sc = _objective(base)
            top.append((sc, base.copy()))
            if sc < best_score:
                best_score, best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new best sum‑edge p={p:.3g} score={sc:.6f}")

            for lam in lam_grid:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                top.append((sc, cand.copy()))
                if sc < best_score:
                    best_score, best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new best mixed p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # 1c) generalized‑normal family
        for q in q_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_generalized_normal(n, q)
            sc = _objective(cand)
            top.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best gn q={q:.2f} score={sc:.6f}")

        # 1d) random Dirichlet (sparse & dense)
        for _ in range(6):
            if time.time() >= deadline:
                break
            # draw concentration parameter between 0.1 and 5.0
            alpha = rng.uniform(0.1, 5.0)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            top.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best dirichlet α={alpha:.2f} score={sc:.6f}")

        # keep only the *very* best seeds (size limited)
        top.sort(key=lambda x: x[0])
        top = top[:30]    # keep a modest number for the next phase

    # ------------------------------------------------------------------
    # 2.  Include the previously submitted construction if it exists
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() > 0.0:
                prev = prev / prev.sum()
                sc = _objective(prev)
                if sc < best_score:
                    best_score, best_seq = sc, prev.copy()
                    print(f"[{time.time():.1f}] previous best score={sc:.6f}")
                top.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Safeguard – fallback to uniform if nothing was generated
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _distribution_uniform(4000)
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Refine *each* top seed with a cheap mirror‑descent run
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        per_seed = max(1.0, remaining / max(1, len(top)))
        print(f"[{time.time():.1f}] Refinement phase – {len(top)} seeds, ~{per_seed:.1f}s each")
        for idx, (_, seed_vec) in enumerate(top):
            sub_deadline = time.time() + per_seed
            if time.time() >= deadline:
                break
            refined_seq, refined_score = _mirror_descent(
                seed_vec.copy(),
                sub_deadline,
                best_seq,
                best_score,
                eta0=0.9,
                eval_every=20,
            )
            if refined_score < best_score - 1e-14:
                best_score, best_seq = refined_score, refined_seq.copy()
                print(f"[{time.time():.1f}] seed {idx} gave new global best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final annealing polish on the current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] Final annealing polish")
        best_seq, best_score = _pairwise_annealing(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            rng,
        )
        print(f"[{time.time():.1f}] after annealing best score {best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective = 2·n·max(conv)/sum²): {best_score:.6f} "
        f"(length={len(best_seq)})"
    )
    # Return a plain Python list – the evaluator will apply its own security checks.
    return best_seq.tolist()
