def _exp_gradient_softmax(a0, deadline, best_seq, best_score,
                         eta0=0.5, beta=0.9, tau0=0.5, tau_max=30.0,
                         eval_every=30):
    a = np.copy(a0)
    n = a.shape[0]
    m = np.zeros_like(a)
    tau = tau0
    t = 0
    while time.time() < deadline:
        t += 1
        grad = _softmax_grad(a, tau)
        m = beta * m + (1 - beta) * grad
        a = a * np.exp(-eta0 * m)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full(n, 1.0/n)
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                # adapt step size: increase a bit

                # increase smoothing temperature (tau)
                tau = min(tau_max, tau * 1.05)

                eta0 *= 0.995

"""
Improved 1000‑second search for a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (∑a)²

Key ideas
---------

*   Many analytic families (uniform, arcsine, Hann, edge‑weighted, sum‑edge,
    exponential‑edge, β‑distribution, generalized‑normal, Dirichlet,
    sparse) are sampled for several lengths (256 … 4096).  The 30 best
    seeds are retained.

*   Mirror‑descent (exponentiated‑gradient) is performed on a *smooth*
    approximation of the max‑convolution using a soft‑max (log‑sum‑exp)
    with an annealing temperature τ.  The temperature is increased when an
    improvement is observed, allowing the algorithm to first smooth many
    peaks and later focus on the true maximum.

*   A momentum term (EMA of the gradient) and an adaptive stepsize give a
    fast, stable refinement.

*   The current best vector is progressively up‑sampled (doubling the
    length while keeping total mass = 1) and refined again.

*   A final pairwise mass‑transfer simulated annealing polishes the result.

All operations respect the 1000 s budget.  The returned list is ready for
``evaluate_sequence`` (non‑negative, scaling‑invariant).
"""

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used occasionally to avoid drift
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto the probability simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based self‑convolution (full length 2·n‑1)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective for a probability vector (∑a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families – each returns a *probability* vector (sum=1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^(-p) where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted_sum(n: int, p: float) -> np.ndarray:
    """Weight ∝ i^(-p) + j^(-p) (sum instead of product)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_exponential(n: int, gamma: float) -> np.ndarray:
    """Weight ∝ exp(-γ·d) where d = distance to the nearer edge."""
    d = np.minimum(np.arange(n, dtype=np.float64) + 0.5,
                   n - np.arange(n, dtype=np.float64) - 0.5)
    w = np.exp(-gamma * d)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(-|x-½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density sampled on cell mid‑points."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # avoid overflow/underflow for extreme α
    w = np.where(
        (x == 0) | (x == 1),
        0.0,
        np.exp((alpha - 1) * np.log(x) + (alpha - 1) * np.log1p(-x))
    )
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector (probability) with given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1‑λ) + uniform·λ."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smooth (soft‑max) gradient of the maximal convolution entry
# ----------------------------------------------------------------------
def _softmax_grad(a: np.ndarray, tau: float) -> np.ndarray:
    """
    Returns a sub‑gradient of the smooth maximum
    Mτ(a) = (1/τ)·log Σ_k exp(τ·b_k)  where b = a*a.
    The resulting gradient is 2·Σ_k w_k·a_{k‑i}  with w_k = softmax(τ·b_k).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)                               # length 2n‑1
    maxb = b.max()
    # stable soft‑max weights
    w = np.exp(tau * (b - maxb))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / (2 * n - 1))
    else:
        w /= w_sum
    # convolution of weights with reversed a
    rev_a = a[::-1]
    conv_wr = _fft_self_convolve(w)                         # convolve w with itself? No, need w * rev_a
    # more efficient: convolve w with rev_a via FFT
    size = 1 << ((len(w) + n - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    R = np.fft.rfft(rev_a, n=size)
    grad_full = np.fft.irfft(W * R, n=size)[: len(w) + n - 1]
    # extract the central n entries (indices n‑1 … n‑1+n‑1)
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    # numerical safety
    grad = np.maximum(grad, 0.0)
    return grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the smooth objective
# ----------------------------------------------------------------------
def _exp_gradient_softmax(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    beta: float = 0.9,
    tau0: float = 0.5,
    tau_max: float = 30.0,
    eval_every: int = 30,
) -> tuple[np.ndarray, float]:
    """
    Mirror descent on the smooth max‑convolution objective.
    Uses an EMA momentum (β) and an annealing temperature τ.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)          # EMA of gradients
    eta = eta0
    tau = tau0
    t = 0
    last_improve = time.time()

    while time.time() < deadline:
        t += 1
        grad = _softmax_grad(a, tau)
        m = beta * m + (1.0 - beta) * grad

        # multiplicative (exponentiated) update on the probability simplex
        a = a * np.exp(-eta * m)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)

        # periodic evaluation
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                eta = min(eta0, eta * 1.02)            # slight step‑size increase
                tau = min(tau_max, tau * 1.03)         # raise temperature → focus on true max
                last_improve = time.time()
                print(f"[{time.time():.1f}] GD it={t:,} new best {best_score:.6f} (τ={tau:.2f})")
            else:
                eta *= 0.995                         # gentle decay
                tau = max(tau0, tau * 0.999)         # slowly cool

        # occasional re‑projection to avoid drift
        if t % (eval_every * 10) == 0:
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise mass‑transfer simulated annealing (same as previous version)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        src = rng.choice(np.where(a > 1e-14)[0])
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1
        delta = a[src] * rng.random() * 0.3
        if delta < 1e-16:
            continue
        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta
        cur = _objective(new_a)
        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.02
        accept = False
        if cur < best_score - 1e-15:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True
        if accept:
            a = new_a
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal {src}->{dst} Δ={delta:.1e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling while preserving total mass (=1)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new

            cand = _distribution_beta(n, float(alpha))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_seq, overall_best_score = cand.copy(), sc
                print(f"[{time.time():.1f}] beta α={alpha:.3g} new best {sc:.6f}")

        # random Dirichlet seeds
        for _ in range(8):
            if time.time() >= deadline:
                break
            log_alpha = rng.uniform(-2.0, 1.0)          # α ∈ [0.01,10]
            alpha = 10 ** log_alpha
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_seq, overall_best_score = cand.copy(), sc
                print(f"[{time.time():.1f}] Dirichlet α={alpha:.3g} new best {sc:.6f}")

        # sparse random seeds
        for density in (0.01, 0.02):
            for _ in range(3):
                if time.time() >= deadline:
                    break
                cand = _distribution_random_sparse(n, rng, density)
                sc = _objective(cand)
                cand_pool.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_seq, overall_best_score = cand.copy(), sc
                    print(f"[{time.time():.1f}] sparse density={density} new best {sc:.6f}")

        # keep only the elite seeds (memory bounded)
        cand_pool.sort(key=lambda x: x[0])
        cand_pool = cand_pool[:30]

    # ------------------------------------------------------------------
    # 2️⃣  Quick adaptive GD on every elite seed (short runs)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0 and cand_pool:
        per_seed = max(0.1, remaining * 0.25 / max(len(cand_pool), 1))
        print(f"[{time.time():.1f}] short GD on {len(cand_pool)} seeds (~{per_seed:.2f}s each)")
        for idx, (sc, a0) in enumerate(cand_pool):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            a_opt, sc_opt = _exp_gradient_softmax(
                a0.copy(),
                sub_dead,
                overall_best_seq,
                overall_best_score,
                eta0=0.7,
                beta=0.9,
                tau0=0.5,
                eval_every=30,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
                print(f"[{time.time():.1f}] seed {idx} improved global best to {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Longer GD on current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_dead = time.time() + (deadline - time.time()) * 0.35
        a_opt, sc_opt = _exp_gradient_softmax(
            overall_best_seq.copy(),
            sub_dead,
            overall_best_seq,
            overall_best_score,
            eta0=0.5,
            beta=0.9,
            tau0=1.0,
            eval_every=60,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
            print(f"[{time.time():.1f}] long GD improved best to {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Progressive up‑sampling (doubling while time permits)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n > 65536:
            break
        print(f"[{time.time():.1f}] up‑sampling {cur_n} → {target_n}")
        up = _upsample(overall_best_seq, target_n)
        up_sc = _objective(up)
        if up_sc < overall_best_score:
            overall_best_seq, overall_best_score = up, up_sc
            print(f"[{time.time():.1f}] up‑sampled new best {overall_best_score:.6f}")

        # medium‑length GD after up‑sample
        sub_dead = time.time() + (deadline - time.time()) * 0.2
        a_opt, sc_opt = _exp_gradient_softmax(
            overall_best_seq.copy(),
            sub_dead,
            overall_best_seq,
            overall_best_score,
            eta0=0.45,
            beta=0.9,
            tau0=2.0,
            eval_every=60,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
            print(f"[{time.time():.1f}] GD after up‑sample new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final pairwise‑mass‑transfer annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing phase")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            overall_best_seq,
            overall_best_score,
            rng,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
            print(f"[{time.time():.1f}] after annealing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best objective = {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    # ``evaluate_sequence`` expects a plain Python list
    return overall_best_seq.tolist()
