    n = len(a)
    b = _fft_self_convolve(a)

    a_rev = a[::-1]

    # Use FFT:

"""
Propose a non‑negative step‑function (probability vector) that minimizes

    2·n·max(conv(a,a)) / (∑a)²

The algorithm is a refined version of the previous search:

1. **Rich seeding** – deterministic analytic families (uniform, arcsine, Hann,
   edge‑weighted, β‑distribution, generalized‑normal, sparse) are sampled
   for several lengths.  The 30 best seeds are retained.

2. **Smoothed exponentiated‑gradient polishing** – instead of using the
   sub‑gradient of the *max* convolution entry we minimise a smooth
   approximation
        g_α(a) = (1/α)·log∑_k exp(α·b_k)       (b = a∗a)
   whose gradient is a weighted sum of the individual lag‑gradients.
   An exponentiated‑gradient with EMA momentum, adaptive step size and a
   geometric schedule for α (α grows from ≈0.5 to ≈64) quickly drives the
   objective down.

3. **Progressive up‑sampling** – the current best vector is repeatedly
   up‑sampled (linear interpolation) up to ≈65 536 entries.
   After each up‑sample a medium‑length smoothed‑gradient run is performed.

4. **Final pairwise mass‑transfer annealing** – a simple Simulated Annealing
   that moves a random fraction of mass between two bins finishes the
   optimisation.

All phases respect a 1000 s budget (2 s safety margin).  The returned list
contains non‑negative values and is already normalised (∑a=1) – the outer
evaluation rescales it internally.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolutions
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a∗a (full length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Linear convolution u∗v via FFT (full length len(u)+len(v)-1)."""
    size = 1 << ((len(u) + len(v) - 1 - 1).bit_length())
    U = np.fft.rfft(u, n=size)
    V = np.fft.rfft(v, n=size)
    conv = np.fft.irfft(U * V, n=size)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
# Exact objective (probability vector: sum = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a∗a)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families (each returns a probability vector)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w[i] ∝ ( (i+½)·(n‑i‑½) )^{-p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(‑|x‑½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) sampled on cell mid‑points."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    w = np.where(
        (x == 0) | (x == 1),
        0.0,
        np.exp((alpha - 1) * np.log(x) + (alpha - 1) * np.log1p(-x))
    )
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1‑λ) + uniform·λ."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smoothed gradient of the max‑convolution (soft‑max over lags)
# ----------------------------------------------------------------------
def _grad_and_approx_max(a: np.ndarray, alpha: float) -> tuple[np.ndarray, float]:
    """
    Returns (grad, approx_max) where approx_max = (1/α)·log∑_k exp(α·b_k)
    with b = a∗a.
    Gradient = 2·(w ⋆ a_rev)_{n‑1..n‑1+n‑1}, where w is the soft‑max weight
    over lags.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)                # length 2·n‑1
    b_max = b.max()
    # soft‑max weights (stable)
    shift = alpha * (b - b_max)               # ≤ 0
    w_exp = np.exp(shift)
    w_sum = w_exp.sum()
    w = w_exp / w_sum                         # length 2·n‑1, sums to 1

    # convolution w * a_rev
    a_rev = a[::-1]
    conv = _fft_convolve(w, a_rev)            # length 3·n‑2
    start = n - 1
    grad = 2.0 * conv[start : start + n]

    # smoothed maximum value
    approx_max = b_max + np.log(w_sum) / alpha
    return grad, approx_max


# ----------------------------------------------------------------------
# Adaptive exponentiated‑gradient using the smoothed loss
# ----------------------------------------------------------------------
def _adaptive_smooth_exp_gradient(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    beta: float = 0.9,
    alpha_start: float = 0.5,
    alpha_end: float = 64.0,
    eval_every: int = 50,
) -> tuple[np.ndarray, float]:
    """
    Mirrors a gradient descent on the smoothed objective.
    Multiplicative updates keep the vector on the simplex.
    The temperature α grows geometrically from alpha_start to alpha_end.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)          # EMA of gradients
    eta = eta0
    t = 0
    alpha = alpha_start
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_and_approx_max(a, alpha)

        # EMA momentum
        m = beta * m + (1.0 - beta) * grad

        # multiplicative (mirror) update
        a = a * np.exp(-eta * m)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # increase α every few hundred steps
        if t % 200 == 0 and alpha < alpha_end:
            alpha = min(alpha_end, alpha * 2.0)

        # periodic exact evaluation
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-15:
                best_score, best_seq = cur, a.copy()
                # reward an improvement with a slightly larger stepsize
                eta = min(eta0, eta * 1.02)
                print(f"[{time.time():.1f}] GD it={t:,} new best {best_score:.6f}")
            else:
                # decay step size when no improvement
                eta *= 0.995

        # occasional re‑projection to avoid drift
        if t % (eval_every * 10) == 0:
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple pairwise mass‑transfer simulated annealing
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        # pick a source with non‑negligible mass
        src = rng.choice(np.where(a > 1e-14)[0])
        # pick a distinct destination uniformly
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # move a random fraction (up to 30 %) of src's mass
        delta = a[src] * rng.random() * 0.3
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        cur = _objective(new_a)

        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.02

        accept = False
        if cur < best_score - 1e-15:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-15:
                best_score, best_seq = cur, a.copy()
                print(
                    f"[{time.time():.1f}] anneal {src}->{dst} Δ={delta:.1e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass = 1)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new
