def _distribution_gaussian(n, sigma_frac):
    i = np.arange(n, dtype=np.float64)
    center = (n-1)/2.0
    sigma = sigma_frac * n
    vals = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    vals = np.clip(vals, 0.0, None)
    s = vals.sum()
    if s == 0:
       return _distribution_uniform(n)
    return vals / s
```

def _distribution_tent(n, width_frac):
    width = max(2, int(width_frac * n))  # half support
    center = (n-1)/2
    i = np.arange(n, dtype=np.float64)
    vals = np.maximum(0.0, 1 - np.abs(i - center) / width)
    s = vals.sum()
    if s == 0:
        return _distribution_uniform(n)
    return vals / s
```

def _distribution_two_spikes(n, pos_frac=0.25, spike_frac=0.5):

"""
Optimise a non‑negative step function (a list of heights) that minimises
the provided ``evaluate_sequence``.  The algorithm consists of five stages:

1️⃣  **Broad analytical seeding** – a large catalogue of parametric families
    (uniform, arcsine, beta‑edge, edge‑weighted, exponential‑edge,
    sine/cosine, quadratic, Gaussian, tent, two‑spike) is sampled on several
    lengths.  The best few candidates are retained.

2️⃣  **Soft‑max (smooth) exponentiated‑gradient descent** – the max‑convolution
    is approximated by a soft‑max with a temperature that is annealed from
    2·10⁻² to 10⁻³.  An exponentiated‑gradient (mirror‑descent on the simplex)
    step follows the exact smoothed sub‑gradient.

3️⃣  **Exact sub‑gradient polishing** – the true L∞ objective is tackled with
    an exponentiated‑gradient using the exact sub‑gradient (computed by
    FFT‑based cross‑correlation).  This sharpens the solution.

4️⃣  **LP‑based “mix‑with‑better‑distribution” step** – mirroring the classical
    iteration described in the problem statement, we solve a linear program
    (``scipy.optimize.linprog``) that, given the current distribution *f* and
    its current max‑convolution *M*, finds a non‑negative vector *g* maximising
    its total mass under the constraint ``f * g ≤ M``.  A convex combination
    ``(1‑t)·f + t·g`` (with ``t≈0.2``) is taken, and the process is repeated a
    few times.  The LP uses a sparse Toeplitz‐style constraint matrix, so it
    is feasible for lengths up to a few thousand.

5️⃣  **Greedy pairwise hill‑climber** – small random mass transfers are kept
    only if they strictly improve the exact objective.

All stages respect the overall 1000 s budget.  The final sequence is returned
as a plain list; ``evaluate_sequence`` will internally normalise it.
"""

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())  # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_cross_correlate_valid(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (valid part) (w ⋆ a)[j] = Σ_i w[i+j]·a[i],
    returning an array of length len(a).
    """
    n = a.shape[0]
    m = w.shape[0]                # = 2·n‑1
    size = 1 << ((m + n - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    A = np.fft.rfft(a, n=size)
    corr = np.fft.irfft(W * np.conj(A), n=size)
    return corr[:n]               # valid region


def _fft_sliding_window_sum(w: np.ndarray, win: int) -> np.ndarray:
    """
    For each j compute Σ_{i=j}^{j+win‑1} w[i] (window length = win).
    Returns an array of length win (i.e. len(a)). Implemented via FFT.
    """
    m = w.shape[0]
    size = 1 << ((m + win - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    O = np.fft.rfft(np.ones(win, dtype=np.float64), n=size)
    conv = np.fft.irfft(W * O, n=size)
    start = win - 1
    return conv[start:start + win]

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_edge(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta‑type density w_i ∝ (i·(n‑i))^{α‑1}, α>0."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = (i * (n - i)) ** (alpha - 1.0)
    vals = np.clip(vals, 0.0, None)
    s = vals.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    return vals / s


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    return w / w_sum


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Centered Gaussian (or “bell”) with σ = sigma_frac·n."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = max(sigma_frac * n, 1e-6)
    w = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    w = np.clip(w, 0.0, None)
    s = w.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    return w / s


def _distribution_tent(n: int, width_frac: float) -> np.ndarray:
    """Symmetric tent (triangular) shape with half‑width = width_frac·n."""
    width = max(1, int(width_frac * n))
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    w = np.maximum(0.0, 1.0 - np.abs(i - centre) / width)
    s = w.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    return w / s


def _distribution_two_spikes(n: int, pos_frac: float = 0.25, spike_frac: float = 0.5) -> np.ndarray:
    """Two symmetric spikes with total mass ``spike_frac``; rest uniform."""
    base = np.full(n, (1.0 - spike_frac) / n, dtype=np.float64)
    i1 = int(pos_frac * n)
    i2 = n - 1 - i1
    base[i1] = spike_frac / 2.0
    base[i2] = spike_frac / 2.0
    base = np.maximum(base, 0.0)
    return base / base.sum()


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.007
) -> np.ndarray:
    """Sparse Dirichlet‑like seed with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    return vals / s


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Exact objective for probability vectors (sum = 1) -> 2·n·max(conv)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth (soft‑max) exponentiated‑gradient descent
# ----------------------------------------------------------------------
def _smooth_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    tau_start: float = 0.02,
    tau_end: float = 0.001,
    lr0: float = 0.6,
    eval_every: int = 250,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a smoothed version of the L∞ objective.
    Temperature τ is linearly annealed from ``tau_start`` to ``tau_end``.
    """
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    it = 0

    while time.time() < time_limit:
        it += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] smooth it={it} new global best {best_score:.6f}")

        # anneal τ
        frac = (time.time() - start) / (time_limit - start)
        tau = tau_start * (1.0 - frac) + tau_end * frac

        # soft‑max weights (stable)
        shift = conv - max_val
        exp_vals = np.exp(shift / tau)          # ≤ 1
        w = exp_vals / exp_vals.sum()           # length 2n‑1, sums to 1

        # gradient of the smoothed objective
        term1 = _fft_cross_correlate_valid(w, a)   # Σ w[i+j]·a[i]
        win_sum = _fft_sliding_window_sum(w, n)   # Σ_{k=j}^{j+n‑1} w[k]
        grad = term1 + a * win_sum

        # EG update (step size ∝ 1/√t)
        eta = lr0 / math.sqrt(it)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if it % eval_every == 0:
            # already evaluated above
            pass

    return best_seq, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (mirror‑descent on the true L∞ objective)
# ----------------------------------------------------------------------
def _subgradient_descent_exact(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    while time.time() < time_limit:
        it += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(it)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if it % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] exact subgrad it={it} new global best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise hill‑climber (mass‑transfer swaps)
# ----------------------------------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# LP‑based improvement (mirroring the “mix with better distribution” step)
# ----------------------------------------------------------------------
def _lp_improve(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_iters: int = 20,
    t_step: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Given a probability vector a0, repeatedly solves the LP:
        maximise Σ g_j   s.t.  g ≥ 0  and  a * g ≤ M,
    where M = max(a * a).  The solution g is renormalised,
    convex‑combined with a, and the process repeats.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_local_score = best_score
    best_local_seq = best_seq.copy()
    it = 0

    # Pre‑create bounds for the LP (g ≥ 0)
    bounds = [(0.0, None)] * n

    while time.time() < time_limit and it < max_iters:
        # Build the Toeplitz‑style constraint matrix A (sparse)
        # A has one diagonal per offset d = 0,…,n‑1 with constant value a[d]
        offsets = np.arange(n)                # 0 … n‑1
        data = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
        A = sp.diags(data, offsets=offsets, shape=(2 * n - 1, n), format='csc')
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # Solve LP: maximise Σ g_j  <=> minimise −Σ g_j
        res = opt.linprog(
            c=-np.ones(n, dtype=np.float64),
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
        )
        if not res.success:
            # if the LP fails, abort the LP phase
            break

        g = res.x
        sum_g = g.sum()
        if sum_g == 0.0:
            g = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            g = g / sum_g

        # convex blend
        a = (1.0 - t_step) * a + t_step * g

        # recompute M and objective
        conv = _fft_self_convolve(a)
        M = conv.max()
        cur_score = 2.0 * n * M
        it += 1

        if cur_score < best_local_score - 1e-14:
            best_local_score = cur_score
            best_local_seq = a.copy()
            print(f"[{time.time():.1f}] LP it={it} new global best {best_local_score:.6f}")

    return best_local_seq, best_local_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` within the allotted time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0    # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, seq)

    # ------------------------------------------------------------------
    # 1️⃣  Analytic seeding on several lengths
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000, 16000]   # a modest spread covering up to 16k

    # parameter grids (kept reasonably sized for the time budget)
    p_grid = np.concatenate(
        [
            np.logspace(-3, -0.2, 12),   # edge‑weight exponent
            np.linspace(0.1, 2.0, 10),   # up to 2
        ]
    )
    alpha_grid = np.linspace(0.05, 1.5, 12)          # beta‑type
    lam_grid = np.linspace(0.0, 1.0, 13)            # mixture with uniform
    c_grid = np.logspace(-3, 0.5, 12)               # exponential‑edge decay
    sigma_grid = np.linspace(0.05, 0.4, 8)          # Gaussian width fraction
    width_grid = np.linspace(0.1, 0.8, 8)           # Tent half‑width fraction
    pos_grid = np.linspace(0.1, 0.4, 7)             # spike position fraction
    spike_grid = np.linspace(0.3, 0.7, 5)           # total spike mass

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # ----- deterministic families -------------------------------------------------
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
        ):
            cand = base_fn(n)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # ----- arcsine mixed with uniform ------------------------------------------------
        arcsine = _distribution_arcsine(n)
        for lam in lam_grid:
            cand = _mix_with_uniform(arcsine, lam)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] len={n} arcsine mix lam={lam:.2f} score={sc:.6f}")

        # ----- beta‑edge families -------------------------------------------------------
        for alpha in alpha_grid:
            beta = _distribution_beta_edge(n, float(alpha))
            for lam in lam_grid:
                cand = _mix_with_uniform(beta, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] len={n} β α={alpha:.3g} lam={lam:.2f} score={sc:.6f}")

        # ----- edge‑weighted families ---------------------------------------------------
        for p in p_grid:
            edge = _distribution_edge_weighted(n, float(p))
            for lam in lam_grid:
                cand = _mix_with_uniform(edge, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] len={n} edge p={p:.4g} lam={lam:.2f} score={sc:.6f}")

        # ----- exponential‑edge families -------------------------------------------------
        for c in c_grid:
            exp_edge = _distribution_exponential_edge(n, float(c))
            for lam in lam_grid:
                cand = _mix_with_uniform(exp_edge, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] len={n} exp c={c:.4g} lam={lam:.2f} score={sc:.6f}")

        # ----- Gaussian families ---------------------------------------------------------
        for sigma in sigma_grid:
            gau = _distribution_gaussian(n, float(sigma))
            for lam in lam_grid:
                cand = _mix_with_uniform(gau, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] len={n} gaussian σ={sigma:.3f} lam={lam:.2f} score={sc:.6f}")

        # ----- Tent families -------------------------------------------------------------
        for width in width_grid:
            tent = _distribution_tent(n, float(width))
            for lam in lam_grid:
                cand = _mix_with_uniform(tent, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] len={n} tent width={width:.3f} lam={lam:.2f} score={sc:.6f}")

        # ----- Two‑spike families --------------------------------------------------------
        for pos in pos_grid:
            for spike_mass in spike_grid:
                two = _distribution_two_spikes(n, pos_frac=float(pos), spike_frac=float(spike_mass))
                for lam in lam_grid:
                    cand = _mix_with_uniform(two, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(f"[{time.time():.1f}] len={n} spikes pos={pos:.2f} mass={spike_mass:.2f} lam={lam:.2f} score={sc:.6f}")

        # ----- random Dirichlet seeds ----------------------------------------------------
        for _ in range(8):
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} score={sc:.6f}")

        # ----- sparse random seeds -------------------------------------------------------
        for _ in range(6):
            a = _distribution_random_sparse(n, rng, density=0.005)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

        # keep only the most promising seeds globally (memory / time bound)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:80]

    # ------------------------------------------------------------------
    # 2️⃣  Incorporate any previously discovered construction (if available)
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3️⃣  Fallback (should never trigger)
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Smooth descent + exact sub‑gradient on the top seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # keep the five best seeds for deeper optimisation
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] start smooth descent on {len(seeds)} seeds (≈{per_seed:.1f}s each)")

        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break

            # --- smooth descent -------------------------------------------------
            smooth_deadline = time.time() + per_seed * 0.55
            a_opt, sc_opt = _smooth_descent(
                seed_seq,
                smooth_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                tau_start=0.03,
                tau_end=0.0008,
                lr0=0.7,
                eval_every=200,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after smooth seed {idx} new global best {overall_best_score:.6f}")

            # --- exact sub‑gradient polishing ------------------------------------
            exact_deadline = time.time() + per_seed * 0.30
            a_opt, sc_opt = _subgradient_descent_exact(
                overall_best_seq,
                exact_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                eval_every=200,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after exact polish seed {idx} new best {overall_best_score:.6f}")

            # --- LP‑based improvement (if time permits) --------------------------
            lp_deadline = time.time() + per_seed * 0.15
            if lp_deadline - time.time() > 5.0:   # need at least a few seconds
                a_opt, sc_opt = _lp_improve(
                    overall_best_seq,
                    lp_deadline,
                    best_seq=overall_best_seq,
                    best_score=overall_best_score,
                    max_iters=12,
                    t_step=0.2,
                )
                if sc_opt < overall_best_score - 1e-12:
                    overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                    print(f"[{time.time():.1f}] after LP improve seed {idx} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final greedy pairwise polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy swap polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=250000,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best objective (2·n·max(conv)) = {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    # Return a plain Python list – the evaluation environment will re‑normalise.
    return overall_best_seq.tolist()
