"""
Hybrid random‑seed + param‑sweep search for a non‑negative step function that
minimises ``evaluate_sequence``.

* Phase 1 – generate a diverse pool of analytic and random seeds (uniform,
  arcsine, beta (asymmetric), edge‑weighted, geometric, harmonic, cosine‑power,
  Dirichlet, sparse, mixtures).  Keep the best few.
* Phase 2 – refine every top seed with a temperature‑smoothed Adam optimiser,
  followed by exact exponentiated‑gradient descent on the true objective.
* Phase 3 – optionally up‑sample the current best vector and repeat the
  refinement (larger N gives a finer discretisation of the underlying density).
* Phase 4 – final pairwise‑annealing polish.

All phases respect the total 1000 s budget and always return a plain ``list``.
"""

import time
import math
import heapq
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities (used throughout)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation of a (len n) with weight vector w (len m) for non‑negative shifts.
    Returns length‑n array where entry i = Σ_j a_j·w_{j+i}.
    Implemented via FFT (convolution with reversed w).
    """
    n = a.shape[0]
    m = w.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()
    rev_w = w[::-1]
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_w, n=size)
    conv = np.fft.irfft(A * B, n=size)
    start = m - 1
    return conv[start:start + n]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a (unnormalised)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(alpha,beta) density on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0)
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{-p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w /= w.sum()
    return w


def _distribution_geometric(n: int, r: float) -> np.ndarray:
    """Decreasing geometric law a_i ∝ r^i, 0<r<1."""
    if r <= 0 or r >= 1:
        r = 0.9
    i = np.arange(n, dtype=np.float64)
    # normalisation factor for finite series
    denom = 1.0 - r ** n
    if denom == 0:
        denom = 1.0
    a = (1.0 - r) * r ** i / denom
    a /= a.sum()
    return a


def _distribution_symmetric_geometric(n: int, r: float) -> np.ndarray:
    """Symmetric version a_i ∝ r^{min(i, n-1-i)}."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    denom = 1.0 - r ** (n // 2 + 1)
    a = (1.0 - r) * r ** d / denom
    a /= a.sum()
    return a


def _distribution_harmonic(n: int) -> np.ndarray:
    """a_i ∝ 1/(i+1)."""
    i = np.arange(n, dtype=np.float64) + 1.0
    w = 1.0 / i
    w /= w.sum()
    return w


def _distribution_cosine_pow(n: int, p: float) -> np.ndarray:
    """a_i ∝ |cos(π·i/(n-1))|^p – a U‑shaped distribution."""
    i = np.arange(n, dtype=np.float64)
    # map i∈[0,n-1] → θ∈[0,π]
    theta = np.pi * i / (n - 1)
    w = np.abs(np.cos(theta)) ** p
    w += 1e-15          # avoid exact zeroes for numerical stability
    w /= w.sum()
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    """Monotone decreasing power‑law (p>0)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.01) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator, alpha: float) -> np.ndarray:
    """Dirichlet draw with identical concentration."""
    vec = rng.dirichlet(np.full(n, alpha))
    vec /= vec.sum()
    return vec


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Sub‑gradient of the (currently) maximal convolution entry
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """Return (sub‑gradient, active index) of max(conv)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))
    grad = np.zeros_like(a)
    # Contribution of a_i to conv_k is a_{k-i} + a_{i-k} when indices are in range
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent (mirror descent on the simplex)
# ----------------------------------------------------------------------
def _exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.9,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the exact objective."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the temperature‑smoothed max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    tau_start: float = 0.5,
    tau_end: float = 5e-4,
    tau_decay: float = 0.998,
    eval_every: int = 150,
    step_size0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Adam on the smoothed objective ``2·n·τ·log‑sum‑exp(conv/τ)``."""
    n = a0.shape[0]
    x = np.log(a0 + 1e-15)               # initialise logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    tau = tau_start
    while time.time() < deadline and tau > tau_end:
        t += 1
        # soft‑max over logits → probabilities
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # convolution
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        exp_vals = np.exp((conv - max_conv) / tau)
        w = exp_vals / exp_vals.sum()               # soft weights over shifts

        # gradient w.r.t. a (derivative of τ·log‑sum‑exp is w)
        corr = _fft_corr(a, w)                       # length‑n correlation
        grad_a = 4.0 * n * corr                     # factor 4 accounts for both
        # chain‑rule to logits (softmax Jacobian)
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = step_size0 / (np.sqrt(v_hat) + eps)
        x = x - lr * m_hat

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} τ={tau:.6f} new best {best_score:.6f}")

        tau = max(tau * tau_decay, tau_end)

    # final projection onto simplex (should already be normalised)
    if best_seq.sum() != 0:
        best_seq = best_seq / best_seq.sum()
    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise annealing polish (simulated‑annealing style)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Random mass transfer between two bins with temperature‑dependent acceptance."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    while time.time() < deadline:
        # pick source with non‑negligible mass
        src_idxs = np.where(a > 1e-12)[0]
        if src_idxs.size == 0:
            break
        i = rng.choice(src_idxs)

        # pick destination distinct from i
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # propose moving a fraction of i's mass to j
        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.02

        accept = cur < best_score - 1e-14 or (temperature > 0 and rng.random() < math.exp(-(cur - best_score) / temperature))
        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling whilst staying on the simplex
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Candidate generator (returns a list of probability vectors)
# ----------------------------------------------------------------------
def _generate_candidates_for_n(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diversified set of seeds for a given length n."""
    cand_list = []

    # deterministic families
    cand_list.append(_distribution_uniform(n))
    cand_list.append(_distribution_arcsine(n))
    cand_list.append(_distribution_hann(n))

    # Edge‑weighted (wide p‑range)
    for p in np.linspace(0.2, 2.5, 6):
        cand_list.append(_distribution_edge_weighted(n, p))

    # Symmetric Beta
    for p in np.logspace(-0.5, 0.5, 6):
        cand_list.append(_distribution_beta_sym(n, p))

    # Asymmetric Beta (random α,β)
    for _ in range(4):
        alpha = 10 ** rng.uniform(-1, 1)   # 0.1 … 10
        beta = 10 ** rng.uniform(-1, 1)
        cand_list.append(_distribution_beta_asym(n, alpha, beta))

    # Geometric families
    for r in rng.uniform(0.6, 0.99, 4):
        cand_list.append(_distribution_geometric(n, r))
        cand_list.append(_distribution_symmetric_geometric(n, r))

    # Harmonic
    cand_list.append(_distribution_harmonic(n))

    # Cosine‑power
    for p in (0.5, 1.0, 2.0, 3.0):
        cand_list.append(_distribution_cosine_pow(n, p))

    # Power‑law
    for p in (0.3, 0.6, 1.0, 1.5, 2.0):
        cand_list.append(_distribution_power_law(n, p))

    # Random Dirichlet draws
    for _ in range(4):
        alpha = rng.uniform(0.01, 2.0)
        cand_list.append(_distribution_dirichlet(n, rng, alpha))

    # Sparse random vectors
    for _ in range(4):
        cand_list.append(_distribution_random_sparse(n, rng, density=0.01))

    # Mixtures with uniform (helps smoothness)
    for lam in (0.2, 0.5, 0.8):
        for base in (cand_list[0], cand_list[1], cand_list[2]):  # uniform, arcsine, hann
            cand_list.append(_mix_with_uniform(base, lam))

    return cand_list

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

            # 1) Adam smoothing (smooth surrogate)

                seed_seq.copy(),
                sub_deadline,
                best_seq=seed_seq,
                best_score=_objective(seed_seq),
                tau_start=0.5,
                tau_end=5e-4,
                tau_decay=0.998,
                eval_every=120,
                step_size0=0.07,
            )
            # 2) Exact EG descent (use whatever time remains)

                a_opt, sc_opt = _exp_gradient_descent(
                    a_opt,
                    deadline,
                    a_opt,
                    sc_opt,
                    eta0=0.9,
                    eval_every=80,
                )
            # 3) Quick annealing polish

                a_opt, sc_opt = _pairwise_annealing(
                    a_opt,
                    deadline,
                    a_opt,
                    sc_opt,
                    rng,

            # keep global best

                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()

        # fallback if nothing improved
        if overall_best_seq is None:
            # just take the best from the pool
            overall_best_score, overall_best_seq = top_candidates[0]

    else:
        # insufficient time – just pick the current best

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

                    overall_best_seq.copy(),
                    sub_deadline,
                    overall_best_seq,
                    overall_best_score,
                    tau_start=0.3,
                    tau_end=3e-4,
                    tau_decay=0.997,
                    eval_every=120,
                    step_size0=0.06,
                )
                # EG fine‑tune
                if time.time() < deadline:
                    up_seq, up_score = _exp_gradient_descent(
                        up_seq,
                        deadline,
                        up_seq,
                        up_score,
                        eta0=0.85,
                        eval_every=80,
                    )
                # final anneal
                if time.time() < deadline:
                    up_seq, up_score = _pairwise_annealing(
                        up_seq,
                        deadline,
                        up_seq,
                        up_score,
                        rng,
                    )
                if up_score < overall_best_score:
                    overall_best_seq = up_seq
                    overall_best_score = up_score
                    print(f"[{time.time():.1f}] after up‑sample refinement new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final polishing (short annealing pass)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final polishing")
        overall_best_seq, overall_best_score = _pairwise_annealing(
            overall_best_seq,
            deadline,
            overall_best_seq,
            overall_best_score,
            rng,
        )

    # ------------------------------------------------------------------
    # Report final outcome
    # ------------------------------------------------------------------
    print(
        f"[{time.time():.1f}] Final best score = {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    # ensure plain python list of floats
    return overall_best_seq.tolist()
