Let's define function _solve_max_sum_lp(f, active_shifts, M):

    n = f.shape[0]
    m = len(active_shifts)
    A = np.zeros((m, n))
    for row, k in enumerate(active_shifts):
         start = max(0, k - (n - 1))
         end = min(n - 1, k)
         if start <= end:
               j = np.arange(start, end+1)
               A[row, start:end+1] = f[k - j]
    # Solve
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=np.full(m, M), bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x

"""
A fast, LP‑driven refinement of step‑functions.

The algorithm:
1.  Generate a diverse set of analytic “seed” distributions (uniform,
    arcsine, beta‑symmetric, edge‑weighted, Hann, generalized‑normal,
    power‑law, Dirichlet, sparse).  Keep the few best.
2.  For each top seed run an *active‑set linear‑program* iteration:
        •  compute the convolution f∗f and collect the K shifts with
           largest values (the “active” constraints);
        •  solve  max ∑g_j   s.t.  f∗g ≤ M on those K shifts,  g≥0.
        •  renormalise g to a probability vector and replace f by a
           convex combination (1‑t) f + t g, picking the best t from a
           short grid.  Repeat while the objective improves.
   This mass‑transfer step implements the Frank‑Wolfe style move described
   in the problem statement, but solves the sub‑problem exactly by a LP.
3.  Polish the best vector with the smooth Adam optimiser (temperature‑
   softened max) and a final exponentiated‑gradient / pairwise annealing
   pass.
4.  Always respect a total 1000 s budget and return a plain ``list[float]``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# FFT utilities (full linear convolution)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a using a real FFT (full linear conv)."""
    n = a.shape[0]
    # size: next power‑of‑two ≥ 2*n‑1
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation of a (len n) with w (len m) for non‑negative shifts.
    Returns an array of length n where element i equals Σ_j a_j · w_{j+i}.
    Implemented via FFT using convolution with the reversed weight vector.
    """
    n = a.shape[0]
    m = w.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()
    rev_w = w[::-1]
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_w, n=size)
    conv = np.fft.irfft(A * B, n=size)
    start = m - 1
    return conv[start:start + n]


# ----------------------------------------------------------------------
# Objective (exact evaluation – scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for (possibly un‑normalised) a."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic families – always return a *probability* vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Linear‑program sub‑problem (mass‑transfer)
# ----------------------------------------------------------------------
def _solve_max_sum_lp(f: np.ndarray, active_shifts: np.ndarray, M: float):
    """
    Solve   max Σ g_j   s.t.   (f ∗ g)[k] ≤ M for every k in active_shifts,
    with g ≥ 0.  Returns the optimal g (un‑normalised) or None on failure.
    """
    n = f.shape[0]
    m = len(active_shifts)
    if m == 0:
        return None

    # Build A where A[row, j] = f[k - j]  (zero when out of range)
    A = np.zeros((m, n), dtype=np.float64)
    for row, k in enumerate(active_shifts):
        start = max(0, k - (n - 1))
        end = min(n - 1, k)
        if start <= end:
            j = np.arange(start, end + 1, dtype=np.int64)
            A[row, start:end + 1] = f[k - j]

    # Linear programme: maximise sum(g)  →  minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n
    res = linprog(
        c,
        A_ub=A,
        b_ub=np.full(m, M, dtype=np.float64),
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Adam optimiser for a temperature‑smoothed surrogate of the max‑conv
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    tau_start: float = 0.2,
    tau_end: float = 0.001,
    tau_decay: float = 0.999,
    eval_every: int = 200,
    step_size0: float = 0.05,
) -> tuple[np.ndarray, float]:
    """Adam on the smoothed objective ``2·n·τ·log‑sum‑exp(conv/τ)``."""
    n = a0.shape[0]
    x = np.log(a0 + 1e-15)               # logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    tau = tau_start
    while time.time() < deadline and tau > tau_end:
        t += 1
        # probabilities from logits
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # convolution and τ‑softmax weights
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        exp_vals = np.exp((conv - max_conv) / tau)
        w = exp_vals / exp_vals.sum()

        # grad w.r.t. a (4·n·cross‑correlation)
        corr = _fft_corr(a, w)                 # length n
        grad_a = 4.0 * n * corr

        # chain rule to logits
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = step_size0 / (np.sqrt(v_hat) + eps)
        x = x - lr * m_hat

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} τ={tau:.6f} new best {best_score:.6f}")

        # cool the temperature
        tau = max(tau * tau_decay, tau_end)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise‑annealing polishing (simple simulated annealing)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins; accept improving
    moves or early‑stage worsening moves with a temperature‑dependent probability."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main search entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the 1000 s budget and
    returns a plain ``list[float]`` ready for evaluation.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # 1. generate a pool of analytic seeds
    # ------------------------------------------------------------------
    lengths = [2048, 4096, 8192]        # sizes we can afford
    all_candidates: list[tuple[float, np.ndarray]] = []

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for fn in (_distribution_uniform, _distribution_arcsine, _distribution_hann):
            seq = fn(n)
            score = _objective(seq)
            all_candidates.append((score, seq.copy()))
            print(f"[{time.time():.1f}]  {fn.__name__:20s} score={score:.6f}")

        # beta‑ and edge‑weighted families
        for p in np.logspace(-0.5, 0.5, 5):
            if time.time() >= deadline:
                break
            for base in (_distribution_beta_sym(n, p), _distribution_edge_weighted(n, p)):
                for lam in (0.0, 0.3, 0.6):
                    seq = _mix_with_uniform(base, lam)
                    score = _objective(seq)
                    all_candidates.append((score, seq.copy()))
        # generalized normal
        for q in (0.5, 1.0, 1.5, 2.0):
            seq = _distribution_generalized_normal(n, q)
            all_candidates.append((_objective(seq), seq.copy()))
        # power‑law
        for pw in (0.3, 0.6, 1.0, 1.5, 2.0):
            seq = _distribution_power_law(n, pw)
            all_candidates.append((_objective(seq), seq.copy()))
        # random Dirichlet
        for _ in range(4):
            alpha = rng.uniform(0.01, 2.0)
            seq = rng.dirichlet(np.full(n, alpha))
            all_candidates.append((_objective(seq), seq.copy()))
        # sparse random seeds
        for _ in range(4):
            seq = _distribution_random_sparse(n, rng, density=0.02)
            all_candidates.append((_objective(seq), seq.copy()))

    # keep the best few for the LP stage
    all_candidates.sort(key=lambda x: x[0])
    top_candidates = all_candidates[:5]  # very few – focus our time

    # ------------------------------------------------------------------
    # 2. LP‑driven active‑set refinement for each top seed
    # ------------------------------------------------------------------
    eps = 1e-12
    K_active = 10               # number of active convolution entries used in the LP
    for idx, (init_score, seq) in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] LP refinement start for seed {idx} (score={init_score:.6f})")
        a = np.maximum(seq, eps)
        a = a / a.sum()
        best_local = a.copy()
        best_local_score = _objective(a)

        it = 0
        while time.time() < deadline and it < 4000:
            it += 1
            conv = _fft_self_convolve(a)
            M = conv.max()
            # select K indices with the largest convolution values
            if K_active < conv.size:
                # use argpartition for speed
                active_idx = np.argpartition(conv, -K_active)[-K_active:]
            else:
                active_idx = np.arange(conv.size)
            # solve LP
            g = _solve_max_sum_lp(a, active_idx, M)
            if g is None:
                break
            sum_g = g.sum()
            if sum_g <= 0.0:
                g_norm = _distribution_uniform(a.shape[0])
            else:
                g_norm = g / sum_g

            # try a small set of convex‑mixing factors
            improved = False
            for t in (0.1, 0.3, 0.5, 0.7, 1.0):
                mixed = (1.0 - t) * a + t * g_norm
                mixed = np.maximum(mixed, eps)
                mixed = mixed / mixed.sum()
                sc = _objective(mixed)
                if sc < best_local_score - 1e-14:
                    best_local_score = sc
                    best_local = mixed.copy()
                    improved = True
            if improved:
                a = best_local.copy()
                print(f"[{time.time():.1f}]  LP it={it} seed={idx} new score {best_local_score:.6f}")
            else:
                break
        # keep global best
        if best_local_score < init_score - 1e-14:
            top_candidates[idx] = (best_local_score, best_local.copy())

    # ------------------------------------------------------------------
    # 3. pick the best among refined seeds
    # ------------------------------------------------------------------
    top_candidates.sort(key=lambda x: x[0])
    best_seq, best_score = top_candidates[0]
    print(f"[{time.time():.1f}] after LP best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. final Adam smoothing (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print(f"[{time.time():.1f}] final Adam smoothing")
        best_seq, best_score = _adam_optimize(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            tau_start=0.2,
            tau_end=0.001,
            tau_decay=0.998,
            eval_every=150,
            step_size0=0.04,
        )

    # ------------------------------------------------------------------
    # 5. final pairwise annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing polish")
        best_seq, best_score = _pairwise_annealing(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            rng,
            max_trials=None,
        )

    print(
        f"[{time.time():.1f}] final best score (objective=2·n·max(conv)/sum²): "
        f"{best_score:.6f} (length={len(best_seq)})"
    )
    return best_seq.tolist()
