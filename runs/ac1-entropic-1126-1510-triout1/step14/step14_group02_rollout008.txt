```
def compute_smooth_obj_and_grad(a, alpha):
    n = a.shape[0]
    conv = _fft_self_convolve(a)   # length 2n-1
    # stable compute softmax
    max_c = conv.max()
    shifted = conv - max_c   # for numerical stability
    exp_alpha = np.exp(alpha * shifted)
    sum_exp = exp_alpha.sum()
    logsumexp = max_c + np.log(sum_exp) / alpha
    # objective = 2 * n * logsumexp
    obj = 2.0 * n * logsumexp
    # weight w_k = exp(alpha * conv) / sum_exp_alpha
    w = exp_alpha / sum_exp
    # compute gradient: grad = 2 * cross_corr(w, a)
    # using FFT
    # pad sizes: need length for convolution of length 2n-1 and n => length = 2n-1 + n -1 = 3n-2
    need = 3*n - 2
    size = 1 << ((need-1).bit_length())
    W = np.fft.rfft(w, n=size)
    a_rev = a[::-1]
    A_rev = np.fft.rfft(a_rev, n=size)
    conv_w_a = np.fft.irfft(W * A_rev, n=size)
    # conv_w_a length size; slice to get length n:
    grad = 2.0 * conv_w_a[n-1:n-1+n]  # as derived
    return obj, grad
```

"""
Propose a non‑negative step function (list of heights) that minimizes
``evaluate_sequence``.  The routine works in three stages:

1. **Seed generation** – a rich collection of analytic families
   (uniform, arcsine, semicircle, beta‑sym, edge‑weighted, edge‑exp,
   sine‑weighted) and random Dirichlet vectors of moderate size
   (≈ 3‑5 k).  The ~10 best seeds (according to the exact non‑smooth
   objective) are retained.

2. **Smooth‑gradient refinement** – for each seed we run an Adam
   optimiser on a *smooth* surrogate of the objective:
   ``log‑sum‑exp`` of the convolution (parameterised by a temperature
   ``α``).  The temperature is increased during the run so that the
   surrogate gradually turns into the original ``max``‐convolution
   objective.  After each update we project back onto the probability
   simplex.  The best solution found over all seeds (and the true
   objective evaluated periodically) is kept.

3. **Final discrete cleanup** – a cheap weighted pairwise mass‐transfer
   refinement followed by a short simulated‑annealing pass are applied
   to the current best sequence.  All steps respect a global time
   budget (default 1000 s).  The returned object is a plain Python list
   of floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = v.astype(np.float64, copy=True)
    # fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based linear convolution (full length 2·n‑1)
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())   # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
# Exact (non‑smooth) objective for probability vectors (sum = 1)
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a with sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_semicircle(n: int) -> np.ndarray:
    """Wigner semicircle on a discrete grid."""
    x = np.linspace(-1.0, 1.0, n, dtype=np.float64)
    w = np.sqrt(np.maximum(0.0, 1.0 - x * x))
    w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Beta(p,p) on [0,1] (symmetrised)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^(-p) with i,j measured from centre."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w /= w.sum()
    return w


def _distribution_edge_exponential(n: int, p: float) -> np.ndarray:
    """Weight ∝ exp(-p·i)+exp(-p·j)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = np.exp(-p * i) + np.exp(-p * j)
    w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet with a single concentration parameter."""
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.006) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, where U is the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Make the vector exactly symmetric and re‑normalise."""
    sym = (seq + seq[::-1]) * 0.5
    sym = np.maximum(sym, 0.0)
    if sym.sum() == 0.0:
        return _distribution_uniform(seq.shape[0])
    return sym / sym.sum()


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient -----------------------
def _smooth_objective_and_grad(a: np.ndarray, alpha: float):
    """
    Returns
    -------
    f   : float  –  2·n·log‑sum‑exp_{α}(conv(a,a))
    g   : ndarray – gradient of f w.r.t. a (sum(a)=1 not enforced)
    The gradient is
        g_i = 2 * Σ_k w_k * a_{k-i},
    where w_k = softmax_{α}(conv_k).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)               # length 2n‑1
    # Numerically stable log‑sum‑exp
    max_c = conv.max()
    shifted = conv - max_c                     # avoid overflow
    exp_alpha = np.exp(alpha * shifted)
    sum_exp = exp_alpha.sum()
    logsumexp = max_c + np.log(sum_exp) / alpha
    f = 2.0 * n * logsumexp

    # softmax weights
    w = exp_alpha / sum_exp                    # length 2n‑1

    # gradient = 2 * cross‑correlation(w, a)
    # compute via FFT: corr(w, a) = conv(w, a_rev) shifted by (n‑1)
    need = 3 * n - 2
    size = 1 << ((need - 1).bit_length())
    W = rfft(w, n=size)
    A_rev = rfft(a[::-1], n=size)
    corr = irfft(W * A_rev, n=size)           # length size
    grad = 2.0 * corr[n - 1:n - 1 + n]        # slice to length n
    return f, grad


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate --------------------------------
def _adam_smooth_opt(a0: np.ndarray,
                     deadline: float,
                     best_seq: np.ndarray,
                     best_score: float,
                     lr: float = 0.4,
                     alpha_start: float = 5.0,
                     alpha_end: float = 200.0,
                     eval_interval: int = 120,
                     patience: int = 2000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth surrogate.  ``alpha`` (temperature)
    is increased linearly from ``alpha_start`` to ``alpha_end``.
    The true non‑smooth objective is evaluated every ``eval_interval``
    steps; the best sequence seen is stored in ``best_seq``.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t_iter = 0
    no_improve = 0
    rng = np.random.default_rng()
    while time.time() < deadline and t_iter < 300_000:
        t_iter += 1
        # schedule temperature
        alpha = alpha_start + (alpha_end - alpha_start) * min(t_iter / 80_000, 1.0)

        f, grad = _smooth_objective_and_grad(a, alpha)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v / (1 - beta2 ** t_iter)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection onto simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)

        # occasional evaluation of the true objective
        if t_iter % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] Adam it={t_iter:,} new global best {best_score:.6f}")
            else:
                no_improve += 1

        # if stuck for a while, jitter a little to escape
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.006, size=n)
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Weighted pairwise deterministic refinement -------------------------------
def _pairwise_weighted_refine(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer guided by the current maximal convolution index."""
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        conv = _fft_self_convolve(a)
        k_max = int(conv.argmax())

        # Impact L_i = a_{k_max-i}
        L = np.zeros(n, dtype=np.float64)
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx = np.arange(lo, hi + 1)
        L[idx] = a[k_max - idx]

        # choose donor i (large impact) and receiver j (small impact)
        if L.sum() > 0.0:
            probs_i = L / L.sum()
            i = rng.choice(n, p=probs_i)
        else:
            i = rng.integers(n)

        Lmax = L.max()
        probs_j = (Lmax - L) + 1e-12
        probs_j /= probs_j.sum()
        j = rng.choice(n, p=probs_j)
        if j == i:
            j = (j + 1) % n

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] weighted swap {i}->{j} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple Metropolis‑type simulated annealing -------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Lightweight local search that accepts occasional uphill moves."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (simple interpolation)
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linearly interpolate a simplex vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 0.0:
        new_seq /= s
    else:
        new_seq = np.full(new_len, 1.0 / new_len, dtype=np.float64)
    return new_seq

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Seed generation (moderate lengths)
    lengths = [2500, 3000, 3500, 4000, 4500, 5000]
    # parameter grids
    p_grid = np.concatenate([
        np.logspace(-3, -1, 9),      # 0.001 … 0.1
        np.logspace(-1, 0, 9),      # 0.1 … 1
        np.logspace(0, 0.5, 7),     # 1 … 3.16
    ])
    lam_grid = np.linspace(0.0, 0.9, 10)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating seeds for length {n}")

        # deterministic families
        for base in (_distribution_uniform(n),
                     _distribution_arcsine(n),
                     _distribution_semicircle(n)):
            sc = _objective(base)
            top_candidates.append((sc, base.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new global best (deterministic) {sc:.6f}")

        # parameterised families
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            edge_exp = _distribution_edge_exponential(n, p)
            sine = _distribution_sine_weighted(n, p)

            for base in (beta, edge, edge_exp, sine):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(f"[{time.time():.1f}] new best (param) len={n} p={p:.3g} lam={lam:.2f} {sc:.6f}")

        # asymmetric beta for extra flexibility
        for _ in range(6):
            if time.time() >= deadline:
                break
            a_par = rng.uniform(0.001, 3.0)
            b_par = rng.uniform(0.001, 3.0)
            cand = _distribution_beta_asym(n, a_par, b_par)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best (asym‑beta) len={n} a={a_par:.3g} b={b_par:.3g} {sc:.6f}")

        # random Dirichlet seeds
        for _ in range(8):
            if time.time() >= deadline:
                break
            cand = _distribution_dirichlet(n, rng)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best (Dirichlet) len={n} {sc:.6f}")

        # sparse random seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best (sparse) len={n} {sc:.6f}")

        # keep only the 25 best seeds overall (across all lengths so far)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:25]

    # ------------------------------------------------------------------
    # 2. incorporate any previously supplied construction (if present)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best improved score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. fallback guard
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Adam‑based smooth optimisation on the few top seeds
    remaining = deadline - time.time()
    if remaining > 40.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:7]]
        per_seed_time = remaining * 0.55 / max(1, len(seeds))
        print(f"[{time.time():.1f}] Adam optimisation on {len(seeds)} seeds, ≈{per_seed_time:.1f}s each")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed_time
            a_opt, sc_opt = _adam_smooth_opt(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr=0.4,
                alpha_start=5.0,
                alpha_end=200.0,
                eval_interval=150,
                patience=1800,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after Adam seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Upsample to larger size if time permits (helps flatten further)
    remaining = deadline - time.time()
    if remaining > 120.0:
        large_n = min(18000, len(overall_best_seq) * 2)
        print(f"[{time.time():.1f}] up‑sampling to {large_n} points for final refinement")
        up_seq = _upsample(overall_best_seq, large_n)

        sub_deadline = time.time() + remaining * 0.55
        up_seq_opt, up_score = _adam_smooth_opt(
            up_seq,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr=0.3,
            alpha_start=5.0,
            alpha_end=200.0,
            eval_interval=120,
            patience=2000,
        )
        if up_score < overall_best_score:
            overall_best_score, overall_best_seq = up_score, up_seq_opt.copy()
            print(f"[{time.time():.1f}] up‑sampled descent gave new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Jitter‑restart cycles (escape possible plateaus)
    while time.time() < deadline - 15.0:
        jitter = rng.exponential(scale=0.018, size=overall_best_seq.shape)
        cand = overall_best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sub_deadline = time.time() + min(22.0, (deadline - time.time()) * 0.5)
        cand_opt, cand_score = _adam_smooth_opt(
            cand,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr=0.35,
            alpha_start=5.0,
            alpha_end=200.0,
            eval_interval=100,
            patience=1500,
        )
        if cand_score < overall_best_score:
            overall_best_score, overall_best_seq = cand_score, cand_opt.copy()
            print(f"[{time.time():.1f}] jitter‑restart improvement to {overall_best_score:.6f}")
        else:
            # stop after a few unsuccessful attempts
            break

    # ------------------------------------------------------------------
    # 7. Symmetrisation test
    sym_seq = _symmetrize(overall_best_seq)
    sym_score = _objective(sym_seq)
    if sym_score + 1e-14 < overall_best_score:
        overall_best_score, overall_best_seq = sym_score, sym_seq
        print(f"[{time.time():.1f}] symmetrised version better: {sym_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Weighted pairwise deterministic refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] weighted pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_weighted_refine(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=200_000,
        )

    # ------------------------------------------------------------------
    # 9. Final cheap simulated‑annealing clean‑up
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
