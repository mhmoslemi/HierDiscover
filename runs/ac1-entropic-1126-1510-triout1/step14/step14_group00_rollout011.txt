def build_A_ub(a: np.ndarray) -> csr_matrix:
    n = a.shape[0]
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        # indices from j_start to j_end inclusive
        num = j_end - j_start + 1
        rows.extend([k] * num)
        cols.extend(list(range(j_start, j_end + 1)))
        data.extend(a[j_start:j_end+1].tolist())
    return csr_matrix((data, (rows, cols)), shape=(2*n - 1, n))

    sumg = g.sum()

def propose_candidate(seed=42, budget_s=1000):

    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix

    random = np.random.default_rng(seed)

    start = time.time()
    deadline = start + budget_s - 1.0

    # Stage 1: choose initial length and candidate from families.
    # We'll test lengths [1024, 2048] for seeds.
    best_score = float('inf')
    best_a = None

    for n in [1024, 2048]:
        # generate candidates
        candidates = []
        # uniform
        candidates.append(_distribution_uniform(n))
        # arcsine
        candidates.append(_distribution_arcsine(n))
        # hann
        candidates.append(_distribution_hann(n))
        # edge-weighted exponents
        for p in np.linspace(0.2, 2.5, 12):
            candidates.append(_distribution_edge_weighted(n, p))
        # beta
        for alpha in np.concatenate((np.logspace(-1.5, -0.2, 6), np.linspace(0.2, 3.0, 6))):
            candidates.append(_distribution_beta(n, alpha))
        # generalized normal
        for q in np.linspace(0.5, 3.0, 6):
            candidates.append(_distribution_generalized_normal(n, q))
        # random sparse
        for _ in range(5):
            candidates.append(_distribution_random_sparse(n, random, density=0.02))

        # evaluate
        for a in candidates:
            sc = _objective(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        # break early if time near end
        if time.time() > deadline - 10:
            break

    # Ensure have best
    if best_a is None:
        best_a = _distribution_uniform(1024)
        best_score = _objective(best_a)

    # Stage 2: LP mixing refinement
    # Choose step-size schedule t decreasing from 0.3 to 0.05
    a = best_a
    n = a.shape[0]

    while time.time() < deadline - 5:

        A_ub = _build_convolution_ub_matrix(a)
        b_ub = np.full(2 * n - 1, M)
        c = -np.ones(n)
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        except Exception as e:
            print("LP exception:", e)
            break
        if not res.success:
            print("LP not successful:", res.message)
            break
        g = res.x
        sumg = g.sum()
        if sumg == 0:
            g = _distribution_uniform(n)
        else:
            g = g / sumg
        # mixing
        a = (1 - t) * a + t * g

        a = _project_onto_simplex(a, s=1.0)
        # recompute score
        sc = _objective(a)

"""
candidate proposer – a multi‑phase optimiser for the
2·n·max(conv(a,a)) / (∑a)² objective.

Phase 1 – diverse analytic seeds (uniform / arcsine / Hann /
edge‑weighted / β / general‑Gaussian / sparse).  The best seed (probability
vector, i.e. ∑a = 1) is kept.

Phase 2 – linear‑program “mix” step:  given a, solve

    max   Σ g_j
    s.t.  (a * g)_k ≤ M   ∀k,   g ≥ 0

where M = max(a*a).  The solution g is normalised to a probability
vector and mixed with a: a ← (1‑t)·a + t·g.  A few iterations (t≈0.2)
drive the maximal self‑convolution down.

Phase 3 – progressive up‑sampling (×2) while a short
exponentiated‑gradient (mirrored‑GD) polish is applied after each up‑sample.

Phase 4 – final pairwise‑mass‑transfer simulated annealing for a
fine‑grained polish.

All work respects the supplied time‑budget (≈1000 s).  The returned list
contains non‑negative numbers; the objective is invariant under scaling,
so the vectors are kept on the simplex (∑=1).
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# Basic utilities

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution via FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """Exact objective assuming Σa = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    w /= w.sum()
    return w


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^(-p) where i,j are distances to the edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(-|x-½|^q) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) sampled on cell mid‑points."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # avoid overflow/underflow for extreme α
    w = np.where(
        (x == 0) | (x == 1),
        0.0,
        np.exp((alpha - 1) * np.log(x) + (alpha - 1) * np.log1p(-x))
    )
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.02
) -> np.ndarray:
    """Sparse random vector with given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals

# Gradient of the maximal convolution entry (sub‑gradient)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Returns (grad, k) where k is the index attaining max conv.
    grad_i = 2·a_{k‑i} for i that participates, otherwise 0.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))
    grad = np.zeros_like(a)
    if k < n:
        # overlap indices 0 … k
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        # overlap indices k‑(n‑1) … n‑1
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k

def _adaptive_exp_gradient(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    beta: float = 0.9,
    eval_every: int = 50,
) -> tuple[np.ndarray, float]:
    """
    Mirrors gradient descent on the exact objective.
    Uses an EMA of the sub‑gradient (momentum) and a simple adaptive
    stepsize that grows on improvement and decays otherwise.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)          # EMA of gradients
    eta = eta0
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)

        a = a * np.exp(-eta * m)

            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()

            else:

# Pairwise mass‑transfer simulated annealing
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        src = rng.choice(np.where(a > 1e-14)[0])
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1
        delta = a[src] * rng.random() * 0.3   # up‑to 30 % move
        if delta < 1e-16:
            continue
        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        cur = _objective(new_a)

        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.02

        accept = False
        if cur < best_score - 1e-15:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal {src}->{dst} Δ={delta:.1e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling while preserving total mass (=1)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new

def _build_convolution_ub_matrix(a: np.ndarray) -> csr_matrix:

    rows, cols, data = [], [], []

        rows.extend([k] * length)
        cols.extend(list(range(j_start, j_end + 1)))
        data.extend(a[j_start:j_end + 1].tolist())
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

        g_sum = g.sum()
        if g_sum == 0.0:
            g = _distribution_uniform(n)
        else:
            g = g / g_sum

        # convex combination (still on the simplex)
        a = (1.0 - t_mix) * a + t_mix * g
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _objective(a)
        if cur_score < best_score - 1e-12:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time():.1f}] LP mix {lp_iter} new best {best_score:.6f}")
            t_mix = min(0.4, t_mix * 1.05)   # slightly increase step if we improve
        else:
            t_mix *= 0.90                     # otherwise decay
        # stop if mixing weight becomes tiny
        if t_mix < 0.015:
            break

    # --------------------------------------------------------------
    # 3️⃣  Progressive up‑sampling + exponentiated‑gradient polishing
    # --------------------------------------------------------------
    while time.time() < deadline - 2:
        cur_n = a.shape[0]
        if cur_n >= 32768:
            break
        target_n = cur_n * 2
        print(f"[{time.time():.1f}] up‑sampling {cur_n} → {target_n}")
        a = _upsample(a, target_n)

        # a medium‑length EG polishing run
        sub_dead = time.time() + min(30.0, deadline - time.time() - 1.0)
        a, _ = _adaptive_exp_gradient(
            a,
            sub_dead,
            best_seq=best_seq,
            best_score=best_score,
            eta0=0.5,
            beta=0.9,
            eval_every=40,
        )
        # update global best if needed
        cur_score = _objective(a)
        if cur_score < best_score - 1e-12:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time():.1f}] up‑sample polished new best {best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final simulated‑annealing polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing phase")
        a, final_score = _pairwise_annealing(
            best_seq.copy(),
            deadline,
            best_seq=best_seq,
            best_score=best_score,
            rng=rng,
        )
        if final_score < best_score:
            best_score, best_seq = final_score, a.copy()
            print(f"[{time.time():.1f}] after annealing new best {best_score:.6f}")

    print(
        f"Returning best score {best_score:.6f} (length={len(best_seq)})"
    )
    # evaluate_sequence expects a plain Python list
    return best_seq.tolist()
