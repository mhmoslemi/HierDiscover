"""
A 1000‑second optimizer for the step‑function bound

Objective (for a probability vector a, Σa=1):
    F(a) = 2·n·max(conv(a,a))

The algorithm combines:
  • a rich library of analytic seeds,
  • a smooth‐gradient descent (mirror descent with an entropy mirror)
    that uses a soft‑max approximation of the max,
  • a tiny LP‑polish for short vectors (n≤2048) that maximises the total
    mass under the current convolution bound,
  • progressive up‑sampling (linear interpolation) and further descent,
  • a final simple pairwise mass‑transfer annealing.

All vectors are kept on the probability simplex (non‑negative, Σ=1)
so the evaluation function ``evaluate_sequence`` can be called directly.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Convolution via FFT (full length)."""
    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:n]


# ----------------------------------------------------------------------
# Exact objective for a probability vector (Σa=1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families (return probability vectors, Σ=1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # avoid 0/1 when α<1
    w = np.where(
        (x == 0) | (x == 1),
        0.0,
        np.exp((alpha - 1) * np.log(x) + (alpha - 1) * np.log1p(-x))
    )
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination ``lam·uniform + (1‑lam)·base``."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smooth gradient of the soft‑max approximation of the max‑convolution
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Returns ∇[2·n·softmax_β(conv(a,a))].  The gradient is
        4·n·( w * a_rev )[k]   for k=0…n‑1,
    where w_i ∝ exp(β·(conv_i‑max)).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)                # length 2n‑1
    M = conv.max()
    # weights (soft‑max, not normalised yet)
    w = np.exp(beta * (conv - M))
    w_sum = w.sum()
    if w_sum == 0.0:          # should never happen
        w_norm = np.full_like(w, 1.0 / w.size)
    else:
        w_norm = w / w_sum

    # convolution of the weight vector with the reversed a
    conv_full = _fft_convolve(w_norm, a[::-1])
    # slice that corresponds to the gradient components
    grad_slice = conv_full[n - 1:n - 1 + n]      # length n
    grad = 4.0 * n * grad_slice
    return grad


# ----------------------------------------------------------------------
# Smooth descent on the simplex (mirror descent with entropy mirror)
# ----------------------------------------------------------------------
def _smooth_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    beta_max: float = 150.0,
    eta0: float = 0.10,
    mu: float = 0.90,
    eval_every: int = 40,
) -> tuple[np.ndarray, float]:
    """
    Projected mirror descent using the smooth (soft‑max) objective.
    The step size eta is adapted: raised when an improvement happens,
    otherwise decayed.
    """
    a = a0.copy()
    n = a.shape[0]
    v = np.zeros_like(a)            # momentum term
    eta = eta0
    it = 0
    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        it += 1
        # linear schedule for the smoothing parameter
        frac = (time.time() - start) / total
        beta = 5.0 + (beta_max - 5.0) * frac

        # gradient of the smooth approximation
        grad = _smooth_gradient(a, beta)

        # normalise the direction to keep steps comparable
        grad_norm = np.linalg.norm(grad) + 1e-12
        step = -eta * grad / grad_norm
        v = mu * v + step
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)                 # non‑negativity
        s = a_new.sum()
        if s == 0.0:
            a_new[:] = 1.0 / n
        else:
            a_new /= s                                 # re‑project onto the simplex

        if it % eval_every == 0:
            cur = _objective(a_new)
            if cur < best_score - 1e-12:
                best_score, best_seq = cur, a_new.copy()
                eta = min(eta0, eta * 1.05)            # encourage larger steps
                print(
                    f"[{time.time():.1f}] descent it={it:,} new best {best_score:.6f}"
                )
            else:
                eta *= 0.97                            # modest decay

        a = a_new
    return best_seq, best_score


# ----------------------------------------------------------------------
# Tiny linear‑program polishing for short vectors (n ≤ 2048)
# ----------------------------------------------------------------------
def _lp_polish(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_n: int = 2048,
    max_iters: int = 4,
) -> tuple[np.ndarray, float]:
    """
    Solve   max Σ b_j   s.t.  b≥0,  a*b ≤ M  (pointwise)
    where M = max(conv(a,a)).  The solution is mixed back into ``a``.
    """
    a = a0.copy()
    n = a.shape[0]
    if n > max_n:
        return a, best_score

    # Build the convolution matrix once (it stays constant while a is fixed)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j : j + n, j] = a

    while time.time() < deadline and max_iters > 0:
        max_iters -= 1
        M = _fft_self_convolve(a).max()
        ub = np.full(rows, M)
        c = -np.ones(n)                         # maximise Σ b_j  → minimise -Σ b_j
        bounds = [(0, None)] * n

        res = linprog(
            c,
            A_ub=A,
            b_ub=ub,
            bounds=bounds,
            method="highs",
            options={"disp": False},
        )
        if not res.success or res.x is None:
            break

        b = res.x
        s = b.sum()
        if s <= 1e-14:
            break
        b_norm = b / s

        # try a few mixing ratios
        for lam in (0.2, 0.3, 0.5):
            a_candidate = (1.0 - lam) * a + lam * b_norm
            cur = _objective(a_candidate)
            if cur < best_score - 1e-12:
                best_score, best_seq = cur, a_candidate.copy()
                a = a_candidate
                print(
                    f"[{time.time():.1f}] LP mix λ={lam:.2f} new best {best_score:.6f}"
                )
                # rebuild A for the new a (required for the next iteration)
                A[:] = 0.0
                for j in range(n):
                    A[j : j + n, j] = a
                break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple pairwise mass‑transfer annealing (as in the reference code)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        src = rng.choice(np.where(a > 1e-14)[0])
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1
        delta = a[src] * rng.random() * 0.3
        if delta < 1e-16:
            continue
        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta
        cur = _objective(new_a)
        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.02
        accept = False
        if cur < best_score - 1e-12:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True
        if accept:
            a = new_a
            if cur < best_score - 1e-12:
                best_score, best_seq = cur, a.copy()
                print(
                    f"[{time.time():.1f}] anneal {src}->{dst} Δ={delta:.1e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass = 1)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new


# ----------------------------------------------------------------------
# Main search entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` (i.e. 2·n·max(conv)/sum²). Returns the best list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    cand_pool: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1️⃣  Seed generation (analytic families, limited lengths)
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096]          # moderate sizes – cheap LP possible
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating seeds, n={n}")

        # deterministic families
        for base in (
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_hann(n),
        ):
            sc = _objective(base)
            cand_pool.append((sc, base.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new global best (det) {sc:.6f}")

        # edge‑weighted with a dense grid of exponents
        p_vals = np.concatenate((
            np.logspace(-3.0, -0.5, 10),   # 0.001 → 0.316
            np.logspace(-0.5, 0.0, 10),    # 0.316 → 1
            np.linspace(0.1, 2.0, 12),     # linear region
            np.logspace(0.0, 0.7, 8),      # 1 → 5
        ))
        lam_vals = np.linspace(0.0, 1.0, 7)      # mixing with uniform
        for p in p_vals:
            if time.time() >= deadline:
                break
            base = _distribution_edge_weighted(n, float(p))
            for lam in lam_vals:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(base, float(lam))
                sc = _objective(cand)
                cand_pool.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] edge p={p:.3g} λ={lam:.2f} new best {sc:.6f}")

        # symmetric Beta family
        for alpha in np.concatenate((
            np.logspace(-2.0, -0.2, 8),
            np.linspace(0.2, 2.0, 10),
            np.linspace(2.0, 5.0, 6),
        )):
            if time.time() >= deadline:
                break
            cand = _distribution_beta(n, float(alpha))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] beta α={alpha:.3g} new best {sc:.6f}")

        # generalized normal family
        for q in np.linspace(0.3, 4.0, 12):
            if time.time() >= deadline:
                break
            cand = _distribution_generalized_normal(n, float(q))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] gn q={q:.2f} new best {sc:.6f}")

        # random Dirichlet seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            log_alpha = rng.uniform(-2.0, 1.0)    # α ∈ [0.01,10]
            alpha = 10 ** log_alpha
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] Dirichlet α={alpha:.3g} new best {sc:.6f}")

        # sparse random seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.02)
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] sparse new best {sc:.6f}")

        # keep top‑30 seeds overall (memory bounded)
        cand_pool.sort(key=lambda x: x[0])
        cand_pool = cand_pool[:30]

    # ------------------------------------------------------------------
    # 2️⃣  Incorporate a possibly‑existing construction
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.abs(prev)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            cand_pool.append((sc, prev.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3️⃣  Ensure we have a starting point
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 2048
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Quick smooth‑descent on all seeds (short budget per seed)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and cand_pool:
        per_seed = max(0.5, remaining * 0.2 / max(len(cand_pool), 1))
        print(f"[{time.time():.1f}] quick descent on {len(cand_pool)} seeds (~{per_seed:.2f}s each)")
        for idx, (sc, a0) in enumerate(cand_pool):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            a_opt, sc_opt = _smooth_descent(
                a0.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                beta_max=80.0,
                eta0=0.10,
                eval_every=30,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] seed {idx} improved best to {overall_best_score:.6f}")

            # tiny LP polish for short vectors
            if a_opt.shape[0] <= 2048 and time.time() < deadline:
                a_opt2, sc_opt2 = _lp_polish(
                    a_opt.copy(),
                    deadline,
                    best_seq=overall_best_seq,
                    best_score=overall_best_score,
                )
                if sc_opt2 < overall_best_score - 1e-12:
                    overall_best_score, overall_best_seq = sc_opt2, a_opt2.copy()
                    print(f"[{time.time():.1f}] LP after seed {idx} improved best to {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Longer descent on the current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_dead = time.time() + (deadline - time.time()) * 0.35
        a_opt, sc_opt = _smooth_descent(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            beta_max=150.0,
            eta0=0.08,
            eval_every=60,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] long descent improved best to {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Progressive up‑sampling with intermediate optimisation
    # ------------------------------------------------------------------
    while time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        if cur_n >= 65536:
            break
        target_n = min(cur_n * 2, 65536)
        print(f"[{time.time():.1f}] up‑sampling {cur_n} → {target_n}")
        up = _upsample(overall_best_seq, target_n)
        up_sc = _objective(up)
        if up_sc < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = up, up_sc
            print(f"[{time.time():.1f}] up‑sampled new best {overall_best_score:.6f}")

        # medium descent on the up‑sampled vector
        sub_dead = time.time() + (deadline - time.time()) * 0.2
        a_opt, sc_opt = _smooth_descent(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            beta_max=180.0,
            eta0=0.06,
            eval_every=50,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
            print(f"[{time.time():.1f}] after up‑sample descent new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣  Final annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing phase")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
            print(f"[{time.time():.1f}] after annealing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score = {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
