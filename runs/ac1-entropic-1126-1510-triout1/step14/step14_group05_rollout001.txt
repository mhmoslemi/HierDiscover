def _annealing_swap(a, time_deadline, best_score):
    a = a.copy()
    n = len(a)
    cur_score = _objective(a)
    T0 = 0.1
    t0 = time.time()
    while time.time() < time_deadline:
        elapsed = time.time() - t0
        T = T0 * (1 - elapsed / (time_deadline - t0))
        # propose move
        i = rng.integers(n)
        j = rng.integers(n)

Propose a step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.

The algorithm is a multi‑stage search that tries to go beyond the previous
implementation:

1️⃣  **Broad seeding** – a rich pool of candidates is generated for several
    lengths (up to 20 000).  Families include the classic analytic shapes,
    random Dirichlet draws with varying concentration, and very sparse
    seeds.  Every candidate is normalised to sum = 1 and its exact objective
    ``2·n·max(conv)`` is evaluated with an FFT based convolution.

2️⃣  **Momentum‑soft‑max descent** – the best few seeds are refined by a
    smooth (soft‑max) approximation of the L∞ objective.  The temperature
    τ is annealed from 2·10⁻² to 10⁻⁴ while an exponentiated‑gradient update
    with a first‑moment momentum term is performed.  This stage is fast and
    explored on many seeds.

3️⃣  **Long exact sub‑gradient polishing** – after the smooth phase the
    current global best is improved by many iterations of exact
    sub‑gradient descent (the same objective used in the original code).
    This uses the full max‑convolution gradient and runs for the rest of the
    time budget.

4️⃣  **Simulated‑annealing hill‑climber** – a final stochastic local search
    performs random mass transfers between two positions and accepts a move
    that either improves the exact objective or is accepted with a
    temperature‑dependent probability.  This helps to escape tiny plateaux
    left by the deterministic optimiser.

All stages constantly check the remaining time (the total budget is 1000 s)
and keep the best vector found so far.  The returned list is a plain Python
list; ``evaluate_sequence`` will re‑normalise it internally.

The code deliberately avoids closures or lambda‑expressions, respects the
CPU limit (uses only NumPy) and prints progress information for debugging.
"""

# Helper utilities

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
    {x ≥ 0, Σ x = s}.
    """
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())          # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_cross_correlate_valid(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (valid part):
    (w ⋆ a)[j] = Σ_i w[i+j]·a[i]   for j = 0 … len(a)-1.
    Implemented with FFT for speed.
    """
    n = a.shape[0]
    m = w.shape[0]          # = 2·n‑1
    size = 1 << ((m + n - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    A = np.fft.rfft(a, n=size)
    corr = np.fft.irfft(W * np.conj(A), n=size)
    return corr[:n]


def _fft_sliding_window_sum(w: np.ndarray, win: int) -> np.ndarray:
    """
    Compute Σ_{k=j}^{j+win‑1} w[k] for each j = 0 … win‑1.
    Implemented via FFT.
    """
    m = w.shape[0]
    size = 1 << ((m + win - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    O = np.fft.rfft(np.ones(win, dtype=np.float64), n=size)
    conv = np.fft.irfft(W * O, n=size)
    start = win - 1
    return conv[start:start + win]


# ----------------------------------------------------------------------
# Analytic families – all return a probability vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_edge(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta‑type density w_i ∝ (i·(n‑i))^{α‑1} for α>0."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = (i * (n - i)) ** (alpha - 1.0)
    vals = np.clip(vals, 0.0, None)
    s = vals.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    return vals / s


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Edges weighted by (i·(n‑i))^{-p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = (i * (n - i)) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    return w / w_sum


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse Dirichlet‑like seed with given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    return vals / s


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Exact objective (assumes sum = 1)  →  2·n·max(conv)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) for a non‑negative vector with sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Soft‑max (smooth) gradient descent with momentum
# ----------------------------------------------------------------------
def _smooth_descent_momentum(
    a0: np.ndarray,
    time_deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    tau_start: float = 0.02,
    tau_end: float = 0.001,
    lr0: float = 0.6,
    beta1: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a soft‑max approximation of the L∞
    objective.  A first‑moment (exponential) moving average of the gradient
    provides simple momentum.  The temperature τ is linearly annealed.
    """
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    it = 0
    grad_avg = np.zeros_like(a)

    while time.time() < time_deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] smooth it={it} new global best {best_score:.6f}"
            )

        # anneal temperature
        frac = (time.time() - start) / (time_deadline - start)
        tau = tau_start * (1.0 - frac) + tau_end * frac

        # soft‑max weights (stable)
        maxc = conv.max()
        shifted = conv - maxc
        exp_vals = np.exp(shifted / tau)      # all ≤ 1
        w = exp_vals / exp_vals.sum()         # length 2n‑1, sums to 1

        # gradient of the smoothed objective
        term1 = _fft_cross_correlate_valid(w, a)      # Σ w[i+j]·a[i]
        win_sum = _fft_sliding_window_sum(w, n)      # Σ_{k=j}^{j+n‑1} w[k]
        grad = term1 + a * win_sum

        # momentum update (exponential moving average)
        grad_avg = beta1 * grad_avg + (1.0 - beta1) * grad

        # EG update with step size η_t = lr0 / sqrt(t)
        eta = lr0 / math.sqrt(it)
        a *= np.exp(-eta * grad_avg)
        # project onto simplex (sum = 1)
        a_sum = a.sum()
        if a_sum <= 0.0:
            a = _distribution_uniform(n)
        else:
            a /= a_sum

    return best_seq, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (same as original)
# ----------------------------------------------------------------------
def _subgradient_descent_exact(
    a0: np.ndarray,
    time_deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        grad = np.zeros_like(a)
        for idx in max_idxs:
            # indices i that contribute to position idx
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = _distribution_uniform(n)
        else:
            a /= s

        if t % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] exact subgrad it={t} new global best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climber (pairwise mass transfer)
# ----------------------------------------------------------------------
def _simulated_annealing_swap(
    a0: np.ndarray,
    time_deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Randomly transfer a fraction of mass from one position to another.
    Moves are accepted if they improve the exact objective, otherwise
    with probability exp((old‑new)/T) where T is linearly cooled.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)

    start = time.time()
    while time.time() < time_deadline:
        # linear temperature schedule
        frac = (time.time() - start) / (time_deadline - start)
        T = max(0.05 * (1.0 - frac), 1e-4)

        # pick donor i (must have some mass) and receiver j≠i
        donors = np.where(a > 1e-12)[0]
        if donors.size == 0:
            break
        i = rng.choice(donors)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction (up to 30 %) of donor mass
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-15:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = _objective(a_new)

        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < math.exp((cur_score - new_score) / T)

        if accept:
            a = a_new
            cur_score = new_score
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] SA swap i={i}->j={j} δ={delta:.6g} new best {best_score:.6f}"
                )
    return best_seq, best_score

    overall_best_seq = None

    candidates: list[tuple[float, np.ndarray]] = []   # (score, seq)

    # ------------------------------------------------------------------

    lengths = [2000, 4000, 6000, 8000, 12000, 16000, 20000]

            rng=rng,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(
                f"[{time.time():.1f}] after SA polishing new best {overall_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    print(
        f"FINAL best objective (2·n·max(conv)) = {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return plain Python list – the external ``evaluate_sequence`` will
    # perform its own clipping and normalisation.
    return overall_best_seq.tolist()
