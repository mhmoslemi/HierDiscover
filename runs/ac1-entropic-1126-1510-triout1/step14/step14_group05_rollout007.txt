"""
Improved search for a non‑negative step function that minimises
``evaluate_sequence``.  The algorithm consists of four stages:

1️⃣  **Exhaustive family scan** – a rich collection of analytic families
    (uniform, arcsine, sine, cosine, quadratic, triangular, Gaussian,
    beta‑edge, power‑edge, edge‑weighted, exponential‑edge) is evaluated
    on several lengths.  Each family is explored on a fine grid of its
    shape parameters and mixed with a uniform component.  All candidates
    are stored and the best few (≈30) are kept for refinement.

2️⃣  **Smooth (soft‑max) gradient descent** – for each promising seed we
    minimise a smoothed version of the L∞ objective
    `τ·log Σ exp(conv/τ)` using an exponentiated‑gradient (EG) update.
    The temperature τ is annealed from 0.1 down to 0.001 and the learning
    rate is scaled as η / √t.  The best result of this stage is retained.

3️⃣  **Exact sub‑gradient descent + mass‑transfer hill‑climbing** – the
    true max‑convolution objective is tackled with an EG step using the
    exact sub‑gradient (derived from the shifts attaining the current
    maximum).  After a limited number of iterations a greedy “mass‑transfer”
    local search is performed: the sub‑gradient identifies a high‑gradient
    source index and a low‑gradient sink index and a small amount of mass
    is moved from source to sink if it strictly improves the objective.
    The process repeats until the time budget expires.

4️⃣  **Final pairwise swap polishing** – a simple random pairwise
    mass‑transfer hill‑climber is run for the remaining time to catch any
    remaining improvements.

All stages respect the global time budget (default 1000 s) and the best
sequence found is returned as a plain Python list.  The code is
self‑contained and uses only NumPy and the standard library.
"""

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_cross_correlate_valid(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (valid part)  (w ⋆ a)[j] = Σ_i w[i+j]·a[i],
    returning an array of length len(a).  Implemented with FFT for speed.
    """
    n = a.shape[0]
    m = w.shape[0]                # = 2·n‑1
    size = 1 << ((m + n - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A = np.fft.rfft(a, n=size)
    corr = np.fft.irfft(W * np.conj(A), n=size)
    return corr[:n]               # valid region


def _fft_sliding_window_sum(w: np.ndarray, win: int) -> np.ndarray:
    """
    For each j compute Σ_{i=j}^{j+win‑1} w[i] (window length = win).
    Returns an array of length ``win`` (i.e. len(a)).  Implemented via FFT.
    """
    m = w.shape[0]
    size = 1 << ((m + win - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    ones = np.ones(win, dtype=np.float64)
    O = np.fft.rfft(ones, n=size)
    conv = np.fft.irfft(W * O, n=size)
    start = win - 1
    return conv[start:start + win]

# Analytic families (all returned vectors sum to 1)

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 1.0 - np.abs(i - (n - 1) / 2.0) / ((n - 1) / 2.0)
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    w /= w.sum()
    return w


def _distribution_beta_edge(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta‑type density  w_i ∝ (i·(n‑i))^{α‑1}  (α>0)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = (i * (n - i)) ** (alpha - 1.0)
    vals = np.clip(vals, 0.0, None)
    s = vals.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    vals /= s
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i·(n‑i))^{‑p}  with p>0."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    w /= w_sum
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_power_edge(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i+½)^{‑p} + (n‑½‑i)^{‑p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = (i) ** (-p) + (n - i) ** (-p)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.005
) -> np.ndarray:
    """Sparse Dirichlet‑like seed with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Exact objective (sum = 1 ⇒ 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth (soft‑max) gradient descent
# ----------------------------------------------------------------------
def _smooth_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    tau_start: float = 0.1,
    tau_end: float = 0.001,
    lr0: float = 0.8,
    eval_every: int = 150,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a smoothed version of the L∞ objective.
    The soft‑max temperature τ is annealed linearly from ``tau_start`` to
    ``tau_end``.  The (exact) gradient of the smoothed objective is obtained
    via FFT‑based cross‑correlation.
    """
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    it = 0
    while time.time() < time_limit:
        it += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] smooth it={it} new global best {best_score:.6f}")

        # anneal temperature
        frac = (time.time() - start) / (time_limit - start)
        tau = tau_start * (1.0 - frac) + tau_end * frac

        # soft‑max weights (stable)
        shifted = conv - max_val            # avoid overflow
        exp_vals = np.exp(shifted / tau)
        w = exp_vals / exp_vals.sum()       # len = 2n‑1, sums to 1

        # gradient of the smooth objective
        term1 = _fft_cross_correlate_valid(w, a)          # Σ w[k+i] a[i]
        win_sum = _fft_sliding_window_sum(w, n)          # Σ_{k=j}^{j+n‑1} w[k]
        grad = term1 + a * win_sum

        # EG update (step‑size ∝ 1/√t)
        eta = lr0 / math.sqrt(it)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if it % eval_every == 0:
            # already reported improvements above
            pass

    return best_seq, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (EG) – used as a polishing stage
# ----------------------------------------------------------------------
def _exact_subgradient_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 120,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    while time.time() < time_limit:
        it += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        # indices attaining the maximum
        max_mask = (conv == max_val).astype(np.float64)

        # sub‑gradient for each coordinate:
        # grad = Σ_{k in max} (a_{k-i} + a_i)  (but we weight correctly)
        term1 = _fft_cross_correlate_valid(max_mask, a)   # Σ_{k∈max} a_{k-i}
        win_sum = _fft_sliding_window_sum(max_mask, n)   # count of max‑shifts covering i
        grad = term1 + a * win_sum

        eta = lr0 / math.sqrt(it)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if it % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] exact it={it} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Mass‑transfer greedy hill‑climber (uses exact sub‑gradient info)
# ----------------------------------------------------------------------
def _mass_transfer_hill_climber(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly moves a small amount of probability mass from a high‑gradient
    index (where the max‑convolution is most sensitive) to a low‑gradient
    index, accepting the move only if it strictly improves the exact
    objective.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        # compute sub‑gradient for the current max‑conv set
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_mask = (conv == max_val).astype(np.float64)

        term1 = _fft_cross_correlate_valid(max_mask, a)
        win_sum = _fft_sliding_window_sum(max_mask, n)
        grad = term1 + a * win_sum

        # identify source (high grad) and sink (low grad)
        src_idx = np.argmax(grad)
        sink_idx = np.argmin(grad)
        if src_idx == sink_idx or a[src_idx] < 1e-15:
            # pick a random alternative if degenerate
            src_candidates = np.where(a > 1e-12)[0]
            if src_candidates.size == 0:
                break
            src_idx = int(rng.choice(src_candidates))
            sink_idx = int(rng.integers(0, n))
            if sink_idx == src_idx:
                continue

        # transfer a fraction of the source mass
        gamma = 0.12  # up to ~12 % of source mass per move
        delta = min(a[src_idx] * gamma, a[src_idx] * rng.random())
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[src_idx] -= delta
        new_a[sink_idx] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] mass‑transfer {src_idx}->{sink_idx} δ={delta:.6g} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise swap hill‑climber (kept as a final polish)
# ----------------------------------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = int(rng.choice(src))
        j = int(rng.integers(0, n - 1))
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 4.0      # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, seq)

    # ------------------------------------------------------------------
    # 1️⃣  Scan analytic families on several lengths
    # ------------------------------------------------------------------
    lengths = [500, 1000, 2000, 4000, 8000, 12000, 16000]

    # Parameter grids (coarse → fine)
    alpha_grid = np.linspace(0.05, 0.60, 8)          # beta‑edge exponent
    p_grid = np.concatenate(
        (np.logspace(-3, -0.5, 6), np.logspace(-0.5, 0.0, 6), np.linspace(0.1, 2.0, 6))
    )
    c_grid = np.logspace(-4, 0, 8)                # exponential‑edge decay
    sigma_frac_grid = np.linspace(0.1, 0.5, 5)     # σ = frac·n for Gaussian
    power_p_grid = np.linspace(0.2, 2.0, 6)       # power‑edge exponent
    lam_grid = np.linspace(0.0, 1.0, 9)           # mixture with uniform

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families (no mixing)
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
            _distribution_triangular,
        ):
            cand = base_fn(n)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # families that we also mix with uniform
        # 1. Beta‑edge
        for alpha in alpha_grid:
            if time.time() >= deadline:
                break
            base = _distribution_beta_edge(n, float(alpha))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] len={n} β α={alpha:.3g} lam={lam:.2f} score={sc:.6f}")

        # 2. Edge‑weighted
        for p in p_grid:
            if time.time() >= deadline:
                break
            base = _distribution_edge_weighted(n, float(p))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] len={n} edge p={p:.4g} lam={lam:.2f} score={sc:.6f}")

        # 3. Exponential‑edge
        for c in c_grid:
            if time.time() >= deadline:
                break
            base = _distribution_exponential_edge(n, float(c))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] len={n} exp c={c:.4g} lam={lam:.2f} score={sc:.6f}")

        # 4. Gaussian edge
        for frac in sigma_frac_grid:
            sigma = max(1e-3, frac * n)
            base = _distribution_gaussian(n, sigma)
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] len={n} gauss σ_frac={frac:.2f} lam={lam:.2f} score={sc:.6f}")

        # 5. Power‑edge
        for p in power_p_grid:
            if time.time() >= deadline:
                break
            base = _distribution_power_edge(n, float(p))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] len={n} power p={p:.2f} lam={lam:.2f} score={sc:.6f}")

        # Random Dirichlet seeds
        for _ in range(12):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} score={sc:.6f}")

        # Sparse random seeds
        for _ in range(12):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.007)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

        # keep a manageable number of promising candidates
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:200]   # retain up to 200 best seeds

    # ------------------------------------------------------------------
    # 2️⃣  Incorporate a previous construction if available
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3️⃣  Fallback (should never trigger)
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Smooth gradient descent on the most promising seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 60.0 and top_candidates:
        # select the best few seeds for deeper optimisation
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:8]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] start smooth descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            # allocate roughly 45 % of per‑seed budget to smooth descent
            sg_deadline = time.time() + per_seed * 0.45
            a_opt, sc_opt = _smooth_descent(
                seed_seq,
                sg_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                tau_start=0.1,
                tau_end=0.001,
                lr0=0.8,
                eval_every=150,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after smooth seed {idx} new global best {overall_best_score:.6f}")

            # quick exact sub‑gradient polish (≈30 % of per‑seed budget)
            eg_deadline = time.time() + per_seed * 0.30
            a_opt, sc_opt = _exact_subgradient_descent(
                overall_best_seq,
                eg_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                eval_every=200,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after exact polish seed {idx} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Mass‑transfer hill‑climber on the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        mt_deadline = time.time() + (deadline - time.time()) * 0.40
        print(f"[{time.time():.1f}] mass‑transfer hill‑climber")
        a_opt, sc_opt = _mass_transfer_hill_climber(
            overall_best_seq.copy(),
            mt_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=None,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after mass‑transfer new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Final greedy pairwise polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final pairwise swap polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣  Report final result
    # ------------------------------------------------------------------
    print(
        f"Final best objective (2·n·max(conv)): {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    # Return a plain Python list – the external runner will evaluate it.
    return overall_best_seq.tolist()
