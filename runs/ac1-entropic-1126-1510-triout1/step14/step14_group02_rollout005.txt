def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and math.isclose(v.sum(), s, rel_tol=1e-12):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    rho = rho_arr[-1] if rho_arr.size else -1
    theta = 0.0 if rho == -1 else cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT (exact up to rounding)."""
    need = len(x) + len(y) - 1
    size = 1 << ((need - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    c = np.fft.irfft(X * Y, n=size)
    return c[:need]


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) under the normalization sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:

    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)

    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)

    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.006) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)

    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Return lam·U + (1‑lam)·base (U = uniform)."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Exact left‑right symmetry, then re‑project onto the simplex."""
    sym = (seq + seq[::-1]) * 0.5
    return sym / sym.sum()


# ----------------------------------------------------------------------
# Smooth‑max mirror descent --------------------------------------------
def _smoothmax_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.7,
    lambda_start: float = 5.0,
    lambda_end: float = 200.0,
    eval_interval: int = 150,
    patience: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the smoothed objective
        L_λ(a) = (1/λ) log Σ_k exp(λ·conv_k)
    with a logarithmic schedule λ∈[lambda_start, lambda_end].
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    # pre‑allocate a_rev (used many times)
    a_rev = a[::-1]

    while time.time() < deadline and t < 1_000_000:
        t += 1
        # current convex combination parameter λ (log‑linear schedule)
        lam = lambda_start * (lambda_end / lambda_start) ** (t / 1_000_000)

        conv = _fft_self_convolve(a)
        max_c = conv.max()

        # smooth weights (stable because exp ≤ 1)
        w = np.exp(lam * (conv - max_c))
        p = w / w.sum()

        # gradient = 2 * correlation(p, a) restricted to n entries
        corr = _fft_convolve(p, a_rev)                # length = 3n‑2
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # exponentiated‑gradient step
        eta = lr0 / math.sqrt(t)
        # clip extreme grad components to avoid overflow/underflow
        grad = np.clip(grad, -30.0, 30.0)
        a = a * np.exp(-eta * grad)

        # simplex projection (multiply then renormalise)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} smooth‑max new best {best_score:.6f}")
            else:
                no_improve += 1

        # simple patience‑based jitter to escape plateaus
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.005, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Weighted pairwise refinement -------------------------------------------
def _pairwise_weighted_refine(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer guided by the location of the current maximal convolution entry."""
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0

    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        conv = _fft_self_convolve(a)
        k_max = int(conv.argmax())

        # impact values L_i = a_{k_max-i} (zero outside support)
        L = np.zeros(n, dtype=np.float64)
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        idx = np.arange(low, high + 1, dtype=np.int64)
        L[idx] = a[k_max - idx]

        # source: favour large impact
        if L.sum() > 0.0:
            probs_i = L / L.sum()
            i = rng.choice(n, p=probs_i)
        else:
            pos = np.where(a > 1e-12)[0]
            if pos.size == 0:
                break
            i = rng.choice(pos)

        # destination: favour small impact
        Lmax = L.max()
        probs_j = (Lmax - L) + 1e-12
        probs_j /= probs_j.sum()
        j = rng.choice(n, p=probs_j)
        if j == i:
            j = (j + 1) % n

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] pairwise swap {i}->{j} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Cheap simulated‑annealing ---------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Lightweight Metropolis‑type local search."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsample helper -------------------------------------------------------
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimizes ``evaluate_sequence``.
    The algorithm follows the three‑stage pipeline described at the top of the file
    and respects the supplied time budget.
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # tiny safety margin

    # ------------------------------------------------------------------
    # 1.  Seed generation (moderate lengths)
    lengths = [2500, 3000, 3500, 4000, 5000]
    # a richer p‑grid – include larger exponents up to 20
    p_grid = np.concatenate(
        [
            np.logspace(-3, -0.5, 7),          # 0.001 … 0.316
            np.logspace(-0.5, 0, 9),          # 0.316 … 1
            np.logspace(0, 0.9, 9),           # 1 … 7.94
            np.logspace(1, 1.3, 5),           # 10 … 20
        ]
    )
    lam_grid = np.linspace(0.0, 0.9, 10)

    top_candidates: list[tuple[float, np.ndarray]] = []
    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Generating seeds of length {n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (base) {sc:.6f}")

        # parametrised families
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            edge2 = _distribution_edge_inversepow(n, p)
            sine = _distribution_sine_weighted(n, p)

            for base in (beta, edge, edge2, sine):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # asymmetric beta
        for _ in range(8):
            if time.time() >= deadline:
                break
            a = rng.uniform(0.001, 5.0)
            b = rng.uniform(0.001, 5.0)
            cand = _distribution_beta_asym(n, a, b)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best asym‑beta len={n} a={a:.3g} b={b:.3g} score={sc:.6f}")

        # Dirichlet random
        for _ in range(8):
            if time.time() >= deadline:
                break
            cand = _distribution_dirichlet(n, rng)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best Dirichlet len={n} score={sc:.6f}")

        # Sparse random
        for _ in range(8):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best sparse len={n} score={sc:.6f}")

        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:20]

    # ------------------------------------------------------------------
    # 2.  If a previous construction exists, include it
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev = prev / prev.sum()
            sc = _objective(prev)
            top_candidates.append((sc, prev.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous seed best {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Guard against empty start
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform – score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Smooth‑max descent on the best few seeds (moderate size)
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]]
        per_seed = remaining * 0.55 / max(1, len(seeds))
        print(
            f"[{time.time():.1f}] smooth‑max descent on {len(seeds)} seeds, ≈{per_seed:.1f}s each"
        )
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _smoothmax_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                lambda_start=5.0,
                lambda_end=150.0,
                eval_interval=120,
                patience=2000,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Upsample and run a longer descent if time permits
    remaining = deadline - time.time()
    if remaining > 120.0:
        large_n = min(20000, len(overall_best_seq) * 2)
        print(f"[{time.time():.1f}] up‑sampling to {large_n} for intensive refinement")
        up_seq = _upsample(overall_best_seq, large_n)

        sub_deadline = time.time() + remaining * 0.6
        up_seq_opt, up_score = _smoothmax_descent(
            up_seq,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.6,
            lambda_start=5.0,
            lambda_end=250.0,
            eval_interval=100,
            patience=2500,
        )
        if up_score < overall_best_score:
            overall_best_score, overall_best_seq = up_score, up_seq_opt.copy()
            print(f"[{time.time():.1f}] up‑sampled descent yielded new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Jitter & restart cycles (help escape possible plateaus)
    while time.time() < deadline - 15.0:
        jitter = rng.exponential(scale=0.016, size=overall_best_seq.shape)
        cand = overall_best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sub_deadline = time.time() + min(18.0, (deadline - time.time()) * 0.5)
        cand_opt, cand_score = _smoothmax_descent(
            cand,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.5,
            lambda_start=5.0,
            lambda_end=200.0,
            eval_interval=80,
            patience=1500,
        )
        if cand_score < overall_best_score:
            overall_best_score, overall_best_seq = cand_score, cand_opt.copy()
            print(f"[{time.time():.1f}] jitter‑restart gave improvement → {overall_best_score:.6f}")
        else:
            break

    # ------------------------------------------------------------------
    # 7.  Symmetrisation test
    sym_seq = _symmetrize(overall_best_seq)
    sym_score = _objective(sym_seq)
    if sym_score + 1e-14 < overall_best_score:
        overall_best_score, overall_best_seq = sym_score, sym_seq
        print(f"[{time.time():.1f}] symmetrised version better: {sym_score:.6f}")

    # ------------------------------------------------------------------
    # 8.  Weighted pairwise deterministic refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] weighted pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_weighted_refine(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=180_000,
        )

    # ------------------------------------------------------------------
    # 9.  Final cheap simulated‑annealing clean‑up
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
