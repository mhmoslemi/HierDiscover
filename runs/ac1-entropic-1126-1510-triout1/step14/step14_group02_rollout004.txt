``evaluate_sequence``.

Algorithm – high‑level
----------------------
1. **Seed generation** – a diverse pool of analytic families (uniform,
   arcsine, symmetric β‑distribution, edge‑weighted, sine‑weighted,
   triangular “tent”) together with random Dirichlet vectors.
   All seeds are normalised to sum = 1 and the ~12 best are kept.

   sub‑gradient we minimise a smooth approximation of ``max(conv)``:
   ``fβ(a)= (1/β)·log Σ_k exp(β·(a*a)_k)``.
   Its gradient is a dense linear combination of the usual
   sub‑gradient and can be obtained via a single FFT‑based cross‑correlation.
   We run an Adam‑style exponential‑gradient update on the simplex,

   decaying the stepsize.  The descent is performed on each top seed for a

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _cross_corr_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (a ⋆ w)[i] = Σ_t a[t]·w[i+t] for i∈[0,n).
    Both inputs are real, a has length n, w has length 2·n‑1.
    """
    n = a.shape[0]
    need = n + (2 * n - 1) - 1               # = 3·n‑2
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    W = rfft(w, n=size)
    corr = irfft(A * np.conj(W), n=size)
    return corr[:n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a with sum(a)=1."""
    n = a.shape[0]
    return 2.0 * n * _fft_self_convolve(a).max()

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    """Symmetric tent shape, rising then falling linearly."""
    i = np.arange(n, dtype=np.float64)
    peak = (n - 1) / 2.0
    w = 1.0 - np.abs(i - peak) / peak
    w[w < 0] = 0.0
    w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.006) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, where U is the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Enforce exact left‑right symmetry and renormalise."""
    sym = (seq + seq[::-1]) * 0.5
    return sym / sym.sum()

def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    up = np.interp(x_new, x_old, seq)
    up = np.maximum(up, 0.0)
    up /= up.sum()
    return up


# ----------------------------------------------------------------------
def _smooth_expgrad_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.5,
    beta_start: float = 1.0,
    beta_end: float = 40.0,
    beta_inc: float = 0.02,
    eval_interval: int = 80,
    patience: int = 1800,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style exponential‑gradient descent on the smoothed objective
    fβ(a) = (1/β)·log Σ_k exp(β·conv_k).  The smoothing parameter β
    is increased slowly from ``beta_start`` to ``beta_end``.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    t = 0
    no_improve = 0
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta = beta_start

    ind_grid = np.arange(n, dtype=np.int32)

    while time.time() < deadline and t < 500_000:
        t += 1

        # ---------- convolution ----------
        conv = _fft_self_convolve(a)          # length 2n‑1
        max_c = conv.max()

        # ---------- update smoothing ----------
        beta = min(beta_end, beta + beta_inc)

        # ---------- smooth max via log‑sum‑exp ----------
        # stable formulation
        shifted = conv - max_c
        exp_terms = np.exp(beta * shifted)
        sum_exp = exp_terms.sum()
        w = exp_terms / sum_exp                # weights ∈ simplex, length 2n‑1

        # ---------- gradient (cross‑correlation) ----------
        grad = 2.0 * _cross_corr_fft(a, w)      # shape (n,)

        # ---------- Adam momentum ----------
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad ** 2)

        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)

        # ---------- stepsize ----------
        eta = lr0 / math.sqrt(t)

        # ---------- exponential‑gradient update ----------
        # protect against overflow/underflow in exponent
        delta = -eta * m_hat / (np.sqrt(v_hat) + 1e-10)
        delta = np.clip(delta, -50.0, 50.0)
        a = a * np.exp(delta)

        # renormalise to the simplex
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # ---------- occasional evaluation ----------
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} new best {best_score:.6f}")
            else:
                no_improve += 1

        # ---------- plateau handling ----------
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.004, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_weighted_refine(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Greedy deterministic mass‑transfer guided by the location of the
    current maximal convolution entry.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0

    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        conv = _fft_self_convolve(a)
        k_max = int(conv.argmax())

        # impact values L_i = a_{k_max-i} where defined
        L = np.zeros(n, dtype=np.float64)
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        idx = np.arange(low, high + 1)
        L[idx] = a[k_max - idx]

        # sample donor (high impact) and receiver (low impact)
        if L.sum() > 0.0:
            probs_i = L / L.sum()
            i = rng.choice(n, p=probs_i)
        else:
            positives = np.where(a > 1e-12)[0]
            if positives.size == 0:
                break
            i = rng.choice(positives)

        Lmax = L.max()
        probs_j = (Lmax - L) + 1e-12
        probs_j /= probs_j.sum()
        j = rng.choice(n, p=probs_j)
        if j == i:
            j = (j + 1) % n

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] weighted swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Lightweight Metropolis‑type annealing (single‑swap moves)."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp

    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-15:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0             # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Seed generation (moderate length, power‑of‑two for FFT speed)
    base_n = 4096
    lengths = [base_n]                         # we concentrate on a single size
    lam_grid = np.linspace(0.0, 0.9, 9)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Generating seeds of length {n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n), _distribution_triangular(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (deterministic) {sc:.6f}")

        # symmetric beta & edge families
        p_grid = np.concatenate(
            [
                np.logspace(-3, -1, 7),   # 0.001 … 0.1
                np.logspace(-1, 0, 7),    # 0.1 … 1
                np.logspace(0, 0.5, 5),   # 1 … 3.16
            ]
        )
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            edge2 = _distribution_edge_inversepow(n, p)
            sine = _distribution_sine_weighted(n, p)
            for base in (beta, edge, edge2, sine):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # asymmetric beta (random)
        for _ in range(8):
            if time.time() >= deadline:
                break
            a = rng.uniform(0.001, 3.0)
            b = rng.uniform(0.001, 3.0)
            cand = _distribution_beta_asym(n, a, b)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best asym a={a:.3g} b={b:.3g} score={sc:.6f}")

        # random Dirichlet vectors
        for _ in range(12):
            if time.time() >= deadline:
                break
            cand = _distribution_dirichlet(n, rng)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best Dirichlet score={sc:.6f}")

        # sparse random vectors
        for _ in range(8):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best sparse score={sc:.6f}")

        # keep only the very best seeds (≈12)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:12]

    # ------------------------------------------------------------------
    # 2. Incorporate any previously supplied construction (if present)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            # interpolate / resample to our base length
            prev = _upsample(prev, base_n)
            prev = np.maximum(prev, 0.0)
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best incorporated, score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback guard
    if overall_best_seq is None:
        overall_best_seq = _distribution_uniform(base_n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform, score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Smoothed exponential‑gradient descent on the top seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:8]]
        per_seed_time = remaining * 0.55 / max(1, len(seeds))
        print(
            f"[{time.time():.1f}] Heavy smoothed descent on {len(seeds)} seeds, ≈{per_seed_time:.1f}s each"
        )
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed_time
            a_opt, sc_opt = _smooth_expgrad_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.7,
                beta_start=1.0,
                beta_end=35.0,
                beta_inc=0.025,
                eval_interval=100,
                patience=1500,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after descent seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5. Upsample to a larger length and refine again (if time permits)
    remaining = deadline - time.time()
    if remaining > 100.0:
        large_n = min(18000, overall_best_seq.shape[0] * 2)
        print(f"[{time.time():.1f}] Upsampling to {large_n} points for final refinement")
        up_seq = _upsample(overall_best_seq, large_n)

        sub_deadline = time.time() + remaining * 0.6
        up_opt, up_score = _smooth_expgrad_descent(
            up_seq,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.5,
            beta_start=1.0,
            beta_end=40.0,
            beta_inc=0.04,
            eval_interval=80,
            patience=1800,
        )
        if up_score < overall_best_score:
            overall_best_score, overall_best_seq = up_score, up_opt.copy()
            print(f"[{time.time():.1f}] Upsampled descent gave new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Jitter + short re‑descent cycles (escape plateaus)
    while time.time() < deadline - 15.0:
        jitter = rng.exponential(scale=0.015, size=overall_best_seq.shape)
        cand = overall_best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sub_deadline = time.time() + min(18.0, (deadline - time.time()) * 0.5)
        cand_opt, cand_score = _smooth_expgrad_descent(
            cand,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.45,
            beta_start=1.0,
            beta_end=30.0,
            beta_inc=0.03,
            eval_interval=70,
            patience=1300,
        )
        if cand_score < overall_best_score:
            overall_best_score, overall_best_seq = cand_score, cand_opt.copy()
            print(f"[{time.time():.1f}] jitter‑restart improved to {overall_best_score:.6f}")
        else:
            # no improvement -> stop after a few attempts
            break

    # ------------------------------------------------------------------
    # 7. Symmetrisation test
    sym_seq = _symmetrize(overall_best_seq)
    sym_score = _objective(sym_seq)
    if sym_score + 1e-14 < overall_best_score:
        overall_best_score, overall_best_seq = sym_score, sym_seq
        print(f"[{time.time():.1f}] Symmetrised version better: {sym_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Weighted pairwise deterministic refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] Weighted pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_weighted_refine(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=250_000,
        )

    # ------------------------------------------------------------------
    # 9. Final cheap simulated‑annealing clean‑up
    if time.time() < deadline:
        print(f"[{time.time():.1f}] Final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
