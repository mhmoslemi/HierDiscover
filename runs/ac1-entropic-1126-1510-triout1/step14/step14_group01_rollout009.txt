def _max_gradient(a):
    n = len(a)
    conv = _fft_self_convolve(a)  # length 2n-1
    k = np.argmax(conv)
    max_conv = conv[k]
    # gradient
    # Indices i from 0 to n-1. For each i, need j = k - i.
    # If 0 <= j < n: grad[i] = 2 * a[j]; else grad[i] = 0.
    grad = np.zeros_like(a)
    # overlapping region: i from max(0, k-(n-1)) to min(n-1, k)
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    # for i in [i_min, i_max]: j = k - i
    # we can fill via vectorized slice:
    # i_range = np.arange(i_min, i_max+1)
    # j_range = k - i_range
    # grad[i_range] = 2 * a[j_range]
    # That's efficient.
    i_range = np.arange(i_min, i_max + 1)
    j_range = k - i_range
    grad[i_range] = 2.0 * a[j_range]
    return grad, max_conv

def _subgradient_descent(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = len(a)
    lr0 = 0.5  # initial step size
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    while time.time() < deadline and t < 1_000_000:
        t += 1
        grad, max_conv = _max_gradient(a)
        # Step size schedule: could use decreasing: eta = lr0 / sqrt(t)
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-30)
        a /= a.sum()
        # periodically evaluate exact J
        if t % 200 == 0:
            cur_score = 2.0 * n * max_conv
            if cur_score + 1e-14 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] subgrad iteration {t} new best {best_score}")
            else:
                no_improve += 1

        # jitter after stagnation
        if no_improve >= 2000:
            jitter = rng.exponential(scale=0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

def _subgradient_descent(a0, deadline, best_seq, best_score,
                         eta0=0.8, eval_every=200, patience=2000):
    a = a0.copy()
    t=0
    no_improve = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        t+=1
        grad, max_val = _max_gradient_and_val(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-30)
        a /= a.sum()
        if t % eval_every == 0:
            cur_score = 2.0 * len(a) * max_val
            if cur_score + 1e-14 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                no_improve = 0
                print("[...]" )
            else:
                no_improve+=1
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.001, size=len(a))
            a += jitter
            a = np.maximum(a, 0)
            a /= a.sum()
            no_improve=0
    return best_seq, best_score
```

```
def _upsample(seq, new_len):
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq
```

def propose_candidate(seed=42, budget_s=1000):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 3.0

    init_n = 5000

    seeds = []
    # deterministic families
    families = [
        lambda n: np.full(n, 1.0/n),
        lambda n: _distribution_arcsine(n),
        lambda n: _distribution_beta_sym(n, 0.4),
        lambda n: _distribution_beta_sym(n, 0.6),
        lambda n: _distribution_edge_weighted(n, 0.5),
        lambda n: _distribution_sine_weighted(n, 2.0),
        lambda n: _distribution_random_sparse(n, rng, density=0.01)
    ]
    # also add a Dirichlet random
    for _ in range(3):
        families.append(lambda n, rng=rng: _distribution_dirichlet(n, rng))

    # add previous global if exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.ndim == 1 and prev.size>0:
            cand = _upsample(prev, init_n) if len(prev)!=init_n else prev
            seeds.append((_objective(cand), cand.copy()))
    except Exception:
        pass

    # generate seeds from families
    for make in families:
        if time.time() >= deadline:
            break
        try:
            cand = make(init_n) if callable(make) else make(init_n)
        except Exception:
            continue
        cand = np.maximum(cand, 0.0)
        if cand.sum() == 0:
            cand[0] = 1.0
        cand = cand / cand.sum()
        score = _objective(cand)
        seeds.append((score, cand.copy()))
        print(f"[{time.time():.1f}] seed score {score:.6f}")

    # keep top seeds
    seeds.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in seeds[:5]]
    best_seq = top_seeds[0].copy()
    best_score = _objective(best_seq)

    # allocate time for refinement each
    remaining = deadline - time.time()
    per_seed_time = remaining * 0.5 / max(1, len(top_seeds))
    for i, seq in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        sub_dead = time.time() + per_seed_time
        seq_opt, seq_score = _subgradient_descent(seq, sub_dead, best_seq, best_score,
                                                 eta0=0.8, eval_every=150, patience=2000)
        if seq_score < best_score:
            best_score = seq_score
            best_seq = seq_opt.copy()
            print(f"[{time.time():.1f}] after refinement seed {i} new best {best_score}")

    # Upsample if time left
    remaining = deadline - time.time()
    if remaining > 30.0:
        new_len = min(18000, len(best_seq) * 2)
        up_seq = _upsample(best_seq, new_len)
        sub_dead = time.time() + remaining * 0.6

"""
Improved search for a low‑value step‑function.

Algorithm (all helpers are top‑level):
1. Build a small set of diverse seed distributions (uniform, arcsine,
   symmetric beta, edge‑weighted, sparse Dirichlet, and the historic
   best if available).
2. Keep the best few seeds and apply a *smoothed* log‑sum‑exp gradient
   descent (mirror‑descent on the simplex) to lower the smooth surrogate.
3. Upsample the current best vector to a larger length (up to 18 000)
   and continue the smoothed descent – a finer discretisation often
   yields a smaller supremum of the auto‑convolution.
4. Switch to a *direct* sub‑gradient descent on the exact max‑convolution
   (the gradient is 2·a shifted by the index of the maximal entry) for
   fast final fine‑tuning.
5. Finish with deterministic pairwise mass‑transfer sweeps and a cheap
   simulated‑annealing clean‑up.
All steps respect a global 1000 s budget and the result is returned as a
plain Python list of non‑negative heights.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σ x = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())  # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using a real FFT."""
    n = a.shape[0]
    m = b.shape[0]
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a simplex vector."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
# Smoothed surrogate ---------------------------------------------------
def _smooth_gradient(a: np.ndarray, lam: float) -> tuple[np.ndarray, float]:
    """Gradient of the log‑sum‑exp surrogate φ_λ."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)               # length 2n‑1
    max_c = conv.max()
    z = lam * conv
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    s = exp_z / exp_z.sum()                    # softmax weights, sum 1
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, s)        # length 3n‑2
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]   # centre block
    return grad, max_c


def _smooth_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    lam_start_factor: float = 0.15,
    lam_inc: float = 1.0015,
    beta: float = 0.9,
    eval_interval: int = 200,
    patience: int = 2500,
) -> tuple[np.ndarray, float]:
    """Annealed exponentiated‑gradient descent on the smoothed surrogate."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    lam = lam_start_factor * n
    rng = np.random.default_rng()

    while time.time() < deadline and t < 600_000:
        t += 1
        grad, max_conv = _smooth_gradient(a, lam)

        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size
        eta = lr0 / math.sqrt(t)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * momentum)

        a = np.maximum(a, 1e-30)
        a /= a.sum()

        # anneal λ
        lam *= lam_inc

        if t % eval_interval == 0:
            cur_score = 2.0 * n * max_conv
            if cur_score + 1e-14 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] smoothed it={t:,}  new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Exact max‑gradient (sub‑gradient) ------------------------------------
def _max_gradient_and_val(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Returns (grad, max_conv) where grad = ∇_a max_k (a*a)_k .
    The gradient is 2·a_{k*-i} for the index k* attaining the maximum.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(conv.argmax())
    max_val = float(conv[k])

    grad = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    if i_min <= i_max:
        idx = np.arange(i_min, i_max + 1, dtype=np.int64)
        grad[idx] = 2.0 * a[k - idx]
    return grad, max_val


def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    eval_every: int = 150,
    patience: int = 1800,
) -> tuple[np.ndarray, float]:
    """Mirror descent with the exact max‑gradient."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        grad, max_conv = _max_gradient_and_val(a)

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        a = np.maximum(a, 1e-30)
        a /= a.sum()

        if t % eval_every == 0:
            cur_score = 2.0 * n * max_conv
            if cur_score + 1e-14 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] subgrad it={t:,}  new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            jitter = rng.exponential(scale=0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

# Deterministic pairwise swap local search ---------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Simple greedy mass‑transfer sweep."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e}  new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Cheap simulated‑annealing clean‑up ----------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that accepts occasional uphill moves."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e}  best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Multi‑start search for a step‑function that minimises
    ``evaluate_sequence``.  The routine combines smoothed‑gradient

    refinements, always respecting the 1000 s budget.
    """

    start = time.time()
    # safety margin so that the orchestrator can still call evaluate_sequence
    deadline = start + budget_s - 3.0

    # ------------------------------------------------------------------
    # 0.  Basic parameters
    base_n = 5000                       # moderate resolution for the
                                        # first phase
    # ------------------------------------------------------------------
    # 1.  Seed generation (few diverse analytic families)
    seeds: list[tuple[float, np.ndarray]] = []

        vec = np.maximum(vec, 0.0)
        if vec.sum() == 0.0:
            vec[0] = 1.0
        vec = vec / vec.sum()
        sc = _objective(vec)
        seeds.append((sc, vec.copy()))
        print(f"[{time.time():.1f}] seed score {sc:.6f}")

    # deterministic families
    _add_seed(_distribution_uniform(base_n))
    _add_seed(_distribution_arcsine(base_n))
    _add_seed(_distribution_beta_sym(base_n, p=0.4))
    _add_seed(_distribution_beta_sym(base_n, p=0.6))
    _add_seed(_distribution_edge_weighted(base_n, p=0.5))
    _add_seed(_distribution_random_sparse(base_n, rng, density=0.012))

    # a few random Dirichlet draws
    for _ in range(4):
        _add_seed(_distribution_dirichlet(base_n, rng))

    # historic best if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            if len(prev) != base_n:
                prev = _upsample(prev, base_n)
            _add_seed(prev)
    except Exception:
        pass

    # keep only the elite seeds
    seeds.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in seeds[:5]]
    overall_best_seq = top_seeds[0].copy()

    # ------------------------------------------------------------------

                lam_inc=1.002,
                beta=0.9,
                eval_interval=150,
                patience=1800,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after smoothed seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Upsample to a larger resolution and continue with smoothed descent
    remaining = deadline - time.time()

    return overall_best_seq.tolist()
