"""
Propose a step‑function (non‑negative sequence) that minimizes
``evaluate_sequence``.

The algorithm works in three phases:

1️⃣  **Broad seeding** – for a variety of lengths we generate many
    candidate distributions: analytic families (uniform, arcsine,
    sine, cosine, quadratic, beta‑edge, edge‑weighted, exponential‑edge,
    Gaussian‑edge, edge‑tophat), random Dirichlet vectors and random
    convex mixtures of several bases.  The best ~80 seeds are kept.

2️⃣  **Continuous optimisation** – each of the top seeds is refined by
    *smooth* exponentiated‑gradient descent on a soft‑max approximation
    of the L∞‑convolution, followed by an exact sub‑gradient descent
    (exponentiated‑gradient on the true max‑convolution).  Temperature
    and learning‑rate schedules are tuned to give a long, low‑temperature
    tail.

3️⃣  **Combinatorial polishing** – a greedy pairwise mass‑transfer
    hill‑climber (and occasional three‑way transfers) is run on the
    current best vector to squeeze out any remaining improvement.

All phases respect the overall time budget (default 1000 s).  The
function finally returns a plain Python list; ``evaluate_sequence``
will re‑normalise it internally.
"""

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# FFT helpers for convolution and related operations

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_cross_correlate_valid(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Valid linear cross‑correlation (w ⋆ a)[j] = Σ_i w[i+j]·a[i],
    returned array has length len(a).
    """
    m = w.shape[0]
    n = a.shape[0]
    size = 1 << ((m + n - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    A = np.fft.rfft(a, n=size)
    corr = np.fft.irfft(W * np.conj(A), n=size)
    return corr[:n]  # valid region


def _fft_sliding_window_sum(w: np.ndarray, win: int) -> np.ndarray:
    """
    For each start j∈[0,win‑1] compute Σ_{i=j}^{j+win‑1} w[i].
    Returns an array of length ``win``.
    """
    m = w.shape[0]
    size = 1 << ((m + win - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    O = np.fft.rfft(np.ones(win, dtype=np.float64), n=size)
    conv = np.fft.irfft(W * O, n=size)
    start = win - 1
    return conv[start:start + win]


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, i.e. sum == 1)
# ----------------------------------------------------------------------


def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_beta_edge(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta‑type density  w_i ∝ (i·(n‑i))^{α‑1}  (α>0)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = (i * (n - i)) ** (alpha - 1.0)
    vals = np.clip(vals, 0.0, None)
    s = vals.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    vals /= s
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·(n‑i))^{‑p}  (p>0)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = (i * (n - i)) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    w /= w_sum
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_gaussian_edge(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(- (i / sigma) ** 2) + np.exp(- ((n - 1 - i) / sigma) ** 2)
    w /= w.sum()
    return w


def _distribution_tophat(n: int, k: int) -> np.ndarray:
    """Uniform mass on the first and last ``k`` entries."""
    k = max(1, min(k, n // 2))
    w = np.zeros(n, dtype=np.float64)
    w[:k] = 1.0
    w[-k:] = 1.0
    w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Exact objective (sum = 1 ⇒ 2·n·max(conv))
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Exact objective for a probability vector (sum == 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth (soft‑max) exponentiated‑gradient descent
# ----------------------------------------------------------------------


def _smooth_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    tau_start: float = 0.02,
    tau_end: float = 0.0005,
    lr0: float = 0.6,
    eval_every: int = 250,
) -> tuple[np.ndarray, float]:
    """
    Soft‑max (temperature τ) EG descent.
    τ anneals linearly from ``tau_start`` to ``tau_end``.
    """
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] smooth it={it} new best {best_score:.6f}")

        # temperature schedule
        frac = (time.time() - start) / (deadline - start)
        tau = tau_start * (1.0 - frac) + tau_end * frac

        # soft‑max weights
        maxc = conv.max()
        shifted = conv - maxc
        # prevent overflow – shift already taken care of
        exp_vals = np.exp(shifted / tau)
        w = exp_vals / exp_vals.sum()               # length 2n‑1

        # gradient of the smoothed objective (derived analytically)
        term1 = _fft_cross_correlate_valid(w, a)      # Σ w[i+j]·a[i]
        win_sum = _fft_sliding_window_sum(w, n)       # Σ_{k=j}^{j+n‑1} w[k]
        grad = term1 + a * win_sum

        # EG update
        eta = lr0 / math.sqrt(it)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = _distribution_uniform(n)
        else:
            a /= s

        if it % eval_every == 0:
            pass  # already evaluated above

    return best_seq, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient EG descent (same as original)
# ----------------------------------------------------------------------


def _subgradient_descent_exact(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 150,
) -> tuple[np.ndarray, float]:
    """EG descent on the genuine L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(it)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = _distribution_uniform(n)
        else:
            a /= s

        if it % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] exact it={it} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise (and tiny three‑way) hill‑climber
# ----------------------------------------------------------------------


def _pairwise_swap(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass from one index to another if it improves."""
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")

        # occasional three‑way transfer (cheap)
        if trials % 1000 == 0 and n > 2:
            i2, j2 = np.random.choice(np.where(a > 1e-14)[0], size=2, replace=False)
            delta2 = a[i2] * np.random.rand() * 0.30
            new_a2 = a.copy()
            new_a2[i2] -= delta2
            new_a2[j2] += delta2
            cur2 = _objective(new_a2)
            if cur2 < best_score - 1e-14:
                a = new_a2
                best_score = cur2
                best_seq = a.copy()
                print(f"[{time.time():.1f}] triple i={i2}→j={j2} δ={delta2:.6g} new best {best_score:.6f}")

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Return a non‑negative step‑function that (hopefully) beats 1.503."""
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    overall_best_seq = None          # numpy array, sum == 1
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, vec)

    # --------------------------------------------------------------
    # 1.  Broad seeding over several lengths
    # --------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000, 16000, 20000]

    # grids for parametric families
    alpha_grid = np.linspace(0.07, 0.45, 9)          # for beta‑edge
    p_grid = np.concatenate(
        [np.logspace(-3, -0.5, 12), np.logspace(-0.5, 0.0, 9)]
    )
    c_grid = np.logspace(-3, -0.1, 10)              # exponential‑edge
    sigma_grid = np.logspace(-1.0, 1.5, 9)          # Gaussian‑edge
    k_grid = [1, 2, 3, 4, 6, 8, 12, 20]            # tophat half‑width
    lam_grid = np.linspace(0.0, 1.0, 21)            # mixing with uniform

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families (no extra parameters)
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
        ):
            cand = base_fn(n)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # beta‑edge (α)
        for alpha in alpha_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_edge(n, float(alpha))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(beta, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] len={n} β α={alpha:.3g} lam={lam:.2f} score={sc:.6f}")

        # edge‑weighted (p)
        for p in p_grid:
            if time.time() >= deadline:
                break
            edge = _distribution_edge_weighted(n, float(p))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(edge, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] len={n} edge p={p:.4g} lam={lam:.2f} score={sc:.6f}")

        # exponential‑edge (c)
        for c in c_grid:
            if time.time() >= deadline:
                break
            exp_edge = _distribution_exponential_edge(n, float(c))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(exp_edge, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] len={n} exp c={c:.4g} lam={lam:.2f} score={sc:.6f}")

        # Gaussian‑edge (σ)
        for sigma in sigma_grid:
            if time.time() >= deadline:
                break
            gauss = _distribution_gaussian_edge(n, float(sigma))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(gauss, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] len={n} gauss σ={sigma:.4g} lam={lam:.2f} score={sc:.6f}")

        # Edge tophat (k)
        for k in k_grid:
            if time.time() >= deadline:
                break
            th = _distribution_tophat(n, int(k))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(th, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] len={n} tophat k={k} lam={lam:.2f} score={sc:.6f}")

        # Random Dirichlet seeds (different concentration)
        for _ in range(12):
            if time.time() >= deadline:
                break
            conc = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, conc))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} conc={conc:.3f} score={sc:.6f}")

        # Random mixtures of already‑generated bases
        # First collect a small pool of base vectors for this n
        base_pool = []
        for _ in range(20):
            # pick a random layout among families above
            which = rng.choice(['beta', 'edge', 'exp', 'gauss', 'tophat'])
            if which == 'beta':
                a = _distribution_beta_edge(n, max(0.05, min(0.5, rng.normal(0.2, 0.1))))
            elif which == 'edge':
                a = _distribution_edge_weighted(n, max(1e-5, rng.exponential(0.2)))
            elif which == 'exp':
                a = _distribution_exponential_edge(n, max(1e-4, rng.exponential(0.2)))
            elif which == 'gauss':
                a = _distribution_gaussian_edge(n, max(0.1, rng.exponential(0.3)))
            else:  # tophat
                kk = max(1, int(rng.integers(1, n // 4)))
                a = _distribution_tophat(n, kk)
            base_pool.append(a)

        # now mix 2‑3 randomly chosen bases
        for _ in range(30):
            if time.time() >= deadline:
                break
            m = rng.integers(2, 4)                # 2 or 3 components
            idxs = rng.choice(len(base_pool), size=m, replace=False)
            weights = rng.dirichlet(np.ones(m))
            mix = np.zeros(n, dtype=np.float64)
            for w, idx in zip(weights, idxs):
                mix += w * base_pool[idx]
            # optional tiny uniform blend
            lam = rng.random() * 0.2
            mix = _mix_with_uniform(mix, lam)
            sc = _objective(mix)
            top_candidates.append((sc, mix.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, mix.copy()
                print(f"[{time.time():.1f}] rand mix len={n} score={sc:.6f}")

        # keep only the most promising seeds (memory bound)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:80]

    # --------------------------------------------------------------
    # 2.  Incorporate previous global construction if available
    # --------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / s
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3.  Fallback (unlikely)
    # --------------------------------------------------------------
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 4.  Smooth descent + exact sub‑gradient on the top seeds
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    # allocate at most 70 % of remaining time to this phase
    smooth_budget = remaining * 0.70
    if smooth_budget > 30.0 and top_candidates:
        # keep the 5 best seeds
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = smooth_budget / max(1, len(seeds))
        print(f"[{time.time():.1f}] smooth+exact on {len(seeds)} seeds (≈{per_seed:.1f}s each)")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            # ---- smooth descent (60 % of per‑seed time) ----------
            deadline_smooth = time.time() + per_seed * 0.60
            a_opt, sc_opt = _smooth_descent(
                seed_seq,
                deadline_smooth,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                tau_start=0.02,
                tau_end=0.0005,
                lr0=0.6,
                eval_every=250,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after smooth seed {idx} new best {overall_best_score:.6f}")

            # ---- exact sub‑gradient (40 % of per‑seed time) -------
            deadline_sub = time.time() + per_seed * 0.40
            a_opt, sc_opt = _subgradient_descent_exact(
                overall_best_seq,
                deadline_sub,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                eval_every=200,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after exact polish seed {idx} new best {overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 5.  Final greedy hill‑climber
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 6.  Report and return
    # --------------------------------------------------------------
    print(
        f"Final best objective (2·n·max(conv)) = {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    # Return as plain Python list – evaluate_sequence will re‑normalise.
    return overall_best_seq.tolist()
