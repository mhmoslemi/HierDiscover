The score returned by `evaluate_sequence` is  

\[
\text{score}= \frac{2\,n\,\max_k (a\star a)[k]}{\bigl(\sum_i a_i\bigr)^2}\; .
\]

Because the denominator scales quadratically with the whole vector, the
objective is **scale‑invariant** – only the *shape* of the non‑negative
vector matters.  
The problem is therefore:

*Choose a probability vector* \(a\in\mathbb{R}_{\ge0}^n,\;\sum a_i=1\) *that
minimises the peak of its auto‑convolution* \(c_k=(a\star a)[k]\).

The code below implements a **multi‑stage search** that improves on the
previous runner:

1. **Diverse seeds** – several analytic families, random Dirichlet draws,
   and a *sorted Dirichlet* (large entries concentrated at one end,
   small entries at the other) which is known to reduce the inner product
   with the reversed vector.
2. **Heavy mirror descent** – the original entropy mirror descent that
   uses only the most‑binding constraints (`active_frac`).  This stage
   quickly finds a good neighbourhood.
3. **Smooth‑max gradient descent** – a new stage that replaces the hard
   active‑set by a **soft‑max weighting** of *all* convolution entries.
   The gradient can be evaluated with two FFTs, thus keeping the cost
   \(O(n\log n)\).  The temperature (β) is increased during the run so
   that the method first moves in a smooth landscape and later focuses
   on the true maximum.
4. **Jitter & restart** – occasional exponential jitter followed by a
   short smooth‑max run to escape plateaus.
5. **Upsampling** – the current best vector is linearly interpolated to a
   *larger* length (up to 50 000).  A further smooth‑max run on this
   longer vector yields a lower peak because the theoretical optimum
   approaches a constant < 2 when \(n\to\infty\).
6. **Weighted pairwise refinement** – greedy mass transfers guided by the
   location of the current maximal convolution entry.
7. **Cheap simulated annealing** – final Metropolis‑type local search.

All stages respect a global time budget (default 1000 s).  The function
returns a plain Python list of floats that can be handed directly to
`evaluate_sequence`.

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution x*y (length len(x)+len(y)-1) via FFT."""
    n, m = x.shape[0], y.shape[0]
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())
    X = rfft(x, n=size)
    Y = rfft(y, n=size)
    conv = irfft(X * Y, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a with sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.006) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_sorted_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw that is sorted increasingly – large mass at the end."""
    a = _distribution_dirichlet(n, rng)
    a.sort()                     # smallest first, largest last
    return a


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, where U is the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Force exact left‑right symmetry and project onto the simplex."""
    sym = (seq + seq[::-1]) * 0.5
    return sym / sym.sum()

# 2. Heavy mirror‑descent (identical to the previous implementation) ---
def _subgradient_descent_momentum(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    active_frac: float = 0.04,
    max_active: int = 30,
    beta: float = 0.9,
    eval_interval: int = 120,
    patience: int = 2000,
) -> tuple[np.ndarray, float]:
    """Entropy‑mirror descent using a hard active set."""
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    ind_grid = np.arange(n, dtype=np.int32)

    while time.time() < deadline and t < 500_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            top_idx = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top_idx]

        grad = np.zeros_like(a)
        for k in act:
            shift = k - ind_grid
            mask = (shift >= 0) & (shift < n)
            grad[mask] += 2.0 * a[shift[mask]]

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * momentum)

        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} new global best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            jitter = rng.exponential(scale=0.004, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

# 3. Smooth‑max gradient descent (new stage) -------------------------
def _smooth_max_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_lr: float = 0.8,
    beta_start: float = 5.0,
    beta_factor: float = 1.05,
    eval_interval: int = 80,
    patience: int = 2500,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a soft‑max approximation of the max‑convolution.
    The temperature β grows geometrically, turning the smooth objective
    gradually into the true max.  The update uses exponentiated‑gradient
    (entropy mirror) steps and a simple patience‑based jitter restart.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    beta = beta_start
    lr = init_lr

    while time.time() < deadline and t < 300_000:
        t += 1

        # 1) full convolution
        conv = _fft_self_convolve(a)
        max_c = conv.max()

        # 2) soft‑max weights
        # keep numbers well‑scaled: subtract max before exponentiation
        w = np.exp(beta * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
            w_sum = 1.0
        w /= w_sum                     # now Σ w_k = 1

        # 3) gradient = 2 * (w * a_rev)[n‑1 : n‑1+n]
        a_rev = a[::-1]
        grad_full = _fft_convolve(w, a_rev)          # length = 2n‑1 + n‑1
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]    # size n

        # 4) entropy mirror step
        eta = lr / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # 5) project back onto simplex (just renormalise)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # 6) occasional evaluation
        if t % eval_interval == 0:
            cur = 2.0 * n * max_c
            if cur + 1e-14 < best_score:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] smooth it={t:,} new best {best_score:.6f}")
            else:
                no_improve += 1

        # 7) increase temperature slowly – focuses more on the true max
        if t % 200 == 0:
            beta = min(200.0, beta * beta_factor)

        # 8) patience based jitter
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.006, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0
            beta = beta_start            # cool down again

    return best_seq, best_score

# Weighted pairwise refinement -------------------------------------------
def _pairwise_weighted_refine(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer guided by the location of the maximal convolution."""
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0

    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        conv = _fft_self_convolve(a)
        k_max = int(conv.argmax())

        # impact values L_i = a_{k_max-i} where defined
        L = np.zeros(n, dtype=np.float64)
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        idx = np.arange(low, high + 1)
        L[idx] = a[k_max - idx]

        # source i (high impact) and destination j (low impact)
        if L.sum() > 0.0:
            probs_i = L / L.sum()
            i = rng.choice(n, p=probs_i)
        else:
            positives = np.where(a > 1e-12)[0]
            if positives.size == 0:
                break
            i = rng.choice(positives)

        Lmax = L.max()
        probs_j = (Lmax - L) + 1e-12
        probs_j /= probs_j.sum()
        j = rng.choice(n, p=probs_j)
        if j == i:
            j = (j + 1) % n

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] weighted swap {i}->{j} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score

# Cheap simulated annealing ---------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Very lightweight Metropolis‑type local search."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling helper -------------------------------------------------------
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The algorithm follows the stages described
    above and respects the given time budget.
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Seed generation (moderate lengths)
    lengths = [1800, 2200, 2600, 3000, 3500, 4200]   # a bit smaller – more seeds
    p_grid = np.concatenate(
        [
            np.logspace(-3, -1, 9),   # 0.001 … 0.1
            np.logspace(-1, 0, 9),   # 0.1 … 1
            np.logspace(0, 0.5, 7),  # 1 … 3.16
        ]
    )
    lam_grid = np.linspace(0.0, 0.9, 10)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] sampling length {n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (uniform/arcsine) {sc:.6f}")

        # parameterised families
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            edge2 = _distribution_edge_inversepow(n, p)
            sine = _distribution_sine_weighted(n, p)

            for base in (beta, edge, edge2, sine):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # asymmetric beta
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = rng.uniform(0.001, 3.0)
            b = rng.uniform(0.001, 3.0)
            cand = _distribution_beta_asym(n, a, b)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best asym‑beta len={n} a={a:.3g} b={b:.3g} score={sc:.6f}"
                )

        # random Dirichlet (symmetrised and unsymmetrised)
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_dirichlet(n, rng)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best Dirichlet len={n} score={sc:.6f}")

            # sorted version – large mass at one side, small at the other
            cand2 = _distribution_sorted_dirichlet(n, rng)
            sc2 = _objective(cand2)
            top_candidates.append((sc2, cand2.copy()))
            if sc2 < overall_best_score:
                overall_best_score, overall_best_seq = sc2, cand2.copy()
                print(f"[{time.time():.1f}] new global best Sorted‑Dirichlet len={n} score={sc2:.6f}")

        # sparse random
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}")

        # keep only the very best seeds (≈ 15)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:15]

    # ------------------------------------------------------------------
    # 2.  Incorporate any previously supplied construction
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Guard against empty result
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Heavy mirror‑descent on a handful of top seeds (moderate size)
    remaining = deadline - time.time()
    if remaining > 40.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]]
        per_seed = remaining * 0.55 / max(1, len(seeds))
        print(
            f"[{time.time():.1f}] heavy mirror‑descent on {len(seeds)} seeds, ≈{per_seed:.1f}s each"
        )
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            a_opt, sc_opt = _subgradient_descent_momentum(
                seed_seq.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=1.0,
                active_frac=0.05,
                max_active=40,
                beta=0.9,
                eval_interval=100,
                patience=1800,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after descent seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Smooth‑max refinement (still on the moderate size)
    remaining = deadline - time.time()
    if remaining > 30.0:
        sub_dead = time.time() + remaining * 0.3
        print(f"[{time.time():.1f}] smooth‑max refinement (size={len(overall_best_seq)})")
        overall_best_seq, overall_best_score = _smooth_max_descent(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_lr=0.9,
            beta_start=5.0,
            beta_factor=1.04,
            eval_interval=80,
            patience=2000,
        )
        print(f"[{time.time():.1f}] after smooth‑max score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Upsample to a larger length (if we have the budget)
    remaining = deadline - time.time()
    if remaining > 120.0:
        # aim for a length up to ~50 000 – still cheap with FFTs
        large_n = min(50000, max(20000, len(overall_best_seq) * 2))
        print(f"[{time.time():.1f}] up‑sampling to {large_n} points")
        up_seq = _upsample(overall_best_seq, large_n)

        sub_dead = time.time() + remaining * 0.55
        # a short mirror‑descent pass (helps get a decent starting point)
        up_seq, up_score = _subgradient_descent_momentum(
            up_seq,
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.6,
            active_frac=0.04,
            max_active=35,
            beta=0.92,
            eval_interval=80,
            patience=1800,
        )
        # now a more precise smooth‑max run
        sub_dead = time.time() + remaining * 0.30
        up_seq, up_score = _smooth_max_descent(
            up_seq,
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_lr=0.5,
            beta_start=4.0,
            beta_factor=1.05,
            eval_interval=70,
            patience=2500,
        )
        if up_score < overall_best_score:
            overall_best_score, overall_best_seq = up_score, up_seq.copy()
            print(f"[{time.time():.1f}] up‑sampled refinement yielded new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Jitter & restart cycles (escape plateaus)
    while time.time() < deadline - 15.0:
        rem = deadline - time.time()
        jitter_scale = 0.018
        jitter = rng.exponential(scale=jitter_scale, size=overall_best_seq.shape)
        cand = overall_best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sub_dead = time.time() + min(18.0, rem * 0.4)
        cand, cand_score = _smooth_max_descent(
            cand,
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_lr=0.5,
            beta_start=5.0,
            beta_factor=1.06,
            eval_interval=60,
            patience=1500,
        )
        if cand_score < overall_best_score:
            overall_best_score, overall_best_seq = cand_score, cand.copy()
            print(f"[{time.time():.1f}] jitter‑restart improved to {overall_best_score:.6f}")
        else:
            break

    # ------------------------------------------------------------------
    # 8.  Symmetrisation test (many good solutions are nearly symmetric)
    sym_seq = _symmetrize(overall_best_seq)
    sym_score = _objective(sym_seq)
    if sym_score + 1e-14 < overall_best_score:
        overall_best_score, overall_best_seq = sym_score, sym_seq
        print(f"[{time.time():.1f}] symmetrised version better: {sym_score:.6f}")

    # ------------------------------------------------------------------
    # 9.  Weighted pairwise deterministic refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] weighted pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_weighted_refine(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=250_000,
        )

    # ------------------------------------------------------------------
    # 10. Final simulated‑annealing clean‑up
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
