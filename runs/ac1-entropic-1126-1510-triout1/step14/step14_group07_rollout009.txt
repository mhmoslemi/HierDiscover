"""
Optimisation of a non‑negative step‑function (a probability vector) for
the objective inside ``evaluate_sequence``.

Approach
--------
1️⃣  **Broad seeding** – generate many analytically‑motivated families
    (uniform, sine‑weighted, left‑exponential, power‑law, linear /
    quadratic decay, logistic, Dirichlet samples, sparse random).  For
    each length in a modest list the best few (≈20) candidates are kept.

2️⃣  **Mirror‑descent (Adam) fine‑tuning** – each retained seed is
    improved by Adam optimisation on the exact objective using the
    active‑set sub‑gradient of the max‑convolution (soft‑max on the log‑
    weights guarantees feasibility).  A short run (≈10 k iterations) is
    performed for every seed; the best refined seed then undergoes a
    long run (≈200 k iterations) that uses a tighter active set.

3️⃣  **Local polishing** – greedy pairwise mass‑transfer swaps followed
    by a cheap simulated‑annealing style search clean up any remaining
    irregularities.

All stages respect the global time budget (default 1000 s).  The final
vector is returned as a plain Python list; ``evaluate_sequence`` will
re‑normalise it anyway.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


# ----------------------------------------------------------------------
# Convolution utilities (full linear convolution via FFT)
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())   # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum(a)=1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families returning probability vectors (sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    w /= w.sum()
    return w


def _distribution_exponential_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-rate * i)
    w /= w.sum()
    return w


def _distribution_exponential_center(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(i - half))
    w /= w.sum()
    return w


def _distribution_power_law(n: int, alpha: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)  # start at 1 to avoid div‑by‑0
    w = i ** (-alpha)
    w /= w.sum()
    return w


def _distribution_linear_decreasing(n: int) -> np.ndarray:
    w = np.arange(n, 0, -1, dtype=np.float64)  # n, n‑1, …, 1
    w /= w.sum()
    return w


def _distribution_quadratic_decreasing(n: int) -> np.ndarray:
    w = np.arange(n, 0, -1, dtype=np.float64) ** 2
    w /= w.sum()
    return w


def _distribution_logistic_decreasing(n: int, centre: float, scale: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    # avoid overflow in exp
    z = (i - centre) / scale
    z = np.clip(z, -50, 50)
    w = 1.0 / (1.0 + np.exp(z))
    w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator, conc: float) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a.astype(np.float64)


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.006) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Adam‑based mirror descent (exact objective, active‑set sub‑gradient)
def _adam_refine(
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 120_000,
    lr0: float = 0.8,
    active_frac: float = 0.025,
    max_active: int = 30,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_interval: int = 250,
    best_seq: np.ndarray | None = None,
    best_score: float = float("inf"),
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex via a soft‑max parametrisation.
    The sub‑gradient is taken over the `max_active` largest entries of the
    self‑convolution that lie within `active_frac` of the current maximum.
    """
    n = init_a.shape[0]
    x = np.log(np.maximum(init_a, 1e-12))   # log‑weights
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    ind_grid = np.arange(n, dtype=np.int32)
    t = 0

    if best_seq is None:
        best_seq = init_a.copy()
        best_score = _objective(init_a)

    while time.time() < deadline and t < max_iters:
        t += 1
        a = _softmax(x)                     # stays on simplex
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        # active set: entries within a fraction of the current max
        thresh = max_c * (1.0 - active_frac)
        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            top = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top]

        # sub‑gradient wrt a (linear combination of active convolution rows)
        grad_a = np.zeros_like(a)
        for idx_k in act:
            offset = idx_k - ind_grid
            mask = (offset >= 0) & (offset < n)
            grad_a[mask] += 2.0 * a[offset[mask]]

        # transform gradient to log‑weight space (mirror‑descent formula)
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update with a decaying step‑size
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = lr0 / math.sqrt(t)
        x -= eta * m_hat / (np.sqrt(v_hat) + eps)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] adam it={t:,} new best {best_score:.6f}")

    # final projection (soft‑max already normalises, but guard)
    best_seq = _project_onto_simplex(best_seq, s=1.0)
    return best_seq, best_score

# Greedy pairwise local refinement (as in original code)
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)            # donor
        j = rng.integers(n - 1)        # receiver (different)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.01,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

    if time.time() < deadline:
        remaining = deadline - time.time()
        long_deadline = time.time() + max(0.0, remaining - 30.0)   # leave a little time for polishing
        print(f"[{time.time():.1f}] long Adam descent (≈{remaining-30:.1f}s remaining)")
        overall_best_seq, overall_best_score = _adam_refine(
            overall_best_seq.copy(),
            long_deadline,
            max_iters=250_000,
            lr0=0.9,
            active_frac=0.02,
            max_active=40,
            beta1=0.9,
            beta2=0.999,
            eval_interval=250,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] after long descent best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=200_000,
        )

    # ------------------------------------------------------------------
    # 5️⃣  final simulated‑annealing polishing
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing polishing")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.02,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(f"Final best score (exact) ≈ {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # The evaluator will normalise – we keep a proper probability vector.
    return overall_best_seq.tolist()
