def _smooth_max_descent(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step_factor = 0.15
    # schedule
    beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0]
    t_global = 0
    for beta in beta_schedule:
        max_iters = 3000
        for _ in range(max_iters):
            if time.time() >= deadline:
                return best_seq, best_score
            t_global += 1
            conv = _fft_self_convolve(a)
            max_val = conv.max()
            shifted = conv - max_val
            exp_beta = np.exp(beta * shifted)
            w = exp_beta / exp_beta.sum()
            a_rev = a[::-1]
            grad_conv = _fft_convolve(w, a_rev)
            grad = 2.0 * grad_conv[n-1:n-1+n]
            grad_J = 2.0 * n * grad
            m = beta1 * m + (1.0 - beta1) * grad_J
            v = beta2 * v + (1.0 - beta2) * (grad_J ** 2)
            m_hat = m / (1.0 - beta1 ** t_global)
            v_hat = v / (1.0 - beta2 ** t_global)
            step = step_factor / (np.sqrt(v_hat) + eps)
            a -= step * m_hat
            a = _project_onto_simplex(a)
            # periodic evaluation
            if t_global % 150 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-13:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] β={beta:.1f} it={t_global} new best {best_score:.6f}")
    return best_seq, best_score
```

"""
Hybrid optimisation for the step‑function minimisation task.

A multi‑stage approach:
1️⃣  Enumerate a rich library of analytic families (uniform, arcsine,
    sine, cosine, quadratic, beta, edge‑weighted, exponential‑edge,
    additive‑power) on several lengths and keep the best few seeds.
2️⃣  Apply a smooth‑max (log‑sum‑exp) mirror‑descent optimiser (Adam)
    with a continuation on the smoothing parameter β (0.5 → 500).  The
    iterates stay on the probability simplex.
3️⃣  Refine the best solution with an exponentiated‑gradient sub‑gradient
    method on the exact L∞‑objective.
4️⃣  A final greedy pairwise mass‑transfer hill‑climber that only
    accepts strict improvements.

The whole pipeline respects a user‑supplied time budget (default 1000 s)
and returns a plain Python list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
#  Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
#  FFT‑based convolutions (real vectors)
# ------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using real FFT."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    c = np.fft.irfft(X * Y, n=size)[:n]
    return c


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a * a (full)."""
    return _fft_convolve(a, a)


# ------------------------------------------------------------
#  Exact objective (Σa = 1)
# ------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ------------------------------------------------------------
#  Analytic families
# ------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    w /= w.sum()
    return w


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    total = w.sum()
    if total == 0.0:
        return _distribution_uniform(n)
    w /= total
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i·(n‑i‑1))^{-p}  (i counted from 0.5)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = n - np.arange(n, dtype=np.float64) - 0.5
    w = (i * j) ** (-p)
    tot = w.sum()
    if tot == 0.0:
        return _distribution_uniform(n)
    w /= tot
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_additive_power(n: int, p: float) -> np.ndarray:
    """w_i ∝ i^{-p} + (n‑i‑1)^{-p}  (i counted from 0.5)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = n - np.arange(n, dtype=np.float64) - 0.5
    w = i ** (-p) + j ** (-p)
    tot = w.sum()
    if tot == 0.0:
        return _distribution_uniform(n)
    w /= tot
    return w


def _distribution_edge_two_point_mix(n: int, lam: float) -> np.ndarray:
    """Mixture λ·uniform + (1‑λ)·{½δ₀+½δₙ₋₁}."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    edges = np.zeros(n, dtype=np.float64)
    edges[0] = 0.5
    edges[-1] = 0.5
    return lam * uniform + (1.0 - lam) * edges


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    return lam * np.full(n, 1.0 / n, dtype=np.float64) + (1.0 - lam) * base


# ------------------------------------------------------------
#  Sparse random seed helper
# ------------------------------------------------------------
def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.01) -> np.ndarray:
    """Sparse random vector with prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.random(k)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    s = a.sum()
    if s == 0.0:
        a[0] = 1.0
        s = 1.0
    a /= s
    return a


# ------------------------------------------------------------
#  Smoothed‑max mirror descent (Adam + β‑continuation)
# ------------------------------------------------------------
def _smooth_max_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    A schedule of β balances smoothness → exactness.
    """
    a = a0.copy()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step_factor = 0.15

    beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0]
    iter_global = 0

    for beta in beta_schedule:
        max_iters = 3000
        for _ in range(max_iters):
            if time.time() >= deadline:
                return best_seq, best_score
            iter_global += 1

            conv = _fft_self_convolve(a)
            max_val = conv.max()
            shifted = conv - max_val        # ≤ 0, stabilises exp
            exp_beta = np.exp(beta * shifted)
            w = exp_beta / exp_beta.sum()   # weights, Σw = 1

            a_rev = a[::-1]
            grad_conv = _fft_convolve(w, a_rev)          # length 3n‑2
            grad = 2.0 * grad_conv[n - 1 : n - 1 + n]    # size n
            grad_J = 2.0 * n * grad

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_J
            v = beta2 * v + (1.0 - beta2) * (grad_J ** 2)
            m_hat = m / (1.0 - beta1 ** iter_global)
            v_hat = v / (1.0 - beta2 ** iter_global)
            step = step_factor / (np.sqrt(v_hat) + eps)
            a -= step * m_hat
            a = _project_onto_simplex(a)

            # periodic exact evaluation
            if iter_global % 150 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-14:
                    best_score, best_seq = cur, a.copy()
                    print(
                        f"[{time.time():.1f}] β={beta:.1f} it={iter_global} new best {best_score:.6f}"
                    )
    return best_seq, best_score


# ------------------------------------------------------------
#  Exact sub‑gradient descent (exponentiated gradient)
# ------------------------------------------------------------
def _subgradient_descent_exact(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true L∞‑objective."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        # sub‑gradient: sum of shifted copies over all max indices
        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)            # maintain positivity
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score, best_seq = cur, a.copy()
                print(
                    f"[{time.time():.1f}] exact subgrad it={t} new global best {best_score:.6f}"
                )
    return best_seq, best_score


# ------------------------------------------------------------
#  Greedy pairwise mass‑transfer hill‑climber
# ------------------------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly relocate a fraction of mass; accept only strict improvements."""
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score, best_seq = cur, a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score
