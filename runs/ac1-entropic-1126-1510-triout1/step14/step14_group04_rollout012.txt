"""
Propose a non‑negative step‑function (list of floats) that minimises

    evaluate_sequence(sequence) = 2·n·max(conv(a,a)) / (∑a)²

The algorithm consists of three phases:

1️⃣  **Broad analytic sweep** – generate a diverse set of candidate
    vectors from many parametric families (uniform, arcsine, sine,
    cosine, quadratic, polynomial, beta, edge‑weighted, exponential‑,
    and Gaussian‑edge).  All candidates are projected onto the
    probability simplex, evaluated (sum = 1) and the best few are kept.

2️⃣  **Smooth‑max mirror descent** – minimise the exact objective via a
    log‑sum‑exp surrogate.  An Adam‑style update with projection onto the
    simplex is performed while the smoothing parameter β is increased
    (β ∈ {0.5,…,5000}).  This yields a refined vector close to a local
    optimum of the non‑smooth problem.

3️⃣  **Exact sub‑gradient polishing + greedy pairwise swaps** – an
    exponentiated‑gradient descent on the true L∞‑objective followed by
    a cheap hill‑climber that transfers a random fraction of mass
    between two positions only when the exact objective improves.

All stages respect the global time budget (default 1000 s).  The final
vector is returned as a plain Python list; `evaluate_sequence` will
re‑scale it internally.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
#  FFT‑based convolution helpers (real inputs)
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    return _fft_convolve(a, a)

# ----------------------------------------------------------------------
#  Exact objective (sum a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# ----------------------------------------------------------------------
#  Analytic families (all return a probability vector)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_polynomial(n: int, r: float) -> np.ndarray:
    """Symmetric polynomial shape: (1‑x²)^r, x∈[‑1,1]."""
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = np.clip(1.0 - x * x, 0.0, None) ** r
    total = w.sum()
    if total == 0.0:
        return _distribution_uniform(n)
    w /= total
    return w


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric beta‑type distribution (α,β may differ)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    total = w.sum()
    if total == 0.0:
        return _distribution_uniform(n)
    w /= total
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i+½)^{‑p}·(n‑i‑½)^{‑p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    total = w.sum()
    if total == 0.0:
        return _distribution_uniform(n)
    w /= total
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    """w_i ∝ exp(‑c·i) + exp(‑c·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_gaussian_edge(n: int, sigma: float) -> np.ndarray:
    """Gaussian‑like spikes at both ends: exp(‑(i/(σ·(n‑1)))²) + symmetric."""
    i = np.arange(n, dtype=np.float64) / (n - 1)
    w = np.exp(- (i ** 2) / (2 * sigma * sigma))
    w += np.exp(- ((1 - i) ** 2) / (2 * sigma * sigma))
    w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.01) -> np.ndarray:
    """Sparse random vector with prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


# ----------------------------------------------------------------------
#  Smooth‑max mirror descent (Adam + β‑continuation)
# ----------------------------------------------------------------------
def _smooth_max_descent(a0: np.ndarray, time_limit: float,
                        best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step_factor = 0.18           # a little larger than the original 0.15

    # β schedule (log‑space, extended)
    beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0,
                     200.0, 500.0, 1000.0, 2000.0, 5000.0]

    it = 0
    for beta in beta_schedule:
        # give each β a finite budget (≈2500 iters) but exit early if time is low
        while time.time() < time_limit:
            it += 1
            # full self‑convolution
            conv = _fft_self_convolve(a)
            max_val = conv.max()
            # soft‑max weights
            shifted = conv - max_val                # ≤0
            exp_beta = np.exp(beta * shifted)
            w = exp_beta / exp_beta.sum()           # Σw = 1

            # gradient of the smoothed max (see analysis)
            a_rev = a[::-1]
            grad_conv = _fft_convolve(w, a_rev)     # length 3n‑2
            grad = 2.0 * grad_conv[n - 1 : n - 1 + n]   # size n
            grad_J = 2.0 * n * grad

            # Adam step
            m = beta1 * m + (1.0 - beta1) * grad_J
            v = beta2 * v + (1.0 - beta2) * (grad_J ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = step_factor / (np.sqrt(v_hat) + eps)
            a -= step * m_hat
            a = _project_onto_simplex(a, s=1.0)

            # periodic exact evaluation
            if it % 150 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-14:
                    best_score, best_seq = cur, a.copy()
                    print(f"[{time.time():.1f}] β={beta:.1f} it={it} new best {best_score:.6f}")

            # stop after a few thousand steps per β to keep total time bounded
            if it % 3000 == 0:
                break
        if time.time() >= time_limit:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Exact sub‑gradient descent (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _subgradient_descent_exact(a0: np.ndarray, deadline: float,
                               best_seq: np.ndarray, best_score: float,
                               lr0: float = 0.9, eval_every: int = 120) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true L∞‑objective."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        # exact sub‑gradient (sum of shifted copies)
        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)          # diminishing step
        a *= np.exp(-eta * grad)          # multiplicative update keeps positivity
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score, best_seq = cur, a.copy()
                print(f"[{time.time():.1f}] exact subgrad it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Greedy pairwise mass‑transfer hill‑climber
# ----------------------------------------------------------------------
def _pairwise_swap(a0: np.ndarray, deadline: float,
                   best_seq: np.ndarray, best_score: float,
                   max_trials: int | None = None) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * np.random.rand() * 0.45
        if delta < 1e-15:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score, best_seq = cur, a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a step‑function (plain Python list) that (approximately)
    minimises ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.5          # tiny safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    MAX_SEEDS = 40
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, vector)

    # ------------------------------------------------------------
    # 1️⃣ Analytic families sweep
    # ------------------------------------------------------------
    lengths = [800, 1200, 1600, 2000, 2400, 2800, 3200, 3600,
               4000, 4800, 5600, 6400]
    # parameter grids (densified compared to the previous version)
    p_grid = np.concatenate((
        np.logspace(-4, -0.5, 15),
        np.logspace(-0.5, 0.0, 15),
        np.linspace(0.2, 3.0, 12)
    ))
    lam_grid = np.linspace(0.0, 1.0, 21)       # finer mixing
    c_grid = np.logspace(-4, 0, 12)           # exponential‑edge decay
    sigma_grid = np.logspace(-3, -0.2, 10)    # Gaussian‑edge width
    alpha_grid = np.logspace(-0.5, 0.7, 14)   # beta‑family α
    beta_grid = np.logspace(-0.5, 0.7, 14)    # beta‑family β (independent)

    def _maybe_add(seq: np.ndarray):
        nonlocal overall_best_seq, overall_best_score, top_candidates
        sc = _objective(seq)
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] new global best len={len(seq)} score={sc:.6f}")
        # maintain a bounded heap‑like list of top seeds
        if len(top_candidates) < MAX_SEEDS:
            top_candidates.append((sc, seq.copy()))
        else:
            # replace worst if better
            worst_idx = max(range(len(top_candidates)),
                            key=lambda i: top_candidates[i][0])
            if sc < top_candidates[worst_idx][0]:
                top_candidates[worst_idx] = (sc, seq.copy())

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # ------------ deterministic families -------------
        for base_fn in (_distribution_uniform,
                        _distribution_arcsine,
                        _distribution_sine,
                        _distribution_cosine,
                        _distribution_quadratic):
            if time.time() >= deadline:
                break
            cand = base_fn(n)
            _maybe_add(cand)

        # ------------ polynomial family (r‑parameter) -------------
        for r in np.linspace(0.2, 5.0, 12):
            if time.time() >= deadline:
                break
            cand = _distribution_polynomial(n, r)
            _maybe_add(cand)

        # ------------ symmetric beta family -------------
        for alpha in alpha_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_beta(n, float(alpha), float(alpha))
            _maybe_add(cand)

        # ------------ asymmetric beta family (α ≠ β) -------------
        for alpha in alpha_grid[::3]:   # thin out a bit to keep time reasonable
            for beta_ in beta_grid[::3]:
                if time.time() >= deadline:
                    break
                cand = _distribution_beta(n, float(alpha), float(beta_))
                _maybe_add(cand)

        # ------------ edge‑weighted + mix -------------
        for p in p_grid:
            if time.time() >= deadline:
                break
            edge = _distribution_edge_weighted(n, float(p))
            _maybe_add(edge)
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(edge, lam)
                _maybe_add(cand)

        # ------------ exponential‑edge + mix -------------
        for c in c_grid:
            if time.time() >= deadline:
                break
            exp_edge = _distribution_exponential_edge(n, float(c))
            _maybe_add(exp_edge)
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(exp_edge, lam)
                _maybe_add(cand)

        # ------------ Gaussian‑edge + mix -------------
        for sigma in sigma_grid:
            if time.time() >= deadline:
                break
            gauss_edge = _distribution_gaussian_edge(n, float(sigma))
            _maybe_add(gauss_edge)
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(gauss_edge, lam)
                _maybe_add(cand)

        for _ in range(6):
            if time.time() >= deadline:

            alpha = rng.uniform(0.01, 2.0)

            if time.time() >= deadline:

            a = _distribution_random_sparse(n, rng, density=0.008)
            _maybe_add(a)

    # ------------------------------------------------------------

    try:

        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            _maybe_add(prev)
    except Exception:

    # ------------------------------------------------------------
    # 3️⃣  Fallback (should never be needed)
    # ------------------------------------------------------------

        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}")

    # ------------------------------------------------------------

    if remaining > 60.0 and top_candidates:
        # pick the very best few (≤8)
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:8]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] smooth‑max descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_max_descent(
                seed_seq,
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after smooth seed {idx} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------
    # 5️⃣  Exact sub‑gradient refinement (exponentiated‑gradient)
    # ------------------------------------------------------------

    if remaining > 40.0:

            overall_best_seq.copy(),

            best_score=overall_best_score,
            lr0=0.9,
            eval_every=150,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after exact subgrad new best {overall_best_score:.6f}")

    # ------------------------------------------------------------
    # 6️⃣  Final greedy pairwise polishing
    # ------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final pairwise swap polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------
    print(f"Final best objective (2·n·max(conv)) = {overall_best_score:.6f} "
          f"(length={len(overall_best_seq)})")
    # The caller will re‑scale internally; return the raw simplex‑normalised vector.
    return overall_best_seq.tolist()
