    best_seq = a.copy()
    best_score = _objective(a)
    t = 0
    η = 1.0
    lr0 = 1.0  # initial learning rate (log domain)
    while time.time() < deadline:
        # compute b
        b = _fft_self_convolve(a)  # conv a*a
        # compute smooth weights
        eta = η
        # Stabilize
        logw = eta * b
        logw -= np.max(logw)
        p = np.exp(logw)
        p /= p.sum()
        # Compute gradient via cross-correlation
        a_rev = a[::-1]
        conv_rev = _fft_convolve(a_rev, p)  # length n+len(p)-1 = n+2n-2 = 3n-2

        start = len(p)//2  # p length = 2n-1, so start = n-1
        grad = 2 * eta * conv_rev[start:start+len(a)]  # size n
        # learning rate schedule
        t += 1
        lr = lr0 / math.sqrt(t)
        # Mirror update
        a = a * np.exp(-lr * grad)
        a_sum = a.sum()

"""
propose_candidate
~~~~~~~~~~~~~~~~~

A multi‑stage optimiser for the step‑function evaluation problem.

The algorithm works in the following phases (all respecting a 1000 s wall‑time
budget):

1. **Seed generation** – a fairly wide set of analytic families
   (uniform, arcsine, edge‑weighted, sum‑edge‑weighted, symmetric Beta,
   generalized normal, random sparse Dirichlet) is sampled for a moderate
   length (≈12 k).  The 10 best seeds (by the exact objective) are retained.

2. **Smooth max descent** – the non‑smooth objective
   ``2·n·max(conv(a,a))/sum(a)^2`` is approximated by a
   log‑sum‑exp (soft‑max) with a temperature parameter η.  For a fixed η we
   obtain a smooth gradient which is the cross‑correlation of the current
   distribution with the soft‑max weights.  An exponentiated‑gradient (mirror)
   update with momentum is applied, while η is slowly increased so that the
   smooth objective converges to the original max‑objective.

3. **Annealing polish** – a cheap pairwise mass‑transfer simulated annealing
   (as in the reference implementation) runs for the remaining time.

All updates keep the vector non‑negative and normalised to sum = 1, i.e.
the search space is the probability simplex.  The code uses FFT‑based
convolutions (real rFFT) for speed and is written without any
filesystem/network access.

The returned list can be directly fed to the supplied ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  FFT utilities – linear convolution via real FFT
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using rFFT."""
    n = x.shape[0]
    m = y.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()  # next power of two
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[: n + m - 1]


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
#  Exact objective used by the evaluator (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a))/sum(a)^2 (a ≥ 0, sum > 0)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
#  Analytic families – all return a *probability* vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i·(n‑i))^{-p}, i measured from the left edge (½‑offset)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_sum_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ i^{-p} + (n‑i)^{-p} (sum of two one‑sided tails)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (-p)) + (j ** (-p))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(-|x‑½|^{q}) on the centre‑scaled grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric/unsymmetric Beta(α,β) density on the (0,1) interval."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # avoid zero‑powers by clipping x∈(0,1)
    eps = np.finfo(np.float64).tiny
    x = np.clip(x, eps, 1 - eps)
    w = x ** (alpha - 1) * (1.0 - x) ** (beta - 1)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1‑lam) + uniform·lam."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
#  Smooth‑max descent (log‑sum‑exp approximation of the max)
# ----------------------------------------------------------------------
def _smooth_descent(
    a: np.ndarray,
    deadline: float,
    best_score: float,
    eta0: float = 0.8,
    lr0: float = 0.4,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """
    Minimise the exact objective using a smooth‑max surrogate.
    An exponentiated‑gradient (mirror) update with momentum is employed.
    """
    n = a.shape[0]
    # small epsilon to avoid log(0) if we ever need it
    eps = 1e-16
    a = np.maximum(a, eps)
    a /= a.sum()

    best_seq = a.copy()
    best_val = best_score
    t = 0
    eta = eta0               # temperature for the soft‑max
    beta_mom = 0.9           # momentum coefficient
    mom = np.zeros_like(a)

    while time.time() < deadline:
        t += 1
        # 1) self‑convolution
        b = _fft_self_convolve(a)               # length 2·n‑1

        # 2) soft‑max weights p_k ∝ exp(η·b_k)
        w = eta * b
        w_max = w.max()
        exp_w = np.exp(w - w_max)               # stabilised
        p = exp_w / exp_w.sum()                 # shape (2·n‑1,)

        # 3) gradient of the smooth surrogate:
        #    grad_i = 2·η·∑_k p_k·a_{k-i} = 2·η·(a ⋆ p)[i]
        a_rev = a[::-1]                         # reversed a
        conv_rev = _fft_convolve(a_rev, p)      # length 3·n‑2
        start = len(p) // 2                     # = n‑1
        grad = 2.0 * eta * conv_rev[start:start + n]

        # 4) momentum + multiplicative update on the simplex
        mom = beta_mom * mom + (1.0 - beta_mom) * grad
        lr = lr0 / math.sqrt(t)                 # diminishing step size
        a = a * np.exp(-lr * mom)               # mirror step
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # 5) occasional evaluation with the *true* objective
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_val - 1e-14:
                best_val = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth descent it={t} new best {best_val:.6f}")

        # 6) slowly raise η to tighten the approximation to max
        if t % 200 == 0:
            eta = min(eta * 1.03, 80.0)          # cap to avoid overflow

    return best_seq, best_val


# ----------------------------------------------------------------------
#  Pairwise‑mass‑transfer annealing (final polish)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Random pairwise mass moves with a simple temperature schedule."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    while time.time() < deadline:
        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4  # move ≤40 % of source mass
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        accept = False
        if cur < best_score - 1e-14:
            accept = True
        else:
            if temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature):
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score

                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best sum p={p:.4g} lam={lam:.2f} score={sc:.6f}")

    # symmetric Beta family (covers arcsine as α=β=0.5)
    alpha_grid = np.linspace(0.2, 3.0, 15)
    for a in alpha_grid:
        if time.time() >= deadline:
            break
        cand = _distribution_beta(n, a, a)
        sc = _objective(cand)
        top.append((sc, cand.copy()))
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] new best beta α={a:.3f} score={sc:.6f}")

    # generalized normal family
    q_grid = np.linspace(0.5, 3.0, 9)
    for q in q_grid:
        if time.time() >= deadline:
            break
        cand = _distribution_generalized_normal(n, q)
        sc = _objective(cand)
        top.append((sc, cand.copy()))
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] new best gn q={q:.2f} score={sc:.6f}")

    # random sparse Dirichlet seeds (few non‑zero entries)
    for _ in range(6):
        if time.time() >= deadline:
            break
        cand = np.zeros(n, dtype=np.float64)
        k = max(2, int(n * 0.008))           # ≈0.8 % non‑zero
        idx = rng.choice(n, size=k, replace=False)
        cand[idx] = rng.random(k)
        cand /= cand.sum()
        sc = _objective(cand)
        top.append((sc, cand.copy()))
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] new best sparse score={sc:.6f}")

    # keep only the very best seeds
    top.sort(key=lambda x: x[0])
    top = top[:10]

    # ------------------------------------------------------------------
    # 2.  Include the previously submitted construction (if present)
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < best_score:
                best_score, best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Guard against an empty result
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _distribution_uniform(5000)
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Smooth‑max descent on the (few) best seeds
    # ------------------------------------------------------------------
    for seed_score, seed_vec in top:
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # allocate at most 30 s per seed (or whatever is left)
        sub_deadline = time.time() + min(remaining * 0.4, 30.0)
        print(f"[{time.time():.1f}] smooth descent on seed (score={seed_score:.6f})")
        cand_seq, cand_score = _smooth_descent(
            seed_vec.copy(),
            sub_deadline,
            best_score,
            eta0=0.5,
            lr0=0.45,
            eval_every=120,
        )
        if cand_score < best_score - 1e-14:
            best_score, best_seq = cand_score, cand_seq.copy()
            print(f"[{time.time():.1f}] ***** new global best after smooth descent {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final annealing polish (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing polish")
        best_seq, best_score = _pairwise_annealing(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            rng,
        )
        print(f"[{time.time():.1f}] after annealing best score {best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective = 2·n·max(conv)/sum²): {best_score:.6f} "
        f"(length={len(best_seq)})"
    )
    return best_seq.tolist()
