def _lp_step(f, M):
    n = len(f)
    rows = np.add.outer(np.arange(n), np.arange(n)).ravel()
    cols = np.tile(np.arange(n), n)
    data = np.repeat(f, n)  # shape (n*n,)
    A = sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)  # maximize sum => minimize -sum
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if res.success:
        b0 = res.x
        sum_b = b0.sum()
        if sum_b <= 0:
            return f, False
        g = b0 / sum_b
        return g, True
    else:
        return f, False

    if sumb <= 0:

    g = b0 / sumb
    return g, True

"""
Step‑function optimizer (≈1 000 s budget).

The algorithm mixes two ideas:

1️⃣  Linear‑programming “tighten‑bound’’ step (the classic
    *f → g* iteration from the paper).  For a current probability
    vector *f* we compute *M = max(f*f)* and solve a LP that finds a
    vector *g* of maximal ℓ₁‑norm under the same convolution bound:
        maximise   1·g
        s.t.      f * g ≤ M   (component‑wise),   g ≥ 0 .
    The solution is a new extreme point of the feasible set.

2️⃣  After a few LP updates the vector is polished with a mirror‑descent
    (Adam) optimiser that works directly on the exact objective

        Φ(a) = 2·n·max(a*a)   (a non‑negative, ∑a = 1)

    The Adam step uses a soft‑max to collect an *active set* of the
    largest convolution entries and projects back onto the simplex.

Finally a cheap greedy pairwise mass‑transfer and a simulated‑annealing
stage clean up any remaining spikes.

All phases respect the global time budget; the best vector found is
returned as a plain Python list (the evaluator will re‑normalise it).
"""

from __future__ import annotations
import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# optional SciPy – only needed for the LP step
try:
    import scipy.sparse as sp
    import scipy.optimize as opt
    _HAS_SCIPY = True
except Exception:                     # pragma: no cover
    _HAS_SCIPY = False

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max (returns a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (a is assumed to sum to 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families (return a probability vector – sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    w /= w.sum()
    return w


def _distribution_exponential_center(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(i - half))
    w /= w.sum()
    return w


def _distribution_exponential_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-rate * i)
    w /= w.sum()
    return w


def _distribution_exponential_right(n: int, rate: float) -> np.ndarray:
    return _distribution_exponential_left(n, rate)[::-1]


def _distribution_dirichlet(n: int, rng: np.random.Generator, conc: float) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a.astype(np.float64)


# ----------------------------------------------------------------------
# Linear‑programming move : f → g
def _lp_optimize(f: np.ndarray, M: float) -> tuple[np.ndarray, bool]:
    """
    Solve  max 1·g  s.t.  f*g ≤ M  (component‑wise),  g ≥ 0 .
    Returns scaled vector g (∑g = 1) and a success flag.
    """
    if not _HAS_SCIPY:                     # pragma: no cover
        return f, False

    n = f.shape[0]
    # Build A where A[k, j] = f_{k-j} (zero when out‑of‑range)
    rows = np.add.outer(np.arange(n), np.arange(n)).ravel()
    cols = np.tile(np.arange(n), n)
    data = np.repeat(f, n)                # f_i repeats for each j
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                       # maximise sum(g)  → minimise –sum(g)
    b = np.full(2 * n - 1, M, dtype=np.float64)

    try:
        res = opt.linprog(c, A_ub=A, b_ub=b,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'presolve': True})
    except Exception:                     # pragma: no cover
        return f, False

    if not res.success:
        return f, False

    g0 = res.x
    s = g0.sum()
    if s <= 0:
        return f, False
    g = g0 / s
    return g, True

# Adam‑based mirror descent on the exact objective
def _adam_refine(
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    lr0: float = 0.8,
    active_frac: float = 0.025,
    max_active: int = 30,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_interval: int = 250,
    best_seq: np.ndarray | None = None,
    best_score: float = float("inf"),
) -> tuple[np.ndarray, float]:
    """Adam on the soft‑max parametrisation; exact objective used."""
    n = init_a.shape[0]
    x = np.log(np.maximum(init_a, 1e-12))
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    ind_grid = np.arange(n, dtype=np.int32)
    t = 0
    if best_seq is None:
        best_seq = init_a.copy()
        best_score = _objective(init_a)

    while time.time() < deadline and t < max_iters:
        t += 1
        a = _softmax(x)                      # stays on the simplex
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            top = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top]

        # sub‑gradient of the max (active set)
        grad_a = np.zeros_like(a)
        for idx_k in act:
            offset = idx_k - ind_grid
            mask = (offset >= 0) & (offset < n)
            grad_a[mask] += 2.0 * a[offset[mask]]

        # gradient w.r.t. log‑weights
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update (learning rate decays like 1/√t)
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = lr0 / math.sqrt(t)
        x -= eta * m_hat / (np.sqrt(v_hat) + eps)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t:,} new best {best_score:.6f}")

    best_seq = _project_onto_simplex(best_seq, s=1.0)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise refinement (same idea as in the reference code)
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing style local search
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a low‑value step‑function under the 1000 s budget."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # small safety margin

    # ------------------------------------------------------------------
    # 1️⃣  a few analytically‑motivated seeds (moderate length → cheap LP)
    n = 800                                    # length kept small for LP speed
    seeds = [
        _distribution_uniform(n),
        _distribution_exponential_left(n, rate=0.004),
        _distribution_exponential_right(n, rate=0.004),
        _distribution_sine_weighted(n, p=2.0),
    ]
    # add a couple of random Dirichlet seeds
    for conc in (0.2, 1.0):
        for _ in range(2):
            seeds.append(_distribution_dirichlet(n, rng, conc))

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 2️⃣  LP‑based “f → g’’ iteration followed by a short Adam run
    for idx, init in enumerate(seeds):
        if time.time() >= deadline:
            break
        f = init.copy()
        cur_score = _objective(f)
        if cur_score < overall_best_score:
            overall_best_score, overall_best_seq = cur_score, f.copy()
            print(f"[{time.time():.1f}] seed {idx} initial best {cur_score:.6f}")

        # ---- LP tightening loop -------------------------------------------------
        mix_factor = 0.2
        for step in range(25):                # a few deterministic updates
            if time.time() >= deadline:
                break
            M = _fft_self_convolve(f).max()
            g, ok = _lp_optimize(f, M)
            if not ok:
                break
            f = (1.0 - mix_factor) * f + mix_factor * g
            f = _project_onto_simplex(f, s=1.0)
            cur_score = _objective(f)
            if cur_score < overall_best_score:
                overall_best_score, overall_best_seq = cur_score, f.copy()
                print(f"[{time.time():.1f}] LP step {step} improved to {cur_score:.6f}")
            mix_factor = max(0.02, mix_factor * 0.90)   # slowly shrink step

        # ---- short Adam refinement of the resulting vector --------------------
        if time.time() < deadline:
            slice_deadline = time.time() + min(30.0,
                                               (deadline - time.time()) * 0.3)
            refined_seq, refined_sc = _adam_refine(
                f.copy(),
                slice_deadline,
                max_iters=150_000,
                lr0=0.7,
                active_frac=0.04,
                max_active=40,
                beta1=0.9,
                beta2=0.999,
                eval_interval=200,
                best_seq=f,
                best_score=cur_score,
            )
            if refined_sc < overall_best_score:
                overall_best_score, overall_best_seq = refined_sc, refined_seq.copy()
                print(f"[{time.time():.1f}] post‑Adam improved to {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Long Adam descent from the globally best seed (uses most of the budget)
    if overall_best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        long_deadline = time.time() + max(0.0, remaining - 30.0)
        print(f"[{time.time():.1f}] long Adam descent – {remaining-30:.1f}s left")
        overall_best_seq, overall_best_score = _adam_refine(
            overall_best_seq.copy(),
            long_deadline,
            max_iters=800_000,
            lr0=0.9,
            active_frac=0.02,
            max_active=60,
            beta1=0.9,
            beta2=0.999,
            eval_interval=250,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] after long Adam best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Greedy pairwise polishing
    if overall_best_seq is not None and time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=250_000,
        )

    # ------------------------------------------------------------------
    # 5️⃣  Final simulated‑annealing polishing
    if overall_best_seq is not None and time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.01,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    # Return the best vector we have found (as plain Python list)
    if overall_best_seq is None:                                 # pragma: no cover
        # fallback – uniform if everything failed
        overall_best_seq = _distribution_uniform(n)

    print(f"Final best score ≈ {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
