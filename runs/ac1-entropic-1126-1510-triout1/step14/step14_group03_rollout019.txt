def solve_lp_step(f, M):
    n = len(f)
    # Build constraint matrix A (2n-1) x n, and b_ub vector of M.
    # Build using convolution: for each shift k, we need sum_i f_i * b_{k-i} <= M
    # This is equivalent to conv(f, b) <= M (elementwise)
    # Represent conv as linear operator: we can use convolve with reversed version?
    # But easiest: Use Toeplitz matrix: each column j (b_j) contributes to conv entries k = j + i, weight f_i.
    # So for k in [0, 2n-2], A[k, j] = f_{k-j} if 0 <= k-j < n else 0.

    n = len(f)
    L = 2*n - 1
    A = np.zeros((L, n), dtype=np.float64)
    for j in range(n):
        # range of i where k - j = i -> i = 0..n-1 => k = i + j, goes from j to j+n-1
        A[j:j+n, j] = f  # fill rows j ... j+n-1 with f

    # objective: maximize sum(b) -> minimize -sum(b)

"""
Hybrid search for a step‑function minimizing ``evaluate_sequence``.

The algorithm proceeds in several stages:

1. **Analytic seeds** – a variety of parametric families (uniform,
   arcsine, sum‑edge‑weighted, random sparse Dirichlet) are generated for a
   few moderate lengths (≈2–4 k).  The best seed per length is kept.

2. **LP refinement** – for the current best seed `f` we solve the linear
   program

       maximize    1ᵀ b
       subject to  f * b ≤ M      (element‑wise)
                   b ≥ 0

   where `M = max(f*f)`.  Solving this LP gives a distribution `b` that can
   be mixed with `f` while keeping the same convolution bound.  After a few
   LP → mix steps the max‑convolution usually drops.

3. **Exponentiated‑gradient descent** – a fast mirror‑descent step that
   uses the sub‑gradient of the active convolution entry to further reduce
   the objective.

4. **Pairwise mass‑transfer annealing** – a final simulated‑annealing polish.

All steps respect the 1000 s wall‑time limit and the evaluator’s safety
checks.  The returned list is a non‑negative probability vector (sum=1);
the evaluator’s scale‑invariance makes this a valid solution.
"""

import time
import math
import numpy as np
from scipy import optimize as opt

# ----------------------------------------------------------------------
#  FFT‑based self‑convolution (real input, full linear convolution)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
#  Exact objective used by the evaluator (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a))/sum(a)² .  ``a`` may be any non‑negative vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
#  Analytic families – all return a *probability* vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_sum_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ i^{‑p} + (n‑i)^{‑p}   (i measured from 0.5)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (-p)) + (j ** (-p))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1‑lam) + uniform·lam."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
#  Sub‑gradient of the *active* convolution entry (max)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """Return (gradient, index_of_max) for the current max entry of a*a."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))

    grad = np.zeros_like(a)
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
#  Exponentiated‑gradient descent (multiplicative mirror descent)
# ----------------------------------------------------------------------
def _exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """Run EG descent on the exact objective until ``deadline``."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Pairwise‑mass‑transfer annealing (final polish)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins; accept improvements."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4        # move at most 40 % of source mass
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
#  LP refinement utilities
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return dense matrix A where (A @ b)[k] = (f * b)[k] (full conv)."""
    n = f.shape[0]
    L = 2 * n - 1
    A = np.zeros((L, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_step(f: np.ndarray, M: float) -> np.ndarray | None:
    """Solve  max 1ᵀb   s.t.  f*b ≤ M, b ≥ 0 ;  return (scaled) b or None."""
    n = f.shape[0]
    A = _build_conv_matrix(f)                     # (2n‑1) × n
    c = -np.ones(n, dtype=np.float64)            # minimise –sum(b)
    bounds = [(0.0, None)] * n
    res = opt.linprog(
        c,
        A_ub=A,
        b_ub=np.full(2 * n - 1, M, dtype=np.float64),
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None
    b = res.x
    return b
