"""
Hybrid search for a step‐function that minimises
evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2).

The method consists of three phases:

1.  Analytic seed generation.
    - Uniform, arcsine, Hann, generalized normal.
    - Edge‑weighted families with a fine grid of small exponents p∈[0.05,0.5].
    - Exponential‑tail distributions a_i ∝ exp(-α·i)+exp(-α·(n‑1‑i)).
    - Power‑law families with p∈[0.15,0.5].
    - Random Dirichlet and sparse seeds.
    All seeds are normalised to sum = 1 and evaluated; the best few are kept.

2.  Smooth optimisation.
    For each of the top seeds we run Adam on the temperature‑smoothed
    objective 2·n·τ·log‑sum‑exp(conv/τ) with a slowly decaying temperature.
    The best refined seed replaces the global best.

3.  Exact refinement.
    A projected exponentiated‑gradient descent (multiplicative mirror
    descent) on the true max‑convolution objective, followed by a short
    pairwise‑annealing polishing step, is performed on the current best
    sequence until the time budget expires.

All steps respect the overall 1000 s budget.  The routine finally returns
a plain list of non‑negative floats ready for ``evaluate_sequence``.
"""
import time
import math
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (FFT based)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence for an un‑normalised vector a."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic families – all return a probability vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w /= w.sum()
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(-|x-½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{-p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w /= w.sum()
    return w


def _distribution_exponential_edge(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    """Monotone decreasing power‑law (p>0)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Gradient / sub‑gradient of the active convolution entry
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> Tuple[np.ndarray, int]:
    """
    Sub‑gradient of the maximal convolution entry.
    Returns (grad, index_of_max).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))
    grad = np.zeros_like(a)
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
# Multiplicative (exponentiated) gradient descent on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_descent(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         eta0: float = 0.7,
                         eval_every: int = 80) -> Tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the exact objective."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the temperature‑smoothed max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   tau_start: float = 0.3,
                   tau_end: float = 5e-4,
                   tau_decay: float = 0.999,
                   eval_every: int = 200,
                   step_size0: float = 0.045) -> Tuple[np.ndarray, float]:
    """Adam on the smoothed objective 2·n·τ·log‑sum‑exp(conv/τ)."""
    n = a0.shape[0]
    x = np.log(a0 + 1e-15)                # logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    tau = tau_start

    while time.time() < deadline and tau > tau_end:
        t += 1
        # softmax probabilities from logits
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # convolution and τ‑softmax weights
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        exp_vals = np.exp((conv - max_conv) / tau)
        w = exp_vals / exp_vals.sum()          # Σ w = 1

        # gradient w.r.t. a (4·n·correlation)
        # correlation of a with weight vector w (same length as a)
        # use FFT based correlation
        n_a = a.shape[0]
        size = 1 << ((n_a + w.shape[0] - 1) - 1).bit_length()
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(w[::-1], n=size)
        corr = np.fft.irfft(A * B, n=size)[:n_a]

        grad_a = 4.0 * n * corr               # ∂ objective / ∂ a
        # map gradient through the softmax
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = step_size0 / (np.sqrt(v_hat) + eps)
        x = x - lr * m_hat

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} τ={tau:.6f} new best {best_score:.6f}")

        tau = max(tau * tau_decay, tau_end)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple pairwise annealing (simulated annealing style)
# ----------------------------------------------------------------------
def _pairwise_annealing(a: np.ndarray,
                        deadline: float,
                        best_seq: np.ndarray,
                        best_score: float,
                        rng: np.random.Generator,
                        max_trials: int | None = None) -> Tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins, accepting improving moves."""
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score
