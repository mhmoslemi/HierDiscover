    M = max(_fft_self_convolve(f))
    # Build Toeplitz matrix A (2n-1 x n)
    c = np.concatenate([f, np.zeros(n-1)])

"""
Improved construction for ``evaluate_sequence``.

The algorithm consists of three stages:

1️⃣  Generate a modest set of diverse seeds (uniform, arcsine, Dirichlet,
    sparse) for a few medium‑length step functions (≈2500–3500).

2️⃣  Apply a *conditional‑gradient* (Frank–Wolfe) refinement.
    For a current sequence `f` we compute `M = max(f*f)`.
    The linear program

        max   1ᵀ g
        s.t.  f * g ≤ M   (element‑wise)
              g ≥ 0

    is solved (the convolution constraints are linear in `g`).
    The optimal direction `g` is mixed with `f` by a few trial
    step‑sizes and the best mixture (in terms of the exact objective
    `2·n·max(conv)`) is kept.  Repeating this step a handful of times
    strictly lowers the bound for most seeds.

3️⃣  Polish the best result with the smooth‐soft‑max EG descent from the
    original code and a final greedy pairwise mass‑transfer hill‑climber.

All stages respect the overall time budget (default 1000 s).  The
function returns a plain Python list – ``evaluate_sequence`` will
re‑normalise it internally.
"""

import time

import numpy as np
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers (FFT based for speed)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """Exact evaluation of 2·n·max(conv) assuming Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, i.e. Σ = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse Dirichlet‑like seed with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals

# Smooth (soft‑max) gradient descent – copied from the previous version
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _fft_cross_correlate_valid(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """Linear cross‑correlation (valid part) (w ⋆ a)[j] = Σ_i w[i+j]·a[i]."""
    n = a.shape[0]
    m = w.shape[0]                # = 2·n‑1
    size = 1 << ((m + n - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A = np.fft.rfft(a, n=size)
    corr = np.fft.irfft(W * np.conj(A), n=size)
    return corr[:n]


def _fft_sliding_window_sum(w: np.ndarray, win: int) -> np.ndarray:
    """For each j compute Σ_{i=j}^{j+win‑1} w[i] (window length = win)."""
    m = w.shape[0]
    size = 1 << ((m + win - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    ones = np.ones(win, dtype=np.float64)
    O = np.fft.rfft(ones, n=size)
    conv = np.fft.irfft(W * O, n=size)
    start = win - 1
    return conv[start:start + win]


def _smooth_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    tau_start: float = 0.02,
    tau_end: float = 0.001,
    lr0: float = 0.6,
    eval_every: int = 200,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient on a smoothed version of the L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    it = 0
    while time.time() < time_limit:
        it += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] smooth it={it} new global best {best_score:.6f}")

        # linear annealing of τ
        frac = (time.time() - start) / (time_limit - start)
        tau = tau_start * (1.0 - frac) + tau_end * frac

        # soft‑max weights
        shifted = conv - max_val
        exp_vals = np.exp(shifted / tau)
        w = exp_vals / exp_vals.sum()               # length 2n‑1, sums to 1

        # gradient of the smoothed objective
        term1 = _fft_cross_correlate_valid(w, a)    # Σ w[i+j]·a[i]
        win_sum = _fft_sliding_window_sum(w, n)    # Σ_{k=j}^{j+n‑1} w[k]
        grad = term1 + a * win_sum

        # EG update
        eta = lr0 / math.sqrt(it)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if it % eval_every == 0:
            pass
    return best_seq, best_score


def _subgradient_descent_exact(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] exact subgrad it={t} new global best {best_score:.6f}")
    return best_seq, best_score


def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Greedy hill‑climber that moves a random fraction of mass between two indices."""
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * np.random.rand() * 0.45
        if delta < 1e-15:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear‑programming oracle for the conditional‑gradient step
# ----------------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve
        max   Σ g_j
        s.t.  (f * g)_k ≤ M  for all k,
              g ≥ 0
    Returns the optimal g (not normalised) or None on failure.
    """
    n = f.shape[0]
    # Build Toeplitz matrix for convolution constraints (2n‑1 × n)
    col = np.concatenate([f, np.zeros(n - 1)])
    row = np.concatenate([f[:1], np.zeros(n - 1)])
    A = toeplitz(col, row)               # dense, but n≈3k ⇒ ~70 MiB
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # Linear programme (note: linprog minimises)
    c = -np.ones(n, dtype=np.float64)    # minimise –Σg ⇒ maximise Σg
    bounds = [(0, None)] * n
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _mix_best(f: np.ndarray, g: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Return the best convex combination (1‑t)·f + t·g (g is already a probability vector)
    among a small set of trial t‑values.
    """
    best_seq = f
    best_score = _objective(f)
    # try a handful of mixing coefficients
    for t in (0.05, 0.1, 0.2, 0.35, 0.5, 0.7):
        cand = (1.0 - t) * f + t * g
        # projection onto the simplex (very cheap because we stay positive)
        cand = _project_onto_simplex(cand, s=1.0)
        sc = _objective(cand)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, cand.copy()
    return best_seq, best_score


def _conditional_gradient_opt(
    f: np.ndarray, time_limit: float, best_score: float
) -> tuple[np.ndarray, float]:
    """
    Perform a few Frank–Wolfe iterations on `f`.  Each iteration solves the LP
    above, mixes the direction with a few step sizes and keeps the best result.
    The routine stops early if no improvement is possible or the time budget
    expires.
    """
    while time.time() < time_limit:
        M = _fft_self_convolve(f).max()
        g = _solve_lp_max_sum(f, M)
        if g is None:
            break
        # normalise the direction to a probability vector
        s = g.sum()
        if s <= 0.0:
            break
        g = g / s
        # try mixing
        new_f, new_score = _mix_best(f, g)
        if new_score < best_score - 1e-14:
            best_score, f = new_score, new_f
            print(
                f"[{time.time():.1f}] CG improved – new best {best_score:.6f}"
            )
        else:
            # no further improvement with the current direction
            break
    return f, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a non‑negative step‑function (as a plain Python list) that minimises
    ``evaluate_sequence`` within the allocated time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # tiny safety margin

    # ------------------------------------------------------------------
    # 0. Retrieve a previous construction if it exists
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
        else:
            prev = None
    except Exception:
        prev = None

    # ------------------------------------------------------------------
    # 1. Build a small, diverse seed pool (length ≈ 2800‑3400)
    # ------------------------------------------------------------------
    seed_lengths = [2600, 3000, 3400]
    seeds: list[np.ndarray] = []

    for n in seed_lengths:
        # deterministic families
        seeds.append(_distribution_uniform(n))
        seeds.append(_distribution_arcsine(n))

        # a few Dirichlet draws with varying concentration
        for _ in range(3):
            alpha = rng.uniform(0.2, 2.0)
            seeds.append(rng.dirichlet(np.full(n, alpha)))

        # sparse random seeds (very low density)
        for _ in range(2):
            seeds.append(_distribution_random_sparse(n, rng, density=0.005))

    if prev is not None:
        seeds.append(prev)

    # ------------------------------------------------------------------
    # 2. Evaluate seeds and keep the most promising ones
    # ------------------------------------------------------------------
    evaluated = [( _objective(s), s.copy() ) for s in seeds]
    evaluated.sort(key=lambda x: x[0])
    top_k = 6
    top_seqs = [seq for _, seq in evaluated[:top_k]]
    best_score, best_seq = evaluated[0]

    print(f"[{time.time():.1f}] initial best objective {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Conditional‑gradient (LP‑based) refinement on the top seeds
    # ------------------------------------------------------------------
    for idx, seq in enumerate(top_seqs):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] CG start on seed {idx} (len={len(seq)})")
        # allocate a slice of the remaining budget to this seed
        seed_deadline = time.time() + (deadline - time.time()) / max(1, len(top_seqs) - idx)
        seq, sc = _conditional_gradient_opt(seq, seed_deadline, best_score)
        if sc < best_score - 1e-12:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] new global best after CG {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Optional smooth EG descent (low‑temperature soft‑max)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print(f"[{time.time():.1f}] smooth descent polishing")
        best_seq, best_score = _smooth_descent(
            best_seq,
            deadline,
            best_seq,
            best_score,
            tau_start=0.015,
            tau_end=0.0008,
            lr0=0.55,
            eval_every=150,
        )

    # ------------------------------------------------------------------
    # 5. Exact sub‑gradient descent (a few more iterations)
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        print(f"[{time.time():.1f}] exact sub‑gradient polishing")
        best_seq, best_score = _subgradient_descent_exact(
            best_seq,
            deadline,
            best_seq,
            best_score,
            lr0=0.85,
            eval_every=250,
        )

    # ------------------------------------------------------------------
    # 6. Final greedy pairwise hill‑climber
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy swap polishing")
        best_seq, best_score = _pairwise_swap(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
        )

    print(
        f"Final best objective (2·n·max(conv)) = {best_score:.6f}, length = {len(best_seq)}"
    )
    # Return as plain Python list – evaluate_sequence will handle re‑normalisation.
    return best_seq.tolist()
