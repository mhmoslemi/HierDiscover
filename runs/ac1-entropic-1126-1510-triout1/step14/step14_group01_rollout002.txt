Multi‑start search for a non‑negative step sequence that minimises
evaluate_sequence.  The pipeline is:

1.  Generate a diverse set of seed vectors (uniform, arcsine,
    Beta‑type, edge‑weighted, sine‑weighted, random Dirichlet,
    sparse random).  Seeds are created for several lengths (2k–12k)
    and mixed with the uniform distribution.

2.  Keep the best ~10 seeds and run an annealed exponentiated‑gradient
    descent on a smooth log‑sum‑exp surrogate of   max(conv(a,a)).
    This is the “smooth descent” from the previous baseline but with
    more aggressive smoothing schedule and longer runtime.

3.  Upsample the current best vector (linear interpolation) to a larger
    length (up to ~15 k) and run the smooth descent again.

4.  Perform a short *sub‑gradient* refinement that directly uses the
    exact gradient of the max‑convolution (no smoothing).  The update
    is an additive step followed by Euclidean projection onto the
    probability simplex.

5.  Deterministic pairwise mass‑transfer sweep and a cheap simulated‑
    annealing clean‑up finish the optimisation.

All operations are FFT‑accelerated, the algorithm respects the 1000 s
budget and prints intermediate scores.

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (real FFT, length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b (real FFT)."""
    n = a.shape[0]
    m = b.shape[0]
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a simplex vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric Beta(a,b) on the regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, where U is the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Smooth log‑sum‑exp surrogate ------------------------------------------------
def _smooth_gradient(a: np.ndarray, lam: float) -> tuple[np.ndarray, float]:
    """
    Returns (grad, max_conv) where grad = ∇_a φ_λ(a) and
    max_conv = max(conv(a,a)).  a is assumed to sum to 1.
    """
    n = a.shape[0]

    # 1) self‑convolution
    conv = _fft_self_convolve(a)               # length 2n‑1
    max_c = conv.max()

    # 2) softmax weights (stable)
    z = lam * conv
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    s = exp_z / exp_z.sum()                    # length 2n‑1, sums to 1

    # 3) gradient = 2 * (a_rev * s) restricted to centre block
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, s)        # length 3n‑2
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]   # slice of length n

    return grad, max_c


def _smooth_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    lam_start_factor: float = 0.12,
    lam_inc: float = 1.0015,
    beta: float = 0.9,
    eval_interval: int = 150,
    patience: int = 1800,
) -> tuple[np.ndarray, float]:
    """
    Annealed exponentiated‑gradient descent on the smooth surrogate.
    Returns the best sequence seen and its exact objective.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    lam = lam_start_factor * n
    rng = np.random.default_rng()

    while time.time() < deadline and t < 600_000:
        t += 1
        grad, max_conv = _smooth_gradient(a, lam)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # learning‑rate schedule
        eta = lr0 / math.sqrt(t)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * momentum)

        # safeguard
        a = np.maximum(a, 1e-30)
        a /= a.sum()

        # anneal λ
        lam *= lam_inc

        if t % eval_interval == 0:
            cur_score = 2.0 * n * max_conv
            if cur_score + 1e-14 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] smooth it={t:,}  new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            # small random jitter to escape flat regions
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

# Exact max‑convolution sub‑gradient -----------------------------------------
def _max_subgradient(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient of max(conv(a,a)) together with the current max value.
    The gradient is g_i = 2 * a_{k-i} for the arg‑max index k.
    """
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))
    n = a.shape[0]

    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[j[mask]]

    max_conv = conv[k]
    return grad, max_conv


def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.15,
    decay: float = 0.9998,
    eval_interval: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Simple additive sub‑gradient descent on the exact max‑convolution.
    Projection onto the simplex guarantees feasibility.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    eta = eta0

    while time.time() < deadline:
        t += 1
        grad, max_conv = _max_subgradient(a)

        a = a - eta * grad
        a = np.maximum(a, 0.0)
        a = _project_onto_simplex(a)

        eta *= decay

        if t % eval_interval == 0:
            cur_score = 2.0 * n * max_conv
            if cur_score + 1e-14 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad it={t:,}  new best {best_score:.6f}")

    return best_seq, best_score

# Local combinatorial refinements --------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Deterministic pairwise mass‑transfer sweep."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e}  new best {best_score:.6f}")

    return best_seq, best_score


def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Cheap simulated‑annealing clean‑up."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e}  best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq

        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Annealed smooth descent on the elite seeds
    remaining = deadline - time.time()
    if remaining > 40.0 and top_candidates:
        # pick the 7 most promising seeds
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:7]]
        per_seed = remaining * 0.55 / max(1, len(seeds))
        print(f"[{time.time():.1f}] smooth descent on {len(seeds)} seeds, ~{per_seed:.1f}s each")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            a_opt, sc_opt = _smooth_descent(
                seed_seq.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                lam_start_factor=0.10,
                lam_inc=1.0017,
                beta=0.88,
                eval_interval=120,
                patience=1500,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after smooth seed {idx} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Upsampling to a larger length (if we have enough time)
    remaining = deadline - time.time()
    if remaining > 150.0:
        # aim for about 12k‑15k points but never exceed 18000
        target_len = min(18000, int(len(overall_best_seq) * 2.5))
        print(f"[{time.time():.1f}] up‑sampling to {target_len}")
        up_seq = _upsample(overall_best_seq, target_len)

        sub_dead = time.time() + remaining * 0.55
        up_opt, up_sc = _smooth_descent(
            up_seq,
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.6,
            lam_start_factor=0.12,
            lam_inc=1.0012,
            beta=0.92,
            eval_interval=100,
            patience=2000,
        )
        if up_sc < overall_best_score:
            overall_best_score, overall_best_seq = up_sc, up_opt.copy()
            print(f"[{time.time():.1f}] up‑sampled descent new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Precise sub‑gradient refinement (targets the exact max)
    remaining = deadline - time.time()
    if remaining > 30.0:
        print(f"[{time.time():.1f}] sub‑gradient refinement on length {len(overall_best_seq)}")
        sub_dead = time.time() + remaining * 0.6
        overall_best_seq, overall_best_score = _subgradient_descent(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.12,
            decay=0.9997,
            eval_interval=80,
        )
        print(f"[{time.time():.1f}] after sub‑grad best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Deterministic pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] deterministic pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=400_000,
        )

    # ------------------------------------------------------------------
    # 8. Final cheap simulated‑annealing clean‑up
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list – evaluate_sequence will enforce final clipping
    return overall_best_seq.tolist()
