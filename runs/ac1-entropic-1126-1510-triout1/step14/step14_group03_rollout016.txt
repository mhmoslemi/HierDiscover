    def _distribution_beta(n, alpha):
        # Beta(alpha,alpha) on discretized grid.
        i = np.arange(n, dtype=np.float64) + 0.5
        x = i / n
        # compute (x*(1-x))^{alpha-1}
        # as x*(1-x) might be zero at edges: i=0.5 and n-0.5 => x approx 0 or 1, but pow with negative exponent may produce inf. So we can add small epsilon.
        vals = (x * (1 - x)) ** (alpha - 1.0)
        # For alpha < 1, exponent negative => near edges singular; we can cap by large value but see division.
        # We'll handle by setting any inf/NaN to 0 then normalize.
        vals = np.where(np.isfinite(vals), vals, 0.0)
        if vals.sum() == 0.0:
            vals = np.full(n, 1.0 / n)
        else:
            vals /= vals.sum()
        return vals

    n = a.shape[0]
    loga = np.log(a + 1e-16)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    t=0
    start = time.time()
    # schedule parameters
    total_opt_time = deadline - start
    alpha_start = 0.5
    alpha_end = 300.0
    # warm-up iterations maybe not needed
    while time.time() < deadline:
        t += 1
        # compute alpha as schedule
        elapsed = time.time() - start
        frac = min(1.0, elapsed / total_opt_time)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        # compute convolution
        conv = _fft_self_convolve(a)
        # compute softmax weights
        max_conv = conv.max()
        # shift by max for numeric stability
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()
        # compute gradient
        grad = _grad_softmax(a, w)   # shape n
        # Adam update on loga
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        # step
        loga = loga - 0.1 * m_hat / (np.sqrt(v_hat) + eps)    # learning rate 0.1
        # map back to a
        a = np.exp(loga)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
            loga = np.log(a)
        else:
            a = a / s
            loga = loga - np.log(s)   # maintain loga after normalization: Actually a = exp(loga)/s, so new_loga = loga - log(s)
        # evaluate objective periodically
        if t % 100 == 0:
            cur = _objective(a)

                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} alpha={alpha:.3f} new best {best_score:.6f}")
    return best_seq, best_score

def _distribution_beta(n, alpha):
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # compute log density
    # Add epsilon to avoid log(0)
    eps = 1e-15
    log_x = np.log(np.clip(x, eps, 1 - eps))
    log_one_minus_x = np.log(np.clip(1 - x, eps, 1 - eps))
    log_vals = (alpha - 1.0) * (log_x + log_one_minus_x)
    # shift to avoid overflow
    max_log = log_vals.max()
    vals = np.exp(log_vals - max_log)
    # renormalize
    sum_vals = vals.sum()
    if sum_vals == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= sum_vals
        return vals

"""
Step‑function optimizer for ``evaluate_sequence``.

The algorithm works in three stages:

1. **Seed generation** – a wide variety of analytic families (uniform,
   arcsine, beta‑shaped, edge‑weighted, sum‑edge‑weighted, generalized‑normal)
   and several random Dirichlet vectors are sampled.  The 20 best seeds are kept.
2. **Soft‑max Adam descent** – starting from the current best seed we minimise a
   smooth approximation of the objective.  The max‑convolution entry is
   replaced by a soft‑max with an annealing temperature that grows from
   ``α_start`` to ``α_end``.  An Adam optimiser works on the log‑variables to keep
   all entries non‑negative, and after each step the vector is projected onto
   the probability simplex.
3. **Pairwise mass‑transfer annealing** – a cheap simulated‑annealing polish
   that moves a fraction of mass between two random positions and accepts
   improvements (or uphill moves with a decreasing temperature).

The whole search respects the 1000 s wall‑time limit and always returns a
non‑negative list of heights ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  FFT‑based self‑convolution (real inputs, full linear convolution)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Return full linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
#  Exact objective (scale invariant, sum(a)=1 ⇒ obj = 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
#  Analytic families (all return a probability vector, sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    s = w.sum()
    return w / s if s != 0.0 else np.full_like(w, 1.0 / n)


def _distribution_edge_sum_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (-p)) + (j ** (-p))
    s = w.sum()
    return w / s if s != 0.0 else np.full_like(w, 1.0 / n)


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    s = w.sum()
    return w / s if s != 0.0 else np.full_like(w, 1.0 / n)


def _distribution_beta(n: int, alpha: float) -> np.ndarray:
    """Beta(α,α) on the discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-15
    log_x = np.log(np.clip(x, eps, 1.0 - eps))
    log_one_minus_x = np.log(np.clip(1.0 - x, eps, 1.0 - eps))
    log_vals = (alpha - 1.0) * (log_x + log_one_minus_x)
    # stabilise exponentials
    log_vals -= log_vals.max()
    vals = np.exp(log_vals)
    s = vals.sum()
    return vals / s if s != 0.0 else np.full_like(vals, 1.0 / n)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
#  Gradient of the soft‑max smoothed objective
# ----------------------------------------------------------------------
def _grad_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of   L(a)=∑_k w_k (a*a)_k   w.r.t. a,
    where w is a probability vector (soft‑max over the convolution).
    The formula reduces to 2·correlation(w, a).
    """
    n = a.shape[0]
    # correlation(w, a) = conv(reverse(w), a) on the appropriate slice
    w_rev = w[::-1]
    conv_len = 3 * n - 2                # length of linear convolution w_rev * a
    size = 1 << ((conv_len - 1).bit_length())   # next power of two
    fw = np.fft.rfft(w_rev, n=size)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fw * fa, n=size)   # linear convolution
    start = n - 1
    end = start + n                       # slice of length n, still reversed
    grad = 2.0 * conv[start:end][::-1]    # reverse to get proper order
    return grad


# ----------------------------------------------------------------------
#  Adam optimiser on the soft‑max objective
# ----------------------------------------------------------------------
def _adam_softmax_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    alpha_start: float = 0.5,
    alpha_end: float = 200.0,
    lr: float = 0.10,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """
    Performs Adam optimisation on the smoothed objective

        F(a) = ∑_k w_k (a*a)_k ,   w_k ∝ exp(α·(a*a)_k),

    where α grows from ``alpha_start`` to ``alpha_end``.
    The iterate is kept on the simplex (non‑negative, sum=1) by working
    in the log‑domain.
    """
    a = a0.copy()
    n = a.shape[0]
    # initialise log‑variables
    loga = np.log(np.maximum(a, 1e-30))
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    it = 0
    start = time.time()
    total_opt_time = deadline - start
    while time.time() < deadline:
        it += 1
        # ----- temperature schedule -----
        elapsed = time.time() - start
        frac = min(1.0, elapsed / total_opt_time)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        # ----- compute soft‑max weights -----
        conv = _fft_self_convolve(a)                 # full self‑convolution
        max_conv = conv.max()
        # stability: shift by max
        exp_weights = np.exp(alpha * (conv - max_conv))
        w = exp_weights / exp_weights.sum()          # probability vector

        # ----- gradient -----
        grad = _grad_softmax(a, w)

        # ----- Adam update on log‑variables -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        loga = loga - lr * m_hat / (np.sqrt(v_hat) + eps)

        # map back to the simplex
        a = np.exp(loga)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        loga = np.log(np.maximum(a, 1e-30))

        # ----- occasional exact evaluation -----
        if it % eval_every == 0:
            cur = 2.0 * n * conv.max()      # exact objective (sum(a)=1)
            if cur < best_score - 1e-14:
                best_score, best_seq = cur, a.copy()
                print(f"[{time.time():.1f}] Adam it={it} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
#  Pairwise‑mass‑transfer simulated‑annealing polish
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        # pick source with non‑negligible mass
        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        # pick a distinct target
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # propose moving a fraction of i's mass to j
        delta = a[i] * rng.random() * 0.4   # at most 40 % of the source
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = 2.0 * n * _fft_self_convolve(new_a).max()
        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score
