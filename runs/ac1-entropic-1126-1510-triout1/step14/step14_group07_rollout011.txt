"""
propose_candidate – heuristic search for a step‑function that minimises
``evaluate_sequence``.
The search consists of

1️⃣  seeding: analytically‑motivated families (uniform, sine‑power,
    centre‑exponential, left‑/right‑exponential, Beta‑U‑shape,
    Dirichlet, sparse random) are sampled for several lengths.
    The best 20 seeds (by the exact objective) are kept.

2️⃣  ADAM mirror‑descent on the soft‑max parametrisation.
    For each seed a short ADAM run refines it, after which the overall
    best seed is refined for the remaining time with a long ADAM run.
    The active‑set for the sub‑gradient is gradually tightened.

3️⃣  sparsification – after the long run entries below a dynamic
    threshold are zeroed, the vector is re‑normalised and ADAM is run
    again for a short burst.  This encourages a “few‑large‑spikes”
    shape that tends to lower the maximal self‑convolution.

4️⃣  local polishing – greedy pairwise mass‑transfer followed by a cheap
    simulated‑annealing style search cleans up any remaining irregularities.

All steps respect the global 1000 s budget.  The returned list is a
plain Python list; ``evaluate_sequence`` will normalise it automatically.
"""
import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a (real) FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    # round up to next power of two for speed
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) with ∑a = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
#  ───────  Analytic families  (all return a probability vector – sum = 1) ───────
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    w /= w.sum()
    return w


def _distribution_exponential_center(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(i - half))
    w /= w.sum()
    return w


def _distribution_exponential_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-rate * i)
    w /= w.sum()
    return w


def _distribution_exponential_right(n: int, rate: float) -> np.ndarray:
    return _distribution_exponential_left(n, rate)[::-1]


def _distribution_dirichlet(n: int, rng: np.random.Generator, conc: float) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a.astype(np.float64)


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """U‑shaped / hump‑shaped beta density discretised on the centre of each cell."""
    t = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    t = np.clip(t, eps, 1.0 - eps)
    w = (t ** (alpha - 1.0)) * ((1.0 - t) ** (beta - 1.0))
    w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
#  ───────  ADAM‑based mirror descent (exact objective)  ───────
def _adam_refine(
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    lr0: float = 0.8,
    active_frac_start: float = 0.03,
    active_frac_end: float = 0.008,
    max_active: int = 30,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_interval: int = 250,
    best_seq: np.ndarray | None = None,
    best_score: float = float("inf"),
) -> tuple[np.ndarray, float]:
    """
    Optimise a probability vector using Adam on the log‑weights.
    ``active_frac`` is linearly annealed from *start* to *end*.
    """
    n = init_a.shape[0]
    x = np.log(np.maximum(init_a, 1e-12))
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    ind_grid = np.arange(n, dtype=np.int32)
    t = 0

    if best_seq is None:
        best_seq = init_a.copy()
        best_score = _objective(init_a)

    while time.time() < deadline and t < max_iters:
        t += 1
        # linearly anneal active fraction
        active_frac = active_frac_start + (active_frac_end - active_frac_start) * (t / max_iters)

        a = _softmax(x)          # lives on the simplex
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        # indices where the convolution is close to its maximum
        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            # keep the largest entries only
            top = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top]

        # sub‑gradient of the max (using the active set)
        grad_a = np.zeros_like(a)
        for idx_k in act:
            offset = idx_k - ind_grid
            mask = (offset >= 0) & (offset < n)
            grad_a[mask] += 2.0 * a[offset[mask]]

        # convert gradient w.r.t. a into gradient w.r.t. log‑weights x
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = lr0 / math.sqrt(t)
        x -= eta * m_hat / (np.sqrt(v_hat) + eps)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] adam it={t:,} new best {best_score:.6f}")

    best_seq = _project_onto_simplex(best_seq, s=1.0)
    return best_seq, best_score

#  ───────  Greedy pairwise refinement (same idea as the reference)  ───────
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score

#  ───────  Simple simulated‑annealing style local search  ───────
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

                top_candidates.append((sc, a.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, a.copy()
                    print(f"[{time.time():.1f}] new global best Dirichlet conc={conc:.3g} score={sc:.6f}")

        for _ in range(3):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}")

        # keep only the very best seeds (global across all lengths)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:20]

    # --------------------------------------------------------------
    # try to use a previously found construction if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
            top_candidates.sort(key=lambda x: x[0])
            top_candidates = top_candidates[:20]
    except Exception:
        pass

    # --------------------------------------------------------------
    # 2️⃣  Short ADAM refinements on each retained seed
    refined: list[tuple[float, np.ndarray]] = []
    for sc, cand in top_candidates:
        if time.time() >= deadline:
            break
        # allocate a tiny slice of the remaining time for each short run
        slice_deadline = time.time() + min(15.0, (deadline - time.time()) * 0.04)
        best_seq, best_sc = _adam_refine(
            cand.copy(),
            slice_deadline,
            max_iters=12_000,
            lr0=0.7,
            active_frac_start=0.04,
            active_frac_end=0.015,
            max_active=20,
            beta1=0.9,
            beta2=0.999,
            eval_interval=200,
            best_seq=cand,
            best_score=sc,
        )
        refined.append((best_sc, best_seq))

    # keep the best after the short runs
    if refined:
        overall_best_score, overall_best_seq = min(refined, key=lambda x: x[0])

    # --------------------------------------------------------------
    # 3️⃣  Long ADAM descent from the overall best seed
    if time.time() < deadline:
        remaining = deadline - time.time()
        long_deadline = time.time() + max(0.0, remaining - 30.0)
        print(f"[{time.time():.1f}] start long ADAM descent (≈{remaining-30:.1f}s left)")
        overall_best_seq, overall_best_score = _adam_refine(
            overall_best_seq.copy(),
            long_deadline,
            max_iters=600_000,
            lr0=0.9,
            active_frac_start=0.02,
            active_frac_end=0.005,
            max_active=30,
            beta1=0.9,
            beta2=0.999,
            eval_interval=250,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] after long descent best score {overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Sparsification + short re‑refinement
    if time.time() < deadline:
        # zero out tiny entries (relative to the current maximum)
        thresh = 0.001 * overall_best_seq.max()
        mask = overall_best_seq > thresh
        if mask.sum() < overall_best_seq.shape[0]:
            sparsified = overall_best_seq * mask
            sparsified = _project_onto_simplex(sparsified, s=1.0)
            print(f"[{time.time():.1f}] sparsified to {mask.sum()} non‑zeros")
            # short re‑run of ADAM on the sparsified vector
            sp_deadline = time.time() + min(30.0, deadline - time.time())
            overall_best_seq, overall_best_score = _adam_refine(
                sparsified,
                sp_deadline,
                max_iters=30_000,
                lr0=0.5,
                active_frac_start=0.015,
                active_frac_end=0.003,
                max_active=20,
                beta1=0.9,
                beta2=0.999,
                eval_interval=200,
                best_seq=sparsified,
                best_score=_objective(sparsified),
            )
            print(f"[{time.time():.1f}] after sparsify‑refine score {overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # --------------------------------------------------------------
    # 6️⃣  Final simulated‑annealing polishing
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.01,
            decay=0.9995,
        )

    # --------------------------------------------------------------
    print(f"Final best score (exact) ≈ {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # The evaluator will re‑normalise; we keep a proper probability vector.
    return overall_best_seq.tolist()
