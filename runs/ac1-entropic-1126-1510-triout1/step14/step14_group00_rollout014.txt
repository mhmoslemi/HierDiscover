The function `propose_candidate` builds a non‑negative step function that tries to minimise  

\[
\text{score}= \frac{2\,n\;\max (a*a)}{(\sum a)^2},
\qquad a\in\mathbb R_{\ge0}^n .
\]

The strategy is a multi‑phase search:

1. **Rich seeding** – a large variety of analytic families (uniform,
   arcsine, Hann, edge‑weighted, symmetric β, generalized‑normal,
   Kaiser‑type) is sampled for several lengths.  
   The 30‑50 best seeds are kept.

2. **Two complementary polishing methods**  
   * *Exponentiated‑gradient* (mirror descent) – the same robust routine
     used in the reference implementation.  
   * *Smoothed‑gradient descent* – we replace the nondifferentiable
     `max` by a log‑sum‑exp with a continuation schedule
     (`γ = 5,10,20,40,…`).  The gradient of the smoothed objective is
     obtained analytically by a single FFT‑based convolution, so each
     iteration is cheap.

3. **Progressive up‑sampling** – the current best vector is linearly
   interpolated to double its length while the total mass stays 1.
   After each up‑sample a short polishing run (both optimisers) is
   performed.  Upsampling is repeated as long as time permits (up to
   ≈ 131 072 entries).

4. **Final annealing** – a simple pair‑wise mass‑transfer simulated
   annealing removes any remaining tiny spikes.

The whole pipeline respects the 1000 s budget (a 2 s safety margin).
All helper functions are pure top‑level functions – no closures, no
file I/O, and only `numpy` is required.

"""
Search for a non‑negative step function that minimises
    2·n·max(conv(a,a)) / (∑a)²

Algorithm
---------
1.  Generate a diverse pool of analytic seeds (uniform, arcsine,
    Hann, edge‑weighted, symmetric Beta, generalized‑normal,
    Kaiser‑type, sparse Dirichlet).  Keep the best ~30‑50.
2.  Polish every seed with a short exponentiated‑gradient run.
3.  Polish the global best with a longer run of the same method.
4.  Repeatedly up‑sample (×2) the best vector while time permits.
    After each up‑sample run a few iterations of
    *exponentiated‑gradient* **and**
    *smoothed‑gradient descent* (log‑sum‑exp continuation).
5.  Finish with pairwise mass‑transfer simulated annealing.

All vectors are normalised to sum = 1, guaranteeing that the objective
is exactly `2·n·max(conv)`.  The implementation uses only NumPy and
respects the 1000 s time budget.
"""

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used by the Euclidean PGD variant
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (linear convolution)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective for a probability vector (∑a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv).  Input must already sum to 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families – every function returns a *probability* vector
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{-p} with i,j distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density sampled on cell mid‑points."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # avoid overflow/underflow for extreme α
    w = np.where(
        (x == 0) | (x == 1),
        0.0,
        np.exp((alpha - 1) * np.log(x) + (alpha - 1) * np.log1p(-x))
    )
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(‑|x‑½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_kaiser(n: int, beta: float) -> np.ndarray:
    """Modified Bessel Kaiser window (β≥0)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = 2.0 * (i / n - 0.5)               # maps to [-1,1]
    # numpy has the order‑0 Modified Bessel function I0
    w = np.i0(beta * np.sqrt(1.0 - x * x))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex blend base·(1‑λ) + uniform·λ."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Sub‑gradient of the *max* entry (used by exponentiated‑gradient)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Returns (grad, k) where k is the index attaining max conv.
    grad_i = 2·a_{k‑i} for i that participates, otherwise 0.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))

    grad = np.zeros_like(a)
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) polishing
# ----------------------------------------------------------------------
def _adaptive_exp_gradient(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    beta: float = 0.9,
    eval_every: int = 50,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact objective.
    Multiplicative (exponentiated) updates keep the vector on the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)          # EMA of gradients
    eta = eta0
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        m = beta * m + (1.0 - beta) * grad
        a = a * np.exp(-eta * m)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                eta = min(eta0, eta * 1.02)      # reward improvement
                print(f"[{time.time():.1f}] exp‑GD it={t:,} new best {best_score:.6f}")
            else:
                eta *= 0.995   # gentle decay

        # occasional re‑projection (numerical drift)
        if t % (eval_every * 10) == 0:
            a = _project_onto_simplex(a, s=1.0)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smoothed‑gradient descent (log‑sum‑exp continuation)
# ----------------------------------------------------------------------
def _smooth_objective_and_grad(
    a: np.ndarray, gamma: float
) -> tuple[float, np.ndarray]:
    """
    Returns (smooth_obj, grad_a) where
        smooth_obj = 2·n·( max(conv) + log Σ exp(gamma·(conv‑max))/γ )
    and grad_a = ∂/∂a smooth_obj.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)               # length 2n‑1
    maxc = conv.max()
    # stable softmax
    exp_term = np.exp(gamma * (conv - maxc))
    sum_exp = exp_term.sum()
    smooth_max = maxc + np.log(sum_exp) / gamma
    obj = 2.0 * n * smooth_max

    # normalized weights p_k = exp(gamma·c_k)/Σexp(...)
    p = exp_term / sum_exp

    # weighted gradient  g_i = 2· Σ_k p_k a_{k‑i}
    # use convolution with reversed p
    rev_p = p[::-1]                               # length 2n‑1
    # convolution length = n + (2n‑1) – 1 = 3n‑2
    size = 1 << ((3 * n - 2 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_p, n=size)
    conv_rev = np.fft.irfft(A * B, n=size)[: 3 * n - 2]
    # extract needed part: indices 2n‑2‑i
    grad = 2.0 * conv_rev[2 * n - 2 - np.arange(n)]
    return obj, grad


def _smooth_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    gamma_schedule: list[float] = (5.0, 10.0, 20.0, 40.0),
    eta0: float = 0.5,
    eval_every: int = 30,
) -> tuple[np.ndarray, float]:
    """
    Mirror descent on the *smoothed* objective.
    The continuation on γ gradually turns the smooth problem into the
    original max‑objective.
    """
    a = a0.copy()
    n = a.shape[0]

    for gamma in gamma_schedule:
        eta = eta0
        t = 0
        while time.time() < deadline:
            t += 1
            obj_smooth, grad = _smooth_objective_and_grad(a, gamma)

            # multiplicative (exponentiated) update
            a = a * np.exp(-eta * grad)
            s = a.sum()
            if s == 0.0:
                a = np.full(n, 1.0 / n, dtype=np.float64)
            else:
                a /= s

            if t % eval_every == 0:
                cur = _objective(a)                 # exact objective
                if cur < best_score - 1e-15:
                    best_score = cur
                    best_seq = a.copy()
                    eta = min(eta0, eta * 1.02)
                    print(f"[{time.time():.1f}] smooth‑GD γ={gamma:.0f} it={t:,} new best {best_score:.6f}")
                else:
                    eta *= 0.995

            # re‑project rarely
            if t % (eval_every * 10) == 0:
                a = _project_onto_simplex(a, s=1.0)

            if time.time() >= deadline:
                break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise mass‑transfer simulated annealing (final polishing)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        # choose a source with non‑negligible mass
        src = rng.choice(np.where(a > 1e-14)[0])
        # pick a distinct destination uniformly
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # move a random fraction (up to 30 %) of src's mass
        delta = a[src] * rng.random() * 0.3
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        cur = _objective(new_a)

        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.02

        accept = False
        if cur < best_score - 1e-15:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal {src}->{dst} Δ={delta:.1e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass = 1)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new

        # keep only the top‑40 seeds overall (memory bounded)
        cand_pool.sort(key=lambda x: x[0])
        cand_pool = cand_pool[:40]

    # ------------------------------------------------------------------
    # 2️⃣  Incorporate possible previous construction
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.abs(prev)
            s = prev.sum()
            if s == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            else:
                prev = prev / s
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
            cand_pool.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3️⃣  Fallback if nothing was generated
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 2048
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Short exponential‑gradient polishing for every seed
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 12.0 and cand_pool:
        per_seed = max(0.1, remaining * 0.25 / max(len(cand_pool), 1))
        print(f"[{time.time():.1f}] quick exp‑GD on {len(cand_pool)} seeds (~{per_seed:.2f}s each)")
        for idx, (sc, a0) in enumerate(cand_pool):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            a_opt, sc_opt = _adaptive_exp_gradient(
                a0.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.7,
                beta=0.9,
                eval_every=30,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after quick exp‑GD seed {idx} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Longer polishing on the *global* best (first exp‑GD, then smooth)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # longer exponentiated‑gradient run
        sub_dead = time.time() + (deadline - time.time()) * 0.30
        a_opt, sc_opt = _adaptive_exp_gradient(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.5,
            beta=0.9,
            eval_every=60,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] long exp‑GD improved best to {overall_best_score:.6f}")

        # followed by a smoothed‑gradient pass
        sub_dead = time.time() + (deadline - time.time()) * 0.20
        a_opt, sc_opt = _smooth_gradient_descent(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            gamma_schedule=[5, 10, 20, 40],
            eta0=0.5,
            eval_every=50,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] smooth‑GD improved best to {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Progressive up‑sampling (doubling size while time permits)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n > 131072:        # keep FFT sizes reasonable
            break
        print(f"[{time.time():.1f}] up‑sampling {cur_n} → {target_n}")
        up = _upsample(overall_best_seq, target_n)
        up_sc = _objective(up)
        if up_sc < overall_best_score:
            overall_best_seq, overall_best_score = up, up_sc
            print(f"[{time.time():.1f}] up‑sampled new best {overall_best_score:.6f}")

        # medium polishing on the up‑sampled vector
        sub_dead = time.time() + (deadline - time.time()) * 0.18
        a_opt, sc_opt = _adaptive_exp_gradient(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.45,
            beta=0.9,
            eval_every=50,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
            print(f"[{time.time():.1f}] post‑up‑sample exp‑GD new best {overall_best_score:.6f}")

        # a quick smooth‑GD pass
        sub_dead = time.time() + (deadline - time.time()) * 0.12
        a_opt, sc_opt = _smooth_gradient_descent(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            gamma_schedule=[5, 10, 20],
            eta0=0.45,
            eval_every=40,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
            print(f"[{time.time():.1f}] post‑up‑sample smooth‑GD new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣  Final annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing phase")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
            print(f"[{time.time():.1f}] after annealing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)) = {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    # ``evaluate_sequence`` expects a plain Python list
    return overall_best_seq.tolist()
