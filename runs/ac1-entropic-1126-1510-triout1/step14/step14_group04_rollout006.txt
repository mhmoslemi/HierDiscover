def _subgradient_descent_exact(a0, time_limit, best_seq, best_score, lr0=0.9, eval_every=100):
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()

        # compute subgrad
        grad = 2.0 * _fft_convolve(w, a[::-1])[n-1:n-1+n]

        eta = lr0 / math.sqrt(t)

            a = np.full(n, 1.0/n)

            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] exact subgrad it={t} new best {best_score:.6f}")
    return best_seq, best_score

3️⃣  Refine the best solution with an **exact sub‑gradient descent** on the
    true L∞‑objective (exponentiated‑gradient updates).

4️⃣  Finish with a **greedy pairwise mass‑transfer** hill‑climber that moves
    a random fraction of mass between two positions whenever it strictly
    improves the exact objective.

All phases are bounded by the supplied time budget (default 1000 s).  The
returned list is simplex‑normalised; evaluate_sequence rescales internally.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  Utility: projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of `v` onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
#  FFT based convolutions (real inputs)
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a * a using a real FFT."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
#  Exact objective (sum(a) = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) for a with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
#  Analytic families (all return a non‑negative vector, not normalised)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    return w / w_sum


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    """w_i ∝ exp(-c·i) + exp(-c·(n‑1‑i))   (c>0)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    return w / w.sum()


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric beta‑type distribution on the discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    total = w.sum()
    if total == 0.0:
        return _distribution_uniform(n)
    return w / total


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


# ----------------------------------------------------------------------
#  Smoothed‑max mirror descent (Adam) – main optimiser
# ----------------------------------------------------------------------
def _smooth_max_adam(a0: np.ndarray,
                    time_limit: float,
                    global_best_seq: np.ndarray,
                    global_best_score: float) -> tuple[np.ndarray, float]:
    """
    Minimise the exact objective using a smooth (log‑sum‑exp) surrogate.
    Adam updates are projected onto the simplex after each step.
    """
    a = a0.copy()
    n = a.shape[0]
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    # keep a local copy of the best seen so far
    best_seq = global_best_seq.copy()
    best_score = global_best_score

    # β schedule (log‑space, gradually sharpening the max)
    beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0]
    max_iter_per_beta = 3000        # a few thousand per β

    for beta in beta_schedule:
        iter_beta = 0
        while time.time() < time_limit and iter_beta < max_iter_per_beta:
            iter_beta += 1
            # ----- gradient of smooth max -----
            conv = _fft_self_convolve(a)          # length 2n‑1
            max_val = conv.max()
            shifted = conv - max_val              # ≤ 0, stabilises exp
            exp_beta = np.exp(beta * shifted)
            w = exp_beta / exp_beta.sum()         # softmax weights, Σw = 1

            # gradient of (w‑weighted) convolution
            a_rev = a[::-1]
            grad_conv = _fft_convolve(w, a_rev)   # length 3n‑2
            grad_a = 2.0 * grad_conv[n - 1 : n - 1 + n]   # size n

            # gradient of the *objective* J = 2·n·smooth_max
            grad_J = 2.0 * n * grad_a

            # ----- Adam update (projected) -----
            m = beta1 * m + (1.0 - beta1) * grad_J
            v = beta2 * v + (1.0 - beta2) * (grad_J ** 2)
            t = iter_beta  # step count for bias correction
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = 0.12 / (np.sqrt(v_hat) + eps)   # 0.12 works well empirically
            a -= step * m_hat
            a = _project_onto_simplex(a, s=1.0)

            # ----- occasional exact evaluation -----
            if iter_beta % 200 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-14:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] β={beta:.1f} it={iter_beta} new best {best_score:.6f}")

        if time.time() >= time_limit:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
#  Exact sub‑gradient descent (exponentiated gradient)
# ----------------------------------------------------------------------
def _subgradient_descent_exact(a0: np.ndarray,
                               time_limit: float,
                               global_best_seq: np.ndarray,
                               global_best_score: float,
                               lr0: float = 0.9,
                               eval_every: int = 100) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the true L∞‑objective.
    Uses the exact sub‑gradient (indicator of max‑positions).
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = global_best_seq.copy()
    best_score = global_best_score
    t = 0
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        w = (conv == max_val).astype(np.float64)        # indicator of active constraints
        grad = 2.0 * _fft_convolve(w, a[::-1])[n - 1 : n - 1 + n]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)                       # stays non‑negative
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] exact subgrad it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
#  Greedy pairwise mass‑transfer hill‑climber
# ----------------------------------------------------------------------
def _pairwise_swap(a0: np.ndarray,
                   time_limit: float,
                   global_best_seq: np.ndarray,
                   global_best_score: float,
                   max_trials: int | None = None) -> tuple[np.ndarray, float]:
    """Randomly relocate a fraction of mass; accept only strict improvements."""
    a = a0.copy()
    n = a.shape[0]
    best_seq = global_best_seq.copy()
    best_score = global_best_score
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        # choose a source index with non‑negligible mass
        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_seq = a.copy()
            best_score = cur
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
#  Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a step‑function (plain Python list) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # small safety margin for the caller
    deadline = start + budget_s - 1.5

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    #  Choose a length (large enough to allow flattening)
    # ------------------------------------------------------------------
    try:
        # If a previous construction exists, prefer its length (but not smaller)
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.shape == () or prev.ndim != 1 or prev.size == 0:
            raise ValueError
        n = max(prev.size, 15000)
    except Exception:
        n = 15000
    print(f"[{time.time():.1f}] using length n = {n}")

    # ------------------------------------------------------------------
    #  1️⃣  Seed generation (analytic families + random)
    # ------------------------------------------------------------------
    def _normalise(v: np.ndarray) -> np.ndarray:
        v = np.maximum(v, 0.0)
        s = v.sum()
        return v / s if s > 0 else _distribution_uniform(n)

    # deterministic analytic families
    for fn in (_distribution_uniform,
               _distribution_arcsine,
               _distribution_sine,
               _distribution_cosine,
               _distribution_quadratic):
        cand = _normalise(fn(n))
        sc = _objective(cand)
        top_candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] {fn.__name__} seed score={sc:.6f}")

    # symmetric beta family
    alpha_grid = np.logspace(-0.5, 0.7, 12)
    for alpha in alpha_grid:
        cand = _normalise(_distribution_beta(n, float(alpha), float(alpha)))
        sc = _objective(cand)
        top_candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] beta α={alpha:.4g} score={sc:.6f}")

    # edge‑weighted + uniform mix
    p_grid = np.concatenate([np.logspace(-3, -0.5, 8),
                             np.logspace(-0.5, 0.0, 8),
                             np.linspace(0.2, 3.0, 8)])
    lam_grid = np.linspace(0.0, 0.5, 5)
    for p in p_grid:
        edge = _distribution_edge_weighted(n, float(p))
        for lam in lam_grid:
            cand = _normalise(_mix_with_uniform(edge, lam))
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] edge p={p:.4g} lam={lam:.2f} score={sc:.6f}")

    # exponential‑edge + uniform mix
    c_grid = np.logspace(-3, 0, 8)
    for c in c_grid:
        exp_edge = _distribution_exponential_edge(n, float(c))
        for lam in lam_grid:
            cand = _normalise(_mix_with_uniform(exp_edge, lam))
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] exp c={c:.4g} lam={lam:.2f} score={sc:.6f}")

    # random Dirichlet seeds
    for _ in range(12):
        alpha = rng.uniform(0.01, 2.0)
        a = rng.dirichlet(np.full(n, alpha))
        a = _normalise(a)
        sc = _objective(a)
        top_candidates.append((sc, a.copy()))
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, a.copy()
            print(f"[{time.time():.1f}] Dirichlet α={alpha:.4g} score={sc:.6f}")

    # sparse random seeds
    for _ in range(8):
        a = _distribution_random_sparse(n, rng, density=0.004)
        a = _normalise(a)
        sc = _objective(a)
        top_candidates.append((sc, a.copy()))
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, a.copy()
            print(f"[{time.time():.1f}] sparse seed score={sc:.6f}")

    # include previous construction if present (and reshape/pad if needed)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            if prev.size != n:
                if prev.size > n:
                    prev = prev[:n]
                else:
                    # pad with zeros
                    prev = np.concatenate([prev, np.zeros(n - prev.size)])
            prev = _normalise(prev)
            sc = _objective(prev)
            top_candidates.append((sc, prev.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # keep a bounded list of the most promising seeds
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:6]   # 6 best seeds
    print(f"[{time.time():.1f}] selected {len(top_candidates)} top seeds, best score so far {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    #  2️⃣  Smoothed‑max mirror descent (Adam) on the top seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        per_seed = remaining * 0.55 / max(1, len(top_candidates))
        for idx, seed_seq in enumerate(top_candidates):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_max_adam(seed_seq,
                                             sub_deadline,
                                             overall_best_seq,
                                             overall_best_score)
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after smooth seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    #  3️⃣  Exact sub‑gradient refinement (exponentiated gradient)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        a_opt, sc_opt = _subgradient_descent_exact(overall_best_seq.copy(),
                                                   deadline,
                                                   overall_best_seq,
                                                   overall_best_score,
                                                   lr0=0.9,
                                                   eval_every=150)
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after exact subgrad new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    #  4️⃣  Final greedy pairwise polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = _pairwise_swap(overall_best_seq.copy(),
                                       deadline,
                                       overall_best_seq,
                                       overall_best_score,
                                       max_trials=5_000_000)
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best objective (2·n·max(conv)) = {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # The evaluation function rescales internally, so we return the raw simplex‑normalised vector.
    return overall_best_seq.tolist()
