"""
Propose a step‑function (non‑negative vector) that minimises ``evaluate_sequence``.

Algorithm
---------

1️⃣  **Diverse seeding** –  many analytically‑motivated families (uniform,
    sine‑powered, Gaussian, centred‑quadratic, “U‑shaped’’ negative‑exponent,
    raised‑cosine, β‑distribution, exponential, Dirichlet, sparse random) are
    sampled for several lengths.  The 15 best seeds (lowest exact objective)
    are retained.

2️⃣  **Adam mirror‑descent** –  each retained seed is refined with Adam on the
    soft‑max parametrisation.  The sub‑gradient uses all convolution entries
    whose value is within a tiny fraction of the current maximum (active set).
    A short run is performed for every seed and a long run for the overall best.

3️⃣  **Random perturb‑restart** –  the best sequence is perturbed by small
    Gaussian noise, re‑projected on the simplex and refined again; a few such
    restarts are carried out if time permits.

4️⃣  **Local polishing** –  a greedy pairwise mass‑transfer step followed by a
    cheap simulated‑annealing search removes any remaining irregularities.

All steps respect the global 1000 s budget.  The returned list can be fed
directly to ``evaluate_sequence`` – the evaluator will normalise it anyway.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())          # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))   (a is assumed to sum to 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families – each returns a probability vector (sum = 1)
# ----------------------------------------------------------------------


def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^p .  p can be negative."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid zero for negative powers
    w = np.sin(theta) ** p
    w = np.where(np.isfinite(w), w, 0.0)
    w[w < 0] = 0.0
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    return w / w_sum


def _distribution_exponential_center(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(i - half))
    w /= w.sum()
    return w


def _distribution_exponential_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-rate * i)
    w /= w.sum()
    return w


def _distribution_exponential_right(n: int, rate: float) -> np.ndarray:
    return _distribution_exponential_left(n, rate)[::-1]


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp( -(i - μ)^2 / (2σ²) )   with μ at the centre."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    return w / w_sum


def _distribution_quadratic(n: int) -> np.ndarray:
    """a_i ∝ i·(n‑1‑i)   a symmetric hill, zero at the edges."""
    i = np.arange(n, dtype=np.float64)
    w = i * (n - 1 - i)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    return w / w_sum


def _distribution_u_shape(n: int, exp: float) -> np.ndarray:
    """
    “U‑shaped’’ family: a_i ∝ ((i+0.5)*(n‑i‑0.5))**exp.
    Negative exp concentrates mass near the ends.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    prod = i * (n - i)
    # protect against zero – prod > 0 always because of 0.5 offset
    w = prod ** exp
    w = np.where(np.isfinite(w), w, 0.0)
    w[w < 0] = 0.0
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    return w / w_sum


def _distribution_raised_cosine(n: int) -> np.ndarray:
    """a_i ∝ 0.5 - 0.5·cos(2π·i/(n‑1))  – a smooth bump."""
    if n == 1:
        return np.array([1.0])
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w /= w.sum()
    return w


def _distribution_beta(n: int, a: float, b: float) -> np.ndarray:
    """Discrete Beta‑like shape: a_i ∝ (i+1)^{a‑1} · (n‑i)^{b‑1}."""
    if a <= 0 or b <= 0:
        a = max(a, 1e-6)
        b = max(b, 1e-6)
    i = np.arange(n, dtype=np.float64)
    w = (i + 1) ** (a - 1) * (n - i) ** (b - 1)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    return w / w_sum


def _distribution_dirichlet(n: int, rng: np.random.Generator, conc: float) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a.astype(np.float64)


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    """Few non‑zero entries, random positions."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Adam‑based mirror descent (hard‑max version)

def _adam_refine(
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    lr0: float = 0.8,
    active_frac: float = 0.02,
    max_active: int = 30,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_interval: int = 250,
    best_seq: np.ndarray | None = None,
    best_score: float = float("inf"),
) -> tuple[np.ndarray, float]:
    """
    Adam on the soft‑max parametrisation.
    Sub‑gradient is assembled from active entries of the self‑convolution.
    """
    n = init_a.shape[0]
    x = np.log(np.maximum(init_a, 1e-12))          # log‑weights
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    ind_grid = np.arange(n, dtype=np.int32)

    if best_seq is None:
        best_seq = init_a.copy()
        best_score = _objective(init_a)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        a = _softmax(x)                 # lies on the simplex
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            # keep the largest max_active entries only (fast)
            top = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top]

        # sub‑gradient of max (active set)
        grad_a = np.zeros_like(a)
        for k in act:
            offset = k - ind_grid
            mask = (offset >= 0) & (offset < n)
            grad_a[mask] += 2.0 * a[offset[mask]]

        # convert to gradient w.r.t. log‑weights (mirror‑descent rule)
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update with decaying step size
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = lr0 / math.sqrt(t)
        x -= eta * m_hat / (np.sqrt(v_hat) + eps)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] adam it={t:,} new best {best_score:.6f}")

    # final safety projection (soft‑max already normalises)
    best_seq = _project_onto_simplex(best_seq, s=1.0)
    return best_seq, best_score

def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Greedy transfer of a random amount of mass from i → j."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


def _random_local_search(
    a: np.ndarray,

            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a probability vector that minimises ``evaluate_sequence``.
    The function respects the global time budget; the returned list can be
    fed directly to ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1️⃣  Seeding – diverse analytic families
    # ------------------------------------------------------------------
    lengths = [1800, 2600, 3400, 4600, 6300, 8800, 12000]   # moderate spread
    # parameter grids
    p_grid = np.logspace(-1, 1.0, 7)          # sine‑weighted exponent (both +/–)
    neg_p_grid = np.array([-0.5, -1.0, -1.5, -2.0])  # negative exponents for “U‑shape’’
    sigma_grid = np.linspace(0.03, 0.25, 5)    # Gaussian widths (as fraction of n)
    rate_grid = np.linspace(0.001, 0.02, 5)   # exponential decay rates
    conc_grid = [0.05, 0.2, 1.0, 5.0]        # Dirichlet concentrations
    lam_grid = np.linspace(0.0, 0.7, 5)      # mixing with uniform

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # ----- deterministic analytic families -----
        families = []

        families.append(_distribution_uniform(n))

        # sine‑weighted, both positive and negative powers
        for p in p_grid:
            families.append(_distribution_sine_weighted(n, p))
        for p in neg_p_grid:
            families.append(_distribution_sine_weighted(n, p))

        families.append(_distribution_exponential_center(n, rate=0.005))
        families.append(_distribution_exponential_left(n, rate=0.003))
        families.append(_distribution_exponential_right(n, rate=0.003))

        # Gaussian with different sigmas (fraction of n)
        for frac in sigma_grid:
            sigma = max(0.5, frac * n)   # ensure sigma not too tiny
            families.append(_distribution_gaussian(n, sigma))

        # Quadratic hill
        families.append(_distribution_quadratic(n))

        # U‑shaped (negative exponent)
        for exp in neg_p_grid:
            families.append(_distribution_u_shape(n, exp))

        # Raised cosine
        families.append(_distribution_raised_cosine(n))

        # Beta‑like symmetric families
        for a_par in [0.3, 0.6, 1.0, 2.0, 5.0]:
            families.append(_distribution_beta(n, a_par, a_par))

        # ----- random Dirichlet / sparse seeds -----
        for conc in conc_grid:
            for _ in range(2):
                families.append(_distribution_dirichlet(n, rng, conc))

        for _ in range(3):
            families.append(_distribution_random_sparse(n, rng, density=0.006))

        # ----- mixing each candidate with uniform (convex combo) -----
        for base in families:
            for lam in lam_grid:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new global best (mixed) len={n} score={sc:.6f}")

        # keep only a limited number of very best seeds (to save time later)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:15]

    # ------------------------------------------------------------------
    # 2️⃣  Adam refinement – short runs on the retained seeds
    # ------------------------------------------------------------------
    refined: list[tuple[float, np.ndarray]] = []
    for sc, cand in top_candidates:
        if time.time() >= deadline:
            break
        # allocate a small time slice for each seed (≈5 s or 3 % of the remaining budget)
        slice_budget = min(6.0, max(0.0, (deadline - time.time()) * 0.03))
        slice_deadline = time.time() + slice_budget
        best_seq, best_sc = _adam_refine(
            cand.copy(),
            slice_deadline,
            max_iters=30_000,
            lr0=0.7,
            active_frac=0.03,
            max_active=25,
            beta1=0.9,
            beta2=0.999,
            eval_interval=150,
            best_seq=cand,
            best_score=sc,
        )
        refined.append((best_sc, best_seq))

    if refined:
        overall_best_score, overall_best_seq = min(refined, key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 3️⃣  Long Adam descent from the current overall best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        long_deadline = time.time() + max(0.0, remaining - 30.0)   # keep ~30 s for polishing
        print(f"[{time.time():.1f}] long Adam descent (≈{remaining-30:.1f}s left)")
        overall_best_seq, overall_best_score = _adam_refine(
            overall_best_seq.copy(),
            long_deadline,
            max_iters=600_000,
            lr0=0.9,
            active_frac=0.015,
            max_active=35,
            beta1=0.9,
            beta2=0.999,
            eval_interval=250,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] after long Adam best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Random perturb‑restart cycles (if time permits)
    # ------------------------------------------------------------------
    while time.time() + 12.0 < deadline:
        # small Gaussian perturbation, re‑normalise, short Adam refinement
        n = overall_best_seq.shape[0]
        noise = rng.normal(scale=0.02, size=n)
        pert = overall_best_seq + noise
        pert = np.maximum(pert, 0.0)
        pert = _project_onto_simplex(pert, s=1.0)
        slice_deadline = time.time() + 9.0
        pert_seq, pert_sc = _adam_refine(
            pert,
            slice_deadline,
            max_iters=40_000,
            lr0=0.6,
            active_frac=0.025,
            max_active=30,
            beta1=0.9,
            beta2=0.999,
            eval_interval=150,
            best_seq=pert,
            best_score=_objective(pert),
        )
        if pert_sc < overall_best_score - 1e-14:
            overall_best_seq, overall_best_score = pert_seq, pert_sc
            print(f"[{time.time():.1f}] restart improved to {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Greedy pairwise refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 6️⃣  Final cheap simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.012,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    # 7️⃣  If a previous construction is available, give it a chance
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best improved score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] FINAL best score ≈ {overall_best_score:.6f} (len={len(overall_best_seq)})")
    # Return a plain Python list; the external evaluator will re‑normalise anyway.
    return overall_best_seq.tolist()
