    on a fine grid of parameters and keep the best few candidates.

3️⃣  Refine the best candidate with an exact sub‑gradient exponentiated‑gradient
    descent (the objective is convex in the probability simplex).  The
    sub‑gradient of  max(a*a) is computed analytically, the update is
    ``a_i ← a_i·exp(-η·grad_i)`` and the vector is projected back onto the
    simplex.  The step‑size η = lr/√t anneals automatically.

4️⃣  Return the best sequence found (a plain Python list).  The evaluator
    will clamp negatives/large values, so we keep the vector normalised
    (sum≈1) and all entries safely below 1000.

The whole routine stays well within the 1000 s budget on a single core.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# Helper utilities
# -------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    # next power‑of‑two that fits 2·n‑1 samples
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# -------------------------------------------------------------
# Analytic families (return a probability vector, i.e. sum = 1)
# -------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_edge(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta‑type density  w_i ∝ (i·(n‑i))^{α‑1} (α>0)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    # avoid overflow / underflow by working in log‑space when α is far from 1
    if abs(alpha - 1.0) > 0.5:
        # log‑space computation
        log_vals = (alpha - 1.0) * (np.log(i) + np.log(n - i))
        vals = np.exp(log_vals - np.max(log_vals))   # normalise later
    else:
        vals = (i * (n - i)) ** (alpha - 1.0)
    vals = np.clip(vals, 0.0, None)
    s = vals.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    vals /= s
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight edges with exponent p (p>0 → stronger edge concentration)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = (i * (n - i)) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    w /= w_sum
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    """Two‑sided exponential decay from both ends."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 0.5) -> np.ndarray:
    """Simple Dirichlet draw – used only for seeding."""
    a = rng.dirichlet(np.full(n, alpha))
    a /= a.sum()
    return a


# -------------------------------------------------------------
# Exact objective (sum = 1 ⇒ 2·n·max(conv))
# -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# -------------------------------------------------------------
# Exact sub‑gradient exponentiated‑gradient descent
# -------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Project‑gradient descent on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            # occasional progress report (kept lightweight)
            if it % 200 == 0:
                print(f"[{time.time():.1f}] subgrad it={it} new best {best_score:.6f}")

        # indices where the maximum is attained
        max_idxs = np.where(conv == max_val)[0]

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]   # derivative of a_i*a_j + a_j*a_i, i≠j; diagonal counted once

        # EG update
        eta = lr0 / math.sqrt(it)
        a *= np.exp(-eta * grad)

        # re‑project onto the simplex (preserves non‑negativity)
        s = a.sum()
        if s <= 0.0:
            a = _distribution_uniform(n)
        else:
            a /= s

    return best_seq, best_score


# -------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine is deterministic (apart from a
    short random seeding) and respects the overall time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0           # 1 s safety margin

    # -----------------------------------------------------------------
    # 1. Choose a sufficiently large length – we keep it fixed to avoid
    #    the overhead of repeatedly reallocating huge arrays.
    # -----------------------------------------------------------------
    # 25 k points provide a good trade‑off between discretisation error
    # and execution time (convolution via FFT costs ≈ 0.01 s).
    n = 25000

    # -----------------------------------------------------------------
    # 2. Scan a compact set of analytic families.
    # -----------------------------------------------------------------
    best_seq = _distribution_uniform(n)   # fallback
    best_score = _objective(best_seq)

    # families without parameters
    families = [
        _distribution_uniform,
        _distribution_arcsine,
        _distribution_sine,
        _distribution_cosine,
        _distribution_quadratic,
    ]

    for fn in families:
        cand = fn(n)
        sc = _objective(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] new best (plain) {best_score:.6f}")

    # parameter grids
    alpha_grid = np.concatenate([
        np.logspace(np.log10(0.01), np.log10(0.3), 12),
        np.linspace(0.3, 2.0, 12),
    ])
    p_grid = np.concatenate([
        np.logspace(-3, -0.5, 12),
        np.logspace(-0.5, 0.0, 12),
        np.linspace(0.1, 2.0, 12),
    ])
    c_grid = np.logspace(-4, 0, 12)
    lam_grid = [0.0, 0.25, 0.5, 0.75, 1.0]   # mixing with uniform

    # beta‑type family
    for alpha in alpha_grid:
        base = _distribution_beta_edge(n, float(alpha))
        for lam in lam_grid:
            cand = _mix_with_uniform(base, lam)
            sc = _objective(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] β α={alpha:.4g} lam={lam:.2f} → {best_score:.6f}")

    # edge‑weighted family
    for p in p_grid:
        base = _distribution_edge_weighted(n, float(p))
        for lam in lam_grid:
            cand = _mix_with_uniform(base, lam)
            sc = _objective(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] edge p={p:.4g} lam={lam:.2f} → {best_score:.6f}")

    # exponential‑edge family
    for c in c_grid:
        base = _distribution_exponential_edge(n, float(c))
        for lam in lam_grid:
            cand = _mix_with_uniform(base, lam)
            sc = _objective(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] exp c={c:.4g} lam={lam:.2f} → {best_score:.6f}")

    # a few random Dirichlet seeds – cheap cheap exploration
    for _ in range(6):
        a = _random_dirichlet(n, rng, alpha=rng.uniform(0.01, 2.0))
        sc = _objective(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time():.1f}] random Dirichlet → {best_score:.6f}")

    # -----------------------------------------------------------------
    # 3. Refine the best seed with exact sub‑gradient descent.
    #    We allocate the remaining time (capped at ~0.9·budget).
    # -----------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] start sub‑gradient polishing (n={n})")
        best_seq, best_score = _subgradient_descent(
            best_seq,
            deadline,
            best_seq,
            best_score,
            lr0=1.0,
        )
        print(f"[{time.time():.1f}] polishing finished – best {best_score:.6f}")

    # -----------------------------------------------------------------
    # 4. Return the plain list (the evaluator will apply its own clamps).
    # -----------------------------------------------------------------
    # Ensure we meet the minimal sum constraint of evaluate_sequence
    if best_seq.sum() < 0.01:
        best_seq = _distribution_uniform(n)   # should never happen
    return best_seq.tolist()
```
