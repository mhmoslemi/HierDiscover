def _smooth_gradient_descent(a, deadline, best_seq, best_score, start_beta=0.5, beta_factor=1.2, beta_max=50.0, step0=0.5, eval_interval=200):
    n = len(a)
    t = 0
    best_seq_local = best_seq.copy()
    best_score_local = best_score
    beta = start_beta
    while time.time() < deadline and beta <= beta_max:
        # number of iterations at this beta:
        inner_iters = 500   # may adapt
        for _ in range(inner_iters):
            if time.time() >= deadline:
                break
            t += 1
            # compute conv and p
            conv = _fft_self_convolve(a)
            max_c = conv.max()
            # compute soft weights
            # shift for numerical stability
            shift = conv - max_c
            # compute exp(beta * shift)
            # if beta large and shift negative large, exp underflows. Use np.exp with clipping maybe.
            exp_shift = np.exp(beta * shift)
            sum_exp = exp_shift.sum()
            p = exp_shift / sum_exp
            
            # compute gradient = 4 * n * (a * p) conv as described
            a_rev = a[::-1]
            # convolution of p (len 2n-1) and a_rev (len n)
            # Use fft
            # compute appropriate size
            need = len(p) + len(a_rev) - 1  # = 3n-2
            size = 1 << ((need - 1).bit_length())
            P_fft = np.fft.rfft(p, n=size)
            A_rev_fft = np.fft.rfft(a_rev, n=size)
            conv_rev = np.fft.irfft(P_fft * A_rev_fft, n=size)[:need]
            # extract middle part
            grad_part = conv_rev[n-1:2*n-1]  # length n
            gradient = 4.0 * n * grad_part
            
            # step size schedule
            eta = step0 / np.sqrt(t)
            a = a - eta * gradient
            # ensure nonnegative by projection onto simplex
            a = _project_onto_simplex(a, s=1.0)
            # optional evaluate actual objective occasionally
            if t % eval_interval == 0:
                cur_score = _objective(a)  # use exact conv max objective
                if cur_score < best_score_local - 1e-14:
                    best_score_local = cur_score
                    best_seq_local = a.copy()
                    print(f"[{time.time():.1f}] smooth descent it={t} β={beta:.2f} new best {best_score_local:.6f}")
        beta = min(beta * beta_factor, beta_max)
    return best_seq_local, best_score_local

"""
A fast multi‑stage optimizer for the “max self‑convolution” problem.

The pipeline:
1. Generate a few diverse seeds (uniform, arcsine, Dirichlet, etc.).
2. For each seed run a *smooth‑gradient descent*:
   - Replace the nonsmooth max by a soft‑max (log‑sum‑exp) with a
     temperature β that is increased gradually.
   - The gradient of the smoothed objective is computed analytically:
       ∇J(a) = 4·n·(a * softmax(conv(a,a)))   (linear convolutions via FFT)
   - After each step the iterate is projected onto the simplex.
3. Upsample the current best vector (linear interpolation) and repeat the
   smooth descent at a larger dimension.
4. Test the exact symmetric version of the best vector.
5. Perform a cheap greedy pairwise refinement.
6. Finish with a light simulated‑annealing clean‑up.

All steps respect the global time budget (default 1000 s).  The returned
list is a plain Python list of non‑negative floats suitable for
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    need = len(x) + len(y) - 1
    size = 1 << ((need - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:need]
    return conv


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a with sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Force exact left‑right symmetry and re‑project onto the simplex."""
    sym = (seq + seq[::-1]) * 0.5
    return sym / sym.sum()


# ----------------------------------------------------------------------
def _smooth_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    start_beta: float = 0.2,
    beta_factor: float = 1.5,
    beta_max: float = 30.0,
    step0: float = 0.12,
    eval_interval: int = 300,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a smoothed version of the objective.

    The nonsmooth max is replaced by a log‑sum‑exp with temperature β.
    The gradient is
        ∇J = 4·n·(a * softmax(conv(a,a)))    (linear convolutions via FFT)
    After each step the iterate is projected onto the simplex.
    β is increased geometrically until beta_max.
    """
    n = a.shape[0]
    t = 0
    beta = start_beta
    best_local_seq = best_seq.copy()
    best_local_score = best_score
    while time.time() < deadline and beta <= beta_max:
        inner_iters = 500
        for _ in range(inner_iters):
            if time.time() >= deadline:
                break
            t += 1

            # 1) self‑convolution
            conv = _fft_self_convolve(a)
            max_c = conv.max()

            # 2) soft‑max weights (stable)
            shift = conv - max_c  # ≤ 0
            exp_shift = np.exp(beta * shift)
            sum_exp = exp_shift.sum()
            if sum_exp == 0.0:
                p = np.full_like(exp_shift, 1.0 / exp_shift.size)
            else:
                p = exp_shift / sum_exp  # shape (2n‑1)

            # 3) gradient = 4·n·(a * p)  (linear convolution of a_rev and p)
            a_rev = a[::-1]
            conv_rev = _fft_convolve(p, a_rev)          # length 3n‑2
            grad_part = conv_rev[n - 1 : n - 1 + n]      # middle n entries
            gradient = 4.0 * n * grad_part

            # 4) step + projection
            eta = step0 / math.sqrt(t)
            a = a - eta * gradient
            a = _project_onto_simplex(a, s=1.0)

            # 5) periodic exact evaluation
            if t % eval_interval == 0:
                cur = _objective(a)
                if cur + 1e-14 < best_local_score:
                    best_local_score = cur
                    best_local_seq = a.copy()
                    print(
                        f"[{time.time():.1f}] smooth descent it={t:,} β={beta:.2f} new best {cur:.6f}"
                    )
        beta = min(beta * beta_factor, beta_max)
    return best_local_seq, best_local_score


# ----------------------------------------------------------------------
def _pairwise_weighted_refine(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer guided by the location of the current maximal
    convolution entry.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        conv = _fft_self_convolve(a)
        k_max = int(conv.argmax())

        # impact values L_i = a_{k_max-i} where defined
        L = np.zeros(n, dtype=np.float64)
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        idx = np.arange(low, high + 1)
        L[idx] = a[k_max - idx]

        # source: favour large impact
        if L.sum() > 0:
            probs_i = L / L.sum()
            i = rng.choice(n, p=probs_i)
        else:
            pos = np.where(a > 1e-12)[0]
            if pos.size == 0:
                break
            i = rng.choice(pos)

        # destination: favour small impact
        Lmax = L.max()
        probs_j = (Lmax - L) + 1e-12
        probs_j /= probs_j.sum()
        j = rng.choice(n, p=probs_j)
        if j == i:
            j = (j + 1) % n

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] weighted swap {i}->{j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.005,
    decay: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Very cheap Metropolis‑type annealing for final polishing.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq

            start_beta=0.2,
            beta_factor=1.5,
            beta_max=30.0,
            step0=0.12,
            eval_interval=300,
        )
        if opt_score < best_score:
            best_score, best_seq = opt_score, opt_seq.copy()
            print(f"[{time.time():.1f}] after smooth descent seed {idx} new global best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) up‑sample and refine at larger resolution if time permits
    if time.time() < deadline - 30.0:
        new_len = min(18000, len(best_seq) * 2)
        print(f"[{time.time():.1f}] up‑sampling to {new_len} points")
        up_seq = _upsample(best_seq, new_len)
        sub_dead = time.time() + min(120.0, deadline - time.time())
        up_seq_opt, up_score = _smooth_gradient_descent(
            up_seq,
            sub_dead,
            best_seq,
            best_score,
            start_beta=0.5,
            beta_factor=1.3,
            beta_max=50.0,
            step0=0.10,
            eval_interval=200,
        )
        if up_score < best_score:
            best_score, best_seq = up_score, up_seq_opt.copy()
            print(f"[{time.time():.1f}] up‑sample smooth descent new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) symmetrisation test
    sym_seq = _symmetrize(best_seq)
    sym_score = _objective(sym_seq)
    if sym_score + 1e-14 < best_score:
        best_score, best_seq = sym_score, sym_seq
        print(f"[{time.time():.1f}] symmetrised version better: {sym_score:.6f}")

    # ------------------------------------------------------------------
    # 5) greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] weighted pairwise refinement")
        best_seq, best_score = _pairwise_weighted_refine(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            max_trials=200_000,
        )

    # ------------------------------------------------------------------
    # 6) final cheap simulated‑annealing clean‑up
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing clean‑up")
        best_seq, best_score = _random_local_search(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            init_temp=0.005,
            decay=0.999,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
