"""
search for a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (∑a)²

The routine follows a three‑stage strategy

1️⃣  **Diversified seeding** – analytic families (uniform, edge‑weighted,
    β‑distribution, generalized‑normal, random Dirichlet and sparse
    vectors) are sampled for several lengths.  The 20‑30 best seeds are
    kept.

2️⃣  **Smooth‑max Adam optimisation** – instead of using a raw
    sub‑gradient of the max‑convolution we minimise a smooth surrogate

        Sγ(a)= (1/γ)·log Σ_k exp(γ·(a*a)_k)

    (γ≈1…50).  Its gradient is cheap to evaluate via a correlation.
    An Adam optimiser (additive updates, simplex projection) runs on
    each seed for a short time slice and finally on the global best
    for a longer period.  γ is increased slowly so that the optimisation
    gradually focuses on the true max.

3️⃣  **Progressive up‑sampling & final annealing** – the best vector is
    repeatedly doubled in length (linear interpolation) while time permits,
    each up‑sampled vector receives a medium‑length Adam run, and finally a
    pairwise‑mass‑transfer simulated annealing polishes the result.

All steps respect the 1000 s budget (2 s safety margin).  The returned
list is a plain Python list ready for ``evaluate_sequence`` (the entries
are non‑negative and automatically normalised inside the objective).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – ensures Σa = 1, a_i ≥ 0
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (length 2n‑1)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective for a probability vector (Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families (all return a *probability* vector, i.e. Σ = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w /= w.sum()
    return w


def _distribution_beta(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    w = np.exp((alpha - 1) * np.log(x) + (alpha - 1) * np.log1p(-x))
    w /= w.sum()
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.02) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    vals /= vals.sum()
    return vals


def _mix(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination  λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smooth‑max gradient (soft‑max of the convolution)
# ----------------------------------------------------------------------
def _smooth_max_grad(a: np.ndarray, gamma: float):
    """
    Returns (grad, max_conv) where grad is the gradient of
    Sγ(a) = (1/γ)·log Σ exp(γ·(a*a)_k) and max_conv is max(a*a).
    """
    conv = _fft_self_convolve(a)                     # length 2n‑1
    max_c = conv.max()
    # stable soft‑max weights
    exp_term = np.exp(gamma * (conv - max_c))
    w = exp_term / exp_term.sum()
    # grad_i = 2·Σ_j w_{i+j}·a_j   → correlation of w with a
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad, max_c


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_opt(a0: np.ndarray, deadline: float,
              best_seq: np.ndarray, best_score: float,
              lr0: float = 0.1, gamma0: float = 1.0,
              gamma_inc: float = 0.3,
              eval_every: int = 30) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on Sγ(a).  γ starts at ``gamma0`` and is slowly raised.
    The function projects back onto the probability simplex after each step.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = lr0
    gamma = gamma0
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _smooth_max_grad(a, gamma)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a)

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-15:
                best_score, best_seq = cur, a.copy()
                # reward: a bit larger step‑size and increase γ
                lr = min(lr0, lr * 1.05)
                gamma = min(80.0, gamma + gamma_inc)
                print(f"[{time.time():.1f}] Adam it={t:,} new best {best_score:.6f} (γ={gamma:.2f})")
            else:
                # slowly decay step‑size and decay γ if we are not improving
                lr *= 0.997
                gamma = max(gamma0, gamma * 0.999)

        # occasional re‑projection to avoid drift
        if t % (eval_every * 10) == 0:
            a = _project_onto_simplex(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise mass‑transfer simulated annealing (as a final polish)
# ----------------------------------------------------------------------
def _pairwise_annealing(a0: np.ndarray, deadline: float,
                        best_seq: np.ndarray, best_score: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        src = rng.choice(np.where(a > 1e-14)[0])
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1
        delta = a[src] * rng.random() * 0.25
        if delta < 1e-16:
            continue
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        cur = _objective(cand)

        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.02

        accept = False
        if cur < best_score - 1e-15:
            accept = True
        elif temperature > 0.0:
            if rng.random() < math.exp(-(cur - best_score) / temperature):
                accept = True

        if accept:
            a = cand
            if cur < best_score - 1e-15:
                best_score, best_seq = cur, cand.copy()
                print(f"[{time.time():.1f}] anneal {src}->{dst} Δ={delta:.1e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling while preserving total mass (=1)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    return a_new

        up = _upsample(overall_best_seq, target_n)
        up_sc = _objective(up)
        if up_sc < overall_best_score:
            overall_best_seq, overall_best_score = up, up_sc
            print(f"[{time.time():.1f}] up‑sampled new best {overall_best_score:.6f}")

        # medium‑length Adam on the up‑sampled vector
        sub_dead = time.time() + (deadline - time.time()) * 0.18
        a_opt, sc_opt = _adam_opt(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.07,
            gamma0=2.0,
            gamma_inc=0.2,
            eval_every=30,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
            print(f"[{time.time():.1f}] Adam after up‑sample new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣  Final pairwise‑mass‑transfer annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing phase")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
            print(f"[{time.time():.1f}] after annealing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score = {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # convert to plain Python list for evaluate_sequence
    return overall_best_seq.tolist()
