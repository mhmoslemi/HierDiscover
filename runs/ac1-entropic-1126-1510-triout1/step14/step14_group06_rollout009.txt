    allowing the optimiser to first explore and then converge to the true
    ``max`` objective.

3.  Refine the resulting vector with an *exponentiated‑gradient* (multiplicative
    mirror descent) on the exact objective.  The step‑size decays as 1/√t
    and the current best vector is projected onto the probability simplex.

4.  Perform a short *Frank‑Wolfe* mass‑transfer descent to eliminate any
    remaining “spikes”.

5.  If time permits, *upsample* the best vector (linear interpolation) to
    a larger length (≈×2) and repeat steps 2‑4 on the finer grid – this
    improves the granularity of the sum‑collision pattern.

6.  Finish with a simple *pairwise annealing* that randomly moves a
    fraction of mass between two bins, accepting improving moves and,
    early on, occasional worsening moves according to a linear cooling
    schedule.

All steps respect the total 1000 s budget and always return a plain
``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (convolution & correlation)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Cross‑correlation of a with w (both non‑negative). Returns length‑n array."""
    n = a.shape[0]
    m = w.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()
    rev_w = w[::-1]
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_w, n=size)
    conv = np.fft.irfft(A * B, n=size)
    start = m - 1
    return conv[start:start + n]


# ----------------------------------------------------------------------
# Exact objective (scale invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a (unspecified scaling)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic families (all return a *probability* vector, i.e. sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w /= w.sum()
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w /= w.sum()
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Beta(a,b) on a uniform grid (asymmetric)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0)
    eps = 1e-16
    x = np.clip(x, eps, 1 - eps)
    w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x))
    w /= w.sum()
    return w


def _distribution_random_decreasing(n: int, rng: np.random.Generator,
                                    density: float = 0.5) -> np.ndarray:
    """Sample a decreasing probability vector by sorting exponentials."""
    k = max(2, int(n * density))
    vals = rng.exponential(scale=1.0, size=k)
    vals.sort()
    vals = vals[::-1]           # decreasing
    full = np.zeros(n, dtype=np.float64)
    # spread the k values uniformly over the n positions
    positions = np.linspace(0, n - 1, k, dtype=int)
    full[positions] = vals
    if full.sum() == 0.0:
        full[0] = 1.0
    full /= full.sum()
    return full


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Sub‑gradient of the *active* convolution entry (used by EG / FW)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """Return sub‑gradient of max entry of a*a and its index."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))
    grad = np.zeros_like(a)
    # the contribution of a_i a_j appears at index i+j.
    # For a given k we have pairs (i, j) with i+j = k.
    # The gradient w.r.t. a_i is 2 * sum_{j: i+j=k} a_j = 2 * a_{k-i}
    # (taking care of boundaries).
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_descent(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float,
                         eta0: float = 0.8,
                         eval_every: int = 80) -> tuple[np.ndarray, float]:
    """Multiplicative mirror descent (EG) on the exact objective."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        if (t % eval_every) == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Frank‑Wolfe style refinement (pairwise mass transfer)
# ----------------------------------------------------------------------
def _frank_wolfe_descent(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float,
                         eval_every: int = 100) -> tuple[np.ndarray, float]:
    """Simple FW iteration: move a fraction of mass towards the coordinate
    with the smallest sub‑gradient entry."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        i_min = int(np.argmin(grad))
        gamma = 2.0 / (t + 2.0)
        a = a * (1.0 - gamma)
        a[i_min] += gamma
        if (t % eval_every) == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] FW it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise‑annealing polishing (simulated‑annealing style)
# ----------------------------------------------------------------------
def _pairwise_annealing(a: np.ndarray, deadline: float,
                        best_seq: np.ndarray, best_score: float,
                        rng: np.random.Generator,
                        max_trials: int | None = None) -> tuple[np.ndarray, float]:
    """Randomly relocate a fraction of mass; accept improving moves or,
    with a decreasing temperature, occasional worsening moves."""
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = rng.choice(np.where(a > 1e-12)[0])
        # pick a distinct destination uniformly
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # move a random fraction of src's mass
        delta = a[src] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal {src}->{dst} Δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling preserving the simplex
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Adam optimisation of the temperature‑smoothed surrogate
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float,
                   best_seq: np.ndarray, best_score: float,
                   tau_start: float = 1.0,
                   tau_end: float = 1e-5,
                   tau_decay: float = 0.9995,
                   step_size0: float = 0.1,
                   eval_every: int = 100) -> tuple[np.ndarray, float]:
    """
    Adam on the surrogate
        fτ(a) = 2·n·τ·log‑sum‑exp( (a*a) / τ )
    The temperature τ is slowly reduced, morphing the loss into the true
    ``max`` form.
    """
    n = a0.shape[0]
    # work in log‑space (logits) to keep positivity automatically
    x = np.log(a0 + 1e-15)

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    t = 0
    tau = tau_start
    while time.time() < deadline and tau > tau_end:
        t += 1

        # Softmax to get probabilities
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # Convolution and softmax over its entries (the log‑sum‑exp trick)
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        exp_vals = np.exp((conv - max_conv) / tau)
        w = exp_vals / exp_vals.sum()          # weights summing to 1

        # Gradient of the surrogate w.r.t. a:
        #   ∂/∂a_i 2·n·τ·log Σ_k exp(conv_k/τ)
        # = 2·n· Σ_k w_k ∂conv_k/∂a_i
        # and ∂conv_k/∂a_i = a_{k-i} + a_{k-i} (mirror) = 2·a_{k-i}
        # which is exactly the cross‑correlation of a with w, scaled.
        corr = _fft_corr(a, w)                 # length n
        grad_a = 4.0 * n * corr                # factor 2·n·2 = 4·n

        # Chain rule through the softmax
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = step_size0 / (np.sqrt(v_hat) + eps)
        x = x - lr * m_hat

        # Periodic evaluation of the true objective
        if (t % eval_every) == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} τ={tau:.6g} new best {best_score:.6f}")

        # Cool the temperature
        tau = max(tau * tau_decay, tau_end)

    # Final projection (numeric drift)
    a = np.exp(x - np.max(x))
    a /= a.sum()
    return a, _objective(a)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the total 1000 s budget
    and returns a plain list ready for evaluation.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Seed generation at a base length
    # ------------------------------------------------------------------
    base_n = 4096
    print(f"[{time.time():.1f}] generating seeds (n={base_n})")

    # deterministic analytic families
    families = [
        _distribution_uniform,
        _distribution_arcsine,
        _distribution_hann,
        lambda n: _distribution_generalized_normal(n, q=1.0),
        lambda n: _distribution_generalized_normal(n, q=1.5),
        lambda n: _distribution_generalized_normal(n, q=2.0),
    ]
    for fn in families:
        cand = fn(base_n)
        sc = _objective(cand)
        top_candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = cand.copy()
            print(f"[{time.time():.1f}] new best (analytic) score={sc:.6f}")

    # beta (symmetric) sweeps
    for p in np.logspace(-0.5, 0.5, 7):
        cand = _distribution_beta_sym(base_n, float(p))
        sc = _objective(cand)
        top_candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = cand.copy()
            print(f"[{time.time():.1f}] new best beta_sym p={p:.3g} score={sc:.6f}")

    # beta (asymmetric) sweeps
    for a_val, b_val in [(0.3, 2.0), (0.5, 2.0), (1.0, 3.0), (2.0, 5.0)]:
        cand = _distribution_beta_asym(base_n, a_val, b_val)
        sc = _objective(cand)
        top_candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = cand.copy()
            print(f"[{time.time():.1f}] new best beta_asym a={a_val} b={b_val} score={sc:.6f}")

    # edge‑weighted / power‑law sweeps
    for p in (0.2, 0.4, 0.6, 0.9):
        cand = _distribution_edge_weighted(base_n, p)
        sc = _objective(cand)
        top_candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = cand.copy()
            print(f"[{time.time():.1f}] new best edge p={p:.2f} score={sc:.6f}")

        cand = _distribution_power_law(base_n, p)
        sc = _objective(cand)
        top_candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = cand.copy()
            print(f"[{time.time():.1f}] new best power p={p:.2f} score={sc:.6f}")

    # random decreasing vectors
    for _ in range(5):
        cand = _distribution_random_decreasing(base_n, rng, density=0.10)
        sc = _objective(cand)
        top_candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = cand.copy()
            print(f"[{time.time():.1f}] new best random‑decr score={sc:.6f}")

    # keep only the elite few for refinement
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:8]

    # ------------------------------------------------------------------
    # 2.  Adam‑based smoothing optimisation on the elite seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        per_seed = remaining * 0.45 / max(1, len(top_candidates))
        print(f"[{time.time():.1f}] Adam optimisation on {len(top_candidates)} seeds")
        for idx, (sc, seed_seq) in enumerate(top_candidates):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _adam_optimize(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                tau_start=1.0,
                tau_end=1e-5,
                tau_decay=0.9995,
                step_size0=0.12,
                eval_every=100,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] after Adam seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Exact exponentiated‑gradient descent (fine‑tune)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 8.0 and overall_best_seq is not None:
        sub_deadline = time.time() + remaining * 0.30
        best_seq, best_score = _exp_gradient_descent(
            overall_best_seq.copy(),
            sub_deadline,
            overall_best_seq,
            overall_best_score,
            eta0=0.7,
            eval_every=80,
        )
        overall_best_seq, overall_best_score = best_seq, best_score

    # ------------------------------------------------------------------
    # 4.  Frank‑Wolfe style refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fw_deadline = time.time() + (deadline - time.time()) * 0.15
        overall_best_seq, overall_best_score = _frank_wolfe_descent(
            overall_best_seq.copy(),
            fw_deadline,
            overall_best_seq,
            overall_best_score,
            eval_every=100,
        )
        print(f"[{time.time():.1f}] after FW best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Multi‑scale up‑sampling (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n <= 64000:
            print(f"[{time.time():.1f}] upsampling {cur_n} → {target_n}")
            up_seq = _upsample(overall_best_seq, target_n)
            up_score = _objective(up_seq)
            if up_score < overall_best_score:
                overall_best_seq = up_seq
                overall_best_score = up_score
                print(f"[{time.time():.1f}] upsampled new best {overall_best_score:.6f}")

            # brief Adam on the up‑sampled vector (shorter horizon)
            sub_deadline = time.time() + (deadline - time.time()) * 0.10
            a_opt, sc_opt = _adam_optimize(
                overall_best_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                tau_start=0.5,
                tau_end=1e-5,
                tau_decay=0.999,
                step_size0=0.08,
                eval_every=120,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_seq = a_opt.copy()
                overall_best_score = sc_opt
                print(f"[{time.time():.1f}] Adam after upsample new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Final annealing polish (full remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing polish")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
            max_trials=None,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq = a_opt.copy()
            overall_best_score = sc_opt
            print(f"[{time.time():.1f}] after final anneal new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"[{time.time():.1f}] FINAL best score (2·n·max(conv)/sum²) = "
        f"{overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # final clipping is performed by ``evaluate_sequence`` – we only need
    # to return a plain list of floats.
    return overall_best_seq.tolist()
