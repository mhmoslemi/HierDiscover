    a = a0.copy()
    n = a.shape[0]

        momentum = beta * momentum + (1-beta) * grad
        eta = lr0 / math.sqrt(t)

        a = np.maximum(a, 0.0)

assistantfinal**Solution Overview**

The goal is to minimise  

\[
J(a)=\frac{2\,n\;\max_k\big((a*a)_k\big)}{\big(\sum_i a_i\big)^2},
\qquad a_i\ge 0,
\]

where `*` denotes the linear convolution.  
Because `J` is *scale‑invariant* we may restrict the search to the probability
simplex `∑a_i = 1`.  
The optimisation is highly non‑convex (the `max` operator) but the
“soft‑max’’ (log‑sum‑exp) surrogate

\[
\phi_{\lambda}(a)=\frac1\lambda\log\!\Bigl(\sum_k
\exp\!\bigl(\lambda (a*a)_k\bigr)\Bigr)
\]

is smooth for any `λ>0`.  Its gradient can be written as a convolution and
evaluated with two FFTs.  Starting with a very smooth surrogate (small `λ`)
and annealing `λ` yields a reliable descent method.

After a broad multi‑start phase we perform three refinement stages

1. **Annealed exponentiated‑gradient descent** on the smooth surrogate
   (the code from the reference solution, but with a longer schedule).
2. **Exact sub‑gradient descent** on the true objective
   (gradient = 2 · shifted‑reversed `a` at the maximising convolution index).
3. **Local cleaning** – deterministic pairwise mass swaps and a cheap simulated‑annealing sweep.

If time permits the best candidate is linearly up‑sampled to a larger
length and the whole pipeline is repeated; larger `n` give a slightly
smaller constant factor in the objective.

All steps respect the 1000 s wall‑clock budget, keep the sequence on the
simplex, and finally return a plain Python list.

    """Project v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolutions
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b (real FFT)."""
    n = a.shape[0]
    m = b.shape[0]
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
# Exact objective
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i)   (rate ≥ 0)"""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-rate * i)
    w /= w.sum()
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)   (p > 0)"""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.006) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, where U is the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smoothed surrogate gradient (log‑sum‑exp)
def _smooth_gradient(a: np.ndarray, lam: float):
    """
    Returns (grad, max_conv) where grad = ∇_a φ_λ(a)
    and max_conv = max(conv(a,a)).  a is assumed to sum to 1.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)               # length 2n‑1
    max_c = conv.max()

    # softmax weights (stable)
    z = lam * conv
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    s = exp_z / exp_z.sum()                    # sums to 1

    # gradient = 2 * (a_rev * s) restricted to centre block
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, s)        # length 3n‑2
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]   # slice of length n
    return grad, max_c


# ----------------------------------------------------------------------
# Exact max‑subgradient (gradient of max(conv(a,a)))
def _max_subgradient(a: np.ndarray):
    """
    Returns (grad, max_conv) where grad = ∇ max(conv(a,a)) (subgradient)
    and max_conv = max(conv(a,a)).  a is on the simplex.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)               # length 2n‑1
    max_c = conv.max()
    # indices that are (almost) maximal – a tiny tolerance avoids numerical ties
    eps = max_c * 1e-12
    idxs = np.where(conv >= max_c - eps)[0]

    # build a sparse indicator vector over the full convolution length
    s = np.zeros(2 * n - 1, dtype=np.float64)
    s[idxs] = 1.0 / len(idxs)                  # average over all active indices

    # convolution with the reversed a gives the summed shift‑gradients
    a_rev = a[::-1]
    full = _fft_convolve(a_rev, s)            # length 3n‑2
    grad = 2.0 * full[n - 1 : n - 1 + n]       # central block, length n
    return grad, max_c


# ----------------------------------------------------------------------
# Annealed exponentiated‑gradient descent (smooth surrogate)
def _smooth_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    lam_start_factor: float = 0.15,
    lam_inc: float = 1.0015,
    beta: float = 0.9,
    eval_interval: int = 200,
    patience: int = 2500,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on φ_λ with an annealing schedule.
    Returns the best sequence seen and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    lam = lam_start_factor * n
    rng = np.random.default_rng()

    while time.time() < deadline and t < 600_000:
        t += 1
        grad, max_conv = _smooth_gradient(a, lam)

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * momentum)          # multiplicative update
        a = np.maximum(a, 1e-30)                # avoid exact zeros
        a /= a.sum()

        lam *= lam_inc

        if t % eval_interval == 0:
            cur_score = 2.0 * n * max_conv
            if cur_score + 1e-14 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,}  smooth new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Exact max‑subgradient descent
def _max_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.5,
    beta: float = 0.85,
    eval_interval: int = 150,
    patience: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Euclidean (projected) sub‑gradient descent on the true objective.
    Returns the best sequence and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)

    while time.time() < deadline and t < 400_000:
        t += 1
        grad, max_c = _max_subgradient(a)

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = lr0 / math.sqrt(t)
        a = a - eta * momentum
        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a.sum()

        if t % eval_interval == 0:
            cur_score = 2.0 * n * max_c
            if cur_score + 1e-14 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,}  exact new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            # a small random nudge to escape plateaus
            rng = np.random.default_rng()
            jitter = rng.exponential(scale=0.002, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Deterministic pairwise sweep
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Move a random fraction of mass from i → j if it improves the exact score."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e}  new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Cheap simulated‑annealing clean‑up
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e}  best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling of a simplex vector
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Multi‑start search for the step‑function that minimises ``evaluate_sequence``.
    Returns a plain Python list of non‑negative heights.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0       # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Seed generation – a modest set of lengths and families
    lengths = [1800, 2000, 2500, 3000, 3500, 4000]
    p_grid = np.concatenate(
        [
            np.logspace(-3, -1, 9),   # 0.001 … 0.1
            np.logspace(-1, 0, 9),   # 0.1 … 1
            np.logspace(0, 0.5, 7),  # 1 … 3.16
        ]
    )
    rate_grid = np.logspace(-4, -1, 8)      # for exponential decay
    pow_grid = np.logspace(-1, 1, 9)        # for power‑law
    lam_grid = np.linspace(0.0, 0.9, 9)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating seeds for n={n}")

        # deterministic bases
        for base in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(base)
            top_candidates.append((sc, base.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new global best (base) {sc:.6f}")

        # parameterised families
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            edge2 = _distribution_edge_inversepow(n, p)
            sine = _distribution_sine_weighted(n, p)
            for base in (beta, edge, edge2, sine):
                for lam in lam_grid:
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(f"[{time.time():.1f}] new best (param) n={n} p={p:.3g} lam={lam:.2f} {sc:.6f}")

        # exponential decay family
        for rate in rate_grid:
            if time.time() >= deadline:
                break
            exp_seq = _distribution_exponential_decay(n, rate)
            sc = _objective(exp_seq)
            top_candidates.append((sc, exp_seq.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, exp_seq.copy()
                print(f"[{time.time():.1f}] new best exp‑decay n={n} rate={rate:.3g} {sc:.6f}")

        # power‑law family
        for pp in pow_grid:
            if time.time() >= deadline:
                break
            pow_seq = _distribution_power_law(n, pp)
            sc = _objective(pow_seq)
            top_candidates.append((sc, pow_seq.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, pow_seq.copy()
                print(f"[{time.time():.1f}] new best power‑law n={n} p={pp:.3g} {sc:.6f}")

        # asymmetric beta
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = rng.uniform(0.001, 5.0)
            b = rng.uniform(0.001, 5.0)
            cand = _distribution_beta_asym(n, a, b)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best asym‑beta n={n} a={a:.3g} b={b:.3g} {sc:.6f}")

        # random Dirichlet
        for _ in range(4):
            if time.time() >= deadline:
                break
            cand = _distribution_dirichlet(n, rng)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best Dirichlet n={n} {sc:.6f}")

        # sparse random vectors
        for _ in range(4):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best sparse n={n} {sc:.6f}")

        # keep only the elite seeds
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2. Bring in any previously discovered sequence
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback in the unlikely case we have no sequence yet
    if overall_best_seq is None:
        n = 2500
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. First heavy descent (smooth surrogate) on the elite seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]]
        per_seed = remaining * 0.55 / max(1, len(seeds))
        print(f"[{time.time():.1f}] smooth descent on {len(seeds)} seeds, ~{per_seed:.1f}s each")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            a_opt, sc_opt = _smooth_descent(
                seed_seq.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                lam_start_factor=0.12,
                lam_inc=1.001,
                beta=0.9,
                eval_interval=120,
                patience=1800,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after smooth seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Exact max‑subgradient refinement on the current best
    remaining = deadline - time.time()
    if remaining > 25.0:
        sub_dead = time.time() + remaining * 0.35
        print(f"[{time.time():.1f}] exact max‑subgradient refinement (n={len(overall_best_seq)})")
        overall_best_seq, overall_best_score = _max_descent(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.4,
            beta=0.85,
            eval_interval=100,
            patience=1500,
        )
        print(f"[{time.time():.1f}] after exact descent best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Upsample to a larger length if we still have budget
    remaining = deadline - time.time()
    if remaining > 120.0:
        large_n = min(18000, len(overall_best_seq) * 2)
        print(f"[{time.time():.1f}] up‑sampling to {large_n}")
        up_seq = _upsample(overall_best_seq, large_n)

        # smooth descent on the upsized sequence
        sub_dead = time.time() + remaining * 0.55
        up_opt, up_sc = _smooth_descent(
            up_seq,
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.6,
            lam_start_factor=0.15,
            lam_inc=1.0018,
            beta=0.92,
            eval_interval=120,
            patience=2000,
        )
        if up_sc < overall_best_score:
            overall_best_score, overall_best_seq = up_sc, up_opt.copy()
            print(f"[{time.time():.1f}] up‑sampled smooth descent new best {overall_best_score:.6f}")

        # exact refinement on the large vector
        remaining = deadline - time.time()
        if remaining > 20.0:
            sub_dead = time.time() + remaining * 0.4
            overall_best_seq, overall_best_score = _max_descent(
                overall_best_seq.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.3,
                beta=0.8,
                eval_interval=80,
                patience=1200,
            )
            print(f"[{time.time():.1f}] final exact refinement best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Deterministic pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] deterministic pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=400_000,
        )

    # ------------------------------------------------------------------
    # 8. Cheap simulated‑annealing clean‑up
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
