"""
Propose a step‑function (non‑negative list of heights) whose
objective

    2·n·max(conv(a,a)) / (sum(a)**2)

is as small as possible.

Algorithm
---------

* Choose a moderate length `n` (≈ 3000–5000).  All candidates are kept on the
  probability simplex (sum = 1), because the objective is scale invariant.
* Build a **library of analytic shapes** (arcsine, beta, Gaussian,
  exponential, semicircle, sine, cosine, uniform, triangular, Cauchy) and a
  few random Dirichlet draws.  Every shape is normalised to sum = 1.
* **Random convex combinations** of a small number of library elements are
  generated (weights drawn from a Dirichlet distribution).  Each combination
  is evaluated with the cheap score `2·n·max(a*a)`.  The best mixture found so
  far is kept.
* Starting from the current best mixture, a **smoothed‑Adam**
  optimisation (log‑sum‑exp surrogate of the max‑convolution) is performed
  for many iterations with a gradually shrinking smoothing parameter `ε`.
  This refines the shape while staying on the simplex.
* A short **projected sub‑gradient descent** is applied as a final polish.
* A lightweight **hill‑climbing** phase transfers tiny amounts of mass
  between random indices, accepting only improvements; occasional jitter
  helps escape flat regions.
* Throughout the run the remaining wall‑clock time is checked; the function
  always returns the best feasible sequence before the 1000 s budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection ---------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# FFT based convolutions -----------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
# Cheap score (exact objective for sum=1) -------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap score = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution ----------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.
    Returns (grad, cur_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)                     # length 2n-1
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
# Smoothed gradient (log‑sum‑exp surrogate) ---------------------------
def _smooth_grad(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smoothed max using a log‑sum‑exp surrogate:
        f_eps(a) = eps * log Σ_i exp(conv_i/eps)
    Returns ∇f_eps(a).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)                     # length 2n‑1
    maxc = conv.max()
    w = np.exp((conv - maxc) / eps)                 # unnormalised softmax weights
    w_sum = w.sum()
    w = w / w_sum

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for the even positions (each appears twice)
    diag_indices = np.arange(conv.size)[(np.arange(conv.size) % 2) == 0] // 2
    grad[diag_indices] -= w[2 * diag_indices] * a[diag_indices]

    return grad


# ----------------------------------------------------------------------
# Adam on the smoothed surrogate ---------------------------------------
def _smooth_adam_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 200_000,
                         eps0: float = 1e-3,
                         step0: float = 0.25,
                         beta1: float = 0.9, beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam optimiser for the log‑sum‑exp surrogate of max(conv)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    eps = eps0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad = _smooth_grad(a, eps)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        # tighten the smoothing occasionally
        if t % 40_000 == 0:
            eps = max(eps * 0.5, 1e-7)

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Plain projected sub‑gradient descent ---------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 2_000_000,
                         eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_w = (p - 1) * (np.log(x) + np.log(1.0 - x))
    log_w = np.clip(log_w, -700, 700)
    w = np.exp(log_w)
    if w.sum() == 0.0:
        return np.full_like(w, 1.0 / n)
    return w / w.sum()


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _gaussian_distribution(n: int, sigma_frac: float = 0.30) -> np.ndarray:
    sigma = sigma_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if vals.sum() == 0.0:
        return np.full_like(vals, 1.0 / n)
    return vals / vals.sum()


def _cauchy_distribution(n: int, scale_frac: float = 0.30) -> np.ndarray:
    scale = scale_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = 1.0 / (1.0 + ((i - centre) / scale) ** 2)
    return vals / vals.sum()


def _semicircle_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0            # map to [-1,1]
    vals = np.sqrt(np.clip(1.0 - x * x, 0.0, None))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    """Dirichlet draw with concentration α."""
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return x / x.sum()

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine stays within the supplied time budget and always returns a feasible
    sequence (sum ≥ 0.01).  It combines random convex mixtures of analytic families,
    a smoothed‑Adam refinement, a short sub‑gradient polishing stage and a final
    hill‑climbing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # ---------------------------------------------------------------
    # Determine a working length.  Prefer the length of a historic best
    # if available, otherwise fall back to a moderate value.
    try:
        base_n = int(len(height_sequence_1))      # type: ignore[name-defined]
    except Exception:
        base_n = 3500
    n = max(2000, min(base_n, 8000))

    # ---------------------------------------------------------------
    # Build a small library of analytic shapes (all normalised to sum=1)
    lib = []
    lib.append(_arcsine_distribution(n))
    lib.append(_uniform_distribution(n))
    lib.append(_triangular_distribution(n))
    lib.append(_semicircle_distribution(n))
    lib.append(_sine_distribution(n))
    lib.append(_cosine_distribution(n))

    for p in [0.3, 0.5, 0.8, 1.2, 2.0]:
        lib.append(_beta_symmetric_distribution(n, p))

    for sf in [0.15, 0.25, 0.35]:
        lib.append(_gaussian_distribution(n, sigma_frac=sf))

    for lam in [0.001, 0.003, 0.008]:
        lib.append(_exponential_distribution(n, lam=lam))

    for sf in [0.15, 0.30, 0.45]:
        lib.append(_cauchy_distribution(n, scale_frac=sf))

    for a in [0.05, 0.2, 0.5]:
        lib.append(_random_dirichlet(n, alpha=a))

    # Normalise (just in case)
    lib = [b / b.sum() for b in lib]

    # ---------------------------------------------------------------
    # Keep the best single‑family shape as a starting point
    best_seq = lib[0].copy()
    best_score = _max_convolution_score(best_seq)
    for b in lib[1:]:
        s = _max_convolution_score(b)
        if s < best_score:
            best_score = s
            best_seq = b.copy()
    print(f"[{time.time():.1f}] initial best single‑family score={best_score:.6f}")

    # ---------------------------------------------------------------
    # Random convex combinations of a few library elements
    num_lib = len(lib)
    while time.time() < deadline - 5.0:
        k = np.random.randint(2, min(6, num_lib) + 1)          # how many components
        idx = np.random.choice(num_lib, size=k, replace=False)
        weights = np.random.dirichlet(np.ones(k))
        a = np.zeros(n)
        for w, i in zip(weights, idx):
            a += w * lib[i]

        # occasional jitter to increase diversity
        if np.random.rand() < 0.2:
            jitter = np.random.randn(n) * 5e-4
            a = a + jitter
            a = np.clip(a, 0, None)
            a = _project_onto_simplex(a, s=1.0)

        s = _max_convolution_score(a)
        if s < best_score:
            best_score, best_seq = s, a.copy()
            print(f"[{time.time():.1f}] new best via convex mix score={best_score:.6f}")

    # ---------------------------------------------------------------
    # Smoothed‑Adam refinement on the current best vector
    # (run as long as we have at least ~30 s left)
    while time.time() < deadline - 30.0:
        # let the helper stop by its own deadline check
        a_ref, _ = _smooth_adam_descent(
            best_seq,
            deadline - 5.0,
            max_iters=400_000,
            eps0=5e-3,
            step0=0.25,
            beta1=0.9,
            beta2=0.999,
        )
        sc = _max_convolution_score(a_ref)
        if sc < best_score:
            best_score, best_seq = sc, a_ref.copy()
            print(f"[{time.time():.1f}] smooth‑Adam refined score={best_score:.6f}")

        # Sub‑gradient polishing (quick, helps escape tiny plateaus)
        if time.time() >= deadline - 10.0:
            break
        a_pol, _ = _subgradient_descent(
            best_seq,
            deadline - 5.0,
            max_iters=1_500_000,
            eta0=0.8,
        )
        sc = _max_convolution_score(a_pol)
        if sc < best_score:
            best_score, best_seq = sc, a_pol.copy()
            print(f"[{time.time():.1f}] sub‑gradient polish score={best_score:.6f}")

        # One refinement round is usually enough; exit if time is tight
        if time.time() > deadline - 20.0:
            break

    # ---------------------------------------------------------------
    # Final hill‑climbing: tiny mass transfers guided by the current score
    a = best_seq.copy()
    cur_score = best_score
    iteration = 0
    max_transfer = 0.03

    while time.time() < deadline:
        iteration += 1
        transfer = max_transfer * (0.5 ** (iteration / 8000.0))

        # pick donor and receiver
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} new best {best_score:.6f}")

        # occasional jitter to shake out of flat regions
        if iteration % 6000 == 0 and np.random.rand() < 0.15:
            jitter = np.random.randn(n) * 6e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    # The sequence already sums to 1, satisfying evaluate_sequence's checks.
    return best_seq.tolist()
