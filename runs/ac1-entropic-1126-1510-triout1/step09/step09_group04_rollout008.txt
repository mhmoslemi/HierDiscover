def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

    """Full linear self‑convolution a*a using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.01):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    """Discrete Gaussian centred at (n‑1)/2."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _smooth_max_mirror_descent(a, deadline,
                               steps=8000,
                               alpha_start=1.0,
                               alpha_end=1e5,
                               lr_start=0.5,
                               lr_end=0.001):
    """Exponentiated‑gradient descent on the smooth max of the self‑convolution."""
    n = a.shape[0]
    eps = 1e-15
    for it in range(steps):
        if time.time() >= deadline:
            break
        t = it / steps
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t

        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # smooth‑max weights (large α ⇒ concentrates on the true maximum)
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of Σ_k w_k (a*a)_k  = 2 * (a_rev * w)_[n‑1:n‑1+n]
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        # exponentiated‑gradient step and projection onto the simplex
        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)          # numerical safety
        a /= a.sum()
    return a


def _subgradient_polish(a, deadline, max_iters=5000):
    """Plain sub‑gradient descent with momentum, used for final polishing."""
    n = a.shape[0]
    eps = 1e-12
    velocity = np.zeros_like(a)
    beta = 0.9
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        eps_rel = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - eps_rel)

        w = np.zeros_like(conv)
        w[active] = 1.0          # sub‑gradient of the max (any convex combination works)

        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = 0.15 / np.sqrt(it)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()
    return a


def _hill_climb(a, deadline, max_iters=20000):
    """Very cheap random mass‑transfer hill‑climber."""
    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # index where the current maximum is attained
        k = int(conv.argmax())
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        best_i, best_j = i_low, k - i_low
        max_prod = a[best_i] * a[best_j]
        for i in range(i_low, i_high + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a_snapshot = a.copy()
        a[source] -= delta
        a[target] += delta
        a = np.maximum(a, eps)
        a /= a.sum()
        conv = _fft_self_convolve(a)
        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
        else:
            a = a_snapshot
            conv = _fft_self_convolve(a)

        if it % 5000 == 0:
            print(f"[{time.time():.1f}] hill‑climb iteration {it}, current score {2*n*M:.6f}")
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found within the allotted wall‑clock time.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # tiny safety margin

    eps = 1e-12
    best_seq = None
    best_score = float('inf')
    best_n = None

    # ------------------------------------------------------------------
    # 1. quick scan of analytic families (hand‑crafted shapes)
    lengths_scan = [500, 1000, 1500, 2000, 2500, 3000]
    for n in lengths_scan:
        if time.time() >= deadline:
            break

        candidates = [
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_triangular(n),
            _distribution_exponential(n, lam=0.008),
            _distribution_exponential(n, lam=0.02),
        ]
        for p in (0.3, 0.6, 1.0, 2.0, 4.0):
            candidates.append(_distribution_beta_sym(n, p))
        for pw in (0.5, 1.0, 2.0, 4.0):
            candidates.append(_distribution_sine_power(n, pw))
        for sigma in (0.2, 0.35, 0.5):
            candidates.append(_distribution_gaussian(n, sigma))
        # a few random Dirichlet draws
        for alpha in (0.2, 0.5, 1.0, 2.0):
            candidates.append(np.random.dirichlet(np.full(n, alpha)))

        # evaluate them (cheap)
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] init – n={n} – new best {sc:.6f}")

    # ------------------------------------------------------------------
    # 2. bring forward a previously discovered construction if present
    try:
        prev = globals().get("height_sequence_1", None)
        if prev is not None:
            prev = np.asarray(prev, dtype=np.float64)
            if prev.ndim == 1 and prev.size > 0:
                prev = np.maximum(prev, eps)
                prev /= prev.sum()
                sc = _max_convolution_score(prev)
                if sc < best_score:
                    best_seq = prev.copy()
                    best_n = prev.size
                    best_score = sc
                    print(f"[{time.time():.1f}] previous global best {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. fallback to a uniform vector if nothing else was found
    if best_seq is None:
        best_n = 2000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform – score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. multi‑restart smooth‑max mirror descent on several lengths
    #    (these lengths give us more optimisation degrees of freedom)
    lengths_opt = [2000, 2500, 3000, 3500, 4000, 4500, 5000]
    # Determine roughly how many restarts we can afford using the remaining budget.
    # 1 restart ≈ 3 s for n≈3500, so we aim for ~30 restarts total.
    restarts_per_len = max(1, int((deadline - time.time()) / 30.0))

    for n in lengths_opt:
        if time.time() >= deadline:
            break
        for r in range(restarts_per_len):
            if time.time() >= deadline:
                break

            # --- initialise a random Dirichlet vector --------------------
            alpha = random.choice([0.2, 0.5, 1.0, 2.0])
            a = np.random.dirichlet(np.full(n, alpha))
            a = np.maximum(a, eps)
            a /= a.sum()

            # optional: start from a shuffled copy of the current best
            if best_seq is not None and best_seq.size == n and random.random() < 0.3:
                a = best_seq.copy()
                # apply a tiny random permutation to escape possible symmetries
                perm = np.arange(n)
                np.random.shuffle(perm)
                a = a[perm]
                a = np.maximum(a, eps)
                a /= a.sum()

            # --- smooth‑max mirror descent --------------------------------
            a = _smooth_max_mirror_descent(a, deadline,
                                           steps=8000,
                                           alpha_start=1.0,
                                           alpha_end=1e5,
                                           lr_start=0.5,
                                           lr_end=0.001)

            # --- short sub‑gradient polish --------------------------------
            a = _subgradient_polish(a, deadline, max_iters=2000)

            # --- hill‑climber for a final touch -------------------------
            a = _hill_climb(a, deadline, max_iters=8000)

            sc = _max_convolution_score(a)
            if sc < best_score:
                best_seq = a.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] restart len={n} iter={r} – new best {sc:.6f}")

    # ------------------------------------------------------------------
    # 5. final sanity: ensure non‑negative, normalised, not too tiny
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()

    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.size})")
    return best_seq.tolist()
