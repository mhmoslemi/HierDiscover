"""
Propose a non‑negative step function minimizing
    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

Algorithm
---------
1.  Choose a modest set of sequence lengths (≈800–2400) and generate a few
    analytic seeds (arcsine, symmetric beta with tiny exponent, uniform,
    Gaussian).  If a historic best `height_sequence_1` is present, it is
    interpolated to each length and added as a seed.

2.  For every seed run a *linear‑programming refinement*:
        •  With the current vector f (∑f=1) compute M = max(conv(f,f)).
        •  Solve    max 1ᵀb   s.t.   conv(f,b) ≤ M ,  b ≥ 0 .
        •  Let g = b / Σb (renormalised to the simplex).  Replace f by
           a convex combination f ← (1‑t) f + t g  (t=0.5) and repeat a few
           times.  The LP uses `scipy.optimize.linprog` and a sparse Toeplitz
           matrix for the convolution constraints.

3.  Starting from the best refined seed run heavy first‑order optimisation
    (exponentiated‑gradient, Adam, smooth‑Adam) with many iterations
    (hundreds of thousands).  All methods keep the iterate on the simplex.

4.  Perform a long sub‑gradient descent and a gradient‑guided hill‑climbing
    phase that transfers a tiny amount of mass between coordinates.

5.  Evaluate every candidate with the provided `evaluate_sequence` and keep
    the best one found before the time budget expires.

The whole routine respects the supplied `budget_s` (≈1000 s) and always
returns a feasible list of non‑negative floats.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a via a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap score = 2·n·max(a*a) assuming Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(conv(a,a)) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0                      # weight = 1 on active positions
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (even‑position terms appear twice)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


def _smooth_grad(a: np.ndarray, eps: float) -> np.ndarray:
    """Gradient of the log‑sum‑exp smoothing of max(conv(a,a))."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    maxc = conv.max()
    w = np.exp((conv - maxc) / eps)
    w /= w.sum()
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction
    diag = np.arange(conv.size)[(np.arange(conv.size) % 2) == 0] // 2
    grad[diag] -= w[2 * diag] * a[diag]
    return grad


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 300_000,
                          eta0: float = 0.9) -> tuple[np.ndarray, float]:
    """Mirror descent with KL (exponentiated‑gradient)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 300_000,
                  step0: float = 0.25,
                  beta1: float = 0.9, beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam with simplex projection."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _smooth_adam_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 200_000,
                         eps0: float = 1e-3,
                         step0: float = 0.25,
                         beta1: float = 0.9, beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp smoothed objective."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    eps = eps0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad = _smooth_grad(a, eps)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        if t % 40_000 == 0:
            eps = max(eps * 0.5, 1e-7)

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 2_000_000,
                         eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Plain sub‑gradient descent (diminishing step‑size)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

# Analytic families (all return a vector that already sums to 1)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_w = (p - 1) * (np.log(x) + np.log(1.0 - x))
    log_w = np.clip(log_w, -700, 700)
    w = np.exp(log_w)
    if w.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return w / w.sum()


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _gaussian_distribution(n: int, sigma_frac: float = 0.30) -> np.ndarray:
    sigma = sigma_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if vals.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return vals / vals.sum()

def _build_conv_constraint_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """
    Build a sparse matrix A (size (2n‑1)×n) such that
        A @ b  ==  conv(f, b)
    for any vector b.
    """
    n = f.size
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)

    # Fill column by column
    pos = 0
    for i in range(n):
        rows[pos:pos + n] = np.arange(i, i + n)        # rows i … i+n‑1
        cols[pos:pos + n] = i
        data[pos:pos + n] = f
        pos += n
    return sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _solve_lp_for_f(f: np.ndarray, M: float) -> np.ndarray | None:
    """Solve   max Σb   s.t. conv(f,b) ≤ M ,  b ≥ 0 .   Returns the optimal b."""
    n = f.size
    A = _build_conv_constraint_matrix(f)                     # (2n‑1)×n
    c = -np.ones(n)                                          # maximise Σb ⇔ minimise -Σb
    ub = np.full(2 * n - 1, M, dtype=np.float64)

    # bounds for each variable: b_i ≥ 0
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
def _lp_refinement(a: np.ndarray, deadline: float,
                   max_iter: int = 5, t: float = 0.5) -> np.ndarray:
    """
    Perform a few LP‑based refinements on a simplex vector a.
    Each iteration solves max Σb s.t. conv(a,b) ≤ M.
    The new point is a convex combination a←(1‑t) a + t (b/Σb).
    """
    for _ in range(max_iter):
        if time.time() >= deadline:
            break
        M = _fft_self_convolve(a).max()
        b = _solve_lp_for_f(a, M)
        if b is None:
            break
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-12:          # no improvement possible
            break
        g = b / sum_b                     # renormalise to the simplex
        a = (1.0 - t) * a + t * g
        a = _project_onto_simplex(a, s=1.0)
    return a

def _hill_climb(a: np.ndarray, best_score: float, deadline: float) -> tuple[np.ndarray, float]:
    """Gradient‑guided hill‑climbing that swaps tiny masses between coordinates."""
    n = a.shape[0]
    cur_score = best_score
    best_a = a.copy()
    iteration = 0
    max_transfer = 0.025

    while time.time() < deadline:
        iteration += 1
        transfer = max_transfer * (0.5 ** (iteration / 8000.0))

        # donor / receiver selection
        if iteration % 9 == 0:
            g, _, _ = _subgrad(a)
            i = int(np.argmax(g))
            j = int(np.argmin(g))
            if i == j:
                i = np.random.randint(n)
                j = (i + np.random.randint(1, n)) % n
        else:
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} new best {best_score:.6f}")

        # occasional jitter to escape flat plateaus
        if iteration % 6000 == 0 and np.random.rand() < 0.15:
            jitter = np.random.randn(n) * 6e-5
            a = best_a + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    return best_a, best_score


# ----------------------------------------------------------------------
def _improve_sequence(seed: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Full improvement pipeline:
        LP refinement → exp‑grad → Adam → smooth‑Adam → sub‑grad → hill‑climb
    Returns the best sequence (on the simplex) and its cheap score.
    """
    n = seed.size
    # -------------------------------------------------
    # 1) Linear‑programming refinement (aggressive)
    lp_deadline = deadline - 120.0               # reserve time for the later stages
    a = _lp_refinement(seed, lp_deadline, max_iter=5, t=0.5)
    best_score = _max_convolution_score(a)
    best_seq = a.copy()

    # -------------------------------------------------
    # 2) Heavy first‑order optimisation
    # (use the same global deadline – each routine stops early on its own)
    seq, sc = _exp_gradient_descent(a, deadline, max_iters=500_000, eta0=0.9)
    if sc < best_score - 1e-14:
        best_seq, best_score = seq, sc
        print(f"[{time.time():.1f}] after exp‑grad  score={best_score:.6f}")

    seq, sc = _adam_descent(best_seq, deadline, max_iters=500_000, step0=0.25)
    if sc < best_score - 1e-14:
        best_seq, best_score = seq, sc
        print(f"[{time.time():.1f}] after Adam      score={best_score:.6f}")

    seq, sc = _smooth_adam_descent(best_seq, deadline,
                                   max_iters=300_000,
                                   eps0=1e-3,
                                   step0=0.20)
    if sc < best_score - 1e-14:
        best_seq, best_score = seq, sc
        print(f"[{time.time():.1f}] after smooth‑Adam score={best_score:.6f}")

    # -------------------------------------------------
    # 3) Plain sub‑gradient descent (long run)
    subgrad_deadline = deadline - 30.0
    if time.time() < subgrad_deadline:
        seq, sc = _subgradient_descent(best_seq, subgrad_deadline,
                                       max_iters=2_000_000,
                                       eta0=0.8)
        if sc < best_score - 1e-14:
            best_seq, best_score = seq, sc
            print(f"[{time.time():.1f}] after sub‑grad   score={best_score:.6f}")

    # -------------------------------------------------
    # 4) Fine‑grained hill‑climb
    final_deadline = deadline
    best_seq, best_score = _hill_climb(best_seq, best_score, final_deadline)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    if seq.size == target_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, seq.size)
    x_new = np.linspace(0.0, 1.0, target_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.clip(new_seq, 0.0, None)
    return _project_onto_simplex(new_seq, s=1.0)

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function (non‑negative, sum ≥ 0.01) that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget
    and prints progress information.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Choose a modest range of lengths (≈800–2400)
    lengths = [800, 1200, 1500, 1800, 2200, 2600]
    seeds = []

    for n in lengths:
        # Analytic families
        seeds.append(_arcsine_distribution(n))
        seeds.append(_beta_symmetric_distribution(n, p=0.001))
        seeds.append(_uniform_distribution(n))
        seeds.append(_gaussian_distribution(n, sigma_frac=0.30))

        # Historic best, if available, resized to the current length
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
            if prev.ndim == 1 and prev.size > 0:
                prev = np.maximum(prev, 0.0)
                if prev.sum() == 0.0:
                    prev = _uniform_distribution(prev.size)
                prev = prev / prev.sum()
                prev_resized = _resize_sequence(prev, n)
                seeds.append(prev_resized)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Evaluate cheap scores and keep a few promising seeds
    candidates = [(s, _max_convolution_score(s)) for s in seeds]
    candidates.sort(key=lambda x: x[1])
    top_k = min(12, len(candidates))
    top_candidates = candidates[:top_k]
    print(f"Selected {top_k} seeds for full refinement; best cheap score={top_candidates[0][1]:.6f}")

    best_overall_seq = None
    best_overall_val = float('inf')

    # ------------------------------------------------------------------
    for idx, (seed_seq, cheap) in enumerate(top_candidates):
        if time.time() >= deadline:
            break

        print(f"[{time.time():.1f}] Refining seed {idx+1}/{top_k} (cheap={cheap:.6f}, length={seed_seq.size})")
        seq, cheap2 = _improve_sequence(seed_seq, deadline)
        val = evaluate_sequence(seq.tolist())
        print(f"[{time.time():.1f}]  → final evaluate {val:.6f} (cheap={cheap2:.6f})")
        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq.tolist()
            print(f"[{time.time():.1f}]  ** NEW GLOBAL BEST **")

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded, return a simple arcsine
    if best_overall_seq is None:
        fallback = _arcsine_distribution(1500)
        best_overall_seq = fallback.tolist()
        best_overall_val = evaluate_sequence(best_overall_seq)

    print(f"Final best evaluate: {best_overall_val:.6f} (length={len(best_overall_seq)})")
    return best_overall_seq
