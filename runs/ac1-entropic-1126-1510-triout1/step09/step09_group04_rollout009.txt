def _build_convolution_rows(f_rev, ks):
    # Return A matrix (len(ks), n)
    n = len(f_rev)
    A = np.zeros((len(ks), n), dtype=np.float64)
    for idx, k in enumerate(ks):
        # Row for convolution at lag k: sum_{j} f_rev[j] * g_{k-j}
        # Equivalent to: for j where 0<=k-j<n => j in [k-n+1, k]
        start_j = max(0, k - (n - 1))
        end_j = min(k, n - 1)
        # For j from start_j to end_j inclusive
        # The coefficient for g_{k-j} is f_rev[j]
        # Let i = k - j (index in g)
        # So g[i] coefficient f_rev[j] = f_rev[k - i] when i in [k - end_j, k - start_j]

        # So we can fill A[idx, i_start:i_end+1] = f_rev[k - np.arange(i_start, i_end+1)]

    """

    M = conv_f.max()
    # Determine ks with highest conv values
    # Use argsort to get top indices descending
    # Ensure we include distinct.
    top_indices = np.argpartition(-conv_f, topk)[:topk]  # unsorted topk
    ks = np.sort(top_indices)  # sort for maybe readability
    # Build A matrix rows for these ks
    f_rev = f[::-1]
    # build rows
    rows = np.zeros((len(ks), n), dtype=np.float64)
    for idx, k in enumerate(ks):
        # fill row
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # for i in range(i_start, i_end+1)
        # rows[idx, i_range] = f_rev[k - i_range]
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        rows[idx, i_range] = f_rev[k - i_range]
    # Setup LP
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(len(ks), M, dtype=np.float64)  # all constraints have same ub M
    # Solve
    try:
        res = linprog(c, A_ub=rows, b_ub=b_ub, bounds=(0, None), method='highs')
    except Exception as e:
        # fallback: return original
        # print(f"LP active set exception: {e}")
        return f, M
    if not res.success:
        return f, M
    g = res.x
    g_sum = g.sum()
    if g_sum < 1e-12:
        return f, M
    g_norm = g / g_sum
    # Try mixing with various t

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of vector v onto the probability simplex."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _fft_self_convolve(a):
    """Full linear self‑convolution a*a using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_convolve(a, b):
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a):
    """2·n·max(a*a) for a vector that already sums to 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)

def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals

def _distribution_exponential(n, lam=0.01):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals

def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    total = w.sum()
    if total == 0 or np.isnan(total):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= total
    return w

def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _distribution_gaussian(n, sigma):
    """Discrete Gaussian centred at (n‑1)/2."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _run_mirror_descent(a, steps, alpha_start, alpha_end, lr_start, lr_end):
    """Exponentiated‑gradient (smooth‑max) descent on the simplex."""
    n = a.shape[0]
    eps = 1e-12
    for it in range(steps):
        t = it / steps
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # soft‑max weights over the convolution entries
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient of the smoothed objective
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]
        # mirror‑step in the log‑domain (exponentiated gradient)
        a = a * np.exp(-lr * grad)
        # projection back onto the simplex
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()
    return a

def _active_set_lp_refine(f, mix, topk, deadline):
    """Linear‑programming refinement using only the top‑k active convolution constraints."""
    n = f.shape[0]
    eps = 1e-12
    conv_f = _fft_self_convolve(f)
    M = conv_f.max()

    # indices of the k largest convolution entries (including the maximum)
    if topk >= conv_f.size:
        ks = np.arange(conv_f.size, dtype=np.int64)
    else:
        # argpartition gives unordered top‑k
        ks = np.argpartition(-conv_f, topk)[:topk]
    ks = np.sort(ks)  # optional

    # Build a tiny convolution matrix A where row k corresponds to (f_rev * g)[k]
    f_rev = f[::-1]
    A = np.zeros((len(ks), n), dtype=np.float64)
    for row_idx, k in enumerate(ks):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        A[row_idx, i_range] = f_rev[k - i_range]

    # Linear program: maximise Σ g_i  <=> minimise –Σ g_i
    c = -np.ones(n, dtype=np.float64)   # objective
    b_ub = np.full(len(ks), M, dtype=np.float64)

    # Solve with HiGHS – it works well for moderate‑size problems
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
    except Exception as e:
        # If the solver crashes we simply return the original vector.
        # print(f"LP active‑set exception: {e}")
        return f, M

    if not res.success:
        return f, M

    g = res.x
    gsum = g.sum()
    if gsum < 1e-12:
        return f, M

    g_norm = g / gsum

    # Try a few mixing ratios (starting from the supplied one)
    for t in (mix, mix * 0.5, mix * 0.25, mix * 0.1, mix * 0.05):
        new_f = (1 - t) * f + t * g_norm
        new_f = np.maximum(new_f, eps)
        new_f /= new_f.sum()
        new_M = _fft_self_convolve(new_f).max()
        if new_M < M - 1e-12:
            return new_f, new_M

    return f, M

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:

    The procedure consists of four stages:
      1.   Exhaustive scan of several analytic families for a handful of lengths.
      2.   A long smooth‑max mirror‑descent (exponentiated‑gradient) on the simplex.
      3.   An active‑set linear‑programming refinement that only enforces the
           most‑violated convolution constraints.
      4.   A cheap random hill‑climber / simulated‑annealing finish.
    The best sequence found before the time budget expires is returned.
    """

    random.seed(seed)

    # Keep a small safety margin for the environment that calls us.
    deadline = start_time + budget_s - 2.0

    best_score = float('inf')

        candidates.append(_distribution_uniform(n))
        candidates.append(_distribution_arcsine(n))
        candidates.append(_distribution_triangular(n))
        candidates.append(_distribution_exponential(n, lam=0.006))
        candidates.append(_distribution_exponential(n, lam=0.02))

"""
Search for a non‑negative step‑function that minimises ``evaluate_sequence``.

The algorithm proceeds in several stages:

  1. Quick scan of a handful of analytic families (uniform, arcsine,
     triangular, exponential, Beta, sine‑power, Gaussian) for several
     lengths and a few random Dirichlet draws. The best of these is kept.
  2. A long smooth‑max mirror‑descent (exponentiated‑gradient) on the
     simplex.  The objective ``max(a*a)`` is approximated by a soft‑max
     (log‑sum‑exp) whose gradient is computed via FFT‑based convolutions.
  3. An active‑set linear‑programming refinement: only the K largest
     convolution constraints are kept, producing a tiny LP that nudges
     mass away from the most‑violating lags.
  4. A series of short random restarts (Dirichlet initialisation,
     brief mirror‑descent + LP polish) to explore different basins.
  5. Final cheap hill‑climbing / simulated‑annealing that repeatedly
     moves a small amount of mass between positions while respecting the
     time budget.

All intermediate vectors are projected onto the probability simplex
(so the sum is 1); the objective is invariant under scaling, therefore
the internal score ``2·n·max(a*a)`` matches the value returned by
``evaluate_sequence`` for a normalised vector.
"""

import time
import random
import numpy as np
from scipy.linalg import toeplitz  # not used for the active‑set LP, kept for safety
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a):
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a):
    """Return 2·n·max(a*a) assuming a already sums to 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families (all return a vector whose entries sum to 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.01):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    total = w.sum()
    if total == 0 or np.isnan(total):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= total
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    """Discrete Gaussian centred at (n‑1)/2."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _run_mirror_descent(a, steps, alpha_start, alpha_end, lr_start, lr_end):
    """
    Exponentiated‑gradient (smooth‑max) descent.
    ``alpha`` controls the smoothness of the max; ``lr`` is the learning rate.
    """
    n = a.shape[0]
    eps = 1e-12
    for it in range(steps):
        t = it / steps
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]
        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()
    return a


def _active_set_lp_refine(f, mix, topk, deadline):
    """
    Solve a tiny LP that only enforces the ``topk`` most‑violated convolution
    constraints.  Returns a potentially improved (f, M) where M = max(f*f).
    """
    n = f.shape[0]
    eps = 1e-12
    conv_f = _fft_self_convolve(f)
    M = conv_f.max()

    # indices of the largest convolution entries
    if topk >= conv_f.size:
        ks = np.arange(conv_f.size, dtype=np.int64)
    else:
        ks = np.argpartition(-conv_f, topk)[:topk]
    ks = np.sort(ks)

    # Build the reduced convolution matrix A (shape |ks| × n)
    f_rev = f[::-1]
    A = np.zeros((len(ks), n), dtype=np.float64)
    for row_idx, k in enumerate(ks):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        A[row_idx, i_range] = f_rev[k - i_range]

    # LP: maximise Σ g_i   <=> minimise –Σ g_i
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(len(ks), M, dtype=np.float64)

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
    except Exception:
        return f, M

    if not res.success:
        return f, M

    g = res.x
    gsum = g.sum()
    if gsum < 1e-12:
        return f, M
    g_norm = g / gsum

    # Try a few mixing ratios
    for t in (mix, mix * 0.5, mix * 0.25, mix * 0.1, mix * 0.05):
        new_f = (1 - t) * f + t * g_norm
        new_f = np.maximum(new_f, eps)
        new_f /= new_f.sum()
        new_M = _fft_self_convolve(new_f).max()
        if new_M < M - 1e-12:
            return new_f, new_M

    return f, M


def _hill_climb(a, deadline):
    """
    Cheap random mass‑transfer hill‑climber.
    Repeatedly moves a small chunk of probability mass and keeps the move
    if it reduces the current maximum convolution value.
    """
    n = a.shape[0]
    eps = 1e-12
    a = a.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    while time.time() < deadline:
        it += 1
        src = random.randint(0, n - 1)
        if a[src] < eps:
            continue
        tgt = random.randint(0, n - 1)
        if tgt == src:
            continue
        max_delta = min(a[src] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta
        a_new = np.maximum(a_new, eps)
        a_new /= a_new.sum()
        new_M = _fft_self_convolve(a_new).max()
        if new_M < M - 1e-12:
            a, M = a_new, new_M
        # periodic jitter to avoid plateaus
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
    return a
