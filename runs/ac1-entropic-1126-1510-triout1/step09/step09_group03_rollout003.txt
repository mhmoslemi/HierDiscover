"""
Propose a step‑function that (hopefully) improves the current best bound.

The algorithm works in three stages:

1️⃣  **Broad sampling** – generate many analytic families (beta, arcsine,
    exponential, power‑law, sinusoid, raised‑cosine, uniform) and random
    Dirichlet draws for several lengths around a base length (≈4000).
    Also try simple mixtures of two families.  All candidates are
    ℓ₁‑normalised (∑a = 1) and evaluated with the cheap surrogate
    ``2·n·max(conv(a,a))`` which coincides with ``evaluate_sequence`` for
    ∑a = 1.

2️⃣  **Intensive refinement** – keep the K best seeds and run three
    optimisation phases on each of them:
        • Adam on the true sub‑gradient of ``max(conv(a,a))``.
        • Adam on a smooth Lᵖ‑approximation of the max (p = 6, 10).
        • A final Frank‑Wolfe style mass‑transfer step.
    The optimisation runs until the overall time budget (≈1000 s) is
    exhausted; we allocate many iterations (the inner loops break
    automatically when the deadline is reached).

3️⃣  **Light annealing** – a simple simulated‑annealing / jitter phase
    that performs random pairwise mass transfers; it helps to escape
    tiny plateaux.

All vectors are kept symmetric (a = a[::-1]) after each projection,
which reduces the dimensionality and usually yields a flatter
convolution.  The best sequence (as a plain list of floats) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce a = a[::-1] (mirror symmetry) and renormalise."""
    a = (a + a[::-1]) * 0.5
    a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """
    Surrogate score 2·n·max(conv(a,a)) – exact when sum(a)=1.
    Faster than the full ``evaluate_sequence``.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) on the simplex (∑a=1).
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices (self‑product terms)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


def _smooth_obj_and_grad(a: np.ndarray, p: float = 6.0):
    """
    Smooth Lp‑approximation of max(conv(a,a)) and its gradient.
    Returns (value, grad) where ``value`` = (∑ conv^p)^{1/p}.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)                # length 2n‑1
    b_pow = b ** p
    S = b_pow.sum()
    if S == 0.0:
        return 0.0, np.zeros_like(a)
    val = S ** (1.0 / p)
    factor = S ** (1.0 / p - 1.0)            # = S^{(1‑p)/p}
    w = factor * (b ** (p - 1))              # length 2n‑1
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return val, grad


# ----------------------------------------------------------------------
def _adam_subgrad(a0: np.ndarray, deadline: float,
                  step0: float = 0.2,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 2_000_000):
    """Adam on the exact sub‑gradient of max‑convolution."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _symmetrize(a)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _adam_smooth(a0: np.ndarray, deadline: float,
                 p: float = 6.0,
                 step0: float = 0.2,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 max_iters: int = 2_000_000):
    """Adam on the smoothed Lp‑approximation (p ≥ 2)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        _, grad = _smooth_obj_and_grad(a, p)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _symmetrize(a)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with concentration ``alpha``."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Analytic families (all return vectors that already sum to 1)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Beta(p,p) density on a regular grid, normalized."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, a: float, b: float) -> np.ndarray:
    """Beta(a,b) density on a regular grid, normalized."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} (monotonically decreasing)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _sinusoidal_distribution(n: int, power: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n) ** power
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _raised_cosine_distribution(n: int, power: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = (1.0 - np.cos(2.0 * np.pi * (i + 0.5) / n)) ** power
    vals /= vals.sum()
    return vals


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
def _mix_two(a: np.ndarray, b: np.ndarray, w: float) -> np.ndarray:
    """Mixture w·a + (1‑w)·b and re‑project onto the simplex."""
    mixed = w * a + (1.0 - w) * b
    return _project_onto_simplex(mixed, s=1.0)

            candidates.append((a, _max_convolution_score(a)))

        # ------------------------------------------------------------------
        # Simple mixtures of two families (arcsine with power‑law or beta)
        base_a = _arcsine_distribution(n)
        for other_gen in [_power_law_distribution,
                          lambda m: _beta_symmetric_distribution(m, 0.2),
                          lambda m: _beta_symmetric_distribution(m, 2.0),
                          _uniform_distribution]:
            other = other_gen(n) if callable(other_gen) else other_gen
            other = _symmetrize(other)
            for w in (0.2, 0.5, 0.8):
                mix = _mix_two(base_a, other, w)
                candidates.append((mix, _max_convolution_score(mix)))

    # ------------------------------------------------------------------
    # 2️⃣ Add the previous best if available
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.ndim == 1 and arr.size > 0:
                arr = _project_onto_simplex(arr / arr.sum(), s=1.0)
                arr = _symmetrize(arr)
                candidates.append((arr, _max_convolution_score(arr)))
                print(f"[{time.time():.1f}] added previous best, score={candidates[-1][1]:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3️⃣ Keep the top seeds (K ≈ 8)
    candidates.sort(key=lambda x: x[1])
    top_k = min(8, len(candidates))
    top_seeds = candidates[:top_k]
    print(f"Generated {len(candidates)} candidates, keeping top {top_k}")

    best_seq, best_score = top_seeds[0][0].copy(), top_seeds[0][1]
    print(f"Initial best surrogate score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Intensive refinement of each seed
    for idx, (seed_seq, _) in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        n = seed_seq.shape[0]
        print(f"--- Refinement of seed #{idx+1}/{top_k}, length={n} ---")

        # a) Adam on true sub‑gradient
        seq_a, sc_a = _adam_subgrad(seed_seq, deadline, step0=0.2)
        if sc_a < best_score - 1e-14:
            best_seq, best_score = seq_a.copy(), sc_a
            print(f"[{time.time():.1f}] Adam sub‑grad improved to {best_score:.6f}")

        # b) Adam on smooth p=6
        seq_s6, sc_s6 = _adam_smooth(seed_seq, deadline, p=6.0, step0=0.2)
        if sc_s6 < best_score - 1e-14:
            best_seq, best_score = seq_s6.copy(), sc_s6
            print(f"[{time.time():.1f}] Adam smooth p=6 improved to {best_score:.6f}")

        # c) Adam on smoother p=10
        seq_s10, sc_s10 = _adam_smooth(seed_seq, deadline, p=10.0, step0=0.2)
        if sc_s10 < best_score - 1e-14:
            best_seq, best_score = seq_s10.copy(), sc_s10
            print(f"[{time.time():.1f}] Adam smooth p=10 improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ Frank‑Wolfe mass‑transfer polishing (more iterations than before)
    a = best_seq.copy()
    cur_score = best_score
    fw_iter = 0
    while time.time() < deadline and fw_iter < 120_000:
        fw_iter += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))   # mass to move away from
        i_dst = int(np.argmin(grad))   # mass to move towards
        if i_src == i_dst:
            break
        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-14:
            break
        # try full transfer
        a_full = a.copy()
        a_full[i_src] -= max_transfer
        a_full[i_dst] += max_transfer
        a_full = _symmetrize(a_full)
        score_full = _max_convolution_score(a_full)

        improved = False
        if score_full < cur_score - 1e-13:
            a, cur_score = a_full, score_full
            improved = True
        else:
            # half transfer
            half = max_transfer * 0.5
            a_half = a.copy()
            a_half[i_src] -= half
            a_half[i_dst] += half
            a_half = _symmetrize(a_half)
            score_half = _max_convolution_score(a_half)
            if score_half < cur_score - 1e-13:
                a, cur_score = a_half, score_half
                improved = True

        if improved and cur_score < best_score - 1e-14:
            best_seq, best_score = a.copy(), cur_score
            print(f"[{time.time():.1f}] FW iter={fw_iter} new best {best_score:.6f}")

        # occasional jitter to escape stagnation
        if fw_iter % 6000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(a.size) * 3e-5
            a = _project_onto_simplex(a + jitter, s=1.0)
            a = _symmetrize(a)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    # 6️⃣ Light simulated‑annealing / jitter phase
    a = best_seq.copy()
    cur_score = best_score
    temp_start = 0.018
    temp_end = 0.001
    anneal_iter = 0
    while time.time() < deadline and anneal_iter < 250_000:
        anneal_iter += 1
        # linear temperature schedule based on remaining time
        remaining = deadline - time.time()
        total = budget_s - 0.5
        t_frac = max(0.0, remaining / total)
        temp = temp_end + (temp_start - temp_end) * t_frac

        i = np.random.randint(a.size)
        j = np.random.randint(a.size - 1)
        if j >= i:
            j += 1
        max_delta = min(a[i], 1.0 - a[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * temp

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _symmetrize(a_new)
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a, cur_score = a_new, new_score
            if cur_score < best_score - 1e-14:
                best_seq, best_score = a.copy(), cur_score
                print(f"[{time.time():.1f}] Anneal iter={anneal_iter} new best {best_score:.6f}")
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_score - cur_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                a, cur_score = a_new, new_score

    # ------------------------------------------------------------------
    print(f"Final best surrogate score: {best_score:.6f} (length={len(best_seq)})")
    # Return a plain Python list as required by the evaluation harness.
    return best_seq.tolist()
