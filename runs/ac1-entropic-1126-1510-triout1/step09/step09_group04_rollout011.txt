"""
Propose a step‑function that (approximately) minimises
``evaluate_sequence``.

Algorithm
---------
1.  Scan a wide collection of analytic families on several lengths.
    * Uniform.
    * Symmetric Beta(p,p) with p ranging from 0.01 up to 4.
    * Asymmetric Beta(p,q) with p and q drawn from a small set.
    * Simple mixtures  (1‑λ)·uniform + λ·Beta(p,q).
    * A few Dirichlet draws for diversity.

    For each candidate the objective
          S(a) = 2·n·max(a*a)     (a is normalised to sum = 1)
    is evaluated with a real‑FFT self‑convolution – this is exactly the
    value returned by ``evaluate_sequence`` up to the invariant scaling.

    The best vector found in this cheap phase is kept as a seed.

2.  **Exponentiated‑gradient (smooth‑max) refinement**.
    Starting from the seed we run several thousand iterations of
    mirror‑descent on the simplex.  The smooth‑max weight
          w ∝ exp(α·(conv−max_conv))
    concentrates the gradient on the currently active convolution
    entries; the update is projected back onto the simplex.

3.  **Simulated‑annealing hill‑climber**.
    With the refined vector we perform many random mass‑transfer moves:
        move a small amount δ from a source index to a random target.
    The move is accepted if it improves the score or with a Metropolis
    probability exp(−Δ/Τ) where the temperature T is linearly cooled.
    This stage is very cheap (one FFT per move) and can explore
    non‑gradient directions that the smooth‑max step may miss.

4.  The overall wall‑clock budget (≈ 1000 s) is honoured; intermediate
    results are printed and the best sequence ever encountered is
    returned as a plain list of Python ``float`` objects.

All numbers are kept non‑negative; the final sequence is normalised to
sum = 1 which makes the objective scale‑invariant and satisfies the
security checks in ``evaluate_sequence``.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Return 2·n·max(a*a) for a vector already normalised to sum = 1.
    This equals the value that ``evaluate_sequence`` would return.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic families (all return a vector with sum = 1)

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """Symmetric or asymmetric Beta(p,q) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑density, safe for very small p or q
    logw = (p - 1.0) * np.log(x) + (q - 1.0) * np.log(1.0 - x)
    w = np.exp(logw)
    if np.isnan(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_mixture(n: int, lam: float, p: float, q: float) -> np.ndarray:
    """(1‑lam)·uniform + lam·Beta(p,q).  lam∈[0,1]."""
    uni = _distribution_uniform(n)
    beta = _distribution_beta(n, p, q)
    mix = (1.0 - lam) * uni + lam * beta
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found within the allotted wall‑clock time.
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # tiny safety margin

    best_seq = None
    best_score = float('inf')
    best_n = None

    # ------------------------------------------------------------------
    # 1.  Quick scan of many analytic families on a variety of lengths.
    lengths = [800, 1200, 1600, 2000, 2600, 3200, 4000, 5000, 6400, 8000]
    beta_params = [0.01, 0.02, 0.05, 0.10, 0.20, 0.30, 0.50, 1.0, 2.0, 4.0]

    for n in lengths:
        if time.time() >= deadline:
            break

        # uniform baseline
        candidates = [_distribution_uniform(n)]

        # symmetric Beta(p,p)
        for p in beta_params:
            candidates.append(_distribution_beta(n, p, p))

        # a few asymmetric Betas
        for p, q in [(0.01, 0.5), (0.01, 2.0), (0.05, 2.0), (0.1, 4.0), (0.2, 5.0)]:
            candidates.append(_distribution_beta(n, p, q))

        # mixtures of uniform and symmetric Beta
        for lam in (0.2, 0.4, 0.6, 0.8):
            for p in (0.01, 0.05, 0.1, 0.2):
                candidates.append(_distribution_mixture(n, lam, p, p))

        # a handful of random Dirichlet vectors (different concentration)
        for alpha in (0.1, 0.3, 0.6, 1.0, 2.0):
            candidates.append(np.random.dirichlet(np.full(n, alpha)))

        # evaluate
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, 1e-14)      # avoid exact zeros
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] init n={n}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  If a historic solution exists, use it as an additional seed.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 1e-14)

            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_seq = prev.copy()
                best_n = prev.size
                best_score = sc
                print(f"[{time.time():.1f}] loaded previous best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback to uniform if everything failed.
    if best_seq is None:
        best_n = 2000
        best_seq = _distribution_uniform(best_n)
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Helper: projection onto the simplex (Euclidean).  Used by the mirror‑
    # descent phase.
    def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
        """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
        if s < 0:
            raise ValueError("Simplex radius must be non‑negative")
        v = np.asarray(v, dtype=np.float64)
        if np.all(v >= 0) and np.isclose(v.sum(), s):
            return v.copy()
        n = v.shape[0]
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - s
        rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w

    # ------------------------------------------------------------------
    # 4.  Smooth‑max mirror‑descent (exponentiated gradient)
    a = best_seq.copy()
    n = a.shape[0]

    # schedule parameters – a few thousand iterations are cheap enough
    total_iters = 8000
    for it in range(total_iters):
        if time.time() >= deadline:
            break
        t = it / total_iters
        # slowly increase the sharpness α and slowly decrease the step‑size
        alpha = 1.0 + 30.0 * t          # from 1 → 31
        lr = 0.8 * (0.02 ** t)          # from 0.8 → 0.02

        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # smooth‑max weights
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth‑max (see analysis notes)
        grad = 2.0 * np.fft.ifft(np.fft.rfft(a[::-1]) * np.fft.rfft(w)).real
        grad = grad[:n]                     # keep the relevant part

        # exponentiated‑gradient update
        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, 1e-14)
        a /= a.sum()

        if it % 500 == 0 or it == total_iters - 1:
            sc = _max_convolution_score(a)
            if sc < best_score:
                best_seq = a.copy()
                best_score = sc
                print(f"[{time.time():.1f}] mirror it={it}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 5.  Simulated‑annealing hill‑climber
    a = best_seq.copy()
    n = a.shape[0]
    cur_score = _max_convolution_score(a)
    best_seq = a.copy()
    best_score = cur_score

    # temperature schedule: start relatively hot, cool to near‑zero
    temp_start = cur_score * 0.15
    temp_end = 1e-6
    move_cnt = 0
    while time.time() < deadline:
        move_cnt += 1
        # linear cooling based on remaining time
        remaining = deadline - time.time()
        frac = remaining / (budget_s - 2.0)          # roughly 0..1
        T = temp_end + (temp_start - temp_end) * frac

        # propose a random mass transfer
        src = random.randint(0, n - 1)
        dst = random.randint(0, n - 1)
        if src == dst:
            continue
        # never move more than 10 % of the source mass
        delta = a[src] * random.random() * 0.10
        if delta < 1e-14:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        a_new = np.maximum(a_new, 1e-14)
        a_new /= a_new.sum()

        new_score = _max_convolution_score(a_new)

        # accept if better, otherwise Metropolis criterion
        if new_score < cur_score or random.random() < math.exp(-(new_score - cur_score) / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_seq = a_new.copy()
                best_score = new_score
                print(f"[{time.time():.1f}] anneal move={move_cnt}  score={new_score:.6f}")

        # occasional full‑re‑normalisation to avoid drift
        if move_cnt % 20000 == 0:
            a = np.maximum(a, 1e-14)
            a /= a.sum()

    # ------------------------------------------------------------------
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
