"""
A multi‑restart, multi‑technique optimizer for the step‑function problem.

The algorithm proceeds in several stages:

1. **Analytic scan** – a handful of classical families (uniform,
   arcsine, triangular, exponential, beta, sine‑power, Gaussian) and a few
   random Dirichlet draws are evaluated for a few lengths.
2. **Interpolation** – the current best vector is linearly interpolated to
   a few larger lengths (up to 5000) and re‑evaluated.
3. **Iterative refinement** – each good candidate is refined by a
   sequence of updates:
   * exponential‑gradient (smooth max) with a decaying learning rate,
   * projected‑gradient descent on the (sub‑)gradient of the max‑convolution,
   * Adam optimisation on the log‑its (so the simplex constraint is handled
     naturally),
   * a cheap hill‑climber that moves a small chunk of mass from the
     largest‑gradient index to the smallest‑gradient index.
4. **Multiple restarts** – while time permits we draw fresh random Dirichlet
   vectors (different lengths) and run the same refinement pipeline,
   keeping the globally best sequence.

The whole search respects a 1000 s wall‑clock budget; intermediate
results are printed and the best sequence found is returned as a plain
list of floats.
"""

import time
import math
import random
import numpy as np
from scipy.linalg import toeplitz  # only needed for legacy LP code (unused)


# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.01):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    """Discrete Gaussian centred at (n‑1)/2."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _smooth_weights(conv, alpha):
    """Return a probability vector approximating the smooth max."""
    maxc = conv.max()
    w = np.exp(alpha * (conv - maxc))
    s = w.sum()
    if s == 0:
        return np.full_like(w, 1.0 / w.size)
    return w / s


# ----------------------------------------------------------------------
def _subgradient_weights(conv, eps_rel=1e-12):
    """Weights for a (sub)gradient of the max‑convolution."""
    m = conv.max()
    w = np.zeros_like(conv)
    w[conv >= m - eps_rel * (abs(m) + 1.0)] = 1.0
    cnt = w.sum()
    if cnt == 0:
        return w
    return w / cnt


# ----------------------------------------------------------------------
def _exponential_gradient_step(a, lr, alpha):
    """One EG step (a ≥ 0, Σa = 1)."""
    conv = _fft_self_convolve(a)
    w = _smooth_weights(conv, alpha)
    grad = 2.0 * _fft_convolve(a[::-1], w)[len(a) - 1 : len(a) - 1 + len(a)]
    a = a * np.exp(-lr * grad)
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def _projected_gradient_step(a, eta, active_w):
    """PGD step on the sub‑gradient defined by active_w."""
    grad = 2.0 * _fft_convolve(a[::-1], active_w)[len(a) - 1 : len(a) - 1 + len(a)]
    a = a - eta * grad
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def _adam_step(logits, m, v, t, step_size):
    """Adam update on logits (unconstrained)."""
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = beta1 * m + (1.0 - beta1) * step_size
    v = beta2 * v + (1.0 - beta2) * (step_size ** 2)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    # small learning‑rate for logits
    lr = 0.02 / (np.sqrt(v_hat) + eps)
    logits = logits - lr * m_hat
    # avoid overflow/underflow
    logits = np.clip(logits, -30, 30)
    return logits, m, v


# ----------------------------------------------------------------------
def _hillclimber_move(a, conv, eps=1e-12):
    """Move a tiny chunk from a large‑gradient index to a small‑gradient one."""
    n = a.shape[0]
    # gradient of the active max entry
    w = _subgradient_weights(conv, eps_rel=eps)
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    src = int(np.argmax(grad))          # biggest contributor to the max
    dst = int(np.argmin(grad))          # smallest contributor
    if src == dst:
        dst = random.randint(0, n - 1)
    max_delta = min(a[src] * 0.5, 0.02)
    delta = max_delta * random.random()
    if delta < eps:
        return a, conv, False
    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    new_conv = _fft_self_convolve(a_new)
    improved = new_conv.max() < conv.max() - 1e-13
    if improved:
        return a_new, new_conv, True
    else:
        return a, conv, False


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found within the allotted time.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1. Quick scan of analytic families
    scan_lengths = [2000, 2500, 3000, 3500, 4000]
    for n in scan_lengths:
        if time.time() >= deadline:
            break
        candidates = []

        # analytic families
        candidates.append(_distribution_uniform(n))
        candidates.append(_distribution_arcsine(n))
        candidates.append(_distribution_triangular(n))
        candidates.append(_distribution_exponential(n, lam=0.008))
        candidates.append(_distribution_exponential(n, lam=0.02))

        for p in (0.3, 0.6, 1.0, 2.0, 4.0):
            candidates.append(_distribution_beta_sym(n, p))

        for pw in (0.5, 1.0, 2.0, 4.0):
            candidates.append(_distribution_sine_power(n, pw))

        for sigma in (0.2, 0.35, 0.5):
            candidates.append(_distribution_gaussian(n, sigma))

        # random Dirichlet draws
        for alpha in (0.2, 0.5, 1.0, 2.0):
            candidates.append(np.random.dirichlet(np.full(n, alpha)))

        # evaluate
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] init n={n}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Pull in the global solution if present (height_sequence_1)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_seq = prev.copy()
                best_n = prev.size
                best_score = sc
                print(f"[{time.time():.1f}] global best {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback uniform if nothing found
    if best_seq is None:
        best_n = 2500
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Interpolation to a few larger sizes (may reduce constant factor)
    extra_lengths = [3500, 4000, 4500, 5000]
    for n in extra_lengths:
        if time.time() >= deadline:
            break
        # linear interpolation of the current best
        x_old = np.linspace(0.0, 1.0, best_seq.size)
        x_new = np.linspace(0.0, 1.0, n)
        vec = np.interp(x_new, x_old, best_seq)
        vec = np.maximum(vec, eps)
        vec /= vec.sum()
        sc = _max_convolution_score(vec)
        if sc < best_score:
            best_seq = vec.copy()
            best_n = n
            best_score = sc
            print(f"[{time.time():.1f}] interp n={n}  new best {sc:.6f}")

    # ------------------------------------------------------------------
    # Helper: run a full optimisation pipeline on a given vector
    def _refine_vector(a, n):
        """Run EG → PGD → Adam → hill‑climber on (a, n); return improved a."""
        nonlocal best_seq, best_score, best_n

        # ---- exponential‑gradient phase ---------------------------------
        a_eg = a.copy()
        steps_eg = 1800
        a0, a1 = 1.0, 3000.0          # smoothing parameter range
        lr0, lr1 = 0.14, 0.0014       # learning‑rate range

                break
            t = it / steps_eg
            alpha = a0 * (a1 / a0) ** t
            lr = lr0 * (lr1 / lr0) ** t
            a_eg = _exponential_gradient_step(a_eg, lr, alpha)

            if it % 300 == 0 or it == steps_eg - 1:
                sc = _max_convolution_score(a_eg)
                if sc < best_score - 1e-13:

        a_pg = a_eg.copy()
        steps_pg = 1200
        eta0, eta1 = 0.30, 0.003      # decreasing step size
        for it in range(steps_pg):
            if time.time() >= deadline:
                break
            # compute active‑set weights (sub‑gradient of the max)
            conv = _fft_self_convolve(a_pg)
            w = _subgradient_weights(conv, eps_rel=1e-12)

                if sc < best_score - 1e-13:

        # initialise logits from the current best
        logits = np.log(a_pg + eps)          # shape (n,)

        steps_adam = 2000

            if time.time() >= deadline:
                break
            a_adam = np.exp(logits)
            a_adam /= a_adam.sum()

            # gradient w.r.t. logits under softmax parametrisation
            grad_l = (grad_a - np.dot(a_adam, grad_a)) * a_adam
            logits, m, v = _adam_step(logits, m, v, it, grad_l)

            if it % 400 == 0 or it == steps_adam:
                a_adam = np.exp(logits)
                a_adam /= a_adam.sum()
                sc = _max_convolution_score(a_adam)
                if sc < best_score - 1e-13:
                    best_seq, best_score, best_n = a_adam.copy(), sc, n
                    print(f"[{time.time():.1f}] Adam it={it}  best={sc:.6f}")

        # ---- simple hill‑climber ----------------------------------------
        a_hc = best_seq if best_seq.shape[0] == n else a_adam
        conv = _fft_self_convolve(a_hc)
        hc_iters = 30000
        for it in range(hc_iters):

            a_hc, conv, improved = _hillclimber_move(a_hc, conv, eps=eps)
            if improved:

    # 5. Perform a refinement on the current best vector
    best_seq = _refine_vector(best_seq, best_n)

    # ------------------------------------------------------------------
    # 6. Multiple random restarts (while time permits)
    restart_lengths = [2000, 2500, 3000, 3500, 4000, 4500, 5000]

    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
