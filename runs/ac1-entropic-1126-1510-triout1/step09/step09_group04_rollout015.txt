```
def _smooth_max_mirror_descent(a, steps, alpha_start, alpha_end, lr_start, lr_end, deadline):
    n = len(a)
    for it in range(steps):
        if time.time() >= deadline:
            break
        t = it / steps
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # compute weights
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        grad = 2.0 * _fft_convolve(a[::-1], w)[n-1:n-1+n]
        # exponentiated update
        a = a * np.exp(-lr * grad)
        # projection onto simplex
        a = _project_onto_simplex(a, 1.0)
        # maintain positivity
        # maybe ensure a_min? Already project ensures non-negative.
    return a
```

```
def _subgrad_descent(a, max_iters, deadline):
    n = a.shape[0]
    velocity = np.zeros_like(a) # optional momentum, but we can keep simple.
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        eps_rel = 1e-10 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - eps_rel)
        w = np.zeros_like(conv)
        w[active] = 1.0 / len(active)  # equal weighting
        grad = 2.0 * _fft_convolve(a[::-1], w)[n-1:n-1+n]
        # step

        a = a - eta * grad
        # project onto simplex
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, 1e-12)
        a /= a.sum()
    return a
```

```
def _hill_climb(a, deadline):
    n = len(a)
    eps = 1e-12
    a = a.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    while time.time() < deadline:
        it += 1
        k = int(conv.argmax())
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        # find pair with max product
        best_i = i_low
        best_j = k - i_low
        max_prod = a[best_i] * a[best_j]
        for i in range(i_low, i_high+1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n-1)
        if target == source:
            continue
        delta = min(a[source]*0.2, 0.01) * random.random()
        if delta < eps:
            continue
        a_new = a.copy()
        a_new[source] -= delta
        a_new[target] += delta
        a_new = np.maximum(a_new, eps)
        a_new /= a_new.sum()
        conv_new = _fft_self_convolve(a_new)
        if conv_new.max() < M - 1e-12:
            a = a_new
            conv = conv_new
            M = conv.max()
        # occasional jitter
        if it % 50000 == 0:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
    return a
```

```
def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0)
    with np.errstate(divide='ignore'):
        w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if np.any(np.isnan(w)) or w.sum() == 0:
        w = np.full_like(x, 1.0 / n)
    else:
        w /= w.sum()
    return w

    return 1 << ((x - 1).bit_length())

    size = _next_pow2(2 * n - 1)

    A = np.fft.rfft(a, n=size)

    return conv[: a.size + b.size - 1]

"""
Propose a step‑function vector that minimises ``evaluate_sequence``.

Algorithm
---------
1.  Evaluate any previously known solution (``height_sequence_1``) and a
    handful of simple analytic families (uniform, arcsine, several
    symmetric Beta shapes).  The best of those becomes the current
    incumbent.

2.  Random restarts:
    * Choose a length `n` from a modest set (1500 … 5000).
    * Initialise a distribution with a random Dirichlet concentration.
    * Refine it by a *smooth‑max mirror descent* (exponentiated‑gradient)
      where the smooth‑max parameter `α` is increased from 1 to a large
      value (≈ 5 000) while the learning rate decays.
    * Finish the restart with a short *sub‑gradient* polishing phase
      (active‑set weights at the current maximum).
    * Keep the best vector seen so far.

3.  Final cheap hill‑climber that repeatedly moves a small amount of mass
    from the pair of entries that contributes most to the current
    convolution peak to a random position, accepting only improving
    moves.  Small random jitter is injected every few thousand steps
    to escape plateaus.

All optimisation steps keep the vector on the probability simplex,
so the final score produced by ``evaluate_sequence`` is simply
`2·n·max(a*a)` (the sum of the vector is 1).  The whole search respects
the 1000 s wall‑clock budget.

The code contains only top‑level helper functions and no closures,
as required by the evaluation harness.
"""

def _score(a: np.ndarray) -> float:

    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Smooth‑max mirror descent (exponentiated‑gradient)
def _smooth_max_mirror_descent(
    a0: np.ndarray,
    steps: int,
    alpha_start: float,
    alpha_end: float,
    lr_start: float,
    lr_end: float,
    deadline: float,
) -> np.ndarray:
    """
    Minimise a smooth approximation of `max(a*a)` using exponentiated‑gradient.
    """
    a = a0.copy()
    n = a.shape[0]

    for it in range(steps):
        if time.time() >= deadline:
            break
        t = it / steps
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t

        conv = _fft_self_convolve(a)
        maxc = conv.max()

        # soft‑max weights (numerically stable)
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth max
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # exponentiated‑gradient update in log‑space
        log_a = np.log(a + 1e-30) - lr * grad
        log_a -= np.max(log_a)               # avoid overflow
        a = np.exp(log_a)
        a = _project_onto_simplex(a, 1.0)

    return a

# Sub‑gradient polishing (active‑set)
def _subgradient_descent(
    a0: np.ndarray, max_iters: int, deadline: float
) -> np.ndarray:
    """A few sub‑gradient steps using the true max (indicator weights)."""
    a = a0.copy()
    n = a.shape[0]

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        eps_rel = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - eps_rel)

        if active.size:
            w[active] = 1.0 / active.size

        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        eta = 0.15 / np.sqrt(it + 1)
        a = a - eta * grad
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, 1e-12)
        a /= a.sum()

    return a

    # 1.  Initialise from any previously discovered sequence.
    try:
        prev = np.asarray(globals().get("height_sequence_1", None), dtype=np.float64)
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 1e-12)
            prev /= prev.sum()
            sc = _score(prev)
            best_seq = prev.copy()
            best_score = sc
            best_n = prev.size
            print(f"[{time.time():.1f}] start from global best n={best_n} score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 2.  Quick scan of a few analytic families.
    lengths = [1500, 2000, 2500, 3000, 3500, 4000, 4500, 5000]
    for n in lengths:
        if time.time() >= deadline:
            break

        # uniform
        uni = np.full(n, 1.0 / n, dtype=np.float64)
        sc = _score(uni)
        if sc < best_score:
            best_seq, best_score, best_n = uni.copy(), sc, n
            print(f"[{time.time():.1f}] uniform n={n} score={sc:.6f}")

        # arcsine (Beta(0.5,0.5))
        arcs = _distribution_beta_sym(n, 0.5)
        sc = _score(arcs)
        if sc < best_score:
            best_seq, best_score, best_n = arcs.copy(), sc, n
            print(f"[{time.time():.1f}] arcsine n={n} score={sc:.6f}")

        # a few symmetric Beta shapes
        for p in (0.2, 0.3, 0.4, 0.6, 0.8, 1.0, 1.5, 2.0):
            beta = _distribution_beta_sym(n, p)
            sc = _score(beta)
            if sc < best_score:
                best_seq, best_score, best_n = beta.copy(), sc, n
                print(f"[{time.time():.1f}] beta p={p} n={n} score={sc:.6f}")

    # fallback in the unlikely case nothing was found
    if best_seq is None:
        n = 3000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_n = n
        best_score = _score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform n={n} score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3.  Random restarts followed by mirror‑descent + sub‑gradient polishing.
    restart = 0
    while time.time() < deadline - 2.0:
        restart += 1
        n = random.choice(lengths)

        # Dirichlet concentration sampled log‑uniformly in [0.01, 1]
        conc = 10 ** np.random.uniform(-2, 0)
        a0 = np.random.dirichlet(np.full(n, conc))
        a0 = np.maximum(a0, 1e-12)
        a0 /= a0.sum()

        # allocate a modest number of mirror‑descent steps (adapt to remaining time)
        remaining = deadline - time.time()
        # crude estimate: ~0.0008 s per step for n≈3000 on usual hardware
        steps = max(500, min(3000, int(remaining / 0.0008)))
        a_opt = _smooth_max_mirror_descent(
            a0,
            steps=steps,
            alpha_start=1.0,
            alpha_end=5000.0,
            lr_start=0.8,
            lr_end=0.001,
            deadline=deadline,
        )

        # short sub‑gradient polishing (800 iterations max)
        a_opt = _subgradient_descent(a_opt, max_iters=800, deadline=deadline)

        sc = _score(a_opt)
        if sc < best_score:
            best_seq, best_score, best_n = a_opt.copy(), sc, n
            print(f"[{time.time():.1f}] restart {restart} n={n} improved score={sc:.6f}")

    # --------------------------------------------------------------
    # 4.  Final hill‑climber on the incumbent solution.
    best_seq = _hill_climb(best_seq, deadline=deadline)

    final_score = _score(best_seq)
    print(f"[{time.time():.1f}] Final best score ≈ {final_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
```
