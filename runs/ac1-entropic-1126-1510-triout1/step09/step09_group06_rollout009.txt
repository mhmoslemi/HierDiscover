"""
Propose a step‑function that minimizes the evaluation function.

Algorithm Overview
------------------
1. **Candidate generation (broad sampling)** – Create a moderate number of
   probability vectors using several analytic families (arcsine, triangular,
   exponential, Gaussian, Cauchy, symmetric/unsymmetric Beta, power‑law) on
   a few lengths around the length of the previously best sequence.
   For each vector also apply three deterministic re‑orderings that tend to
   flatten the self‑convolution.

   scheme on a smooth approximation of the max‑convolution:
       smooth_max_α(a) = max_i (a∗a)_i  (approximated by a log‑sum‑exp with
       temperature α).  The gradient is computed analytically via FFT and an
   Adam update is performed on the simplex (Euclidean projection after each
   step).  The temperature α is increased geometrically (0.5 → … → 80) to
   approach the true max.

   2·n·max(a∗a)).  Further improve it with a few Frank‑Wolfe style mass‑transfer
   steps (move mass from the index with the largest sub‑gradient to the one

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate for `evaluate_sequence` when Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _gaussian_distribution(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    w /= w.sum()
    return w


def _cauchy_distribution(n: int, scale_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    scale = scale_factor * n
    w = 1.0 / (1.0 + ((idx - centre) / scale) ** 2)
    w /= w.sum()
    return w


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Unsymmetric Beta(α,β) on a discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide='ignore', invalid='ignore'):
        w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    if np.isnan(w).any() or np.isinf(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    w = 1.0 / (idx ** p)
    w /= w.sum()
    return w


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex from Dirichlet(α,…,α)."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Re‑ordering heuristics -----------------------------------------------
def order_alternate_extremes(desc_vals: np.ndarray) -> np.ndarray:
    """Place the largest, second‑largest, … alternately at the left‑most
    and right‑most positions, spreading large masses."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    left = 0
    right = n - 1
    toggle = True
    for v in desc_vals:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    return out


def order_center_outwards(desc_vals: np.ndarray) -> np.ndarray:
    """Put the biggest value at the centre and then fill alternatingly
    to the left and to the right, moving outwards."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    positions = []
    mid = (n - 1) // 2
    positions.append(mid)
    for d in range(1, n):
        left = mid - d
        if left >= 0:
            positions.append(left)
        right = mid + d
        if right < n:
            positions.append(right)
    for i, pos in enumerate(positions[:n]):
        out[pos] = desc_vals[i]
    return out


def order_binary_tree(desc_vals: np.ndarray) -> np.ndarray:
    """Spread large coefficients by recursively filling the midpoint of
    the remaining interval (binary‑tree layout)."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)

    def recurse(l: int, r: int):
        if l > r:
            return
        m = (l + r) // 2
        positions.append(m)
        recurse(l, m - 1)
        recurse(m + 1, r)

    positions = []
    recurse(0, n - 1)
    out[positions] = desc_vals
    return out


# ----------------------------------------------------------------------
# Smooth‑max gradient utilities -----------------------------------------
def _smooth_max_and_grad(a: np.ndarray, alpha: float) -> tuple[float, np.ndarray]:
    """
    Compute a smooth approximation of max(conv) and its gradient.
    Returns (smooth_max, gradient) where gradient has length len(a).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    # self‑convolution
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)[: 2 * n - 1]          # length 2n‑1
    maxc = conv.max()
    # safe exponentiation
    w = np.exp(alpha * (conv - maxc))               # max weight = 1
    sum_w = w.sum()
    # gradient: 2 * (a_rev * w) slice / sum_w
    conv_rev = _fft_convolve(a[::-1], w)            # full convolution
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n] / sum_w
    # smooth max value (stable)
    smooth_max = maxc + np.log(sum_w) / alpha
    return smooth_max, grad


# ----------------------------------------------------------------------
# Adam optimiser on the smooth objective --------------------------------
def _adam_smooth_descent(a0: np.ndarray,
                         deadline: float,
                         alphas: list[float],
                         iters_per_alpha: int = 2000,
                         step0: float = 0.2,
                         beta1: float = 0.9,
                         beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max‑convolution.
    Returns the best vector (exact objective) found during the run.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    # exact score (objective = 2·n·max(conv))
    conv = _fft_self_convolve(a)
    best_score = 2.0 * n * conv.max()
    t = 0
    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score
            t += 1
            smooth_max, grad = _smooth_max_and_grad(a, alpha)
            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            eta = step0 / (np.sqrt(v_hat) + 1e-8)
            a = a - eta * m_hat
            a = _project_onto_simplex(a, s=1.0)
            # exact objective (no extra FFT, reuse conv)
            conv = _fft_self_convolve(a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _frank_wolfe_refine(a: np.ndarray,
                        deadline: float,
                        max_iters: int = 50000) -> tuple[np.ndarray, float]:
    """Simple Frank‑Wolfe style mass transfer on the current best."""
    n = a.shape[0]
    best_a = a.copy()
    conv = _fft_self_convolve(a)
    best_score = 2.0 * n * conv.max()
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # exact sub‑gradient (same as in the original code)
        conv = _fft_self_convolve(best_a)
        maxc = conv.max()
        eps = 1e-12 * (abs(maxc) + 1.0)
        active = np.flatnonzero(conv >= maxc - eps)
        w = np.zeros_like(conv)
        w[active] = 1.0
        conv_rev = _fft_convolve(best_a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        delta = min(best_a[i_src], 1.0 - best_a[i_dst])
        if delta <= 0:
            break
        step = 0.7 * delta
        best_a[i_src] -= step
        best_a[i_dst] += step
        cur_score = 2.0 * n * _fft_self_convolve(best_a).max()
        if cur_score < best_score:
            best_score = cur_score
            # occasionally re‑project to the simplex to correct numeric drift
            best_a = _project_onto_simplex(best_a, s=1.0)
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the supplied time budget (≈ 1000 s) and returns the
    best vector discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose a base length (around the length of the previous best if known)
    try:
        base_n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        base_n = 3500
    # exploration lengths (moderate range)
    lengths = set()
    for factor in (0.8, 1.0, 1.2):
        n = int(base_n * factor)
        if 800 <= n <= 8000:
            lengths.add(n)
    # a few random lengths for diversity
    for _ in range(3):
        n = np.random.randint(800, min(8000, base_n * 3) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # Helper to normalise a vector and store its exact surrogate score
    candidates = []   # list of (array, exact_score)

    def add_candidate(arr: np.ndarray):
        a = np.asarray(arr, dtype=np.float64)
        if a.ndim != 1 or a.size == 0:
            return
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            return
        a = a / s
        score = _max_convolution_score(a)
        candidates.append((a, score))

    # ------------------------------------------------------------------
    # 1. Broad sampling (families + deterministic reorderings)
    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- Simple analytic families --------------------------------
        families = [
            (_arcsine_distribution, "arcsine"),
            (_triangular_distribution, "triangular"),
            (_exponential_distribution, "exponential"),
            (_gaussian_distribution, "gaussian"),
            (_cauchy_distribution, "cauchy")
        ]
        for gen, _ in families:
            a = gen(n)
            add_candidate(a)
            desc = np.sort(a)[::-1]
            add_candidate(order_alternate_extremes(desc))
            add_candidate(order_center_outwards(desc))
            add_candidate(order_binary_tree(desc))

        # ----- Symmetric beta (wide grid) -------------------------------
        beta_ps = list(np.arange(0.05, 0.55, 0.05)) + list(np.arange(0.6, 2.1, 0.2))
        for p in beta_ps:
            if time.time() >= deadline:
                break
            a = _beta_symmetric_distribution(n, p)
            add_candidate(a)
            desc = np.sort(a)[::-1]
            add_candidate(order_alternate_extremes(desc))
            add_candidate(order_center_outwards(desc))
            add_candidate(order_binary_tree(desc))

        # ----- Unsymmetric beta (coarse grid) ---------------------------
        beta_params = [0.2, 0.5, 1.0, 2.0, 5.0]
        for alpha in beta_params:
            for beta in beta_params:
                if time.time() >= deadline:
                    break
                a = _beta_distribution(n, alpha, beta)
                add_candidate(a)
                desc = np.sort(a)[::-1]
                add_candidate(order_alternate_extremes(desc))

        # ----- Power‑law families ---------------------------------------
        for pwr in [0.5, 1.0, 1.5, 2.0]:
            a = _power_law_distribution(n, pwr)
            add_candidate(a)
            desc = np.sort(a)[::-1]
            add_candidate(order_alternate_extremes(desc))

        # ----- Random Dirichlet draws -----------------------------------
        for alpha in [0.2, 0.5, 1.0, 2.0]:
            if time.time() >= deadline:
                break
            a = _random_dirichlet_alpha(n, alpha)
            add_candidate(a)

    # ------------------------------------------------------------------
    # 2. Insert the previously best construction (if any)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        if prev.sum() > 0:
            prev = prev / prev.sum()
            add_candidate(prev)
            print(f"Added previous best candidate, score={candidates[-1][1]:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Keep a small elite set of seeds
    candidates.sort(key=lambda x: x[1])
    elite_k = min(12, len(candidates))
    elite = candidates[:elite_k]
    print(f"Generated {len(candidates)} candidates, keeping top {elite_k}.")
    print(f"Initial elite best score = {elite[0][1]:.6f}")

    # ------------------------------------------------------------------
    # 4. Smooth‑max Adam optimisation for each elite seed
    best_seq = elite[0][0].copy()
    best_score = elite[0][1]
    alphas_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0]

    for idx, (seed_vec, _) in enumerate(elite):
        if time.time() >= deadline:
            break
        seq, sc = _adam_smooth_descent(seed_vec, deadline,
                                       alphas=alphas_schedule,
                                       iters_per_alpha=1500,
                                       step0=0.2)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] elite {idx+1}/{elite_k} Adam‑smooth -> {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Frank‑Wolfe style refinement on the global best
    best_seq, best_score = _frank_wolfe_refine(best_seq, deadline,
                                               max_iters=70000)

    # ------------------------------------------------------------------
    # 6. Final re‑ordering trial (keep the ordering that gives the smallest exact score)
    desc = np.sort(best_seq)[::-1]
    reorderings = [
        order_alternate_extremes,
        order_center_outwards,
        order_binary_tree,
        lambda x: x  # identity (no reordering)
    ]
    for fn in reorderings:
        cand = fn(desc)
        sc = _max_convolution_score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"Re‑ordering improvement: new best {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
