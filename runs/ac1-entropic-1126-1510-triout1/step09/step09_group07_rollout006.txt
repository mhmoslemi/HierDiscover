"""
Propose a step‑function that minimises ``evaluate_sequence``.

The algorithm consists of three stages:

1️⃣  **Broad coarse sweep** – Enumerate many analytically‑motivated families
    (arcsine, smoothed arcsine, product‑power, symmetric/asymmetric Beta,
    sine, cosine, triangular, exponential) on several lengths.
    For each family we also try a few random Dirichlet draws and simple
    mixtures (e.g. arcsine + exponential).  The cheapest 20 candidates are
    kept.

2️⃣  **Intensive refinement** – The best seeds are polished with several
    optimisation primitives that operate directly on the probability simplex:
        * exponentiated‑gradient (mirror descent)
        * Adam
        * a continuation method that smooths the max‑operator
        * heavy‑ball sub‑gradient
    After each primitive we additionally try a small set of power‑transforms
    (a_i←a_i^β) and a tiny Gaussian blur.  The best obtained vector is kept.

3️⃣  **Simulated‑annealing hill climbing** – Starting from the refined
    vector we perform a large number of random mass‑transfer moves.
    Moves are accepted either if they improve the cheap objective
    ``2·n·max(conv(a,a))`` or with a Metropolis probability that slowly
    cools down.  This helps to escape shallow local minima.

All vectors are projected onto the probability simplex after every
operation, guaranteeing feasibility (non‑negative, sum = 1).  The routine
respects the global time budget (default 1000 s) and returns a plain
list of floats ready for ``evaluate_sequence``.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap surrogate = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (k = 2*i)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

# Families of probability vectors
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _smoothed_arcsine_distribution(n: int, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5 + eps) * (n - i - 0.5 + eps))
    return vals / vals.sum()


def _product_power_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-alpha)
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, p: float, q: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    w = np.where(np.isfinite(w), w, 0.0)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    return vals / vals.sum()


def _random_dirichlet(rng: np.random.Generator, n: int) -> np.ndarray:
    x = rng.gamma(1.0, 1.0, size=n)
    return x / x.sum()


def _mix_vectors(a: np.ndarray, b: np.ndarray, w: float) -> np.ndarray:
    """Return w·a + (1‑w)·b, projected onto the simplex."""
    mix = w * a + (1.0 - w) * b
    return _project_onto_simplex(mix, s=1.0)

# Transformations used during refinement
def _power_transform(a: np.ndarray, beta: float) -> np.ndarray:
    # a_i^beta  (beta>0) then renormalise
    b = np.power(a, beta)
    if b.sum() == 0:
        return a
    return b / b.sum()


def _gaussian_smooth(a: np.ndarray, sigma: float = 1.0) -> np.ndarray:
    # discrete Gaussian kernel (odd length)
    radius = max(1, int(3 * sigma))
    x = np.arange(-radius, radius + 1, dtype=np.float64)
    w = np.exp(-0.5 * (x / sigma) ** 2)
    w /= w.sum()
    sm = np.convolve(a, w, mode='same')
    sm = np.maximum(sm, 0.0)
    sm_sum = sm.sum()
    if sm_sum == 0:
        return a
    return sm / sm_sum

def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 150_000, eta0: float = 0.8):
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a /= a.sum()
        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 150_000, step0: float = 0.2,
                  beta1: float = 0.9, beta2: float = 0.999):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 400_000, eta0: float = 0.6):
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _smooth_gradient(a: np.ndarray, gamma: float):
    n = a.shape[0]
    b = _fft_self_convolve(a)
    b_max = b.max()
    w = np.exp(gamma * (b - b_max))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    diag_w = w[2 * np.arange(n)]
    grad -= a * diag_w
    return grad, b_max


def _smooth_gradient_descent(a0: np.ndarray, deadline: float,
                             max_iters: int = 300_000,
                             eta0: float = 0.7,
                             gamma0: float = 5.0,
                             gamma_factor: float = 1.001):
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    gamma = gamma0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _ = _smooth_gradient(a, gamma)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
        gamma *= gamma_factor
    return best_a, best_score


def _heavy_ball_subgradient_descent(a0: np.ndarray, deadline: float,
                                   max_iters: int = 300_000,
                                   eta0: float = 0.6, momentum: float = 0.6):
    a_prev = a0.copy()
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * grad + momentum * (a - a_prev)
        a_new = _project_onto_simplex(a_new, s=1.0)
        a_prev, a = a, a_new
        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

def _simulated_annealing(a0: np.ndarray, deadline: float,
                         max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """Random mass‑transfer hill‑climber with a simple temperature schedule."""
    rng = np.random.default_rng()
    a = a0.copy()
    cur_score = _max_convolution_score(a)
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    start = time.time()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # linear cooling from T0 to ~0
        T = max(0.02 * (1.0 - (time.time() - start) / (deadline - start)), 1e-6)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        # transfer at most a fraction of the mass at i
        delta = rng.random() * min(0.02, a[i])
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)
        accept = (new_score < cur_score) or (rng.random() < math.exp((cur_score - new_score) / T))
        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_a = new_score, a_new.copy()
    return best_a, best_score

            candidates.sort(key=lambda x: x[0])
        elif sc < candidates[-1][0]:
            candidates[-1] = (sc, vec.copy())
            candidates.sort(key=lambda x: x[0])

    for n in lengths:
        if time.time() >= deadline:
            break

        # arcsine and smoothed variants
        for eps in (0.0, 0.1, 0.5, 1.0, 2.0):
            a = _arcsine_distribution(n) if eps == 0.0 else _smoothed_arcsine_distribution(n, eps)
            _push_candidate(a)

        # product‑power family
        for alpha in np.arange(0.30, 0.71, 0.05):
            a = _product_power_distribution(n, alpha)
            _push_candidate(a)

        # symmetric Beta
        for p in np.arange(0.30, 0.81, 0.05):
            a = _beta_symmetric_distribution(n, p)
            _push_candidate(a)

        # asymmetric Beta (few random draws)
        for _ in range(4):
            p = rng.uniform(0.2, 1.5)
            q = rng.uniform(0.2, 1.5)
            a = _beta_asymmetric_distribution(n, p, q)
            _push_candidate(a)

        # sine / cosine
        _push_candidate(_sine_distribution(n))
        _push_candidate(_cosine_distribution(n))

        # triangular & exponential
        _push_candidate(_triangular_distribution(n))
        for lam in (0.001, 0.003, 0.005, 0.01):
            a = _exponential_distribution(n, lam)
            _push_candidate(a)

        # a few random Dirichlet draws
        for _ in range(8):
            a = _random_dirichlet(rng, n)
            _push_candidate(a)

        # mixtures arcsine + exponential
        a_arc = _arcsine_distribution(n)
        a_exp = _exponential_distribution(n, lam=0.003)
        for w in (0.2, 0.5, 0.8):
            _push_candidate(_mix_vectors(a_arc, a_exp, w))

    # Keep the best one so far as a starting point
    best_seq, best_score = candidates[0][1].copy(), candidates[0][0]

    # also try to import a previously known sequence if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.size > 0:
            prev = prev / prev.sum()
            _push_candidate(prev)
            # maybe replace best_seq now
            if candidates[0][0] < best_score:
                best_score, best_seq = candidates[0][0], candidates[0][1].copy()
    except Exception:
        pass

    print(f"[{time.time():.1f}] coarse sweep best cheap={best_score:.6f} length={len(best_seq)}")

    # ------------------------------------------------------------------
    # 2️⃣ Heavy refinement – allocate most of the remaining budget here
    remaining = deadline - time.time()
    if remaining <= 0:
        return best_seq.tolist()

    # fraction of remaining time per method
    fractions = [0.20, 0.20, 0.30, 0.15, 0.10]   # last slot kept for SA
    method_funs = [
        (_exp_gradient_descent, {'eta0': 1.0, 'max_iters': 150_000}),
        (_adam_descent, {'step0': 0.3, 'max_iters': 150_000}),
        (_smooth_gradient_descent,
         {'eta0': 0.8, 'gamma0': 2.0, 'gamma_factor': 1.005, 'max_iters': 250_000}),
        (_heavy_ball_subgradient_descent,
         {'eta0': 0.6, 'momentum': 0.6, 'max_iters': 250_000}),
    ]

    for (fun, kwargs), frac in zip(method_funs, fractions):
        sub_deadline = time.time() + remaining * frac
        if time.time() >= deadline:
            break
        seq, sc = fun(best_seq, sub_deadline, **kwargs)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time():.1f}] after {fun.__name__:20s} cheap={best_score:.6f}")

        # ----- power‑transform sweep -----
        for beta in (0.6, 0.8, 1.0, 1.2, 1.4):
            a_pow = _power_transform(best_seq, beta)
            sc_pow = _max_convolution_score(a_pow)
            if sc_pow < best_score:
                best_seq, best_score = a_pow, sc_pow
                print(f"[{time.time():.1f}] power β={beta:.2f} cheap={best_score:.6f}")

        # ----- Gaussian smoothing sweep -----
        for sigma in (0.5, 1.0, 1.5):
            a_smooth = _gaussian_smooth(best_seq, sigma)
            sc_s = _max_convolution_score(a_smooth)
            if sc_s < best_score:
                best_seq, best_score = a_smooth, sc_s
                print(f"[{time.time():.1f}] gaussian σ={sigma:.1f} cheap={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Simulated‑annealing hill climbing (final polishing)
    if time.time() < deadline:
        seq, sc = _simulated_annealing(best_seq, deadline,
                                       max_iter=300_000)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time():.1f}] after simulated‑anneal cheap={best_score:.6f}")

    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
