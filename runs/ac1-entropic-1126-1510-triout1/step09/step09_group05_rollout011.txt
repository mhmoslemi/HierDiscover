    n = f.size

    t = 0.2
    # main loop

                                         method='highs', options={'presolve': True, 'time_limit': max(deadline - time.time(), 0)})

            break
        if not res.success:

def _lp_refine(f, deadline, t_start=0.2):
    n = f.shape[0]
    best_f = f.copy()
    best_score = _max_convolution_score(best_f)
    t = t_start
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        conv = _fft_self_convolve(f)
        M = conv.max()
        eps = 1e-12 * (abs(M) + 1.0)
        active = np.flatnonzero(conv >= M - eps)
        if active.size == 0:
            break
        A = _build_active_conv_matrix(f, active)
        c = -np.ones(n)
        b_ub = np.full(active.shape[0], M)
        bounds = [(0.0, None)] * n
        # solve
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            print(f"LP failed: {res.message}")
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break
        g = b / sum_b
        # convex combination
        f = (1 - t) * f + t * g
        f = f / f.sum()
        # evaluate current score
        cur_score = _max_convolution_score(f)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_f = f.copy()
            print(f"[{time.time():.1f}] LP iter {iteration} obj={cur_score:.6f} M={M:.6f} active={len(active)}")
        # step-size decay
        if iteration % 10 == 0:
            t = max(min_t, t * t_decay)
        # early exit maybe
        if iteration >= 5000:  # safety
            break
    return best_f, best_score

def _hill_climb(a, deadline, best_score):
    n = a.shape[0]
    cur_score = best_score
    best_a = a.copy()
    iteration = 0
    max_transfer = 0.025
    while time.time() < deadline:
        iteration += 1
        transfer = max_transfer * (0.5 ** (iteration / 8000.0))
        # compute subgrad
        grad, _, _ = _subgrad(a)
        # choose donor/receiver based on gradient
        if np.random.rand() < 0.5:
            i = int(np.argmax(grad))
            j = int(np.argmin(grad))
        else:
            i = np.random.randint(n)
            j = np.random.randint(n-1)
            if j >= i:
                j += 1
        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)
        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill climb iter={iteration} new best {best_score:.6f}")
        # occasional jitter
        if iteration % 6000 == 0 and np.random.rand() < 0.15:
            jitter = np.random.randn(n) * 6e-5
            a = best_a + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)
    return best_a, best_score

"""
Search for a step‑function minimizing ``evaluate_sequence``.

Algorithm
---------
1️⃣ Generate a modest pool of analytic families (arcsine, beta,
   exponential, uniform, triangular) for a few lengths around the
   size of the previously best construction (if available).

2️⃣ Keep the top few seeds according to the cheap surrogate
   ``2·n·max(conv)`` (the true objective when the sum of the sequence is
   1).

3️⃣ For each seed run an **active‑set linear‑program refinement**:
   * Let ``M = max(conv(f,f))``.
   * Identify the active indices where the convolution attains its
     maximum (up to a tiny tolerance).
   * Build a sparse convolution matrix that maps a candidate ``b`` to
     ``conv(f,b)`` only on those active rows.
   * Solve the LP   ``max sum(b)``  s.t. ``conv(f,b) ≤ M``  and ``b≥0``.
   * Mix the solution with the current iterate:
        ``f←(1−t)·f + t·(b / sum(b))``   with a decaying step size ``t``.
   * Iterate while time permits.  The objective equals ``2·n·M`` (since
     the sequence is normalised to sum 1), so reduction of ``M`` directly
     improves the true bound.

4️⃣ After the LP phase perform a **gradient‑guided hill‑climb** using the
   exact sub‑gradient of ``max(conv)`` on the simplex.  Small amounts of
   mass are transferred from high‑gradient indices to low‑gradient ones,
   with occasional random jitter to escape plateaus.

5️⃣ Return the best sequence found before the deadline.  All entries are
   non‑negative and the sum is ≥ 0.01, satisfying the verifier’s checks.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap score = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.
    Returns (grad, cur_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
# Analytic families -------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_w = (p - 1) * (np.log(x) + np.log(1.0 - x))
    log_w = np.clip(log_w, -700, 700)
    w = np.exp(log_w)
    if w.sum() == 0.0:
        return np.full_like(w, 1.0 / n)
    return w / w.sum()


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)

def _build_active_conv_matrix(f: np.ndarray, active: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Build a sparse matrix A such that (A @ b)[k] = conv(f, b) on the
    rows indexed by ``active`` (0‑based).  Each row corresponds to one
    convolution index.
    """
    n = f.shape[0]
    row_ind = []
    col_ind = []
    data = []

    for r, k in enumerate(active):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        length = i_end - i_start + 1
        row_ind.extend([r] * length)
        col_ind.extend(range(i_start, i_end + 1))
        data.extend(f[i_start : i_end + 1])

    return scipy.sparse.csr_matrix((data, (row_ind, col_ind)), shape=(len(active), n))


# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray, deadline: float, t_start: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Refine ``f`` by repeatedly solving the active‑set LP:
        maximise   sum(b)
        subject to  conv(f, b) ≤ M   on active rows,  b ≥ 0,
    and mixing the solution back into ``f``.
    Returns the best encountered sequence and its cheap score.
    """
    n = f.shape[0]
    best_f = f.copy()
    best_score = _max_convolution_score(best_f)
    t = t_start
    no_improve = 0
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        conv = _fft_self_convolve(f)
        M = conv.max()
        eps = 1e-12 * (abs(M) + 1.0)
        active = np.flatnonzero(conv >= M - eps)
        if active.size == 0:
            break

        A = _build_active_conv_matrix(f, active)
        c = -np.ones(n)                # maximise sum(b)  ⇔ minimise -sum(b)
        b_ub = np.full(active.shape[0], M)
        bounds = [(0.0, None)] * n

        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                    bounds=bounds, method='highs')
        if not res.success:
            # LP failure – abort this refinement
            print(f"[{time.time():.1f}] LP failed: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            break
        g = b / sum_b

        # convex combination and renormalisation
        f = (1.0 - t) * f + t * g
        f = f / f.sum()

        cur_score = _max_convolution_score(f)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_f = f.copy()
            no_improve = 0
            print(f"[{time.time():.1f}] LP iter {iteration} improved obj={best_score:.6f} M={M:.6f} active={len(active)}")
        else:
            no_improve += 1

        # decay the mixing weight
        if iteration % 5 == 0:
            t = max(0.01, t * 0.95)

        if no_improve >= 200 or iteration >= 5000:
            break

    return best_f, best_score


# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray, deadline: float, start_score: float) -> tuple[np.ndarray, float]:
    """
    Gradient‑guided hill‑climbing: repeatedly move a tiny amount of
    mass from a high‑gradient index to a low‑gradient one, accepting
    only improvements.
    """
    n = a.shape[0]
    cur_score = start_score
    best_a = a.copy()
    best_score = start_score
    iteration = 0
    max_transfer = 0.025

    while time.time() < deadline:
        iteration += 1
        transfer = max_transfer * (0.5 ** (iteration / 8000.0))

        # donor / receiver selection (gradient‑biased half the time)
        if np.random.rand() < 0.5:
            grad, _, _ = _subgrad(a)
            i = int(np.argmax(grad))
            j = int(np.argmin(grad))
            if i == j:
                i = np.random.randint(n)
                j = (i + np.random.randint(1, n)) % n
        else:
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} new best {best_score:.6f}")

        # occasional jitter to leave flat plateaus
        if iteration % 6000 == 0 and np.random.rand() < 0.15:
            jitter = np.random.randn(n) * 6e-5
            a = best_a + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function (non‑negative, sum ≥ 0.01) that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget
    and applies analytic seeding → LP refinement → hill‑climb.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Base length – use length of previous best if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        n_base = prev.shape[0]
        prev = np.maximum(prev, 0.0)
        if prev.sum() == 0.0:
            raise Exception
        prev = prev / prev.sum()
    except Exception:
        n_base = 3000
        prev = _arcsine_distribution(n_base)

    # ------------------------------------------------------------------
    # Lengths to explore (moderate neighbourhood + a few random ones)
    lengths = set()
    for f in (0.8, 1.0, 1.2):
        lengths.add(max(500, int(n_base * f)))
    for _ in range(3):
        lengths.add(np.random.randint(500, min(20000, n_base * 2) + 1))
    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # Generate a pool of analytic seeds
    seeds = []
    beta_ps = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0]
    exp_lams = [0.001, 0.003, 0.006, 0.01]

    for n in lengths:
        # arcsine, uniform, triangular
        seeds.append(_arcsine_distribution(n))
        seeds.append(_uniform_distribution(n))
        seeds.append(_triangular_distribution(n))

        # exponential family
        for lam in exp_lams:
            seeds.append(_exponential_distribution(n, lam=lam))

        # symmetric beta family
        for p in beta_ps:
            seeds.append(_beta_symmetric_distribution(n, p))

    # add the previous best if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        if prev.sum() > 0.0:
            prev = prev / prev.sum()
            seeds.append(prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Keep the most promising seeds (cheap score)
    scored = [(s, _max_convolution_score(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [s for s, _ in scored[:top_k]]
    print(f"Selected {len(top_seeds)} seeds for refinement; best cheap score={scored[0][1]:.6f}")

    best_f = top_seeds[0].copy()
    best_score = _max_convolution_score(best_f)

    # ------------------------------------------------------------------
    # Refine each seed: LP phase → hill‑climb phase
    for idx, seed in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Refinement start for seed {idx+1}/{len(top_seeds)}")
        # LP‑based active‑set refinement
        f_refined, score_refined = _lp_refine(seed.copy(), deadline, t_start=0.2)
        if score_refined < best_score - 1e-14:
            best_score, best_f = score_refined, f_refined.copy()
            print(f"[{time.time():.1f}] LP improvement from seed {idx+1}: {best_score:.6f}")

        # Hill‑climb polish (short, keep a few seconds spare)
        hill_deadline = deadline - 5.0
        if time.time() < hill_deadline:
            f_hill, score_hill = _hill_climb(best_f.copy(), hill_deadline, best_score)
            if score_hill < best_score - 1e-14:
                best_score, best_f = score_hill, f_hill.copy()
                print(f"[{time.time():.1f}] Hill‑climb improvement after seed {idx+1}: {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final cheap objective: {best_score:.6f} (length={len(best_f)})")
    # Ensure non‑negativity, sum≥0.01 (already true) and return as list
    best_f = np.maximum(best_f, 0.0)
    return best_f.tolist()
