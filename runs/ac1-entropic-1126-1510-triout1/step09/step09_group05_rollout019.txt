# Helper functions
def _project_onto_simplex(v, s=1.0):
    """ Euclidean projection onto the simplex {x>=0, sum x = s}. """
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if ind.size == 0:
        theta = 0.0
    else:
        rho = ind[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _fft_self_convolve(a):
    """ full self‑convolution (length 2n‑1) via real FFT """
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_convolve(a, b):
    """ full convolution via real FFT """
    size = 1 << ((a.size + b.size - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a):
    """ cheap score = 2·n·max(conv) (sum a = 1) """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

def _subgrad(a):
    """ exact sub‑gradient of max(conv) on the simplex """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)
    w = np.zeros_like(b)
    w[active] = 1.0
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active

def _smooth_grad(a, eps):
    """ gradient of smooth max via log‑sum‑exp """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    maxc = conv.max()
    w = np.exp((conv - maxc) / eps)
    w_sum = w.sum()
    w /= w_sum
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    diag_idx = np.arange(conv.size)[(np.arange(conv.size) % 2) == 0] // 2
    grad[diag_idx] -= w[2 * diag_idx] * a[diag_idx]
    return grad

def _smooth_adam_descent(a0, deadline,
                         max_iters=300_000,
                         eps0=5e-3,
                         step0=0.25,
                         beta1=0.9, beta2=0.999):
    """ Adam on smoothed max‑convolution. eps is halved every 30k iters. """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    eps = eps0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad = _smooth_grad(a, eps)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, 1.0)
        if t % 30_000 == 0:
            eps = max(eps * 0.5, 1e-9)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

def _subgradient_descent(a0, deadline,
                         max_iters=1_500_000,
                         eta0=0.8):
    """ Simple sub‑gradient descent with diminishing step size """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, 1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

# Analytic families
def _arcsine_distribution(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()

def _powerlaw_distribution(n, p):
    """ a_i ∝ ((i+0.5)*(n-i-0.5))^{-p} """
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.power((i + 0.5) * (n - i - 0.5), p)
    return vals / vals.sum()

def _edge_block_distribution(n, edge_frac, ratio):
    """two‑zone distribution: edge region width = edge_frac·n, edge weight = ratio·interior weight."""
    edge_len = max(1, int(round(n * edge_frac)))
    interior_len = n - 2 * edge_len
    if interior_len < 0:
        edge_len = n // 2
        interior_len = 0
    # interior weight = 1, edge weight = ratio
    a = np.empty(n, dtype=np.float64)
    a[:edge_len] = ratio
    a[-edge_len:] = ratio
    if interior_len > 0:
        a[edge_len:n - edge_len] = 1.0
    a = a / a.sum()
    return a

def _beta_symmetric_distribution(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_w = (p - 1) * (np.log(x) + np.log(1.0 - x))
    log_w = np.clip(log_w, -700, 700)
    w = np.exp(log_w)
    if w.sum() == 0.0:
        return np.full_like(w, 1.0 / n)
    return w / w.sum()

def _triangular_distribution(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()

def _uniform_distribution(n):
    return np.full(n, 1.0 / n)

def _random_dirichlet(n, alpha=1.0):
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return x / x.sum()

"""
search for a step‑function (non‑negative, sum ≈ 1) that minimises
``evaluate_sequence``.  The algorithm consists of:

1️⃣ **Broad sampling** – analytic families (including new power‑law and
   edge‑block shapes) plus many Dirichlet draws, evaluated with the cheap
   objective 2·n·max(conv(a,a)).  The best K candidates are kept.

2️⃣ **Heavy refinement** – each seed is optimised with a smooth‑Adam
   optimiser that minimises a log‑sum‑exp surrogate of the max‑convolution.
   The smoothing parameter ε is halved regularly, giving a very accurate
   approximation of the true objective.

3️⃣ **Polishing** – a long sub‑gradient descent (projected onto the simplex)
   is applied to the current best sequence.

4️⃣ **Final hill‑climb** – a short simulated‑annealing loop guided by the
   exact sub‑gradient moves a tiny amount of mass between indices, occasionally
   accepting uphill moves to escape plateaus.

All steps respect the supplied time budget (default 1000 s) and the routine
always returns a feasible list of floats.

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if ind.size == 0:
        theta = 0.0
    else:
        rho = ind[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """
    Cheap score = 2·n·max(conv(a,a)) assuming Σa = 1.
    This equals the value returned by ``evaluate_sequence`` for normalized a.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(conv) on the simplex.
    Returns (grad, cur_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


def _smooth_grad(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smoothed max using a log‑sum‑exp surrogate:
        f_eps(a) = eps * log Σ_i exp(conv_i/eps)
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    maxc = conv.max()
    w = np.exp((conv - maxc) / eps)
    w_sum = w.sum()
    w /= w_sum

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    diag_idx = np.arange(conv.size)[(np.arange(conv.size) % 2) == 0] // 2
    grad[diag_idx] -= w[2 * diag_idx] * a[diag_idx]

    return grad

def _smooth_adam_descent(a0: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         eps0: float = 5e-3,
                         step0: float = 0.25,
                         beta1: float = 0.9,
                         beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp smooth surrogate of max(conv).
    ``eps`` is halved every 30 000 iterations (but never below 1e‑9).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    eps = eps0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad = _smooth_grad(a, eps)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, 1.0)

        if t % 30_000 == 0:
            eps = max(eps * 0.5, 1e-9)

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         max_iters: int = 1_500_000,
                         eta0: float = 0.7) -> tuple[np.ndarray, float]:
    """Plain projected sub‑gradient descent with diminishing step‑size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        a = a - eta * grad
        a = _project_onto_simplex(a, 1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:

    return best_a, best_score

# Analytic families (including new ones)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _powerlaw_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+0.5)*(n‑i‑0.5))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.power((i + 0.5) * (n - i - 0.5), p)
    return vals / vals.sum()


def _edge_block_distribution(n: int, edge_frac: float, ratio: float) -> np.ndarray:
    """
    Two‑zone shape:
        * edge region width = edge_frac·n (both ends)
        * weight inside the edge region = ratio·interior weight
    interior weight = 1.0.
    """
    edge_len = max(1, int(round(n * edge_frac)))
    interior_len = n - 2 * edge_len
    if interior_len < 0:
        # degenerate case – fall back to pure edge
        edge_len = n // 2
        interior_len = 0
    a = np.empty(n, dtype=np.float64)
    a[:edge_len] = ratio
    a[-edge_len:] = ratio
    if interior_len > 0:
        a[edge_len:n - edge_len] = 1.0
    a = a / a.sum()
    return a


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_w = (p - 1) * (np.log(x) + np.log(1.0 - x))
    log_w = np.clip(log_w, -700, 700)
    w = np.exp(log_w)
    if w.sum() == 0.0:
        return np.full_like(w, 1.0 / n)
    return w / w.sum()


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    """Dirichlet draw with concentration α."""
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return x / x.sum()
