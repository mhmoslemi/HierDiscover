def _mirror_descent(a, deadline):
    n = a.shape[0]
    eps = 1e-12
    schedule = [
        (1.0, 100.0, 5000, 0.3, 0.03),
        (100.0, 1000.0, 8000, 0.03, 0.003),
        (1000.0, 20000.0, 12000, 0.003, 0.0003),
        (20000.0, 200000.0, 15000, 0.0003, 0.00003),
    ]
    for alpha0, alpha1, steps, lr0, lr1 in schedule:
        for it in range(steps):
            if time.time() >= deadline:
                return a
            t = it / steps
            alpha = alpha0 * (alpha1 / alpha0) ** t
            lr = lr0 * (lr1 / lr0) ** t
            conv = _fft_self_convolve(a)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            grad = 2.0 * _fft_convolve(a[::-1], w)[n-1:n-1+n]
            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()
    return a

def _subgradient_descent(a, deadline):

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    it = 0
    best_local_score = _max_convolution_score(a)
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)
        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n-1:n-1+n]
        # update momentum
        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** it)
        # line search
        step = step0
        while True:
            a_candidate = a - step * v_hat
            a_candidate = _project_onto_simplex(a_candidate, 1.0)
            a_candidate = np.maximum(a_candidate, eps)
            a_candidate /= a_candidate.sum()
            cand_sc = _max_convolution_score(a_candidate)
            if cand_sc < best_local_score - 1e-12 or step < 1e-8:
                break
            step *= 0.5
        a = a_candidate

        # occasional jitter

            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:

def _hillclimber(a, deadline):
    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    while time.time() < deadline:
        it += 1
        # locate the index of max conv
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # find pair (i,j) maximizing a[i]*a[j]
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.random.randint(0, n)
        if target == source:
            continue
        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * np.random.rand()
        if delta <= eps:
            continue
        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
    return a

"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.  The search proceeds in four stages:

1. **Broad exploration** – for several lengths we create many analytic
   families (uniform, triangular, exponential, symmetric / general Beta,
   sine‑power, cosine‑window, Gaussian) and a few random Dirichlet draws.
   The best few candidates (by the exact objective 2·n·max(conv)) are kept.

2. **Smooth‑max mirror descent** – we minimise a softened version of the
   max‑convolution using an exponentiated‑gradient (mirror‑descent) update.
   The temperature (α) is annealed from ~1 up to ~2⋅10⁵ while the learning
   rate decays, giving a long, high‑precision optimisation of the shape.

3. **Projected sub‑gradient polishing** – a true sub‑gradient of the
   max‑convolution (active‑set gradient) is used together with an Adam‑style
   momentum estimate and a back‑tracking line‑search for rapid final descent.

4. **Hill‑climber (mass‑transfer)** – a cheap local search repeatedly moves a
   small amount of mass from a location that contributes most to the current
   peak towards a random target, accepting only improvements.  Occasional
   jitter avoids stagnation.

All stages respect a global time budget (default 1000 s).  The best
distribution found is returned as a plain list of floats.  The code
makes no assumptions on the length of the sequence – several lengths are
tried and the best result overall is output.
"""

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a):
    """Full linear self‑convolution a * a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a):
    """
    Exact objective 2·n·max(a * a) for a probability vector (sum = 1).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_general(n, a, b):
    """General (non‑symmetric) Beta(a,b) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1 - x)
    w = np.exp(logw)

    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    """Discrete Gaussian centred at (n‑1)/2 with relative width sigma."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _generate_candidates(n):
    """Return a list of candidate probability vectors of length n."""
    candidates = []

    # Analytic families
    candidates.append(_distribution_uniform(n))
    candidates.append(_distribution_triangular(n))
    candidates.append(_distribution_exponential(n, lam=0.003))
    candidates.append(_distribution_exponential(n, lam=0.02))

    # Symmetric Beta
    for p in (0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        candidates.append(_distribution_beta_sym(n, p))

    # General Beta grid (a,b)
    for a in (0.05, 0.1, 0.2, 0.5, 1.0):
        for b in (0.05, 0.1, 0.2, 0.5, 1.0):
            candidates.append(_distribution_beta_general(n, a, b))

    # Sine‑power family
    for pw in (0.5, 1.0, 2.0, 4.0, 8.0, 12.0):
        candidates.append(_distribution_sine_power(n, pw))

    # Cosine windows
    for coeff in (0.3, 0.5, 0.7, 0.9):
        candidates.append(_distribution_cosine_window(n, coeff))

    # Gaussian windows
    for sigma in (0.12, 0.18, 0.25, 0.35, 0.45):
        candidates.append(_distribution_gaussian(n, sigma))

    # Random Dirichlet draws (different concentration)
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0):
        candidates.append(np.random.dirichlet(np.full(n, alpha)))

    # Sparse random start (few non‑zeros)
    sparse = np.zeros(n, dtype=np.float64)
    k = max(2, n // 300)
    idx = np.random.choice(n, k, replace=False)
    sparse[idx] = np.random.rand(k)
    sparse /= sparse.sum()
    candidates.append(sparse)

    return candidates

def _mirror_descent(a, deadline):
    """Smooth‑max (exponentiated‑gradient) mirror descent."""
    n = a.shape[0]
    eps = 1e-12
    schedule = [
        (1.0,   100.0,   5000, 0.30, 0.03),
        (100.0, 1000.0,  8000, 0.03, 0.003),
        (1000.0, 20000.0,12000,0.003,0.0003),
        (20000.0,200000.0,15000,0.0003,0.00003),
    ]
    for alpha0, alpha1, steps, lr0, lr1 in schedule:
        for it in range(steps):
            if time.time() >= deadline:
                return a
            t = it / steps
            alpha = alpha0 * (alpha1 / alpha0) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()
    return a


def _subgradient_descent(a, deadline):
    """Projected Adam‑style sub‑gradient descent on the true max‑convolution."""
    n = a.shape[0]
    eps = 1e-12
    velocity = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    step0 = 0.5
    it = 0
    best_local_score = _max_convolution_score(a)

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam‑style momentum
        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** it)

        # back‑tracking line search
        step = step0
        while True:
            a_candidate = a - step * v_hat
            a_candidate = _project_onto_simplex(a_candidate, 1.0)
            a_candidate = np.maximum(a_candidate, eps)
            a_candidate /= a_candidate.sum()
            cand_sc = _max_convolution_score(a_candidate)
            if cand_sc < best_local_score - 1e-12 or step < 1e-8:
                break
            step *= 0.5

        a = a_candidate
        if cand_sc < best_local_score - 1e-12:
            best_local_score = cand_sc

        # occasional jitter to escape plateaus
        if it % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    return a


def _hillclimber(a, deadline):
    """Simple mass‑transfer hill‑climber that only accepts improvements."""
    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    while time.time() < deadline:
        it += 1
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # locate the pair (i,j) that gives the largest product contribution
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M

        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step function that minimises evaluate_sequence."""
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin
    eps = 1e-12

    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1. Broad exploration over several lengths
    lengths = [2500, 3000, 3500, 4000, 5000, 8000]
    pool = []                     # keep the best few candidates overall
    pool_size = 10

    for n in lengths:
        if time.time() >= deadline:
            break
        for vec in _generate_candidates(n):
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            # insert into the global pool of best candidates
            if len(pool) < pool_size:
                pool.append((sc, vec.copy()))
                pool.sort(key=lambda x: x[0])
            else:
                if sc < pool[-1][0]:
                    pool[-1] = (sc, vec.copy())
                    pool.sort(key=lambda x: x[0])
            # update global best for logging
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] init n={n} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Incorporate any previously supplied sequence
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = len(prev)
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            pool.append((sc, prev.copy()))
            pool.sort(key=lambda x: x[0])
            if len(pool) > pool_size:
                pool = pool[:pool_size]
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never be needed)
    if best_seq is None:
        n = 4000
        best_seq = _distribution_uniform(n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        best_n = n
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Refine each candidate from the pool
    for idx, (init_sc, a0) in enumerate(pool):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] polishing candidate {idx+1}/{len(pool)} start={init_sc:.6f}")
        a = a0.copy()
        a = _mirror_descent(a, deadline)
        a = _subgradient_descent(a, deadline)
        a = _hillclimber(a, deadline)
        final_sc = _max_convolution_score(a)
        print(f"[{time.time():.1f}] candidate {idx+1} final_score={final_sc:.6f}")
        if final_sc < best_score:
            best_score = final_sc
            best_seq = a.copy()
            best_n = a.shape[0]
            print(f"[{time.time():.1f}] NEW GLOBAL BEST {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final sanitisation and return
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()
    print(f"Return best score ≈ {best_score:.6f} (length={best_n})")
    return best_seq.tolist()
