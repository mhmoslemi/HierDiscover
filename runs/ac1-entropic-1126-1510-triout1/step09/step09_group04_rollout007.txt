def _smooth_max_adam_optimize(a, max_iters, time_deadline):

    n = a.shape[0]

    logp = np.log(a + 1e-20)   # initial parameters x.

    m = np.zeros_like(logp)

    v = np.zeros_like(logp)

    beta1, beta2 = 0.9, 0.999

    eps = 1e-8

    # schedule parameters
    alpha0 = 0.1
    alpha_max = 1e5
    lr0 = 0.2
    lr_end = 0.001

    step = 0

    best_local_seq = a.copy()
    best_local_score = _eval_score(a)

    while step < max_iters and time.time() < time_deadline:

        t = step / max_iters

        alpha = alpha0 * (alpha_max / alpha0) ** t

        lr = lr0 * (lr_end / lr0) ** t

        # compute current a via softmax to ensure sum=1
        # enforces a = exp(logp)/sum(exp(logp))

        # Use stable softmax
        max_logp = np.max(logp)
        exp_shift = np.exp(logp - max_logp)
        a = exp_shift / np.sum(exp_shift)

        conv = _fft_self_convolve(a)
        # compute smooth max weights
        # w_i = exp(alpha * (conv_i - conv_max)) normalized.
        conv_max = conv.max()

        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t a
        grad_a = 2.0 * _fft_convolve(a[::-1], w)[n-1 : n-1+n]

        # compute grad in log-space: x gradient = a * (grad_a - avg)
        avg = np.dot(a, grad_a)

        # Adam update
        step += 1
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)
        m_hat = m / (1 - beta1 ** step)

"""
Propose a step‑function (non‑negative list) that minimises `evaluate_sequence`.

Algorithm
---------
1. Load an existing solution (`height_sequence_1`) if it exists.
2. Choose a few candidate lengths (the current best length and ±500).
   For each length we obtain an initial distribution by resizing the best
   known vector (or using a uniform distribution).
3. For each initial vector run a **smooth‑max Adam optimisation**:
   - Represent the vector as a soft‑max of free parameters `logp`
     (automatically stays on the simplex).
   - At iteration `t` use a smoothing parameter `α(t)` that grows from
     0.1 to 1e5, turning the smooth‑max into the true max.
   - Compute the weighted self‑convolution and its gradient
     (the same expression used in the reference code).
   - Apply Adam on the log‑probabilities with a learning‑rate schedule
     (`lr(t)` from 0.2 → 0.001).  This performs a projected mirror‑descent
     on the simplex and finds a lower bound on the maximum self‑convolution.
4. Keep the best vector seen across all lengths and all optimisation runs.
5. A cheap **hill‑climber** (mass‑transfer moves) is applied for a short
   final polishing phase.
6. Return the best sequence as a plain list of floats.

The whole procedure respects a wall‑clock budget of `budget_s` seconds
(the default is 1000 s).  Progress is printed to the console.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _eval_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at (n‑1)/2."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _resize_vector(a: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a probability vector by linear interpolation of its PDF."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    new_a = np.interp(new_x, old_x, a, left=0.0, right=0.0)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0:
        new_a = _distribution_uniform(new_n)
    else:
        new_a /= s
    return new_a


# ----------------------------------------------------------------------
def _smooth_max_adam_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 60000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate of max(a*a).

    Parameters
    ----------
    a0 : np.ndarray
        Starting probability vector (sum = 1, non‑negative).
    deadline : float
        Wall‑clock timeout (seconds since epoch).
    max_iters : int, optional
        Upper bound on iterations (the loop also stops by deadline).

    Returns
    -------
    best_a : np.ndarray
        The best vector encountered (sum = 1).
    best_score : float
        Corresponding value of 2·n·max(a*a).
    """
    eps = 1e-20
    n = a0.shape[0]

    # free parameters – log‑softmax
    logp = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(logp)
    v = np.zeros_like(logp)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # schedules
    alpha_start, alpha_end = 0.1, 1e5        # smooth‑max concentration
    lr_start, lr_end = 0.2, 0.001          # learning‑rate schedule

    best_a = a0.copy()
    best_score = _eval_score(a0)

    step = 0
    while time.time() < deadline and step < max_iters:
        step += 1

        # current probability vector from log‑softmax
        max_lp = np.max(logp)
        exp_shift = np.exp(logp - max_lp)
        a = exp_shift / np.sum(exp_shift)

        # current convolution & score
        conv = _fft_self_convolve(a)
        conv_max = conv.max()
        cur_score = 2.0 * n * conv_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            if step % 200 == 0:
                print(f"[{time.time():.1f}] Adam step={step}  best {cur_score:.6f}")

        # schedules (t ∈ [0,1])
        t = step / max_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t

        # smooth‑max weights (stable: max exponent = 0)
        w = np.exp(alpha * (conv - conv_max))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth‑max objective w.r.t. a
        grad_a = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # convert to gradient w.r.t. log‑softmax parameters
        avg = np.dot(a, grad_a)
        grad_logp = a * (grad_a - avg)

        # Adam update on log‑parameters
        m = beta1 * m + (1 - beta1) * grad_logp
        v = beta2 * v + (1 - beta2) * (grad_logp ** 2)
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        logp -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional jitter to escape plateaus
        if step % 1000 == 0:
            logp += np.random.randn(n) * 0.01

    return best_a, best_score


# ----------------------------------------------------------------------
def _hill_climber(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 25000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap random mass‑transfer hill‑climber that nudges the vector
    while preserving the simplex constraint.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()

    eps = 1e-12
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        k = int(conv.argmax())
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        # find the pair with largest product contributing to this lag
        best_i, best_j = i_low, k - i_low
        max_prod = a[best_i] * a[best_j]
        for i in range(i_low, i_high + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue
        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        snapshot = a.copy()
        a[source] -= delta
        a[target] += delta
        a = np.maximum(a, eps)
        a /= a.sum()
        conv_new = _fft_self_convolve(a)
        new_score = 2.0 * n * conv_new.max()
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a.copy()
            conv = conv_new
        else:
            a = snapshot
            conv = _fft_self_convolve(a)   # restore exact convolution

        if it % 5000 == 0:
            # small random jitter to avoid stagnation
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found within the allotted wall‑clock time.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 5.0   # small safety margin

    best_seq = None
    best_score = float('inf')
    best_n = None

    # --------------------------------------------------------------
    # 0.  Load a previously discovered sequence if available.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev /= prev.sum()
            sc = _eval_score(prev)
            best_seq = prev.copy()
            best_n = prev.size
            best_score = sc
            print(f"[{time.time():.1f}] loaded previous best length={best_n} score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1.  If nothing loaded, start from a uniform distribution of size ~3000.
    if best_seq is None:
        best_n = 3000
        best_seq = _distribution_uniform(best_n)
        best_score = _eval_score(best_seq)
        print(f"[{time.time():.1f}] start uniform length={best_n} score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2.  Define a small set of lengths to explore (current ±500).
    lengths = [best_n]
    for delta in (-500, 500):
        cand = best_n + delta
        if 200 <= cand <= 8000:
            lengths.append(cand)
    lengths = sorted(set(lengths))

    # --------------------------------------------------------------
    # 3.  For each length: initialise, run Adam optimisation, optionally hill‑climb.
    for n in lengths:
        if time.time() >= deadline:
            break
        # initialise a vector of length n
        if best_seq is not None and best_seq.size != n:
            init = _resize_vector(best_seq, n)
        else:
            init = best_seq.copy()
        init = np.maximum(init, 1e-12)
        init /= init.sum()

        # --- smooth‑max Adam optimisation ---------------------------------
        a_opt, score_opt = _smooth_max_adam_optimize(init, deadline)
        if score_opt < best_score - 1e-12:
            best_seq = a_opt.copy()
            best_n = n
            best_score = score_opt
            print(f"[{time.time():.1f}] improved after Adam length={n} score={score_opt:.6f}")

        # --- final cheap hill‑climber ------------------------------------
        if time.time() < deadline:
            a_hc, score_hc = _hill_climber(best_seq, deadline)
            if score_hc < best_score - 1e-12:
                best_seq = a_hc.copy()
                best_n = best_seq.size
                best_score = score_hc
                print(f"[{time.time():.1f}] improved after hill‑climb length={best_n} score={score_hc:.6f}")

    # --------------------------------------------------------------
    # 4.  Final sanity check and output.
    final_score = _eval_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.size})")
    return best_seq.tolist()
