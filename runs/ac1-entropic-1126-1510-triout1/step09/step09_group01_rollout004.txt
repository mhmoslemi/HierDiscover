"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.  The search consists of three phases:

1. **Broad exploration** – generate a handful of analytic families
   (uniform, arcsine, triangular, exponential, beta, sine‑power,
   cosine window, Gaussian) for several lengths.  The best normalised
   candidate is kept.

2. **Logit‑Adam optimisation of a smoothed max‑convolution** – we
   represent the vector as a softmax of unconstrained logits ``u``.
   The objective is the smooth approximation
   `f_α(a)= (1/α)·log Σ exp(α·conv(a))`.  Adam updates are performed on
   the logits while the smoothing parameter ``α`` is annealed from 1
   to 2·10⁵ and the learning‑rate from 0.5 to 5·10⁻⁴.  This refines the
   distribution far beyond the earlier exponentiated‑gradient scheme.

3. **Simulated‑annealing polishing** – small random mass‑transfer moves
   are accepted with a Metropolis criterion.  Temperature is linearly
   cooled.  Finally a targeted hill‑climber (mass transfer away from the
   pair responsible for the current maximal convolution entry) finishes
   the optimisation.

All stages respect a total wall‑clock budget (default 1000 s).  The
function returns the best sequence found as a plain ``list[float]``.
"""

# Helper utilities
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    For a vector a already normalised to sum = 1,
    the exact value returned by evaluate_sequence is
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families (all return a vector of length n with sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n: int, coeff: float = 0.5) -> np.ndarray:
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` within the given time budget.

    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin
    eps = 1e-12

    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1. Broad exploration over several lengths and families
    lengths = [2000, 4000, 8000, 15000, 25000]
    for n in lengths:
        if time.time() >= deadline:
            break

        candidates = []

        # Analytic families (already normalised)
        candidates.append(_distribution_uniform(n))
        candidates.append(_distribution_arcsine(n))
        candidates.append(_distribution_triangular(n))
        candidates.append(_distribution_exponential(n, lam=0.003))
        candidates.append(_distribution_exponential(n, lam=0.02))

        # Symmetric Beta with a dense grid of shape parameters
        for p in np.geomspace(0.02, 5.0, num=20):
            candidates.append(_distribution_beta_sym(n, float(p)))

        # Sine‑power family
        for pw in (0.5, 1.0, 2.0, 4.0, 8.0, 12.0):
            candidates.append(_distribution_sine_power(n, pw))

        # Cosine window variants
        for coeff in (0.3, 0.5, 0.7, 0.9):
            candidates.append(_distribution_cosine_window(n, coeff))

        # Gaussian variants
        for sigma in (0.12, 0.18, 0.25, 0.35, 0.45):
            candidates.append(_distribution_gaussian(n, sigma))

        # Random Dirichlet draws (different concentration)
        for alpha in (0.2, 0.5, 1.0, 2.0):
            candidates.append(np.random.dirichlet(np.full(n, alpha)))

        # Sparse random start (few non‑zeros)
        sparse = np.zeros(n, dtype=np.float64)
        k = max(2, n // 300)
        idx = np.random.choice(n, k, replace=False)
        sparse[idx] = np.random.rand(k)
        sparse /= sparse.sum()
        candidates.append(sparse)

        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] n={n} new init best {sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Incorporate a previously discovered sequence if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_seq = prev.copy()
                best_n = prev.size
                best_score = sc
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never trigger)
    if best_seq is None:
        best_n = 4000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Logit‑Adam optimisation of a smoothed max‑convolution
    a = best_seq.copy()
    n = a.shape[0]
    u = np.log(a + eps)                     # logits, unconstrained
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(u)
    v = np.zeros_like(u)

    # we run a bounded number of iterations; the loop also stops at the time deadline
    it = 0
    max_it = 20000                          # enough for a solid refinement
    while time.time() < deadline and it < max_it:
        it += 1
        # schedules (geometric/exponential interpolation)
        frac = min(it / max_it, 1.0)
        alpha = 1.0 * (2e5) ** frac                # smoothness → 2·10⁵
        lr = 0.5 * (5e-4) ** frac                  # learning‑rate → 5·10⁻⁴

        conv = _fft_self_convolve(a)
        maxc = conv.max()

        # soft‑max weights (stable because conv‑maxc ≤ 0)
        shifted = alpha * (conv - maxc)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # gradient of the smooth‑max (vector of length n)
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam update on logits
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        u = u - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # softmax to obtain a (automatically on the simplex)
        u_max = np.max(u)
        exp_u = np.exp(u - u_max)
        a = exp_u / exp_u.sum()
        a = np.maximum(a, eps)
        a /= a.sum()

        if it % 500 == 0 or time.time() >= deadline - 5:
            cur_sc = _max_convolution_score(a)
            if cur_sc < best_score - 1e-12:
                best_score = cur_sc
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={it} new best {cur_sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Simulated‑annealing polishing
    a = best_seq.copy()
    cur_score = best_score
    # temperature schedule
    T_start = 1e-3
    T_end = 1e-6
    max_sa_iters = 150000
    sa_deadline = time.time() + (deadline - time.time()) * 0.5
    it_sa = 0
    while time.time() < sa_deadline and it_sa < max_sa_iters:
        it_sa += 1
        frac = it_sa / max_sa_iters
        T = T_start * (1 - frac) + T_end * frac

        # move a modest amount of probability mass away from the current largest entry
        source = int(np.argmax(a))
        target = random.randint(0, n - 1)
        if target == source:
            continue
        delta = min(a[source] * 0.1, 0.05) * random.random()
        if delta <= eps:
            continue

        a_new = a.copy()
        a_new[source] -= delta
        a_new[target] += delta
        a_new = np.maximum(a_new, eps)
        a_new /= a_new.sum()

        new_sc = _max_convolution_score(a_new)
        diff = new_sc - cur_score
        if diff < 0 or random.random() < math.exp(-diff / T):
            a = a_new
            cur_score = new_sc
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] SA iter={it_sa} new best {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final targeted hill‑climber (mass‑transfer from the most offending pair)
    a = best_seq.copy()
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    it_hc = 0
    while time.time() < deadline:
        it_hc += 1
        # locate the convolution index attaining the current maximum
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # find the pair (i,j) inside the active window with the largest product a[i]*a[j]
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        # choose the larger probability of the pair as the source
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        delta = min(a[source] * 0.2, 0.03) * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()

        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M
            cur_score = 2.0 * n * M
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Hill iter={it_hc} new best {cur_score:.6f}")

        # occasional jitter to prevent stagnation
        if it_hc % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

    final_score = 2.0 * n * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
