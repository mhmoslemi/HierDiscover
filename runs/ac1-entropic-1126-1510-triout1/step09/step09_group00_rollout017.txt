assistantfinal**Solution Overview**

The objective is  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2},
\qquad a_i\ge 0 .
\]

Because the formula is *scale‑invariant* we may (and will) keep the
vector normalised to `sum = 1`.  Then  

\[
F(a)=2\,n\;\max (a*a) .
\]

The problem is non‑convex, but a very good family of candidates is the
symmetric Beta distribution  

\[
a_i\propto x_i^{p-1}\,(1-x_i)^{p-1},\qquad
x_i=\frac{i+0.5}{n},
\]

which for \(p=\tfrac12\) equals the classical *arcsine* distribution.
Scanning this one–parameter family with a modest size `n≈2000`
finds the best shape parameter `p`.  The same shape can be transferred
to a much larger `n` (e.g. 20000) – the value of the objective depends
almost only on the shape, not on `n`.

Starting from the best Beta‑shape we run a **mirror (exponentiated)
sub‑gradient descent** on the exact maximum–convolution objective.
The sub‑gradient for a given maximal index `k` is simply the reversed
vector shifted by `k`.  When several indices share the maximal value we
average their sub‑gradients.  After each update the vector is projected
onto the probability simplex (Euclidean projection).

Finally a cheap **pairwise‑swap local search** (randomly moving a small
amount of mass from one position to another) is performed for the
remaining time budget.

All steps respect the global time limit; progress is printed so the
run can be monitored.

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()       # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    """Beta(0.5,0.5) – the classical arcsine shape."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    if p <= 0.0:
        p = 1e-12
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((p - 1) * np.log(np.clip(x, eps, 1.0)) +
               (p - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Sparse random start – useful for explorative restarts."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float,
                         lr0: float = 0.7,
                         max_iter: int = 15000,
                         eval_interval: int = 200) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the true max‑convolution
    objective.  Handles ties by averaging the sub‑gradients.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        idxs = np.where(conv == max_val)[0]

        # averaged sub‑gradient
        grad = np.zeros_like(a)
        for idx in idxs:
            ind = idx - np.arange(n)
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * a[ind[mask]]
        grad /= len(idxs)                       # average if several maxima

        eta = lr0 / math.sqrt(t)               # classic 1/√t schedule
        a = a * np.exp(-eta * grad)            # exponentiated‑gradient step
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s                           # stay on the simplex

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad it={t} new best {best_score:.6f}")

    return best_seq, best_score

def _pairwise_swap_local(a0: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """
    Tiny simulated‑annealing walk: move a random amount of mass from one
    coordinate to another and keep the move only if the true objective
    improves.
    """
    a = a0.copy()
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # move at most 30 % of the source mass
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The algorithm:

    1. Scan the 1‑parameter family Beta(p,p) on a modest problem size and
       keep the best few shape parameters.
    2. Transfer the best shape(s) to a larger problem size (≈20 k entries)
       – the objective is essentially independent of the length.
    3. Refine each promising candidate with mirror‑descent on the exact
       max‑convolution objective.
    4. Finish with a cheap pairwise‑swap local optimiser.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0            # safety margin

    # ------------------------------------------------------------------
    # 1. coarse scan on a small n
    n_small = 2000
    p_values = np.geomspace(0.02, 1.0, num=45)        # from very U‑shaped to uniform
    small_candidates = []

    print(f"[{time.time():.1f}] scanning Beta(p,p) family (n={n_small})")
    for p in p_values:
        if time.time() >= deadline:
            break
        vec = _distribution_beta_sym(n_small, float(p))
        score = _objective(vec)
        small_candidates.append((score, p, vec))
        # keep a few best seeds (we will refine later)
    # also add arcsine and uniform explicitly
    for name, vec in (("arcsine", _distribution_arcsine(n_small)),
                      ("uniform", _distribution_uniform(n_small))):
        score = _objective(vec)
        small_candidates.append((score, name, vec))

    # sort and keep the best 7 distinct shapes
    small_candidates.sort(key=lambda x: x[0])
    top_shapes = []
    seen = set()
    for sc, key, vec in small_candidates:
        if isinstance(key, str):
            ident = key
        else:
            ident = round(key, 5)                     # numeric shape identifier
        if ident in seen:
            continue
        seen.add(ident)
        top_shapes.append((sc, ident, vec))
        if len(top_shapes) >= 7:
            break

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 2. upscale each promising shape to a large n (≈20 k)
    n_big = 20000
    print(f"[{time.time():.1f}] up‑scaling to n={n_big}")
    big_candidates = []
    for sc_small, ident, _ in top_shapes:
        if time.time() >= deadline:
            break
        if isinstance(ident, str) and ident == "uniform":
            vec_big = _distribution_uniform(n_big)
        else:
            # interpret ident as a float p when possible
            try:
                p = float(ident)
            except Exception:
                p = 0.5
            vec_big = _distribution_beta_sym(n_big, p)
        sc_big = _objective(vec_big)
        big_candidates.append((sc_big, ident, vec_big))
        if sc_big < overall_best_score:
            overall_best_score = sc_big
            overall_best_seq = vec_big.copy()
            print(f"[{time.time():.1f}] new global best from shape {ident} score={sc_big:.6f}")

    # keep a small set of the very best seeds for refinement
    big_candidates.sort(key=lambda x: x[0])
    refinement_seeds = [seq for _, _, seq in big_candidates[:3]]

    # ------------------------------------------------------------------
    # 3. mirror‑descent (sub‑gradient) on the selected seeds
    remaining = deadline - time.time()
    if remaining > 60 and refinement_seeds:
        slice_time = remaining * 0.7 / max(1, len(refinement_seeds))
        for seed_vec in refinement_seeds:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + slice_time
            overall_best_seq, overall_best_score = _subgradient_descent(
                seed_vec,
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.7,
                max_iter=15000,
                eval_interval=200,
            )
            print(f"[{time.time():.1f}] after sub‑grad best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. final cheap local polishing
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq,
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] final best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # return the best sequence found (as a plain Python list)
    print(f"Finished – best objective {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
