that minimises ``evaluate_sequence``.

The algorithm is a refined version of the previous approach:

*   Work with larger problem sizes (≈20 k coefficients) – larger n gives a
    lower theoretical lower bound (≈1) and more degrees of freedom.
*   Keep the sequence symmetric (a[i]=a[-i‑1]), which both respects the
    structure of the optimal solution and lets us reuse the simple
    sub‑gradient formula from the reference code.
*   Sample a rich set of analytic families (arcsine, beta, raised‑cosine,
    sinusoidal, exponential, power‑law) together with symmetric Dirichlet
    draws.
*   Keep the K best seeds and improve them with three optimisers:
    – Adam on the exact sub‑gradient of ``max(conv(a,a))`` (projected onto
      the simplex and re‑symmetrised each step);

    – Adam on a smooth Lp‑approximation of the max (p=8 gives a tighter
      approximation than p=4).
*   The best seed is polished further by a Frank‑Wolfe style mass‑transfer
    move and a lightweight simulated‑annealing phase.
*   All loops respect the overall 1000 s budget; the best sequence seen

The implementation is completely self‑contained, uses only NumPy,

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Force symmetry a[i] = a[-i‑1] (average with its reverse)."""
    return (a + a[::-1]) * 0.5


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Surrogate objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) on the simplex (∑a=1) for a **symmetric**
    sequence. Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # For symmetric a the two convolutions are identical;
    # we compute only one and double it, then correct the diagonal.
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices (self‑product terms)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.15,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 300_000):
    """Adam on the true sub‑gradient (max‑convolution) with symmetry enforcement."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        a = _symmetrize(a)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          eta0: float = 0.6,
                          max_iters: int = 250_000):
    """Exponentiated‑gradient (KL‑mirror) descent with symmetry enforcement."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        a = _symmetrize(a)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _smooth_obj_and_grad(a: np.ndarray, p: float = 8.0):
    """
    Smooth Lp‑approximation of max(conv(a,a)) and its gradient.
    Returns ``(value, grad)`` where ``value = (∑ conv^p)^{1/p}``.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)                # length 2n‑1
    b_pow = b ** p
    S = b_pow.sum()
    if S == 0.0:
        return 0.0, np.zeros_like(a)
    val = S ** (1.0 / p)
    factor = S ** (1.0 / p - 1.0)            # = S^{(1‑p)/p}
    w = factor * (b ** (p - 1))              # length 2n‑1
    # For symmetric a we can reuse the same “double‑conv” trick.
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    # diagonal correction (same as in _subgrad)
    even_mask = ((np.arange(2 * n - 1) % 2) == 0)
    diag_idx = (np.arange(2 * n - 1)[even_mask] // 2)
    # Only keep those within the original range
    valid = diag_idx < n
    if valid.any():
        grad[diag_idx[valid]] -= a[diag_idx[valid]] * (p - 1) * (b[diag_idx[valid]] ** (p - 2)) * factor
    return val, grad


def _adam_smooth_descent(a0: np.ndarray, deadline: float,
                         p: float = 8.0,
                         step0: float = 0.15,
                         beta1: float = 0.9,
                         beta2: float = 0.999,
                         max_iters: int = 300_000):
    """
    Adam on the smoothed Lp objective (p≥2).  Symmetry is enforced after each step.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        _, grad = _smooth_obj_and_grad(a, p)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        a = _symmetrize(a)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet vector (size n) with concentration ``alpha``."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


def _random_symmetric_dirichlet(n: int, alpha: float) -> np.ndarray:
    """Draw a symmetric Dirichlet vector (size n) with concentration ``alpha``."""
    x = _random_dirichlet_alpha(n, alpha)
    a = (x + x[::-1]) * 0.5
    a /= a.sum()
    return a

# Analytic families (all produce symmetric vectors)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _sinusoidal_distribution(n: int, power: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n) ** power
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _raised_cosine_distribution(n: int, power: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = (1.0 - np.cos(2.0 * np.pi * (i + 0.5) / n)) ** power
    vals /= vals.sum()
    return vals

            (lambda m: _exponential_distribution(m, lam=0.005), "exp0.005"),
            (lambda m: _sinusoidal_distribution(m, power=1.0), "sinus1"),
            (lambda m: _sinusoidal_distribution(m, power=2.0), "sinus2"),
            (lambda m: _raised_cosine_distribution(m, power=1.0), "raised_cos1"),
        ]
        for gen, name in fams:
            a = gen(n)
            candidates.append((a, _max_convolution_score(a)))
            print(f"[{time.time():.1f}] n={n:<5d} {name:12s}")

        # exponential with a few λ values
        for lam in (0.001, 0.005, 0.02):
            a = _exponential_distribution(n, lam=lam)
            candidates.append((a, _max_convolution_score(a)))
            print(f"[{time.time():.1f}] n={n:<5d} exponential(lam={lam})")

        # symmetric beta (grid)
        for p in beta_ps:
            a = _beta_symmetric_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))

        # asymmetric beta (a few hand‑picked pairs – still symmetrised later)
        for a_alpha, a_beta in [(0.3, 0.7), (0.5, 0.5), (0.7, 0.3),
                                (0.2, 1.0), (1.0, 0.2)]:
            a = _beta_asymmetric_distribution(n, a_alpha, a_beta)
            candidates.append((a, _max_convolution_score(a)))

        # random symmetric beta
        for _ in range(3):
            p = 10 ** np.random.uniform(-1.5, 0.5)   # between ~0.03 and ~3.16
            a = _beta_symmetric_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))

        # power‑law families
        for p in power_ps:
            a = _power_law_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))

        # random symmetric Dirichlet draws
        for alpha in dirichlet_alphas:
            a = _random_symmetric_dirichlet(n, alpha)
            candidates.append((a, _max_convolution_score(a)))

    # ------------------------------------------------------------------
    # 3️⃣ Add previous best if available (symmetrised)
    if prev_seq is not None:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.ndim == 1 and arr.size > 0:
                arr = _symmetrize(arr)
                arr = arr / arr.sum()
                candidates.append((arr, _max_convolution_score(arr)))
                print(f"[{time.time():.1f}] added previous best, score={candidates[-1][1]:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 4️⃣ Keep the top K seeds
    candidates.sort(key=lambda x: x[1])
    top_k = min(20, len(candidates))
    top_seeds = candidates[:top_k]
    print(f"Generated {len(candidates)} candidates, keeping top {top_k} for refinement.")

    best_seq, best_score = top_seeds[0][0].copy(), top_seeds[0][1]
    print(f"Initial best surrogate score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ Refinement of each seed
    for idx, (seed_seq, _) in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        n = seed_seq.shape[0]

        # 5a) Adam on exact sub‑gradient
        seq_adam, sc_adam = _adam_descent(seed_seq, deadline,
                                          step0=0.18, max_iters=30_000)
        if sc_adam < best_score - 1e-14:
            best_seq, best_score = seq_adam.copy(), sc_adam
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} Adam improved to {best_score:.6f}")

        # 5b) Exponentiated‑gradient descent (short run)
        seq_eg, sc_eg = _exp_gradient_descent(seed_seq, deadline,
                                              eta0=0.7, max_iters=20_000)
        if sc_eg < best_score - 1e-14:
            best_seq, best_score = seq_eg.copy(), sc_eg
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} ExpGrad improved to {best_score:.6f}")

        # 5c) Adam on smooth Lp surrogate (p=8)
        seq_sad, sc_sad = _adam_smooth_descent(seed_seq, deadline,
                                                p=8.0, step0=0.15,
                                                max_iters=30_000)
        if sc_sad < best_score - 1e-14:
            best_seq, best_score = seq_sad.copy(), sc_sad
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} SmoothAdam improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣ Frank‑Wolfe mass‑transfer polishing (more iterations)
    a = best_seq.copy()
    cur_score = best_score
    fw_iter = 0
    while time.time() < deadline and fw_iter < 100_000:
        fw_iter += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))   # mass to move away from
        i_dst = int(np.argmin(grad))   # mass to move towards
        if i_src == i_dst:
            break
        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-14:
            break

        # try full transfer
        a_full = a.copy()
        a_full[i_src] -= max_transfer
        a_full[i_dst] += max_transfer
        score_full = _max_convolution_score(a_full)

        improved = False
        if score_full < cur_score - 1e-13:
            a = a_full
            cur_score = score_full
            improved = True
        else:
            # try half transfer
            half = max_transfer * 0.5
            a_half = a.copy()
            a_half[i_src] -= half
            a_half[i_dst] += half
            score_half = _max_convolution_score(a_half)
            if score_half < cur_score - 1e-13:
                a = a_half
                cur_score = score_half
                improved = True

        if improved:
            if cur_score < best_score - 1e-14:
                best_seq, best_score = a.copy(), cur_score
                print(f"[{time.time():.1f}] FW iter={fw_iter} new best {best_score:.6f}")
            continue

        # occasional jitter to escape tiny plateaus
        if fw_iter % 5000 == 0 and np.random.rand() < 0.2:
            jitter = np.random.randn(a.size) * 2e-5
            a = _project_onto_simplex(a + jitter, s=1.0)
            a = _symmetrize(a)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    # 7️⃣ Light simulated‑annealing / jitter phase
    a = best_seq.copy()
    cur_score = best_score
    temp_start = 0.02
    temp_end = 0.001
    anneal_iter = 0
    while time.time() < deadline and anneal_iter < 150_000:
        anneal_iter += 1
        remaining = deadline - time.time()
        total = budget_s - 0.5
        t_frac = remaining / total
        temp = temp_end + (temp_start - temp_end) * t_frac

        i = np.random.randint(a.size)
        j = np.random.randint(a.size - 1)
        if j >= i:
            j += 1
        max_delta = min(a[i], 1.0 - a[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * temp

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _symmetrize(a_new)                 # keep symmetry
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if cur_score < best_score - 1e-14:
                best_seq, best_score = a.copy(), cur_score
                print(f"[{time.time():.1f}] Anneal iter={anneal_iter} new best {best_score:.6f}")
        else:
            prob = np.exp(-(new_score - cur_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                a = a_new
                cur_score = new_score

    # ------------------------------------------------------------------
    print(f"Final best surrogate score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
