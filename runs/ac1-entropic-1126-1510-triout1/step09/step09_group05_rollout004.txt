def lp_step(f, M):
    n = len(f)
    K = 2*n-1
    # Build A matrix; use scipy.sparse.lil_matrix for building then convert.
    rows = []
    cols = []
    data = []
    for k in range(K):
        # i from max(0, k-(n-1)) to min(k, n-1). Then j = k - i.
        i_min = max(0, k - (n-1))
        i_max = min(k, n-1)
        # for each i, coeff is f[i] for variable g_{k-i}
        for i in range(i_min, i_max+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    # After loops, construct csr.
    from scipy.sparse import csr_matrix
    A = csr_matrix((data, (rows, cols)), shape=(K, n))
    # b_ub = M * np.ones(K)
    # c = -np.ones(n)
    # solve
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(K, M), bounds=[(0, None)] * n, method='highs')
    if not res.success:
        return None
    g = res.x
    return g

def _build_constraint_matrix(f):
    n = len(f)
    K = 2*n - 1
    total = n * n
    rows = np.empty(total, dtype=np.int32)

def _lp_step(f, M):
    A = _build_constraint_matrix(f)
    K = A.shape[0]
    c = -np.ones(f.shape[0])
    b = np.full(K, M)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * f.shape[0], method='highs')
    if not res.success:
        return None
    return res.x

"""
Propose a step‑function (non‑negative list) that minimizes

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of three stages:

1️⃣  Analytic seeding – a handful of simple families (arcsine, beta,
    Gaussian, Cauchy) are generated on a modest length (≈2000).  The
    cheap score 2·n·max(conv) (sum is 1) is used to keep the few best
    seeds.

2️⃣  Linear‑programming refinement – for a current distribution f we
    solve

        max  Σ g_j
        s.t. f * g ≤ M  (component‑wise)
             g ≥ 0,

    where M = max(f * f).  The solution g is feasible for the same
    bound M, its total mass Σ g_j is typically tiny (≈10⁻²).  Updating

        f ← (1 – Σg)·f + g

    preserves Σ f = 1 and strictly reduces M.  Repeating this step a
    few times yields a dramatic drop in the objective.

3️⃣  Light polishing – a very fast hill‑climbing that repeatedly moves a
    small amount of mass from an “over‑weighted’’ index to an “under‑
    weighted’’ one (guided by the sub‑gradient).  Occasional jitter
    helps to escape flat plateaus.

All stages respect the global time budget (default 1000 s).  The
function returns the best sequence (as plain Python floats) found before
the deadline.
"""

from scipy.optimize import linprog

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap score = 2·n·max(a*a) for a probability vector (Σa=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_w = (p - 1) * (np.log(x) + np.log(1.0 - x))
    log_w = np.clip(log_w, -700, 700)
    w = np.exp(log_w)
    if w.sum() == 0.0:
        return np.full_like(w, 1.0 / n)
    return w / w.sum()


def _gaussian_distribution(n: int, sigma_frac: float = 0.30) -> np.ndarray:
    sigma = sigma_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if vals.sum() == 0.0:
        return np.full_like(vals, 1.0 / n)
    return vals / vals.sum()


def _cauchy_distribution(n: int, scale_frac: float = 0.30) -> np.ndarray:
    scale = scale_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = 1.0 / (1.0 + ((i - centre) / scale) ** 2)
    return vals / vals.sum()


# ----------------------------------------------------------------------
def _build_constraint_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build the (2n‑1) × n matrix A for the constraints  (f * g)_k ≤ M.
    Row k contains the coefficients f[i] for the variable g_{k‑i}
    (i runs over all indices where the term is defined).
    """
    n = f.shape[0]
    K = 2 * n - 1
    total = n * n
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    data = np.empty(total, dtype=np.float64)

    ar = np.arange(n, dtype=np.int32)

    for i in range(n):
        start = i * n
        rows[start:start + n] = i + ar               # k = i + j
        cols[start:start + n] = ar                  # column = j
        data[start:start + n] = f[i]               # coefficient = f[i]

    return csr_matrix((data, (rows, cols)), shape=(K, n))


# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max Σ g_j   s.t.   f * g ≤ M ,   g ≥ 0.
    Returns the optimal g (may be all‑zero) or None on failure.
    """
    A = _build_constraint_matrix(f)
    K = A.shape[0]
    c = -np.ones(f.shape[0])           # linprog does minimisation → maximise Σg
    b = np.full(K, M, dtype=np.float64)

    # Use HiGHS (fast, supports sparse matrices)
    res = linprog(
        c, A_ub=A, b_ub=b,
        bounds=[(0, None)] * f.shape[0],
        method="highs",
        options={"presolve": True, "time_limit": 30.0},
    )
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
def _lp_refine(a0: np.ndarray, deadline: float, max_iters: int = 8) -> tuple[np.ndarray, float]:
    """
    Repeatedly apply the LP step described above.
    Each iteration updates f ← (1‑Σg)·f + g, keeping the vector on the simplex.
    Returns the best vector found and its cheap score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        # current bound M = max(f*f)
        M = _fft_self_convolve(a).max()
        g = _lp_step(a, M)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g < 1e-14:
            break
        # convex combination that preserves Σ = 1
        a = (1.0 - sum_g) * a + g
        # numeric cleanup
        a = np.maximum(a, 0.0)
        a = a / a.sum()

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time():.1f}] LP iter {it+1} → {cur_score:.6f}")

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.
    Returns (grad, cur_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with the active‑mask weight vector
    conv_rev = np.fft.irfft(np.fft.rfft(a[::-1]) * np.fft.rfft(w))[: len(a) + len(w) - 1]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even positions
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


def _hill_climb(a: np.ndarray, deadline: float, init_score: float) -> tuple[np.ndarray, float]:
    """Simple stochastic hill‑climb guided by the sub‑gradient."""
    cur_score = init_score
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    iteration = 0
    max_transfer = 0.025  # 2.5 % of total mass at the very start

    while time.time() < deadline:
        iteration += 1
        transfer = max_transfer * (0.5 ** (iteration / 8000.0))

        if iteration % 9 == 0:
            g, _, _ = _subgrad(a)
            i = int(np.argmax(g))
            j = int(np.argmin(g))
            if i == j:
                i = np.random.randint(n)
                j = (i + np.random.randint(1, n)) % n
        else:
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score, best_a = new_score, a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} new best {best_score:.6f}")

        # occasional jitter to escape flat plateaus
        if iteration % 6000 == 0 and np.random.rand() < 0.15:
            jitter = np.random.randn(n) * 6e-5
            a = best_a + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine stays within the supplied time budget and returns the best
    sequence (as a plain Python list) found so far.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin

    # --------------------------------------------------------------
    # Choose a working length.  A moderate size (≈2000) keeps the LP cheap
    # while still providing a good objective value.  If a historic best
    # is available we honour its length if it is not too large.
    try:
        base_n = int(len(height_sequence_1))   # type: ignore[name-defined]
    except Exception:
        base_n = 2000
    n = max(1200, min(2500, base_n))

    # --------------------------------------------------------------
    # 1️⃣  Analytic seeds (few).  Build a small pool and keep the best 4.
    seed_list = []

    # arcsine (classical strong performer)
    seed_list.append(_arcsine_distribution(n))

    # symmetric beta with very small exponents – heavy tails
    for p in (0.001, 0.003, 0.008, 0.02):
        seed_list.append(_beta_symmetric_distribution(n, p))

    # Gaussian – try a couple of widths
    for sf in (0.20, 0.30, 0.40):
        seed_list.append(_gaussian_distribution(n, sigma_frac=sf))

    # Cauchy – a few scales
    for sf in (0.15, 0.25, 0.35):
        seed_list.append(_cauchy_distribution(n, scale_frac=sf))

    # Include the historic optimum if it exists (rescaled to current n)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, 0.0)
            # resample to current n by linear interpolation
            x_old = np.linspace(0.0, 1.0, prev.size)
            x_new = np.linspace(0.0, 1.0, n)
            prev = np.interp(x_new, x_old, prev)
            prev = _project_onto_simplex(prev, s=1.0)
            seed_list.append(prev)
    except Exception:
        pass

    # cheap evaluation and keep the most promising seeds
    candidates = []
    for a in seed_list:
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        sc = _max_convolution_score(a)
        candidates.append((a, sc))
        print(f"[{time.time():.1f}] seed cheap score = {sc:.6f}")

    candidates.sort(key=lambda x: x[1])
    top_k = 4
    seeds = [c[0] for c in candidates[:top_k]]
    print(f"Selected {len(seeds)} seeds for LP refinement; best cheap = {candidates[0][1]:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  LP refinement – run a handful of iterations for each seed.
    best_seq = None
    best_score = np.inf

    for idx, seed_seq in enumerate(seeds):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] LP refinement start for seed {idx+1}/{len(seeds)}")
        seq, sc = _lp_refine(seed_seq, deadline, max_iters=8)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] new best after LP (seed {idx+1}) score = {sc:.6f}")

    if best_seq is None:
        # fallback – return a simple uniform distribution
        best_seq = np.full(n, 1.0 / n)
        best_score = _max_convolution_score(best_seq)

    # --------------------------------------------------------------
    # 3️⃣  Light polishing with hill‑climb (sub‑gradient guided)
    remaining = deadline - time.time()
    if remaining > 10.0:
        hill_deadline = time.time() + remaining - 2.0   # reserve a couple seconds
        best_seq, best_score = _hill_climb(best_seq, hill_deadline, best_score)

    # --------------------------------------------------------------
    # Final report
    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    # convert to plain Python list of floats
    return best_seq.tolist()
