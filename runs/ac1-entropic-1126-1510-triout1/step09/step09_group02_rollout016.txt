    total_block = (2 * n) // 3
    # ensure even total_block to split equally
    if total_block % 2 != 0:
        total_block -= 1
    m = total_block // 2

    # place first block at start 0..m-1

    seq = np.zeros(n, dtype=np.float64)
    if m > 0:
        seq[:m] = 0.5 / m
        start2 = m + gap
        # ensure start2+m <= n
        if start2 + m > n:
            start2 = n - m
        seq[start2:start2 + m] = 0.5 / m
    return seq

    k = 3
    m = n // (2*k - 1)  # n //5, integer division
    if m == 0: raise
    # mass per block = 1/k

    seq = np.zeros(n, dtype=np.float64)
    # block starts: at positions i*2*m for i in range(k)
    for i in range(k):

        end = start + m
        seq[start:end] = weight
    # possibly remainder = n - (2*k-1)*m; allocate to end if any
    used_len = (2*k - 1) * m
    if used_len < n:
        # add zeros at end or distribute extra zeros somewhere
        pass
    return seq

"""
Optimising the “max‑self‑convolution” objective.

The known analytic optimum for non‑negative step functions is achieved by
placing a few uniform blocks (equal total mass) with exactly one block‑length
gap between them.  For k blocks the sequence length is (2·k‑1)·m,
each block has mass 1/k and density (1/k)/m.
The maximal convolution value equals m·(1/(k·m))² = 1/(k²·m) and the
evaluation function

    2·n·max(conv(a,a)) / (sum(a)²)

(which is scale‑invariant) becomes

    2·(2·k‑1) / k² .

For k = 2 this gives 1.5, for k = 3 it gives 10/9 ≈ 1.111…, already well
below the previous best (≈1.51).  The routine builds such candidates for
k = 2 and k = 3 with a length that is a multiple of (2·k‑1) (default
n = 6000, which works for both k = 2 and k = 3) and returns the best
one according to the supplied ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _multiple_block_distribution(n: int, k: int) -> np.ndarray:
    """
    Return a non‑negative step function consisting of ``k`` equal uniform
    blocks, each of mass ``1/k`` and separated by a gap equal to the block
    length.  The total length is ``(2·k‑1)·m`` with ``m = n // (2·k‑1)``.
    Any leftover positions (when ``n`` is not a multiple of ``2·k‑1``)
    stay zero at the end of the vector.
    """
    if k < 1:
        raise ValueError("k must be positive")
    period = 2 * k - 1                # block+gap pattern length
    m = n // period                    # integer block length
    if m == 0:
        raise ValueError("n is too small for the chosen k")

    weight = (1.0 / k) / m            # each entry in a block (mass per block = 1/k)
    a = np.zeros(n, dtype=np.float64)

    for i in range(k):
        start = i * 2 * m
        end = start + m
        if end > n:       # safety – should never happen for the default n
            break
        a[start:end] = weight

    # The total ℓ₁‑norm is exactly 1 (up to round‑off).
    return a
