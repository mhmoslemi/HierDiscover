# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap surrogate = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.
    Returns (grad, cur_max, active_set).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (k = 2*i)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _smoothed_arcsine_distribution(n: int, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5 + eps) * (n - i - 0.5 + eps))
    return vals / vals.sum()


def _product_power_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-alpha)
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, p: float, q: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    w = np.where(np.isfinite(w), w, 0.0)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    return vals / vals.sum()


def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Γ–Dirichlet, α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()

def _beta_mixture_distribution(n: int, raw_weights: np.ndarray,
                               p_vals: np.ndarray) -> np.ndarray:
    """
    Two‑component symmetric Beta mixture.
    raw_weights are unrestricted; they are turned into proper simplex weights.
    p_vals are the shape parameters (must be >0).
    """
    w = np.exp(raw_weights)
    w /= w.sum()
    out = np.zeros(n, dtype=np.float64)
    for wi, pi in zip(w, p_vals):
        out += wi * _beta_symmetric_distribution(n, pi)
    # Numerical safety – ensure non‑negative and normalised
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s == 0:
        out = np.full(n, 1.0 / n)
    else:
        out /= s
    return out


def _de_optimize_beta_mixture(n: int, max_iter: int = 70,
                              popsize: int = 10) -> tuple[np.ndarray, float]:
    """
    Very cheap global optimisation of a 2‑component Beta mixture using
    differential evolution.  Returns (sequence, cheap_score).
    """
    # ------------------------------------------------------------------
    def _obj(x):
        # x : [w0_raw, w1_raw, p0, p1]
        seq = _beta_mixture_distribution(n, x[:2], x[2:])
        return _max_convolution_score(seq)

    bounds = [(-8, 8), (-8, 8), (0.05, 2.0), (0.05, 2.0)]
    # DE is deterministic with the supplied seed.
    res = differential_evolution(
        _obj,
        bounds,
        maxiter=max_iter,
        popsize=popsize,
        polish=False,
        init='latinhypercube',
        seed=42,
        updating='deferred',
        workers=1,
        disp=False,
        tol=1e-6,
    )
    best_seq = _beta_mixture_distribution(n, res.x[:2], res.x[2:])
    best_score = _max_convolution_score(best_seq)
    return best_seq, best_score

def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 200_000, eta0: float = 1.0):
    """Mirror‑descent (exponentiated‑gradient) on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        a = a / s if s != 0 else np.full_like(a, 1.0 / a.size)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 200_000,
                  step0: float = 0.3,
                  beta1: float = 0.9,
                  beta2: float = 0.999):
    """Adam‑style adaptive gradient optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 400_000, eta0: float = 0.6):
    """Plain sub‑gradient descent with diminishing step size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _smooth_gradient(a: np.ndarray, gamma: float):
    """Weighted gradient of the soft‑max of a*a (γ controls smoothness)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    b_max = b.max()
    w = np.exp(gamma * (b - b_max))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    diag_w = w[2 * np.arange(n)]
    grad -= a * diag_w
    return grad, b_max


def _smooth_gradient_descent(a0: np.ndarray, deadline: float,
                             max_iters: int = 300_000,
                             eta0: float = 0.8,
                             gamma0: float = 2.0,
                             gamma_factor: float = 1.005):
    """Continuation method: start with a smooth max (γ≈2) and increase γ."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    gamma = gamma0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _ = _smooth_gradient(a, gamma)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score, best_a = cur_score, a.copy()
        gamma *= gamma_factor
    return best_a, best_score


def _heavy_ball_subgradient_descent(a0: np.ndarray, deadline: float,
                                   max_iters: int = 300_000,
                                   eta0: float = 0.6,
                                   momentum: float = 0.6):
    """Sub‑gradient descent with a heavy‑ball momentum term."""
    a_prev = a0.copy()
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * grad + momentum * (a - a_prev)
        a_new = _project_onto_simplex(a_new, s=1.0)
        a_prev, a = a, a_new
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

# --- stochastic annealing hill‑climb ------------------------------------
def _annealing_hill_climb(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Random mass transfers accepted with a Metropolis rule.
    Temperature decays exponentially.
    Returns (best_sequence, best_score).
    """
    a = a0.copy()
    cur_score = _max_convolution_score(a)
    best_seq = a.copy()
    best_score = cur_score
    n = a.shape[0]
    it = 0
    # schedule parameters
    init_temp = 0.02
    decay = 0.99995

    temp = init_temp
    while time.time() < deadline:
        it += 1
        # choose two distinct indices
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        # amount to move: fraction of current mass at i
        delta = np.random.rand() * min(0.05 * a[i], a[i])
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        # acceptance test
        if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            a, cur_score = a_new, new_score
            if new_score < best_score - 1e-15:
                best_seq, best_score = a_new.copy(), new_score
                print(f"[{time.time():.1f}] anneal iter={it} new best {best_score:.6f}")
        # temperature schedule
        temp *= decay
        if it % 5000 == 0:
            # occasional jitter to keep exploration alive
            jitter = np.random.randn(n) * 5e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    return best_seq, best_score

    try:
        base_n = len(height_sequence_1)   # type: ignore[name-defined]
        if base_n < 500:
            base_n = 500
    except Exception:
        base_n = 3500

    # a few nearby lengths – the objective depends weakly on n, but we try a few
    lengths = [int(base_n * f) for f in (0.9, 1.0, 1.1)]
    lengths = [max(500, n) for n in lengths]

    # ------------------------------------------------------------------
    # 1️⃣  Coarse sweep (families + Beta‑mixture via DE)
    best_seq = None
    best_score = float('inf')

    for n in lengths:
        if time.time() >= deadline:
            break

        # -- classic families ------------------------------------------------
        for eps in (0.0, 0.1, 0.5, 1.0, 2.0, 5.0):
            a = _arcsine_distribution(n) if eps == 0.0 else _smoothed_arcsine_distribution(n, eps)
            score = _max_convolution_score(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()

        for alpha in np.arange(0.30, 0.71, 0.05):
            a = _product_power_distribution(n, alpha)
            score = _max_convolution_score(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()

        for p in np.arange(0.30, 0.81, 0.05):
            a = _beta_symmetric_distribution(n, p)
            score = _max_convolution_score(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()

        for lam in (0.001, 0.003, 0.005, 0.010):
            a = _exponential_distribution(n, lam)
            score = _max_convolution_score(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()

        for _ in range(3):
            a = _random_dirichlet(n)
            score = _max_convolution_score(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()

        # -- beta mixture (2‑component) ------------------------------------
        # small DE run; we cap the total time spent on DE to stay safe.
        if time.time() < deadline - 2.0:   # leave a little slack
            mix_seq, mix_score = _de_optimize_beta_mixture(
                n,
                max_iter=70,   # ~ 700 evaluations → ≈ 1 s per length
                popsize=10,
            )
            if mix_score < best_score:
                best_score, best_seq = mix_score, mix_seq.copy()

    # incorporate the previous best, if any
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            score = _max_convolution_score(prev)
            if score < best_score:
                best_score, best_seq = score, prev.copy()
    except Exception:
        pass

    print(f"[{time.time():.1f}] after coarse sweep best cheap={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Heavy refinement – split the remaining budget
    remaining = deadline - time.time()
    if remaining <= 0.0:
        return best_seq.tolist()

    # 20 % Expo‑grad, 20 % Adam, 30 % smooth, 15 % heavy‑ball,
    # the rest (15 %) kept for annealing hill‑climb.
    # (All sub‑routines respect their own per‑method deadline.)

    # 2.1 exp‑grad
    sub_deadline = time.time() + remaining * 0.20
    seq, sc = _exp_gradient_descent(best_seq, sub_deadline,
                                    max_iters=250_000, eta0=1.0)
    if sc < best_score:
        best_seq, best_score = seq, sc
        print(f"[{time.time():.1f}] after exp‑grad   {best_score:.6f}")

    # 2.2 Adam
    sub_deadline = time.time() + remaining * 0.20
    seq, sc = _adam_descent(best_seq, sub_deadline,
                            max_iters=250_000, step0=0.3)
    if sc < best_score:
        best_seq, best_score = seq, sc
        print(f"[{time.time():.1f}] after Adam      {best_score:.6f}")

    # 2.3 smoothed‑max continuation
    sub_deadline = time.time() + remaining * 0.30
    seq, sc = _smooth_gradient_descent(best_seq, sub_deadline,
                                        max_iters=350_000,
                                        eta0=0.9,
                                        gamma0=2.0,
                                        gamma_factor=1.005)
    if sc < best_score:
        best_seq, best_score = seq, sc
        print(f"[{time.time():.1f}] after smooth    {best_score:.6f}")

    # 2.4 heavy‑ball sub‑gradient
    sub_deadline = time.time() + remaining * 0.15
    seq, sc = _heavy_ball_subgradient_descent(best_seq, sub_deadline,
                                              max_iters=350_000,
                                              eta0=0.6,
                                              momentum=0.6)
    if sc < best_score:
        best_seq, best_score = seq, sc
        print(f"[{time.time():.1f}] after heavy‑ball {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Fine‑grained stochastic annealing hill‑climb
    a = best_seq.copy()
    cur_score = best_score
    remaining = deadline - time.time()
    if remaining > 0.0:
        a, cur_score = _annealing_hill_climb(a, deadline)

    # final report
    print(f"Final cheap score: {cur_score:.6f} (length={len(a)})")
    return a.tolist()
