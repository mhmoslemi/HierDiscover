"""
Improved search for a non‑negative step‑function that minimises
`evaluate_sequence`.  The algorithm proceeds as follows

1. **Pick a reasonably large length** (up to 12 000) – the constant
   2·n·max(conv) tends to the theoretical limit 1.5 when the shape
   approaches the continuous arcsine law, and a larger `n` reduces the
   finite‑size error.

2. **Broad sampling** – generate many analytic families
   (arcsine, symmetric‑Beta, power‑transformed arcsine, exponential,
   random Dirichlet) and three deterministic re‑orderings that spread
   large coefficients.  All vectors are ℓ₁‑normalised; the cheap surrogate
   `2·n·max(conv)` (exact for normalised vectors) is used to rank the
   candidates.

3. **Intensive refinement** – the top‑k candidates are polished with a
   projected sub‑gradient method that uses an explicit back‑tracking line
   search.  The sub‑gradient of `max(conv)` is computed analytically
   (see `_subgrad`).  After each step we re‑project onto the probability
   simplex.

4. **Frank‑Wolfe mass transfer** – on the current global best we repeatedly
   move a fraction of mass from the coordinate with the largest
   sub‑gradient to the one with the smallest, which is a provably descent
   direction for the max‑objective.

5. **Return** the best sequence found within the time budget.

The whole routine respects the 1000 s budget, uses only NumPy, SciPy and
standard library modules, and never writes to disk.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
from typing import List, Tuple

# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolutions -----------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
# Objective helpers ----------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate for `evaluate_sequence` when Σa = 1:
       returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray) -> Tuple[np.ndarray, float, np.ndarray]:
    """
    Exact sub‑gradient of the max‑convolution functional on the simplex.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives raw gradient
    conv_rev = _fft_convolve(a[::-1], w)
    # slice the central n entries (the gradient w.r.t. each a_i)
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]

    # diagonal correction for the terms i=j (they appear only once)
    even_mask = (active % 2 == 0)
    if even_mask.any():
        diag_idx = (active[even_mask] // 2)
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, alpha: float) -> np.ndarray:
    """Beta(α,α) on the discrete grid (U[0,1] with mid‑point rule)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # using log‑space for stability
    with np.errstate(divide='ignore', invalid='ignore'):
        logw = (alpha - 1) * (np.log(x) + np.log(1.0 - x))
    # replace -inf with a finite value
    logw[np.isneginf(logw)] = -1e30
    w = np.exp(logw)
    if w.sum() == 0 or np.isnan(w).any():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex from Dirichlet(α,…,α)."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Re‑ordering heuristics -----------------------------------------------
def order_alternate_extremes(desc_vals: np.ndarray) -> np.ndarray:
    """Spread the largest values alternately left‑most / right‑most."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    left, right, toggle = 0, n - 1, True
    for v in desc_vals:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    return out


def order_center_outwards(desc_vals: np.ndarray) -> np.ndarray:
    """Place the biggest value at the centre and fill outwards."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    mid = (n - 1) // 2
    positions = [mid]
    for d in range(1, n):
        left = mid - d
        if left >= 0:
            positions.append(left)
        right = mid + d
        if right < n:
            positions.append(right)
    for i, pos in enumerate(positions[:n]):
        out[pos] = desc_vals[i]
    return out


def order_binary_tree(desc_vals: np.ndarray) -> np.ndarray:
    """Recursive midpoint filling – a binary‑tree layout."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    positions = []

    def recurse(l: int, r: int):
        if l > r:
            return
        m = (l + r) // 2
        positions.append(m)
        recurse(l, m - 1)
        recurse(m + 1, r)

    recurse(0, n - 1)
    out[positions] = desc_vals
    return out


# ----------------------------------------------------------------------
# Local optimiser (projected sub‑gradient with line‑search) ----------
def _refine_subgradient(a0: np.ndarray,
                        deadline: float,
                        max_iters: int = 5000) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with back‑tracking line search.
    The iterate always stays on the probability simplex (ℓ₁ = 1).
    """
    a = a0.copy()
    best = a.copy()
    best_score = _max_convolution_score(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        direction = -grad

        # simple back‑tracking line search
        step = 1.0
        improved = False
        for _ in range(12):
            a_new = _project_onto_simplex(a + step * direction, s=1.0)
            sc = _max_convolution_score(a_new)
            if sc < best_score - 1e-12:
                a = a_new
                best = a_new
                best_score = sc
                improved = True
                break
            step *= 0.5

        if not improved:
            # tiny random jitter to escape plateaus
            a = _project_onto_simplex(a + np.random.randn(a.size) * 1e-6, s=1.0)

    return best, best_score


# ----------------------------------------------------------------------
# Frank‑Wolfe style mass transfer --------------------------------------
def _frank_wolfe_transfer(a: np.ndarray,
                          deadline: float,
                          max_iters: int = 200_000) -> Tuple[np.ndarray, float]:
    """Move mass from the highest‑gradient index to the lowest‑gradient index."""
    cur = a.copy()
    best_score = _max_convolution_score(cur)
    best = cur.copy()
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(cur)
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        delta = min(cur[i_src], 1.0 - cur[i_dst])
        if delta <= 0:
            break
        step = 0.7 * delta
        cur[i_src] -= step
        cur[i_dst] += step
        sc = _max_convolution_score(cur)
        if sc < best_score:
            best_score, best = sc, cur.copy()
    return best, best_score


# ----------------------------------------------------------------------
# Candidate generation -------------------------------------------------
def _add_candidate(arr: np.ndarray,
                   cand_list: List[Tuple[np.ndarray, float]]) -> None:
    """Normalise `arr` to ℓ₁=1 and append (arr, score) to `cand_list`."""
    a = np.asarray(arr, dtype=np.float64)
    if a.ndim != 1 or a.size == 0:
        return
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total < 1e-12:
        return
    a /= total
    cand_list.append((a, _max_convolution_score(a)))


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the supplied time budget (≈ 1000 s) and returns the
    best vector discovered.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5            # safety margin

    # ------------------------------------------------------------------
    # Choose a working length (larger lengths reduce finite‑size error)
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 4000

    # we aim for roughly double the base size but never exceed 12 000
    n = min(12_000, max(2_000, int(base_n * 2)))
    print(f"[{time.time():.1f}] Using length n = {n}")

    # ------------------------------------------------------------------
    # Build a pool of raw candidates
    candidates: List[Tuple[np.ndarray, float]] = []

    # ---- analytic families -------------------------------------------------
    # arcsine (the classic near‑optimal shape)
    a_arc = _arcsine_distribution(n)
    _add_candidate(a_arc, candidates)

    # a few reorderings of arcsine
    desc = np.sort(a_arc)[::-1]
    _add_candidate(order_alternate_extremes(desc), candidates)
    _add_candidate(order_center_outwards(desc), candidates)
    _add_candidate(order_binary_tree(desc), candidates)

    # symmetric Beta for a fine grid of exponents (U‑shaped distributions)
    beta_alphas = np.concatenate((
        np.arange(0.02, 0.21, 0.02),
        np.arange(0.25, 0.71, 0.05),
        np.arange(0.80, 1.51, 0.10),
        [2.0]
    ))
    for alpha in beta_alphas:
        if time.time() >= deadline:
            break
        a_beta = _beta_symmetric_distribution(n, alpha)
        _add_candidate(a_beta, candidates)
        desc = np.sort(a_beta)[::-1]
        _add_candidate(order_alternate_extremes(desc), candidates)

    # power‑transformed arcsine (helps fine‑tune the tails)
    power_exps = [0.6, 0.8, 1.0, 1.2, 1.5, 2.0]
    base_arc = _arcsine_distribution(n)
    for pwr in power_exps:
        if time.time() >= deadline:
            break
        a = np.power(base_arc, pwr)
        a /= a.sum()
        _add_candidate(a, candidates)

    # exponential (allocates mass near the centre)
    for lam in [0.001, 0.003, 0.005, 0.01]:
        if time.time() >= deadline:
            break
        a_exp = _exponential_distribution(n, lam=lam)
        _add_candidate(a_exp, candidates)

    # random Dirichlet draws (very low concentration → heavy‑tailed)
    for alpha in [0.05, 0.1, 0.2, 0.5]:
        if time.time() >= deadline:
            break
        a_dir = _random_dirichlet_alpha(n, alpha)
        _add_candidate(a_dir, candidates)

    # ------------------------------------------------------------------
    # Inject the previous best if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        if prev.sum() > 0:
            prev = prev / prev.sum()
            _add_candidate(prev, candidates)
            print(f"Added previous best candidate, score={candidates[-1][1]:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Keep the most promising seeds
    candidates.sort(key=lambda x: x[1])
    top_k = min(12, len(candidates))
    top_candidates = candidates[:top_k]
    print(f"Generated {len(candidates)} raw candidates, keeping top {top_k}.")
    print(f"Best seed score = {top_candidates[0][1]:.6f}")

    best_seq = top_candidates[0][0].copy()
    best_score = top_candidates[0][1]

    # ------------------------------------------------------------------
    # Intensive refinement of each seed (projected sub‑gradient + line search)
    for idx, (seed_vec, _) in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        seq, sc = _refine_subgradient(seed_vec, deadline, max_iters=4000)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} refined → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Frank‑Wolfe style mass transfer on the global best
    if time.time() < deadline:
        seq_fw, sc_fw = _frank_wolfe_transfer(best_seq, deadline, max_iters=120_000)
        if sc_fw < best_score:
            best_score, best_seq = sc_fw, seq_fw.copy()
            print(f"[{time.time():.1f}] Frank‑Wolfe boost → {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best surrogate score: {best_score:.6f} (n={len(best_seq)})")
    return best_seq.tolist()
