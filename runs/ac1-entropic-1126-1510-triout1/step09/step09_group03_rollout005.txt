def _lp_improve(seq, t_init=0.2, min_improve=1e-8, deadline=None):
    n = len(seq)
    # compute M = max(conv(seq, seq))
    b = _fft_self_convolve(seq)
    M = b.max()
    # Build A matrix
    from scipy.sparse import diags
    offsets = np.arange(n)
    # Data list: each diag array is np.full(n, seq[i]) shape (n,)
    data = [np.full(n, seq[i]) for i in range(n)]
    A = diags(data, offsets, shape=(2*n-1, n), format='csc')
    # linear objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    b_ub = np.full(2*n-1, M)
    # solve
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if res.status != 0:
        return seq, False  # no improvement
    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= 0:
        return seq, False
    # scale g0 to sum=1
    g = g0 / sum_g0

    # compute delta
    delta = g - seq

    # precompute conv_f_Δ, conv_Δ_Δ
    conv_f_delta = _fft_convolve(seq, delta)
    conv_delta_delta = _fft_self_convolve(delta)

    # Precompute conv_f_f already have b

    # Find best t in [0,1] via ternary search (maybe 30 iterations)

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Surrogate 2·n·max(conv(a,a)) – exact when sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) on the simplex (∑a=1).
    Returns (grad, cur_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.15,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 500_000):
    """Adam on the true sub‑gradient (max‑convolution)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with concentration ``alpha``."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
def _lp_improve(seq: np.ndarray, deadline: float,
                min_improve: float = 1e-12) -> tuple[np.ndarray, bool]:
    """
    One LP‑based boost step.

    Returns (new_seq, improved_flag).  The step consists of:
      * solving max 1ᵀg  s.t. conv(seq,g) ≤ M, g ≥ 0,
      * scaling g to unit ℓ₁,
      * line‑searching the optimal mixing coefficient t∈[0,1] that
        minimises the peak of conv(seq+t·Δ, seq+t·Δ), where Δ = g‑seq,
      * accepting the move only if the peak strictly drops.
    """
    if time.time() >= deadline:
        return seq, False

    n = seq.size
    conv_ff = _fft_self_convolve(seq)
    M = conv_ff.max()

    # --- build the (2n‑1)×n convolution matrix A (Toeplitz with constant diagonals) ---
    # each sub‑diagonal i (i≥0) is filled with seq[i]
    data = [np.full(n, seq[i], dtype=np.float64) for i in range(n)]
    offsets = np.arange(n)               # 0,…,n‑1
    A = diags(data, offsets, shape=(2 * n - 1, n), format='csc')

    # --- linear program: maximise ∑g  ⇔ minimise -∑g ---
    c = -np.ones(n, dtype=np.float64)    # minimise -sum(g)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})

    # clean up the sparse matrix early
    del A

    if not res.success:
        return seq, False

    g0 = res.x
    s = g0.sum()
    if s <= 0:
        return seq, False
    g = g0 / s                         # unit ℓ₁‑norm

    delta = g - seq

    # Pre‑compute terms needed for the quadratic expression of the peak
    conv_f_delta = _fft_convolve(seq, delta)          # conv(f,Δ)
    conv_delta_delta = _fft_self_convolve(delta)      # conv(Δ,Δ)

    # --- ternary search for the best mixing coefficient t ---
    lo, hi = 0.0, 1.0
    for _ in range(30):
        t1 = lo + (hi - lo) / 3.0
        t2 = hi - (hi - lo) / 3.0
        val1 = np.max(conv_ff + 2 * t1 * conv_f_delta + (t1 ** 2) * conv_delta_delta)
        val2 = np.max(conv_ff + 2 * t2 * conv_f_delta + (t2 ** 2) * conv_delta_delta)
        if val1 < val2:
            hi = t2
        else:
            lo = t1
    t_opt = (lo + hi) / 2.0

    M_new = np.max(conv_ff + 2 * t_opt * conv_f_delta + (t_opt ** 2) * conv_delta_delta)

    if M_new < M - min_improve:
        seq_new = seq + t_opt * delta
        # safeguard: enforce feasibility (non‑negativity and simplex)
        seq_new = np.maximum(seq_new, 0.0)
        seq_new = _project_onto_simplex(seq_new, s=1.0)
        return seq_new, True
    else:
        return seq, False


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects a 1000 s time budget and returns the best list found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 0️⃣ Starting point – reuse the previous best if present
    prev_seq = globals().get('height_sequence_1')
    if prev_seq is not None:
        try:
            a0 = np.asarray(prev_seq, dtype=np.float64)
            if a0.ndim == 1 and a0.size >= 2:
                a0 = a0 / a0.sum()
            else:
                raise ValueError
        except Exception:
            a0 = None
    else:
        a0 = None

    # 0️⃣ If no usable previous construction, initialise randomly
    if a0 is None:
        init_len = 3000
        a0 = _random_dirichlet_alpha(init_len, alpha=0.5)

    best_seq = a0.copy()
    best_score = _max_convolution_score(best_seq)
    print(f"[{time.time():.1f}] start length={len(best_seq)} surrogate={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣ LP‑based boosting iterations
    lp_iter = 0
    while time.time() < deadline and lp_iter < 40:
        lp_iter += 1
        new_seq, improved = _lp_improve(best_seq, deadline)
        if not improved:
            print(f"[{time.time():.1f}] LP iteration {lp_iter} – no improvement, stop.")
            break
        best_seq = new_seq
        cur_score = _max_convolution_score(best_seq)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            print(f"[{time.time():.1f}] LP iteration {lp_iter} improved surrogate to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Short Adam refinement (if we still have time)
    remaining = deadline - time.time()
    if remaining > 5.0:
        sub_dead = time.time() + remaining * 0.6
        seq_adam, score_adam = _adam_descent(best_seq, sub_dead)
        if score_adam < best_score:
            best_seq, best_score = seq_adam, score_adam
            print(f"[{time.time():.1f}] Adam refinement improved surrogate to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Optional very light Frank–Wolfe polishing (few thousand steps)
    a = best_seq.copy()
    cur = best_score
    fw_iter = 0
    while time.time() < deadline and fw_iter < 30_000:
        fw_iter += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-14:
            break
        a[i_src] -= max_transfer
        a[i_dst] += max_transfer
        a = _project_onto_simplex(a, s=1.0)
        new = _max_convolution_score(a)
        if new < cur - 1e-13:
            cur = new
            if new < best_score:
                best_seq = a.copy()
                best_score = new
                print(f"[{time.time():.1f}] FW iter={fw_iter} new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation (real objective)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] final evaluate_sequence = {final_val}")
    except Exception as exc:
        print("[warning] evaluation failed:", exc)

    print(f"returning sequence of length {len(best_seq)} with surrogate {best_score:.6f}")
    return best_seq.tolist()
