"""
propose_candidate – search for a step‑function minimizing
evaluate_sequence().

The algorithm proceeds in three stages:

1. **Analytic sweep** – For several lengths (up to 16000) the code
   builds a variety of simple families (uniform, arcsine,
   triangular, exponential, Gaussian, beta, sine‑power) and many
   random mixtures of a base distribution with the uniform one.
   Each candidate is normalised to sum = 1 and its objective
   2 · n · max(a*a) is evaluated via a fast FFT self‑convolution.
   The best vector so far is kept.

2. **Smooth‑max mirror descent** – Starting from the best candidate,
   we perform exponentiated‑gradient updates on the simplex.
   The loss is a smoothed max: w ∝ exp(α·(conv−max(conv))) and the
   gradient is 2·(a_rev * w).  The smoothing parameter α and the
   learning rate are annealed.  The best vector encountered during
   this phase replaces the incumbent.

3. **Simple hill‑climber** – Small random transfers of mass are
   tried; a move is kept only when it strictly lowers the maximum
   convolution entry.  Periodic jitter helps escape plateaus.

All steps respect the wall‑clock budget (default = 1000 s).  The
function finally returns the best sequence (as a plain list of
floats).  If a global variable `height_sequence_1` is defined, it is
also considered as a starting point.
"""

import time

import math
from typing import List

import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families – each returns a non‑negative vector with sum = 1.
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.01) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # Use log‑space to avoid underflow for extreme p
    logw = (p - 1) * (np.log(x) + np.log(1 - x))
    w = np.exp(logw)
    if np.all(w == 0):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at (n‑1)/2."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _mixture(v1: np.ndarray, v2: np.ndarray, w: float) -> np.ndarray:
    """Convex combination w·v1 + (1‑w)·v2, re‑normalised to sum = 1."""
    mixed = w * v1 + (1.0 - w) * v2
    mixed = np.maximum(mixed, 1e-15)
    mixed /= mixed.sum()
    return mixed

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best list of floats found within the wall‑clock budget.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0   # a small safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1. Analytic sweep (short list of lengths)
    lengths = [2000, 4000, 8000, 12000, 16000]
    for n in lengths:
        if time.time() >= deadline:
            break

        # a small set of base families
        families = [
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_triangular(n),
            _distribution_exponential(n, lam=0.008),
            _distribution_exponential(n, lam=0.02),
            _distribution_gaussian(n, sigma=0.30),
            _distribution_gaussian(n, sigma=0.45),
        ]

        # beta family with a range of p
        for p in (0.2, 0.35, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
            families.append(_distribution_beta_sym(n, p))

        # sine‑power family
        for pw in (0.5, 1.0, 2.0, 3.0):
            families.append(_distribution_sine_power(n, pw))

        # evaluate them
        for vec in families:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] init n={n}  score={sc:.6f}")

        # random Dirichlet draws – a few per length
        for _ in range(30):
            if time.time() >= deadline:
                break
            alpha = random.choice([0.2, 0.5, 1.0, 2.0])
            vec = np.random.dirichlet(np.full(n, alpha))
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] dirichlet n={n}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Random mixtures of a base shape with the uniform distribution
    if best_seq is not None:
        base = best_seq.copy()
    else:
        # fallback uniform
        best_n = 8000
        base = _distribution_uniform(best_n)

    for mix_iter in range(200):
        if time.time() >= deadline:
            break
        n = best_n
        # pick a random base family
        family = random.choice([
            _distribution_arcsine,
            _distribution_triangular,
            _distribution_exponential,
            _distribution_beta_sym,
            _distribution_sine_power,
            _distribution_gaussian,
        ])
        # random parameters

            lam = 10 ** random.uniform(-4, -1)
            vec = family(n, lam=lam)
        elif family is _distribution_beta_sym:
            p = 10 ** random.uniform(-1, 0.7)   # range ~0.1–5
            vec = family(n, p)
        elif family is _distribution_gaussian:
            sigma = random.uniform(0.15, 0.5)
            vec = family(n, sigma)
        elif family is _distribution_sine_power:
            pw = random.uniform(0.3, 4.0)
            vec = family(n, pw)
        elif family is _distribution_triangular:
            vec = family(n)
        else:   # arcsine or exponential fallback
            vec = family(n)

        # mix with uniform
        w = random.random()
        mixed = _mixture(vec, _distribution_uniform(n), w)

        sc = _max_convolution_score(mixed)
        if sc < best_score:
            best_seq = mixed.copy()
            best_score = sc
            print(f"[{time.time():.1f}] mix iter={mix_iter}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3. Bring in the global `height_sequence_1` if it exists
    try:

        if prev.ndim == 1 and prev.size > 0:

            prev /= prev.sum()

            if sc < best_score:
                best_seq = prev.copy()
                best_n = prev.size
                best_score = sc
                print(f"[{time.time():.1f}] global variable score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Smooth‑max mirror descent (exponentiated gradient)
    a = best_seq.copy()
    n = a.shape[0]

    # schedule: start with low smoothing, increase; learning rate decays
    total_steps = 5000
    for it in range(total_steps):
        if time.time() >= deadline:
            break

        conv = _fft_self_convolve(a)
        maxc = conv.max()

        # smoothing parameter α grows from 1 to ~2000 (log‑linear)
        t = it / total_steps
        alpha = 1.0 * (2000.0 / 1.0) ** t

        # soft‑max weights
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of weighted sum of convolutions
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # learning rate schedule – from 0.6 down to 0.03
        lr = 0.6 * (0.03 / 0.6) ** t

        # exponentiated‑gradient update
        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()

    # 5. Simple hill‑climber (mass transfers)
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    while time.time() < deadline:
        it += 1

        # pick an index that participates in the current maximal entry
        k = int(conv.argmax())
        i_low = max(0, k - (n - 1))

        # find the pair (i,j) with largest product a[i]*a[j] among those giving k
        best_i, best_j = i_low, k - i_low
        max_prod = a[best_i] * a[best_j]
        for i in range(i_low, i_high + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        # transfer a small chunk from the larger of the two to a random slot
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue
        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a_new = a.copy()
        a_new[source] -= delta
        a_new[target] += delta
        a_new = np.maximum(a_new, eps)
        a_new /= a_new.sum()
        new_conv = _fft_self_convolve(a_new)
        new_M = new_conv.max()

            a = a_new
            conv = new_conv
            M = new_M
            sc = 2.0 * n * M
            if sc < best_score - 1e-12:
                best_seq = a.copy()
                best_score = sc
                if it % 500 == 0:
                    print(f"[{time.time():.1f}] hill it={it}  score={sc:.6f}")
        else:
            # occasionally inject a tiny jitter to avoid stagnation
            if it % 4000 == 0:
                jitter = np.random.randn(n) * 1e-5
                a = a + jitter
                a = np.maximum(a, eps)
                a /= a.sum()
                conv = _fft_self_convolve(a)
                M = conv.max()
                print(f"[{time.time():.1f}] jitter applied at hill it={it}")

    final_obj = 2.0 * best_n * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_obj:.6f} (length={best_n})")
    return best_seq.tolist()
