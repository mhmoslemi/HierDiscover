def smooth_max_and_grad(a, lam):
    conv = _fft_self_convolve(a)   # length 2n-1
    # compute softmax weights
    maxv = conv.max()
    # to avoid overflow, subtract maxv
    exps = np.exp(lam * (conv - maxv))
    sum_exp = exps.sum()
    softmax = exps / sum_exp   # shape same
    # smooth max: maxv + log(sum_exp)/lam
    smooth_max = maxv + np.log(sum_exp) / lam
    # gradient: 2 * conv_rev of reversed a with softmax weights
    grad = 2.0 * _fft_convolve(a[::-1], softmax)[len(a)-1 : len(a)-1 + len(a)]
    return smooth_max, grad
```

"""
Propose a non‑negative step function that minimises `evaluate_sequence`.

Algorithm
---------

1. **Broad sampling** – generate a large pool of candidates from many analytic
   families (arcsine, triangular, exponential, Gaussian, Cauchy,
   symmetric/unsymmetric Beta, Dirichlet draws) for several lengths.
   For each raw vector three deterministic reorderings are tried
   (`alternate_extremes`, `center_outwards`, `binary_tree`) because spreading
   large values tends to flatten the self‑convolution.
   All candidates are ℓ₁‑normalised and the cheap surrogate
   `2·n·max(conv(a,a))` (exact for a normalised vector) is used to keep the
   best **K** seeds (K≈3).

2. **Continuation gradient descent** – on the globally best seed we run three
   consecutive optimisers that all minimise a *smooth* approximation of the
   max‑convolution:

   * `smooth_max` = log‑sum‑exp(λ·conv) / λ  (λ grows from 5 to ≈200)
   * Gradient of `smooth_max` can be obtained analytically via FFT
     (no diagonal correction needed).

   The three schemes are:
   * Adam (first‑order moment, second‑order moment)
   * Exponentiated‑gradient (mirror descent on the simplex)
   * Projected SGD

   Each optimiser is allocated a large portion of the remaining time
   (≈ 30 % / 30 % / 30 %).  The current best (real) objective
   `2·n·max(conv)` is monitored and the best sequence is remembered.

3. **Frank‑Wolfe mass transfer** – starting from the best sequence a
   Frank‑Wolfe style move transfers mass from the index with the largest
   sub‑gradient to the index with the smallest.  The step size is a fixed
   fraction (70 %) of the admissible mass.  Up to 150 k iterations are
   performed (or until the time budget is exhausted).

4. **Return** – the best vector found (as a plain `list[float]`) is returned.
   All numbers are non‑negative; scaling does not affect the evaluation
   because `evaluate_sequence` is homogeneous.

The implementation respects the 1000 s budget, uses only NumPy and the
standard library, and places all helper functions at top level.

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _actual_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector normalised to sum‑1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


def _smooth_max_and_grad(a: np.ndarray, lam: float):
    """
    Smoothed max (log‑sum‑exp) of the self‑convolution and its gradient.
    Returns (smooth_max, grad) where grad has the same shape as a.
    """
    conv = _fft_self_convolve(a)                     # length 2n‑1
    max_val = conv.max()
    # stability: shift by max_val
    exps = np.exp(lam * (conv - max_val))
    sum_exps = exps.sum()
    w = exps / sum_exps                               # soft‑max weights, sum=1
    smooth_max = max_val + np.log(sum_exps) / lam

    # gradient = 2 * (rev(a) * w) restricted to the middle n entries
    grad = 2.0 * _fft_convolve(a[::-1], w)[len(a) - 1:
                                           len(a) - 1 + len(a)]
    return smooth_max, grad

# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _gaussian_distribution(n: int, sigma_factor: float = 0.15) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    w /= w.sum()
    return w


def _cauchy_distribution(n: int, scale_factor: float = 0.15) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    scale = scale_factor * n
    w = 1.0 / (1.0 + ((idx - centre) / scale) ** 2)
    w /= w.sum()
    return w


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide='ignore', invalid='ignore'):
        w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    if np.isnan(w).any() or np.isinf(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Re‑ordering heuristics -----------------------------------------------
def order_alternate_extremes(desc_vals: np.ndarray) -> np.ndarray:
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    left, right = 0, n - 1
    toggle = True
    for v in desc_vals:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    return out


def order_center_outwards(desc_vals: np.ndarray) -> np.ndarray:
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    mid = (n - 1) // 2
    pos = [mid]
    for d in range(1, n):
        if mid - d >= 0:
            pos.append(mid - d)
        if mid + d < n:
            pos.append(mid + d)
    for i, p in enumerate(pos[:n]):
        out[p] = desc_vals[i]
    return out


def order_binary_tree(desc_vals: np.ndarray) -> np.ndarray:
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)

    positions = []

    def recurse(l: int, r: int):
        if l > r:
            return
        m = (l + r) // 2
        positions.append(m)
        recurse(l, m - 1)
        recurse(m + 1, r)

    recurse(0, n - 1)
    out[positions] = desc_vals
    return out

# Sub‑gradient of the (hard) max – needed for Frank‑Wolfe ----------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of the max‑convolution functional on the simplex.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]

    # diagonal correction for active even indices (self‑product terms)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active

def _exp_opt(a0: np.ndarray, deadline: float,

            add_candidate(order_binary_tree(desc))

        # Symmetric beta family
        for p in (0.2, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
            a = _beta_symmetric_distribution(n, p)
            add_candidate(a)
            desc = np.sort(a)[::-1]
            add_candidate(order_alternate_extremes(desc))

        # Unsymmetric beta (limited grid)
        for alpha in (0.5, 1.0, 2.0):
            for beta in (0.5, 1.0, 2.0):
                a = _beta_distribution(n, alpha, beta)
                add_candidate(a)
                desc = np.sort(a)[::-1]
                add_candidate(order_alternate_extremes(desc))

        # Dirichlet draws
        for alpha in (0.2, 0.5, 1.0, 2.0):
            a = _random_dirichlet_alpha(n, alpha)
            add_candidate(a)

        # Regular “comb” vectors (equally spaced spikes)
        spikes = max(2, int(np.sqrt(n)))
        idxs = np.linspace(0, n - 1, spikes, dtype=int)
        comb = np.zeros(n)
        comb[idxs] = 1.0 / spikes
        add_candidate(comb)

    # ------------------------------------------------------------------
    # 2. Incorporate previous best if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        add_candidate(prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Keep only the top few seeds (K≈3)
    candidates.sort(key=lambda x: x[1])
    top_k = min(3, len(candidates))
    top_seeds = [c[0] for c in candidates[:top_k]]
    print(f"Broad sampling produced {len(candidates)} candidates, "
          f"keeping top {top_k} for intensive optimisation.")

    # ------------------------------------------------------------------
    # 3. Intensive optimisation on the global best seed
    best_seq = top_seeds[0].copy()
    best_score = _actual_score(best_seq)
    print(f"Initial best surrogate score = {best_score:.6f}")

    # Allocate time slices
    remaining = deadline - time.time()
    # give roughly equal thirds to Adam, ExpGrad, SGD
    third = remaining / 3.0

    # ---- Adam ----
    a_adam, sc_adam = _adam_opt(best_seq, deadline=time.time() + third,
                                max_iters=200_000, step0=0.25,
                                lam_start=5.0, lam_end=150.0)
    if sc_adam < best_score:
        best_score, best_seq = sc_adam, a_adam.copy()
        print(f"Adam improved best to {best_score:.6f}")

    # ---- Exponentiated gradient ----
    a_exp, sc_exp = _exp_opt(best_seq, deadline=time.time() + third,
                             max_iters=150_000,
                             eta0=0.7,
                             lam_start=5.0, lam_end=150.0)
    if sc_exp < best_score:
        best_score, best_seq = sc_exp, a_exp.copy()
        print(f"ExpGrad improved best to {best_score:.6f}")

    # ---- SGD ----
    a_sgd, sc_sgd = _sgd_opt(best_seq, deadline=time.time() + third,
                             max_iters=180_000,
                             step0=0.6,
                             lam_start=5.0, lam_end=150.0)
    if sc_sgd < best_score:
        best_score, best_seq = sc_sgd, a_sgd.copy()
        print(f"SGD improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Frank‑Wolfe style mass transfer (pairwise re‑allocation)
    a = best_seq.copy()
    cur_best = best_score
    it_fw = 0
    while time.time() < deadline - 2.0 and it_fw < 200_000:
        it_fw += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        delta = min(a[i_src], 1.0 - a[i_dst])
        if delta <= 0:
            break
        # move 70 % of admissible mass
        step = 0.7 * delta
        a[i_src] -= step
        a[i_dst] += step
        val = _actual_score(a)
        if val < cur_best:
            cur_best, best_seq = val, a.copy()
            print(f"[FW] iteration {it_fw} improved best to {cur_best:.6f}")

        # occasional small random jitter to avoid stagnation
        if it_fw % 5000 == 0 and np.random.rand() < 0.15:
            a = _project_onto_simplex(a + np.random.randn(a.size) * 1e-5, s=1.0)
    # ------------------------------------------------------------------
    print(f"Final best surrogate score = {cur_best:.6f} "
          f"(length={len(best_seq)})")
    return best_seq.tolist()
