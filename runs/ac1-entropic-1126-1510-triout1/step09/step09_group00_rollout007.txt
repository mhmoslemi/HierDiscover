"""
Propose a non‑negative step function (list of heights) that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

The evaluation is scaling‑invariant, so we always normalise candidates to
sum = 1.  The algorithm consists of three phases:

1. **Seed generation** – a wide variety of analytic families (uniform,
   symmetric/asymmetric Beta, power‑law, exponential, two‑block spikes,
   sparse random) are sampled for several lengths.  The best few seeds are
   retained.

2. **Smoothed mirror descent** – starting from each retained seed we run a
   gradient descent on a *soft‑max* approximation of the max‑convolution.
   The soft‑max temperature (parameter α) is raised gradually so that the
   early stage explores widely while the late stage focuses on the true
   maximum.  The update is an exponentiated‑gradient (mirror) step followed
   by a projection onto the simplex.

3. **Fine‑grained local optimisation** – a cheap Metropolis walk that moves
   a small amount of mass between random indices and accepts the move only
   if the exact objective improves.

All phases respect the global time budget (default 1000 s).  The best
vector found is returned as a plain ``list[float]``.

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Small helper – smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (size = next pow‑2)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families – return a normalised (sum = 1) density on a discrete grid
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric beta (Beta(alpha,beta))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _seed_pool_for_length(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Broad pool of analytic seeds for a given length."""
    cand = []

    # classic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_two_blocks(n, block_frac=0.05))

    # symmetric Beta – a broad log‑scale sweep
    for p in np.logspace(-2, 0.5, num=12):   # 0.01 … ~3.16
        cand.append(_distribution_beta_sym(n, float(p)))

    # asymmetric Beta – random draws on a log‑scale
    for _ in range(6):
        alpha = 10 ** rng.uniform(-2, 1)   # 0.01 … 10
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # monotone power‑law
    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))

    # monotone exponential
    for lam in (0.001, 0.005, 0.01, 0.02, 0.05):
        cand.append(_distribution_exponential_increasing(n, lam))

    # sparse random start (~1 % non‑zeros)
    cand.append(_distribution_random_sparse(n, rng, density=0.01))

    return cand


# ----------------------------------------------------------------------
# Smoothed mirror‑descent on the true max‑convolution objective
# ----------------------------------------------------------------------
def _softmax_gradient_descent(a0: np.ndarray,
                              deadline: float,
                              best_seq: np.ndarray,
                              best_score: float,
                              init_alpha: float = 5.0,
                              final_alpha: float = 150.0,
                              lr0: float = 0.5,
                              max_iter: int = 12000,
                              eval_interval: int = 300) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent using a soft‑max approximation of the max.  The temperature
    parameter α is increased linearly from ``init_alpha`` to ``final_alpha``.
    """
    n = a0.shape[0]
    a = a0.copy()
    a = a / a.sum()
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        # ----- exact convolution and exact objective -----
        conv = _fft_self_convolve(a)                # length 2n‑1
        maxb = conv.max()
        cur_score = 2.0 * n * maxb
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] iter={t} new best {best_score:.6f}")

        # ----- soft‑max weights -----
        alpha = init_alpha + (final_alpha - init_alpha) * (t / max_iter)
        # shift by max to avoid overflow
        w = np.exp(alpha * (conv - maxb))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # ----- gradient of the smooth objective -----
        # we need g_i = 2 * Σ_k w_k a_{k-i} - w_{2i} a_i
        a_rev = a[::-1]
        size = _next_pow2(3 * n - 2)                     # length for correlation
        W = np.fft.rfft(w, n=size)
        A_rev = np.fft.rfft(a_rev, n=size)
        corr = np.fft.irfft(W * A_rev, n=size)[: 3 * n - 2]   # correlation vector

        # term1 = 2 * Σ_k w_k a_{k-i}
        term1 = 2.0 * corr[n - 1 :: -1][:n]                  # reversed slice of length n

        # term2 = w_{2i} * a_i (remove double‑count on the diagonal)
        diag_idx = 2 * np.arange(n)
        term2 = np.zeros_like(a)
        mask = diag_idx < w.shape[0]
        term2[mask] = w[diag_idx[mask]] * a[mask]

        grad = term1 - term2

        # ----- mirror (exponentiated) update -----
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # re‑project onto the simplex (simple scaling suffices)
        s = a.sum()
        if s <= 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            # occasional exact evaluation printed above already
            pass

    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise mass transfer (final polishing)
# ----------------------------------------------------------------------
def _pairwise_swap_local(a0: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Tiny Metropolis walk – keep a move only if the true objective improves."""
    a = a0.copy()
    n = a.shape[0]
    while time.time() < deadline:
        src_idx = np.where(a > 1e-14)[0]
        if src_idx.size == 0:
            break
        i = np.random.choice(src_idx)

        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.30
        if delta <= 0:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best vector found (as a plain Python list of floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1.   Choose a handful of lengths (larger ⇒ finer discretisation)
    # ------------------------------------------------------------------
    lengths = [5000, 10000, 20000, 30000]

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 2.   Scan analytic families for each length
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning analytic families, length={n}")

        for cand in _seed_pool_for_length(n, rng):
            if time.time() >= deadline:
                break
            cand = cand / cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))

        # keep a modest shortlist for later optimisation
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:12]

    # ------------------------------------------------------------------
    # 3.   Add the previously discovered sequence (if any)
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best improved score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4.   If for any reason we have no candidate, fall back to uniform
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 8000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform seq, score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.   Main optimisation pipeline
    # ------------------------------------------------------------------
    # 5a – smoothed mirror descent on a few top seeds
    remaining = deadline - time.time()
    if remaining > 60:
        # pick the best few distinct seeds (by length) for intensive optimisation
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:4]]
        per_seed_time = (deadline - time.time()) / max(1, len(seeds))
        for seed_seq in seeds:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed_time
            overall_best_seq, overall_best_score = _softmax_gradient_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                init_alpha=5.0,
                final_alpha=200.0,
                lr0=0.5,
                max_iter=15000,
                eval_interval=500,
            )
            print(f"[{time.time():.1f}] after descent best score={overall_best_score:.6f}")

    # 5b – final greedy pairwise swaps / tiny simulated‑annealing fine‑tuning
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    # 6.   Return the best vector found
    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
