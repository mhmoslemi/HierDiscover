"""
Propose a non‑negative step‑function (probability vector) that
minimises ``evaluate_sequence``.
--------------------------------------------------------------------
Idea
----
* Work with the normalised problem (sum = 1).  The objective then is
  ``2·n·max(conv)`` where ``conv = a * a`` is the self‑convolution.
* Use a multi‑start scheme: generate many analytic families (uniform,
  arcsine, beta, sine‑power, Gaussian, exponential) and random
  Dirichlet draws for several lengths.
* Keep the best few candidates from the coarse scan.
* Refine each candidate with two complementary first‑order methods:
  1. **Smooth‑max mirror descent** (exponentiated‑gradient) with a
     geometric temperature schedule that ends at a huge value
     (≈10⁶) – this directly minimises a smooth approximation of the
     max‑convolution.
  2. **Projected sub‑gradient descent** on the true max‑convolution,
     with an adaptive back‑tracking line‑search.
* All updates stay in the simplex using the Euclidean projection.
* The whole search respects the supplied CPU‑time budget (default 1000 s).
* The best vector discovered is returned as a plain Python list of
  floats.
--------------------------------------------------------------------
"""

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    For a vector already normalised to sum = 1 returns the exact value
    that ``evaluate_sequence`` would produce, i.e. ``2·n·max(a*a)``.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families – each returns a vector with sum = 1

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    """Discrete Gaussian centred at (n‑1)/2 with relative width sigma."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _smooth_max_mirror_descent(a0, deadline,
                               alpha_start=1.0, alpha_end=5e5,
                               steps=8000, lr_start=0.5, lr_end=5e-4,
                               eps=1e-12):
    """
    Exponentiated‑gradient (mirror descent) minimising a smooth approximation
    of the max‑convolution.  Returns the improved vector and its score.
    """
    a = a0.copy()
    n = a.shape[0]

    # geometric schedules for temperature (alpha) and learning‑rate (lr)
    for it in range(steps):
        if time.time() >= deadline:
            break
        t = it / steps
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t

        # self‑convolution and smooth‑max weights
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth‑max (see derivation in the analysis)
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # exponentiated‑gradient step and re‑projection onto the simplex
        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()

    return a, _max_convolution_score(a)


def _projected_subgradient(a0, deadline,
                           init_step=0.5, max_iter=200000,
                           eps=1e-12):
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Uses an active‑set sub‑gradient (all indices attaining the maximum)
    and a simple back‑tracking line search.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    # Adam‑style momentum (helps in the flat regions)
    m = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        # active indices where the max is essentially attained
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam-like first‑moment update
        m = beta1 * m + (1.0 - beta1) * grad
        m_hat = m / (1.0 - beta1 ** it)

        # back‑tracking line search
        step = init_step
        while True:
            cand = a - step * m_hat
            cand = _project_onto_simplex(cand, 1.0)
            cand = np.maximum(cand, eps)
            cand /= cand.sum()
            cand_score = _max_convolution_score(cand)

            if cand_score < best_score - 1e-13 or step < 1e-9:
                break
            step *= 0.5

        a = cand
        if cand_score < best_score - 1e-13:
            best_score = cand_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {cand_score:.6f}")

        # occasional small random jitter to escape shallow plateaus
        if it % 3000 == 0:
            jitter = np.random.randn(n) * 5e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        # safeguard against over‑run
        if time.time() + 0.25 > deadline:
            break

    return best_a, best_score
