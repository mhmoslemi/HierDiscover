"""
search_step_function.py

A deterministic/heuristic search for a non‑negative step function that minimizes
the provided ``evaluate_sequence``.  The objective (with the sum of the
coefficients normalized to 1) is

    2·n·max(conv(a,a))

where ``conv`` is the ordinary (zero‑padded) convolution.  The algorithm works
in three stages:

0️⃣  Broad sampling of simple analytic families (uniform, monotone exponential,
    power‑law, random Dirichlet) to obtain a few promising seeds.

1️⃣  For each seed run a *linear‑program* sub‑routine that, given the current
    vector a and its current maximal self‑convolution M,
    maximises ``sum(b)`` under the linear constraints ``conv(a,b) ≤ M``.
    Mixing the old vector with the normalised LP solution (Frank‑Wolfe step)
    strictly reduces M.  Repeating this step a few times drives the objective
    well below the naïve uniform bound.

2️⃣  A short Adam refinement and a light simulated‑annealing polishing phase
    to escape tiny plateaus and to exploit the exact non‑smooth objective.

All loops respect a 1000 s wall‑clock budget; the best sequence seen is
returned as a plain ``list[float]``.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Surrogate score 2·n·max(conv(a,a)) (exact when sum(a)=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# 1. Linear‑program sub‑routine
def _solve_lp_max_sum(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve
        maximise   sum(b)
        subject to conv(a,b) ≤ M   (elementwise),
                    b ≥ 0 .
    Returns the optimal b (may be all‑zero on failure).
    """
    n = a.size
    rows = 2 * n - 1

    # build the constraint matrix A where (A @ b)[k] = conv(a,b)[k]
    # column j is the vector a placed starting at row j
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a

    # linprog solves a *minimisation*; we minimise –sum(b)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n
    ub = np.full(rows, M, dtype=np.float64)

    # ``highs`` is fast, deterministic and works well for pure LPs
    res = linprog(
        c,
        A_ub=A,
        b_ub=ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 5.0},
    )
    if not res.success or res.x is None:
        # fall back to a neutral vector (uniform) – should never dominate
        return np.full_like(a, 1.0 / n)

    b = np.maximum(res.x, 0.0)  # guard against tiny negatives
    return b


# ----------------------------------------------------------------------
# 2. Adam optimiser on the *exact* sub‑gradient (same code as before)
def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) on the simplex (∑a=1).
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = np.fft.irfft(
        np.fft.rfft(a[::-1]) * np.fft.rfft(w),
        n=1 << ((a.size + w.size - 1) - 1).bit_length(),
    )
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.15,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 300_000):
    """Adam on the true sub‑gradient (max‑convolution)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects a 1000 s time budget and returns the best list found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # A modest size makes the LP‑based refinement cheap while still
    # allowing enough resolution to approach the 0.75 constant.
    n = 1800

    # --------------------------------------------------------------
    # 0️⃣  Produce a few diverse seed vectors (all ℓ₁‑normalised)
    seeds = []

    # uniform
    seeds.append(np.full(n, 1.0 / n))

    # monotone exponential (left‑hand side)
    for lam in np.logspace(-4, -1, num=8):
        vals = np.exp(-lam * np.arange(n, dtype=np.float64))
        seeds.append(vals / vals.sum())

    # decreasing power‑law
    for p in [0.5, 0.8, 1.0, 1.2, 1.5, 2.0]:
        idx = np.arange(1, n + 1, dtype=np.float64)
        vals = idx ** (-p)
        seeds.append(vals / vals.sum())

    # random Dirichlet draws
    for alpha in [0.2, 0.5, 1.0, 2.0, 5.0]:
        vals = np.random.gamma(alpha, 1.0, size=n)
        seeds.append(vals / vals.sum())

    # incorporate the previous best if it exists
    prev = globals().get('height_sequence_1')
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.ndim == 1 and arr.size >= n:
                arr = arr[:n] / arr[:n].sum()
                seeds.append(arr)
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1️⃣  Keep the K most promising seeds (according to the cheap surrogate)
    scored = [(s, _max_convolution_score(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = min(5, len(scored))
    top_seeds = [s for s, _ in scored[:top_k]]
    best_seq = top_seeds[0].copy()
    best_score = _max_convolution_score(best_seq)
    print(f"[{time.time():.1f}] Initial best surrogate = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  LP‑based Frank‑Wolfe style refinement
    for idx, a_start in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] LP refinement of seed {idx + 1}/{top_k}")
        a = a_start.copy()
        for it in range(30):                               # ≤30 FW steps per seed
            if time.time() >= deadline:
                break
            M = _fft_self_convolve(a).max()
            b = _solve_lp_max_sum(a, M)
            sb = b.sum()
            if sb <= 0:
                break
            b_norm = b / sb
            # convex combination (step size 0.5 is a good compromise)
            a = 0.5 * a + 0.5 * b_norm
            a = _project_onto_simplex(a, s=1.0)            # numerical safety
            cur_score = _max_convolution_score(a)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}]  → new best {best_score:.6f} (FW iter {it+1})")
            # tiny early exit if improvement stalls
            if it > 0 and abs(cur_score - best_score) < 1e-15:
                break

        # 2b️⃣  short Adam fine‑tune of the locally improved vector
        if time.time() < deadline:
            a_adj, score_adj = _adam_descent(a, deadline,
                                             step0=0.08,
                                             max_iters=120_000)
            if score_adj < best_score - 1e-12:
                best_score = score_adj
                best_seq = a_adj.copy()
                print(f"[{time.time():.1f}]  → Adam fine‑tune improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Final light simulated‑annealing / jitter polishing
    a = best_seq.copy()
    cur_score = best_score
    temp_start = 0.02
    temp_end = 0.0005
    while time.time() < deadline:
        # linear cooling based on remaining time
        remaining = deadline - time.time()
        total = budget_s - 0.5
        t_frac = remaining / total
        temp = temp_end + (temp_start - temp_end) * t_frac

        i = np.random.randint(a.size)
        j = np.random.randint(a.size - 1)
        if j >= i:
            j += 1
        max_delta = min(a[i], 1.0 - a[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * temp

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}]  → anneal improved to {best_score:.6f}")
        else:
            # Metropolis‑type acceptance to keep the walk ergodic
            prob = np.exp(-(new_score - cur_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                a = a_new
                cur_score = new_score

    print(f"[{time.time():.1f}] Final surrogate score = {best_score:.6f}")
    return best_seq.tolist()
