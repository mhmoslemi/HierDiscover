    # random partition of n into segments
    cuts = np.sort(np.random.choice(n-1, segments-1, replace=False) + 1)
    starts = np.concatenate(([0], cuts))

"""
Enhanced search for a non‑negative step‑function that minimizes
``evaluate_sequence``.  The algorithm consists of:

1️⃣  A very broad sampling of analytical families (uniform, arcsine,
    generalized arcsine, geometric, power‑law, triangular, sine,
    cosine, beta, piece‑wise constant) over several lengths and
    parameter sweeps.  All candidates are normalised to sum = 1 and
    scored with the cheap surrogate 2·n·max(conv(a,a)).

2️⃣  The top k candidates are refined by two fast first‑order schemes:
    – a smoothed Adam optimiser acting on a log‑sum‑exp surrogate of
      the max‑convolution (gradients are obtained via FFT);
    – a projected sub‑gradient descent on the exact max‑convolution.
    The loops run only a few × 10⁴ iterations each, allowing many
    seeds to be inspected within the 1000 s budget.

3️⃣  The current best sequence is finally polished by a long simulated‑
    annealing / hill‑climbing phase that repeatedly moves a tiny
    amount of mass between random indices, occasionally accepting
    uphill moves and injecting small Gaussian jitter.

All helper functions are top‑level, use only NumPy and the provided
``evaluate_sequence`` (which is called only at the very end), and the
routine always returns a feasible list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _cheap_score(a: np.ndarray) -> float:
    """Cheap surrogate of ``evaluate_sequence`` (assumes sum(a)=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.
    Returns (grad, cur_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with the active‑mask weight vector
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for indices where the mask hits an even position
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _smooth_grad(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smoothed max using a log‑sum‑exp surrogate:
        f_eps(a) = eps * log Σ_i exp(conv_i/eps)
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)                     # length 2n‑1
    maxc = conv.max()
    w = np.exp((conv - maxc) / eps)                 # unnormalised weights
    w_sum = w.sum()
    w /= w_sum

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for the even positions (each appears twice)
    diag_indices = np.arange(conv.size)[(np.arange(conv.size) % 2) == 0] // 2
    grad[diag_indices] -= w[2 * diag_indices] * a[diag_indices]

    return grad


# ----------------------------------------------------------------------
def _smooth_adam_descent(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 150_000,
    eps0: float = 0.05,
    step0: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp smoothing of the max‑convolution."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _cheap_score(a)
    t = 0
    eps = eps0

    while time.time() < deadline and t < max_iters:
        t += 1
        grad = _smooth_grad(a, eps)

        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad ** 2)

        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        # slowly tighten the smoothing
        if t % 30_000 == 0:
            eps = max(eps * 0.5, 1e-8)

        cur = _cheap_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()

    return best_a, best_score


def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """Plain projected sub‑gradient descent with a diminishing step‑size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _cheap_score(a)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)

        cur = _cheap_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()

    return best_a, best_score

# Analytic families -------------------------------------------------------
def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _generalized_arcsine_distribution(n: int, p: float) -> np.ndarray:
    """Arcsine‑like shape with exponent p (p=1 recovers the classic arcsine)."""
    i = np.arange(n, dtype=np.float64)
    # log‑space to avoid overflow for large p
    log_w = -0.5 * p * (np.log(i + 0.5) + np.log(n - i - 0.5))
    log_w = np.clip(log_w, -700, 700)
    w = np.exp(log_w)
    if w.sum() == 0.0:
        return _uniform_distribution(n)
    return w / w.sum()


def _geometric_distribution(n: int, r: float) -> np.ndarray:
    """Decreasing geometric law a_i ∝ r**i, 0<r<1."""
    idx = np.arange(n, dtype=np.float64)
    w = r ** idx
    if w.sum() == 0.0:
        return _uniform_distribution(n)
    return w / w.sum()


def _power_law_distribution(n: int, q: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^{-q}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    w = idx ** (-q)
    if w.sum() == 0.0:
        return _uniform_distribution(n)
    return w / w.sum()


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_w = (p - 1) * (np.log(x) + np.log(1.0 - x))
    log_w = np.clip(log_w, -700, 700)
    w = np.exp(log_w)
    if w.sum() == 0.0:
        return _uniform_distribution(n)
    return w / w.sum()


def _random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    """Dirichlet draw with concentration α."""
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0.0:
        return _uniform_distribution(n)
    return x / x.sum()


def _piecewise_constant_distribution(n: int, segments: int) -> np.ndarray:
    """Random piece‑wise constant density with `segments` blocks."""
    if segments <= 1:
        return _uniform_distribution(n)
    cuts = np.sort(np.random.choice(n - 1, segments - 1, replace=False) + 1)
    starts = np.concatenate(([0], cuts))
    ends = np.concatenate((cuts, [n]))
    w = np.empty(n, dtype=np.float64)
    seg_weights = np.random.rand(segments)
    seg_weights /= seg_weights.sum()
    for s, e, weight in zip(starts, ends, seg_weights):
        length = e - s
        w[s:e] = weight / length
    if w.sum() == 0.0:
        return _uniform_distribution(n)
    return w / w.sum()

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function (non‑negative, sum > 0.01) that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget,
    prints progress information and returns the best list it discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # tiny safety margin

    # --------------------------------------------------------------
    # Choose a base length – use the length of a historic best if present.
    try:
        base_n = int(len(height_sequence_1))        # type: ignore[name-defined]
    except Exception:
        base_n = 3000

    # Lengths to explore (moderate neighbourhood + a few larger random ones)
    lengths = set()
    for f in (0.80, 0.90, 1.00, 1.10, 1.30):
        lengths.add(max(500, int(base_n * f)))
    for _ in range(10):
        lengths.add(np.random.randint(500, min(20000, base_n * 3) + 1))
    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # 1️⃣ Broad sampling
    candidates = []   # (array, cheap_score)

    # parametrisations for the families
    arcsine_ps = [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]
    geom_rs = [0.7, 0.8, 0.85, 0.9, 0.95]
    power_qs = [0.5, 0.8, 1.0, 1.2, 1.5]
    beta_ps = [0.1, 0.2, 0.5, 1.0, 2.0, 3.0]
    dirichlet_alphas = [0.03, 0.05, 0.1, 0.2, 0.5, 0.8, 1.0]
    piecewise_segs = [2, 3, 4]

    for n in lengths:
        if time.time() >= deadline:
            break

        # Uniform
        a = _uniform_distribution(n)
        candidates.append((a, _cheap_score(a)))

        # Arcsine‑like (different exponents)
        for p in arcsine_ps:
            a = _generalized_arcsine_distribution(n, p)
            candidates.append((a, _cheap_score(a)))

        # Geometric decreasing
        for r in geom_rs:
            a = _geometric_distribution(n, r)
            candidates.append((a, _cheap_score(a)))

        # Power‑law tails
        for q in power_qs:
            a = _power_law_distribution(n, q)
            candidates.append((a, _cheap_score(a)))

        # Classical families
        for gen in (_triangular_distribution, _sine_distribution,
                    _cosine_distribution, _arcsine_distribution):
            a = gen(n)
            candidates.append((a, _cheap_score(a)))

        # Symmetric beta
        for p in beta_ps:
            a = _beta_symmetric_distribution(n, p)
            candidates.append((a, _cheap_score(a)))

        # Piece‑wise constant
        for seg in piecewise_segs:
            a = _piecewise_constant_distribution(n, seg)
            candidates.append((a, _cheap_score(a)))

        # Random Dirichlet draws
        for alpha in dirichlet_alphas:
            for _ in range(2):
                a = _random_dirichlet(n, alpha)
                candidates.append((a, _cheap_score(a)))

        # Jittered arcsine (helps to escape symmetry)
        a = _arcsine_distribution(n)
        jitter = np.random.randn(n) * 4e-4
        a = a + jitter
        a = np.clip(a, 0, None)
        a = _project_onto_simplex(a, s=1.0)
        candidates.append((a, _cheap_score(a)))

    # ------------------------------------------------------------------
    # Insert the previous best solution (if available) as a seed.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        if prev.sum() > 0.0:
            prev = prev / prev.sum()
        else:
            prev = _uniform_distribution(prev.size)
        candidates.append((prev, _cheap_score(prev)))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Keep the most promising seeds.
    candidates.sort(key=lambda x: x[1])
    top_k = 30
    seeds = [c[0] for c in candidates[:top_k]]
    best_seq = seeds[0].copy()
    best_score = _cheap_score(best_seq)
    print(
        f"Broad sampling finished – {len(seeds)} seeds kept; "
        f"initial best cheap score = {best_score:.6f}"
    )

    # ------------------------------------------------------------------
    # 2️⃣ Heavy refinement (smooth‑Adam → sub‑gradient) on each seed
    for idx, seed_a in enumerate(seeds):
        if time.time() >= deadline:
            break
        # 2a – smoothed Adam
        seq, sc = _smooth_adam_descent(
            seed_a,
            deadline,
            max_iters=120_000,
            eps0=0.05,
            step0=0.18,
        )
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(seeds)} smooth‑Adam → {sc:.6f}")

        # 2b – projected sub‑gradient (quick fine‑tuning)
        seq, sc = _subgradient_descent(
            best_seq,
            deadline,
            max_iters=250_000,
            eta0=0.85,
        )
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(seeds)} sub‑grad → {sc:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Long‑run polishing – simulated‑annealing / hill‑climbing
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    iteration = 0
    max_transfer = 0.02            # at most 2 % of the total mass per move
    start_temp = 8e-3

    while time.time() < deadline:
        iteration += 1
        # ----------- propose a tiny transfer of mass ------------------
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = np.random.rand() * min(max_transfer, a[i])
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _cheap_score(a_new)

        # ----------- accept if improvement, otherwise Metropolis ----------
        if new_score < cur_score - 1e-15:
            a, cur_score = a_new, new_score
            if new_score < best_score - 1e-14:
                best_score, best_seq = new_score, a_new.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} new best {best_score:.6f}")
        else:
            # temperature decays slowly
            temp = start_temp * (0.995 ** (iteration // 4000))
            if np.random.rand() < math.exp(-(new_score - cur_score) / max(temp, 1e-12)):
                a, cur_score = a_new, new_score

        # occasional jitter to shake out of flat plateaus
        if iteration % 7_000 == 0 and np.random.rand() < 0.18:
            jitter = np.random.randn(n) * 5e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _cheap_score(a)

    print(f"Final cheap score: {best_score:.6f} (length = {len(best_seq)})")

    # The returned list is already normalised (sum = 1) – the evaluator
    # treats any scaling as equivalent, so this is a valid solution.
    return best_seq.tolist()
