assistantfinal**Solution Overview**

The goal is to find a non‑negative vector `a` (the step‑function) that minimises  

```
evaluate_sequence(a) = 2·n·max( a * a )   (when Σa = 1)
```

`a*a` is the linear self‑convolution of `a`.  
If the sum of the coefficients is forced to one, the objective is simply  

```
F(a) = 2·n·max( a*a )
```

Hence we have to make the convolution as flat as possible.

The optimiser works in three phases  

1. **Broad exploration** – a short sweep over many analytic families
   (uniform, arcsine, triangular, exponential, β‑distribution, sine‑power,
   cosine‑window, Gaussian, power‑law) and a few Dirichlet draws for several
   lengths. The best few candidates are kept.

2. **Smooth‑max Adam optimisation** – the vector is represented by logits
   `z` (`a = softmax(z)`), guaranteeing positivity and the simplex constraint.
   The smoothed maximum  

   ```
   Sα(a) = (1/α)·log Σk exp( α·( (a*a)[k] – max(a*a) ) )
   ```

   is differentiable; its gradient can be obtained by a convolution with the
   soft‑max weights `w`.  Adam updates on the logits together with an
   exponentially increasing temperature `α` rapidly decrease the true
   maximum `max(a*a)`.  

3. **Targeted hill‑climbing** – after the Adam phase a cheap greedy
   “mass‑transfer” routine repeatedly moves a small amount of probability
   from the index that contributes most to the current peak to a random
   index, accepting the move only if the peak strictly decreases.

All steps respect the 1000 s budget, print progress information and finally
return the best vector found (as a plain Python list).  The code only uses
`numpy` and the already‑provided `evaluate_sequence`.

"""
Propose a step‑function (non‑negative vector) that minimises
evaluate_sequence.  The algorithm consists of

1. broad exploration of analytic families and random Dirichlet draws,
2. Adam optimisation on the soft‑max parametrisation using a smooth‑max
   surrogate of the infinity‑norm of the self‑convolution, and
3. a final greedy hill‑climber that transfers probability mass away
   from the dominant convolution entry.

The implementation respects the 1000 s time budget, works for any
vector length, and returns the best sequence found as a list of floats.
"""

import time

# Helper utilities -------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a * a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a * b (real FFT based)."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """2·n·max(a*a) for a vector already normalised to Σa = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


def _softmax(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()

# Analytic families (return vectors already normalised to sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n: int, coeff: float = 0.5) -> np.ndarray:
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_power_law(n: int, exponent: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-exponent}  (decreasing power‑law)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = idx ** (-exponent)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step function that minimises evaluate_sequence.

    The function respects a CPU‑time budget of ``budget_s`` seconds.
    It returns the best sequence found as a plain ``list`` of floats.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    eps = 1e-12
    best_seq: np.ndarray | None = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1. Broad exploration over a few lengths and many analytic families
    lengths = [2000, 3000, 4000, 5000, 8000, 12000]

    for n in lengths:
        if time.time() >= deadline:
            break

        candidates: list[np.ndarray] = []

        # analytic families (already normalised)
        candidates.append(_distribution_uniform(n))
        candidates.append(_distribution_arcsine(n))
        candidates.append(_distribution_triangular(n))
        candidates.append(_distribution_exponential(n, lam=0.003))
        candidates.append(_distribution_exponential(n, lam=0.02))

        for p in (0.02, 0.04, 0.06, 0.08, 0.12, 0.15, 0.2,
                  0.25, 0.3, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
            candidates.append(_distribution_beta_sym(n, p))

        for pw in (0.5, 1.0, 2.0, 4.0, 8.0, 12.0):
            candidates.append(_distribution_sine_power(n, pw))

        for coeff in (0.3, 0.5, 0.7, 0.9):
            candidates.append(_distribution_cosine_window(n, coeff))

        for sigma in (0.12, 0.18, 0.25, 0.35, 0.45):
            candidates.append(_distribution_gaussian(n, sigma))

        for exp in (0.0, 0.2, 0.5, 1.0, 1.5, 2.0):
            candidates.append(_distribution_power_law(n, exp))

        # Random Dirichlet draws with a wide range of concentrations
        for alpha in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
            candidates.append(np.random.dirichlet(np.full(n, alpha)))

        # Sparse start (few non‑zeros)
        sparse = np.zeros(n, dtype=np.float64)
        k = max(2, n // 300)                     # at most a few hundred non‑zeros
        idx = np.random.choice(n, k, replace=False)
        sparse[idx] = np.random.rand(k)
        sparse /= sparse.sum()
        candidates.append(sparse)

        # Evaluate all candidates (single cheap score)
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] n={n} new init best {sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Use the globally supplied previous construction if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = len(prev)
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback – should never be needed
    if best_seq is None:
        best_n = 4000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Smooth‑max Adam optimisation on the soft‑max parametrisation
    a = best_seq.copy()
    n = a.shape[0]
    logits = np.log(a + eps)                   # initialise logits
    m = np.zeros_like(logits)                  # Adam first moment
    v = np.zeros_like(logits)                  # Adam second moment
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # schedule (total iterations are capped by the remaining time)
    max_iters = 5000
    it = 0
    while time.time() < deadline and it < max_iters:
        # ----- recompute a from logits (soft‑max) -----
        a = _softmax(logits)

        # ----- objective and best‑track -----
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            best_n = n
            print(f"[{time.time():.1f}] Adam it={it} new best {cur_score:.6f}")

        # ----- temperature (α) and learning‑rate schedules -----
        t = it / max_iters
        alpha = 1.0 * (40000.0 / 1.0) ** t          # exponential growth
        lr = 0.08 * (0.001 / 0.08) ** t            # exponential decay

        # ----- smooth‑max weights -----
        w = np.exp(alpha * (conv - cur_max))       # ≤ 1, max at the peak
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient w.r.t. a -----
        grad_a = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # ----- gradient w.r.t. logits (softmax Jacobian) -----
        grad_log = a * (grad_a - np.dot(grad_a, a))

        # ----- Adam update on logits -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

    # ------------------------------------------------------------------
    # 5. Final greedy hill‑climber (mass‑transfer) for polishing
    a = best_seq.copy()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _fft_self_convolve(a)
        M = conv.max()
        k = int(conv.argmax())                     # index of the current peak

        # indices that contribute to this peak
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = k - idx
        # find the pair with the largest product a[i]*a[j]
        prod = a[idx] * a[jdx]
        max_pair = np.argmax(prod)
        i_star, j_star = idx[max_pair], jdx[max_pair]

        # source = the larger of the two contributors
        source = i_star if a[i_star] >= a[j_star] else j_star
        target = random.randint(0, n - 1)
        if target == source:
            continue

        # transfer a modest amount of probability mass
        max_delta = min(a[source] * 0.5, 0.01)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()

        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2
            M = new_M
            new_score = 2.0 * n * M
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hill new best {new_score:.6f}")

        # occasional tiny random jitter to escape shallow plateaus
        if random.random() < 0.001:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

    # ------------------------------------------------------------------
    final_score = best_score
    print(f"Final best score ≈ {final_score:.6f} (length={best_n})")
    # Return a plain Python list (the outer judge will call evaluate_sequence)
    return best_seq.tolist()
