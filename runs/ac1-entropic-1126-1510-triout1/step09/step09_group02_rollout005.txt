def _lp_improve(a, M, target_sum=1.0, bound_g=1000.0):
    n = len(a)
    # Build constraint matrix A (2n-1, n)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(a[i])
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)  # maximize sum => minimize -sum
    bounds = [(0, bound_g) for _ in range(n)]
    res = linprog(c, A_ub=A, b_ub=M*np.ones(2*n-1), bounds=bounds, method='highs')
    if not res.success:
        return None
    g = res.x
    return g
```

"""
Propose a low‑score non‑negative step function.

The algorithm combines a very broad sampling of analytic families,
a large‑scale smoothed‑gradient descent (soft‑max approximation of
the max–convolution), a short Adam/Exponentiated‑Gradient polishing,
Frank‑Wolfe style mass‑transfer, and finally a handful of linear‑program
steps that move mass from “heavy’’ to “light’’ coordinates while keeping
the convolution bound fixed.  All work is performed on the simplex
Σa_i = 1, because the evaluation function is scale‑invariant.

The search respects a 1000 s budget and prints progress.  If a
SciPy installation is present we also solve a small LP to obtain extra
improvement; otherwise the code falls back to the pure gradient method.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact objective when Σa = 1: 2·n·max(conv(a,a))."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """Sub‑gradient of max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (i=j terms)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _double_bump_distribution(n: int,
                              block_frac: float = 0.25,
                              gap_frac: float = 0.1) -> np.ndarray:
    total_block = max(2, int(block_frac * n))
    m = total_block // 2
    g = int(gap_frac * n)
    a = np.zeros(n, dtype=np.float64)
    a[:m] = 1.0 / (2 * m)
    start2 = m + g
    if start2 + m > n:
        start2 = n - m
    a[start2:start2 + m] = 1.0 / (2 * m)
    return a


def _power_symmetric_distribution(n: int, s: float) -> np.ndarray:
    """w_i ∝ ((i+0.5)*(n-i-0.5))^{-s}  (s>0)."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-s)
    vals[np.isinf(vals)] = 0.0
    if vals.sum() == 0:
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with concentration α."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Smoothed max‑convolution gradient ------------------------------------
def _smooth_gradient(a: np.ndarray, T: float) -> np.ndarray:
    """
    Gradient of the smoothed objective  φ_T(b) = T·log Σ exp(b_i/T),
    where b = conv(a,a).  Returns ∇_a φ_T(b).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    maxb = b.max()
    # soft‑max weights, numerically stable
    w = np.exp((b - maxb) / T)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # grad = 2 * (a_rev * w) restricted to length n
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad


def _smooth_descent(a0: np.ndarray, deadline: float,
                    T0: float = 0.1, T_decay: float = 0.5,
                    steps_per_T: int = 4000) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max approximation.
    The temperature is gradually reduced.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    T = T0
    while time.time() < deadline and T > 1e-6:
        step0 = 0.4 * T            # base step size scales with temperature
        for _ in range(steps_per_T):
            if time.time() >= deadline:
                break
            grad = _smooth_gradient(a, T)
            # simple back‑tracking line‑search
            step = step0
            for _ in range(5):
                a_new = _project_onto_simplex(a - step * grad, 1.0)
                sc_new = _max_convolution_score(a_new)
                if sc_new < best_score - 1e-15:
                    best_score, best_a = sc_new, a_new.copy()
                if sc_new < _max_convolution_score(a) - 1e-12:
                    a = a_new
                    break
                step *= 0.5
        T *= T_decay
    return best_a, best_score

# Adam and exponentiated‑gradient (re‑used from the previous code) ---
def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.20,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 1_200_000):

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          eta0: float = 0.7,
                          max_iters: int = 800_000):
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score

# Linear‑program improvement (requires SciPy) ------------------------
def _lp_improve(a: np.ndarray, M: float, bound: float = 1000.0):
    """
    Solve    max Σ g_j  s.t.  a * g ≤ M (elementwise), 0 ≤ g_j ≤ bound.
    Returns the optimal g (or None on failure).
    """
    try:
        from scipy.optimize import linprog
        import scipy.sparse as sp
    except Exception:          # SciPy not available
        return None

    n = a.shape[0]
    rows = []
    cols = []
    data = []

    # Build the (2n‑1)×n constraint matrix.
    # Row k corresponds to convolution index k (0‑based).
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # i runs over indices of a that appear in this convolution term.
        for i in range(i_start, i_end + 1):
            j = k - i                     # the g‑index that pairs with a[i]
            rows.append(k)
            cols.append(j)
            data.append(a[i])

    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)                       # maximise Σ g ⇔ minimise −Σ g
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, bound) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None
    return res.x

        seq_eg, sc_eg = _exp_gradient_descent(seed_seq, deadline, max_iters=500_000)
        if sc_eg < best_score - 1e-14:
            best_score, best_seq = sc_eg, seq_eg.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} ExpGrad improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Frank‑Wolfe style mass‑transfer polishing
    a = best_seq.copy()
    cur_score = best_score
    fw_iter = 0
    while time.time() < deadline and fw_iter < 150_000:
        fw_iter += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))   # mass to move away from
        i_dst = int(np.argmin(grad))   # mass to move towards
        if i_src == i_dst:
            break
        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-15:
            break
        # try moving the whole transferable chunk (or half/quarter of it)
        improved = False
        for frac in (1.0, 0.5, 0.25):
            delta = max_transfer * frac
            a_new = a.copy()
            a_new[i_src] -= delta
            a_new[i_dst] += delta
            sc = _max_convolution_score(a_new)
            if sc < cur_score - 1e-13:
                a = a_new
                cur_score = sc
                improved = True
                break
        if improved and cur_score < best_score - 1e-14:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time():.1f}] FW iter={fw_iter} new best {best_score:.6f}")
        # occasional jitter to escape tiny plateaus
        if fw_iter % 6000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(a.size) * 3e-5
            a = _project_onto_simplex(a + jitter, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    # Linear‑program improvement (if SciPy is present)
    a = best_seq.copy()
    cur_score = best_score
    lp_iter = 0
    while time.time() < deadline and lp_iter < 8:
        b = _fft_self_convolve(a)
        M = b.max()
        g = _lp_improve(a, M)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-9:
            # no useful direction found
            break
        # scale g back to the simplex (preserves the convolution bound)
        g = g / sum_g
        # convex combo – a modest step size usually works well
        t = 0.3
        a_new = (1.0 - t) * a + t * g
        new_score = _max_convolution_score(a_new)
        if new_score < cur_score - 1e-13:
            a = a_new
            cur_score = new_score
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time():.1f}] LP iter={lp_iter} new best {best_score:.6f}")
        else:
            # if the step didn’t help, abort further LP attempts
            break
        lp_iter += 1

    # ------------------------------------------------------------------
    # Light simulated‑annealing / jitter phase (helps escape very flat regions)
    a = best_seq.copy()
    cur_score = best_score
    temp_start = 0.025
    temp_end = 0.001
    anneal_iter = 0
    while time.time() < deadline and anneal_iter < 200_000:
        anneal_iter += 1
        t_rem = deadline - time.time()
        fraction = max(0.0, t_rem / (budget_s - 0.5))
        temp = temp_end + (temp_start - temp_end) * fraction

        i = np.random.randint(a.size)
        j = np.random.randint(a.size - 1)
        if j >= i:
            j += 1
        max_delta = min(a[i], 1.0 - a[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * temp
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if cur_score < best_score - 1e-14:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time():.1f}] Anneal iter={anneal_iter} new best {best_score:.6f}")
        else:
            # Metropolis acceptance
            prob = math.exp(-(new_score - cur_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                a = a_new
                cur_score = new_score

    # ------------------------------------------------------------------
    print(f"Final best surrogate score: {best_score:.6f} (length={len(best_seq)})")
    # Return plain Python list – ``evaluate_sequence`` will handle final scaling/clipping.
    return best_seq.tolist()
