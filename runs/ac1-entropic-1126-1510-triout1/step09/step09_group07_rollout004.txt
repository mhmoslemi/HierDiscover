#!/usr/bin/env python
"""
Propose a step‑function (non‑negative, sum = 1) that minimises
``evaluate_sequence``.

Algorithm
---------

1️⃣ **Broad sweep** – Over a variety of analytical families
   (arcsine, smoothed arcsine, symmetric/asymmetric Beta,
   product‑power, asymmetric product‑power, geometric, exponential‑uniform
   mixtures) and over several lengths (up to 12 000).  For each set of
   parameters the cheap surrogate
   ``2·n·max(conv(a,a))`` (exact when ``sum(a)=1``) is evaluated.
   The best few candidates are kept for later refinement.

2️⃣ **Heavy refinement** – Each retained candidate is polished with a
   portfolio of simplex‑aware optimisation routines:
   * exponentiated‑gradient (mirror descent)
   * Adam‑style adaptive step
   * continuation on a smooth max (log‑sum‑exp)
   * heavy‑ball sub‑gradient descent
   Each routine respects a per‑method deadline and updates the global
   best if an improvement is found.

3️⃣ **Fine‑grained hill‑climbing** – Random mass transfers with an
   exponentially decaying step size, plus occasional Gaussian jitter,
   are performed until the overall time budget expires.

The whole search runs within the supplied time budget (default 1000 s)
and returns a plain list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """ Euclidean projection of v onto {x ≥ 0, sum(x)=s}. """
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a power‑of‑two FFT length."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a power‑of‑two FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap surrogate = 2·n·max(a*a)  (assumes sum(a)=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.
    Returns (grad, cur_max, active_set).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for terms where i=j (k = 2*i)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
#  Distribution families (always return a vector that sums to 1)
# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _smoothed_arcsine_distribution(n: int, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5 + eps) * (n - i - 0.5 + eps))
    return vals / vals.sum()


def _product_power_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-alpha)
    return vals / vals.sum()


def _asymmetric_product_power_distribution(
    n: int, alpha_l: float, alpha_r: float
) -> np.ndarray:
    """a_i ∝ (i+½)^{-α_l} (n‑i‑½)^{-α_r}."""
    i = np.arange(n, dtype=np.float64)
    vals = (i + 0.5) ** (-alpha_l) * (n - i - 0.5) ** (-alpha_r)
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, p: float, q: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    w = np.where(np.isfinite(w), w, 0.0)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _geometric_distribution(n: int, p: float) -> np.ndarray:
    """Left‑skewed geometric (a_i ∝ p^i, 0<p<1)."""
    i = np.arange(n, dtype=np.float64)
    vals = p ** i
    return vals / vals.sum()


def _exponential_one_sided_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i).  Normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * i)
    return vals / vals.sum()


def _mix_exponential_uniform(
    n: int, weight: float, lam: float
) -> np.ndarray:
    """
    Convex combination of a left‑exponential and the uniform distribution.
    weight ∈ [0,1] controls the exponential share.
    """
    uniform = np.full(n, 1.0 / n)
    expo = _exponential_one_sided_distribution(n, lam)
    vals = (1.0 - weight) * uniform + weight * expo
    return vals / vals.sum()


def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Γ–Dirichlet, α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()

def _exp_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    eta0: float = 1.0,
):
    """Mirror‑descent (exponentiated‑gradient) on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        a = a / s if s != 0 else np.full_like(a, 1.0 / a.size)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _adam_descent(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    step0: float = 0.3,
    beta1: float = 0.9,
    beta2: float = 0.999,
):
    """Adam‑style adaptive gradient optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _smooth_gradient(a: np.ndarray, gamma: float):
    """Weighted gradient of the soft‑max of a*a (γ controls smoothness)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    b_max = b.max()
    w = np.exp(gamma * (b - b_max))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    diag_w = w[2 * np.arange(n)]
    grad -= a * diag_w
    return grad, b_max


def _smooth_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
    gamma0: float = 2.0,
    gamma_factor: float = 1.005,
):
    """Continuation method: start with a smooth max (γ≈2) and increase γ."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    gamma = gamma0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _ = _smooth_gradient(a, gamma)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score, best_a = cur_score, a.copy()
        gamma *= gamma_factor
    return best_a, best_score


def _heavy_ball_subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.6,
    momentum: float = 0.6,
):
    """Sub‑gradient descent with a heavy‑ball momentum term."""
    a_prev = a0.copy()
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * grad + momentum * (a - a_prev)
        a_new = _project_onto_simplex(a_new, s=1.0)
        a_prev, a = a, a_new
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

        seq, sc = _adam_descent(best_seq, sub_deadline, max_iters=300_000, step0=0.35)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] Adam (seed {idx}) {best_score:.6f}")

        sub_deadline = time.time() + remaining * 0.30
        seq, sc = _smooth_gradient_descent(best_seq, sub_deadline,
                                           max_iters=400_000,
                                           eta0=0.9,
                                           gamma0=2.5,
                                           gamma_factor=1.004)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] smooth (seed {idx}) {best_score:.6f}")

        sub_deadline = time.time() + remaining * 0.15
        seq, sc = _heavy_ball_subgradient_descent(best_seq, sub_deadline,
                                                  max_iters=400_000,
                                                  eta0=0.7,
                                                  momentum=0.7)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] heavy‑ball (seed {idx}) {best_score:.6f}")

        remaining = deadline - time.time()
        if remaining <= 0.0:
            break

    # ------------------------------------------------------------------
    # 3️⃣ Fine‑grained hill‑climbing (random mass transfers + jitter)
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    it = 0
    max_transfer = 0.08                     # start with up‑to 8 % of total mass

    while time.time() < deadline:
        it += 1
        # geometric decay of the transferred amount
        transfer = max_transfer * (0.5 ** (it / 12000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = _max_convolution_score(a_new)
        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_seq, best_score = a_new.copy(), new_score
                print(f"[{time.time():.1f}] hill‑climb iter={it} new best {best_score:.6f}")

        # occasional Gaussian jitter to shake out tiny plateaus
        if it % 4000 == 0 and np.random.rand() < 0.2:
            jitter = np.random.randn(n) * 4e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
