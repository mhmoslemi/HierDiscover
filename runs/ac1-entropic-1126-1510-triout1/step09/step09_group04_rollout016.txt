assistantfinal**Solution Overview**

The optimizer builds a step‑function `a` (non‑negative, summed to 1) and
minimises  

\[
\text{score}(a)=2\,n\;\max (a*a)
\]

because `evaluate_sequence` is invariant under a global scaling of the
vector.  
The search consists of

1. **Broad initial scan** – analytic families (uniform, arcsine,
   Beta, sine‑power, Gaussian, exponential, triangular) on a set of
   lengths.  The parameter grid is much finer than in the reference
   implementation (e.g. Beta p​∈ {0.05,…,4}).

2. **Smooth‑max mirror descent** – an exponentiated‑gradient step that
   uses a soft‑max approximation of the max‑convolution.  The smoothing
   parameter `α` and the learning‑rate `η` are increased/decreased
   logarithmically over 6000 iterations.

3. **Adam‑accelerated sub‑gradient descent** – the true sub‑gradient is
   formed from the active set of maximal convolution entries and
   accumulated with Adam‑style momentum.

4. **Linear‑programming refinement** – for the current vector `f` we
   solve  

   \[
   \max\;\sum_i g_i\quad\text{s.t.}\; (f*g)_\ell\le M,\;g_i\ge0,
   \]

   where \(M=\max(f*f)\).  
   The feasible `g` is mixed into the current iterate; several
   refinements with a decreasing mixing factor are performed.

5. **Final hill‑climber** – a cheap “move‑mass” routine that repeatedly
   transfers a tiny chunk from the pair contributing most to the current
   maximum towards a random location, accepting only improvements.

All stages respect a global wall‑clock budget (default 1000 s).  The
best vector found is printed and returned as a plain Python list of
floats, ready for `evaluate_sequence`.

"""
Propose a step‑function (non‑negative list) minimising ``evaluate_sequence``.

The algorithm combines
   • a dense scan of analytic families (uniform, arcsine, Beta, sine‑power,
     Gaussian, exponential, triangular, random Dirichlet) on several
     lengths,
   • a long smooth‑max mirror‑descent (exponentiated gradient) with a
     log‑space schedule for the soft‑max temperature and learning rate,
   • Adam‑accelerated sub‑gradient descent on the true ℓ∞‑objective,
   • a few LP‑based refinements (maximise Σ gᵢ  s.t. (f∗g)≤M) and
   • a final cheap hill‑climber that moves a small amount of mass away
     from the pair causing the current maximal convolution entry.

All steps stop when the 1000 s wall‑clock budget is exhausted.
The best sequence encountered is printed and returned.
"""

import time
import random
import math

import numpy as np
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) for a probability vector (sum=1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.01) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑space to avoid overflow when p<1
    with np.errstate(divide='ignore'):
        logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if np.isnan(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """General Beta(p,q) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide='ignore'):
        logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    w = np.exp(logw)
    if np.isnan(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at (n‑1)/2."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _lp_refine(f: np.ndarray, mix_factor: float, deadline: float):
    """One LP‑based refinement step. Returns potentially improved (f, M)."""
    n = f.shape[0]
    conv_f = _fft_self_convolve(f)
    M = conv_f.max()

    # Build dense convolution matrix T such that T @ g = f * g
    col = np.concatenate([f, np.zeros(n - 1, dtype=np.float64)])
    row = np.concatenate([f[:1], np.zeros(n - 1, dtype=np.float64)])
    T = toeplitz(col, row)                     # shape (2n‑1) × n

    c = -np.ones(n, dtype=np.float64)          # maximise Σ g_i  ⇔ minimise –Σ g_i
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=(0, None), method='highs')
    except Exception as e:
        # fall back silently – the caller will just keep the old vector
        return f, M

    if not res.success:
        return f, M

    g = res.x
    gsum = g.sum()
    if gsum < 1e-12:
        return f, M

    g_norm = g / gsum

    # Try a few mixing ratios, starting from the supplied one and decreasing
    for t in (mix_factor, mix_factor * 0.5, mix_factor * 0.25,
              mix_factor * 0.1, mix_factor * 0.05):
        new_f = (1 - t) * f + t * g_norm
        new_f = np.maximum(new_f, 1e-12)
        new_f /= new_f.sum()
        new_M = _fft_self_convolve(new_f).max()
        if new_M < M - 1e-12:
            return new_f, new_M

    return f, M

        w_vec[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w_vec)[n - 1:n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step_size = 0.3 / (np.sqrt(v_hat) + eps_adam)

        a = a - step_size * m_hat
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()

        if it % 500 == 0:
            sc = _max_convolution_score(a)
            if sc < best_score - 1e-12:
                best_seq = a.copy()
                best_score = sc
                print(f"[{time.time():.1f}] Adam subgrad it={it} best {sc:.6f}")

        # a tiny safety break to allow final stages
        if time.time() + 0.5 > deadline:
            break

    # --------------------------------------------------------------
    # 6️⃣  Linear‑programming refinement (several passes)
    a = best_seq.copy()
    n = a.shape[0]
    mix = 0.2
    lp_iters = 8
    while time.time() < deadline and lp_iters > 0:
        a, M = _lp_refine(a, mix, deadline)
        sc = 2.0 * n * M
        if sc < best_score - 1e-12:
            best_seq = a.copy()
            best_score = sc
            print(f"[{time.time():.1f}] LP refine iter={lp_iters} best {sc:.6f}")
        mix = max(0.01, mix * 0.85)
        lp_iters -= 1

    # --------------------------------------------------------------
    # 7️⃣  Final cheap hill‑climber (mass‑transfer moves)
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    while time.time() < deadline:
        it += 1
        k = int(conv.argmax())

        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        best_i, best_j = i_low, k - i_low
        max_prod = a[best_i] * a[best_j]

        for i in range(i_low, i_high + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a_tmp = a.copy()
        a_tmp[source] -= delta
        a_tmp[target] += delta
        a_tmp = np.maximum(a_tmp, eps)
        a_tmp /= a_tmp.sum()
        conv_new = _fft_self_convolve(a_tmp)
        new_M = conv_new.max()
        if new_M < M - 1e-12:
            a = a_tmp
            conv = conv_new
            M = new_M
            sc = 2.0 * n * M
            if sc < best_score - 1e-12:
                best_seq = a.copy()
                best_score = sc
        else:
            # reject move
            pass

        if it % 2000 == 0:
            # occasional jitter to escape flat plateaus
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] jitter at hill iter {it}")

    final_score = 2.0 * best_seq.shape[0] * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
