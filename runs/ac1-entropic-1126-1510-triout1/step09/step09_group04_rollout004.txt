---------

   fast score  2·n·max(a*a)  is evaluated via FFT.  The best
   construction is kept.

   a long exponentiated‑gradient (smooth‑max) descent on the simplex.
   The smoothing parameter α grows exponentially (from ≈0.1 to ≈20)
   while the learning rate decays.  After every few hundred steps the
   true score is recomputed and the global optimum is updated.

3. **Simulated annealing** – a temperature‑controlled random walk that
   moves a small amount of mass from a source index to a target index.
   Moves are accepted if they improve the score or with probability
   exp(‑Δ/ T).  The temperature is geometrically cooled.

   repeatedly shifts a little mass away from the pair (i,j) that

All steps respect the global 1000 s wall‑clock budget and intermediate
best scores are printed.  The returned list is already normalised and
contains only non‑negative numbers (tiny eps is added to avoid exact

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a: np.ndarray) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a vector already normalised to sum = 1.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * float(conv.max())

def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)

    return vals

def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n

    if w.sum() == 0 or np.any(np.isinf(w)):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

        if (it % 500 == 0) or (it == total_steps - 1):
            sc = _max_convolution_score(a)
            if sc < best_score - 1e-12:
                best_seq = a.copy()
                best_score = sc
                best_n = n
                print(f"[{time.time():.1f}] mirror it={it}  best={sc:.6f}")

    # --------------------------------------------------------------
    # 4.  Simulated annealing (random mass transfers)
    a = best_seq.copy()
    cur_score = best_score
    # start temperature at about 1 % of the current score
    T = max(cur_score * 0.01, 1e-5)
    T_min = cur_score * 1e-7
    decay = 0.9995
    max_sa_iters = 15000

    for it in range(max_sa_iters):
        if time.time() >= deadline:
            break
        # biased choice of source (more mass → higher chance)
        source = np.random.choice(n, p=a)
        target = random.randint(0, n - 1)
        if target == source:
            continue

        # propose moving a modest amount of mass
        delta = min(a[source] * 0.3, 0.02) * random.random()
        if delta < eps:
            continue

        a_new = a.copy()
        a_new[source] -= delta
        a_new[target] += delta
        a_new = np.maximum(a_new, eps)
        a_new /= a_new.sum()

        new_score = _max_convolution_score(a_new)
        dE = new_score - cur_score

        if dE < 0 or random.random() < math.exp(-dE / T):
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_seq = a.copy()
                best_score = new_score
                best_n = n
                print(f"[{time.time():.1f}] SA it={it}  best={new_score:.6f}")

        # geometric cooling
        T = max(T * decay, T_min)

    # --------------------------------------------------------------
    # 5.  Final cheap hill‑climber (mass‑transfer moves)
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    hill_iters = 0
    while time.time() < deadline:
        hill_iters += 1
        # locate the convolution index attaining the current maximum
        k = int(conv.argmax())
        # restrict the pair (i,j) to indices that can produce k
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        # find the pair with the largest product a_i * a_j
        best_i, best_j = i_low, k - i_low
        max_prod = a[best_i] * a[best_j]
        for i in range(i_low, i_high + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        # move a tiny chunk from the larger contributor to a random slot
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        max_delta = min(a[source] * 0.5, 0.015)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a_snapshot = a.copy()
        a[source] -= delta
        a[target] += delta
        a = np.maximum(a, eps)
        a /= a.sum()
        conv = _fft_self_convolve(a)
        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
            sc = 2.0 * n * M
            if sc < best_score - 1e-12:
                best_seq = a.copy()
                best_score = sc
                print(f"[{time.time():.1f}] hill it={hill_iters}  best={sc:.6f}")
        else:
            a = a_snapshot
            conv = _fft_self_convolve(a)

        if hill_iters % 5000 == 0:
            # tiny jitter to shake the system
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] jitter applied at hill iter {hill_iters}")

    final_score = 2.0 * best_seq.shape[0] * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    # Ensure the returned list meets the required format
    return best_seq.tolist()
