def gradient_softmax(a, beta):
    conv = fft_self_convolve(a) (len 2n-1)
    # compute softmax weights p
    # may need to avoid overflow: subtract max conv before exponent.
    maxc = conv.max()
    exp = np.exp(beta * (conv - maxc)) # safe
    p = exp / exp.sum()
    # reverse p
    p_rev = p[::-1]
    # convolution a with p_rev
    size = 1 << ((len(a) + len(p_rev) - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    P = np.fft.rfft(p_rev, n=size)
    conv_rev = np.fft.irfft(A * P, n=size)[:len(a)+len(p_rev)-1]
    # gradient slice
    start = len(a)-1
    end = start + len(a)
    grad = conv_rev[start:end][::-1]  # shape n
    # multiply by 2 if factor required? Need gradient of (softmax sum) = sum p_k conv_k => factor 2 from derivative
    grad *= 2.0
    return grad

def grad_softmax(a, conv, beta):
    # conv is self-convolution
    maxc = conv.max()
    # compute stable softmax weights
    exp_term = np.exp(beta * (conv - maxc))
    w = exp_term / exp_term.sum()
    # reverse w
    w_rev = w[::-1]
    # convolution using FFT
    n = a.shape[0]
    size = 1 << ((n + len(w_rev) - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w_rev, n=size)
    full = np.fft.irfft(A * W, n=size)[: n + len(w_rev) - 1]
    # gradient slice
    start = n - 1
    end = start + n
    grad = full[start:end][::-1]  # shape n
    grad *= 2.0
    return grad

def _smoothed_descent(a, beta_start, beta_end, steps, lr0, deadline, best_seq, best_score):
    n = a.shape[0]
    for t in range(1, steps+1):
        if time.time() >= deadline:
            break
        # compute size for FFT: can compute each iteration but may want same size for consistency.
        size = 1 << ((2 * n - 1).bit_length())  # we need length at least 2n-1 for conv.
        # For gradient we need length n + (2n-1) -1 = 3n-2; set size accordingly:
        size = 1 << ((3 * n - 2).bit_length())
        # Compute FFT of a
        A = np.fft.rfft(a, n=size)
        # self-convolution
        conv_full = np.fft.irfft(A * A, n=size)[: 2 * n - 1]   # conv length 2n-1
        # Update beta schedule per iteration
        beta = beta_start * (beta_end / beta_start) ** (t / steps)
        # compute softmax weights w
        maxc = conv_full.max()
        exp_term = np.exp(beta * (conv_full - maxc))
        w = exp_term / exp_term.sum()
        # compute gradient using w
        w_rev = w[::-1]
        # Compute convolution a * w_rev
        # Use same size maybe for FFT: new size as length = n + len(w_rev) - 1 = n + (2n -1) -1 = 3n-2, we already have size >= 3n-2.
        W = np.fft.rfft(w_rev, n=size)
        full = np.fft.irfft(A * W, n=size)[: n + len(w_rev) - 1]
        start = n - 1
        end = start + n
        grad = full[start:end][::-1] * 2.0
        # step size
        eta = lr0 / np.sqrt(t)
        a = a * np.exp(-eta * grad)
        # renormalize
        s = a.sum()
        if s == 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s
        # occasionally compute true objective
        if t % 200 == 0 or t == steps:
            cur = _objective(a)
            if cur < best_score - 1e-15:

"""
Propose a step‑function (non‑negative vector) that minimises
evaluate_sequence() = 2·n·max(conv(a,a))/(sum a)².

Algorithm
---------

1. **Broad seed generation** – Classic analytic families (uniform,
   arcsine, symmetric/asymmetric Beta, power‑law, exponential,
   two‑block spikes, sparse random) are instantiated for several
   lengths (up to 30 000).  The best few are kept as starting points.

2. **Smoothed‑max mirror descent** – Instead of using the hard max
   convolution we minimise the *soft* maximum
   `smaxβ(a) = log‑sum‑exp(β·conv(a,a))/β`.
   For a given β the gradient is
   `∇smaxβ = 2·a *⁎ w_rev` where
   `w = softmax(β·conv)` and `w_rev` is w reversed.
   The update is an exponentiated‑gradient (mirror descent) step
   `a_i ← a_i·exp(-η·∇_i)` followed by a Euclidean projection onto the
   simplex.  β is increased exponentially from 0.2 to 20, so the method
   smoothly drives the true max down.

3. **Exact‑max refinement** – After the smoothed phase a few
   sub‑gradient steps (using the exact maximal convolution index) are
   performed to polish the result.

4. **Greedy local swaps** – A cheap “move mass from i to j’’ walk that
   accepts only improvements is finally applied.

All stages respect the overall time budget (default 1000 s).  The best
vector encountered is returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum x = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# --------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    conv = _fft_self_convolve(a)
    return 2.0 * a.shape[0] * conv.max()

# Analytic families (all produce a probability vector, i.e. sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric beta (Beta(alpha,beta))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^p (monotone increasing)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    """a_i ∝ exp(lam·i) (monotone increasing)."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Large pool of diverse seeds for a given length."""
    cand = []

    # Classic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # Symmetric Beta with a wide range of p (including extreme U‑shapes)
    for p in (0.02, 0.05, 0.1, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0, 2.0):
        cand.append(_distribution_beta_sym(n, p))

    # Asymmetric Beta – random log‑scale sampling
    for _ in range(4):
        alpha = 10 ** rng.uniform(-2, 1)   # 0.01 … 10
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # Monotone power‑law
    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))

    # Monotone exponential
    for lam in (0.001, 0.005, 0.01, 0.02, 0.05):
        cand.append(_distribution_exponential_increasing(n, lam))

    # Two‑block spikes
    for block_frac in (0.05, 0.10, 0.15):
        cand.append(_distribution_two_blocks(n, block_frac))

    # Sparse random start
    cand.append(_distribution_random_sparse(n, rng, density=0.01))

    return cand

def _gradient_softmax(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the soft‑max of the self‑convolution.
    Parameters
    ----------
    a : ndarray (n,)
        Current probability vector (sum = 1).
    conv : ndarray (2·n‑1,)
        Full self‑convolution a*a.
    beta : float
        Inverse temperature of the soft‑max.
    Returns
    -------
    grad : ndarray (n,)
        Gradient ∂/∂a of  log‑sum‑exp(β·conv) .
    """
    n = a.shape[0]

    # soft‑max weights (stable)
    maxc = conv.max()
    w = np.exp(beta * (conv - maxc))
    w /= w.sum()

    # reverse the weight vector
    w_rev = w[::-1]

    # size for FFT large enough for a (n) + w_rev (2n‑1) – 1 = 3n‑2
    size = 1 << ((3 * n - 2).bit_length())

    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w_rev, n=size)

    full = np.fft.irfft(A * W, n=size)[: n + len(w_rev) - 1]

    # slice the part that corresponds to the gradient and reverse it
    start = n - 1
    end = start + n
    grad = full[start:end][::-1] * 2.0
    return grad


# --------------------------------------------------------------
def _smoothed_descent(a0: np.ndarray,
                      beta_start: float,
                      beta_end: float,
                      steps: int,
                      lr0: float,
                      deadline: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the *soft* maximum of the self‑convolution.
    The temperature β is increased exponentially from beta_start to beta_end.
    """
    a = a0.copy()
    n = a.shape[0]

    for t in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # compute self‑convolution once per iteration
        conv = _fft_self_convolve(a)

        # temperature schedule (geometric)
        beta = beta_start * (beta_end / beta_start) ** (t / steps)

        # gradient of the smooth max
        grad = _gradient_softmax(a, conv, beta)

        # step size (mirrored exponentiated‑gradient)
        eta = lr0 / math.sqrt(t)

        # exponentiated update (guarantees non‑negativity)
        a = a * np.exp(-eta * grad)

        # Euclidean projection onto the simplex (sum = 1)
        a = _project_onto_simplex(a, s=1.0)

        # occasional true‑objective evaluation
        if t % 250 == 0 or t == steps:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.2f}] smoothed‑descent it={t} β={beta:.3f} new best {cur:.6f}")

    return best_seq, best_score


# --------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         lr0: float = 0.5,
                         max_iter: int = 15000,
                         eval_interval: int = 200) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient mirror descent on the *exact* max(conv) objective.
    """
    a = a0.copy()
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _fft_self_convolve(a)
        idx = int(conv.argmax())

        # sub‑gradient: 2·a shifted by idx
        ind = idx - np.arange(n)
        mask = (ind >= 0) & (ind < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[ind[mask]]

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.2f}] subgrad it={t} new best {cur:.6f}")

    return best_seq, best_score

def _pairwise_swap_local(a0: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer walk that accepts only improvements."""
    a = a0.copy()
    n = a.shape[0]

    while time.time() < deadline:
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        # move a random fraction (up to 30 %) of the source mass
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.2f}] swap i={i}→j={j} δ={delta:.3e} new best {cur:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence().  Returns the best vector found as a list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # small safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Choose a handful of lengths (larger ≈ more freedom)
    lengths = [8000, 12000, 20000, 30000]

    # ------------------------------------------------------------------
    # 2. Scan analytic families for each length and keep the very best seeds
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.2f}] scanning length {n}")

        for cand in _initial_candidates_extended(n, rng):
            if time.time() >= deadline:
                break
            cand = cand / cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.2f}] new global best len={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))

        # keep only the elite seeds globally (not per length)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:12]

    # ------------------------------------------------------------------
    # 3. Add a previously discovered construction if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.2f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
            top_candidates.sort(key=lambda x: x[0])
            top_candidates = top_candidates[:12]
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. If nothing was found (should never happen) fall back to uniform
    if overall_best_seq is None:
        n = 8000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.2f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Main optimisation pipeline
    #   5a – Smoothed‑max mirror descent on a few elite seeds
    remaining_time = deadline - time.time()
    if remaining_time > 30.0:
        elite = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:4]]
        for idx, seed_seq in enumerate(elite):
            if time.time() >= deadline:
                break
            # allocate a fair share of the remaining budget
            sub_deadline = time.time() + (deadline - time.time()) / max(1, len(elite) - idx)
            overall_best_seq, overall_best_score = _smoothed_descent(
                seed_seq.copy(),
                beta_start=0.20,
                beta_end=20.0,
                steps=25000,
                lr0=1.0,
                deadline=sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
            )
            print(f"[{time.time():.2f}] after smoothed descent seed {idx} best {overall_best_score:.6f}")

    #   5b – Exact‑max sub‑gradient refinement (short)
    if time.time() < deadline:
        sub_deadline = time.time() + (deadline - time.time()) * 0.3
        overall_best_seq, overall_best_score = _subgradient_descent(
            overall_best_seq.copy(),
            deadline=sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.4,
            max_iter=12000,
            eval_interval=150,
        )
        print(f"[{time.time():.2f}] after exact sub‑gradient best {overall_best_score:.6f}")

    #   5c – Final greedy pairwise swaps / tiny simulated‑annealing
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.2f}] final best score {overall_best_score:.6f} (length={len(overall_best_seq)})")

    # ------------------------------------------------------------------
    return overall_best_seq.tolist()
