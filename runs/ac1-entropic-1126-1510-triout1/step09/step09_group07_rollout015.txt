def _conv_matrix(f):
    n = f.shape[0]
    rows = 2*n - 1
    # Build data arrays for sparse matrix
    data = []
    row_ind = []
    col_ind = []
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1)

        vals = f[i_vals]
        data.extend(vals.tolist())
        row_ind.extend([k] * len(i_vals))

    return sparse.csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))

"""
Advanced search for a step function minimising evaluate_sequence.

The algorithm combines three phases:
1. Coarse sweep over several analytic families (arcsine, power‑beta,
   exponential, …) to obtain a good initial seed.
2. Iterative improvement using a linear‑programming direction
   (maximise L1‑mass of a vector g under the linear constraint f * g≤M)
   followed by a cheap line‑search over convex combinations (1−t)f+t g.
   The LP is solved with SciPy’s high‑performance ‘highs’ solver.
3. Final refinement with random hill‑climbing (tiny mass transfers) and
   optional gradient‑descent refiners from the previous baseline.

All sequences are kept on the probability simplex (sum = 1) so that the
objective reduces to 2·n·max(f∗f).  The routine respects the global time
budget and prints progress.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# --------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# --------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap surrogate = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.
    Returns (grad, cur_max, active_set).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (k = 2*i)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# --------------------------------------------------------------
# families of probability vectors
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _smoothed_arcsine_distribution(n: int, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5 + eps) * (n - i - 0.5 + eps))
    return vals / vals.sum()


def _product_power_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-alpha)
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Γ–Dirichlet, α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


# --------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """Build the (2n‑1)×n matrix representing convolution with the fixed kernel f."""
    n = f.shape[0]
    rows = 2 * n - 1
    data = []
    row_ind = []
    col_ind = []
    # loop over output index k
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k - i_vals
        vals = f[i_vals]
        data.extend(vals.tolist())
        row_ind.extend([k] * len(i_vals))
        col_ind.extend(j_vals.tolist())
    return sparse.csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))


def _lp_direction(f: np.ndarray, M: float):
    """Solve the LP max 1ᵀg s.t. f∗g ≤ M, g≥0. Returns g (not normalised) or None."""
    n = f.shape[0]
    A = _conv_matrix(f)  # sparse (2n‑1)×n
    # objective: maximise sum(g) → minimise -sum(g)
    c = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    # SciPy’s high‑level solver handles sparse A_ub
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


# --------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 150_000, eta0: float = 0.8):
    """Mirror‑descent (exponentiated‑gradient) on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        a = a / s if s != 0 else np.full_like(a, 1.0 / a.size)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 150_000, step0: float = 0.2,
                  beta1: float = 0.9, beta2: float = 0.999):
    """Adam‑style adaptive gradient optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _smooth_gradient_descent(a0: np.ndarray, deadline: float,
                             max_iters: int = 300_000,
                             eta0: float = 0.7,
                             gamma0: float = 5.0,
                             gamma_factor: float = 1.001):
    """Continuation method: start with a smooth max (γ≈5) and increase γ."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    gamma = gamma0
    while time.time() < deadline and it < max_iters:
        it += 1
        # weighted gradient of a smoothed max
        n = a.shape[0]
        b = _fft_self_convolve(a)
        b_max = b.max()
        w = np.exp(gamma * (b - b_max))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
        diag_w = w[2 * np.arange(n)]
        grad -= a * diag_w

        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score, best_a = cur_score, a.copy()
        gamma *= gamma_factor
    return best_a, best_score


def _heavy_ball_subgradient_descent(a0: np.ndarray, deadline: float,
                                   max_iters: int = 300_000,
                                   eta0: float = 0.6, momentum: float = 0.6):
    a_prev = a0.copy()
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * grad + momentum * (a - a_prev)
        a_new = _project_onto_simplex(a_new, s=1.0)
        a_prev, a = a, a_new
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# --------------------------------------------------------------
def _coarse_seeds(n: int):
    """Yield a handful of analytically‑motivated seeds for length n."""
    # arcsine and smoothed arcsine
    yield _arcsine_distribution(n)
    for eps in (0.1, 0.5, 1.0, 2.0):
        yield _smoothed_arcsine_distribution(n, eps)

    # product‑power family
    for alpha in np.arange(0.30, 0.71, 0.05):
        yield _product_power_distribution(n, float(alpha))

    # symmetric beta
    for p in np.arange(0.30, 0.81, 0.05):
        yield _beta_symmetric_distribution(n, float(p))

    # exponential tails
    for lam in (0.001, 0.003, 0.005, 0.010):
        yield _exponential_distribution(n, lam)

    # a few random Dirichlet draws
    for _ in range(2):
        yield _random_dirichlet(n)

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine is deterministic (apart from the seed) and respects the global
    time budget.  It returns a list of floats ready to be evaluated.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # safety margin

    # ------------------------------------------------------------------
    # Choose a set of lengths to explore
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
        if base_n < 500:
            base_n = 2000
    except Exception:
        base_n = 2000

    lengths = [int(base_n * f) for f in (0.75, 1.0, 1.25)]
    lengths = [max(500, n) for n in lengths]

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 1️⃣ Coarse sweep over analytic families
    for n in lengths:
        if time.time() >= deadline:
            break
        for cand in _coarse_seeds(n):
            score = _max_convolution_score(cand)
            if score < best_score:
                best_score, best_seq = score, cand.copy()
    # incorporate previous best if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            score = _max_convolution_score(prev)
            if score < best_score:
                best_score, best_seq = score, prev.copy()
    except Exception:
        pass

    print(f"[{time.time():.1f}] after coarse sweep best cheap={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Iterative LP‑driven refinement
    # Allocate roughly half of the remaining budget to the LP loop.
    while time.time() < deadline - 5.0:
        # compute current M and build LP direction
        conv_f = _fft_self_convolve(best_seq)
        M = conv_f.max()
        # solve LP
        g_raw = _lp_direction(best_seq, M)
        if g_raw is None:
            print("LP failed – exiting LP loop")
            break
        total = g_raw.sum()
        if total < 1e-12:
            # Direction does not provide any mass – stop LP loop
            break
        g = g_raw / total     # normalise to sum=1

        improved = False
        # try a few mixing coefficients
        for t in (0.15, 0.30, 0.45, 0.60):
            cand = (1.0 - t) * best_seq + t * g
            cand = _project_onto_simplex(cand, s=1.0)  # ensure feasibility
            score = _max_convolution_score(cand)
            if score < best_score - 1e-12:
                best_seq, best_score = cand, score
                improved = True
                print(f"[{time.time():.1f}] LP mix t={t:.2f} new cheap={best_score:.6f}")
                break   # accept first improvement and recompute direction
        if not improved:
            # No improvement with mixing; stop the LP loop
            break

    # ------------------------------------------------------------------
    # 3️⃣ Heavy refinement (gradient‑type methods) – truncated time budget
    remaining = deadline - time.time()
    if remaining > 0.0:
        # split remaining budget among a few refiners
        sub_deadline = time.time() + remaining * 0.25
        seq, sc = _exp_gradient_descent(best_seq, sub_deadline,
                                        max_iters=120_000, eta0=1.0)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time():.1f}] after exp‑grad   {best_score:.6f}")

        sub_deadline = time.time() + remaining * 0.25
        seq, sc = _adam_descent(best_seq, sub_deadline,
                                max_iters=120_000, step0=0.25)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time():.1f}] after Adam      {best_score:.6f}")

        sub_deadline = time.time() + remaining * 0.30
        seq, sc = _smooth_gradient_descent(best_seq, sub_deadline,
                                            max_iters=200_000,
                                            eta0=0.8,
                                            gamma0=2.0,
                                            gamma_factor=1.003)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time():.1f}] after smooth   {best_score:.6f}")

        sub_deadline = time.time() + remaining * 0.20
        seq, sc = _heavy_ball_subgradient_descent(best_seq, sub_deadline,
                                                  max_iters=200_000,
                                                  eta0=0.5, momentum=0.5)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time():.1f}] after heavy‑ball {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Fine‑grained hill‑climbing
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    it = 0
    max_transfer = 0.05

    while time.time() < deadline:
        it += 1
        transfer = max_transfer * (0.5 ** (it / 8000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_seq, best_score = a_new.copy(), new_score
                print(f"[{time.time():.1f}] hill‑climb iter={it} new best {best_score:.6f}")

        if it % 6000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 4e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()

"""
Advanced search for a step function minimising ``evaluate_sequence``.

The algorithm proceeds in four phases:

1️⃣  Coarse sweep over several analytic families (arcsine, power‑Beta,
    exponential, random Dirichlet) to obtain a good seed.

2️⃣  Iterative LP‑driven refinement.  For the current f we compute
    M = max(f∗f) and solve the linear program

        maximise    1ᵀg
        subject to   f∗g ≤ M ,   g ≥ 0 .

    The solution g (after normalisation) is mixed with f via a small
    convex combination (1‑t)f + t g; if the cheap objective
    2·n·max((1‑t)f+t g) improves we accept the move and recompute the LP
    direction.  This step often escapes the local minima that gradient
    descent gets stuck in.

3️⃣  Heavy refinement with a handful of gradient‑type methods
    (exponentiated‑gradient, Adam, smooth‑max continuation, heavy‑ball
    sub‑gradient).  The time budget is split among them.

4️⃣  Final fine‑grained hill‑climbing (tiny random mass transfers and
    occasional jitter).

All sequences are kept on the probability simplex (sum = 1), so the
objective reduces to ``2·n·max(conv)`` where ``conv`` is the full
self‑convolution.  The routine respects the global time budget and
returns a list ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# --------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# --------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap surrogate = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (k = 2*i)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

# families of probability vectors
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _smoothed_arcsine_distribution(n: int, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5 + eps) * (n - i - 0.5 + eps))
    return vals / vals.sum()


def _product_power_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-alpha)
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Γ–Dirichlet, α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()

def _lp_direction(f: np.ndarray, M: float):
    """
    Solve  max 1ᵀg  s.t.  f∗g ≤ M,  g ≥ 0.
    Returns the raw solution vector (may be un‑normalised) or None.
    """
    n = f.shape[0]
    rows = 2 * n - 1

    # Build the dense convolution matrix A where A[k, j] = f_{k-j} (if in range)
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f

    c = -np.ones(n)                 # maximise sum(g)  → minimise -sum(g)
    b_ub = M * np.ones(rows)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 150_000, eta0: float = 0.8):
    """Mirror‑descent (exponentiated‑gradient) on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        a = a / s if s != 0 else np.full_like(a, 1.0 / a.size)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 150_000, step0: float = 0.2,
                  beta1: float = 0.9, beta2: float = 0.999):
    """Adam‑style adaptive gradient optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _smooth_gradient_descent(a0: np.ndarray, deadline: float,
                             max_iters: int = 300_000,
                             eta0: float = 0.7,
                             gamma0: float = 5.0,
                             gamma_factor: float = 1.001):
    """Continuation method for a smooth approximation of the max."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    gamma = gamma0
    while time.time() < deadline and it < max_iters:
        it += 1
        n = a.shape[0]
        b = _fft_self_convolve(a)
        b_max = b.max()
        w = np.exp(gamma * (b - b_max))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
        diag_w = w[2 * np.arange(n)]
        grad -= a * diag_w

        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score, best_a = cur_score, a.copy()
        gamma *= gamma_factor
    return best_a, best_score


def _heavy_ball_subgradient_descent(a0: np.ndarray, deadline: float,
                                   max_iters: int = 300_000,
                                   eta0: float = 0.6, momentum: float = 0.6):
    """Sub‑gradient descent with a heavy‑ball momentum term."""
    a_prev = a0.copy()
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * grad + momentum * (a - a_prev)
        a_new = _project_onto_simplex(a_new, s=1.0)
        a_prev, a = a, a_new
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

def _coarse_seeds(n: int):

        g_raw = _lp_direction(best_seq, M)
        if g_raw is None:
            # LP infeasible or unbounded – give up on this phase
            break
        total = g_raw.sum()
        if total < 1e-12:
            break
        g = g_raw / total                 # normalise to sum = 1

        improved = False
        # try a small set of mixing coefficients
        for t in (0.15, 0.30, 0.45, 0.60):
            cand = (1.0 - t) * best_seq + t * g
            cand = _project_onto_simplex(cand, s=1.0)
            score = _max_convolution_score(cand)
            if score < best_score - 1e-12:
                best_seq, best_score = cand, score
                improved = True
                print(f"[{time.time():.1f}] LP mix t={t:.2f} new cheap={best_score:.6f}")
                break
        if not improved:
            # no improvement possible with the current direction
            break

    # ------------------- 3️⃣ Heavy gradient‑type refinement -------
    remaining = deadline - time.time()
    if remaining > 0.0:
        # 1) exponentiated‑gradient
        sub_deadline = time.time() + remaining * 0.25
        seq, sc = _exp_gradient_descent(best_seq, sub_deadline,
                                        max_iters=120_000, eta0=1.0)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time():.1f}] after exp‑grad   {best_score:.6f}")

        # 2) Adam
        sub_deadline = time.time() + remaining * 0.25
        seq, sc = _adam_descent(best_seq, sub_deadline,
                                max_iters=120_000, step0=0.25)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time():.1f}] after Adam      {best_score:.6f}")

        # 3) smooth‑max continuation
        sub_deadline = time.time() + remaining * 0.30
        seq, sc = _smooth_gradient_descent(best_seq, sub_deadline,
                                            max_iters=200_000,
                                            eta0=0.8,
                                            gamma0=2.0,
                                            gamma_factor=1.003)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time():.1f}] after smooth    {best_score:.6f}")

        # 4) heavy‑ball sub‑gradient
        sub_deadline = time.time() + remaining * 0.20
        seq, sc = _heavy_ball_subgradient_descent(best_seq, sub_deadline,
                                                  max_iters=200_000,
                                                  eta0=0.5, momentum=0.5)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time():.1f}] after heavy‑ball {best_score:.6f}")

    # ------------------- 4️⃣ fine‑grained hill‑climbing ----------
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    it = 0
    max_transfer = 0.05

    while time.time() < deadline:
        it += 1
        transfer = max_transfer * (0.5 ** (it / 8000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_seq, best_score = a_new.copy(), new_score
                print(f"[{time.time():.1f}] hill‑climb iter={it} new best {best_score:.6f}")

        if it % 6000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 4e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
