"""
Propose a step‐function (non‑negative list) that (approximately) minimizes
``evaluate_sequence``.  The algorithm follows three phases:

1️⃣  *Broad sampling* – generate many analytic families (arcsine, beta,
    power‑law, exponential, …) and random Dirichlet draws, all projected onto
    the simplex (∑a=1).  Keep the top few candidates (by the cheap surrogate
    ``2·n·max(conv(a,a))``).

2️⃣  *Local refinement* – for each seed run three first‑order methods:
    Adam on the true sub‑gradient, exponentiated‑gradient descent, and Adam on
    an ``L^p`` smooth approximation (p=8).  The best result among all seeds
    is retained.

3️⃣  *Global polishing* – a Frank–Wolfe style pairwise mass‑transfer move,
    followed by an extended simulated‑annealing/jitter phase, both
    respecting the remaining time budget.

All operations stay on the probability simplex, therefore the invariant
``evaluate_sequence(seq) == 2·n·max(conv(seq,seq))`` holds.  The routine
honours a 1000 s wall‑clock limit and prints intermediate scores.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap surrogate: 2·n·max(conv(a,a)) for a simplex (∑a=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of ``max(conv(a,a))`` on the unit simplex.
    Returns (grad, cur_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices (self‑product terms)
    even = (active % 2 == 0)
    if np.any(even):
        diag_idx = (active[even] // 2)
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.30,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 1_000_000):
    """Adam on the exact sub‑gradient of ``max(conv(a,a))``."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          eta0: float = 0.80,
                          max_iters: int = 600_000):
    """Exponentiated‑gradient (KL‑mirror) descent."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


def _smooth_obj_and_grad(a: np.ndarray, p: float = 8.0):
    """
    Smoothed ``L^p`` approximation of ``max(conv(a,a))``.
    Returns (value, grad) where ``value = (∑ conv^p)^{1/p}``.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    b_pow = b ** p
    S = b_pow.sum()
    if S == 0.0:
        return 0.0, np.zeros_like(a)
    val = S ** (1.0 / p)
    factor = S ** (1.0 / p - 1.0)          # = S^{(1‑p)/p}
    w = factor * (b ** (p - 1))
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return val, grad


def _adam_smooth_descent(a0: np.ndarray, deadline: float,
                         p: float = 8.0,
                         step0: float = 0.30,
                         beta1: float = 0.9,
                         beta2: float = 0.999,
                         max_iters: int = 600_000):
    """Adam on the smoothed L^p objective."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        _, grad = _smooth_obj_and_grad(a, p)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score

def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with concentration ``alpha``."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Analytic families
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _sparse_uniform_distribution(n: int, fraction: float = 0.1) -> np.ndarray:
    """Place equal mass on a random subset of the indices."""
    k = max(2, int(n * fraction))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return a

            break
        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-14:
            break

        improved = False
        # try three fractions: 1, 0.5, 0.25
        for frac in (1.0, 0.5, 0.25):
            delta = max_transfer * frac
            a_candidate = a.copy()
            a_candidate[i_src] -= delta
            a_candidate[i_dst] += delta
            cand_score = _max_convolution_score(a_candidate)
            if cand_score < cur_score - 1e-13:
                a = a_candidate
                cur_score = cand_score
                improved = True
                break
        if improved:
            if cur_score < best_score - 1e-14:
                best_seq, best_score = a.copy(), cur_score
                print(f"[{time.time():.1f}] FW iter={fw_iter} new best {best_score:.6f}")
            continue

        # occasional jitter to avoid stagnation
        if fw_iter % 5000 == 0 and np.random.rand() < 0.2:
            jitter = np.random.randn(a.size) * 1e-5
            a = _project_onto_simplex(a + jitter, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    # 7️⃣  Extended simulated‑annealing / jitter phase
    a = best_seq.copy()
    cur_score = best_score
    temp_start = 0.08
    temp_end = 0.001
    anneal_iter = 0
    while time.time() < deadline and anneal_iter < 300_000:
        anneal_iter += 1
        remaining = deadline - time.time()
        total = budget_s - 0.5
        t_frac = max(0.0, remaining / total)
        temp = temp_end + (temp_start - temp_end) * t_frac

        i = np.random.randint(a.size)
        j = np.random.randint(a.size - 1)
        if j >= i:
            j += 1
        max_delta = min(a[i], 1.0 - a[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * temp

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if cur_score < best_score - 1e-14:
                best_seq, best_score = a.copy(), cur_score
                print(f"[{time.time():.1f}] Anneal iter={anneal_iter} new best {best_score:.6f}")
        else:
            prob = np.exp(-(new_score - cur_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                a = a_new
                cur_score = new_score

    print(f"Final best surrogate score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
