def lp_step(f, M, deadline):
    n = len(f)
    # Build A (2n-1) x n sparse matrix: each row i has f[i - j] for j where 0 <= i - j < n
    data = []
    rows = []
    cols = []
    for i in range(2*n-1):
        j_start = max(0, i-(n-1))
        j_end = min(i, n-1)
        # col = k = j, coefficient f[i - j]
        # loop over k
        for k in range(j_start, j_end+1):
            coeff = f[i - k]
            rows.append(i)
            cols.append(k)
            data.append(coeff)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # inequality: A * g <= M * ones
    # g >= 0
    # objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)  # minimize -sum = maximize sum
    bounds = [(0, None)] * n
    ub = np.full(2*n-1, M)
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                                 method='highs', options={'disp': False})
    if not res.success:
        return None
    g = res.x
    return g
```

"""
Propose a non‑negative step‑function that minimises
``evaluate_sequence``.

The method builds on the previous “three‑phase” pipeline
(cheap sweep → heavy continuous optimisation → hill‑climbing) but
adds a few important upgrades:

*     The sweep now covers **triangular, sine, cosine** families and a
      **mixture of arcsine & cosine** – these were missing in the
      original code and provide better seeds.
*     Inside the heavy refinement the time budget is split more
      aggressively among the four gradient‑based updates and a **linear‑
      programming refinement** (the LP described in the paper) is
      performed after the gradient stage.  The LP enforces the bound
      `(f*g)_k ≤ M` and moves the current vector towards a feasible
      direction that lowers the maximum convolution.
*     The optimisation runs for a few more iterations (the step size
      decays as `1/√t` as before) and all stages respect the global
      deadline.
*     The final hill‑climbing stage now uses a slowly decaying
      transfer amount and occasional Gaussian jitter, which is often
      enough to capture the last few‑hundredths of improvement.

All helper routines are top‑level; the function is deterministic
apart from the supplied seed.

The returned list can be fed straight into ``evaluate_sequence``.
"""

import time
import numpy as np

from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap surrogate = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.
    Returns (grad, cur_max, active_set).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution with the indicator of the active set (reversed a)
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for indices where k = 2*i
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

# ----- families of probability vectors ---------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _smoothed_arcsine_distribution(n: int, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5 + eps) * (n - i - 0.5 + eps))
    return vals / vals.sum()


def _product_power_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-alpha)
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, p: float, q: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    w = np.where(np.isfinite(w), w, 0.0)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _random_dirichlet(n: int) -> np.ndarray:
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


def _mix_two(a: np.ndarray, b: np.ndarray, w: float) -> np.ndarray:
    """Linear mixture a*w + b*(1-w) then renormalise."""
    mixed = w * a + (1.0 - w) * b
    mixed = mixed / mixed.sum()
    return mixed

def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 200_000, eta0: float = 1.0):
    """Mirror‑descent (exponentiated‑gradient) on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = a / a.sum()
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 200_000, step0: float = 0.3,
                  beta1: float = 0.9, beta2: float = 0.999):
    """Adam‑style adaptive gradient optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _smooth_gradient_descent(a0: np.ndarray, deadline: float,
                             max_iters: int = 300_000,
                             eta0: float = 0.8,
                             gamma0: float = 2.0,
                             gamma_factor: float = 1.005):
    """Continuation method: start with a smooth max (γ≈2) and increase γ."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    gamma = gamma0
    while time.time() < deadline and it < max_iters:
        it += 1
        # smooth gradient of the soft‑max of a*a
        n = a.shape[0]
        b = _fft_self_convolve(a)
        b_max = b.max()
        w = np.exp(gamma * (b - b_max))
        if w.sum() == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w.sum()
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
        diag_w = w[2 * np.arange(n)]  # entries where k = 2*i
        grad -= a * diag_w

        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score, best_a = cur_score, a.copy()
        gamma *= gamma_factor
    return best_a, best_score


def _heavy_ball_subgradient_descent(a0: np.ndarray, deadline: float,
                                   max_iters: int = 300_000,
                                   eta0: float = 0.6, momentum: float = 0.6):
    """Sub‑gradient descent with a heavy‑ball momentum term."""
    a_prev = a0.copy()
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * grad + momentum * (a - a_prev)
        a_new = _project_onto_simplex(a_new, s=1.0)
        a_prev, a = a, a_new
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

# ----- linear‑programming refinement ------------------------------------
def _lp_refine(a: np.ndarray, deadline: float):
    """
    Perform a few linear‑programming updates as described in the
    “iteration” paragraph of the statement.  Each step solves

        max 1ᵀg
        s.t. f*g ≤ M,   g ≥ 0,

    then mixes the result back into f.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution_score(best_a)

    while time.time() < deadline:
        # current bound
        M = _fft_self_convolve(best_a).max()
        # Build convolution matrix A (shape (2n‑1, n)):
        #   (A g)_k = Σ_j best_a[j] * g_{k-j}
        #   which is a Toeplitz matrix with diagonals = best_a.
        offsets = -np.arange(n)                     # -0, -1, …, -(n‑1)
        data = [np.full(n, best_a[i]) for i in range(n)]
        A = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csc')

        # Linear programme: maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n)
        ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        # Use HiGHS interior‑point solver (fast for moderate sizes)
        res = linprog(
            c,
            A_ub=A,
            b_ub=ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            break

        g = res.x
        if g.sum() == 0:
            break
        g = g / g.sum()               # renormalise to sum = 1
        # convex combination – the original paper uses a small t
        t = 0.25
        new_a = (1.0 - t) * best_a + t * g
        new_a = _project_onto_simplex(new_a, s=1.0)

        new_score = _max_convolution_score(new_a)
        if new_score < best_score - 1e-15:
            best_a, best_score = new_a, new_score
            print(f"[{time.time():.1f}] LP refinement improved to {best_score:.6f}")
        else:
            # no progress – stop early
            break
    return best_a, best_score

        for eps in (0.0, 0.1, 0.5, 1.0, 2.0, 5.0):
            a = _arcsine_distribution(n) if eps == 0.0 else _smoothed_arcsine_distribution(n, eps)
            score = _max_convolution_score(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()

        # product‑power
        for alpha in np.arange(0.25, 0.81, 0.07):
            a = _product_power_distribution(n, alpha)
            score = _max_convolution_score(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()

        # symmetric beta
        for p in np.arange(0.20, 1.01, 0.10):
            a = _beta_symmetric_distribution(n, p)
            score = _max_convolution_score(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()

        # asymmetric beta (a few extreme combos)
        for (p, q) in ((0.2, 0.8), (0.3, 0.7), (0.8, 0.2)):
            a = _beta_asymmetric_distribution(n, p, q)
            score = _max_convolution_score(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()

        # triangular, sine, cosine
        for fam in (_triangular_distribution, _sine_distribution, _cosine_distribution):
            a = fam(n)
            score = _max_convolution_score(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()

        # exponential tails
        for lam in (0.001, 0.003, 0.005, 0.010):
            a = _exponential_distribution(n, lam)
            score = _max_convolution_score(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()

        # mixture of arcsine and cosine (weights 0.2‑0.8)
        a_arc = _arcsine_distribution(n)
        a_cos = _cosine_distribution(n)
        for w in (0.2, 0.4, 0.6, 0.8):
            a = _mix_two(a_arc, a_cos, w)
            score = _max_convolution_score(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()

        # a few random Dirichlet draws
        for _ in range(5):
            a = _random_dirichlet(n)
            score = _max_convolution_score(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()

    # include the previously best construction, if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            score = _max_convolution_score(prev)
            if score < best_score:
                best_score, best_seq = score, prev.copy()
    except Exception:
        pass

    print(f"[{time.time():.1f}] after coarse sweep best cheap={best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ Heavy refinement – allocate the bulk of the remaining budget
    remaining = deadline - time.time()
    if remaining <= 0.0:
        return best_seq.tolist()

    # split remaining time among optimiser blocks
    # 1) exponentiated‑gradient
    sub_deadline = time.time() + remaining * 0.15
    seq, sc = _exp_gradient_descent(best_seq, sub_deadline,
                                    max_iters=300_000, eta0=1.2)
    if sc < best_score:
        best_seq, best_score = seq, sc
        print(f"[{time.time():.1f}] after exp‑grad   {best_score:.6f}")

    # 2) Adam
    sub_deadline = time.time() + remaining * 0.15
    seq, sc = _adam_descent(best_seq, sub_deadline,
                            max_iters=300_000, step0=0.35)
    if sc < best_score:
        best_seq, best_score = seq, sc
        print(f"[{time.time():.1f}] after Adam      {best_score:.6f}")

    # 3) smooth‑max continuation
    sub_deadline = time.time() + remaining * 0.25
    seq, sc = _smooth_gradient_descent(best_seq, sub_deadline,
                                       max_iters=400_000,
                                       eta0=0.9,
                                       gamma0=2.5,
                                       gamma_factor=1.004)
    if sc < best_score:
        best_seq, best_score = seq, sc
        print(f"[{time.time():.1f}] after smooth   {best_score:.6f}")

    # 4) heavy‑ball sub‑gradient
    sub_deadline = time.time() + remaining * 0.15
    seq, sc = _heavy_ball_subgradient_descent(best_seq, sub_deadline,
                                              max_iters=400_000,
                                              eta0=0.7, momentum=0.65)
    if sc < best_score:
        best_seq, best_score = seq, sc
        print(f"[{time.time():.1f}] after heavy‑ball {best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣ Linear‑programming refinement (if time permits)
    remaining = deadline - time.time()
    if remaining > 10.0:                     # need a few seconds for the LP
        lp_deadline = time.time() + remaining * 0.30
        seq, sc = _lp_refine(best_seq, lp_deadline)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time():.1f}] after LP step   {best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣ Fine‑grained hill‑climbing
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    it = 0
    max_transfer = 0.04  # start with 4 % of total mass

    while time.time() < deadline:
        it += 1
        # geometric decay of transferred amount
        transfer = max_transfer * (0.5 ** (it / 9000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_seq, best_score = a_new.copy(), new_score
                print(
                    f"[{time.time():.1f}] hill‑climb iter={it} new best {best_score:.6f}"
                )

        # occasional jitter to escape tiny plateaus
        if it % 6000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 6e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
