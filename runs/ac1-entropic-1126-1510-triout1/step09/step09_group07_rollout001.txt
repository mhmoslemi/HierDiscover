"""
Improved step‑function optimizer.

The algorithm follows three stages:

1️⃣  **Broad sweep** – generate many candidate probability vectors from
    a rich collection of analytical families (arcsine, smoothed arcsine,
    product‑power, Beta, Gaussian, exponential, uniform and simple
    mixtures).  For each choice of length (≈ 0.9×, 1×, 1.1× and 1.3× a
    reference length) the cheap surrogate 2·n·max(a*a) (exact for a ∈ Δ₁)
    is evaluated and the best few seeds are kept.

2️⃣  **Intensive refinement** – each of the top‑k seeds is polished by a
    projected sub‑gradient descent on the true objective (the same
    surrogate, because the sum is kept equal to 1).  The sub‑gradient is
    computed analytically (see ``_subgrad``).  A simple adaptive step‑size
    and a momentum term are employed.  The refinement runs for the
    majority of the remaining time budget.

3️⃣  **Fine‑grained hill‑climbing** – a stochastic local‑search that moves
    tiny amounts of probability mass between random coordinates.  The
    step size decays geometrically, and occasional “jitter’’ steps help
    to escape shallow plateaus.

All vectors are projected onto the probability simplex after every
update, guaranteeing admissibility for ``evaluate_sequence``.  The final
candidate is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap surrogate = 2·n·max(a*a) (valid because sum(a)=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.
    Returns (grad, cur_max, active_set).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (k = 2*i)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
#  Families of probability vectors (all return a vector that sums to 1)
# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _smoothed_arcsine_distribution(n: int, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5 + eps) * (n - i - 0.5 + eps))
    return vals / vals.sum()


def _product_power_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-alpha)
    # avoid division by zero when alpha is huge
    if np.any(np.isinf(vals)):
        vals = np.where(np.isfinite(vals), vals, 0.0)
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w = w / w.sum()
    return w


def _gaussian_distribution(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _mix_two(a: np.ndarray, b: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture lam*a + (1‑lam)*b (both already normalized)."""
    mix = lam * a + (1.0 - lam) * b
    # Renormalise only to cure possible rounding errors
    s = mix.sum()
    if s == 0:
        return _uniform_distribution(len(a))
    return mix / s


def _random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a Dirichlet point with concentration `conc`."""
    x = np.random.gamma(conc, 1.0, size=n)
    s = x.sum()
    return x / s if s > 0 else _uniform_distribution(n)


# ----------------------------------------------------------------------
#  Optimisation primitives
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float,
                         eta0: float = 0.8, max_iters: int = 500_000,
                         momentum: float = 0.6) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with simple momentum."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    a_prev = a.copy()
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        # Diminishing step size
        eta = eta0 / np.sqrt(it)
        # Momentum update
        a_new = a - eta * grad + momentum * (a - a_prev)
        a_new = _project_onto_simplex(a_new, s=1.0)
        a_prev, a = a, a_new
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-14:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _hill_climb(a0: np.ndarray, deadline: float,
                init_transfer: float = 0.05) -> tuple[np.ndarray, float]:
    """Tiny random mass transfers with geometrically decaying step size."""
    a = a0.copy()
    cur_score = _max_convolution_score(a)
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it += 1
        # decay the allowed transfer
        transfer = init_transfer * (0.5 ** (it / 8000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a, cur_score = a_new, new_score
            if new_score < best_score - 1e-14:
                best_a, best_score = a_new.copy(), new_score
                print(f"[{time.time():.1f}] hill‑climb iter={it} new best {best_score:.6f}")

        # occasional jitter to shake out tiny plateaus
        if it % 6000 == 0 and np.random.rand() < 0.2:
            jitter = np.random.randn(n) * 5e-5
            a = best_a + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    return best_a, best_score
