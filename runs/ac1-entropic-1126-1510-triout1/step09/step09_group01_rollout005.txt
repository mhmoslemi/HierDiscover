"""
Algorithm:  Hybrid exploration + active‑set linear‑program refinement.

1. Broad exploration – test a moderate set of analytic families
   (Beta, arcsine, sine‑power, Gaussian, power‑law) for a handful of
   lengths (2 k … 50 k).  Keep the best normalised vector (sum = 1).

2. Active‑set LP – given a current vector a (∑a=1) with M = max(a*a),
   solve the LP

       maximise   1ᵀb
       subject to  a * b ≤ M   (convolution bound)
                  b ≥ 0

   Only the constraints that are tight (those attaining the current
   maximum) are kept initially; any violating constraint discovered
   after solving is added to the active set (cutting‑plane style).  The
   LP is solved with ``scipy.optimize.linprog`` (highs).

3. Mixing – normalise the LP solution b to a sum of 1, then replace
   a ← (1‑t)·a + t·b with a decreasing schedule t∈[0.6,0.01].
   This convex combination provably does not increase the maximum
   convolution and typically reduces it.

4. Polishing – a short exponentiated‑gradient (smooth‑max) pass
   finishes the optimisation.

The whole search respects the 1000 s budget and returns the best
sequence found (as a plain list of floats).  All numbers are non‑negative,
their sum is 1, and they lie well below the 1000 cap required by
``evaluate_sequence``.
"""

import time
import math
import random
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# FFT‑based convolution utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families – each returns a vector with sum = 1
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # Work in log‑space to avoid under/overflow for tiny p
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian, centre at (n‑1)/2, relative width sigma∈(0,1)."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_power_law(n: int, exponent: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-exponent} on 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-exponent)
    vals /= vals.sum()
    return vals

# Linear‑program helper (active‑set)
def _build_sparse_constraints(a: np.ndarray, active: list[int]) -> csr_matrix:
    """Return a sparse matrix A where each row encodes (a*b)_k ≤ M."""
    n = a.shape[0]
    rows = []
    cols = []
    data = []
    for row, k in enumerate(active):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # i ranges i_start … i_end, corresponding column j = k - i
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = (k - i_vals).astype(np.int64)
        rows.extend([row] * len(i_vals))
        cols.extend(j_vals.tolist())
        data.extend(a[i_vals].tolist())
    return csr_matrix((data, (rows, cols)), shape=(len(active), n), dtype=np.float64)


def _lp_optimize_active(a: np.ndarray, M: float,
                        initial_active: list[int],
                        max_outer_iters: int = 12,
                        tol: float = 1e-12) -> np.ndarray:
    """
    Solve  max 1ᵀb  s.t. a * b ≤ M, b ≥ 0,
    gradually adding violated constraints (cutting‑plane).
    Returns a feasible b (not normalised).
    """
    n = a.shape[0]
    active = list(initial_active)  # copy
    for _ in range(max_outer_iters):
        A = _build_sparse_constraints(a, active)
        b_ub = np.full(A.shape[0], M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)  # minimise –sum(b)
        bounds = [(0, None)] * n

        # Use the high‑performance HiGHS solver (default in recent SciPy)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            # Fallback: uniform distribution (feasible for any M≥1/n)
            return np.full(n, 1.0 / n, dtype=np.float64)

        b = res.x
        conv_ab = _fft_convolve(a, b)
        max_val = conv_ab.max()
        if max_val <= M + max(tol, 1e-12):
            return b

        viol = int(conv_ab.argmax())
        if viol not in active:
            active.append(viol)
        else:
            # No progress – exit to avoid infinite loop
            break
    return b  # last feasible (or near‑feasible) solution


def _mix_and_project(a: np.ndarray, b: np.ndarray, t: float) -> np.ndarray:
    """Convex combine a with a normalised version of b (sum = 1)."""
    s = b.sum()
    if s <= 0:
        b_norm = _distribution_uniform(len(a))
    else:
        b_norm = b / s
    a_new = (1.0 - t) * a + t * b_norm
    eps = 1e-12
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step function that minimises evaluate_sequence.

    The routine respects a CPU‑time budget of ``budget_s`` seconds.
    It returns the best sequence (as a plain Python list) found.
    """
    random.seed(seed)
    np.random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    best_seq = None
    best_score = float("inf")
    best_n = None

    eps = 1e-12

    # --------------------------------------------------------------
    # 1. Broad exploration over a few lengths and analytic families
    lengths = [3000, 6000, 12000, 25000, 50000]          # up to 50 k elements
    for n in lengths:
        if time.time() >= deadline:
            break

        # Uniform baseline
        cand = _distribution_uniform(n)
        sc = _max_convolution_score(cand)
        if sc < best_score:
            best_score, best_seq, best_n = sc, cand.copy(), n
            print(f"[{time.time():.1f}] n={n} uniform best {sc:.6f}")

        # Arcsine
        cand = _distribution_arcsine(n)
        sc = _max_convolution_score(cand)
        if sc < best_score:
            best_score, best_seq, best_n = sc, cand.copy(), n
            print(f"[{time.time():.1f}] n={n} arcsine best {sc:.6f}")

        # Beta family – a dense grid of tiny shape parameters
        beta_ps = [0.001, 0.003, 0.006, 0.008, 0.01, 0.015,
                   0.02, 0.04, 0.06, 0.08, 0.12, 0.2, 0.3, 0.5, 1.0]
        for p in beta_ps:
            if time.time() >= deadline:
                break
            cand = _distribution_beta_sym(n, p)
            sc = _max_convolution_score(cand)
            if sc < best_score:
                best_score, best_seq, best_n = sc, cand.copy(), n
                print(f"[{time.time():.1f}] n={n} beta p={p:.4f} best {sc:.6f}")

        # Sine‑power family
        for pw in (0.5, 1.0, 2.0, 4.0, 8.0, 12.0):
            if time.time() >= deadline:
                break
            cand = _distribution_sine_power(n, pw)
            sc = _max_convolution_score(cand)
            if sc < best_score:
                best_score, best_seq, best_n = sc, cand.copy(), n
                print(f"[{time.time():.1f}] n={n} sine‑pow {pw:.1f} best {sc:.6f}")

        # Gaussian windows
        for sigma in (0.10, 0.15, 0.20, 0.25, 0.30):
            if time.time() >= deadline:
                break
            cand = _distribution_gaussian(n, sigma)
            sc = _max_convolution_score(cand)
            if sc < best_score:
                best_score, best_seq, best_n = sc, cand.copy(), n
                print(f"[{time.time():.1f}] n={n} gauss σ={sigma:.2f} best {sc:.6f}")

        # Power‑law tails
        for exp in (0.5, 1.0, 1.5, 2.0, 3.0):
            if time.time() >= deadline:
                break
            cand = _distribution_power_law(n, exp)
            sc = _max_convolution_score(cand)
            if sc < best_score:
                best_score, best_seq, best_n = sc, cand.copy(), n
                print(f"[{time.time():.1f}] n={n} power‑law α={exp:.1f} best {sc:.6f}")

        # Random Dirichlet (few draws)
        for alpha in (0.1, 0.3, 0.6, 1.0):
            if time.time() >= deadline:
                break
            vec = np.random.dirichlet(np.full(n, alpha))
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n} Dirichlet α={alpha:.1f} best {sc:.6f}")

    # --------------------------------------------------------------
    # 2. If a previous solution is available, consider it
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score, best_seq, best_n = sc, prev.copy(), len(prev)
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3. Fallback in the unprecedented case
    if best_seq is None:
        best_n = 4000
        best_seq = _distribution_uniform(best_n)
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Active‑set LP refinement
    a = best_seq.copy()                          # always normalised to sum=1
    n = a.shape[0]

    # schedule for mixing parameter t (larger → aggressive)
    t_schedule = [0.6, 0.4, 0.25, 0.15, 0.08, 0.04, 0.02, 0.01]

    for outer, t in enumerate(t_schedule):
        if time.time() >= deadline:
            break
        conv_self = _fft_self_convolve(a)
        M = conv_self.max()
        # initialise active set with every index that is (almost) tight
        init_active = list(np.flatnonzero(conv_self >= M - 1e-12))
        b_opt = _lp_optimize_active(a, M, init_active, max_outer_iters=12)

        # Mix the new direction into the current iterate
        a_new = _mix_and_project(a, b_opt, t)

        new_score = _max_convolution_score(a_new)
        if new_score < best_score - 1e-12:
            best_score, best_seq = new_score, a_new.copy()
            print(f"[{time.time():.1f}] LP iter {outer} t={t:.3f} new best {new_score:.6f}")

        a = a_new

        # safety‑stop if we have already reached the target
        if best_score < 1.503:
            break

    # --------------------------------------------------------------
    # 5. Light smooth‑max mirror‑descent polishing (few hundred iters)
    #   – this is cheap compared to the earlier long mirror‑descent.
    al_start, al_end = 500.0, 20000.0
    steps = 800
    lr_start, lr_end = 0.05, 0.0005
    for it in range(steps):
        if time.time() >= deadline:
            break
        t_frac = it / steps
        alpha = al_start * (al_end / al_start) ** t_frac
        lr = lr_start * (lr_end / lr_start) ** t_frac

        conv = _fft_self_convolve(a)
        maxc = conv.max()
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        a = a * np.exp(-lr * grad)
        a = np.maximum(a, eps)
        a /= a.sum()

        if (it % 100 == 0) or (it == steps - 1):
            cur_sc = _max_convolution_score(a)
            if cur_sc < best_score - 1e-12:
                best_score, best_seq = cur_sc, a.copy()
                print(f"[{time.time():.1f}] mirror polishing it={it} new best {cur_sc:.6f}")

    # --------------------------------------------------------------
    # 6. Final simple hill‑climber (mass‑transfer) – very cheap
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    while time.time() < deadline:
        it += 1
        # pick the (i,j) pair that gives the current maximum product
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M
            cur_sc = 2.0 * n * M
            if cur_sc < best_score - 1e-12:
                best_score, best_seq = cur_sc, a.copy()
                print(f"[{time.time():.1f}] hill‑climb it={it} new best {cur_sc:.6f}")

        if it % 2000 == 0:
            # occasional jitter to avoid plateaus
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] hill jitter at it {it}")

    final_score = 2.0 * n * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
