    import numpy as np, random, time, math, sys
    from scipy.special import i0

    start = time.time()
    deadline = start + budget_s - 0.5   # margin

    np.random.seed(seed)
    random.seed(seed)

    eps = 1e-12

    # define helper functions: _next_pow2, _fft_self_convolve, _max_convolution_score etc.
    # define families:
    # * uniform
    # * exponential lam list
    # * beta p list
    # * sine_power list
    # * gaussian sigma list
    # * kaiser beta list
    # * tukey alpha list
    # * random Dirichlet draws (alpha list)
    # Collect candidates for each n in lengths list.

    best_seq = None; best_score = float('inf'); best_n=0; best_alpha=0
    lengths = [800,1200,1600,2000,2400,2800,3200]   # maybe keep within budget.

    families loops:

    for n in lengths:
        if time.time() >= deadline: break
        # for each param set:
        generate vectors v
        for v:
            v = np.maximum(v, eps); v /= v.sum()
            sc = _max_convolution_score(v)
            if sc < best_score:
                best_seq=v.copy(); best_score=sc; best_n=n
                print(f"[{time.time():.1f}] init n={n} schema={name} score={sc:.6f}")

    # now refine best_seq using Adam smoothing for several lengths
    # we might also attempt to improve all lengths by random restarts, but limited time.
    if best_seq is None:
        # fallback uniform
        n = 2000; best_seq = np.full(n, 1/n); best_score = _max_convolution_score(best_seq)

    # local refinement using Adam
    def adam_refine(a, max_iters, deadline):
        n = a.shape[0]
        # initialize raw parameters z = log a (or small random)
        z = np.log(a + eps)
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-8
        alpha_smooth_start = 1.0
        alpha_smooth_end = 200.0   # large to approximate max
        # schedule alpha linearly in log
        for t in range(1, max_iters+1):
            if time.time() >= deadline:
                break
            # compute a = softmax(z)
            exp_z = np.exp(z - np.max(z))  # stable
            a = exp_z / exp_z.sum()
            # compute convolution
            conv = _fft_self_convolve(a)
            maxc = conv.max()
            # smooth parameter
            # using exponential schedule for alpha: alpha = alpha_start * (alpha_end/alpha_start) ** ((t-1)/max_iters)
            alpha = alpha_smooth_start * (alpha_smooth_end/alpha_smooth_start) ** ((t-1)/max_iters)
            # compute softmax over conv
            w_exp = np.exp(alpha * (conv - maxc))   # shift for numerical stability
            w = w_exp / w_exp.sum()
            # gradient w.r.t a
            grad_a = 2.0 * _fft_convolve(a[::-1], w)[n-1:n-1+n]
            # gradient w.r.t z
            inner = np.dot(a, grad_a)
            grad_z = a * (grad_a - inner)
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * (grad_z**2)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            lr = 0.1  # base learning rate
            step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
            z = z - step
            # periodic evaluation
            if t % 200 == 0:
                sc = 2.0 * n * maxc
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] Adam it={t} n={n} score={sc:.6f}")

        return a

    # run Adam until time.
    a = best_seq.copy()
    a = adam_refine(a, max_iters=20000, deadline=deadline)

    # Optionally perform a short hill climb as earlier to adjust
    # but not necessary

    # final best_seq is best_seq.

    return best_seq.tolist()

def _distribution_tukey(n, alpha):
    # alpha within (0,1]
    if not 0 < alpha <=1:
        alpha = 1.0
    r = np.arange(n, dtype=float) / (n-1)
    # Define region length = alpha/2 at each end
    w = np.empty_like(r)
    # first region
    t = r / alpha
    w[:int(alpha*(n-1)/2)+1] = 0.5 * (1 + np.cos(np.pi*(t[:int(alpha*(n-1)/2)+1] - 1)))
    # middle region
    w[int(alpha*(n-1)/2): n - int(alpha*(n-1)/2)] = 1.0

    idx = np.arange(n - int(alpha*(n-1)/2), n)

    w[idx] = 0.5 * (1 + np.cos(np.pi * t2))
    w = np.maximum(w,0)
    w /= w.sum()
    return w

def _adam_optimize(a_init, max_iters, deadline):
    n = a_init.shape[0]
    z = np.log(a_init + 1e-12)  # ensure nonzero
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    # schedule alpha start small (like 1) to large (like 2000)
    alpha_start = 0.5
    alpha_end = 5000.0 # final smoothing large
    best_local_seq = a_init.copy()
    best_local_score = 2.0 * n * _fft_self_convolve(a_init).max()
    for t in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # compute a
        max_z = np.max(z)  # for stability
        exp_z = np.exp(z - max_z)
        a = exp_z / exp_z.sum()
        # conv
        conv = _fft_self_convolve(a)
        # get smooth max weights
        alpha = alpha_start * (alpha_end / alpha_start) ** ((t-1) / max_iters)
        # compute w = softmax(alpha * conv)
        max_c = conv.max()
        exp_w = np.exp(alpha * (conv - max_c))
        w = exp_w / exp_w.sum()
        # gradient wrt a
        grad_a = 2.0 * _fft_convolve(a[::-1], w)[n-1:n-1+n]
        # inner product a·grad_a
        inner = np.dot(a, grad_a)
        grad_z = a * (grad_a - inner)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        # step size schedule: maybe decreasing 0.5 / sqrt(t)
        lr = 0.5 / np.sqrt(t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        # occasionally evaluate true score
        if t % 200 == 0:
            sc = 2.0 * n * conv.max()
            if sc < best_local_score:
                best_local_score = sc
                best_local_seq = a.copy()
                print(f"[{time.time():.1f}] Adam iter {t}/{max_iters} n={n} score={sc:.6f}")
        # optionally add jitter after some steps? Not needed.
    return best_local_seq, best_local_score
```

        # Use min(target_alpha, 1.0 + (it/1000.0))

        # compute w

        grad_a = 2.0 * _fft_convolve(a[::-1], w)[n-1:n-1+n]
        # softmax gradient for z

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z

        # Evaluate current score occasionally

                best_a = a.copy()
                print(f"[{time.time():.1f}] Adam it {it}  best {cur_score:.6f}")

"""
Propose a step‑function vector (list of non‑negative floats) that minimises
``evaluate_sequence``.  The optimiser works in three stages:

1️⃣  Fast scan of many analytic families (uniform, triangular, exponential,
    beta, sine‑power, Gaussian, Kaiser, Tukey, random Dirichlet) for several
    lengths.  The best normalised candidate is kept.

2️⃣  A continuation‑style Adam optimiser on the simplex.  The vector is
    parameterised by log‑weights (soft‑max) so positivity and Σa=1 are
    automatic.  The objective is a smooth approximation of the max
    convolution (log‑sum‑exp with a gradually increasing smoothing parameter).
    Adam’s adaptive step‑size and momentum give a fast, stable descent.

3️⃣  A cheap hill‑climber that repeatedly moves a tiny amount of mass from the
    pair that contributes most to the current maximum towards a random index,
    plus occasional jitter to escape plateaus.

The whole routine respects a 1000 s wall‑clock budget (a small safety margin
is subtracted).  Intermediate improvements are printed, and the best sequence
found is returned as a plain list of Python ``float`` objects.
"""
import time
import random
import math

import numpy as np
from scipy import signal
from scipy.special import i0

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to Σa=1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families (all return a vector with Σ=1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_triangular(n):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = 1.0 - np.abs((idx - centre) / centre)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.01):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    """Discrete Gaussian centred at (n‑1)/2."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_kaiser(n, beta):
    """Kaiser window (SciPy) normalised to a probability vector."""
    w = signal.windows.kaiser(n, beta, sym=False)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _distribution_tukey(n, alpha):
    """Tukey window (SciPy) normalised."""
    w = signal.windows.tukey(n, alpha, sym=False)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _adam_optimize(a_init: np.ndarray, deadline: float):
    """
    Adam optimiser on the simplex using a smooth‐max surrogate.
    Returns an improved vector (still Σ=1) and its exact score.
    """
    n = a_init.shape[0]
    eps = 1e-12
    # log‑softmax parameters (guarantee positivity and sum=1)
    z = np.log(a_init + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    best_a = a_init.copy()
    best_score = _max_convolution_score(best_a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- soft‑max to obtain a -----
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        a = exp_z / exp_z.sum()

        # ----- convolution and smooth‑max weights -----
        conv = _fft_self_convolve(a)
        # smoothing parameter grows from ~1 to a few thousand
        alpha = 1.0 + min(2000.0, it / 200.0)   # linear increase, capped
        max_c = conv.max()
        exp_w = np.exp(alpha * (conv - max_c))
        w = exp_w / exp_w.sum()

        # ----- gradient w.r.t. a (sub‑gradient of the smooth max) -----
        grad_a = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        # ----- translate to gradient w.r.t. z (soft‑max Jacobian) -----
        inner = np.dot(a, grad_a)
        grad_z = a * (grad_a - inner)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.5 / np.sqrt(it)                # decreasing learning rate
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z - step

        # ----- occasional exact evaluation -----
        if it % 400 == 0:
            cur_score = _max_convolution_score(a)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] Adam it={it} n={n} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
def _hill_climber(a: np.ndarray, deadline: float):
    """Simple mass‑transfer hill‑climber (few thousand cheap steps)."""
    eps = 1e-12
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    best = a.copy()
    best_score = 2.0 * n * M
    it = 0

    while time.time() < deadline:
        it += 1
        k = int(conv.argmax())                     # shift with current maximum
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i_range = np.arange(i_low, i_high + 1)
        j_range = k - i_range
        prod = a[i_range] * a[j_range]
        idx = prod.argmax()
        i, j = i_range[idx], j_range[idx]

        # move a little mass from the larger contributor to a random slot
        source = i if a[i] >= a[j] else j
        target = random.randint(0, n - 1)
        if target == source:
            continue
        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        conv2 = _fft_self_convolve(a2)
        M2 = conv2.max()

        if M2 < M - 1e-12:
            a, conv, M = a2, conv2, M2
            cur_score = 2.0 * n * M
            if cur_score < best_score - 1e-12:
                best, best_score = a.copy(), cur_score
                print(f"[{time.time():.1f}] hill it={it} best={cur_score:.6f}")

        # occasional jitter to escape flat plateaus
        if it % 5000 == 0:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

    return best, best_score
