"""
Propose a step‑function (non‑negative vector) that minimizes
``evaluate_sequence`` = 2·n·max(conv(a,a)) / (sum(a)^2).

Algorithm
---------

* **Seed selection** – try a few classic analytic families (uniform,
  arcsine, sparse random) and the optional previous solution
  ``height_sequence_1``.  The best of them (by the exact objective) is
  taken as the starting point.

* **p‑norm smoothing gradient descent** – instead of the
  sub‑gradient of the ∞‑norm (max) we minimise the smooth functional  

        F_p(a) = ( Σ_k ( (a∗a)_k )^p )^{1/p}

  whose gradient can be written in closed form using a correlation.
  For increasing values of p (4, 8, 16, 32, 64) we run a projected
  exponentiated‑gradient descent (mirror descent on the simplex).
  The step‑size decays like 1/√t and the update is clipped to avoid
  overflow.  After every few iterations the true objective
  (max‑convolution) is evaluated and the best vector is stored.

* **Optional up‑sampling** – if time permits the current best vector is
  linearly interpolated to twice the resolution and the smoothing
  descent is repeated.  More points give a finer discretisation and
  allow a flatter autocorrelation.

* **Fine‑tuning** – a cheap greedy pairwise mass‑transfer walk (the
  former ``_pairwise_swap_local``) is finally applied until the time
  budget expires.

All heavy operations (convolution, correlation) use real FFTs, giving
O(n log n) per iteration.  The routine respects the 1000 s budget and
returns the best sequence found as a plain ``list[float]``.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full (linear) self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_correlation(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full correlation (x ⋆ y) = Σ_i x[i]·y[i+shift].
    Returned length is len(x)+len(y)-1 (full mode).
    """
    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    corr = np.fft.irfft(np.conj(X) * Y, n=size)
    return corr[:n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic seeds -------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals

def _smooth_gradient(a: np.ndarray, p: int) -> np.ndarray:
    """
    Gradient of the smoothed p‑norm objective:
        G_j = 2 * Σ_i (conv_i)^{p-1} * a_{i-j}
    computed via an FFT‑based correlation.
    """
    conv = _fft_self_convolve(a)
    eps = 1e-30
    power = np.maximum(conv, eps) ** (p - 1)          # (conv)^{p‑1}
    corr = _fft_correlation(power, a)                # length = 3n‑2
    n = a.shape[0]
    # correlation index for shift = -j is (n‑1)‑j
    grad = 2.0 * corr[(n - 1) - np.arange(n)]
    return grad

def _smooth_descent(a0: np.ndarray, deadline: float,
                    best_seq: np.ndarray, best_score: float,

    increasing p values.  The current best vector and its score are
    updated whenever a true (max‑convolution) improvement is found.
    """
    a = a0.copy()
    for p in p_schedule:

        remaining = deadline - time.time()

            break
        # rough cost estimate: ~0.004 s per iteration for n≈15000

            break

            grad = _smooth_gradient(a, p)

            eta = base_lr / (math.sqrt(it) * (p ** 0.25))

            # exponentiated‑gradient update, clipping exponent to avoid overflow
            upd = -eta * grad
            upd = np.clip(upd, -50, 50)               # safe region for exp

        if time.time() >= deadline:

def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Tiny greedy mass‑transfer walk – keep the move only if the true objective improves."""
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e} "
                  f"new best {best_score:.6f}")

    return best_seq, best_score

def _linear_upscale(a: np.ndarray, factor: int = 2) -> np.ndarray:
    """Linear interpolation to a finer grid (length multiplied by factor)."""
    if factor == 1:
        return a.copy()
    n = len(a)
    old_x = np.arange(n, dtype=np.float64)
    new_n = n * factor
    new_x = np.linspace(0, n - 1, new_n, dtype=np.float64)
    up = np.interp(new_x, old_x, a)
    up /= up.sum()
    return up

        if n > 120000:    # guard against exploding memory/time
            break

    # ---- 5. final greedy local search -----------------------------------
    if time.time() < deadline:
        best_seq, best_score = _pairwise_swap_local(best_seq.copy(),
                                                    deadline,
                                                    best_seq.copy(),
                                                    best_score)

    # ---- 6. final sanity & return ---------------------------------------
    final = np.clip(best_seq, 0.0, 1000.0)
    final = final.astype(float).tolist()
    print(f"Final best score: {best_score:.6f} (length={len(final)})")
    return final
