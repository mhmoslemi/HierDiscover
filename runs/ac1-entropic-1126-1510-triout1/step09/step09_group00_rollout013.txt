def _softmax_gradient(a, alpha):
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    # compute softmax weights
    max_c = conv.max()
    # To prevent overflow: shift
    exp_vals = np.exp(alpha * (conv - max_c))
    w = exp_vals / exp_vals.sum()
    # compute gradient g = 2 * (w ⋆ a)
    # cross-correlation
    rev_a = a[::-1]
    size = 1 << ((len(w) + n - 1 - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(rev_a, n=size)
    g_full = np.fft.irfft(W * A_rev, n=size)[: len(w) + n - 1]
    # take slice (n-1) : (n-1 + n)
    start = n - 1
    g = 2.0 * g_full[start:start+n]
    return g
```

"""
Search for a non‑negative step‑function that minimizes
``evaluate_sequence`` (i.e. 2·n·max(conv(a,a)) for a normalised vector).

Algorithm
---------

* Build a diverse pool of seeds:
  – classic analytic families (uniform, arcsine, beta, power‑law, exponential,
    two‑block spikes, sparse random);
  – a few random Dirichlet vectors with varying concentration.
* Keep the best few seeds (by exact objective) and start a *soft‑max*
  mirror‑descent from each one.
  The smooth loss Lα(a)= (1/α)·log∑ exp(α·conv(a,a)) approaches the true
  max‑convolution as α grows.  Its gradient can be computed by a single FFT
  convolution, yielding cheap O(n log n) updates.
  We increase α during optimisation (annealing) and use an exponentiated‑
  gradient update together with Euclidean projection onto the simplex.
* After the annealing stage a short greedy pairwise‑mass‑transfer walk
  (always accepting exact improvements) polish the final vector.

The whole routine respects a 1000 s wall‑time budget and returns the best
vector found as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Broad pool of seeds for a given length."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    for p in (0.05, 0.1, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0):
        cand.append(_distribution_beta_sym(n, p))

    for _ in range(5):
        alpha = 10 ** rng.uniform(-2, 1)   # 0.01 … 10
        beta = 10 ** rng.uniform(-2, 1)    # 0.01 … 10
        cand.append(_distribution_beta_asym(n, alpha, beta))

    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))

    for lam in (0.001, 0.005, 0.01, 0.02, 0.05):
        cand.append(_distribution_exponential_increasing(n, lam))

    for block_frac in (0.05, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac))

    cand.append(_distribution_random_sparse(n, rng, density=0.01))

    return cand


# ----------------------------------------------------------------------
# Gradient of the soft‑max (log‑sum‑exp) approximation of max‑conv ---
def _softmax_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Returns ∇_a  (1/α)·log∑_k exp(α·conv_k(a)).
    Gradient formula: 2·(w ⋆ a) where w are the soft‑max weights.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)

    # stable soft‑max
    max_c = conv.max()
    exp_vals = np.exp(alpha * (conv - max_c))
    w = exp_vals / exp_vals.sum()

    # cross‑correlation w ⋆ a  (equivalently conv(w, rev(a)) slice)
    rev_a = a[::-1]
    size = 1 << ((len(w) + n - 1 - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(rev_a, n=size)
    full = np.fft.irfft(W * A_rev, n=size)[: len(w) + n - 1]

    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smooth log‑sum‑exp loss.
    Starts with a moderate α and slowly anneals it upwards.
    Returns the best vector (and its exact objective) seen before the deadline.
    """
    a = a0.astype(np.float64).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = _objective(a)

    eta0 = 0.5          # base step size
    alpha = 1.0         # start temperature
    t = 0

    while time.time() < deadline:
        t += 1

        # gradient of the smooth loss
        g = _softmax_gradient(a, alpha)

        # mirror‑descent update (exponentiated)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)

        # simple re‑normalisation to the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # occasional exact evaluation
        if t % 200 == 0:
            cur_sc = _objective(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] smooth it={t} new best {best_sc:.6f}")

        # anneal α – larger α makes the loss a tighter approximation of max
        if t % 500 == 0:
            alpha = min(alpha * 1.3, 128.0)

    return best_a, best_sc


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Randomly move a little mass; keep move only if true objective improves."""
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur_sc = _objective(new_a)

        if cur_sc < best_score - 1e-13:
            a = new_a
            best_score = cur_sc
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best vector (as a plain ``list[float]``) found within the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    best_seq = None
    best_score = float("inf")
    candidate_pool = []

    # ------------------------------------------------------------------
    # 1. Analytic families for a handful of lengths
    lengths = [2000, 4000, 8000, 12000, 20000, 30000]
    for n in lengths:
        if time.time() >= deadline:
            break
        for cand in _initial_candidates_extended(n, rng):
            if time.time() >= deadline:
                break
            cand = cand.astype(np.float64)
            cand /= cand.sum()
            sc = _objective(cand)
            candidate_pool.append((sc, cand))
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best (seed) len={n} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Random Dirichlet / sparse seeds – add a few hundred more
    while time.time() < deadline and len(candidate_pool) < 300:
        n = int(rng.integers(2000, 30001))
        # concentration between 0.2 and 5.0 (sparse to fairly smooth)
        conc = rng.uniform(0.2, 5.0)
        a = rng.gamma(conc, 1.0, size=n)
        a /= a.sum()
        sc = _objective(a)
        candidate_pool.append((sc, a.copy()))
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time():.1f}] new global best (rand) len={n} score={sc:.6f}")

        # occasional sparse vector
        if rng.random() < 0.15:
            sparse = _distribution_random_sparse(n, rng, density=0.01)
            sc = _objective(sparse)
            candidate_pool.append((sc, sparse.copy()))
            if sc < best_score:
                best_score = sc
                best_seq = sparse.copy()
                print(f"[{time.time():.1f}] new global best (sparse) len={n} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3. Keep only the very best seeds for the costly optimisation phase
    candidate_pool.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in candidate_pool[:12]]

    # ------------------------------------------------------------------
    # 4. Smooth log‑sum‑exp mirror descent on the top seeds
    remaining = deadline - time.time()
    if remaining > 60 and top_seeds:
        per_seed = remaining / max(1, len(top_seeds))
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            opt_vec, opt_sc = _smooth_descent(seed_vec.copy(), sub_deadline)
            if opt_sc < best_score:
                best_score = opt_sc
                best_seq = opt_vec.copy()
                print(f"[{time.time():.1f}] after smooth descent new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final tiny Metropolis‑type polishing
    if best_seq is not None and time.time() < deadline:
        best_seq, best_score = _pairwise_swap_local(
            best_seq.copy(),
            deadline,
            best_seq.copy(),
            best_score,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length={len(best_seq) if best_seq is not None else 0})")
    return best_seq.tolist() if best_seq is not None else []
