def order_by_step(desc_vals, step):
    n = desc_vals.shape[0]
    out = np.empty_like(desc_vals)
    indices = (np.arange(n) * step) % n
    out[indices] = desc_vals
    return out

The strategy is a three‑stage pipeline:

1️⃣ **Broad generation** – Create many analytic families (arcsine, beta,
   Gaussian, exponential, …) on a handful of lengths.  All vectors are
   ℓ₁‑normalised.

2️⃣ **Ordering search** – For each family take the multiset of values,
   sort it descending and try many deterministic and random permutations.
   In particular we use a “step‑permutation’’ (i·step mod n) with a
   coprime step, several random shuffles and a few classic spreads
   (alternate extremes, centre‑outwards, binary‑tree).  This cheap search
   typically discovers a much flatter self‑convolution than the
   deterministic orders used before.

3️⃣ **Local polishing** – The best few candidates are refined by a short
   Adam‐style projected gradient descent (sub‑gradient of the max‑convolution)
   and by a simple stochastic mass‑transfer hill‑climb that repeatedly
   moves a tiny amount of probability mass from a high‑gradient index to a
   low‑gradient one.  Small convex blends (e.g. with the reversed sequence)
   are also tried.

All steps respect a global time budget (≈ 1000 s).  The routine returns a

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a via a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a⋆b via a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate of ``evaluate_sequence`` for a ℓ₁‑normalised vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of the max‑convolution functional.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives raw gradient
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even‑indexed active positions
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _gaussian_distribution(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    w /= w.sum()
    return w


def _cauchy_distribution(n: int, scale_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    scale = scale_factor * n
    w = 1.0 / (1.0 + ((idx - centre) / scale) ** 2)
    w /= w.sum()
    return w


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Unsymmetric Beta(α,β) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide='ignore', invalid='ignore'):
        w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    if np.isnan(w).any() or np.isinf(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex from Dirichlet(α,…,α)."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Deterministic re‑orderings --------------------------------------------
def order_alternate_extremes(desc_vals: np.ndarray) -> np.ndarray:
    """Largest, second‑largest, … alternately left‑most / right‑most."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    left, right = 0, n - 1
    toggle = True
    for v in desc_vals:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    return out


def order_center_outwards(desc_vals: np.ndarray) -> np.ndarray:
    """Put the largest value at the centre then expand outwards."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    positions = []
    mid = (n - 1) // 2
    positions.append(mid)
    for d in range(1, n):
        left = mid - d
        if left >= 0:
            positions.append(left)
        right = mid + d
        if right < n:
            positions.append(right)
    for i, pos in enumerate(positions[:n]):
        out[pos] = desc_vals[i]
    return out


def order_binary_tree(desc_vals: np.ndarray) -> np.ndarray:
    """Recursive midpoint filling – spreads large coefficients."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)

    def recurse(l: int, r: int, it: list):
        if l > r:
            return
        m = (l + r) // 2
        it.append(m)
        recurse(l, m - 1, it)
        recurse(m + 1, r, it)

    positions = []
    recurse(0, n - 1, positions)
    out[positions] = desc_vals
    return out

def order_by_step(desc_vals: np.ndarray, step: int) -> np.ndarray:
    """Place values according to i·step (mod n); step must be coprime to n."""
    n = desc_vals.shape[0]
    out = np.empty_like(desc_vals)
    idx = (np.arange(n) * step) % n
    out[idx] = desc_vals
    return out

def _adam_descent(a0: np.ndarray, deadline: float,

def _sgd_subgrad(a0: np.ndarray, deadline: float,

# Small stochastic mass‑transfer hill‑climb -------------------------------
def _hill_climb_mass_transfer(a: np.ndarray, deadline: float,
                             max_trials: int = 30_000,
                             delta_frac: float = 0.001) -> tuple[np.ndarray, float]:

    best_a = a.copy()

    while time.time() < deadline and trials < max_trials:
        trials += 1
