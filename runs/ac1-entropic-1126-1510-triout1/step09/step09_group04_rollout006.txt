"""
Propose a step‑function (non‑negative list) that minimises
``evaluate_sequence``.

Algorithm
---------
1. **Analytic scan** – generate several textbook families (uniform,
   arcsine, triangular, exponential, symmetric beta, sine‑power,
   Gaussian) for a handful of lengths (up to 6000).  Keep the best
   normalised vector.
2. **Adam optimisation** – treat the vector as a soft‑max of logits.
   Minimise a smoothed version of the objective
   `max(a*a)` using the smooth‑max (log‑sum‑exp) with a temperature
   that grows from 1 to 1e5 while Adam’s learning‑rate decays.
   The simplex constraint is satisfied automatically by the soft‑max.
3. **Mass‑transfer hill‑climber** – starting from the Adam solution,
   repeatedly move a small amount of mass from a high‑contribution
   index to a random site and update the full self‑convolution by a
   cheap O(n) shift.  Accept only improving moves (or occasional
   uphill moves with an annealing temperature) and keep the best
   sequence seen.
4. **Fallback** – if no sequence was found, fall back to the globally
   supplied ``height_sequence_1`` or a uniform vector.

All steps respect the wall‑clock budget (default 1000 s) and the
returned list consists of Python ``float`` objects, ready for
``evaluate_sequence``.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) for a vector already normalised to sum = 1.
    This equals the value returned by ``evaluate_sequence`` for the same
    normalised vector.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families – all return a *non‑negative* vector with sum = 1
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.01) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at (n‑1)/2."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The function respects the supplied
    time budget and returns the best vector (as a plain list of Python
    floats) it found.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    best_seq = None
    best_score = float('inf')
    best_n = None
    eps = 1e-12

    # ------------------------------------------------------------------
    # 1) quick scan of analytic families on several lengths
    lengths = [2000, 3000, 4000, 5000, 6000]
    for n in lengths:
        if time.time() >= deadline:
            break

        # families to try
        families = [
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_triangular,
            lambda nn: _distribution_exponential(nn, lam=0.006),
            lambda nn: _distribution_exponential(nn, lam=0.018),
            lambda nn: _distribution_beta_sym(nn, p=0.7),
            lambda nn: _distribution_beta_sym(nn, p=1.4),
            lambda nn: _distribution_sine_power(nn, power=1.4),
            lambda nn: _distribution_gaussian(nn, sigma=0.33),
        ]

        for gen in families:
            vec = gen(n).astype(np.float64)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] init n={n:4d}  score={sc:.6f}")

        # few random Dirichlet draws
        for alpha in (0.2, 0.6, 1.0, 2.0):
            vec = np.random.dirichlet(np.full(n, alpha))
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] rand n={n:4d} α={alpha:.1f}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2) bring forward the global variable if it exists and is better
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_seq = prev.copy()
                best_n = prev.size
                best_score = sc
                print(f"[{time.time():.1f}] prior best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3) fallback to uniform if nothing was found
    if best_seq is None:
        best_n = 3000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Adam optimisation on the smoothed max objective
    n = best_n
    # initialise logits (softmax) from the best vector
    y = np.log(best_seq + 1e-30)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    total_iters = 6000
    alpha_start = 1.0
    alpha_end = 1e5   # strong smoothing → true max

    for t in range(1, total_iters + 1):
        if time.time() >= deadline:
            break

        # ---- softmax to obtain a on the simplex ----
        y_shift = y - np.max(y)                     # numeric stability
        a = np.exp(y_shift)
        a /= a.sum()

        # ---- smooth‑max weights ----
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        alpha = alpha_start * (alpha_end / alpha_start) ** (t / total_iters)
        w = np.exp(alpha * (conv - maxc))
        s = w.sum()
        if s == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= s

        # ---- gradient w.r.t. a (see analysis) ----
        grad_a = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        # ---- chain rule for softmax (gradient w.r.t. logits) ----
        mu = np.dot(a, grad_a)
        grad_y = a * (grad_a - mu)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = 0.25 / (1.0 + 5e-4 * t)               # slowly decreasing
        y -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional reporting and external best‑track update
        if t % 300 == 0 or t == total_iters:
            a_eval = np.exp(y - np.max(y))
            a_eval /= a_eval.sum()
            sc = _max_convolution_score(a_eval)
            if sc < best_score:
                best_seq = a_eval.copy()
                best_score = sc
                print(f"[{time.time():.1f}] Adam it={t:4d}  best={sc:.6f}")

    # ------------------------------------------------------------------
    # 5) cheap hill‑climber / simulated annealing on the exact max
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    n = a.size

    # annealing schedule (temperature for occasional uphill moves)
    temp_start = 0.02
    temp_end = 1e-5
    hill_iters = 200_000

    for it in range(1, hill_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        temp = temp_start * (temp_end / temp_start) ** (it / hill_iters)

        # pick a source index weighted by current mass (prefer big entries)
        source = np.random.choice(n, p=a / a.sum())
        if a[source] < eps:
            continue

        # pick a distinct target uniformly
        target = random.randint(0, n - 1)
        if target == source:
            continue

        # step size – a modest fraction of the source mass
        epsilon = min(0.02 * a[source], 0.004)

        # propose new a
        a_new = a.copy()
        a_new[source] -= epsilon
        a_new[target] += epsilon
        if a_new[source] < 0.0:
            continue

        # update convolution for the proposed move (O(n))
        conv_new = conv.copy()
        # subtract contribution of the source
        conv_new[source:source + n] -= 2.0 * epsilon * a
        # add contribution of the target
        conv_new[target:target + n] += 2.0 * epsilon * a
        # correct the diagonal quadratic terms
        conv_new[2 * source] += epsilon * epsilon
        conv_new[2 * target] += epsilon * epsilon

        new_M = conv_new.max()
        delta = new_M - M

        # Metropolis acceptance
        if delta < -1e-12 or random.random() < math.exp(-delta / (temp + 1e-12)):
            a = a_new
            conv = conv_new
            M = new_M
            # every few thousand iterations we check whether we've set a new record
            if it % 5000 == 0:
                sc = 2.0 * n * M
                if sc < best_score:
                    best_seq = a.copy()
                    best_score = sc
                    print(f"[{time.time():.1f}] hill it={it:6d}  best={sc:.6f}")

    # ------------------------------------------------------------------
    # final reporting
    final_score = 2.0 * n * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
