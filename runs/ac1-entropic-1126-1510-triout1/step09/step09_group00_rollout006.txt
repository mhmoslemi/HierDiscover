def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_full_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a*b using real FFT (length = len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    size = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[:m]


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length = 2·len(a)-1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a vector with sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families – many parameters, very cheap to generate ----------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric beta (Beta(alpha,beta))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a

    for p in np.logspace(-4, 0, num=13):  # 1e‑4 … 1
        pool.append(_distribution_beta_sym(n, float(p)))

    # asymmetric beta – random sampling on a wide log‑scale
    for _ in range(6):
        alpha = 10 ** rng.uniform(-4, 2)   # 1e‑4 … 100
        beta = 10 ** rng.uniform(-4, 2)
        pool.append(_distribution_beta_asym(n, alpha, beta))

    # monotone power‑law (incl. very steep)
    for p in (0.5, 1.0, 2.0, 4.0, 6.0):
        pool.append(_distribution_power_increasing(n, p))

    # exponential – both shallow and steep
    for lam in (0.001, 0.003, 0.005, 0.01, 0.02, 0.05):
        pool.append(_distribution_exponential_increasing(n, lam))

    # two‑block spikes – different block fractions
    for block_frac in (0.02, 0.05, 0.10):
        pool.append(_distribution_two_blocks(n, block_frac))

    # sparse random start (≈1 % non‑zeros)
    pool.append(_distribution_random_sparse(n, rng, density=0.01))

    return pool

def _smooth_gradient_descent(a: np.ndarray,
                             deadline: float,
                             best_seq: np.ndarray,
                             best_score: float,
                             lr0: float = 0.30,
                             max_iter: int = 20000,
                             gamma_start: float = 1.0,
                             gamma_mult: float = 1.5,
                             gamma_step: int = 500,
                             eval_interval: int = 200) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the *smoothed* max‑convolution objective.
    Uses a soft‑max over the full convolution to obtain a smooth
    gradient, then performs an additive step with simplex projection.
    """
    n = a.shape[0]
    gamma = gamma_start
    t = 0

    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)                     # length 2n‑1
        # soft‑max weights (numerically stable)
        maxc = conv.max()
        w = np.exp(gamma * (conv - maxc))
        w /= w.sum()

        # gradient = 2 * (a_rev * w)  (cross‑correlation)
        a_rev = a[::-1]
        conv_rev = _fft_full_convolve(a_rev, w)          # length 2n‑1
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]          # slice size n

        # step size schedule η_t = lr0 / sqrt(t)
        eta = lr0 / math.sqrt(t)
        a = a - eta * grad

        # keep us on the simplex (non‑negative, sum = 1)
        a = _project_onto_simplex(a, s=1.0)

        # occasionally evaluate the true (non‑smoothed) objective
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] grad it={t} γ={gamma:.3f} new best {best_score:.6f}")

        # increase smoothness parameter γ periodically
        if t % gamma_step == 0:
            gamma = min(gamma * gamma_mult, 1e5)

    return best_seq, best_score


def _annealing_pairwise_swap(a: np.ndarray,
                             deadline: float,
                             best_seq: np.ndarray,
                             best_score: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing walk: move a random portion of mass from a
    source index to a destination index.  Accept improvements always,
    accept deteriorations with probability exp(–Δ/T).
    """
    n = a.shape[0]
    start = time.time()
    total_time = max(deadline - start, 1e-3)

    # temperature schedule (linear cooling)
    T_start = 5e-3
    T_end = 1e-7

    while time.time() < deadline:
        elapsed = time.time() - start
        T = max(T_end, T_start * (1 - elapsed / total_time))

        # choose source (must have some mass) and destination
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # amount to move – up to 30 % of source mass
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        diff = cur - best_score

        if diff < 0 or np.random.rand() < math.exp(-diff / T):
            a = new_a
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}→j={j} δ={delta:.3e} T={T:.2e} new best {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best vector found within the
    given time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # a small safety margin

    # ------------------------------------------------------------------
    # 1. Lengths to explore – a mix of medium and large sizes
    lengths = [2000, 4000, 8000, 12000, 20000, 30000]

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 2. Scan analytic families for every length
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        for cand in _seed_pool_for_length(n, rng):
            if time.time() >= deadline:
                break
            cand = cand / cand.sum()          # enforce sum = 1
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

            top_candidates.append((sc, cand.copy()))

        # keep only a handful of the very best seeds for later work
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:20]

    # ------------------------------------------------------------------
    # 3. If a previous solution is available, inject it
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Fallback – should never happen
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Main optimisation pipeline
    #    – allocate roughly half the remaining budget to gradient descent,
    #      the rest to annealing.
    remaining = deadline - time.time()
    if remaining <= 0:
        print("[warning] no time left for optimisation")
        return overall_best_seq.tolist()

    # number of seeds to devote gradient descent to (most promising)
    seeds_to_use = min(5, len(top_candidates))
    seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:seeds_to_use]]

    # 5a – smoothed gradient descent on each seed (share time)
    grad_deadline = time.time() + remaining * 0.55
    for seed_seq in seeds:
        if time.time() >= grad_deadline:
            break
        sub_dead = time.time() + (grad_deadline - time.time()) / max(1, len(seeds))
        best_seq, overall_best_score = _smooth_gradient_descent(
            seed_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.30,
            max_iter=25000,
            gamma_start=1.0,
            gamma_mult=1.5,
            gamma_step=500,
            eval_interval=250,
        )
        overall_best_seq = best_seq.copy()

    # 5b – final annealing / pairwise swaps
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _annealing_pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # make sure we return plain Python floats
    return overall_best_seq.tolist()
