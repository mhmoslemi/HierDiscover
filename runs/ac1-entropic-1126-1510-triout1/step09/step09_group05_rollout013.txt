        if M_new >= M - 1e-12:

"""
Propose a near‑optimal non‑negative step‑function.
The strategy is deliberately simple but very time‑focused:

1️⃣  Choose a single “promising” length – a little larger than the
    historically best length (≈ +10 %).  A longer sequence allows the
    maximal convolution value to shrink roughly like 1/n, which in
    turn lowers the objective 2·n·max(conv)/sum².

2️⃣  Initialise the distribution with a few analytic families
    (arcsine, symmetric beta, uniform, Gaussian, Cauchy, Dirichlet).
    Keep the best one as a seed.

3️⃣  Run a long smoothed‑Adam optimisation (log‑sum‑exp surrogate of
    the max‑convolution) with a slowly halving smoothing parameter.
    This gives a high‑quality smooth approximation of the true objective.

4️⃣  Refine the result with a plain projected sub‑gradient descent
    (the exact sub‑gradient of max(conv)).
    The remaining time is spent on this step, which is the only part
    that directly minimises the exact objective.

5️⃣  Finish with a tiny hill‑climbing pass that moves a few‑percent
    of the total mass between two indices, accepting only strictly
    improving moves.

All vectors are kept on the probability simplex (sum = 1); the final
sequence is returned as a plain list of floats.  The routine respects
the supplied time budget (default 1000 s) and prints progress
information.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap score = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.
    Returns (grad, cur_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even positions
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _smooth_grad(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smoothed max using the log‑sum‑exp surrogate:
        f_eps(a) = eps * log Σ_i exp(conv_i/eps)
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)               # length 2n‑1
    maxc = conv.max()
    w = np.exp((conv - maxc) / eps)           # unnormalised softmax weights
    w_sum = w.sum()
    w /= w_sum

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (even indices appear twice)
    diag = np.arange(conv.size)[(np.arange(conv.size) % 2) == 0] // 2
    grad[diag] -= w[2 * diag] * a[diag]

    return grad


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 500_000,
                          eta0: float = 0.9) -> tuple[np.ndarray, float]:
    """Mirror descent with KL divergence (exponentiated‑gradient)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / a.size)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 500_000,
                  step0: float = 0.25,
                  beta1: float = 0.9, beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam‑style adaptive gradient optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _smooth_adam_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 400_000,
                         eps0: float = 1e-3,
                         step0: float = 0.25,
                         beta1: float = 0.9, beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp smoothing of the max‑convolution."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    eps = eps0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad = _smooth_grad(a, eps)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        # halve the smoothing every 40 000 iterations
        if t % 40_000 == 0:
            eps = max(eps * 0.5, 1e-7)

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 1_500_000,
                         eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Plain sub‑gradient descent with a diminishing step‑size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

# Analytic families (all return a probability vector, i.e. sum = 1)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_w = (p - 1) * (np.log(x) + np.log(1.0 - x))
    log_w = np.clip(log_w, -700, 700)          # keep in float range
    w = np.exp(log_w)
    if w.sum() == 0.0:
        return np.full_like(w, 1.0 / n)
    return w / w.sum()


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _gaussian_distribution(n: int, sigma_frac: float = 0.30) -> np.ndarray:
    sigma = sigma_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if vals.sum() == 0.0:
        return np.full_like(vals, 1.0 / n)
    return vals / vals.sum()


def _cauchy_distribution(n: int, scale_frac: float = 0.30) -> np.ndarray:
    scale = scale_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = 1.0 / (1.0 + ((i - centre) / scale) ** 2)
    return vals / vals.sum()


def _random_dirichlet(n: int, alpha: float = 0.05) -> np.ndarray:
    """Highly‑sparse Dirichlet draw."""
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return x / x.sum()

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The routine respects the time budget,
    uses a single well‑chosen length (slightly larger than the
    historic best), starts from several analytical families,
    spends the bulk of the time on smoothed‑Adam + sub‑gradient
    descent, and finally performs a light hill‑climb.  The returned
    list is suitable for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    # small safety margin – we must return before the budget expires
    deadline = time.time() + budget_s - 0.5

    # --------------------------------------------------------------
    # 1. Choose a length a bit larger than the historic best.
    try:
        base_n = int(len(height_sequence_1))        # type: ignore[name-defined]
    except Exception:
        base_n = 3200

    # enlarge by 10‑12 % (more points → smaller max‑convolution)
    n = int(min(5000, base_n * 1.12))
    if n < 1500:
        n = 1500

    print(f"[{time.time():.1f}] chosen length n={n}")

    # --------------------------------------------------------------
    # 2. Build a tiny pool of analytic seeds and keep the best one.
    seed_vectors = []

    # a few classic families
    seed_vectors.append(_arcsine_distribution(n))
    seed_vectors.append(_beta_symmetric_distribution(n, p=0.2))
    seed_vectors.append(_beta_symmetric_distribution(n, p=0.5))   # arcsine again
    seed_vectors.append(_beta_symmetric_distribution(n, p=0.8))
    seed_vectors.append(_uniform_distribution(n))
    seed_vectors.append(_gaussian_distribution(n, sigma_frac=0.25))
    seed_vectors.append(_cauchy_distribution(n, scale_frac=0.25))
    seed_vectors.append(_random_dirichlet(n, alpha=0.02))
    seed_vectors.append(_random_dirichlet(n, alpha=0.01))

    # if a previous construction exists, resize it and add it
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            if prev.size != n:
                # linear interpolation to the new length
                x_old = np.linspace(0.0, 1.0, prev.size)
                x_new = np.linspace(0.0, 1.0, n)
                from scipy.interpolate import interp1d
                f = interp1d(x_old, prev, kind="linear", fill_value="extrapolate")
                prev = f(x_new)
            prev = np.clip(prev, 0.0, None)
            prev = _project_onto_simplex(prev, s=1.0)
            seed_vectors.append(prev)
    except Exception:
        pass

    # Tiny jitter for robustness
    for i, vec in enumerate(seed_vectors):
        jitter = np.random.randn(n) * 5e-5
        vec = np.clip(vec + jitter, 0.0, None)
        seed_vectors[i] = _project_onto_simplex(vec, s=1.0)

    # keep the ten best seeds according to the cheap score
    scored = [(vec, _max_convolution_score(vec)) for vec in seed_vectors]
    scored.sort(key=lambda x: x[1])
    top_seeds = [vec for vec, _ in scored[:8]]
    print(f"[{time.time():.1f}] kept {len(top_seeds)} seeds for optimisation "
          f"(best cheap score {scored[0][1]:.6f})")

    # --------------------------------------------------------------
    # 3. Main optimisation loop – we give the bulk of the budget to the
    #    best seed only (the others act as fall‑backs).
    best_seq = None
    best_score = float("inf")

    # allocate roughly 70 % of the time to the refinement of the top seed
    refine_deadline = deadline - (deadline - time.time()) * 0.30

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # start from the seed
        a = seed_vec.copy()
        cur_score = _max_convolution_score(a)

        # ----- smoothed‑Adam (log‑sum‑exp surrogate) -----------------
        a_smooth, sc_smooth = _smooth_adam_descent(
            a, deadline if idx == 0 else refine_deadline,
            max_iters=400_000, eps0=1e-3, step0=0.30
        )
        if sc_smooth < cur_score:
            a, cur_score = a_smooth, sc_smooth
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_seeds)} "
                  f"smooth‑Adam → {cur_score:.6f}")

        # ----- exponentiated‑gradient (mirror descent) ---------------
        a_exp, sc_exp = _exp_gradient_descent(
            a, deadline if idx == 0 else refine_deadline,
            max_iters=300_000, eta0=0.9
        )
        if sc_exp < cur_score:
            a, cur_score = a_exp, sc_exp
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_seeds)} "
                  f"exp‑grad → {cur_score:.6f}")

        # ----- Adam -------------------------------------------------
        a_adam, sc_adam = _adam_descent(
            a, deadline if idx == 0 else refine_deadline,
            max_iters=300_000, step0=0.25
        )
        if sc_adam < cur_score:
            a, cur_score = a_adam, sc_adam
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_seeds)} "
                  f"Adam → {cur_score:.6f}")

        # ----- long sub‑gradient descent (exact objective) ----------
        sub_deadline = deadline - 5.0   # keep ~5 s for final hill‑climb
        if time.time() < sub_deadline:
            a_sub, sc_sub = _subgradient_descent(
                a, sub_deadline,
                max_iters=1_200_000,
                eta0=0.8
            )
            if sc_sub < cur_score:
                a, cur_score = a_sub, sc_sub
                print(f"[{time.time():.1f}] seed {idx+1}/{len(top_seeds)} "
                      f"sub‑grad → {cur_score:.6f}")

        # keep the globally best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] new global best cheap {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Final cheap hill‑climbing (tiny mass transfers) – always cheap
    if best_seq is None:
        # fallback – uniform distribution of modest size
        best_seq = _uniform_distribution(1500)

    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    it = 0
    max_transfer = 0.03  # up to 3 % of total mass at the very start

    while time.time() < deadline:
        it += 1
        # geometric decay of the transferred mass
        transfer = max_transfer * (0.5 ** (it / 9000.0))

        # pick donor / receiver
        if it % 7 == 0:
            g, _, _ = _subgrad(a)
            i = int(np.argmax(g))
            j = int(np.argmin(g))
            if i == j:
                i = np.random.randint(n)
                j = (i + np.random.randint(1, n)) % n
        else:
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={it} new best {best_score:.6f}")

        # occasional jitter to shake out flat plateaus
        if it % 6000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 6e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
