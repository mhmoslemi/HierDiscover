def _simulated_annealing(a0, deadline, max_iters=200_000, init_temp=0.01, cooling=0.9995):
    a = a0.copy()
    cur_score = _max_convolution_score(a)
    best = a.copy()
    best_score = cur_score
    n = a.shape[0]
    temp = init_temp
    it = 0
    # to sample moves: choose i!=j and delta from distribution proportional to a[i].
    while time.time()<deadline and it<max_iters:
        it+=1
        # propose move
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j>=i:
            j+=1
        # delta fraction of total mass (maybe relative to a[i])
        max_delta = min(a[i], 0.05)  # at most 5%
        delta = np.random.rand() * max_delta
        # apply move
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)
        dE = new_score - cur_score
        if dE < 0 or np.random.rand() < np.exp(-dE / temp):
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-15:
                best = a_new.copy()
                best_score = new_score
        # cooling
        temp *= cooling
        # occasionally reheat to avoid stuck (increase temp).
        if it % 5000 == 0 and np.random.rand() < 0.1:
            temp = init_temp
    return best, best_score

# ----------------------------------------------------------------------
#  Simplex utilities

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
#  Convolution helpers (FFT based)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap surrogate = 2·n·max(a*a) (∑a=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

#  Sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.
    Returns (grad, cur_max, active_set).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    # uniform weights on the active set (any convex combination works)
    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (k = 2*i)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _smoothed_arcsine_distribution(n: int, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5 + eps) * (n - i - 0.5 + eps))
    return vals / vals.sum()


def _product_power_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-alpha)
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, p: float, q: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    w = np.where(np.isfinite(w), w, 0.0)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _gaussian_distribution(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle. sigma = sigma_frac * n."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = max(sigma_frac * n, 1e-9)
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _random_dirichlet(n: int, concentration: float = 1.0) -> np.ndarray:
    """Dirichlet draw with equal concentration."""
    x = np.random.gamma(concentration, 1.0, size=n)
    return x / x.sum()

def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 200_000, eta0: float = 1.0):

    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:

        a = _project_onto_simplex(a)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _subgradient_descent_line_search(a0: np.ndarray, deadline: float,
                                    max_iters: int = 200_000,
                                    step0: float = 1.0):
    """Projected sub‑gradient descent with a simple back‑tracking line search."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    step = step0
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, cur_max, active = _subgrad(a)
        direction = -grad
        improved = False
        # try a few decreasing step‑sizes
        for factor in (1.0, 0.5, 0.25, 0.125):
            cand = a + step * factor * direction
            cand = _project_onto_simplex(cand)
            score = _max_convolution_score(cand)
            if score < best_score - 1e-15:
                a = cand
                best_score, best_a = score, cand.copy()
                step *= 1.2   # push a little further next time
                improved = True
                break
        if not improved:
            step *= 0.5   # shrink if we cannot improve
    return best_a, best_score


def _simulated_annealing(a0: np.ndarray, deadline: float,
                         max_iters: int = 200_000,
                         init_temp: float = 0.02,
                         cooling: float = 0.9995):
    """Mass‑transfer annealing on the simplex."""
    a = a0.copy()
    cur_score = _max_convolution_score(a)
    best = a.copy()
    best_score = cur_score
    n = a.shape[0]
    temp = init_temp
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # propose a small transfer
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        max_delta = min(a[i], 0.05)   # at most 5 % of total mass
        delta = np.random.rand() * max_delta
        if delta <= 0:

        a_new = a.copy()

        new_score = _max_convolution_score(a_new)
        dE = new_score - cur_score
        if dE < 0 or np.random.rand() < np.exp(-dE / max(temp, 1e-12)):

            if new_score < best_score - 1e-15:

        temp *= cooling
        # occasional reheating
        if it % 5000 == 0 and np.random.rand() < 0.1:

        # fallback – use the first seed normalised
        global_best = candidate_seeds[0] / candidate_seeds[0].sum()
        global_best_score = _max_convolution_score(global_best)

    a = global_best.copy()
    cur_score = global_best_score
    n = a.shape[0]
    it = 0
    max_transfer = 0.02  # smaller jitter in the final stage

    while time.time() < deadline:
        it += 1
        # tiny random mass transfer
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = np.random.rand() * min(max_transfer, a[i])
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)
        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < global_best_score - 1e-14:
                global_best_score, global_best = new_score, a_new.copy()
                print(f"[{time.time() - start_time:.1f}s] final jitter improvement {global_best_score:.6f}")

        # occasional Gaussian jitter around the current best
        if it % 4000 == 0:
            jitter = np.random.randn(n) * 2e-5
            a = global_best + jitter
            a = _project_onto_simplex(a)
            cur_score = _max_convolution_score(a)

    print(f"[{time.time() - start_time:.1f}s] finished – cheap score {global_best_score:.6f}, length={len(global_best)}")
    return global_best.tolist()
