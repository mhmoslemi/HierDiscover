Search for a non‑negative step‑function (a list of floats) that minimises
``evaluate_sequence``.  The implementation follows three ideas:

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _normalise(seq):
    """Clamp to allowed range, enforce non‑negativity and project onto the simplex."""
    seq = np.asarray(seq, dtype=np.float64)

def _hann(n):
    """Classic Hann window, normalised to sum = 1."""
    if n == 1:

    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return w / w.sum()


def _kaiser(n, beta):
    """Kaiser window (beta may be any non‑negative float)."""
    if n == 1:
        return np.array([1.0])
    from math import i0  # modified Bessel of the first kind, order 0
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha
    w = i0(beta * np.sqrt(1.0 - t * t)) / i0(beta)
    return w / w.sum()


def _tukey(n, alpha):
    """Tukey (tapered cosine) window."""
    if n == 1:
        return np.array([1.0])
    i = np.arange(n, dtype=np.float64)
    w = np.ones_like(i)

    left = i < (alpha * (n - 1) / 2.0)
    right = i > ((n - 1) * (1 - alpha / 2.0))
    w[left] = 0.5 * (1 + np.cos(np.pi * (2 * i[left] / (alpha * (n - 1)) - 1)))
    w[right] = 0.5 * (1 + np.cos(np.pi * (2 * i[right] / (alpha * (n - 1)) - 2 / alpha + 1)))
    return w / w.sum()


def _dirichlet_draw(n, alpha):
    """Return a Dirichlet sample with concentration parameter *alpha* (scalar)."""
    if alpha == 0:

        vec[random.randrange(n)] = 1.0
        return vec
    gam = np.random.gamma(shape=alpha, scale=1.0, size=n)
    vec = gam / gam.sum()
    return vec
