def _lp_refine(f: np.ndarray, M: float, t: float = 0.5, verbose: bool=False) -> np.ndarray:
    n = f.shape[0]
    m = 2*n - 1
    # Build convolution matrix A
    # Use scipy.sparse.diags
    from scipy.sparse import diags
    A = diags(f, offsets=np.arange(n), shape=(m, n), format='csr')
    # Solve LP: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n, dtype=np.float64)
    b_ub = M * np.ones(m, dtype=np.float64)
    bounds = [(0, None)] * n
    # Use linprog with HiGHS
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        if verbose:
            print("LP failed:", res.message)
        return f
    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return f
    # Scale g to same sum as f
    g = g / sum_g
    # Convex combination
    new_f = (1 - t) * f + t * g
    # Project onto simplex to correct sum to 1
    new_f = new_f / new_f.sum()
    return new_f

def _lp_refine(f: np.ndarray, max_iter: int, t: float, deadline: float):
    from scipy.optimize import linprog
    from scipy.sparse import diags
    n = f.shape[0]
    m = 2 * n - 1
    # prebuild offsets array
    offsets = np.arange(n)
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # compute current M
        conv = _fft_self_convolve(f)
        M = conv.max()
        # Build convolution matrix A: each diagonal offset i has constant f[i]
        A = diags(f, offsets=offsets, shape=(m, n), format='csr')
        # Solve LP: maximize sum(g) => minimize -sum(g)
        c = -np.ones(n)
        b_ub = M * np.ones(m)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            # if fails, break
            print(f"LP iteration {it} failed: {res.message}")
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            # nothing to improve
            break
        # scale g to sum = 1
        g = g / sum_g
        # convex combination
        f = (1 - t) * f + t * g
        # project onto simplex (should already sum to 1, but ensure)
        f = _project_onto_simplex(f, s=1.0)
    return f
```

# Simplex projection
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# FFT‑based convolutions (real)
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

# Cheap surrogate (identical to true objective when sum=1)
def _max_convolution_score(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Exact sub‑gradient of max(a*a) on the simplex
def _subgrad(a: np.ndarray):
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

def _smooth_gradient(a: np.ndarray, gamma: float):
    n = a.shape[0]
    b = _fft_self_convolve(a)
    b_max = b.max()
    w = np.exp(gamma * (b - b_max))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    diag_w = w[2 * np.arange(n)]
    grad -= a * diag_w
    return grad, b_max

# LP‑based refinement (small scale)
def _lp_refine(f: np.ndarray, max_iter: int, t: float, deadline: float) -> np.ndarray:
    """
    Perform a few LP refinement steps on a small problem.
    Each step solves:
        maximise   ∑ g_j
        subject to (f * g)_k ≤ M   ∀k,   g ≥ 0,
    where M = max(f*f).  The solution g is rescaled to sum 1 and mixed
    with f (weight t).  The process stops when the deadline is reached.
    """
    n = f.shape[0]
    m = 2 * n - 1
    offsets = np.arange(n)

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        # current peak
        M = _fft_self_convolve(f).max()
        # convolution matrix (Toeplitz) – each diagonal i contains the constant f[i]
        A = diags(f, offsets=offsets, shape=(m, n), format="csr")
        # Linear programme via HiGHS (maximise sum g  <=> minimise -sum g)
        c = -np.ones(n, dtype=np.float64)
        b_ub = M * np.ones(m, dtype=np.float64)
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method="highs", options={"presolve": True})
        if not res.success:
            # If LP fails we abort the refinement.
            print(f"LP step failed: {res.message}")
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            break
        # Scale to the simplex and mix.
        g = g / sum_g
        f = (1.0 - t) * f + t * g
        f = _project_onto_simplex(f, s=1.0)
    return f

        base_n = 3500

    # Try a few nearby lengths (the objective is almost length‑invariant)
    lengths = [int(base_n * f) for f in (0.8, 0.9, 1.0, 1.1, 1.2)]
    lengths = [max(500, n) for n in lengths]

    # --------------------------------------------------------------
    # 1️⃣  Coarse sweep – keep the K best cheap candidates
    K = 5
    best_candidates = []   # list of (cheap_score, vector)

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- arcsine & smoothed arcsine ---------------------------------
        for eps in (0.0, 0.01, 0.1, 0.5, 1.0, 2.0):
            a = _arcsine_distribution(n) if eps == 0.0 else _smoothed_arcsine_distribution(n, eps)
            score = _max_convolution_score(a)
            if len(best_candidates) < K or score < best_candidates[-1][0]:
                best_candidates.append((score, a.copy()))
                best_candidates.sort(key=lambda x: x[0])
                best_candidates = best_candidates[:K]

        # ----- product‑power family ---------------------------------------
        for alpha in np.arange(0.30, 0.71, 0.05):
            a = _product_power_distribution(n, alpha)
            score = _max_convolution_score(a)
            if len(best_candidates) < K or score < best_candidates[-1][0]:
                best_candidates.append((score, a.copy()))
                best_candidates.sort(key=lambda x: x[0])
                best_candidates = best_candidates[:K]

        # ----- symmetric Beta (very fine grid) ---------------------------
        p_vals = np.concatenate([np.arange(0.05, 0.51, 0.05),
                                 np.arange(0.55, 0.91, 0.05)])
        for p in p_vals:
            a = _beta_symmetric_distribution(n, p)
            score = _max_convolution_score(a)
            if len(best_candidates) < K or score < best_candidates[-1][0]:
                best_candidates.append((score, a.copy()))
                best_candidates.sort(key=lambda x: x[0])
                best_candidates = best_candidates[:K]

        # ----- asymmetric Beta --------------------------------------------
        for p in (0.05, 0.10, 0.20, 0.30):
            for q in (0.5, 0.8, 1.0, 1.5):
                a = _beta_asymmetric_distribution(n, p, q)
                score = _max_convolution_score(a)
                if len(best_candidates) < K or score < best_candidates[-1][0]:
                    best_candidates.append((score, a.copy()))
                    best_candidates.sort(key=lambda x: x[0])
                    best_candidates = best_candidates[:K]

        # ----- exponential tails ------------------------------------------
        for lam in (0.001, 0.003, 0.005, 0.010, 0.020):
            a = _exponential_distribution(n, lam)
            score = _max_convolution_score(a)
            if len(best_candidates) < K or score < best_candidates[-1][0]:
                best_candidates.append((score, a.copy()))
                best_candidates.sort(key=lambda x: x[0])
                best_candidates = best_candidates[:K]

        # ----- uniform ----------------------------------------------------
        a = _uniform_distribution(n)
        score = _max_convolution_score(a)
        if len(best_candidates) < K or score < best_candidates[-1][0]:
            best_candidates.append((score, a.copy()))
            best_candidates.sort(key=lambda x: x[0])
            best_candidates = best_candidates[:K]

        # ----- mixture arcsine + uniform ----------------------------------
        for w in np.arange(0.0, 0.51, 0.1):
            a = _mix_arcsine_uniform(n, w)
            score = _max_convolution_score(a)
            if len(best_candidates) < K or score < best_candidates[-1][0]:
                best_candidates.append((score, a.copy()))
                best_candidates.sort(key=lambda x: x[0])
                best_candidates = best_candidates[:K]

        # ----- random Dirichlet (small concentration → spiky) ------------
        for alpha in (0.5, 0.3, 0.1):
            a = _random_dirichlet(n, alpha=alpha)
            score = _max_convolution_score(a)
            if len(best_candidates) < K or score < best_candidates[-1][0]:
                best_candidates.append((score, a.copy()))
                best_candidates.sort(key=lambda x: x[0])
                best_candidates = best_candidates[:K]

    # ----- incorporate the previous best, if any ------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev = prev / prev.sum()
            score = _max_convolution_score(prev)
            if len(best_candidates) < K or score < best_candidates[-1][0]:
                best_candidates.append((score, prev.copy()))
                best_candidates.sort(key=lambda x: x[0])
                best_candidates = best_candidates[:K]
    except Exception:
        pass

    print(f"[{time.time():.1f}] after coarse sweep best cheap={best_candidates[0][0]:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Small‑scale LP refinement (only on the overall best candidate)
    best_seq = best_candidates[0][1].copy()
    best_score = best_candidates[0][0]

    # Down‑sample to a size where the LP is cheap
    small_n = max(200, min(900, best_seq.shape[0] // 3))
    if small_n < best_seq.shape[0] and time.time() < deadline:
        a_small = _resize_distribution(best_seq, small_n)
        a_small = a_small / a_small.sum()
        lp_deadline = time.time() + (deadline - time.time()) * 0.15
        a_small = _lp_refine(a_small, max_iter=6, t=0.4, deadline=lp_deadline)
        # Upsample back to original length
        a_up = _resize_distribution(a_small, best_seq.shape[0])
        a_up = a_up / a_up.sum()
        up_score = _max_convolution_score(a_up)
        if up_score < best_score:
            best_seq, best_score = a_up, up_score
            print(f"[{time.time():.1f}] after LP refinement cheap={best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Heavy refinement – allocate the remaining budget here
    remaining = deadline - time.time()
    if remaining <= 0.0:
        return best_seq.tolist()

    # 3.1 Exponentiated‑gradient (mirror descent)
    sub_deadline = time.time() + remaining * 0.20
    seq, sc = _exp_gradient_descent(best_seq, sub_deadline,
                                    max_iters=120_000, eta0=1.0)
    if sc < best_score:
        best_seq, best_score = seq, sc
        print(f"[{time.time():.1f}] after exp‑grad cheap={best_score:.6f}")

    # 3.2 Adam
    sub_deadline = time.time() + remaining * 0.20
    seq, sc = _adam_descent(best_seq, sub_deadline,
                            max_iters=120_000, step0=0.3)
    if sc < best_score:
        best_seq, best_score = seq, sc
        print(f"[{time.time():.1f}] after Adam cheap={best_score:.6f}")

    # 3.3 Smooth‑max continuation
    sub_deadline = time.time() + remaining * 0.30
    seq, sc = _smooth_gradient_descent(best_seq, sub_deadline,
                                       max_iters=250_000,
                                       eta0=0.8,
                                       gamma0=2.0,
                                       gamma_factor=1.005)
    if sc < best_score:
        best_seq, best_score = seq, sc
        print(f"[{time.time():.1f}] after smooth cheap={best_score:.6f}")

    # 3.4 Heavy‑ball sub‑gradient
    sub_deadline = time.time() + remaining * 0.15
    seq, sc = _heavy_ball_subgradient_descent(best_seq, sub_deadline,
                                              max_iters=250_000,
                                              eta0=0.6, momentum=0.6)
    if sc < best_score:
        best_seq, best_score = seq, sc
        print(f"[{time.time():.1f}] after heavy‑ball cheap={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Fine‑grained hill‑climbing (random mass transfers + jitter)
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    it = 0
    max_transfer = 0.05

    while time.time() < deadline:
        it += 1
        # geometric decay of the transferred amount
        transfer = max_transfer * (0.5 ** (it / 8000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_seq, best_score = a_new.copy(), new_score
                print(f"[{time.time():.1f}] hill‑climb iter={it} new best {best_score:.6f}")

        # occasional jitter to escape plateaus
        if it % 4000 == 0 and np.random.rand() < 0.2:
            jitter = np.random.randn(n) * 1e-4
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
