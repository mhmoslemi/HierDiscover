The code below implements a more aggressive search for a non‑negative step function that minimizes  

\[
F(a)=\frac{2\,n\,\max (a*a)}{(\sum a)^2}\ .
\]

Because the objective is homogeneous we work with vectors whose entries sum to 1.  
The search consists of three stages

1. **Wide seed generation** – families of analytic distributions (uniform, arcsine,
   symmetric Beta with a fine grid of parameters, two‑block spikes, random sparse,
   asymmetric Beta) are instantiated for several lengths.  The best few seeds are kept.

2. **Smooth‑max mirror descent** – the non‑smooth `max` is replaced by a
   `log‑sum‑exp` with a temperature parameter α.  For a fixed α the gradient is
   obtained analytically (soft‑max weights of the convolution) and an
   exponentiated‑gradient (mirror) update is performed, followed by a projection
   onto the simplex.  α is increased slowly so that the algorithm first moves
   quickly on a smooth landscape and finally concentrates on the true maximum.
   The procedure runs for a time‑controlled number of iterations and keeps the
   best vector found.

3. **Local polishing** – a cheap Metropolis walk moves a random fraction of mass
   from one index to another; moves are kept only if they improve the exact
   objective.  This fine‑tunes the solution obtained by the previous stage.

All heavy operations (self‑convolution, gradient convolution) are carried out
with real FFTs, giving an \(O(n\log n)\) cost per iteration.  The algorithm
honours the 1000 s budget, gradually allocating time to seed evaluation,
descent on the best seeds and the final polishing step.

```python
"""
Propose a step‑function (list of non‑negative heights) that minimises
evaluate_sequence().  The algorithm:
1) builds a large pool of analytic seeds (uniform, arcsine, fine‑grid
   symmetric‑Beta, two‑block spikes, random sparse, random asymmetric‑Beta);
2) evaluates them cheaply using FFT‑based convolution and keeps the best few;
3) refines each of those seeds with a smooth‑max mirror‑descent (log‑sum‑exp
   approximation of the max) that uses exact gradients computed via FFT;
4) finishes with a greedy pairwise‑mass‑transfer walk that only accepts
   improving moves.
The whole search respects a 1000 s wall‑clock budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = s}.
    """
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (real FFT, linear convolution)
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*b using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    n = a.shape[0]
    m = b.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: n + m - 1]


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
# Objective (exact, for the normalised case Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families (all return a vector that sums to 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑space to avoid overflow for p close to 0
    logw = (p - 1) * (np.log(x) + np.log(1.0 - x))
    w = np.exp(logw - np.max(logw))   # subtract max for stability
    w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric beta (Beta(alpha,beta))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    logw = (alpha - 1) * np.log(np.clip(x, eps, 1.0)) + \
           (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0))
    w = np.exp(logw - np.max(logw))
    w /= w.sum()
    return w


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    """Two equal blocks at the ends, each of length ≈ block_frac·n."""
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Sparse random vector with prescribed density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals

def _candidate_pool(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """
    Build a broad pool of seeds for length n.
    """
    pool = []

    # classic families
    pool.append(_distribution_uniform(n))

                                            alpha=float(alpha),
                                            beta=float(beta)))

# ----------------------------------------------------------------------
# Smooth‑max mirror descent (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    deadline: float,
                    best_seq: np.ndarray,
                    best_score: float,
                    alpha0: float = 5.0,
                    alpha_factor: float = 1.02,
                    alpha_max: float = 3000.0,
                    eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smoothed objective
        f_α(a) = log‑sum‑exp(α·(a*a)) / α
    with α increasing slowly.  The true objective 2·n·max(a*a) is evaluated
    intermittently and the best vector is retained.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    alpha = alpha0

    while time.time() < deadline:
        t += 1

        # ----- convolution a*a -----
        conv = _fft_self_convolve(a)               # length 2n‑1

        # ----- soft‑max weights -----
        maxc = conv.max()
        exps = np.exp(alpha * (conv - maxc))        # safe from overflow
        w = exps / exps.sum()                       # sums to 1, length 2n‑1

        # ----- gradient of f_α -----
        w_rev = w[::-1]                             # reversed weights
        grad_full = _fft_convolve(a, w_rev)         # length 3n‑2
        L = conv.shape[0]                           # 2n‑1
        # extract 2·Σ_j a_j·w_{i+j}  for i=0…n‑1
        grad = 2.0 * grad_full[L - 1 - np.arange(n)]

        # ----- mirror (exponentiated) update -----
        # Clip the gradient to avoid overflow in the exponentiation
        grad = np.clip(grad, -50.0, 50.0)
        eta = eta0 / math.sqrt(t)                   # diminishing stepsize
        a = a * np.exp(-eta * grad)

        # ----- projection onto simplex Σa = 1, a ≥ 0 -----
        a = _project_onto_simplex(a)

        # ----- temperature schedule -----
        alpha = min(alpha * alpha_factor, alpha_max)

        # ----- exact evaluation every few iterations -----
        if t % 80 == 0:
            cur = 2.0 * n * conv.max()
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth it={t} α={alpha:.1f} "
                      f"new best {best_score:.6f}")

        # safety break if we run out of time
        if time.time() >= deadline:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Local greedy pairwise mass transfer (polishing)
# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Randomly move a small fraction of mass; keep move only if it improves the true objective."""
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = int(np.random.choice(src))
        # pick a destination different from i
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # transfer up to 30 % of the source mass
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    evaluate_sequence().  The function respects the given time budget (seconds)
    and returns the best vector found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Choose a set of lengths (larger n → finer discretisation)
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000, 20000, 30000]

    overall_best_seq = None
    overall_best_score = float("inf")
    global_top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 2. Scan analytic families for each length
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        for cand in _candidate_pool(n, rng):
            if time.time() >= deadline:
                break
            # normalise to Σ=1 (the objective is homogeneous)
            cand = cand / cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] NEW GLOBAL BEST len={n} score={sc:.6f}")
            global_top_candidates.append((sc, cand.copy()))

        # keep only a few elite seeds per length (to limit later work)
        global_top_candidates.sort(key=lambda x: x[0])
        global_top_candidates = global_top_candidates[:12]

    # ------------------------------------------------------------------
    # 3. If a previous solution exists, inject it as a seed
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] PREVIOUS BEST score={sc:.6f}")
            global_top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Fallback (should never happen)
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] FALLBACK uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Main optimisation pipeline
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    # allocate roughly half the remaining time to smooth descent on the elites
    descent_budget = remaining * 0.55
    swap_budget    = remaining - descent_budget

    # ----- 5a. Smooth‑max mirror descent on top seeds -----
    top_seeds = [seq for _, seq in sorted(global_top_candidates, key=lambda x: x[0])[:4]]
    if top_seeds:
        per_seed_deadline = time.time() + descent_budget / max(1, len(top_seeds))
        for seed_seq in top_seeds:
            if time.time() >= deadline:
                break
            # each seed runs its own descent (time‑controlled)
            overall_best_seq, overall_best_score = _smooth_descent(
                a0=seed_seq.copy(),
                deadline=per_seed_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                alpha0=5.0,
                alpha_factor=1.025,
                alpha_max=3000.0,
                eta0=0.5,
            )

    # ----- 5b. Final local greedy polishing -----
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            a=overall_best_seq.copy(),
            deadline=deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    # 6. Return the best vector (as a plain Python list)
    # ------------------------------------------------------------------
    print(f"FINAL BEST score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
```
