def _smooth_max_descent(a, steps, lr0, lr1, beta0, beta1, deadline):
    best = a.copy(); best_score = _max_convolution_score(a)
    for t in range(steps):
        if time.time() >= deadline:
            break
        frac = t / steps
        lr = lr0 * (lr1/lr0) ** frac
        beta = beta0 * (beta1/beta0) ** frac
        conv = _fft_self_convolve(a)
        w = conv ** (beta - 1)
        # Avoid overflow: if conv all zeros, set uniform weight
        # But conv values positive due to positivity. May have small values.
        # norm w maybe not needed; gradient can be computed with w directly.
        grad = 2.0 * _fft_convolve(a[::-1], w)[len(a)-1:len(a)-1+len(a)]
        # Exponentiated gradient
        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)

    return best, best_score

def _lp_refine_subset(f, mix_factor, deadline, top_k=20):
    n = len(f)
    conv_f = _fft_self_convolve(f)
    M = conv_f.max()
    # Identify top_k indices with largest conv entries
    idx = np.argpartition(conv_f, -top_k)[-top_k:]
    # Build matrix T for those indices only: each row corresponds to conv index.
    # For each k, row vector = f shifted? i.e., (f*g)[k] = sum_i f_i * g_{k-i}
    # Equivalent: T_ki = f_{k-i} if 0 <= k-i < n else 0.
    # We can build rows as Toeplitz columns pieces.
    rows = np.zeros((len(idx), n), dtype=np.float64)
    for row_idx, k in enumerate(idx):
        # compute f shifted
        # for each i, g_i multiplies f_{k-i}
        # we fill row: for i from max(0, k-(n-1)) to min(n-1, k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # for i in [i_start,i_end], weight = f[k-i]
        # we can assign using slice
        rows[row_idx, i_start:i_end+1] = f[k - np.arange(i_start, i_end+1)]
    # Now solve LP: maximize sum(g) subject to rows @ g <= M, g>=0
    c = -np.ones(n)
    b_ub = np.full(len(idx), M)
    # Solve
    try:
        res = linprog(c, A_ub=rows, b_ub=b_ub, bounds=(0, None), method='highs')
    except Exception as e:
        print(f"LP subset exception: {e}")
        return f, M

    if not res.success:
        return f, M

    g = res.x
    g_sum = g.sum()
    if g_sum < 1e-12:
        return f, M
    g_norm = g / g_sum

    # mixing as before
    best_f = f; best_M = M
    for t in (mix_factor, mix_factor * 0.5, mix_factor * 0.25):
        new_f = (1 - t) * f + t * g_norm
        new_f = np.maximum(new_f, 1e-12)
        new_f /= new_f.sum()
        new_M = _fft_self_convolve(new_f).max()
        if new_M < best_M - 1e-12:
            best_f = new_f; best_M = new_M
            break
    return best_f, best_M

"""
Hybrid search for a low‑value step‑function.

The algorithm proceeds in several stages within the 1000 s wall‑clock limit:

1. **Analytic families + parameter sweep**  
   Scan a handful of lengths (≈2000–3500) and a broad grid of
   parameters for several symmetric distributions (arcsine, Beta,
   sine‑power, Gaussian, semicircle, exponential, triangular, uniform,
   random Dirichlet).  The best candidate is kept.

2. **Down‑sampling of the best vector**  
   The best vector found in (1) is repeatedly down‑sampled to shorter
   lengths (by block‑averaging and re‑normalisation).  This often yields
   a lower objective because the factor `2·n` in the score is reduced.

3. **Smooth‑max gradient descent with an increasing “β”**  
   A mirror‑descent on the simplex is performed on the current best
   vector.  At iteration `t` the smooth‑max weight is
   `w_i = (conv_i+ε)^{β-1}` with `β` growing from 2 to 64, turning the
   smooth objective into the true max.  The update is
   `a ← a·exp(−lr·grad)` followed by Euclidean projection onto the
   simplex.

4. **Selective LP refinement**  
   Instead of the full `2n‑1` constraints (which is memory heavy) we
   build a tiny LP only for the `k` largest convolution entries
   (`k≈20`).  Solving `max Σ g_i  s.t.  f*g ≤ M` on this reduced set
   yields a feasible direction that can lower the maximum when mixed
   with the current vector.

5. **Hill‑climber / mass‑transfer moves**  
   Repeatedly move a small chunk of probability mass from the two
   coordinates that realise the current maximal convolution entry to a
   random coordinate, accepting the move only if the max decreases.

All stages respect the shared deadline.  The best sequence seen so far
is printed regularly and finally returned as a plain list of Python
floats, ready to be evaluated by ``evaluate_sequence``.
"""

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full linear self‑convolution a*a (real FFT)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a):
    """Score for a vector already normalised to sum = 1:
       2·n·max(a*a)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.01):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    """Discrete Gaussian centred at (n‑1)/2."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_semicircle(n):
    """Symmetric discrete semicircle distribution."""
    x = (np.arange(n, dtype=np.float64) - (n - 1) / 2) / ((n - 1) / 2)
    vals = np.sqrt(1.0 - x ** 2)
    vals[x > 1] = 0.0
    vals[x < -1] = 0.0
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals

def _downsample(vec, target_n):
    """Block‑average down‑sampling from len(vec) to target_n (target_n < len)."""
    src_n = vec.shape[0]
    if target_n >= src_n:
        raise ValueError("target_n must be smaller than source length")
    # size of each block (may be non‑integer)
    block = src_n / target_n
    out = np.empty(target_n, dtype=np.float64)
    for i in range(target_n):
        start = int(round(i * block))
        end = int(round((i + 1) * block))
        if start >= end:
            end = start + 1
        out[i] = vec[start:end].mean()
    out = np.maximum(out, 1e-15)
    out /= out.sum()
    return out

def _smooth_max_descent(a, steps, lr0, lr1, beta0, beta1, deadline):
    """Mirror‑descent on the simplex with an increasing smooth‑max exponent."""
    eps = 1e-15
    best = a.copy()
    best_score = _max_convolution_score(a)
    n = a.shape[0]
    for it in range(steps):
        if time.time() >= deadline:
            break
        t = it / steps
        lr = lr0 * (lr1 / lr0) ** t
        beta = beta0 * (beta1 / beta0) ** t
        conv = _fft_self_convolve(a)
        # smooth‑max weight (β‑1) power, add eps to avoid 0**neg.
        w = (conv + eps) ** (beta - 1.0)
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]
        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()
        score = _max_convolution_score(a)
        if score < best_score - 1e-12:
            best = a.copy()
            best_score = score
            print(f"[{time.time():.1f}] smooth it={it:04d} β={beta:.1f} score={score:.6f}")
    return best, best_score

def _lp_refine_subset(f, mix_factor, deadline, top_k=20):
    """LP refinement using only the top_k convolution entries."""
    n = f.shape[0]
    conv_f = _fft_self_convolve(f)
    M = conv_f.max()
    # indices of the k largest entries
    if top_k >= conv_f.size:
        idx = np.arange(conv_f.size)
    else:
        idx = np.argpartition(conv_f, -top_k)[-top_k:]
    rows = np.zeros((len(idx), n), dtype=np.float64)
    for r, k in enumerate(idx):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # range of i where g_i contributes to (f*g)_k
        rng = np.arange(i_start, i_end + 1, dtype=int)
        rows[r, i_start:i_end + 1] = f[k - rng]
    c = -np.ones(n, dtype=np.float64)          # maximise Σ g_i  <=> minimise –Σ g_i
    b_ub = np.full(len(idx), M, dtype=np.float64)
    try:
        res = linprog(c, A_ub=rows, b_ub=b_ub, bounds=(0, None), method='highs')
    except Exception as e:
        print(f"LP subset exception: {e}")
        return f, M
    if not res.success:
        return f, M
    g = res.x
    gsum = g.sum()
    if gsum < 1e-12:
        return f, M
    g_norm = g / gsum
    # try a few mixing ratios
    for t in (mix_factor, mix_factor * 0.5, mix_factor * 0.25, mix_factor * 0.1):
        new_f = (1 - t) * f + t * g_norm
        new_f = np.maximum(new_f, 1e-12)
        new_f /= new_f.sum()
        new_M = _fft_self_convolve(new_f).max()
        if new_M < M - 1e-12:
            return new_f, new_M
    return f, M

def _hill_climber(a, deadline):
    """Simple mass‑transfer hill‑climber."""
    eps = 1e-12
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    best = a.copy()
    best_score = 2.0 * n * M
    while time.time() < deadline:
        it += 1
        k = int(conv.argmax())
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        # locate the pair (i,j) giving the maximal product for this k
        best_i, best_j = i_low, k - i_low
        max_prod = a[best_i] * a[best_j]
        for i in range(i_low, i_high + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue
        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta < eps:
            continue
        snapshot = a.copy()
        a[source] -= delta
        a[target] += delta
        a = np.maximum(a, eps)
        a /= a.sum()
        conv = _fft_self_convolve(a)
        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
            score = 2.0 * n * M
            if score < best_score - 1e-12:
                best = a.copy()
                best_score = score
                if it % 500 == 0:
                    print(f"[{time.time():.1f}] hill it={it} score={score:.6f}")
        else:
            a = snapshot
            conv = _fft_self_convolve(a)
        if it % 5000 == 0:
            # small jitter to escape plateaus
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] jitter at hill iter {it}")
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:

    best_score = float('inf')
    best_n = None

        cand.append(_distribution_arcsine(n))
        cand.append(_distribution_triangular(n))
        cand.append(_distribution_semicircle(n))

        cand.append(_distribution_exponential(n, lam=0.004))
        cand.append(_distribution_exponential(n, lam=0.01))
        cand.append(_distribution_exponential(n, lam=0.03))

        # beta family with a fine grid
        for p in np.geomspace(0.2, 5.0, num=9):
            cand.append(_distribution_beta_sym(n, p))

        # sine‑power family
        for pw in (0.2, 0.5, 1.0, 2.0, 4.0, 8.0):
            cand.append(_distribution_sine_power(n, pw))

        # Gaussian family
        for sigma in (0.15, 0.25, 0.35, 0.45):
            cand.append(_distribution_gaussian(n, sigma))

        # random Dirichlet draws (several concentration levels)
        for alpha in (0.1, 0.3, 0.7, 1.5, 3.0):
            cand.append(np.random.dirichlet(np.full(n, alpha)))

        # evaluate all candidates
        for vec in cand:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] init n={n} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Down‑sample to explore shorter lengths
    if best_seq is not None:

                continue
            try:
                ds = _downsample(best_seq, target_n)

                    best_score = sc

            except Exception:
                continue

    # ------------------------------------------------------------------
    # 3. Smooth‑max mirror descent (multiple restarts)
    if best_seq is None:
        # fallback: uniform vector of moderate size
        best_n = 2500
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)

    # a few restarts with jittered starts
    restarts = 3
    for r in range(restarts):
        if time.time() >= deadline:
            break
        a = best_seq.copy()
        # add tiny jitter to escape static local minima
        a += np.random.randn(a.size) * 1e-5
