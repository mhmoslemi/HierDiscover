"""
Propose a step‑function (non‑negative list) that minimizes
    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)^2).

The algorithm works on the probability simplex (sum = 1), because the
objective is scale‑invariant.  It consists of three phases:

1️⃣  Broad sampling – generate a modest pool of analytic families (uniform,
    triangular, beta‑laws, Dirichlet draws, etc.) for a few different
    lengths (500 – 2000).  All candidates are evaluated by the cheap proxy
    `2·n·max(conv(a,a))` and the best few are kept.

2️⃣  Deep smooth‑Adam refinement – starting from each seed we minimise the
    smooth log‑sum‑exp surrogate of the max‑convolution.
    The smoothing parameter ε is decreased gradually and Adam is run for a
    large number of iterations (up to the remaining time budget).  The best
    sequence over all seeds is retained.

3️⃣  Final hill‑climb – a cheap, gradient‑guided transfer of a tiny amount of
    mass between two indices is performed repeatedly until the deadline.
    Occasionally a small random jitter is added to escape flat plateaus.

The whole routine respects the supplied 1000 s budget and always returns a
feasible list of floats (non‑negative, sum > 0.01).  The returned list is
already normalised (sum = 1) so `evaluate_sequence` will compute exactly the
cheap score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap proxy = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _smooth_grad(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smoothed max using a log‑sum‑exp surrogate:
        f_eps(a) = eps * log Σ_i exp(conv_i/eps)
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)          # length 2n‑1
    maxc = conv.max()
    w = np.exp((conv - maxc) / eps)       # un‑normalised softmax weights
    w_sum = w.sum()
    w /= w_sum

    # convolution of reversed a with the soft weights
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even positions (each appears twice)
    diag_idx = np.arange(conv.size)[(np.arange(conv.size) % 2) == 0] // 2
    grad[diag_idx] -= w[2 * diag_idx] * a[diag_idx]

    return grad


# ----------------------------------------------------------------------
def _smooth_adam_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 500_000,
                         eps0: float = 5e-3,
                         step0: float = 0.25,
                         beta1: float = 0.9, beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑sum‑exp smoothed max‑convolution.
    The smoothing parameter ε is decreased geometrically.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    t = 0
    eps = eps0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad = _smooth_grad(a, eps)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        # decrease ε every few thousand steps (geometric schedule)
        if t % 20000 == 0:
            eps = max(eps * 0.7, 1e-9)

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(conv(a,a)) on the simplex.
    Returns (grad, cur_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with the active‑mask weight vector
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for indices where the mask hits an even position
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 2_000_000,
                         eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Plain sub‑gradient descent with a diminishing step‑size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

# Analytic families (all return a vector that already sums to 1)
def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """f(x) ∝ x^{p‑1} (1‑x)^{p‑1} on the grid { (i+½)/n }."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑space to avoid overflow for tiny / huge p
    log_w = (p - 1) * (np.log(x) + np.log(1.0 - x))
    log_w = np.clip(log_w, -700, 700)
    w = np.exp(log_w)
    if w.sum() == 0.0:
        return np.full_like(w, 1.0 / n)
    return w / w.sum()


def _random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    """Dirichlet draw with concentration α."""
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return x / x.sum()

    # ------------------------------------------------------------------
    # Keep the most promising seeds (few of them)
    raw_candidates.sort(key=lambda x: x[1])
    top_k = 6
    seeds = [c[0] for c in raw_candidates[:top_k]]
    print(f"[{time.time():.1f}] selected {len(seeds)} seeds, best cheap score={raw_candidates[0][1]:.6f}")

    # ------------------------------------------------------------------
    # 2. Deep smooth‑Adam refinement on each seed (most of the time budget)
    best_seq = seeds[0].copy()
    best_score = _max_convolution_score(best_seq)

    # allocate the bulk of the budget to the refinement stage
    refinement_deadline = deadline - 30.0   # keep a few seconds for final hill‑climb

    for idx, seed_seq in enumerate(seeds):
        if time.time() >= refinement_deadline:
            break
        seq, sc = _smooth_adam_descent(seed_seq,
                                       deadline=refinement_deadline,
                                       max_iters=500_000,
                                       eps0=5e-3,
                                       step0=0.25)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(seeds)} smooth‑Adam → {sc:.6f}")

    # ------------------------------------------------------------------
    # 3. Heavy sub‑gradient polishing (optional – run only if time)
    if time.time() < refinement_deadline:
        seq, sc = _subgradient_descent(best_seq,
                                       deadline=refinement_deadline,
                                       max_iters=2_000_000,
                                       eta0=0.7)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] sub‑gradient polishing → {sc:.6f}")

    # ------------------------------------------------------------------
    # 4. Final hill‑climbing (tiny transfers, occasional jitter)
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    iter_cnt = 0
    max_transfer = 0.02   # start with 2 % of total mass

    while time.time() < deadline:
        iter_cnt += 1
        # geometric decay of the transferred mass
        transfer = max_transfer * (0.5 ** (iter_cnt / 6000.0))

        # pick donor / receiver
        if iter_cnt % 7 == 0:
            # use the exact sub‑gradient to locate a promising pair
            g, _, _ = _subgrad(a)
            i = int(np.argmax(g))          # donor (large gradient)
            j = int(np.argmin(g))          # receiver (small gradient)
        else:
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={iter_cnt} new best {best_score:.6f}")

        # occasional jitter to shake off plateaus
        if iter_cnt % 8000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 5e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] final cheap score: {best_score:.6f} (length={len(best_seq)})")
    # ensure the output is a plain list of floats
    return best_seq.tolist()
