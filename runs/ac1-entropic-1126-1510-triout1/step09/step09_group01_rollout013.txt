def propose_candidate(seed=123, budget_s=1000):
    import time, random, math, numpy as np
    start = time.time()
    deadline = start + budget_s - 0.5
    np.random.seed(seed)
    random.seed(seed)

    # set epsilon
    eps = 1e-14

    # Step 1: broad exploration
    best_seq = None
    best_score = float('inf')
    best_n = None
    lengths = [1600, 3200, 6400, 12800, 25600]   # moderate
    for n in lengths:
        if time.time() >= deadline:
            break
        # generate candidates
        cand_list = []
        # analytic families
        cand_list.append(_distribution_uniform(n))
        cand_list.append(_distribution_arcsine(n))
        cand_list.append(_distribution_triangular(n))
        for lam in (0.001, 0.003, 0.01, 0.04):
            cand_list.append(_distribution_exponential(n, lam))
        # Beta general grid
        for a in (0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0, 3.0):
            for b in (0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0, 3.0):
                cand_list.append(_distribution_beta_general(n, a, b))
        # sine-power
        for pw in (0.5, 1.0, 2.0, 4.0, 6.0, 10.0):
            cand_list.append(_distribution_sine_power(n, pw))
        # cosine window
        for coeff in (0.2, 0.4, 0.6, 0.8):
            cand_list.append(_distribution_cosine_window(n, coeff))
        # Gaussian
        for sigma in (0.08, 0.12, 0.2, 0.3):
            cand_list.append(_distribution_gaussian(n, sigma))
        # random Dirichlet
        for alpha in (0.1, 0.3, 0.7, 1.5):
            cand_list.append(np.random.dirichlet(np.full(n, alpha)))
        # sparse random
        k = max(2, n // 500)
        sparse = np.zeros(n)
        idx = np.random.choice(n, k, replace=False)
        sparse[idx] = np.random.rand(k)
        sparse /= sparse.sum()
        cand_list.append(sparse)

        # evaluate each
        for vec in cand_list:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] len={n} new best {sc:.6f}")

    # If no candidate (unlikely)
    if best_seq is None:
        best_n = 2000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"fallback uniform {best_score}")

    # Step 2: Build a small population from variations of best_seq
    pop_size = 30
    pop = []
    # Insert best
    pop.append((best_seq.copy(), best_score))
    # Generate rest by mutations of best
    while len(pop) < pop_size and time.time() < deadline:
        # random perturbation
        a = best_seq.copy()
        # add gaussian noise
        noise = np.random.randn(a.shape[0]) * 0.05
        a += noise
        a = np.maximum(a, eps)
        a /= a.sum()
        sc = _max_convolution_score(a)
        pop.append((a, sc))
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time():.1f}] pop init improved to {sc:.6f}")

    # Ensure sorted by score ascending
    pop.sort(key=lambda x: x[1])

    # Step 3: Evolutionary loop
    it = 0
    while time.time() < deadline:
        it += 1
        # select parents via tournament (top 10)
        # Choose two from top half of population
        idxs = np.random.choice(len(pop)//2, 2, replace=False)
        a1, s1 = pop[idxs[0]]
        a2, s2 = pop[idxs[1]]
        # Crossover
        w = random.random()
        child = w * a1 + (1 - w) * a2
        # Mutation
        sigma = 0.02 / (1 + it * 0.0001)  # decaying
        child += np.random.randn(child.size) * sigma
        child = np.maximum(child, eps)
        child /= child.sum()
        sc = _max_convolution_score(child)
        # Insert into population if good
        if sc < best_score - 1e-12:
            best_score, best_seq = sc, child.copy()
            print(f"[{time.time():.1f}] iter {it} new global best {sc:.6f}")
        # Replace worst if child better than worst
        if sc < pop[-1][1]:
            pop[-1] = (child, sc)
            pop.sort(key=lambda x: x[1])

        # occasional local hill climb on best_seq
        if it % 2000 == 0:
            best_seq, best_score = _local_mass_transfer(best_seq, best_score, deadline, eps)

    # Step 4: final small hill climb (maybe 2000 steps)
    best_seq, best_score = _local_mass_transfer(best_seq, best_score, deadline, eps)

    # return as list
    print(f"Final best score {best_score:.6f} length {best_seq.shape[0]}")
    return best_seq.tolist()

def _local_mass_transfer(a, cur_score, deadline, eps):
    n = a.shape[0]
    # compute conv once
    while time.time() < deadline - 0.5:
        conv = _fft_self_convolve(a)
        M = conv.max()
        if M < cur_score / (2*n) - 1e-12: # not needed but cannot guarantee
            pass
        # Find k where max occurs
        k = int(conv.argmax())
        # Determine range of i values giving this sum
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # Find pair with largest product among active
        best_i = i_start
        best_j = k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j
        # choose source among them (the larger component)
        source = best_i if a[best_i] >= a[best_j] else best_j
        # choose random target not equal source
        target = random.randint(0, n-1)
        if target == source:
            continue
        # determine delta
        max_delta = min(a[source] * 0.5, 0.02)
        if max_delta <= eps:
            break
        delta = max_delta * random.random()
        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_score = _max_convolution_score(a2)
        # accept if better
        if new_score < cur_score:
            a = a2
            cur_score = new_score
        else:
            # maybe accept with small probability based on temperature
            # simple simulated annealing
            T = 0.001 * (deadline - time.time()) / (deadline - start)  # decaying
            if random.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
                a = a2
                cur_score = new_score
        # optional break after many steps
        if random.random() < 0.001:
            break
    return a, cur_score

"""
Propose a non‑negative step function (list of floats) that minimises
``evaluate_sequence``.  The algorithm consists of four stages:

1️⃣ **Broad analytic search** – for several lengths we evaluate many
   parametric families (uniform, arcsine, triangular, exponential,
   asymmetric Beta, power‑law, heavy‑tail, sine‑power, cosine‑window,
   Gaussian) and a few random Dirichlet draws.  The best few candidates
   for each length are kept.

2️⃣ **Smooth‑max mirror descent** – each kept candidate is refined by
   exponentiated‑gradient (mirror) descent on a soft‑max approximation
   of the convolution maximum.  The temperature (α) is increased from
   1 to 5·10⁴ while the learning rate decays, which first explores the
   simplex and then optimises the true objective.

3️⃣ **Evolutionary polishing** – a small population (≈30) of the
   refined vectors is evolved with crossover, Gaussian mutation and
   simplex projection.  The population always contains the current
   best vector.

4️⃣ **Mass‑transfer hill‑climber** – a final cheap local search moves
   tiny amounts of mass from indices that contribute most to the
   current maximum convolution towards random targets.  The move is
   accepted if it lowers the objective (or with a tiny annealing
   probability).

All steps respect a user‑supplied time budget (default 1000 s).  The
returned list is already normalised (∑aᵢ = 1) and contains only
non‑negative numbers, which makes ``evaluate_sequence`` invariant to
the scaling.
"""

import time
import random
import math
import numpy as np

# Helper functions ------------------------------------------------------

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a):
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a):
    """
    Objective 2·n·max(a*a) for a vector already normalised to sum = 1.
    This equals the value returned by evaluate_sequence (apart from the
    scaling‑invariant denominator).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


def _normalise(vec, eps=1e-14):
    """Clip negatives, enforce minimal positive mass and renormalise."""
    vec = np.maximum(vec, eps)
    s = vec.sum()
    if s == 0:
        vec[:] = 1.0 / vec.size
        s = 1.0
    return vec / s

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    w = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    w /= w.sum()
    return w


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    w = np.concatenate([left, right])
    if w.size != n:               # pad if odd length was mismatched
        w = np.pad(w, (0, n - w.size), constant_values=0)
    w /= w.sum()
    return w


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-lam * np.abs(idx - centre))
    w /= w.sum()
    return w


def _distribution_beta(n, a, b):
    """Beta(a,b) on the discrete grid {0,…,n‑1} (a,b>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use log‐space for stability
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full(n, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_law(n, s):
    """Symmetric power‑law: w_i ∝ (i+1)^{-s} + (n-i)^{-s}."""
    i = np.arange(n, dtype=np.float64) + 1
    w = i ** (-s) + (n - i + 1) ** (-s)
    w /= w.sum()
    return w


def _distribution_heavy_tail(n, lam=0.001, gamma=0.5):
    """Heavy‑tail: exp(‑lam * |i‑centre|^γ) with 0<γ≤1."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-lam * np.abs(i - centre) ** gamma)
    w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    i = np.arange(n, dtype=np.float64)
    w = np.sin(np.pi * (i + 0.5) / n) ** power
    w = np.maximum(w, 0)
    w /= w.sum()
    return w


def _distribution_cosine_window(n, coeff=0.5):
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    w = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    w = np.maximum(w, 0)
    w /= w.sum()
    return w


def _distribution_gaussian(n, sigma):
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    w = np.maximum(w, 0)
    w /= w.sum()
    return w

def _smooth_max_mirror_descent(
    a0,
    schedule,
    time_deadline,
    eps=1e-14,
):
    """Mirror descent on the smooth‑max of the self‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_score(best)

    for (alpha0, alpha1, steps, lr0, lr1) in schedule:
        if time.time() >= time_deadline:
            break
        for it in range(steps):
            if time.time() >= time_deadline:
                break
            t = it / steps
            alpha = alpha0 * (alpha1 / alpha0) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of smooth‑max (conv of a with w)
            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

            # exponentiated‑gradient step
            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = _normalise(a, eps)

            if (it & 511) == 0:      # every 512 iterations
                cur = _max_convolution_score(a)
                if cur < best_score - 1e-12:
                    best_score = cur
                    best = a.copy()
    return best, best_score


def _mass_transfer_hill_climb(
    a,
    cur_score,
    time_deadline,
    eps=1e-14,
    max_iter=5000,
):
    """
    Very cheap local search that repeatedly moves an amount of mass from a
    “source” index (participating in the current maximal convolution term)
    to a random “target” index.  The move is kept if it improves the
    objective (or with a tiny annealing probability).
    """
    n = a.shape[0]
    while max_iter > 0 and time.time() < time_deadline:
        max_iter -= 1
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # find the pair (i,j) with largest product among those that give k
        best_i, best_j = i_start, k - i_start
        best_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > best_prod:
                best_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        max_delta = min(a[source] * 0.5, 0.02)
        if max_delta <= eps:
            continue
        delta = max_delta * random.random()
        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = _normalise(a2, eps)
        new_score = _max_convolution_score(a2)
        if new_score < cur_score:
            a = a2
            cur_score = new_score
        else:
            # occasional simulated‑annealing acceptance to escape plateaus
            T = 0.001 * (time_deadline - time.time())
            if random.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
                a = a2
                cur_score = new_score
    return a, cur_score

    pop = refined_candidates[:5]

    # ------------------------------------------------------------------
    # 4️⃣  Evolutionary polishing (population based)
    # ------------------------------------------------------------------
    POP_SIZE = 30
    # fill the rest of the population with random perturbations of the best
    while len(pop) < POP_SIZE and time.time() < deadline:
        noise = np.random.randn(best_n) * 0.03
        vec = best_seq + noise
        vec = _normalise(vec, eps)
        sc = _max_convolution_score(vec)
        pop.append((vec, sc))
        if sc < best_score:
            best_score, best_seq = sc, vec.copy()
            print(f"[{time.time():.1f}] initial pop improved to {sc:.6f}")

    # always keep the list sorted (ascending scores)
    pop.sort(key=lambda t: t[1])

    it = 0
    while time.time() < deadline:
        it += 1
        # tournament selection from the top half of the population
        parent_idx = np.random.choice(len(pop) // 2, 2, replace=False)
        a1, _ = pop[parent_idx[0]]
        a2, _ = pop[parent_idx[1]]

        # crossover
        w = random.random()
        child = w * a1 + (1 - w) * a2

        # mutation – decaying Gaussian noise
        sigma = 0.02 / (1 + it * 0.0005)
        child += np.random.randn(child.size) * sigma
        child = _normalise(child, eps)

        child_score = _max_convolution_score(child)

        # keep global best
        if child_score < best_score - 1e-12:
            best_score, best_seq = child_score, child.copy()
            print(f"[{time.time():.1f}] evo it {it} new global best {child_score:.6f}")

        # replace the worst individual if the child is better
        if child_score < pop[-1][1]:
            pop[-1] = (child, child_score)
            pop.sort(key=lambda t: t[1])

        # every few thousand iterations perform a cheap hill‑climb on the best
        if it % 3000 == 0:
            best_seq, best_score = _mass_transfer_hill_climb(
                best_seq, best_score, deadline, eps, max_iter=2000
            )
            # keep the population consistent
            pop[-1] = (best_seq, best_score)
            pop.sort(key=lambda t: t[1])

    # ------------------------------------------------------------------
    # 5️⃣  Final mass‑transfer polishing (if time permits)
    # ------------------------------------------------------------------
    best_seq, best_score = _mass_transfer_hill_climb(
        best_seq, best_score, deadline, eps, max_iter=5000
    )

    print(f"Final best score ≈ {best_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
