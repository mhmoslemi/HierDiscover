"""
Propose a step‑function sequence that (hopefully) improves the
objective 2·n·max(conv(a,a)) / (sum(a)**2).

Algorithm
---------
1. Start from the best known sequence (global variable ``height_sequence_1``)
   or from an analytically‑generated arcsine distribution.
   Scale the vector to the canonical sum √(2 n) (the objective is
   scale‑invariant).

2. Repeatedly solve a linear programme that, for the current vector `f`,
   finds a vector `g` with the **largest possible total mass**
   while keeping the convolution bound `f*g ≤ M` (where `M` is the current
   worst auto‑correlation).  This is the Frank‑Wolfe sub‑problem described
   in the statement.

3. Mix the current vector and the LP solution:
      f← (1‑τ)·f + τ·g   (τ is a small step, chosen by a tiny line‑search)
   then re‑project onto the simplex of sum √(2 n).

4. Whenever a blend yields a lower cheap score, keep it as the current
   best.  The loop stops when the time budget is almost exhausted.

5. As a final polish, run a few iterations of the sub‑gradient descent
   from the original code (projected onto the simplex) to squeeze out
   any remaining gain.

All heavy work respects the 1000 s budget; the function returns a list of
non‑negative floats that can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy import sparse

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _cheap_score(a: np.ndarray) -> float:
    """Score = 2·n·max(conv) / (sum(a)**2);  works for any positive scaling."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max() / (a.sum() ** 2)


# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """
    Build the matrix A (size (2n‑1)×n) such that A @ g  =  f * g.
    Row k contains the coefficients f_i for which g_{k-i} participates.
    """
    n = f.shape[0]
    # each diagonal i (offset i) consists of the constant f[i]
    data = [np.full(n, f[i]) for i in range(n)]
    offsets = np.arange(n)
    return sparse.diags(data, offsets, shape=(2 * n - 1, n), format='csr')


# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float, target_sum: float) -> np.ndarray:
    """
    Solve   max sum(g)   s.t.   f * g ≤ M , g ≥ 0 .
    Returned vector g is scaled to have sum equal to ``target_sum``.
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)                     # (2n‑1)×n sparse matrix
    b_ub = np.full(2 * n - 1, M)

    # objective: minimise -∑ g   (i.e. maximise ∑g)
    c = -np.ones(n)

    bounds = [(0, None)] * n

    # Solve with HiGHS (fast, works with sparse matrices)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs')
    if not res.success or res.x is None:
        # fallback to a uniform vector (still feasible)
        g0 = np.full(n, target_sum / n)
    else:
        g0 = res.x

    g_sum = g0.sum()
    if g_sum == 0.0:
        return np.full(n, target_sum / n)
    scale = target_sum / g_sum
    return g0 * scale


# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 500_000,
                         eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Plain sub‑gradient descent (projected onto the simplex of sum 1)."""
    a = a0 / a0.sum()              # work on the probability simplex
    n = a.shape[0]
    best_a = a.copy()
    best_score = _cheap_score(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        # exact sub‑gradient of max(conv) on the simplex
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps)

        w = np.zeros_like(b)
        w[active] = 1.0
        conv_rev = np.fft.irfft(
            np.fft.rfft(a[::-1], n=1 << ((a.size + b.size - 1) - 1).bit_length()) *
            np.fft.rfft(w, n=1 << ((a.size + b.size - 1) - 1).bit_length())
        )
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # diagonal correction for even positions
        even = (active % 2 == 0)
        diag_idx = active[even] // 2
        if diag_idx.size:
            grad[diag_idx] -= a[diag_idx]

        it += 1
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _cheap_score(a)
        if cur_score < best_score - 1e-15:
            best_score, best_a = cur_score, a.copy()
    # bring back to the original scaling (sum = sqrt(2n))
    return best_a * (np.sqrt(2 * n) / best_a.sum()), best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine stays within the given time budget and returns the best
    sequence it discovers.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # tiny safety margin

    # --------------------------------------------------------------
    # 1️⃣  Initial vector (use previous best if available)
    try:
        a0 = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if a0.ndim != 1:
            raise ValueError
    except Exception:
        # fall back to a moderately sized analytic distribution
        n_default = 3000
        a0 = np.sin(np.pi * (np.arange(n_default) + 0.5) / n_default)  # sine‑like
    a0 = np.maximum(a0, 0.0)
    if a0.sum() == 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)

    n = a0.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # Scale to the canonical sum √(2 n)
    a = a0 / a0.sum() * target_sum
    best_a = a.copy()
    best_score = _cheap_score(a)
    print(f"[{time.time():.1f}] start length={n} initial cheap score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Frank‑Wolfe‑style LP refinement loop
    # parameters
    max_lp_iters = 25            # hard limit; each iteration may cost a few sec
    step_choices = [0.1, 0.2, 0.3, 0.4]   # candidate mixing factors

    lp_iter = 0
    while time.time() < deadline - 5.0 and lp_iter < max_lp_iters:
        lp_iter += 1
        # current worst autocorrelation
        conv_f = _fft_self_convolve(a)
        M = conv_f.max()

        # solve LP sub‑problem
        g = _lp_step(a, M, target_sum)

        # try a few mixing coefficients and keep the best blend
        best_local_a = None
        best_local_score = np.inf
        for tau in step_choices:
            blended = (1.0 - tau) * a + tau * g
            # re‑scale back to the exact target sum (projection onto the affine set)
            blended = blended * (target_sum / blended.sum())
            sc = _cheap_score(blended)
            if sc < best_local_score:
                best_local_score = sc
                best_local_a = blended

        # accept if we made progress
        if best_local_score < best_score - 1e-12:
            best_score = best_local_score
            best_a = best_local_a.copy()
            a = best_local_a
            print(f"[{time.time():.1f}] LP iter {lp_iter} → new cheap score={best_score:.6f}")
        else:
            # even if no improvement, still move a little towards g to escape plateaus
            a = (0.9) * a + 0.1 * g
            a = a * (target_sum / a.sum())

    # --------------------------------------------------------------
    # 3️⃣  Final polishing with sub‑gradient descent (on the simplex)
    subgrad_deadline = deadline - 2.0
    if time.time() < subgrad_deadline:
        seq, sc = _subgradient_descent(best_a, subgrad_deadline,
                                       max_iters=300_000, eta0=0.5)
        if sc < best_score:
            best_score = sc
            best_a = seq
            print(f"[{time.time():.1f}] sub‑gradient polish → cheap score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Return the best sequence found (any positive scaling is fine)
    final_seq = best_a.tolist()
    final_obj = evaluate_sequence(final_seq)  # sanity check – will be printed only
    print(f"Final cheap score: {best_score:.6f} (objective={final_obj:.6f}) length={len(final_seq)}")
    return final_seq
```

This implementation builds on the previous exhaustive sampling approach but adds a **Frank‑Wolfe linear‑programming refinement** that directly maximises the feasible mass under the current convolution bound.  By repeatedly mixing the LP solution back into the current vector (with a tiny line‑search over the step size) we obtain a monotone decrease of the worst auto‑correlation.  A short sub‑gradient polishing stage finishes the optimisation.  The routine respects the 1000‑second budget and always returns a feasible list of non‑negative floats.
