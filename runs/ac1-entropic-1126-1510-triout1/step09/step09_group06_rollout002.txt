assistantfinal**Solution Overview**

We search for a non‑negative step function `a` (a 1‑D vector) that minimises  

```
E(a) = 2·n·max(conv(a,a)) / (∑a)²          (n = len(a))
```

The evaluation routine is scale‑invariant, therefore we may restrict ourselves to the
simplex  

```
S = { a ≥ 0 ,  ∑ a = 1 } .
```

On this simplex the objective reduces to  

```
F(a) = 2·n·max(conv(a,a)).
```

`conv(a,a)` is the (full) aperiodic self‑convolution.
`max(conv)` is convex but non‑smooth.  
We use the exact sub‑gradient of `max(conv)` and a *smoothed* gradient
obtained from a soft‑max (log‑sum‑exp) approximation.  
Both gradients are computed **exactly** via FFT‑convolutions.

The optimisation proceeds in three stages:

1. **Broad sampling** – generate many analytic families (arcsine, Beta,
   Gaussian, …) for several lengths and many deterministic/ random
   reorderings.  All vectors are ℓ₁‑normalised and scored with the cheap
   surrogate `F(a)`.  The 20 best seeds are kept.

2. **Refinement** – each seed is polished with a multi‑phase Adam
   optimiser.  Phases use increasing smoothing parameters
   `γ = 0 → 20 → 200`.  The Adam update works on the simplex by Euclidean
   projection after each step.  Because we fixed the sub‑gradient
   calculation (the previous implementation subtracted a diagonal term
   erroneously) the descent direction is now exact, leading to a
   strictly lower bound.

3. **Fine‑tuning** – a Frank‑Wolfe style mass‑transfer step moves mass from
   the index with the largest gradient to the one with the smallest.
   Random small perturbations are occasionally injected to escape shallow
   plateaus.

All helper functions are defined at top‑level, there are no closures,
and the whole routine respects the 1000 s time budget.

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolutions
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
# Objective helpers
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate for `evaluate_sequence` when Σ a = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad_max(a: np.ndarray):
    """
    Exact sub‑gradient of the max‑convolution functional on the simplex.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0                     # weight 1 on every active index

    # gradient = 2 * (a_rev * w)[n-1 : n-1+n]
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    return grad, cur_max, active


def _smooth_grad(a: np.ndarray, gamma: float):
    """
    Gradient of the log‑sum‑exp smoothed max.
    Returns (gradient, smooth_max).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    maxb = b.max()
    # stability shift
    shifted = b - maxb
    exp_term = np.exp(gamma * shifted)
    sum_exp = exp_term.sum()
    # softmax weights (sum to 1)
    w = exp_term / sum_exp

    # gradient = 2 * (a_rev * w)[n-1 : n-1+n]
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # smooth approximation of max(conv) = maxb + log(sum_exp)/gamma
    smooth_max = maxb + np.log(sum_exp) / gamma
    return grad, smooth_max

# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _gaussian_distribution(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    w /= w.sum()
    return w


def _cauchy_distribution(n: int, scale_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    scale = scale_factor * n
    w = 1.0 / (1.0 + ((idx - centre) / scale) ** 2)
    w /= w.sum()
    return w


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Unsymmetric Beta(α,β) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide='ignore', invalid='ignore'):
        w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    if np.isnan(w).any() or np.isinf(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex from Dirichlet(α,…,α)."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Deterministic re‑orderings --------------------------------------------
def order_alternate_extremes(desc_vals: np.ndarray) -> np.ndarray:
    """Place largest values alternately left‑most / right‑most."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    left, right, toggle = 0, n - 1, True
    for v in desc_vals:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    return out


def order_center_outwards(desc_vals: np.ndarray) -> np.ndarray:
    """Biggest value at centre, then fill outwards."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    positions = []
    mid = (n - 1) // 2
    positions.append(mid)
    for d in range(1, n):
        left = mid - d
        if left >= 0:
            positions.append(left)
        right = mid + d
        if right < n:
            positions.append(right)
    for i, pos in enumerate(positions[:n]):
        out[pos] = desc_vals[i]
    return out


def order_binary_tree(desc_vals: np.ndarray) -> np.ndarray:
    """Spread large coefficients by recursively filling the midpoint."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)

    def recurse(l: int, r: int, pos_list):
        if l > r:
            return
        m = (l + r) // 2
        pos_list.append(m)
        recurse(l, m - 1, pos_list)
        recurse(m + 1, r, pos_list)

    positions = []
    recurse(0, n - 1, positions)
    out[positions] = desc_vals
    return out


def order_random_shuffle(desc_vals: np.ndarray) -> np.ndarray:
    """Simple random permutation."""
    perm = np.random.permutation(desc_vals.shape[0])
    return desc_vals[perm]

# Adam optimiser --------------------------------------------------------
def _adam_descent(a0: np.ndarray,
                  deadline: float,
                  max_iters: int = 250_000,
                  step0: float = 0.2,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  gamma: float = 0.0) -> tuple[np.ndarray, float]:
    """
    Adam on the simplex.
    * if gamma == 0  → exact sub‑gradient of max.
    * else           → smoothed gradient with parameter γ.
    Returns best vector (on simplex) and its surrogate score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _max_convolution_score(a)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        if gamma == 0.0:
            grad, _, _ = _subgrad_max(a)
        else:
            grad, _ = _smooth_grad(a, gamma)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

            desc = np.sort(a)[::-1]
            add_candidate(order_alternate_extremes(desc))
            add_candidate(order_center_outwards(desc))
            add_candidate(order_binary_tree(desc))

        # ----- unsymmetric beta
        for alpha in beta_params:
            for beta_ in beta_params:
                if time.time() >= deadline:
                    break
                a = _beta_distribution(n, alpha, beta_)
                add_candidate(a)
                desc = np.sort(a)[::-1]
                add_candidate(order_alternate_extremes(desc))

        # ----- power‑transformed arcsine / beta
        base_vec = _arcsine_distribution(n)
        for pwr in [0.3, 0.5, 0.8, 1.2, 1.5, 2.0]:
            if time.time() >= deadline:
                break
            a = np.power(base_vec, pwr)
            a /= a.sum()
            add_candidate(a)

        # ----- random Dirichlet draws
        for a_par in dirichlet_alphas:
            if time.time() >= deadline:
                break
            a = _random_dirichlet_alpha(n, a_par)
            add_candidate(a)

    # ------------------------------------------------------------------
    # 2. Inject previous best if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        prev = np.maximum(prev, 0.0)
        if prev.sum() > 0:
            prev = prev / prev.sum()
            add_candidate(prev)
            print(f"added previous best candidate, score={candidates[-1][1]:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Keep top seeds
    candidates.sort(key=lambda x: x[1])
    top_k = min(20, len(candidates))
    top_candidates = candidates[:top_k]
    print(f"generated {len(candidates)} candidates, keeping top {top_k}")
    print(f"best seed score = {top_candidates[0][1]:.6f}")

    best_seq = top_candidates[0][0].copy()
    best_score = top_candidates[0][1]

    # ------------------------------------------------------------------
    # 4. Intensive refinement (multi‑phase Adam)
    for idx, (seed_seq, _) in enumerate(top_candidates):
        if time.time() >= deadline:
            break

        # Phase A – smooth (γ=20)
        seq, sc = _adam_descent(seed_seq, deadline,
                                max_iters=200_000,
                                step0=0.25,
                                gamma=20.0)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} "
                  f"Adam γ=20 → {best_score:.6f}")

        # Phase B – less smooth (γ=200)
        seq, sc = _adam_descent(best_seq, deadline,
                                max_iters=150_000,
                                step0=0.15,
                                gamma=200.0)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} "
                  f"Adam γ=200 → {best_score:.6f}")

        # Phase C – exact max sub‑gradient
        seq, sc = _adam_descent(best_seq, deadline,
                                max_iters=200_000,
                                step0=0.1,
                                gamma=0.0)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} "
                  f"Adam max → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Frank‑Wolfe style mass transfer on the global best
    a = best_seq.copy()
    cur_score = best_score
    it_fw = 0
    while time.time() < deadline and it_fw < 120_000:
        it_fw += 1
        grad, _, _ = _subgrad_max(a)
        i_src = int(np.argmax(grad))          # where decreasing a helps most
        i_dst = int(np.argmin(grad))          # where increasing a helps most
        if i_src == i_dst:
            break
        delta = min(a[i_src], 1.0 - a[i_dst])
        if delta <= 0:
            break
        step = 0.7 * delta
        a[i_src] -= step
        a[i_dst] += step
        cur_score = _max_convolution_score(a)
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time():.1f}] FW iter {it_fw} new best {best_score:.6f}")

        # occasional jitter to escape plateaus
        if it_fw % 4000 == 0 and np.random.rand() < 0.1:
            a = _project_onto_simplex(a + np.random.randn(a.size) * 1e-5, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    # 6. Final random perturbation + short Adam run (optional)
    while time.time() < deadline - 1.0:
        # Add tiny noise, re‑normalise, run a few Adam steps
        noisy = best_seq + np.random.randn(best_seq.size) * 5e-4
        noisy = _project_onto_simplex(noisy, s=1.0)
        seq, sc = _adam_descent(noisy, deadline,
                                max_iters=40_000,
                                step0=0.08,
                                gamma=0.0)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] noise‑Adam improved best to {best_score:.6f}")
        else:
            break

    # ------------------------------------------------------------------
    print(f"final best surrogate score: {best_score:.6f}  (length={len(best_seq)})")
    # The surrogate equals the true evaluation when the vector is ℓ₁‑normalised.
    return best_seq.tolist()
