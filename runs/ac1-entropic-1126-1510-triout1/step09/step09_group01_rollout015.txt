"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.  The search proceeds in three phases:

1. **Wide exploration** – a rich palette of analytic families
   (uniform, arcsine, triangular, exponential, symmetric / asymmetric
   Beta, power‑law, sine‑power, cosine / Blackman / Kaiser windows,
   Gaussian) is generated for several lengths.  The best few candidates
   of each length are kept.

2. **Smooth‑max mirror descent** – each retained candidate is refined
   by minimising a softened version of the max‑convolution
   ``max(a*a)``.  An exponentiated‑gradient (mirror‑descent) update is
   performed with a geometric schedule for the smoothing parameter
   ``α`` (from 1 up to 1e6) and a decaying learning‑rate.  The iterates
   stay on the probability simplex.

3. **Simulated‑annealing polishing** – starting from the best vector
   found so far, random mass‑transfer moves are proposed.  Moves that
   lower the true objective ``2·n·max(conv)`` are always accepted; moves
   that increase the objective are accepted with a temperature‑dependent
   probability.  The temperature cools geometrically.  This stage can
   escape shallow local minima and typically yields the final best
   value.

All stages respect a CPU‑time budget (default 1000 s).  The returned
sequence is a plain ``list`` of non‑negative floats whose sum is 1,
hence ``evaluate_sequence`` will only apply harmless clipping and
return the objective value.
"""

# Helper utilities

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a):
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full linear convolution of two real vectors using a real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a probability vector (sum = 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n, p, q):
    """Beta(p,q) (asymmetric) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_law(n, exp):
    """a_i ∝ (i+1)^{-exp}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-exp)
    vals /= vals.sum()
    return vals


def _distribution_sine_power(n, power):
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_blackman_window(n):
    """Standard Blackman window."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    a0 = 0.42
    a1 = 0.5
    a2 = 0.08
    vals = (a0
            - a1 * np.cos(2.0 * np.pi * i / (n - 1))
            + a2 * np.cos(4.0 * np.pi * i / (n - 1)))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_kaiser_window(n, beta):
    """Kaiser window with parameter beta (beta≥0)."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha
    # scipy.special.i0 may not be present – use math.i0 (since Python 3.8)
    w = np.array([math.i0(math.pi * beta * math.sqrt(1 - ti * ti)) for ti in t])
    w0 = math.i0(math.pi * beta)
    vals = w / w0
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    """Discrete Gaussian centred at (n‑1)/2 with width sigma·n."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

        # deterministic families
        candidates.append(_distribution_uniform(n))
        candidates.append(_distribution_arcsine(n))
        candidates.append(_distribution_triangular(n))
        candidates.append(_distribution_exponential(n, lam=0.003))
        candidates.append(_distribution_exponential(n, lam=0.02))

        # symmetric Beta
        for p in (0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
            candidates.append(_distribution_beta_sym(n, p))

        # asymmetric Beta (randomly sampled)
        for _ in range(3):
            p = 10 ** random.uniform(-2, 0)          # 0.01 … 1
            q = 10 ** random.uniform(-2, 0)
            candidates.append(_distribution_beta_asym(n, p, q))

        # power‑law
        for exp in (0.3, 0.5, 0.8, 1.0, 1.5):
            candidates.append(_distribution_power_law(n, exp))

        # sine‑power
        for pw in (0.5, 1.0, 2.0, 4.0, 8.0):
            candidates.append(_distribution_sine_power(n, pw))

        # cosine window variants
        for coeff in (0.3, 0.5, 0.7, 0.9):
            candidates.append(_distribution_cosine_window(n, coeff))

        # Blackman
        candidates.append(_distribution_blackman_window(n))

        # Kaiser windows with a few betas
        for beta in (3.0, 5.0, 8.0, 12.0):
            candidates.append(_distribution_kaiser_window(n, beta))

        # Gaussian
        for sigma in (0.12, 0.18, 0.25, 0.35):
            candidates.append(_distribution_gaussian(n, sigma))

        # Random Dirichlet draws
        for alpha in (0.2, 0.5, 1.0, 2.0):
            vec = np.random.dirichlet(np.full(n, alpha))
            candidates.append(vec)

        # Sparse random start (few non‑zeros)
        sparse = np.zeros(n, dtype=np.float64)
        k = max(2, n // 400)          # a few thousand non‑zeros max
        idx = np.random.choice(n, k, replace=False)
        sparse[idx] = np.random.rand(k)
        sparse /= sparse.sum()
        candidates.append(sparse)

        # ---- evaluate candidates and keep the best few for this length ----
        scored = []
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec.astype(np.float64), eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            scored.append((sc, vec))
        # keep the top `keep_per_n` of this length
        scored.sort(key=lambda x: x[0])
        for sc, vec in scored[:keep_per_n]:
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n} initial best {sc:.6f}")

    # ------------------------------------------------------------------
    # 2. If a previous solution exists, try it as well.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # noqa: F821
        if prev.ndim == 1 and prev.size > 1:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score, best_seq, best_n = sc, prev.copy(), len(prev)
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback – should never happen.
    if best_seq is None:
        best_n = 4000
        best_seq = _distribution_uniform(best_n)
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Smooth‑max mirror‑descent refinement for every kept candidate.
    #    We repeat the whole refinement for the current global best and
    #    for a few of the top candidates discovered in step 1.
    #    This batch approach makes better use of the time budget.
    # ------------------------------------------------------------------
    # Assemble a small pool of promising starting points (global best + top 3)
    # to diversify the local search.
    pool = [best_seq.copy()]
    # add a couple of other good vectors from the previous scan (if any)
    # (they are stored in the local variable `scored`; we reuse the last
    #  generated `scored` list which belongs to the final length in the loop)
    try:
        extra = [vec for _, vec in scored[:3]]
        pool.extend(extra)
    except Exception:
        pass

    # progress reporting helper
    def _maybe_print(it, cur_sc):
        if it % 800 == 0 or it == steps - 1:
            print(f"[{time.time():.1f}] mirror it={it} cur best {cur_sc:.6f}")

    for start_vec in pool:
        a = start_vec.copy()
        n = a.shape[0]

        # schedule for the smoothing parameter α and the stepsize lr
        schedule = [
            # (α_start, α_end, steps, lr_start, lr_end)
            (1.0, 1e2, 3000, 0.20, 0.02),
            (1e2, 1e4, 3000, 0.02, 0.002),
            (1e4, 1e6, 4000, 0.002, 0.0002),
        ]

        for (a0, a1, steps, lr0, lr1) in schedule:
            if time.time() >= deadline:
                break
            for it in range(steps):
                if time.time() >= deadline:
                    break
                t = it / steps
                alpha = a0 * (a1 / a0) ** t
                lr = lr0 * (lr1 / lr0) ** t

                conv = _fft_self_convolve(a)
                maxc = conv.max()
                # soft‑max weights
                w = np.exp(alpha * (conv - maxc))
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # gradient of the smooth‑max (see derivation in the description)
                conv_rev = _fft_convolve(a[::-1], w)
                grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

                # exponentiated‑gradient step + simplex projection
                a = a * np.exp(-lr * grad)
                a = _project_onto_simplex(a, 1.0)
                a = np.maximum(a, eps)
                a /= a.sum()

                if (it % 800 == 0) or (it == steps - 1):
                    cur_sc = _max_convolution_score(a)
                    if cur_sc < best_score - 1e-12:
                        best_score, best_seq = cur_sc, a.copy()
                        print(f"[{time.time():.1f}] new best after mirror {cur_sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Simulated annealing / stochastic hill‑climbing polish.
    #    This stage can escape shallow local minima.
    # ------------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]
    cur_score = best_score
    temperature = 0.05 * cur_score          # initial temperature
    final_temperature = 1e-7
    decay_factor = 0.995                   # geometric cooling
    iter_sa = 0
    max_sa_iters = 200_000                  # generous upper bound

    while time.time() < deadline and iter_sa < max_sa_iters:
        iter_sa += 1
        # pick a source with enough mass
        src = np.random.choice(np.where(a > eps)[0])
        # pick a distinct target
        tgt = random.randint(0, n - 1)
        if tgt == src:
            continue

        # propose a transfer (up to 5 % of the source mass)
        max_delta = min(a[src] * 0.5, 0.03)   # keep moves modest
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta
        # ensure positivity
        a_new = np.maximum(a_new, eps)
        a_new /= a_new.sum()

        new_score = _max_convolution_score(a_new)
        diff = new_score - cur_score

        if diff < 0 or random.random() < math.exp(-diff / temperature):
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time():.1f}] SA it={iter_sa} new best {new_score:.6f}")

        # cool down
        if iter_sa % 500 == 0:
            temperature = max(temperature * decay_factor, final_temperature)

        # occasional jitter to avoid stagnation
        if iter_sa % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    # 6. Final polishing with a few sub‑gradient steps (optional but cheap).
    # ------------------------------------------------------------------
    a = best_seq.copy()
    velocity = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    step0 = 0.5
    it = 0
    while time.time() < deadline and it < 5000:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** it)

        step = step0
        while True:
            a_cand = a - step * v_hat
            a_cand = _project_onto_simplex(a_cand, 1.0)
            a_cand = np.maximum(a_cand, eps)
            a_cand /= a_cand.sum()
            cand_sc = _max_convolution_score(a_cand)
            if cand_sc < best_score - 1e-12 or step < 1e-8:
                break
            step *= 0.5

        a = a_cand
        if cand_sc < best_score - 1e-12:
            best_score, best_seq = cand_sc, a.copy()
            print(f"[{time.time():.1f}] sub‑grad it={it} new best {cand_sc:.6f}")

    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
