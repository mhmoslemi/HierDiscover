Propose a non‑negative step function minimizing
    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)²).

Approach
--------
1️⃣  Generate a small pool of analytic seeds (arcsine, symmetric Beta,
    Dirichlet draws, uniform).  All seeds are normalised to sum = 1.

2️⃣  For the most promising seeds run a *linear‑programming refinement*:
    given the current distribution `f` and its current worst‑case
    convolution value `M = max(conv(f,f))`, solve the LP

        maximise   Σ b_j
        subject to A_f·b ≤ M·1,   b ≥ 0

    where `A_f` is the convolution matrix defined by (A_f b)[k] = Σ_i f_i·b_{k‑i}.
    The LP finds a distribution `b` that carries as much total mass as
    possible while never exceeding the current bound `M`.  Normalising
    `b` to sum = 1 and mixing it with `f` (geometric step `t = 0.3`)
    strictly reduces the objective after a few iterations.

3️⃣  Polish the best candidate with a short exponentiated‑gradient run
    (mirror descent w.r.t. KL) and a cheap hill‑climbing that moves tiny
    amounts of mass between two coordinates.

The whole routine respects the supplied time budget (default 1000 s) and
always returns a feasible list of floats.

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a via real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap score = 2·n·max(a*a) assuming Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(conv(a,a)) on the simplex.
    Returns (grad, cur_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with the active‑mask weight vector
    # (full convolution then slice the relevant part)
    rev_a = a[::-1]
    conv_rev = np.fft.irfft(np.fft.rfft(rev_a, n=2 * n - 1) *
                            np.fft.rfft(w, n=2 * n - 1))
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even positions
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 200_000,
                          eta0: float = 0.9) -> tuple[np.ndarray, float]:
    """Mirror descent with KL (exponentiated‑gradient)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / a.size)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray):
    """
    Build the sparse convolution matrix A_f (shape (2n‑1, n))
    such that (A_f @ b) = conv(f, b).
    """
    n = f.size
    # each diagonal k (offset) carries the constant weight f[k]
    diag_data = [np.full(n - k, f[k]) for k in range(n)]
    offsets = np.arange(n)  # 0 .. n‑1
    A = diags(diag_data, offsets, shape=(2 * n - 1, n), format='csr')
    return A


# ----------------------------------------------------------------------
def _lp_max_sum(f: np.ndarray, M: float):
    """
    Solve   maximise Σ b_j   subject to   conv(f, b) ≤ M,   b ≥ 0.
    Returns the optimal b (may be dense).  If LP fails, returns None.
    """
    n = f.size
    A = _build_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)                     # minimise -Σb  ⇒ maximise Σb
    bounds = [(0, None)] * n
    # linprog with HiGHS resists large problems; give it a short internal limit.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success and res.x is not None:
        return res.x
    return None

# Analytic families (all return a normalised probability vector)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_w = (p - 1) * (np.log(x) + np.log(1.0 - x))
    log_w = np.clip(log_w, -700, 700)
    w = np.exp(log_w)
    if w.sum() == 0.0:
        return np.full_like(w, 1.0 / n)
    return w / w.sum()


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0.0:
        return _uniform_distribution(n)
    return x / x.sum()

def propose_candidate(seed: int = 43, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function (non‑negative, sum ≥ 0.01) that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # pick a base length – favour the length of an existing solution,
    # otherwise fall back to a moderate size.
    try:
        base_n = int(len(height_sequence_1))               # type: ignore[name-defined]
    except Exception:
        base_n = 2000

    lengths = {max(500, int(base_n * f)) for f in (0.8, 1.0, 1.2)}
    # add a few random larger lengths (up to 1.8·base_n)
    for _ in range(5):
        lengths.add(np.random.randint(500, min(18000, int(base_n * 1.8)) + 1))
    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # 1. Broad sampling of analytic families
    seed_vectors = []
    for n in lengths:
        # arcsine
        seed_vectors.append(_arcsine_distribution(n))
        # symmetric beta with a few exponents surrounding 0.5
        for p in (0.3, 0.5, 0.7):
            seed_vectors.append(_beta_symmetric_distribution(n, p))
        # uniform
        seed_vectors.append(_uniform_distribution(n))
        # random Dirichlet – very sparse draws and a dense one
        for a in (0.01, 0.05, 0.3, 1.0):
            seed_vectors.append(_random_dirichlet(n, a))

    # ------------------------------------------------------------------
    # include the historic best if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.clip(prev, 0.0, None)
            if prev.sum() == 0.0:
                prev = _uniform_distribution(prev.size)
            else:
                prev = _project_onto_simplex(prev / prev.sum())
            seed_vectors.append(prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Compute cheap scores and keep the most promising seeds
    scored = [(a, _max_convolution_score(a)) for a in seed_vectors]
    scored.sort(key=lambda x: x[1])
    top_k = 12
    best_candidates = [a for a, _ in scored[:top_k]]
    best_score = scored[0][1]
    best_seq = best_candidates[0].copy()
    print(f"[{time.time():.1f}] initial best cheap score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. LP‑based refinement
    t_mix = 0.3                    # mixing weight for the new LP solution
    max_lp_iters = 12

    for idx, cand in enumerate(best_candidates):
        if time.time() >= deadline:
            break
        f = cand.copy()
        f = f / f.sum()
        for lp_it in range(max_lp_iters):
            if time.time() >= deadline:
                break
            # current worst‑case convolution
            M = _fft_self_convolve(f).max()
            b = _lp_max_sum(f, M)
            if b is None:
                break
            # normalise the LP solution to sum = 1
            b = b / b.sum()
            # geometric mixing step
            f_new = (1 - t_mix) * f + t_mix * b
            f_new = _project_onto_simplex(f_new, s=1.0)
            cur_score = _max_convolution_score(f_new)
            if cur_score < best_score - 1e-12:
                best_score, best_seq = cur_score, f_new.copy()
                print(f"[{time.time():.1f}] seed {idx+1}/{len(best_candidates)} "
                      f"LP‑iter {lp_it+1} → {best_score:.6f}")
            f = f_new

    # ------------------------------------------------------------------
    # 3. Light exponential‑gradient polishing (few hundred thousand steps)
    if time.time() < deadline:
        seq, sc = _exp_gradient_descent(best_seq, deadline,
                                        max_iters=150_000, eta0=0.9)
        if sc < best_score - 1e-12:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] exp‑grad polishing → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Simple hill‑climbing with mass transfers
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    it = 0
    max_transfer = 0.02   # up to 2 % of the total mass at start

    while time.time() < deadline:
        it += 1
        # geometrically decaying transfer magnitude
        transfer = max_transfer * (0.5 ** (it / 6000.0))

        # pick donor / receiver indices
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter {it} new best {best_score:.6f}")

        # occasional jitter to escape plateaus
        if it % 5000 == 0 and np.random.rand() < 0.1:
            jitter = np.random.randn(n) * 4e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    print(f"Final cheap score: {best_score:.6f} (length = {len(best_seq)})")
    return best_seq.tolist()
