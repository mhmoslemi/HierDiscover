def smooth_l_bfgs_optimize(a0, deadline, gamma=20.0, max_iter=200):
    n = len(a0)
    # convert to logit vector z (softmax)
    eps = 1e-12
    a0 = np.maximum(a0, eps)
    z0 = np.log(a0)
    # define objective and gradient wrt z
    def fun(z):
        a = np.exp(z)
        a = a / a.sum()
        b = _fft_self_convolve(a)
        b_max = b.max()
        # compute smooth approx of max:
        exp_terms = np.exp(gamma * (b - b_max))
        w_sum = exp_terms.sum()
        approx = b_max + np.log(w_sum) / gamma
        obj = 2 * n * approx
        # gradient w.r.t a (approx_max)
        w = exp_terms / w_sum
        conv_rev = _fft_convolve(a[::-1], w)
        grad_a = 2 * conv_rev[n-1:n-1+n]
        diag_w = w[2*np.arange(n)]
        grad_a -= a * diag_w
        # chain rule for softmax:
        # grad_z = a * (grad_a - dot(grad_a, a))
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)
        return obj, grad_z.astype(np.float64)
    # Use SciPy minimize L-BFGS-B with jac=True.
    res = minimize(fun, z0, method='L-BFGS-B', jac=True, options={'maxiter': max_iter, 'gtol':1e-12, 'disp': False})
    # get a from solution
    z_opt = res.x
    a_opt = np.exp(z_opt)
    a_opt = a_opt / a_opt.sum()
    # compute final cheap score exactly
    final_score = _max_convolution_score(a_opt)
    return a_opt, final_score

#!/usr/bin/env python
"""Optimise step function to minimise evaluate_sequence.

The algorithm performs:
1. Coarse sweep over analytic families and random Dirichlet draws.
2. Long projected sub‑gradient descent on the simplex.
3. A simulated‑annealing phase that accepts occasional uphill moves.
4. Final polishing with a few more sub‑gradient iterations.

All stages respect the global time budget.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
import math

# Helper: projection onto simplex
def _project_onto_simplex(v, s=1.0):
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# FFT based convolutions
def _fft_self_convolve(a):
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_convolve(a, b):
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a):
    """Cheap surrogate = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Exact sub‑gradient of max(a*a)
def _subgrad(a):
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (k = 2*i)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

# Families --------------------------------------------------------------

def _arcsine_distribution(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()

def _smoothed_arcsine_distribution(n, eps):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5 + eps) * (n - i - 0.5 + eps))
    return vals / vals.sum()

def _product_power_distribution(n, alpha):
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-alpha)
    return vals / vals.sum()

def _beta_symmetric_distribution(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use log to avoid overflow for very small p

    eps = 1e-14
    a0 = np.maximum(a0, eps)
    z0 = np.log(a0)

    def fun(z):
        a = np.exp(z)
        a_sum = a.sum()
        a = a / a_sum
        b = _fft_self_convolve(a)
        b_max = b.max()
        exp_term = np.exp(gamma * (b - b_max))
        w_sum = exp_term.sum()
        approx = b_max + np.log(w_sum) / gamma
        obj = 2.0 * n * approx

        w = exp_term / w_sum
        conv_rev = _fft_convolve(a[::-1], w)
        grad_a = 2.0 * conv_rev[n - 1 : n - 1 + n]
        diag_w = w[2 * np.arange(n)]
        grad_a -= a * diag_w

        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)
        return obj, grad_z.astype(np.float64)

    res = minimize(fun, z0, method='L-BFGS-B', jac=True,
                   options={'maxiter': max_iter, 'gtol': 1e-12, 'disp': False})
    if not res.success:
        # fallback to current a0
        a_opt = a0
    else:
        z_opt = res.x
        a_opt = np.exp(z_opt)
        a_opt = a_opt / a_opt.sum()
    final_score = _max_convolution_score(a_opt)
    return a_opt, final_score

# Simplex projection
def _project_onto_simplex(v, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

    Returns (grad, cur_max, active_set) for the function max(a*a).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for terms with i=j (k = 2*i)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

            candidates.append(_smoothed_arcsine_distribution(n, eps))
        for alpha in (0.35, 0.45, 0.55, 0.7, 0.9, 1.2):
            candidates.append(_product_power_distribution(n, alpha))
        for p in (0.15, 0.25, 0.35, 0.45, 0.55):
            candidates.append(_beta_symmetric_distribution(n, p))
        for lam in (0.0008, 0.0015, 0.003, 0.006, 0.01):
            candidates.append(_exponential_distribution(n, lam))
        candidates.append(_triangular_distribution(n))
        candidates.append(_sine_distribution(n))

        # a few random dirichlet draws
        for _ in range(4):
            candidates.append(_random_dirichlet(n))

        # evaluate cheap scores and keep the best
        for a in candidates:
            score = _max_convolution_score(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()

        # also try the previous incumbent if it exists
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
            if prev.ndim == 1 and prev.size > 0:
                prev = prev / prev.sum()
                score = _max_convolution_score(prev)
                if score < best_score:
                    best_score, best_seq = score, prev.copy()
        except Exception:
            pass

    print(f"[{time.time():.1f}] after coarse sweep best cheap={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Heavy refinement – long sub‑gradient descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        # 60 % of the remaining time for a deep sub‑gradient phase
        sub_deadline = time.time() + remaining * 0.60
        seq, sc = _subgradient_descent(best_seq, sub_deadline,
                                       max_iters=800_000, eta0=0.9)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time():.1f}] after long sub‑grad cheap={best_score:.6f}")

        # optional L‑BFGS polishing (uses SciPy, may be skipped automatically)
        try:
            sub_deadline = time.time() + remaining * 0.25
            seq, sc = smooth_l_bfgs_optimize(best_seq, sub_deadline,
                                            gamma=15.0, max_iter=300)
            if sc < best_score:
                best_seq, best_score = seq, sc
                print(f"[{time.time():.1f}] after L‑BFGS cheap={best_score:.6f}")
        except Exception as e:
            print("L‑BFGS failed:", e)

    # ------------------------------------------------------------------
    # 3️⃣ Simulated‑annealing fine search
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        seq, sc = simulated_annealing(best_seq, deadline)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time():.1f}] after SA cheap={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Very short final polishing (sub‑gradient)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        sub_deadline = time.time() + remaining
        seq, sc = _subgradient_descent(best_seq, sub_deadline,
                                       max_iters=200_000, eta0=0.7)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time():.1f}] final sub‑grad cheap={best_score:.6f}")

    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
