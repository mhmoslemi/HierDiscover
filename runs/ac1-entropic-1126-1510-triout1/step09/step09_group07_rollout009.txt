def _lp_optimize_g(f, M, max_time):
    n = len(f)
    conv_len = 2*n-1
    # Build A matrix via sparse toeplitz.
    # Use sp.diags for each shift? Could simply use convolution-of-identity?
    # Approach: compute each row of A (k) corresponds to f reversed? Actually (f*g)[k] = sum_i f[i] * g[k-i]; treat g as variable; so for each k we have coefficients such that g[j] multiplies f[k-j] if 0<=k-j<n.
    # So A is Toeplitz matrix where row k has f[0..] placed.
    # We can implement via sp.lil_matrix and fill.
    rows = []
    cols = []
    data = []
    for k in range(conv_len):
        i_min = max(0, k-(n-1))
        i_max = min(n-1, k)
        # f indices = i where i = k - j; j = k - i.
        # So for each i, coefficient is f[i] for variable g_{k-i}
        # So j = k - i
        # So variable index j = k - i
        # For i in [i_min,i_max], j in [k-i_max, k-i_min]
        # So add entries.
        for i in range(i_min, i_max+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A = sp.csr_matrix((data, (rows, cols)), shape=(conv_len, n))

def _double_peak_distribution(n, width_frac, peak_mass):
    w = max(1, int(round(width_frac * n)))
    # allocate uniform mass to peaks and interior
    interior_len = n - 2*w
    if interior_len < 0:
        w = n // 2
        interior_len = 0
    a = np.zeros(n)
    if interior_len > 0:
        a[w:-w] = (1 - peak_mass) / interior_len
    a[:w] = peak_mass / (2*w)
    a[-w:] = peak_mass / (2*w)
    # ensure sum=1
    a = a / a.sum()
    return a

#!/usr/bin/env python
"""
Propose a non‑negative step function that (approximately) minimises
``evaluate_sequence``.  The approach is a three‑stage pipeline:

1️⃣ **Extended coarse sweep** – many analytic families (arcsine, smoothed
   arcsine, power‑law, symmetric Beta and a simple double‑peak) are
   evaluated on a range of lengths, up to ≈15000 entries.  The cheap
   surrogate 2·n·max(a*a) (exact for a normalised sum) is used to keep
   the best seed.

2️⃣ **Focused refinement** – the current best vector is polished with a
   handful of first‑order methods that respect the simplex:
   exponentiated‑gradient, Adam, a continuation method for a smooth‑max,
   and a heavy‑ball sub‑gradient scheme.  Each routine runs only for the
   time remaining after the sweep.

3️⃣ **Fine‑grained hill‑climbing** – random tiny mass transfers and
   occasional jitter finish the search, ensuring that any last marginal
   improvement is captured before the deadline.

All numbers are non‑negative and normalised to sum = 1 (the objective is
scale‑invariant).  The routine respects a global time budget and returns
a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast convolution helpers (real FFT)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap surrogate = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Exact sub‑gradient of the max‑convolution (used by the first‑order
# optimisers below).
# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.
    Returns (grad, cur_max, active_set).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    # build weight vector w that is 1 on active positions, 0 elsewhere
    w = np.zeros_like(b)
    w[active] = 1.0

    # gradient = 2 * reverse(a) * w  (convolution)
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for indices where k = 2*i
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
# Families of probability vectors (all return a normalised vector)
# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Classic arcsine density on a discrete grid."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _smoothed_arcsine_distribution(n: int, eps: float) -> np.ndarray:
    """Arcsine with an additive smoothing ε."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5 + eps) * (n - i - 0.5 + eps))
    return vals / vals.sum()


def _product_power_distribution(n: int, alpha: float) -> np.ndarray:
    """Shape a_i ∝ ((i+½)(n‑i‑½))^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-alpha)
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑space to avoid underflow for extreme p
    logw = (p - 1) * (np.log(x) + np.log(1.0 - x))
    w = np.exp(logw)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _double_peak_distribution(n: int, width_frac: float, peak_mass: float) -> np.ndarray:
    """
    Two symmetric rectangular peaks at the ends.
    * ``width_frac`` – fraction of the total length occupied by each peak.
    * ``peak_mass`` – total probability mass carried by the two peaks.
    The interior (if any) receives the remaining mass uniformly.
    """
    w = max(1, int(round(width_frac * n)))
    interior = n - 2 * w
    a = np.zeros(n, dtype=np.float64)

    if interior > 0:
        a[w:-w] = (1.0 - peak_mass) / interior
    # distribute the peak mass uniformly inside each edge block
    a[:w] = (peak_mass / 2.0) / w
    a[-w:] = (peak_mass / 2.0) / w
    # normalise (tiny numerical errors may appear)
    a /= a.sum()
    return a


def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Γ–Dirichlet, α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# First‑order optimisation primitives (all respect the simplex)
# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 200_000, eta0: float = 1.0):
    """Mirror‑descent (exponentiated‑gradient) on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        a = a / s if s != 0 else np.full_like(a, 1.0 / a.size)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 200_000, step0: float = 0.3,
                  beta1: float = 0.9, beta2: float = 0.999):
    """Adam‑style adaptive‑gradient optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 300_000, eta0: float = 0.6):
    """Plain sub‑gradient descent with diminishing step size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / math.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _smooth_gradient(a: np.ndarray, gamma: float):
    """Weighted gradient of the soft‑max of a*a (γ controls smoothness)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    b_max = b.max()
    w = np.exp(gamma * (b - b_max))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    # diagonal correction (k = 2*i)
    diag_w = w[2 * np.arange(n)]
    grad -= a * diag_w
    return grad, b_max


def _smooth_gradient_descent(a0: np.ndarray, deadline: float,
                             max_iters: int = 250_000,
                             eta0: float = 0.8,
                             gamma0: float = 2.0,
                             gamma_factor: float = 1.005):
    """Continuation method: start with a smooth max and raise γ."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    gamma = gamma0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _ = _smooth_gradient(a, gamma)
        eta = eta0 / math.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score, best_a = cur_score, a.copy()
        gamma *= gamma_factor
    return best_a, best_score


def _heavy_ball_subgradient_descent(a0: np.ndarray, deadline: float,
                                   max_iters: int = 250_000,
                                   eta0: float = 0.6, momentum: float = 0.6):
    """Sub‑gradient descent with a heavy‑ball momentum term."""
    a_prev = a0.copy()
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / math.sqrt(it)
        a_new = a - eta * grad + momentum * (a - a_prev)
        a_new = _project_onto_simplex(a_new, s=1.0)
        a_prev, a = a, a_new
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine is deterministic (apart from the provided seed) and respects
    the global time budget.  It returns a list of floats ready to be evaluated.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # 0️⃣ Determine a base length (use a previous construction if available)
    # ------------------------------------------------------------------
    try:
        base_n = int(len(height_sequence_1))  # type: ignore[name-defined]
        if base_n < 500:
            base_n = 500
    except Exception:
        base_n = 3500

    # ------------------------------------------------------------------
    # 1️⃣ Extended coarse sweep – many lengths & families
    # ------------------------------------------------------------------
    # We deliberately push well beyond the previous maximal length.
    length_factors = (0.8, 1.0, 1.1, 1.5, 2.0, 2.5, 3.0)
    lengths = sorted({max(500, int(base_n * f)) for f in length_factors})
    # Upper cap – 15000 is comfortably fast for the FFT‑based scoring.
    lengths = [min(L, 15000) for L in lengths]

    best_seq = None
    best_score = float('inf')
    best_config = None   # (factory, args) – useful for possible up‑scaling

    # families to be examined
    for n in lengths:
        if time.time() >= deadline:
            break

        # ---------- arcsine & smoothed arcsine ----------
        for eps in (0.0, 0.1, 0.5, 1.0, 2.0):
            a = _arcsine_distribution(n) if eps == 0.0 else _smoothed_arcsine_distribution(n, eps)
            score = _max_convolution_score(a)
            if score < best_score:
                best_score, best_seq, best_config = score, a.copy(), (_arcsine_distribution if eps == 0.0 else _smoothed_arcsine_distribution, (n, eps))

        # ---------- power‑law family ----------
        for alpha in np.arange(0.30, 0.71, 0.025):
            a = _product_power_distribution(n, alpha)
            score = _max_convolution_score(a)
            if score < best_score:
                best_score, best_seq, best_config = score, a.copy(), (_product_power_distribution, (n, alpha))

        # ---------- symmetric Beta ----------
        for p in np.arange(0.15, 0.90, 0.025):
            a = _beta_symmetric_distribution(n, p)
            score = _max_convolution_score(a)
            if score < best_score:
                best_score, best_seq, best_config = score, a.copy(), (_beta_symmetric_distribution, (n, p))

        # ---------- double‑peak ----------
        for width_frac in (0.02, 0.05, 0.10):
            for peak_mass in (0.5, 0.7, 0.9):
                a = _double_peak_distribution(n, width_frac, peak_mass)
                score = _max_convolution_score(a)
                if score < best_score:
                    best_score, best_seq, best_config = score, a.copy(), (_double_peak_distribution, (n, width_frac, peak_mass))

        # ---------- a few random Dirichlet draws ----------
        for _ in range(2):
            a = _random_dirichlet(n)
            score = _max_convolution_score(a)
            if score < best_score:
                best_score, best_seq, best_config = score, a.copy(), (_random_dirichlet, (n,))

    # incorporate previous best if it improves the cheap score
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            score = _max_convolution_score(prev)
            if score < best_score:
                best_score, best_seq, best_config = score, prev.copy(), (None, None)
    except Exception:
        pass

    print(f"[{time.time():.1f}] after coarse sweep best cheap={best_score:.6f} length={len(best_seq)}")

    # ------------------------------------------------------------------
    # 2️⃣ Focused refinement – we stay on the current length.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.5:
        # allocate fractions of the remaining budget to each optimiser
        # (they themselves respect their individual sub‑deadline)
        # 1) exponentiated‑gradient
        sub_deadline = time.time() + remaining * 0.20
        seq, sc = _exp_gradient_descent(best_seq, sub_deadline,
                                        max_iters=400_000, eta0=1.0)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time():.1f}] after exp‑grad   {best_score:.6f}")

        # 2) Adam
        sub_deadline = time.time() + remaining * 0.20
        seq, sc = _adam_descent(best_seq, sub_deadline,
                                max_iters=400_000, step0=0.3)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time():.1f}] after Adam      {best_score:.6f}")

        # 3) smoothed‑max continuation
        sub_deadline = time.time() + remaining * 0.30
        seq, sc = _smooth_gradient_descent(best_seq, sub_deadline,
                                            max_iters=400_000,
                                            eta0=0.8,
                                            gamma0=2.0,
                                            gamma_factor=1.005)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time():.1f}] after smooth   {best_score:.6f}")

        # 4) heavy‑ball sub‑gradient
        sub_deadline = time.time() + remaining * 0.15
        seq, sc = _heavy_ball_subgradient_descent(best_seq, sub_deadline,
                                                  max_iters=400_000,
                                                  eta0=0.6,
                                                  momentum=0.6)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time():.1f}] after heavy‑ball {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Fine‑grained hill‑climbing (tiny random mass transfers)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    it = 0
    max_transfer = 0.05  # initial max fraction of total mass that may move

    while time.time() < deadline:
        it += 1
        # Geometric decay of the transferred amount
        transfer = max_transfer * (0.5 ** (it / 10000.0))

        i = np.random.randint(n)
        # pick a different index j
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_seq, best_score = a_new.copy(), new_score
                print(f"[{time.time():.1f}] hill‑climb iter={it} new best {best_score:.6f}")

        # occasional jitter to escape tiny plateaus
        if it % 5000 == 0 and np.random.rand() < 0.15:
            jitter = np.random.randn(n) * 5e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
