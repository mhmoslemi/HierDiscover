"""
Propose a step‑function (non‑negative sequence) that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

The algorithm works in three stages:

1️⃣  A deterministic analytic seed (arcsine shape) is taken on a modest
    length (≈2000).  The seed already yields a cheap score ≈1.5.

2️⃣  **Linear‑programming improvement** – a classic “LP‑boost” step:
    given the current f, compute its worst convolution value M.
    Solve the LP

        maximise   Σ b_j
        subject to  conv(f,b) ≤ M ,   b ≥ 0

    The optimum b is typically much larger than f in ℓ₁‑norm; after
    normalising b we obtain a new distribution g that strictly reduces
    the worst convolution value.  A convex combination
        f ← (1‑τ)·f + τ·g
    (τ slowly decays) iteratively pushes the objective down.
    The LP has only O(n) variables and O(n) constraints; for n≈2000 the
    dense constraint matrix fits easily into memory and ECOS solves it
    in a few hundred milliseconds.

3️⃣  **Fine polishing** – a short projected sub‑gradient descent (the
    same routine used in the previous state‑of‑the‑art) finishes the
    optimisation.

The whole routine respects the supplied time budget (default 1000 s)
and always returns a feasible list of floats.
"""

import time
import numpy as np
import cvxpy as cp
import scipy.linalg as la

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap objective = 2·n·max(a*a) under the convention Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Linear operator matrix B ↦ conv(f,B) (full linear convolution).
    Shape = (2·n‑1, n).  Uses a Toeplitz construction.
    """
    n = f.shape[0]
    m = 2 * n - 1
    col = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    row = np.concatenate([f[:1], np.zeros(n - 1, dtype=f.dtype)])
    T = la.toeplitz(col, row)[:m, :]
    return T


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
def _downsample(seq: np.ndarray, n: int) -> np.ndarray:
    """Resample by simple nearest‑neighbour indexing and renormalise."""
    if seq.size == n:
        return seq.copy()
    idx = np.linspace(0, seq.size - 1, n).astype(int)
    out = seq[idx]
    out_sum = out.sum()
    if out_sum == 0.0:
        out = np.full_like(out, 1.0 / n)
    else:
        out = out / out_sum
    return out


# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 500_000,
                         eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Plain projected sub‑gradient descent on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # compute sub‑gradient of max(conv) (exact)
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - eps)

        # weight mask
        w = np.zeros_like(conv)
        w[active] = 1.0

        # gradient = 2 * (a_rev * w) restricted to original length
        grad = 2.0 * np.fft.irfft(
            np.fft.rfft(a[::-1]) * np.fft.rfft(w, n=1 << ((2 * a.size - 1) - 1).bit_length()),
            n=1 << ((2 * a.size - 1) - 1).bit_length()
        )[:a.size]

        # diagonal correction for even positions (pairs (i,i))
        even = (active % 2 == 0)
        diag_idx = active[even] // 2
        if diag_idx.size:
            grad[diag_idx] -= a[diag_idx]

        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score
