The task is to minimise  

\[
F(a)=\frac{2\,n\;\max\limits_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\]

where `a` is a non‑negative vector (the “step heights”).  Because the
objective is homogeneous, we may always normalise `∑a_i = 1`.  After the
normalisation the problem becomes

\[
\min_{a\ge0,\; \sum a_i = 1}\; 2\,n\;\max_k (a*a)_k .
\]

The function is **non‑convex** (the convolution terms are quadratic) but
the pointwise‐maximum is a convex, piece‑wise linear functional of the
quadratic forms.  A standard trick for such problems is to replace the
hard maximum by a smooth **log‑sum‑exp** surrogate

\[
M_\beta(a)=\frac{1}{\beta}\log\!\Bigl(\sum_k e^{\beta\,(a*a)_k}\Bigr),
\qquad\beta>0 .
\]

When \(\beta\) is small the surrogate is very smooth, when \(\beta\) is
large it approaches the true maximum.  The gradient of the surrogate
has a closed form:

* let \(c_k=(a*a)_k\) and \(w_k=\operatorname{softmax}_\beta(c)_k\);
* the gradient is  

\[
\nabla_a M_\beta = 2\,\bigl(w\star a\bigr),
\]

where \(\star\) is the **cross‑correlation** (equivalently a convolution
with the reversed weight vector).  All convolutions are performed with a
real FFT, i.e. in \(\mathcal O(n\log n)\) time.

The optimiser uses **mirror (exponentiated) descent** on the simplex:
\[
a_i \leftarrow a_i\,\exp\bigl(-\eta_t\,(\nabla_a M_\beta)_i\bigr),
\qquad
a\gets\Pi_{\Delta}(a)
\]
with a step‑size \(\eta_t= \eta_0/\sqrt{t}\) and a slowly increasing
temperature \(\beta_t\) (annealing).  The algorithm is run from a large
set of analytic “seed” distributions (uniform, arcsine, Beta,
power‑law, exponential, two‑block spikes, random sparse) and from an
interpolation of the previously best known vector.  After the smooth‑
max phase a cheap **pairwise mass‑transfer** local search is applied to
polish the result.

The whole pipeline respects a global time budget (default **1000 s**)
and always returns the best vector found so far.

```python
"""
Propose a non‑negative step sequence that minimises `evaluate_sequence`.

Algorithm
---------

*   Build a diverse pool of analytic seeds for several lengths.
*   For each seed run a *smooth‑max* mirror‑descent optimiser:
    -   replace `max(conv)` by a log‑sum‑exp surrogate,
    -   compute its gradient via a cross‑correlation (FFT‑based),
    -   perform exponentiated‑gradient updates on the probability simplex.
    -   increase the smoothing parameter β during the run (annealing).
*   Keep the best vector (exact objective measured regularly).
*   Finish with a greedy pairwise‑mass transfer walk that accepts only
    improving moves.
*   Return the best sequence found before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full (a*a) via real FFT. Returns length 2·n‑1 array."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact 2·n·max(conv) for a with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic seed families ------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Broad pool of seeds for a given length."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    for p in (0.05, 0.1, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0):
        cand.append(_distribution_beta_sym(n, p))

    for _ in range(5):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))

    for lam in (0.001, 0.005, 0.01, 0.02, 0.05):
        cand.append(_distribution_exponential_increasing(n, lam))

    for block_frac in (0.05, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac))

    cand.append(_distribution_random_sparse(n, rng, density=0.01))
    return cand


# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max of β·conv."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max objective:
        ∇ = 2 * (w ⋆ a)   (cross‑correlation)
    Returns a vector of length n.
    """
    n = a.shape[0]
    rev_w = w[::-1]                     # reverse for cross‑correlation
    size = 1 << ((n + len(rev_w) - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_w, n=size)
    conv_full = np.fft.irfft(A * B, n=size)[: n + len(rev_w) - 1]
    # slice indices n‑1 … 2n‑2 (inclusive) and reverse
    segment = conv_full[n - 1: 2 * n - 1][::-1]
    return 2.0 * segment


# ----------------------------------------------------------------------
def _smooth_max_optimize(a0: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float,
                         eta0: float = 0.5, beta0: float = 0.5,
                         beta_max: float = 50.0, max_iter: int = 200_000,
                         eval_interval: int = 200) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the log‑sum‑exp surrogate of max(conv).
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    beta = beta0

    while t < max_iter and time.time() < deadline:
        t += 1

        # exact convolution and true objective (periodically for logging)
        conv = _fft_self_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score - 1e-13:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[{time.time():.1f}] iter {t:5d} (β={beta:.2f}) new best {best_score:.6f}")

        # smooth max weights
        w = _softmax_weights(conv, beta)

        # gradient of the smooth surrogate
        grad = _gradient_from_weights(a, w)

        # step‑size schedule (mirror descent)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # project onto probability simplex
        a = _project_onto_simplex(a, s=1.0)

        # anneal β slowly
        if t % eval_interval == 0 and beta < beta_max:
            beta = min(beta * 1.2, beta_max)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Tiny Monte‑Carlo walk: move a random fraction of mass from i→j."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.2e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def _resample_sequence(seq: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a distribution to a new length n."""
    m = len(seq)
    if m == n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    new /= new.sum()
    return new

                eval_interval=400,
            )
            print(f"[{time.time():.1f}] after smooth‑max pass best={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4b – final local greedy polishing
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] after swaps best={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
```
