# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate (Σa = 1) → 2·n·max(conv(a,a))."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) under Σa = 1.
    Returns (gradient, current max value, active indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (terms i=j that appear twice in the max)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.18,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 1_500_000):
    """Adam on the simplex (projected). Returns the best point seen."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          eta0: float = 0.6,
                          max_iters: int = 900_000):
    """Exponentiated‑gradient (KL‑mirror) descent on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


def _sgd_descent(a0: np.ndarray, deadline: float,
                 step0: float = 0.12,
                 max_iters: int = 800_000):
    """Plain projected sub‑gradient descent (step ~ 1/√t)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = step0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score

def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Dirichlet draw with concentration alpha."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


def _random_fft_distribution(n: int) -> np.ndarray:
    """Generate a real‑valued sequence by randomising its Fourier spectrum."""
    # Random complex spectrum with Hermitian symmetry
    rnd = np.random.normal(size=n) + 1j * np.random.normal(size=n)
    rnd[0] = np.random.rand()                     # real DC term
    if n % 2 == 0:
        rnd[n // 2] = np.random.rand()            # Nyquist term (real)
    # Enforce Hermitian symmetry
    rnd[1:n // 2] = np.conj(rnd[-1:-n // 2:-1])
    seq = np.fft.ifft(rnd).real
    seq = np.maximum(seq, 0.0)
    if seq.sum() == 0:
        seq = np.full_like(seq, 1.0 / n)
    else:
        seq = seq / seq.sum()
    return seq

def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _double_bump_distribution(n: int,
                              block_frac: float = 0.4,
                              gap_frac: float = 0.05) -> np.ndarray:
    """Two equal uniform blocks, total occupied fraction = block_frac."""
    total_block = max(2, int(block_frac * n))
    m = total_block // 2
    g = int(gap_frac * n)
    a = np.zeros(n, dtype=np.float64)
    a[:m] = 1.0 / (2 * m)
    start2 = m + g
    if start2 + m > n:
        start2 = n - m
    a[start2:start2 + m] = 1.0 / (2 * m)
    return a


def _k_block_distribution(n: int, k: int) -> np.ndarray:
    """
    k equally‑spaced uniform blocks, each carries mass 1/k.
    Minimal gap of one zero between blocks is enforced.
    """
    if k < 2:
        raise ValueError("k must be ≥2")
    # Ensure we have enough room for the blocks + (k‑1) gaps
    max_block_len = (n - (k - 1)) // k
    if max_block_len <= 0:
        # fallback to uniform distribution
        return np.full(n, 1.0 / n)
    m = max_block_len
    a = np.zeros(n, dtype=np.float64)
    for i in range(k):
        start = i * (m + 1)
        a[start:start + m] = 1.0 / (k * m)
    # Normalise (tiny mass loss due to truncation is compensated)
    a /= a.sum()
    return a
