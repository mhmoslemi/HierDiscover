def evaluate_sequence(sequence):
    # security checks etc.
    # Clip to nonnegative, clip values above 1000.

def _smooth_objective_and_gradient(a, tau):
    # assume a sum=1
    n = a.shape[0]
    b = _fft_self_convolve(a)  # length 2n-1
    # compute softmax weights
    max_b = b.max()
    # to improve numerics,
    exp_vals = np.exp(tau * (b - max_b))  # subtract max to avoid overflow
    sum_exp = exp_vals.sum()
    w = exp_vals / sum_exp
    # compute smooth max value:
    smooth_max = (np.log(sum_exp) / tau) + max_b
    # compute conv_rev = convolution of a_rev and w
    conv_rev = _fft_convolve(a[::-1], w)
    grad_b = 2.0 * conv_rev[n-1:n-1+n]
    # gradient of objective 2*n*max(b):
    grad = 2.0 * n * grad_b
    obj = 2.0 * n * smooth_max
    return obj, grad
```

"""
Propose a low‑score non‑negative step function.

Algorithm
---------
1. Pick a modest set of lengths (≈ 5 k–12 k).  
2. Generate a rich pool of seeds:
   – analytic families (beta, arcsine, exponential, power‑law, double‑bump …)
   – extreme beta parameters (p down to 0.01) and mixture of a uniform with each family.
   – random Dirichlet draws with many concentration values.
3. Keep the best k (k≈30) seeds according to the fast surrogate
   ``_max_convolution_score`` (exact value when Σa=1).
4. Refine each seed with a *smooth* Adam optimiser:
   • replace the non‑smooth max by a log‑sum‑exp (temperature τ) that is
     gradually increased (continuation) – this gives a well‑behaved gradient.
   • use the correct gradient of the smoothed max (no ad‑hoc diagonal correction).
5. After the smooth‑Adam phase run a few rounds of the classic sub‑gradient Adam
   (the code from the previous solution) to clean up residual kinks.
6. Finish with a cheap Frank‑Wolfe/ mass‑transfer polishing and a short simulated‑annealing jitter.
7. Return the best sequence found before the 1000 s budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors via a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact objective when Σa = 1 → 2·n·max(conv(a,a))."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _smooth_max_objective_and_grad(a: np.ndarray, tau: float) -> tuple[float, np.ndarray]:
    """
    Smooth approximation of the max‑convolution objective and its gradient.
    Uses ``smooth_max(b) = (log Σ exp(tau·b))/tau`` with b = a*a.
    Returns (objective, gradient) where the gradient is ∂/∂a
    of 2·n·smooth_max(b).  ``a`` is assumed to satisfy Σa = 1.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)                     # length 2n‑1
    # stable log‑sum‑exp
    b_max = b.max()
    exp_shift = np.exp(tau * (b - b_max))
    sum_exp = exp_shift.sum()
    w = exp_shift / sum_exp                       # soft‑max weights, Σw = 1
    smooth_max = b_max + np.log(sum_exp) / tau

    # gradient of smooth max = Σ_k w_k ∂b_k/∂a = 2 * conv_rev (no diagonal correction)
    conv_rev = _fft_convolve(a[::-1], w)          # length n+2n‑2‑1
    grad_b = 2.0 * conv_rev[n - 1 : n - 1 + n]    # length n
    grad = 2.0 * n * grad_b                       # chain rule for 2·n·smooth_max
    obj = 2.0 * n * smooth_max
    return float(obj), grad


# ----------------------------------------------------------------------
def _adam_smooth(a0: np.ndarray,
                 deadline: float,
                 tau_start: float = 0.5,
                 tau_end: float = 30.0,
                 step0: float = 0.15,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 max_iters: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the *smooth* objective.  The temperature τ is raised
    linearly from ``tau_start`` to ``tau_end`` (continuation) while Adam
    runs.  Returns the best point (according to the *exact* objective).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        # schedule τ (linear in iteration count)
        tau = tau_start + (tau_end - tau_start) * (t / max_iters)

        obj, grad = _smooth_max_objective_and_grad(a, tau)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        # monitor the true (non‑smooth) objective
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score

def _adam_subgrad(a0: np.ndarray,
                  deadline: float,
                  step0: float = 0.20,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """
    The original Adam implementation that works with the *non‑smooth* sub‑gradient.
    Kept as a final polishing stage.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        # sub‑gradient of max(conv) (the version used previously)
        n = a.shape[0]
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps)

        w = np.zeros_like(b)
        w[active] = 1.0
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        even_mask = (active % 2 == 0)
        diag_idx = (active[even_mask] // 2)
        if diag_idx.size > 0:
            grad[diag_idx] -= a[diag_idx]

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score

def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with concentration ``alpha``."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Analytic families (all return a vector that already sums to 1)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _double_bump_distribution(n: int,
                              block_frac: float = 0.25,
                              gap_frac: float = 0.1) -> np.ndarray:
    total_block = max(2, int(block_frac * n))
    m = total_block // 2
    g = int(gap_frac * n)
    a = np.zeros(n, dtype=np.float64)
    a[:m] = 1.0 / (2 * m)
    start2 = m + g
    if start2 + m > n:
        start2 = n - m
    a[start2:start2 + m] = 1.0 / (2 * m)
    return a


def _triple_bump_distribution(n: int,
                              block_frac: float = 0.2,
                              gap_frac: float = 0.05) -> np.ndarray:
    """Three equally‑sized uniform blocks separated by equal gaps."""
    total_block = max(3, int(block_frac * n))
    m = total_block // 3
    g = int(gap_frac * n)
    a = np.zeros(n, dtype=np.float64)
    a[:m] = 1.0 / (3 * m)
    start2 = m + g
    a[start2:start2 + m] = 1.0 / (3 * m)
    start3 = start2 + m + g
    if start3 + m > n:
        start3 = n - m
    a[start3:start3 + m] = 1.0 / (3 * m)
    return a


def _mix_with_uniform(base: np.ndarray, alpha: float) -> np.ndarray:
    """Mixture (1‑α)·uniform + α·base, returned already normalised."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n)
    mix = (1.0 - alpha) * uniform + alpha * base
    mix /= mix.sum()
    return mix

            a = _power_law_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))

        # multi‑bump families ----------------------------------------------------
        for block_frac in (0.2, 0.25, 0.3):
            for gap_frac in (0.05, 0.1, 0.2):
                a = _double_bump_distribution(n, block_frac, gap_frac)
                candidates.append((a, _max_convolution_score(a)))
                a = _triple_bump_distribution(n, block_frac, gap_frac)
                candidates.append((a, _max_convolution_score(a)))

        # random Dirichlet draws -------------------------------------------------
        for alpha in dirichlet_alphas:
            a = _random_dirichlet_alpha(n, alpha)
            candidates.append((a, _max_convolution_score(a)))

    # ------------------------------------------------------------------
    # Bring in the previously known best (if any)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            if prev.sum() == 0:
                prev = np.full_like(prev, 1.0 / prev.size)
            else:
                prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
            print(f"[{time.time():.1f}] added previous best, score={candidates[-1][1]:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Keep the top‑k seeds for intensive refinement
    candidates.sort(key=lambda x: x[1])
    top_k = min(30, len(candidates))
    top_seeds = candidates[:top_k]
    print(f"Generated {len(candidates)} candidates, keeping top {top_k} for refinement.")

    best_seq = top_seeds[0][0].copy()
    best_score = top_seeds[0][1]
    print(f"Initial best surrogate score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Intensive refinement of each seed
    for idx, (seed_seq, _) in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        n = seed_seq.shape[0]

        # 1) Smooth‑Adam with continuation on τ
        seq_smooth, sc_smooth = _adam_smooth(seed_seq, deadline,
                                             tau_start=0.5, tau_end=30.0,
                                             step0=0.15,
                                             max_iters=600_000)
        if sc_smooth < best_score - 1e-14:
            best_score, best_seq = sc_smooth, seq_smooth.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} smooth‑Adam → {best_score:.6f}")

        # 2) Classic sub‑gradient Adam (fine‑tuning)
        seq_adam, sc_adam = _adam_subgrad(seed_seq, deadline,
                                          step0=0.20,
                                          max_iters=400_000)
        if sc_adam < best_score - 1e-14:
            best_score, best_seq = sc_adam, seq_adam.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} sub‑grad Adam → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Frank‑Wolfe style mass‑transfer polishing (cheaper than before)
    a = best_seq.copy()
    cur_score = best_score
    fw_iter = 0
    while time.time() < deadline and fw_iter < 80_000:
        fw_iter += 1
        # exact sub‑gradient of the *max* (no diagonal correction – correct)
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps)

        w = np.zeros_like(b)
        w[active] = 1.0
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]

        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-15:
            break
        # try a few transfer fractions
        improved = False
        for frac in (1.0, 0.5, 0.25):
            delta = max_transfer * frac
            a_new = a.copy()
            a_new[i_src] -= delta
            a_new[i_dst] += delta
            sc = _max_convolution_score(a_new)
            if sc < cur_score - 1e-13:
                a = a_new
                cur_score = sc
                improved = True
                break
        if improved and cur_score < best_score - 1e-14:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time():.1f}] FW iter={fw_iter} new best {best_score:.6f}")

        # occasional jitter to get unstuck
        if fw_iter % 5000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(a.size) * 2e-5
            a = _project_onto_simplex(a + jitter, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    # Short simulated‑annealing jitter phase (helps escape tiny plateaus)
    a = best_seq.copy()
    cur_score = best_score
    temp_start = 0.02
    temp_end = 0.001
    anneal_iter = 0
    while time.time() < deadline and anneal_iter < 120_000:
        anneal_iter += 1
        t_rem = deadline - time.time()
        fraction = max(0.0, t_rem / (budget_s - 0.5))
        temp = temp_end + (temp_start - temp_end) * fraction

        i = np.random.randint(a.size)
        j = np.random.randint(a.size - 1)
        if j >= i:
            j += 1
        max_delta = min(a[i], 1.0 - a[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * temp
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if cur_score < best_score - 1e-14:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time():.1f}] Anneal iter={anneal_iter} new best {best_score:.6f}")
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_score - cur_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                a = a_new
                cur_score = new_score

    # ------------------------------------------------------------------
    print(f"Final best surrogate score: {best_score:.6f} (length={len(best_seq)})")
    # Return a plain Python list – ``evaluate_sequence`` will take care of clipping.
    return best_seq.tolist()
