"""
Propose a step‑function that minimizes `evaluate_sequence`.

The algorithm expands the previous “broad‑sampling → intensive‑refinement”
pipeline with a much larger parametric family search (including very
U‑shaped Beta and exponential families), deeper local optimisation
(Adam, exponentiated‑gradient, SGD) and a final stochastic “mass‑transfer”
local search.  The code respects a 1000 s budget, prints progress
information and returns the best vector found (normalized to ℓ₁ = 1,
which is scale‑invariant for the evaluation function).
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Exact surrogate for `evaluate_sequence` when Σa = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of the max‑convolution functional on the simplex.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives raw gradient
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for terms a_i * a_i appearing only once in the max
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

# Analytic families (all return a normalized (ℓ₁=1) vector)

def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _gaussian_distribution(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    w /= w.sum()
    return w


def _cauchy_distribution(n: int, scale_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    scale = scale_factor * n
    w = 1.0 / (1.0 + ((idx - centre) / scale) ** 2)
    w /= w.sum()
    return w


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide='ignore', invalid='ignore'):
        w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if np.isnan(w).any() or np.isinf(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Unsymmetric Beta(α,β) on a discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide='ignore', invalid='ignore'):
        w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    if np.isnan(w).any() or np.isinf(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Sample a point on the simplex from Dirichlet(α,…,α)."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    x /= x.sum()
    return x

def order_alternate_extremes(desc_vals: np.ndarray) -> np.ndarray:
    """Spread the largest and the second‑largest values to opposite ends."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    left = 0
    right = n - 1
    toggle = True
    for v in desc_vals:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    return out


def order_center_outwards(desc_vals: np.ndarray) -> np.ndarray:
    """Place the biggest value in the centre and then fill alternatingly left/right."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    mid = (n - 1) // 2
    pos = [mid]
    for d in range(1, n):
        left = mid - d
        right = mid + d
        if left >= 0:
            pos.append(left)
        if right < n:
            pos.append(right)
    for i, p in enumerate(pos[:n]):
        out[p] = desc_vals[i]
    return out


def order_binary_tree(desc_vals: np.ndarray) -> np.ndarray:
    """Place values recursively at the midpoint of remaining intervals."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    positions = []

    def recurse(l: int, r: int):
        if l > r:
            return
        m = (l + r) // 2
        positions.append(m)
        recurse(l, m - 1)
        recurse(m + 1, r)

    recurse(0, n - 1)
    out[positions] = desc_vals
    return out

def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 300_000,
                  step0: float = 0.2,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 200_000,
                          eta0: float = 0.8) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _sgd_subgrad(a0: np.ndarray, deadline: float,
                 max_iters: int = 300_000,
                 step0: float = 0.5) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        step = step0 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def _mass_transfer_search(a0: np.ndarray, deadline: float,
                         max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Randomly transfer a fraction of probability mass from a donor index
    to a receiver index.  Accept only improving moves (simple hill‑climb).
    """
    a = a0.copy()
    best = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    n = a.size
    rng = np.random.default_rng()
    while time.time() < deadline and it < max_iters:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 1e-12:
            continue
        # move a random fraction (up to 50 %) of donor's mass
        delta = a[i] * rng.random() * 0.5
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # a_new stays on the simplex automatically
        sc = _max_convolution_score(a_new)
        if sc < best_score:
            best_score = sc
            best = a_new.copy()
            a = a_new  # continue searching from the improved point
    return best, best_score

    print(f"Best seed surrogate = {top_seeds[0][1]:.6f}")

    # Global best tracker
    best_seq = top_seeds[0][0].copy()
    best_score = top_seeds[0][1]

    # ------------------------------------------------------------------
    # 4. Intensive refinement of each top seed (Adam → ExpGrad → SGD)
    # ------------------------------------------------------------------
    for idx, (seed_vec, seed_score) in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        # Adam
        seq, sc = _adam_descent(seed_vec, deadline, max_iters=500_000, step0=0.2)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] Seed {idx+1}/{top_k} Adam improved to {best_score:.6f}")
        # Exponentiated‑gradient
        seq, sc = _exp_gradient_descent(seed_vec, deadline, max_iters=300_000, eta0=0.9)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] Seed {idx+1}/{top_k} ExpGrad improved to {best_score:.6f}")
        # Simple SGD
        seq, sc = _sgd_subgrad(seed_vec, deadline, max_iters=400_000, step0=0.5)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] Seed {idx+1}/{top_k} SGD improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Frank‑Wolfe style pairwise mass transfer on the current global best.
    # ------------------------------------------------------------------
    a = best_seq.copy()
    it_fw = 0
    while time.time() < deadline and it_fw < 120_000:
        it_fw += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        delta = min(a[i_src], 0.7 * (1.0 - a[i_dst]))   # move up to 70 % of admissible mass
        if delta <= 0:
            break
        a[i_src] -= delta
        a[i_dst] += delta
        cur_score = _max_convolution_score(a)
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time():.1f}] Frank‑Wolfe iter {it_fw} new best {best_score:.6f}")
        # occasional jitter to escape plateaus
        if it_fw % 4000 == 0 and np.random.rand() < 0.12:
            a = _project_onto_simplex(a + np.random.randn(a.size) * 1e-5, s=1.0)

    # ------------------------------------------------------------------
    # 6. Stochastic mass‑transfer local search (hill‑climbing)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    a, sc = _mass_transfer_search(a, deadline, max_iters=250_000)
    if sc < best_score:
        best_score, best_seq = sc, a.copy()
        print(f"[{time.time():.1f}] Mass‑transfer local search improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final report
    # ------------------------------------------------------------------
    print(f"Final best surrogate = {best_score:.6f} (length={len(best_seq)})")
    # Return as plain Python list (any scaling works – we keep ℓ₁ = 1)
    return best_seq.tolist()
