def _adam_smooth_descent(a0, deadline, eps_schedule, max_iters=200_000):
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t=0
    eps_idx = 0
    eps = eps_schedule[eps_idx]
    # set maybe inner loop to iterate until next eps threshold
    while time.time() < deadline and t < max_iters:
        t += 1
        # compute gradient of smooth loss (with current eps)
        b = _fft_self_convolve(a)
        # compute stable softmax
        maxb = b.max()
        w = np.exp((b - maxb) / eps)
        sum_w = w.sum()
        w /= sum_w
        # gradient
        grad = 2.0 * _fft_convolve(a[::-1], w)[a.size-1:a.size-1+a.size]

        # Adam update
        m = 0.9*m + 0.1*grad
        v = 0.999*v + 0.001*grad**2
        m_hat = m/(1-0.9**t)
        v_hat = v/(1-0.999**t)
        eta = 0.2/(np.sqrt(v_hat)+1e-8)
        a = a - eta*m_hat
        a = _project_onto_simplex(a)

        # Evaluate exact objective occasionally (maybe every 200 steps)
        if t % 200 == 0:
            cur_score = _max_convolution_score(a)
            if cur_score < best_score:

        # E.g., after every 4000 steps reduce epsilon if more remain
        if t % 4000 == 0 and eps_idx < len(eps_schedule)-1:
            eps_idx += 1
            eps = eps_schedule[eps_idx]

    return best_a, best_score

"""
Propose a non‑negative step‑function that minimises the evaluation score.

Algorithm overview
------------------
1. **Seed generation** – many analytic families (arcsine, beta,
   exponential, Gaussian, Cauchy, triangular) are instantiated for a
   single length (taken from the previous best if available, otherwise
   5000).  For each family three deterministic reorderings (alternating
   extremes, centre‑outwards, binary‑tree) are applied and the vectors are
   ℓ₁‑normalised.  Random Dirichlet draws are also added.  The *surrogate*
   objective `2·n·max(conv(a,a))` (exact for a normalised vector) is used
   to keep the best ~12 seeds.

2. **Smoothed gradient descent** – the non‑smooth `max` is replaced by the
   smooth surrogate  
   `L_ε(a) = ε·log∑_k exp((a∗a)_k/ε)` .  Its gradient is
   `∇L = 2·conv_rev(a, softmax((a∗a)/ε))`.  For each seed we run a short
   Adam, exponentiated‑gradient and vanilla SGD optimisation on a
   decreasing schedule of ε (0.05 → 0.001).  All updates are projected
   onto the probability simplex.

3. **Frank‑Wolfe pairwise mass transfer** – on the current global best we
   repeatedly move mass from the index with the largest gradient to the
   one with the smallest, which is a descent direction for the original
   `max`‑objective.

4. **Local random refinement** – small Gaussian perturbations are added
   to the best vector, projected back onto the simplex and accepted if
   they improve the exact score.

The whole pipeline respects the 1000 s budget and returns the best
vector found.  No file I/O or lambdas are used; all helpers are defined
at top level.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolutions -------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a∗a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
# Exact surrogate (max‑convolution) --------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate for `evaluate_sequence` when Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the *max* objective (used for Frank‑Wolfe) -------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of the max‑convolution functional on the simplex.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (even active indices correspond to a_i² terms)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
# Analytic families ------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _gaussian_distribution(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    w /= w.sum()
    return w


def _cauchy_distribution(n: int, scale_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    scale = scale_factor * n
    w = 1.0 / (1.0 + ((idx - centre) / scale) ** 2)
    w /= w.sum()
    return w


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide='ignore', invalid='ignore'):
        w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    if np.isnan(w).any() or np.isinf(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex from Dirichlet(α,…,α)."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Deterministic reorderings ---------------------------------------------
def order_alternate_extremes(desc_vals: np.ndarray) -> np.ndarray:
    """Spread the largest values alternately to the left‑most and right‑most slots."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    left, right, toggle = 0, n - 1, True
    for v in desc_vals:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    return out


def order_center_outwards(desc_vals: np.ndarray) -> np.ndarray:
    """Place the biggest value in the centre and then fill alternatingly outwards."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    positions = []
    mid = (n - 1) // 2
    positions.append(mid)
    for d in range(1, n):
        left = mid - d
        if left >= 0:
            positions.append(left)
        right = mid + d
        if right < n:
            positions.append(right)
    for i, pos in enumerate(positions[:n]):
        out[pos] = desc_vals[i]
    return out


def order_binary_tree(desc_vals: np.ndarray) -> np.ndarray:
    """Recursively fill the midpoint of the remaining interval (binary‑tree layout)."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)

    def recurse(l: int, r: int, pos_list):
        if l > r:
            return
        m = (l + r) // 2
        pos_list.append(m)
        recurse(l, m - 1, pos_list)
        recurse(m + 1, r, pos_list)

    positions = []
    recurse(0, n - 1, positions)
    out[positions] = desc_vals
    return out

# Smoothed loss and gradient (log‑sum‑exp surrogate) --------------------
def _smooth_loss_and_grad(a: np.ndarray, eps: float):
    """Return (smooth loss, gradient) for the log‑sum‑exp surrogate."""
    b = _fft_self_convolve(a)
    maxb = b.max()
    # stable softmax
    w = np.exp((b - maxb) / eps)
    sum_w = w.sum()
    w /= sum_w
    # gradient of the surrogate
    grad = 2.0 * _fft_convolve(a[::-1], w)[a.size - 1 : a.size - 1 + a.size]
    loss = maxb + eps * np.log(sum_w)          # = eps * logsumexp(b/eps)
    return loss, grad

def _adam_smooth_descent(a0: np.ndarray, deadline: float,

                         max_iters: int = 200_000,

    best_score = _max_convolution_score(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    t = 0
    eps_idx = 0
    eps = eps_schedule[eps_idx]

    while time.time() < deadline and t < max_iters:
        t += 1
        _, grad = _smooth_loss_and_grad(a, eps)

        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad ** 2)

        v_hat = v / (1.0 - 0.999 ** t)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a -= eta * m_hat
        a = _project_onto_simplex(a)

        if t % 200 == 0:
            cur = _max_convolution_score(a)

            eps = eps_schedule[eps_idx]

            break

        # Adam on smooth surrogate
        a, sc = _adam_smooth_descent(seed_vec, deadline, eps_sched,
                                      max_iters=200_000, step0=0.2)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_seeds)} Adam‑smooth -> {best_score:.6f}")

        # Exponentiated‑gradient on smooth surrogate
        a, sc = _exp_gradient_smooth(seed_vec, deadline, eps_sched,
                                     max_iters=150_000, eta0=0.8)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_seeds)} EG‑smooth -> {best_score:.6f}")

        # Plain SGD on smooth surrogate
        a, sc = _sgd_smooth(seed_vec, deadline, eps_sched,
                            max_iters=200_000, step0=0.5)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_seeds)} SGD‑smooth -> {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Frank‑Wolfe style pairwise mass transfer on the current best
    a = best_seq.copy()
    cur_score = best_score
    it_fw = 0
    while time.time() < deadline and it_fw < 200_000:
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        delta = min(a[i_src], 1.0 - a[i_dst])
        if delta <= 0:
            break
        step = 0.9 * delta                # move almost all admissible mass
        a[i_src] -= step
        a[i_dst] += step
        cur_score = _max_convolution_score(a)
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time():.1f}] FW iter {it_fw} new best {best_score:.6f}")
        it_fw += 1
        if it_fw % 5000 == 0:
            # occasional jitter to escape plateaus
            jitter = np.random.randn(a.size) * 1e-5
            a = _project_onto_simplex(a + jitter)

    # ------------------------------------------------------------------
    # 6. Final local random refinement
    sigma = 1e-3
    while time.time() < deadline:
        cand = best_seq + sigma * np.random.randn(best_seq.size)
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s
        sc = _max_convolution_score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand
            print(f"[{time.time():.1f}] random perturbation improved to {best_score:.6f}")
            sigma = max(sigma * 0.9, 5e-5)   # shrink step size when successful
        else:
            sigma = min(sigma * 1.02, 2e-3)  # slowly increase to explore

    # ------------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
