def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """
    Exact surrogate when Σa = 1:
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) under Σa = 1.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # gradient of the (non‑smooth) max‑convolution:
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for terms where i=j (even active indices)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.30,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 2_000_000):
    """Adam on the simplex (projected). Returns best point seen."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          eta0: float = 0.8,
                          max_iters: int = 1_500_000):
    """Exponentiated‑gradient (KL‑mirror) descent on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with concentration ``alpha``."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _linear_decreasing_distribution(n: int) -> np.ndarray:
    """Simple linear decay from left to right."""
    i = np.arange(n, dtype=np.float64)
    vals = (n - i)
    vals /= vals.sum()
    return vals


def _exponential_decreasing_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    """Purely decreasing exponential (no centre symmetry)."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * i)
    vals /= vals.sum()
    return vals


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _spike_distribution(n: int, k: int = 20) -> np.ndarray:
    """k random positions get random positive weight, everything else zero."""
    a = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, size=min(k, n), replace=False)
    vals = np.random.rand(len(idx)) + 1e-12
    a[idx] = vals
    a /= a.sum()
    return a


def _double_bump_distribution(n: int,
                              block_frac: float = 0.25,
                              gap_frac: float = 0.1) -> np.ndarray:
    """
    Two equal uniform blocks separated by a gap.
    ``block_frac`` is the fraction of entries occupied by both blocks.
    ``gap_frac`` is the fraction left empty between them.
    """
    total_block = max(2, int(block_frac * n))
    m = total_block // 2  # size of each block
    g = int(gap_frac * n)
    a = np.zeros(n, dtype=np.float64)
    a[:m] = 1.0 / (2 * m)
    start2 = m + g
    if start2 + m > n:
        start2 = n - m
    a[start2:start2 + m] = 1.0 / (2 * m)
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best list found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose a base length (use the previous best if available)
    try:
        base_n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        base_n = 6000
    n = max(2000, min(int(base_n), 12000))

    # --------------------------------------------------------------
    # Generate a pool of candidates (all normalised to ℓ₁‑norm = 1)
    candidates = []   # (array, surrogate_score)

    # deterministic families ------------------------------------------------
    families = [
        (_arcsine_distribution, "arcsine"),
        (_linear_decreasing_distribution, "linear‑dec"),
        (lambda m: _exponential_decreasing_distribution(m, lam=0.005), "exp‑dec0.005"),
    ]
    for gen, name in families:
        a = gen(n)
        candidates.append((a, _max_convolution_score(a)))
        print(f"[{time.time():.1f}] n={n:<5d} family={name}")

    # exponential decays with several lambdas
    for lam in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]:
        a = _exponential_decreasing_distribution(n, lam=lam)
        candidates.append((a, _max_convolution_score(a)))
        print(f"[{time.time():.1f}] n={n:<5d} exp‑dec(lam={lam})")

    # power‑law family ----------------------------------------------------
    for p in np.arange(0.1, 3.01, 0.2):
        a = _power_law_distribution(n, p)
        candidates.append((a, _max_convolution_score(a)))
        # no per‑candidate print to keep output short

    # random Dirichlet draws (wide range of concentrations)
    for alpha in [0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]:
        a = _random_dirichlet_alpha(n, alpha)
        candidates.append((a, _max_convolution_score(a)))

    # spike‑like vectors ---------------------------------------------------
    for k in [5, 10, 20, 50]:
        a = _spike_distribution(n, k)
        candidates.append((a, _max_convolution_score(a)))

    # double‑bump with a few hyper‑parameters
    for block_frac in (0.15, 0.20, 0.25):
        for gap_frac in (0.02, 0.05, 0.1):
            a = _double_bump_distribution(n, block_frac, gap_frac)
            candidates.append((a, _max_convolution_score(a)))

    # ------------------------------------------------------------------
    # Add previously known best (if present)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            if prev.sum() == 0:
                prev = np.full_like(prev, 1.0 / prev.size)
            else:
                prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
            print(f"[{time.time():.1f}] added previous best, score={candidates[-1][1]:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Keep the top‑k seeds for intensive refinement
    candidates.sort(key=lambda x: x[1])
    top_k = min(15, len(candidates))
    top_seeds = candidates[:top_k]
    print(f"Generated {len(candidates)} candidates, keeping top {top_k} for refinement.")
    best_seq = top_seeds[0][0].copy()
    best_score = top_seeds[0][1]
    print(f"Initial best surrogate score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Intensive refinement of each seed (most of the remaining time)
    for idx, (seed_seq, seed_score) in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        # Adam descent
        seq_adam, sc_adam = _adam_descent(seed_seq, deadline,
                                          step0=0.30,
                                          max_iters=800_000)
        if sc_adam < best_score:
            best_score, best_seq = sc_adam, seq_adam.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} Adam improved to {best_score:.6f}")

        # Exponentiated‑gradient descent
        seq_eg, sc_eg = _exp_gradient_descent(seed_seq, deadline,
                                              eta0=0.8,
                                              max_iters=600_000)
        if sc_eg < best_score:
            best_score, best_seq = sc_eg, seq_eg.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} EG improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Frank‑Wolfe style mass‑transfer polishing (deterministic)
    a = best_seq.copy()
    cur_score = best_score
    fw_iter = 0
    while time.time() < deadline and fw_iter < 80_000:
        fw_iter += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))   # source (mass to remove)
        i_dst = int(np.argmin(grad))   # destination (mass to add)

        if i_src == i_dst:
            break

        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-15:
            break

        # try a few fractions
        improved = False
        for frac in (1.0, 0.5, 0.25):
            delta = max_transfer * frac
            a_new = a.copy()
            a_new[i_src] -= delta
            a_new[i_dst] += delta
            sc = _max_convolution_score(a_new)
            if sc < cur_score - 1e-13:
                a = a_new
                cur_score = sc
                improved = True
                break
        if improved and cur_score < best_score - 1e-14:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time():.1f}] FW iter={fw_iter} new best {best_score:.6f}")

        # occasional jitter to move out of tiny plateaus
        if fw_iter % 5000 == 0 and np.random.rand() < 0.13:
            jitter = np.random.randn(a.size) * 2e-5
            a = _project_onto_simplex(a + jitter, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    # Simulated‑annealing / jitter phase (stochastic)
    a = best_seq.copy()
    cur_score = best_score
    temp_start = 0.022
    temp_end = 0.0008
    while time.time() < deadline:
        # linear temperature schedule (more aggressive early)
        fraction = max(0.0, (deadline - time.time()) / (budget_s - 0.5))
        temp = temp_end + (temp_start - temp_end) * fraction

        i = np.random.randint(a.size)
        j = np.random.randint(a.size - 1)
        if j >= i:
            j += 1

        max_delta = min(a[i], 1.0 - a[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * temp

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if cur_score < best_score - 1e-14:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time():.1f}] Anneal improved to {best_score:.6f}")
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_score - cur_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                a = a_new
                cur_score = new_score

    # ------------------------------------------------------------------
    print(f"Final best surrogate score: {best_score:.6f} (length={len(best_seq)})")
    # Return plain Python list – ``evaluate_sequence`` will handle any clipping.
    return best_seq.tolist()
