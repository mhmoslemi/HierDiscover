"""
propose_candidate: multi‑stage search for a non‑negative step function that
minimises ``evaluate_sequence``.

The routine follows three main phases within the given time budget:

1. **Broad sampling** – generate many analytic families (uniform, arcsine,
   triangular, symmetric Beta, asymmetric Beta, sine‑power, cosine‑window,
   Gaussian) for several lengths (up to 20 000) together with random
   Dirichlet draws.  The ten best normalised vectors (according to the exact
   objective ``2·n·max(conv)``) are retained.

2. **Local continuous optimisation** – each retained candidate is refined
   separately.
   *   A smooth‑max (log‑sum‑exp) surrogate of the max‑convolution is
       minimised by an Adam‑style projected gradient descent.  The smoothing
       parameter ``α`` is annealed from 1 to 5·10⁴.
   *   The surrogate is then discarded and a true projected sub‑gradient
       descent on the exact max‑convolution is performed, again with
       Adam‑style momentum and a back‑tracking line‑search.
   *   Finally a cheap mass‑transfer hill‑climber (move a tiny amount of mass
       from the most “responsible’’ index to a low‑mass index) prunes the
       remaining peak.

3. **Selection** – the best sequence seen over the whole run is returned.

All vectors are kept on the probability simplex (sum = 1) so the scaling
in ``evaluate_sequence`` does not affect the objective.  The code respects a
hard time limit (default 1000 s) and prints occasional progress reports.

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of a vector onto the simplex {x≥0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x):
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a):
    """Full linear self‑convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full linear convolution of real vectors a and b using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a):
    """
    Exact objective for a normalised vector (sum=1):
    2·n·max(a*a).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


def _smooth_gradient(a, alpha):
    """
    Gradient of the smooth‑max surrogate
        fα(a) = (1/α)·log Σ_i exp(α·conv_i)
    where conv = a*a (full convolution).  The gradient is
        grad = 2·(a_rev * w)_{n‑1..n‑1+n}
    with w being a softmax over conv entries.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    maxc = conv.max()
    # softmax weights, numerically stable
    w = np.exp(alpha * (conv - maxc))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # convolution of reversed a with w
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, maxc


def _max_convolution_gradient(a, tol_factor=1e-12):
    """
    Sub‑gradient of the true max‑convolution (i.e. of max(conv)).
    Returns (grad, current_max).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    cur_max = conv.max()
    # indices where conv is essentially maximal
    tol = tol_factor * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - tol)
    w = np.zeros_like(conv)
    w[active] = 1.0
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    return grad, cur_max

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n, p, q):
    """Asymmetric Beta(p,q)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    """Discrete Gaussian centred at (n‑1)/2 with relative width sigma (relative to n)."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _adam_optimize_smooth(a, alpha_start, alpha_end, steps,
                          lr_start, lr_end, deadline):
    """
    Adam‑style projected gradient descent on the smooth‑max surrogate.
    Returns the improved vector (still summed to 1) and the best score seen.
    """
    eps = 1e-12
    n = a.shape[0]
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_local = a.copy()
    best_score = _max_convolution_score(a)

    for it in range(steps):
        if time.time() >= deadline:
            break
        t = it / steps
        # geometric interpolation of parameters
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t

        grad, _ = _smooth_gradient(a, alpha)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = a - step
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()

        if (it % 500 == 0) or (it == steps - 1):
            cur_score = _max_convolution_score(a)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_local = a.copy()
                print(f"[{time.time():.1f}]   smooth iter={it} score={cur_score:.6f}")

    return best_local


def _subgradient_optimize(a, deadline):
    """
    Projected sub‑gradient descent on the true max‑convolution with
    simple back‑tracking line‑search and Adam‑style momentum.
    """
    eps = 1e-12
    n = a.shape[0]
    beta1 = 0.9
    v = np.zeros_like(a)          # first moment (Adam‑style)
    step0 = 0.5
    it = 0
    best_score = _max_convolution_score(a)

    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_convolution_gradient(a)
        # Adam momentum
        v = beta1 * v + (1.0 - beta1) * grad
        v_hat = v / (1.0 - beta1 ** it)

        # back‑tracking line search
        step = step0
        while True:
            a_candidate = a - step * v_hat
            a_candidate = _project_onto_simplex(a_candidate, 1.0)
            a_candidate = np.maximum(a_candidate, eps)
            a_candidate /= a_candidate.sum()
            cand_score = _max_convolution_score(a_candidate)
            if cand_score < best_score - 1e-14 or step < 1e-8:
                break
            step *= 0.5

        a = a_candidate
        if cand_score < best_score - 1e-14:
            best_score = cand_score
            print(f"[{time.time():.1f}]   subgrad it={it} score={cand_score:.6f}")

        # occasional jitter to avoid stagnation
        if it % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            print(f"[{time.time():.1f}]   jitter at subgrad it={it}")

        # early exit if we are very close to the deadline
        if time.time() + 0.5 > deadline:
            break

    return a


def _hillclimb(a, deadline):
    """
    Very cheap mass‑transfer hill‑climber.  Repeatedly moves a small amount
    of probability mass from the index participating in the current maximal
    product to a low‑mass index, accepting moves that decrease the peak.
    """
    eps = 1e-12
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    best_score = 2.0 * n * M
    best_seq = a.copy()

    while time.time() < deadline:
        it += 1
        # location of maximal convolution coefficient
        k = int(conv.argmax())
        # find the pair (i,j) with i+j=k giving the largest product a[i]*a[j]
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.argmin(a)   # move towards the current smallest entry
        if target == source:
            continue

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-13:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M
            best_seq = a.copy()
            best_score = 2.0 * n * M
            if it % 200 == 0:
                print(f"[{time.time():.1f}]   hill it={it} score={best_score:.6f}")

        if it % 5000 == 0:
            # gentle jitter to shake out flat plateaus
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}]   hill jitter at it={it}")

        if time.time() + 0.5 > deadline:
            break

    return best_seq

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:

    best_score = float("inf")

    def _maybe_push(vec, n):
