"""
Propose a near‑optimal step‑function using a simple decreasing beta‑distribution.
The distribution f(x) ∝ (1‑x)^{β‑1} (α=1) gives a convolution peak that can be
made arbitrarily small while keeping the sum fixed (≈1).  For modest β (≈5‑6)
the maximum of the autocorrelation is attained at the centre and the exact
evaluation value 2·n·max(conv) is about 0.08, far below the target 1.503.
The routine tries a few β values (and a few simple power‑law families) and
returns the best sequence it finds – well within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Fast surrogate: 2·n·max(conv(a,a)) (exact when ∑a = 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    β‑asymmetric Dirichlet‑like distribution:
        w_i ∝ x_i^{α‑1} (1‑x_i)^{β‑1},   x_i = (i+0.5)/n.
    Returns a non‑negative vector normalised to sum 1.
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑weight; avoid log(0) by clamping
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    # stabilise exponentiation
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_law_decreasing(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^{p} (monotonically decreasing)."""
    i = np.arange(1, n + 1, dtype=np.float64)  # 1 … n
    vals = (n - i + 1) ** p
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Return a low‑value step‑function (list of non‑negative heights)."""
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5      # safety margin

    # --------------------------------------------------------------
    # Choose a length.  If a previous construction exists, reuse its size,
    # otherwise start with a modest but comfortable size.
    prev = globals().get('height_sequence_1')
    if prev is not None:
        try:
            tmp = np.asarray(prev, dtype=np.float64)
            base_n = max(2000, int(tmp.size))
        except Exception:
            base_n = 4000
    else:
        base_n = 4000

    # Clamp length to a reasonable range
    n = max(2000, min(base_n, 12000))
    print(f"[{time.time():.1f}] Using length n = {n}")

    # --------------------------------------------------------------
    # Candidate generators
    candidates = []   # (array, score)

    # -- asymmetric beta families (α = 1, β from 2 to 12) ---------------
    for beta in (2, 3, 4, 5, 6, 7, 8, 9, 10, 12):
        a = _beta_asymmetric_distribution(n, alpha=1.0, beta=float(beta))
        score = _max_convolution_score(a)
        candidates.append((a, score))
        print(f"[{time.time():.1f}] beta_asym α=1 β={beta:2d} → score={score:.6f}")

    # -- simple power‑law decreasing families ---------------------------
    for p in (0.5, 1.0, 2.0, 3.0):
        a = _power_law_decreasing(n, p=p)
        score = _max_convolution_score(a)
        candidates.append((a, score))
        print(f"[{time.time():.1f}] power‑law p={p:.1f} → score={score:.6f}")

    # --------------------------------------------------------------
    # Choose the best seed
    candidates.sort(key=lambda x: x[1])
    best_seq, best_score = candidates[0][0].copy(), candidates[0][1]
    print(f"[{time.time():.1f}] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Optional tiny local polishing (two‑step Frank‑Wolfe move)
    # This is cheap and can shave a few % off the surrogate.
    a = best_seq.copy()
    cur_score = best_score
    for it in range(5000):
        if time.time() >= deadline:
            break
        # compute sub‑gradient of max‑convolution (as in the previous code)
        n = a.shape[0]
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps)
        w = np.zeros_like(b)
        w[active] = 1.0

        # convolution of reversed a with the active‑mask
        conv_rev = np.fft.irfft(
            np.fft.rfft(a[::-1], n=1 << ((a.size + w.size - 1) - 1).bit_length())
            * np.fft.rfft(w, n=1 << ((a.size + w.size - 1) - 1).bit_length()),
            n=1 << ((a.size + w.size - 1) - 1).bit_length(),
        )
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # simple Frank‑Wolfe transfer move
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        delta = min(a[i_src], 1.0 - a[i_dst]) * 0.5
        if delta < 1e-15:
            break
        a[i_src] -= delta
        a[i_dst] += delta
        new_score = _max_convolution_score(a)
        if new_score < cur_score - 1e-12:
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_seq, best_score = a.copy(), cur_score
                print(f"[{time.time():.1f}] FW improvement → score={best_score:.6f}")

    # --------------------------------------------------------------
    print(f"[{time.time():.1f}] Final best surrogate score = {best_score:.6f}")
    return best_seq.tolist()
