    a /= a.sum()
    # smoothing stage: exponentiated gradient (mirror descent)
    for it in range(1000):

        maxc = conv.max()

def _exp_gradient_descent(a, steps, alpha0, alpha1, lr0, lr1, deadline):
    n = a.shape[0]; eps = 1e-12
    for it in range(steps):
        if time.time() >= deadline: break
        t = it / steps
        alpha = alpha0 + (alpha1 - alpha0) * t
        lr = lr0 * ((lr1 / lr0) ** t) if lr0 != 0 else 0

def _hill_climb(a, deadline, max_iters=100000):
    n = a.shape[0]
    eps = 1e-12
    a = np.maximum(a, eps)
    a /= a.sum()
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        k = int(conv.argmax())
        # compute contributions of each i to conv[k]
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i_range = np.arange(i_low, i_high + 1, dtype=int)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]
        source_idx = i_range[np.argmax(contrib)]
        # Determine target that does not affect conv[k]
        # indices outside the window [i_low, i_high]
        idx_arr = np.arange(n)
        mask_outside = (idx_arr < i_low) | (idx_arr > i_high)
        if np.any(mask_outside):
            # choose index with minimum a among those
            candidates = idx_arr[mask_outside]
            target_idx = int(candidates[np.argmin(a[candidates])])
        else:
            # fallback to global min
            target_idx = int(np.argmin(a + (np.arange(n) == source_idx) * 1e6))
        if target_idx == source_idx:
            continue
        delta = min(0.5 * a[source_idx], 0.02)
        if delta < eps:
            break
        a_temp = a.copy()
        a_temp[source_idx] -= delta
        a_temp[target_idx] += delta
        a_temp = np.maximum(a_temp, eps)
        a_temp /= a_temp.sum()
        conv_new = _fft_self_convolve(a_temp)
        new_M = conv_new.max()
        if new_M < M - 1e-12:
            a = a_temp
            conv = conv_new
            M = new_M
        else:
            # possibly try smaller delta or different target
            # try small random jitter after many failures
            if it % 1000 == 0:
                # random jitter
                jitter = np.random.randn(n) * 1e-5
                a = a + jitter
                a = np.maximum(a, eps)
                a /= a.sum()
                conv = _fft_self_convolve(a)
                M = conv.max()
    return a

"""
Propose a step‑function (non‑negative list) that minimises ``evaluate_sequence``.

Main ideas
---------
* Initialise the distribution with the analytic *arcsine* law – it already has a
  low max‑self‑convolution.
* Refine it by a **smooth‑max mirror descent** (exponentiated gradient) that
  directly follows the sub‑gradient of the max‐convolution.
* Follow with a **targeted hill‑climber**:
  – locate the index `k` where the convolution attains its maximum,
  – find the pair `(i,j)` that contributes most to this entry,
  – move a small amount of mass from the heavier partner `i` to a position
    that does *not* affect the current peak (i.e. outside the window that
    contributes to `k`).  Accept the move only if the peak decreases.
* When time permits, jitter the current best solution and run the whole
  refinement again – this gives a cheap multi‑restart strategy.
* All operations are performed on the simplex (sum = 1, non‑negative) and use
  FFT‑based convolutions (O(n log n)) so that even a few thousand iterations are
  cheap for n≈4000.

The function respects a 1000 s wall‑clock limit and returns the best vector
found.
"""

import time
import random
import numpy as np
from math import log

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _exp_gradient_descent(
    a: np.ndarray,
    steps: int,
    alpha0: float,
    alpha1: float,
    lr0: float,
    lr1: float,
    deadline: float,
) -> np.ndarray:
    """Smooth‑max (exponential‑weights) mirror descent on the simplex."""
    n = a.shape[0]
    eps = 1e-12
    for it in range(steps):
        if time.time() >= deadline:
            break
        t = it / steps
        # linear schedule for the smooth‑max temperature
        alpha = alpha0 + (alpha1 - alpha0) * t
        # geometric schedule for the learning rate
        lr = lr0 * ((lr1 / lr0) ** t) if lr0 != 0 else 0.0

        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # soft‑max weights (the gradient of the smooth max)
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of Σ w_i (a*a)_i   (see derivation in the analysis)
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # exponentiated‑gradient update and projection back to the simplex
        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray, deadline: float, max_iters: int = 200_000) -> np.ndarray:
    """Deterministic hill‑climber that moves mass away from the current peak."""
    n = a.shape[0]
    eps = 1e-12
    a = np.maximum(a, eps)
    a /= a.sum()
    conv = _fft_self_convolve(a)
    cur_max = conv.max()
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        k = int(conv.argmax())                     # index of the worst convolution entry
        # indices that actually contribute to conv[k]
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i_vals = np.arange(i_low, i_high + 1, dtype=int)
        j_vals = k - i_vals
        contrib = a[i_vals] * a[j_vals]
        src = i_vals[np.argmax(contrib)]           # source index (largest product)

        # Choose a target that does **not** affect conv[k] (outside the window)
        mask_outside = (np.arange(n) < i_low) | (np.arange(n) > i_high)
        if np.any(mask_outside):
            cand = np.where(mask_outside)[0]
            tgt = int(cand[np.argmin(a[cand])])
        else:
            # fallback – move to the globally smallest entry
            tgt = int(np.argmin(a + (np.arange(n) == src) * 1e6))

        if tgt == src:
            continue

        delta = min(0.5 * a[src], 0.02)   # keep moves modest
        if delta < eps:
            break

        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta
        a_new = np.maximum(a_new, eps)
        a_new /= a_new.sum()
        conv_new = _fft_self_convolve(a_new)
        new_max = conv_new.max()

        # Accept only genuine improvement (tiny tolerance to avoid oscillation)
        if new_max < cur_max - 1e-12:
            a = a_new
            conv = conv_new
            cur_max = new_max
        else:
            # occasionally shake the vector to get out of plateaus
            if it % 1500 == 0:
                jitter = np.random.randn(n) * 1e-5
                a += jitter
                a = np.maximum(a, eps)
                a /= a.sum()
                conv = _fft_self_convolve(a)
                cur_max = conv.max()
    return a


# ----------------------------------------------------------------------
def _local_optimize(start_vec: np.ndarray, deadline: float) -> np.ndarray:
    """Full local refinement: smooth‑max descent followed by hill‑climbing."""
    # stage 1 – smooth max (4000 steps, fairly aggressive schedule)
    a = _exp_gradient_descent(
        start_vec,
        steps=4000,
        alpha0=1.0,
        alpha1=500.0,
        lr0=0.15,
        lr1=0.001,
        deadline=deadline,
    )
    # stage 2 – deterministic hill climbing
    a = _hill_climb(a, deadline, max_iters=200_000)
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The algorithm concentrates on a single length (≈4000) that trades
    off evaluation cost and quality, then spends the remaining wall‑clock
    time on intensive refinement and cheap random restarts.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a length that is large enough to exploit the asymptotic
    #    improvement but still cheap for repeated FFTs.
    n = 4000
    eps = 1e-12

    # 2. Base distribution – the arcsine law already beats uniform heavily.
    base = _distribution_arcsine(n)
    base = np.maximum(base, eps)
    base /= base.sum()

    # 3. Main optimisation phase.
    best_a = _local_optimize(base.copy(), deadline)
    best_score = _score(best_a)
    print(f"[{time.time():.1f}] after first optimisation: score {best_score:.6f}")

    # 4. Cheap multi‑restart: jitter the current best and re‑optimise while time permits.
    while time.time() < deadline - 5.0:
        # small random perturbation
        jitter = np.random.randn(n) * 5e-4
        a_try = best_a + jitter
        a_try = np.maximum(a_try, eps)
        a_try /= a_try.sum()
        a_try = _local_optimize(a_try, deadline)
        s_try = _score(a_try)
        if s_try < best_score - 1e-12:
            best_a = a_try
            best_score = s_try
            print(f"[{time.time():.1f}] new best after restart: {best_score:.6f}")

    print(f"[{time.time():.1f}] final best score ≈ {best_score:.6f} (length={n})")
    return best_a.tolist()
