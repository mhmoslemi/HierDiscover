"""

minimises ``evaluate_sequence``.  The search consists of:

3️⃣  A final hill‑climbing phase that makes tiny random transfers and
    occasional jitter to capture any remaining marginal gain.

All updates respect the simplex constraint (non‑negative, sum = 1) by
projecting onto the probability simplex.  The routine respects the global
budget (default 1000 s) and returns the best sequence found.
"""

#  Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
#  Convolution utilities (FFT based)
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
#  Cheap objective (exact when sum(a)=1)
def _max_convolution_score(a: np.ndarray) -> float:
    """Returns 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
#  Exact sub‑gradient of the max‑convolution (used by several optimisers)
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.
    Returns (grad, cur_max, active_set).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution with reversed a (as in derivation)
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (k = 2*i)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
#  Families of distributions (all return a normalized probability vector)

def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _smoothed_arcsine_distribution(n: int, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5 + eps) * (n - i - 0.5 + eps))
    return vals / vals.sum()


def _product_power_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-alpha)
    return vals / vals.sum()


def _inverse_product_distribution(n: int, alpha: float, delta: float) -> np.ndarray:
    """Generalised arcsine‑type shape with an additive offset delta."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5 + delta) * (n - i - 0.5 + delta)) ** (-alpha)
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # safeguard against log(0)
    eps = np.finfo(float).eps
    x = np.clip(x, eps, 1 - eps)
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Γ–Dirichlet, α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
#  Optimisation primitives ------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 200_000, eta0: float = 1.0):
    """Mirror‑descent (exponentiated‑gradient) on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        a = a / s if s != 0 else np.full_like(a, 1.0 / a.size)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 200_000, step0: float = 0.3,
                  beta1: float = 0.9, beta2: float = 0.999):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _smooth_gradient(a: np.ndarray, gamma: float):
    """Weighted gradient of the soft‑max of a*a (γ controls smoothness)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    b_max = b.max()
    w = np.exp(gamma * (b - b_max))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    diag_w = w[2 * np.arange(n)]
    grad -= a * diag_w
    return grad, b_max


def _smooth_gradient_descent(a0: np.ndarray, deadline: float,
                             max_iters: int = 300_000,
                             eta0: float = 0.8,
                             gamma0: float = 2.0,
                             gamma_factor: float = 1.005):
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    gamma = gamma0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _ = _smooth_gradient(a, gamma)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score, best_a = cur_score, a.copy()
        gamma *= gamma_factor
    return best_a, best_score


def _simulated_annealing(a0: np.ndarray, deadline: float,
                         max_iters: int = 500_000,
                         init_temp_factor: float = 0.02,
                         cooling: float = 0.9999,
                         max_move_frac: float = 0.02):
    """Random mass‑transfer SA on the simplex."""
    n = a0.shape[0]
    a = a0.copy()
    cur_score = _max_convolution_score(a)
    best_a = a.copy()
    best_score = cur_score

    # temperature proportional to current score (a few percent)
    T = cur_score * init_temp_factor
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # pick distinct i, j
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        max_delta = min(a[i], max_move_frac)
        if max_delta <= 0:
            continue
        delta = np.random.rand() * max_delta

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = _max_convolution_score(a_new)
        dE = new_score - cur_score
        if dE < 0 or np.random.rand() < np.exp(-dE / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

        # temperature schedule
        T *= cooling
        # slowly shrink the allowed move size
        if it % 5000 == 0:
            max_move_frac = max_move_frac * 0.9
            max_move_frac = max(max_move_frac, 1e-4)

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the global time budget and returns a Python list ready
    to be evaluated.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0          # safety margin

    # --------------------------------------------------------------
    # Pick a representative length (try to use the previous best if available)
    try:
        base_n = len(height_sequence_1)   # type: ignore[name-defined]
        if base_n < 500:
            base_n = 500
    except Exception:
        base_n = 3500

    # a few nearby lengths – slightly wider than before
    lengths = [int(base_n * f) for f in (0.8, 0.9, 1.0, 1.1, 1.2)]
    lengths = [max(500, n) for n in lengths]

    # --------------------------------------------------------------
    # 1️⃣  Enrich the coarse parameter sweep
    best_seq = None
    best_score = float('inf')

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- product‑power with offset (generalised arcsine) -------------
        alphas = np.arange(0.35, 0.71, 0.02)
        deltas = [0.0, 0.1, 0.2, 0.5, 1.0]
        for alpha in alphas:
            for delta in deltas:
                a = _inverse_product_distribution(n, alpha, delta)
                score = _max_convolution_score(a)
                if score < best_score:
                    best_score, best_seq = score, a.copy()

        # ----- smoothed arcsine (different epsilons) -----------------------
        epsilons = [0.0, 1e-4, 1e-3, 5e-3, 1e-2, 2e-2, 5e-2, 1e-1,
                    2e-1, 5e-1, 1.0]
        for eps in epsilons:
            a = _smoothed_arcsine_distribution(n, eps)
            score = _max_convolution_score(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()

        # ----- symmetric Beta  (p close to 0 gives very U‑shape) ----------
        ps = np.arange(0.05, 0.91, 0.02)
        for p in ps:
            a = _beta_symmetric_distribution(n, p)
            score = _max_convolution_score(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()

        # ----- exponential tails -------------------------------------------
        lams = [0.0005, 0.001, 0.002, 0.003, 0.005, 0.010]
        for lam in lams:
            a = _exponential_distribution(n, lam)
            score = _max_convolution_score(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()

        # ----- a few random Dirichlet draws --------------------------------
        for _ in range(4):
            a = _random_dirichlet(n)
            score = _max_convolution_score(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()

    # ----- incorporate the previous best, if any ------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev = prev / prev.sum()
            score = _max_convolution_score(prev)
            if score < best_score:
                best_score, best_seq = score, prev.copy()
    except Exception:
        pass

    print(f"[{time.time():.1f}] after coarse sweep best cheap={best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Mixed refinement stage
    remaining = deadline - time.time()
    if remaining <= 0.0:
        return best_seq.tolist()

    # 2.1 exponentiated‑gradient (10% of remaining)
    sub_deadline = time.time() + remaining * 0.10
    seq, sc = _exp_gradient_descent(best_seq, sub_deadline,
                                    max_iters=300_000, eta0=1.0)
    if sc < best_score:
        best_seq, best_score = seq, sc
        print(f"[{time.time():.1f}] after exp‑grad   {best_score:.6f}")

    remaining = deadline - time.time()
    if remaining <= 0.0:
        return best_seq.tolist()

    # 2.2 simulated annealing (45% of remaining)
    sub_deadline = time.time() + remaining * 0.45
    seq, sc = _simulated_annealing(best_seq, sub_deadline,
                                   max_iters=400_000,
                                   init_temp_factor=0.02,
                                   cooling=0.9997,
                                   max_move_frac=0.02)
    if sc < best_score:
        best_seq, best_score = seq, sc
        print(f"[{time.time():.1f}] after SA        {best_score:.6f}")

    remaining = deadline - time.time()
    if remaining <= 0.0:
        return best_seq.tolist()

    # 2.3 smooth‑max continuation (30% of remaining)
    sub_deadline = time.time() + remaining * 0.30
    seq, sc = _smooth_gradient_descent(best_seq, sub_deadline,
                                        max_iters=300_000,
                                        eta0=0.8,
                                        gamma0=2.0,
                                        gamma_factor=1.005)
    if sc < best_score:
        best_seq, best_score = seq, sc
        print(f"[{time.time():.1f}] after smooth   {best_score:.6f}")

    remaining = deadline - time.time()
    if remaining <= 0.0:
        return best_seq.tolist()

    # 2.4 short Adam pass (remaining time)
    sub_deadline = time.time() + remaining * 0.90
    seq, sc = _adam_descent(best_seq, sub_deadline,
                            max_iters=200_000, step0=0.25)
    if sc < best_score:
        best_seq, best_score = seq, sc
        print(f"[{time.time():.1f}] after Adam      {best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Final fine‑grained hill‑climbing
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    it = 0
    max_transfer = 0.02  # start with up to 2 % of total mass per move

    while time.time() < deadline:
        it += 1
        # geometric decay of transfer magnitude
        transfer = max_transfer * (0.5 ** (it / 10000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_seq, best_score = a_new.copy(), new_score
                print(f"[{time.time():.1f}] hill‑climb iter={it} new best {best_score:.6f}")

        # occasional jitter to escape shallow plateaus
        if it % 4000 == 0 and np.random.rand() < 0.2:
            jitter = np.random.randn(n) * 3e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
