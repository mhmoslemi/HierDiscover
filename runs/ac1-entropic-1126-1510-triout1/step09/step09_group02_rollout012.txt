def _smooth_gradient(a, lam):
    conv = _fft_self_convolve(a)   # length 2n-1
    maxc = conv.max()
    # compute weights stable: w = np.exp(lam * (conv - maxc))
    w = np.exp(lam * (conv - maxc))
    w_sum = w.sum()
    w_norm = w / w_sum
    rev = _fft_convolve(a[::-1], w_norm)   # length 3n-2 (or n+2n-2)
    grad = 2.0 * rev[a.shape[0]-1 : a.shape[0]-1 + a.shape[0]]
    return grad

def _smooth_adam(a0, deadline, lam_start=5.0, lam_end=200.0, step0=0.2, beta1=0.9,beta2=0.999, max_iters=800_000):
    a = a0.copy()
    m=v etc.

    best_a = a.copy()
    best_score = _max_convolution_score(a)
    for t in range(1, max_iters+1):
        if time.time() >= deadline: break
        lam = lam_start + (lam_end - lam_start) * min(1.0, t / max_iters)
        grad = _smooth_gradient(a, lam)

        # update Adam
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1-beta1**t)
        v_hat = v / (1-beta2**t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)

        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score

"""
Propose a low‑score non‑negative step function.

Algorithm Overview
------------------
* Choose a handful of sequence lengths (≈4000‑15000).
* Generate a large pool of analytic and random seeds (beta, arcsine,
  exponential, power‑law, double‑bump, Dirichlet draws).  All seeds are
  normalised to ℓ₁‑norm = 1.
* Keep the best *K* seeds (K≈30) according to the exact objective
  ``2·n·max(conv(a,a))``.
* Refine each seed with a *smooth* Adam optimiser.  The non‑smooth
  max‑operator is replaced by a soft‑max (log‑sum‑exp) with a gradually
  increasing temperature λ; the resulting gradient is cheap to compute
  via FFTs and respects the simplex constraint via Euclidean projection.
* After the Adam passes, run a short Frank‑Wolfe style mass‑transfer
  polishing and a final simulated‑annealing jitter phase.
* Return the best sequence found before the 1000 s deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact objective (sum of a = 1): 2·n·max(conv(a,a))."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the smooth approximation
        f_λ(a) = (1/λ)·log Σ_k exp(λ·conv_k(a))
    where conv = a*a (full).  Returns a shape‑(n,) array.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)               # length 2n‑1
    maxc = conv.max()
    # stable softmax weights (max weight = 1)
    w = np.exp(lam * (conv - maxc))
    w_sum = w.sum()
    w_norm = w / w_sum
    # convolution of reversed a with the weight vector
    rev = _fft_convolve(a[::-1], w_norm)       # length 3n‑2
    grad = 2.0 * rev[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _smooth_adam(a0: np.ndarray, deadline: float,
                 lam_start: float = 5.0,
                 lam_end: float = 200.0,
                 step0: float = 0.20,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 max_iters: int = 500_000):
    """
    Adam on the simplex using the smooth max‑convolution gradient.
    Returns the best point seen and its surrogate score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _max_convolution_score(a)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        # linearly increase λ during the run
        lam = lam_start + (lam_end - lam_start) * min(1.0, t / max_iters)

        grad = _smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score

def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with concentration ``alpha``."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _double_bump_distribution(n: int,
                              block_frac: float = 0.25,
                              gap_frac: float = 0.1) -> np.ndarray:
    """Two equal uniform blocks separated by a gap."""
    total_block = max(2, int(block_frac * n))
    m = total_block // 2
    g = int(gap_frac * n)
    a = np.zeros(n, dtype=np.float64)
    a[:m] = 1.0 / (2 * m)
    start2 = m + g
    if start2 + m > n:
        start2 = n - m
    a[start2:start2 + m] = 1.0 / (2 * m)
    return a
