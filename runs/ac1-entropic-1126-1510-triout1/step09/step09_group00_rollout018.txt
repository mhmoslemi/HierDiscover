"""
Simple yet effective search for a non‑negative step function that minimises
``evaluate_sequence``.  The algorithm works in three stages:

1. **Grid search** over symmetric Beta(α,α) densities on a few modest
   lengths (2000‑6000).  The convolution is evaluated with a fast FFT
   implementation, so we can scan dozens of shapes cheaply.

2. **Exponentiated‑gradient (mirror) descent** on the best candidate
   discovered in step 1.  At each iteration we locate the maximal
   convolution entry, form the exact sub‑gradient
   (the shifted vector) and update with a decaying step size, projecting
   back onto the probability simplex.

3. **Fine‑grained random pairwise swaps**, accepting only moves that
   improve the true objective.  This cleans up any remaining
   irregularities.

The whole routine respects the global 1000 s budget, reports progress,
and returns the best sequence found as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # Fast path when already feasible
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (fast, O(n log n))."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # Work in log‑space to avoid overflow/underflow
    logw = (p - 1) * (np.log(x) + np.log(1.0 - x))
    # Clip extreme values
    logw = np.clip(logw, -700, 700)
    w = np.exp(logw)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _subgradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 1.0,
    max_iter: int = 200_000,
    eval_interval: int = 500,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the true max‑convolution objective.
    The sub‑gradient comes from the currently maximal convolution entry.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        # 1) compute full convolution and the index of its maximum
        conv = _fft_self_convolve(a)
        idx = int(conv.argmax())

        # 2) sub‑gradient: 2·a shifted by idx (only where the shift stays inside the array)
        grad = np.zeros_like(a)
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        i_range = np.arange(i_start, i_end + 1)
        j_range = idx - i_range
        grad[i_range] = 2.0 * a[j_range]

        # 3) step size schedule η_t = lr0 / √t
        eta = lr0 / math.sqrt(t)

        # 4) exponentiated‑gradient update and projection onto the simplex
        a = a * np.exp(-eta * grad)
        a = _project_onto_simplex(a)

        # 5) occasional exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise mass transfer (tiny simulated‑annealing step) -------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a small amount of mass from one index to another.
    Keep the move only if the true objective improves.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        # ensure destination differs from source
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.30  # move up to 30 % of the source mass
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  Returns the best vector found as a plain list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # a small safety margin

    # ------------------------------------------------------------------
    # 1. Coarse analytic sweep: symmetric Beta(p,p) on several lengths
    lengths = [2000, 3000, 4000, 5000, 6000]
    p_grid = np.linspace(0.05, 1.5, 150)  # dense enough for a fine scan

    best_seq = None
    best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # a few classic families for completeness
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            if time.time() >= deadline:
                break
            score = _objective(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(
                    f"[{time.time():.1f}] new global best (uniform/arc) len={n} score={score:.6f}"
                )

        # dense Beta(p,p) sweep
        for p in p_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_beta_sym(n, p)
            score = _objective(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(
                    f"[{time.time():.1f}] new global best len={n} p={p:.4f} score={score:.6f}"
                )

    # ------------------------------------------------------------------
    # 2. If nothing was found (extremely unlikely) fall back to a uniform vector
    if best_seq is None:
        n = 2000
        best_seq = _distribution_uniform(n)
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Refine the current best with exponentiated‑gradient descent
    if time.time() < deadline:
        # allocate most of the remaining budget to refined optimisation
        sub_deadline = deadline - 5.0  # keep a little slack for the final walk
        best_seq, best_score = _subgradient_descent(
            best_seq.copy(),
            sub_deadline,
            best_seq.copy(),
            best_score,
            lr0=1.2,          # a slightly larger initial step to move faster
            max_iter=500_000,
            eval_interval=800,
        )

    # ------------------------------------------------------------------
    # 4. Final local random walk for polishing
    if time.time() < deadline:
        best_seq, best_score = _pairwise_swap_local(
            best_seq.copy(),
            deadline,
            best_seq.copy(),
            best_score,
        )

    # ------------------------------------------------------------------
    print(
        f"Final estimated best score: {best_score:.6f} (length={len(best_seq)})"
    )
    # The returned list is fed to ``evaluate_sequence`` afterwards.
    return best_seq.tolist()
