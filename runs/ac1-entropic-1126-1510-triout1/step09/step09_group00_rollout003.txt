The program searches for a non‑negative step function  
\(a=(a_0,\dots ,a_{n-1})\) that minimises  

\[
\text{score}(a)=\frac{2\,n\;\max\bigl((a*a)\bigr)}{\bigl(\sum a_i\bigr)^2},
\]

i.e. the value returned by `evaluate_sequence`.  
The objective is scale‑invariant, therefore we keep the vector on the
simplex \(\sum a_i=1,\ a_i\ge 0\).

The search consists of three stages

1. **Rich seed pool** – a few classic analytic families (uniform,
   arcsine, beta, power‑law, exponential, two‑block spikes, sparse
   random) are generated for several lengths. The best few seeds are kept.

2. **Smoothed exponentiated‑gradient descent** – instead of using the
   nonsmooth max‑convolution we minimise a *smoothed* \(L_p\) version

\[
F_p(a)=\Bigl(\sum_i (a*a)_i^{\,p}\Bigr)^{1/p},
\qquad p\!\uparrow\!\infty .
\]

   For a fixed \(p\) the gradient is
   \[
   \nabla F_p(a)=\frac{2}{F_p^{p-1}}\;(a^{\text{rev}}*w)_{[n-1:n-1+n]},
   \qquad
   w_i=(a*a)_i^{\,p-1},
   \]
   where `*` is a full convolution and `a_rev` is `a` reversed.
   An exponentiated‑gradient (mirror descent) update
   \(a_i\leftarrow a_i\exp(-\eta\,\nabla_i)\) is performed and the vector
   is projected back onto the simplex.  
   The smoothing parameter is increased geometrically
   \(p\in\{4,8,16,32,64\}\); for each \(p\) a few hundred iterations are
   executed.  This gives a much richer descent direction than the pure
   sub‑gradient on the max‑index and drives the distribution toward the
   true optimum.

3. **Local pairwise mass transfers** – after the gradient phase a cheap
   Metropolis walk moves a random amount of mass from one index to another.
   Only improving moves are kept; this polishes the final vector.

All stages share the global 1000 s budget.  The best vector ever seen is
returned as a plain `list[float]`.

"""

   lengths.  Keep the best few seeds.

   minimise the L_p‑norm of the self‑convolution (p = 4,8,16,32,64).
   The exact gradient of the L_p‑objective is computed by FFTs,
   an exponentiated‑gradient (mirror descent) step is taken and the
   vector is projected onto the probability simplex.

3. Finish with a cheap greedy pairwise mass‑transfer walk that accepts
   only moves that strictly improve the true max‑convolution objective.

The best vector found within the time budget is returned.

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    n = len(x)
    m = len(y)
    size = 1 << ((n + m - 1) - 1).bit_length()
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[: n + m - 1]


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(alpha,beta)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^p (monotone increasing)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    """a_i ∝ exp(lam·i) (monotone increasing)."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    """Two symmetric blocks at the ends."""
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Broad pool of seeds for a given length."""
    cand = []

    # classic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # symmetric beta
    for p in (0.05, 0.1, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0):
        cand.append(_distribution_beta_sym(n, p))

    # asymmetric beta – a few random samples
    for _ in range(5):
        alpha = 10 ** rng.uniform(-2, 1)   # 0.01 … 10
        beta = 10 ** rng.uniform(-2, 1)    # 0.01 … 10
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # monotone power‑law
    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))

    # monotone exponential
    for lam in (0.001, 0.005, 0.01, 0.02, 0.05):
        cand.append(_distribution_exponential_increasing(n, lam))

    # two‑block spikes
    for block_frac in (0.05, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac))

    # sparse random start (~1 % non‑zeros)
    cand.append(_distribution_random_sparse(n, rng, density=0.01))

    return cand

# Smoothed exponentiated‑gradient descent -------------------------------
def _smooth_exponential_descent(seed: np.ndarray,
                               deadline: float,
                               best_seq: np.ndarray,
                               best_score: float,
                               lr0: float = 0.5,
                               steps_per_p: int = 300) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed Lp‑norm of the self‑convolution using
    exponentiated‑gradient updates.
    The smoothing parameter p is increased geometrically: 4 → 8 → 16 → 32 → 64.
    """
    a = seed.copy()
    n = a.shape[0]
    t = 0
    p = 4
    eps = 1e-12

    while time.time() < deadline and p <= 64:
        for _ in range(steps_per_p):
            if time.time() >= deadline:
                break
            t += 1

            # current convolution and true objective
            conv = _fft_self_convolve(a)
            cur = 2.0 * n * conv.max()
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] p={p} it={t} new best {best_score:.6f}")

            # weights for the Lp‑norm (normalised to avoid overflow)
            w = np.power(np.maximum(conv, eps), p - 1.0)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient = 2 * (a_rev * w)[n-1:n-1+n]
            a_rev = a[::-1]
            grad = 2.0 * _fft_convolve(a_rev, w)[n - 1: n - 1 + n]

            # exponentiated‑gradient step
            eta = lr0 / math.sqrt(t)
            a = a * np.exp(-eta * grad)

            # simplex projection (keeps sum == 1)
            a = _project_onto_simplex(a)

        p = min(p * 2, 64)

    return best_seq, best_score

# Greedy pairwise mass transfer ----------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Randomly move a small amount of mass; keep move only if objective improves."""
    n = a.shape[0]
    while time.time() < deadline:
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)

        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")

    return best_seq, best_score

    Returns the best vector found (as a Python list).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a handful of lengths (larger gives finer discretisation)
    lengths = [2000, 4000, 8000, 12000, 20000]

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 2. Scan analytic families for each length
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        for cand in _initial_candidates_extended(n, rng):
            if time.time() >= deadline:
                break
            cand = cand / cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))

        # keep only the best few seeds (global, not per‑length)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:12]

    # ------------------------------------------------------------------
    # 3. Incorporate a previously discovered sequence if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Fallback – should never be needed
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Main optimisation pipeline
    # --------------------------------------------------------------
    # 5a – Smoothed exponentiated‑gradient descent on a few top seeds
    remaining = deadline - time.time()
    if remaining > 60:
        # choose up to 4 best seeds for the heavy optimisation
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:4]]
        for seed_seq in seeds:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + (deadline - time.time()) / max(1, len(seeds))
            overall_best_seq, overall_best_score = _smooth_exponential_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.5,
                steps_per_p=400,
            )

    # --------------------------------------------------------------
    # 5b – final greedy pairwise swaps / tiny simulated‑annealing fine‑tuning
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
