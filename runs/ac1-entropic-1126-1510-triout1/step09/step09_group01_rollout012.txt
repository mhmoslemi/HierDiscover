"""
Propose a non‑negative step function (a probability vector) that
minimises ``evaluate_sequence``.

Algorithm
---------
1️⃣  **Broad exploration** – generate many analytic families
    (uniform, arcsine, triangular, exponential, symmetric Beta,
    sine‑power, cosine‑window, Gaussian) for a few moderate lengths,
    plus a few random Dirichlet draws.  Keep the best normalised vector.

2️⃣  **Upscaling** – linearly interpolate the current best to a larger
    length (≈40 k) – the objective is essentially scale‑invariant,
    so a finer discretisation can approach the continuous optimum more
    closely.

3️⃣  **Smooth‑max mirror descent** – minimise the soft maximum
    ``Lα(a)= (1/α) log Σ exp(α·(a*a))`` using an exponentiated‑gradient
    (entropy mirror) step.  The temperature `α` is annealed from 1 to
    4·10⁵ while the stepsize decays from 0.4 to 2·10⁻³.
    The iteration count is chosen dynamically from the remaining time.

4️⃣  **Adam‑style sub‑gradient polishing** – switch to the true
    non‑smooth objective `M(a)=max_k (a*a)[k]`.  The active set (all
    indices attaining the current maximum up to a tiny tolerance) gives
    a sub‑gradient.  An Adam momentum update with back‑tracking line‑search
    refines the solution.

5️⃣  **Stochastic hill‑climber** – occasional random mass transfers
    (source → target) are tried; a transfer is kept whenever it reduces
    the maximal convolution value.  Small Gaussian jitter is injected
    periodically to escape shallow plateaus.

All steps respect the overall 1000 s budget.  The returned vector is a
plain list of ``float`` and is guaranteed to have non‑negative entries
and a positive sum (the optimiser works on the simplex, i.e. ∑a_i=1).

The code is deliberately self‑contained – only ``numpy`` and ``scipy``
are required (both are available in the execution environment).
"""

import time
import math
import random
import numpy as np
from scipy.optimize import linprog  # used only for a cheap sanity check


# ----------------------------------------------------------------------
#  Utility functions
# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Return the exact value that ``evaluate_sequence`` would give for
    a normalised vector (∑a = 1).  It is 2·n·max(a*a).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families (all already normalised to unit sum)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n: int, coeff: float = 0.5) -> np.ndarray:
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a list of non‑negative heights (a probability vector) that
    minimises ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin
    eps = 1e-12

    best_seq = None
    best_score = float("inf")
    best_n = None

    # ---------------------------------------------------------------
    # 1️⃣  Broad exploration on moderate lengths
    # ---------------------------------------------------------------
    lengths = [1500, 3000, 6000, 12000]              # cheap but varied
    for n in lengths:
        if time.time() >= deadline:
            break

        candidates = []

        # Analytic families ------------------------------------------------
        candidates.append(_distribution_uniform(n))
        candidates.append(_distribution_arcsine(n))
        candidates.append(_distribution_triangular(n))
        candidates.append(_distribution_exponential(n, lam=0.002))
        candidates.append(_distribution_exponential(n, lam=0.02))

        for p in (0.01, 0.02, 0.04, 0.08, 0.12, 0.2, 0.4, 0.6, 1.0, 1.5, 2.0, 3.0):
            candidates.append(_distribution_beta_sym(n, p))

        for pw in (0.5, 1.0, 2.0, 4.0, 6.0, 8.0, 12.0):
            candidates.append(_distribution_sine_power(n, pw))

        for coeff in (0.3, 0.5, 0.7, 0.9):
            candidates.append(_distribution_cosine_window(n, coeff))

        for sigma in (0.10, 0.15, 0.20, 0.30, 0.40):
            candidates.append(_distribution_gaussian(n, sigma))

        # Random Dirichlet draws -----------------------------------------
        for alpha in (0.1, 0.2, 0.4, 0.8, 1.5):
            candidates.append(np.random.dirichlet(np.full(n, alpha)))

        # Sparse start ----------------------------------------------------
        sparse = np.zeros(n, dtype=np.float64)
        k = max(2, n // 250)                     # a few hundred non‑zeros
        idx = np.random.choice(n, k, replace=False)
        sparse[idx] = np.random.rand(k)
        sparse /= sparse.sum()
        candidates.append(sparse)

        # Evaluate all candidates (already sum‑to‑1)
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] n={n}  new init best {sc:.6f}")

    # ---------------------------------------------------------------
    # 2️⃣  Re‑use a previously discovered vector, if available
    # ---------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score, best_seq, best_n = sc, prev.copy(), len(prev)
                print(f"[{time.time():.1f}] Re‑using previous best {sc:.6f}")
    except Exception:
        pass

    # ---------------------------------------------------------------
    # 3️⃣  Fallback (should never happen)
    # ---------------------------------------------------------------
    if best_seq is None:
        best_n = 4000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ---------------------------------------------------------------
    # 4️⃣  Upscale to a finer grid (≈40 k) – the objective is scale‑invariant
    # ---------------------------------------------------------------
    upscale_n = 40000
    if best_n < upscale_n:
        # linear interpolation on the cumulative distribution ensures monotonicity
        x_old = np.linspace(0.0, 1.0, best_n, endpoint=False) + 0.5 / best_n
        x_new = np.linspace(0.0, 1.0, upscale_n, endpoint=False) + 0.5 / upscale_n
        best_seq = np.interp(x_new, x_old, best_seq)
        best_seq = np.maximum(best_seq, eps)
        best_seq /= best_seq.sum()
        best_n = upscale_n
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] Upscaled to n={best_n}, score {best_score:.6f}")

    # ---------------------------------------------------------------
    # 5️⃣  Smooth‑max mirror descent (entropy mirror)
    # ---------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]

    # Determine remaining budget and allocate it to the mirror phase
    time_left = deadline - time.time()
    # Aim for around 30 k iterations but never exceed the budget.
    total_iters = min(30000, int(time_left * 30))   # rough 30 it/s on modest hardware
    if total_iters < 1000:
        total_iters = int(time_left * 60) + 200   # very short fallback

    # Annealing schedule (geometric)
    alpha_start, alpha_end = 1.0, 4.0e5
    lr_start, lr_end = 0.4, 0.001

    print(f"[{time.time():.1f}] Mirror‑descent: {total_iters} iters, n={n}")

    for it in range(total_iters):
        if time.time() >= deadline:
            break
        t = it / total_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t

        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # soft‑max weights (stable)
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth‑max (see docstring)
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # exponentiated‑gradient step on the simplex
        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()

        if (it % 2500 == 0) or (it == total_iters - 1):
            cur_sc = _max_convolution_score(a)
            if cur_sc < best_score - 1e-12:
                best_score, best_seq = cur_sc, a.copy()
                print(f"[{time.time():.1f}] mirror it={it} new best {cur_sc:.6f}")

    # ---------------------------------------------------------------
    # 6️⃣  Adam‑style sub‑gradient polishing (exact max)
    # ---------------------------------------------------------------
    a = best_seq.copy()
    velocity = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        # active set (allow a tiny tolerance)
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0               # sub‑gradient uses unit weight on active shifts
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam momentum update
        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** it)

        # back‑tracking line search (start from step0, halve until improvement)
        step = step0
        while True:
            a_candidate = a - step * v_hat
            a_candidate = _project_onto_simplex(a_candidate, 1.0)
            a_candidate = np.maximum(a_candidate, eps)
            a_candidate /= a_candidate.sum()
            cand_sc = _max_convolution_score(a_candidate)
            if cand_sc < best_score - 1e-12 or step < 1e-8:
                break
            step *= 0.5

        a = a_candidate
        if cand_sc < best_score - 1e-12:
            best_score, best_seq = cand_sc, a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {cand_sc:.6f}")

        # occasional jitter to keep the dynamics alive
        if it % 1500 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    # ---------------------------------------------------------------
    # 7️⃣  Stochastic hill‑climber (mass‑transfer) for final polishing
    # ---------------------------------------------------------------
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    while time.time() < deadline:
        it += 1
        # identify the shift that currently attains the maximum
        k = int(conv.argmax())
        # all index pairs (i,j) with i+j = k
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # pick the pair contributing the largest product (greedy)
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod, best_i, best_j = prod, i, j

        # move a tiny amount from the larger entry to a random target
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue
        # limit the transferred mass (5 % of the source, capped at 0.01)
        max_delta = min(a[source] * 0.05, 0.01)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M
            best_seq = a.copy()
            best_score = 2.0 * n * M
            if it % 200 == 0:
                print(f"[{time.time():.1f}] hill it={it}  new best {best_score:.6f}")

        # occasional small jitter
        if it % 4000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] hill jitter at it {it}")

    final_score = 2.0 * n * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
