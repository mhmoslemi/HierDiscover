def _interpolate_seq(seq, new_n):
    old_n = len(seq)
    x_old = np.linspace(0, 1, old_n, endpoint=False) + 0.5/old_n
    x_new = np.linspace(0, 1, new_n, endpoint=False) + 0.5/new_n
    new_seq = np.interp(x_new, x_old, seq, left=0, right=0)  # but seq is zero beyond boundaries; while interpolation may extrapolate.
    # Then ensure nonnegative and renormalize.
    new_seq = np.maximum(new_seq, 0)
    if new_seq.sum() == 0:
        new_seq = np.full(new_n, 1.0 / new_n)
    else:
        new_seq /= new_seq.sum()
    return new_seq
```

def _resample_seq(seq, new_n):
    old_n = len(seq)
    factor = new_n // old_n
    remainder = new_n % old_n
    seq_rep = np.repeat(seq, factor)
    if remainder > 0:
        seq_rep = np.concatenate([seq_rep, seq[:remainder]])
    # Renormalize
    seq_rep = np.maximum(seq_rep, 0)
    seq_rep /= seq_rep.sum()
    return seq_rep
```

evaluate_sequence().  The algorithm proceeds in three phases:

1. **Broad exploration** – for a few moderate lengths we generate many
   analytic families (uniform, arcsine, triangular, exponential, symmetric
   beta, sine‑power, cosine‑window, Gaussian) together with random Dirichlet
   draws and a sparse “spike” pattern.  The best candidate (normalised to
   sum = 1) is retained.

2. **Multi‑scale refinement** – the current best shape is repeatedly
   up‑sampled to larger lengths (1024 → 2048 → 4096 → …).  For each length we
   run two optimisation stages:
   * a smooth‑max mirror‑descent (exponentiated‑gradient) with an annealed
     temperature α that drives the smoothed objective towards the true max,
   * an Adam‑style projected sub‑gradient descent with back‑tracking line‑search.
   Both stages respect the simplex constraint via Euclidean projection.

3. **Hill‑climber polishing** – a simple mass‑transfer local search moves a
   small amount of probability mass from the index contributing most to the
   current maximal convolution towards a random target; moves are kept if they
   strictly lower the maximal self‑convolution.  Occasional jitter helps to
   escape shallow plateaus.

# Simplex projection (Euclidean)
def _project_onto_simplex(v, s=1.0):
    """Project a vector onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# FFT‑based convolutions (real)
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Return the value of evaluate_sequence() for a vector already
    normalised to sum = 1.  This is 2·n·max(a*a).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]

    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals

    idx = np.arange(n, dtype=np.float64)

    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n

    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

    if n == 1:

    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

    """Discrete Gaussian centred at (n‑1)/2 with relative width sigma."""
    x = np.arange(n, dtype=np.float64)

    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _resample_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Upsample a probability vector by nearest‑neighbour repetition.
    The result is re‑normalised to sum = 1.
    """
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    factor = new_n // old_n
    remainder = new_n % old_n
    up = np.repeat(seq, factor)
    if remainder:
        up = np.concatenate([up, seq[:remainder]])
    up = np.maximum(up, 0.0)
    up_sum = up.sum()
    if up_sum == 0:
        up = np.full(new_n, 1.0 / new_n, dtype=np.float64)
    else:
        up /= up_sum
    return up

                conv = _fft_self_convolve(a)
                M = new_M
                best_seq = a.copy()
                best_score = 2.0 * n * M
                best_n = n
                if hill_it % 200 == 0:
                    print(f"[{time.time():.1f}] hill n={n} it={hill_it} new best {best_score:.6f}")

            if hill_it % 2000 == 0:
                # small random jitter to keep dynamics alive
                jitter = np.random.randn(n) * 1e-5
                a = np.maximum(a + jitter, eps)
                a /= a.sum()
                conv = _fft_self_convolve(a)
                M = conv.max()

        # sync best_seq after this scale
        best_seq = a.copy()
        best_n = n

    # ------------------------------------------------------------------
    final_score = _max_convolution_score(best_seq)
    print(f"[{time.time():.1f}] Finished – best score ≈ {final_score:.6f} (length={best_n})")
    # Convert to plain Python list as required by the evaluation harness
    return best_seq.tolist()
