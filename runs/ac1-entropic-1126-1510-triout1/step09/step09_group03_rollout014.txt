"""
search for a non‑negative step function that (approximately) minimises
evaluate_sequence.  The algorithm

• samples a handful of lengths (2000–12000),
• builds several cheap seeds (arcsine, random Dirichlet, symmetric β‑law),
• then refines each seed with a continuation of smooth “log‑sum‑exp’’
  optimisation.  The smooth objective  f(a)=2·n·(log‑sum‑exp(β·(a*a))) / β
  approximates the true max‑convolution; we minimise it with L‑BFGS‑B
  (variables are logits, so the simplex constraint is satisfied automatically).
  The β parameter is increased in stages (5 → 15 → 30) to approach the true
  max more closely.
• the best refined sequence from all lengths is returned.  The routine
  respects the 1000 s budget and prints simple progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Fast surrogate 2·n·max(a*a) (exact when sum(a)=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families (used as cheap seeds)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Beta(p,p) on the discrete grid {1,…,n}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with concentration ``alpha``."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Smooth log‑sum‑exp max‑convolution objective (gradient in log‑space)
def _smooth_obj_and_grad_logits(
    logits: np.ndarray, n: int, beta: float
) -> tuple[float, np.ndarray]:
    """
    Returns (objective, gradient) for the smooth max‑convolution
    surrogate with temperature 1/β.
    Variables are the logits (unconstrained); the probability vector is
    a = softmax(logits).
    """
    # a = exp(logits) / sum(exp(logits))
    exp_x = np.exp(logits - np.max(logits))          # stabilise
    a = exp_x / exp_x.sum()

    # convolution a*a
    b = _fft_self_convolve(a)                       # length 2n‑1

    # stable log‑sum‑exp
    b_max = b.max()
    exp_term = np.exp(beta * (b - b_max))
    sum_exp = exp_term.sum()
    smooth_max = b_max + np.log(sum_exp) / beta
    obj = 2.0 * n * smooth_max

    # softmax weights for the gradient (∂smooth_max/∂b = w)
    w = exp_term / sum_exp                           # length 2n‑1, sums to 1

    # gradient w.r.t. a : 2 * conv_rev – diagonal correction
    conv_rev = _fft_convolve(a[::-1], w)             # length 2n‑1
    grad_a = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for a_i² terms (i such that 2*i is a valid index)
    idx = np.arange(n)
    diag_mask = (2 * idx < w.size)
    grad_a[diag_mask] -= a[diag_mask] * w[2 * idx[diag_mask]]

    # convert gradient to logits (chain rule for a = softmax(logits))
    avg = np.dot(a, grad_a)                          # <a,∇ₐf>
    grad_logits = a * (grad_a - avg)                 # ∂f/∂x_j = a_j(∂f/∂a_j‑⟨a,∇ₐf⟩)

    return obj, grad_logits


def _l_bfgs_optimize(
    a0: np.ndarray, n: int, beta: float, maxfun: int, deadline: float
) -> np.ndarray:
    """
    L‑BFGS‑B optimisation of the smooth surrogate for a single β.
    Returns the refined probability vector (sum = 1).
    """
    from scipy.optimize import minimize

    # start from logits = log a0  (add a tiny constant to avoid log(0))
    eps = 1e-16
    logits0 = np.log(np.maximum(a0, eps))

    # combine objective and gradient in one callable (required by scipy)
    def fun_and_grad(x):
        obj, grad = _smooth_obj_and_grad_logits(x, n, beta)
        return obj, grad

    options = {
        "maxfun": maxfun,
        "maxiter": maxfun,
        "disp": False,
    }

    # run L‑BFGS‑B; it respects the iteration limits we give.
    res = minimize(fun_and_grad, logits0, method="L-BFGS-B", jac=True, options=options)

    # recover a from the final logits
    exp_x = np.exp(res.x - np.max(res.x))
    a = exp_x / exp_x.sum()
    return a


def _refine_sequence(a0: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Continuation scheme: smooth optimisation with β = 5 → 15 → 30,
    followed by a short exact‑subgradient Adam run.
    Returns the best vector (sum = 1) and its surrogate score.
    """
    betas = [5.0, 15.0, 30.0]
    a = a0.copy()

    for beta in betas:
        if time.time() >= deadline:
            break
        # allocate a modest budget to each β‑stage (max 200 function evals)
        maxfun = 200
        a = _l_bfgs_optimize(a, n, beta, maxfun, deadline)

    # final polishing with the exact sub‑gradient (few hundred Adam steps)
    if time.time() < deadline:
        a, _ = _adam_descent(a, deadline, step0=0.12, max_iters=200_000)

    score = _max_convolution_score(a)
    return a, score


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.15,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 200_000):
    """Adam on the true sub‑gradient (max‑convolution)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) on the simplex (∑a=1).
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices (self‑product terms)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects a 1000 s budget and returns the best list found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # a set of lengths to explore (keep the work‑load moderate)
    lengths = [2000, 3000, 5000, 8000, 12000]
    lengths = [ln for ln in lengths if 2000 <= ln <= 12000]

    print(f"[{time.time():.1f}] planning to test {len(lengths)} lengths")

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] ---- length {n} ----")

        # ---- 1) build cheap seeds -------------------------------------------------
        seeds = []

        # deterministic analytic families
        seeds.append(_arcsine_distribution(n))
        seeds.append(_beta_symmetric_distribution(n, p=0.5))   # arcsine already, but keep
        seeds.append(_beta_symmetric_distribution(n, p=1.0))   # uniform
        seeds.append(_beta_symmetric_distribution(n, p=2.0))   # bell‑shaped
        # a few random Dirichlet draws
        for alpha in (0.3, 0.7, 1.5, 3.0):
            seeds.append(_random_dirichlet_alpha(n, alpha))

        # ------------------------------------------------------------------
        # 2) refine each seed
        for idx, seed_vec in enumerate(seeds):
            if time.time() >= deadline:
                break
            # project onto simplex just in case
            seed_vec = _project_onto_simplex(seed_vec, s=1.0)

            a_refined, sc = _refine_sequence(seed_vec, n, deadline)

            if sc < best_score - 1e-14:
                best_score = sc
                best_seq = a_refined.copy()
                print(
                    f"[{time.time():.1f}] n={n} seed{idx+1}/{len(seeds)} "
                    f"new best surrogate={best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 3) final polishing with a very light Frank‑Wolfe / jitter loop
    if best_seq is not None:
        a = best_seq.copy()
        cur_score = best_score
        fw_iter = 0
        while time.time() < deadline and fw_iter < 30_000:
            fw_iter += 1
            grad, _, _ = _subgrad(a)
            i_src = int(np.argmax(grad))
            i_dst = int(np.argmin(grad))
            if i_src == i_dst:
                break
            step = min(a[i_src], 1.0 - a[i_dst])
            if step < 1e-15:
                break
            a_new = a.copy()
            a_new[i_src] -= step
            a_new[i_dst] += step
            new_score = _max_convolution_score(a_new)
            if new_score < cur_score - 1e-13:
                a = a_new
                cur_score = new_score
                if cur_score < best_score - 1e-14:
                    best_seq = a.copy()
                    best_score = cur_score
                    print(
                        f"[{time.time():.1f}] FW iter={fw_iter} improved best={best_score:.6f}"
                    )
                continue
            # occasional tiny jitter to escape plateaus
            if fw_iter % 5000 == 0 and np.random.rand() < 0.2:
                jitter = np.random.randn(a.size) * 1e-5
                a = _project_onto_simplex(a + jitter, s=1.0)
                cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    if best_seq is None:
        # as a fallback return a simple uniform vector (should never happen)
        best_seq = np.full(2000, 1.0 / 2000, dtype=np.float64)

    print(
        f"Search finished – best surrogate score {best_score:.6f} "
        f"(length={len(best_seq)})"
    )
    return best_seq.tolist()
