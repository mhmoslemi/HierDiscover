def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact objective (2·n·max(a*a)) when sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


def _smooth_grad(a: np.ndarray, eps: float) -> np.ndarray:
    """Gradient of log‑sum‑exp surrogate of max(a*a)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    maxc = conv.max()
    w = np.exp((conv - maxc) / eps)
    w_sum = w.sum()
    w /= w_sum
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    diag_indices = np.arange(conv.size)[(np.arange(conv.size) % 2) == 0] // 2
    grad[diag_indices] -= w[2 * diag_indices] * a[diag_indices]
    return grad

def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 300_000,
                          eta0: float = 0.9) -> tuple[np.ndarray, float]:
    """Mirror descent with KL (exponentiated‑gradient)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 300_000,
                  step0: float = 0.25,
                  beta1: float = 0.9, beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _smooth_adam_descent(a0: np.ndarray, deadline: float,

def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))

def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_w = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    log_w = np.clip(log_w, -700, 700)
    w = np.exp(log_w)
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / n)
    return w / s


def _power_sine_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n) ** p
    return vals / vals.sum()


def _poly_power_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = (i * (n - 1 - i)) ** p
    return vals / vals.sum()


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _gaussian_distribution(n: int, sigma_frac: float = 0.30) -> np.ndarray:
    sigma = sigma_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = vals.sum()
    if s == 0.0:
        return np.full_like(vals, 1.0 / n)
    return vals / s


def _cauchy_distribution(n: int, scale_frac: float = 0.30) -> np.ndarray:
    scale = scale_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = 1.0 / (1.0 + ((i - centre) / scale) ** 2)
    return vals / vals.sum()


def _semicircle_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    vals = np.sqrt(np.clip(1.0 - x * x, 0.0, None))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:

def _cosine_distribution(n: int) -> np.ndarray:

def _uniform_distribution(n: int) -> np.ndarray:

def _random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    x = np.random.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:

    scale_fracs = [0.12, 0.20, 0.30, 0.40]
    dirichlet_alphas = [0.02, 0.05, 0.1, 0.2, 0.5, 0.8, 1.0]

    for n in lengths:
        if time.time() >= sample_deadline:
            break

        # --------------------------------------------------------------
        # Analytic families (few per length for speed)
        fams = [
            (_arcsine_distribution, "arcsine"),
            (_uniform_distribution, "uniform"),
            (_triangular_distribution, "triangular"),
            (_sine_distribution, "sine"),
            (_cosine_distribution, "cosine"),
            (_semicircle_distribution, "semicircle"),
        ]
        for gen, name in fams:
            a = gen(n)
            score = _max_convolution_score(a)
            candidates.append((score, a))
            print(f"[{time.time():.1f}] n={n:5d} {name:10s} score={score:.6f}")

        # --------------------------------------------------------------
        # Beta family (symmetric & asymmetric)
        for (alpha, beta) in beta_params:
            a = _beta_distribution(n, alpha, beta)
            score = _max_convolution_score(a)
            candidates.append((score, a))
        # --------------------------------------------------------------
        # Power‑sine family
        for p in power_sine_ps:
            a = _power_sine_distribution(n, p)
            score = _max_convolution_score(a)
            candidates.append((score, a))
        # --------------------------------------------------------------
        # Polynomial‑power family
        for p in poly_ps:
            a = _poly_power_distribution(n, p)
            score = _max_convolution_score(a)
            candidates.append((score, a))
        # --------------------------------------------------------------
        # Exponential
        for lam in exp_lams:
            a = _exponential_distribution(n, lam=lam)
            score = _max_convolution_score(a)
            candidates.append((score, a))
        # --------------------------------------------------------------
        # Gaussian
        for sf in sigma_fracs:
            a = _gaussian_distribution(n, sigma_frac=sf)
            score = _max_convolution_score(a)
            candidates.append((score, a))
        # --------------------------------------------------------------
        # Cauchy
        for sf in scale_fracs:
            a = _cauchy_distribution(n, scale_frac=sf)
            score = _max_convolution_score(a)
            candidates.append((score, a))
        # --------------------------------------------------------------
        # Random Dirichlet draws
        for alpha in dirichlet_alphas:
            for _ in range(2):
                a = _random_dirichlet(n, alpha=alpha)
                score = _max_convolution_score(a)
                candidates.append((score, a))

        # --------------------------------------------------------------
        # Small jittered copies of a few families (helps escape plateaus)
        for base_gen in [_arcsine_distribution, _triangular_distribution,
                         _uniform_distribution]:
            a = base_gen(n)
            jitter = np.random.randn(n) * 1e-4
            a = np.clip(a + jitter, 0, None)
            a = _project_onto_simplex(a, s=1.0)
            score = _max_convolution_score(a)
            candidates.append((score, a))

        # --------------------------------------------------------------
        # Insert historic best (if any) for this length
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
            if prev.size == n:
                prev = np.clip(prev, 0, None)
                if prev.sum() == 0.0:
                    prev = _uniform_distribution(n)
                else:
                    prev = prev / prev.sum()
                score = _max_convolution_score(prev)
                candidates.append((score, prev))
                print(f"[{time.time():.1f}] n={n:5d} previous best score={score:.6f}")
        except Exception:
            pass

    # Keep only the top candidates (by cheap exact score)
    candidates.sort(key=lambda x: x[0])
    top_k = 60
    top_candidates = candidates[:top_k]
    print(f"Stage 1 done – retained {len(top_candidates)} seeds (best score={top_candidates[0][0]:.6f})")

    # ------------------------------------------------------------------
    # Stage 2 – heavy refinement (≈55 % of time)
    best_seq = top_candidates[0][1].copy()
    best_score = top_candidates[0][0]

    for idx, (seed_seq, _) in enumerate(top_candidates):
        if time.time() >= deadline:
            break

        # Exponentiated‑gradient
        seq, sc = _exp_gradient_descent(seed_seq, deadline,
                                        max_iters=250_000, eta0=0.9)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} exp‑grad → {sc:.6f}")

        # Adam
        seq, sc = _adam_descent(seed_seq, deadline,
                                 max_iters=250_000, step0=0.25)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} Adam → {sc:.6f}")

        # Smooth‑Adam
        seq, sc = _smooth_adam_descent(seed_seq, deadline,
                                        max_iters=180_000,
                                        eps0=1e-3, step0=0.20)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} smooth‑Adam → {sc:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final polishing (≈15 % of time)
    # Sub‑gradient descent on the current champion
    subgrad_deadline = deadline - 15.0
    if time.time() < subgrad_deadline:
        seq, sc = _subgradient_descent(best_seq, subgrad_deadline,
                                       max_iters=2_000_000, eta0=0.8)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] sub‑gradient descent → {sc:.6f}")

    # Gradient‑guided hill climbing
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    iteration = 0
    max_transfer = 0.025  # starting transfer magnitude (fraction of total mass)

    while time.time() < deadline:
        iteration += 1
        # geometric decay of transfer magnitude
        transfer = max_transfer * (0.5 ** (iteration / 8000.0))

        # pick donor/receiver indices
        if iteration % 9 == 0:
            grad, _, _ = _subgrad(a)
            i = int(np.argmax(grad))
            j = int(np.argmin(grad))
            if i == j:
                i = np.random.randint(n)
                j = (i + np.random.randint(1, n)) % n
        else:
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} new best {best_score:.6f}")

        # occasional random jitter to shake out flat regions
        if iteration % 6_000 == 0 and np.random.rand() < 0.15:
            jitter = np.random.randn(n) * 5e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    print(f"Finished – best cheap score={best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
