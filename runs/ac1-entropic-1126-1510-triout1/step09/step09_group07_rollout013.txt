#!/usr/bin/env python
"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1️⃣  **Coarse seeding** – generate a handful of analytic candidates
    (arcsine, smoothed arcsine, power‑law, symmetric Beta and a few
    random Dirichlet draws).  All candidates are normalised to sum = 1.

2️⃣  **LP‑based refinement** – given a current distribution *a*,
    solve the linear programme

        maximise    Σ g_i
        subject to  g_i ≥ 0
                    (a * g)_k ≤ M   for every convolution index k
        where M = max_k (a * a)_k.

    The optimal *g* is then scaled so that its convolution with *a*
    attains the bound *M* again, renormalised to the simplex and mixed
    with *a* (convex combination with a decreasing step‑size).  This
    is an implementation of the “partner LP” described in the problem
    statement and drives the maximum autocorrelation down much faster
    than a pure gradient descent.

3️⃣  **Heavy gradient polishing** – the best LP‑refined seed is handed
    to a suite of first‑order methods (exponentiated‑gradient,
    Adam, smooth‑max continuation and a heavy‑ball sub‑gradient) that
    respect the remaining time budget.

4️⃣  **Fine‑grained hill‑climbing** – tiny random mass transfers and
    occasional jitter finish the search, ensuring that any last marginal
    improvement is captured before the deadline.

All sequences are non‑negative, projected onto the probability simplex
(after every operation) and finally returned as a plain ``list`` ready
for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy import sparse  # keep the import for compatibility with previous code

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap surrogate = 2·n·max(a*a) assuming Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (k = 2*i)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
# families of probability vectors (all return a vector that sums to 1)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _smoothed_arcsine_distribution(n: int, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5 + eps) * (n - i - 0.5 + eps))
    return vals / vals.sum()


def _product_power_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-alpha)
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, p: float, q: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    w = np.where(np.isfinite(w), w, 0.0)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _random_dirichlet(n: int) -> np.ndarray:
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()

# optimisation primitives -------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 120_000, eta0: float = 1.0):
    """Exponentiated‑gradient (mirror descent) on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = _project_onto_simplex(a, 1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_a, best_score = a.copy(), cur
    return best_a, best_score


def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 120_000, step0: float = 0.3,
                  beta1: float = 0.9, beta2: float = 0.999):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, 1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_a, best_score = a.copy(), cur
    return best_a, best_score


def _smooth_gradient(a: np.ndarray, gamma: float):
    """Weighted gradient of the soft‑max of a*a."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    b_max = b.max()
    w = np.exp(gamma * (b - b_max))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    diag_w = w[2 * np.arange(n)]
    grad -= a * diag_w
    return grad, b_max


def _smooth_gradient_descent(a0: np.ndarray, deadline: float,
                             max_iters: int = 200_000,
                             eta0: float = 0.8,
                             gamma0: float = 2.0,
                             gamma_factor: float = 1.005):
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    gamma = gamma0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _ = _smooth_gradient(a, gamma)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, 1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_a, best_score = a.copy(), cur
        gamma *= gamma_factor
    return best_a, best_score


def _heavy_ball_subgradient_descent(a0: np.ndarray, deadline: float,
                                   max_iters: int = 200_000,
                                   eta0: float = 0.6, momentum: float = 0.6):
    a_prev = a0.copy()
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * grad + momentum * (a - a_prev)
        a_new = _project_onto_simplex(a_new, 1.0)
        a_prev, a = a, a_new
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_a, best_score = a.copy(), cur
    return best_a, best_score

# LP‑based refinement ----------------------------------------------------
def _lp_refine(a0: np.ndarray, iters: int, deadline: float,
               t0: float = 0.4) -> np.ndarray:
    """
    Perform a few LP‑based partner steps.
    At each iteration we solve:
        maximise Σ g_i
        s.t.   g_i ≥ 0,
               (a * g)_k ≤ M  (M = max_k (a*a)_k).

    The resulting g is scaled so that (a*g) attains the bound M,
    renormalised to the simplex and mixed into a.
    """
    a = a0.copy()
    n = a.shape[0]
    rows = 2 * n - 1

    while time.time() < deadline and iters > 0:
        # current bound
        b = _fft_self_convolve(a)
        M = b.max()

        # build the constraint matrix A_ub (shape: rows × n)
        # column j contributes a to rows j … j+n‑1
        A = np.zeros((rows, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = a

        # linear programme (maximise sum(g)  <=> minimise -sum(g))
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(rows, M),
                      bounds=bounds, method='highs')
        if not res.success:
            break
        g0 = res.x

        # scale g0 so that its convolution with a hits the bound M again
        conv_ag0 = _fft_convolve(a, g0)
        max_ag0 = conv_ag0.max()
        if max_ag0 <= 0:
            break
        r = M / max_ag0
        g = g0 * r

        # normalise to the simplex (sum = 1)
        s = g.sum()
        if s == 0:
            break
        g = g / s

        # convex combination with a (step size decays)
        t = t0 * (0.95 ** (iters - 1))
        a = (1 - t) * a + t * g
        a = _project_onto_simplex(a, 1.0)

        iters -= 1

    return a
