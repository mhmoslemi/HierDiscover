"""
Propose a low‑score non‑negative step function.

Algorithm
---------
1. Choose a set of lengths around a “base” size (≈ 5 000).  
   For each length generate a diverse pool of candidate shapes:
   * analytic families (uniform, arcsine, triangular, beta, exponential,
     power‑law, double‑bump, cosine‑bump) with many hyper‑parameters,
   * Dirichlet random draws with a wide range of concentration,
   * smoothed versions of the above (small moving‑average kernel),
   * the current global best resized to the new length.
2. Normalise every candidate to ℓ₁‑norm = 1 and compute the exact surrogate
   objective 2·n·max(conv(a,a)).  Keep the top‑k (k≈30) candidates.
3. Refine every seed with three first‑order methods:
   – Adam (projected on the simplex),
   – Exponentiated‑gradient (mirror) descent,
   – Simple projected gradient descent with a back‑tracking line‑search.
   The best point seen during each run is retained.
4. Polish the current global best:
   * Frank‑Wolfe style mass‑transfer steps (move mass from the index
     with the largest sub‑gradient to the smallest, try several
     transfer fractions and keep the best improvement),
   * a longer simulated‑annealing phase with a slower temperature decay.
5. Try to improve further by scaling the best sequence to a few larger
   lengths (1.1×, 1.2×, 1.5×) and re‑optimising it with the same
   three gradient methods.
6. Return the best sequence found before the 1000 s deadline.
All heavy work respects the time budget; simple FFT based
convolutions are used throughout.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """
    Exact surrogate objective when Σa = 1:
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) under Σa = 1.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # gradient of the (non‑smooth) max‑convolution:
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for squared terms on even active positions
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.20,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 400_000):
    """Adam on the simplex (projected). Returns best point seen."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          eta0: float = 0.7,
                          max_iters: int = 300_000):
    """Exponentiated‑gradient (KL‑mirror) descent on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


def _gd_line_search(a0: np.ndarray, deadline: float,
                    step0: float = 1.0,
                    max_iters: int = 250_000):
    """Projected gradient descent with a simple back‑tracking line search."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    step = step0
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        grad, _, _ = _subgrad(a)
        direction = -grad
        improved = False
        # try several shrinking step factors
        for factor in (1.0, 0.5, 0.25, 0.125):
            cand = _project_onto_simplex(a + step * factor * direction, s=1.0)
            cand_score = _max_convolution_score(cand)
            if cand_score < best_score - 1e-15:
                a = cand
                best_a = cand
                best_score = cand_score
                step = step * factor  # keep the successful size
                improved = True
                break
        if not improved:
            step *= 0.5
            if step < 1e-8:
                break
    return best_a, best_score

def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with concentration ``alpha``."""
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0:
        return np.full_like(x, 1.0 / n)
    return x / x.sum()

# Analytic families (all return a normalised vector – ℓ₁‑norm = 1)
def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _double_bump_distribution(n: int,
                              block_frac: float = 0.25,
                              gap_frac: float = 0.1) -> np.ndarray:
    """Two equal uniform blocks separated by a gap."""
    total_block = max(2, int(block_frac * n))
    m = total_block // 2
    g = int(gap_frac * n)
    a = np.zeros(n, dtype=np.float64)
    a[:m] = 1.0 / (2 * m)
    start2 = m + g
    if start2 + m > n:
        start2 = n - m
    a[start2:start2 + m] = 1.0 / (2 * m)
    return a


def _cosine_bump_distribution(n: int,
                              block_frac: float = 0.3,
                              gap_frac: float = 0.1) -> np.ndarray:
    """Two cosine‑shaped bumps (smooth edges)."""
    total_block = max(2, int(block_frac * n))
    m = total_block // 2
    g = int(gap_frac * n)
    a = np.zeros(n, dtype=np.float64)
    # first bump (cosine half‑wave)
    t = np.arange(m, dtype=np.float64) / (m - 1)
    bump = 0.5 * (1 - np.cos(np.pi * t))
    a[:m] = bump / bump.sum() / 2.0
    # second bump
    start2 = m + g
    if start2 + m > n:
        start2 = n - m
    a[start2:start2 + m] = bump[::-1] / bump.sum() / 2.0
    return a


def _smooth_distribution(base: np.ndarray, width: int = 3) -> np.ndarray:
    """Simple moving‑average smoothing (width must be odd)."""
    if width <= 1:
        return base.copy()
    kernel = np.ones(width, dtype=np.float64) / width
    sm = np.convolve(base, kernel, mode='same')
    sm = np.maximum(sm, 0.0)
    if sm.sum() == 0:
        return np.full_like(base, 1.0 / base.size)
    return sm / sm.sum()


def _resize_sequence(a: np.ndarray, m: int) -> np.ndarray:
    """Linear interpolation resizing to length ``m`` (preserves positivity)."""
    if a.size == m:
        return a.copy()
    x_old = np.linspace(0, 1, a.size)
    x_new = np.linspace(0, 1, m)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0:
        a_new = np.full_like(a_new, 1.0 / m)
    else:
        a_new = a_new / a_new.sum()
    return a_new

    a = best_seq.copy()
    cur_score = best_score
    temp_start = 0.04
    temp_end = 0.0008
    anneal_iter = 0
    while time.time() < deadline and anneal_iter < 300_000:
        anneal_iter += 1
        # linear cooling schedule
        fraction = max(0.0, (deadline - time.time()) / (budget_s - 0.5))
        temp = temp_end + (temp_start - temp_end) * fraction

        i = np.random.randint(a.size)
        j = np.random.randint(a.size - 1)
        if j >= i:
            j += 1
        max_delta = min(a[i], 1.0 - a[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * temp
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if cur_score < best_score - 1e-14:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time():.1f}] Anneal iter={anneal_iter} new best {best_score:.6f}")
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_score - cur_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                a = a_new
                cur_score = new_score

    # ------------------------------------------------------------------
    # Try to improve by enlarging the best length and re‑optimising
    base_len = best_seq.shape[0]
    for factor in (1.1, 1.25, 1.5):
        if time.time() >= deadline:
            break
        n_new = int(base_len * factor)
        if n_new > 20000 or n_new < 1500:
            continue
        a_resized = _resize_sequence(best_seq, n_new)
        # short optimisation on the new length
        seq_adam, sc_adam = _adam_descent(a_resized, deadline, max_iters=200_000)
        seq_eg, sc_eg = _exp_gradient_descent(seq_adam, deadline, max_iters=150_000)
        seq_gd, sc_gd = _gd_line_search(seq_eg, deadline, max_iters=100_000)

        cand_seq = seq_gd
        cand_score = _max_convolution_score(cand_seq)
        if cand_score < best_score - 1e-14:
            best_score, best_seq = cand_score, cand_seq.copy()
            print(f"[{time.time():.1f}] up‑scaled to n={n_new} → new best {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final surrogate best: {best_score:.6f} (length={len(best_seq)})")
    # Return plain Python list – evaluate_sequence will normalise again if needed.
    return best_seq.tolist()
