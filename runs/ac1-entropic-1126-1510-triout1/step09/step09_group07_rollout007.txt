Search for a near‑optimal non‑negative step function.

The algorithm consists of three stages:

1️⃣  **Broad parameter sweep** – many analytic families (arcsine,
    smoothed arcsine, product‑power, symmetric / asymmetric Beta,
    exponential tail, triangular, sine) are evaluated on several
    lengths (≈0.8 × to 2.5 × the length of the previous best).  The
    cheap surrogate 2·n·max(a*a) (identical to the true objective when
    the vector is normalised) filters the best seed.

2️⃣  **Multi‑start non‑smooth optimisation** – from the current best
    seed we repeatedly launch short runs of several first‑order
    methods (exponentiated‑gradient, Adam, sub‑gradient, smoothed‑max
    continuation, heavy‑ball) with different random seeds.  Each run
    respects the remaining time budget and updates the global best.

3️⃣  **Simulated‑annealing hill‑climber** – tiny random mass transfers
    are proposed; moves that improve the cheap score are always accepted,
    worse moves are accepted with a temperature‑controlled probability.
    Temperature decays geometrically, allowing occasional escapes from
    shallow local minima.

All vectors are projected onto the probability simplex after every
update, guaranteeing feasibility for ``evaluate_sequence``.  The final
sequence is returned as a plain ``list[float]`` ready for evaluation.
"""

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# FFT based convolutions (real)
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
# Cheap surrogate (exact when sum=1)
def _max_convolution_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) – the true objective when sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (exact)
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.
    Returns (grad, cur_max, active_set).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with indicator of active positions
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for indices where k = 2*i
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
# Smooth‑max gradient (continuation)
def _smooth_gradient(a: np.ndarray, gamma: float):
    """Weighted gradient of the soft‑max of a*a (γ controls smoothness)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    b_max = b.max()
    w = np.exp(gamma * (b - b_max))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    # diagonal terms (k = 2*i)
    diag_w = w[2 * np.arange(n)]
    grad -= a * diag_w
    return grad, b_max

# Families of probability vectors (analytic shapes)
def _arcsine_distribution(n: int) -> np.ndarray:
    """Classic arcsine density on a discrete grid."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _smoothed_arcsine_distribution(n: int, eps: float) -> np.ndarray:
    """Arcsine with an additive smoothing ε."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5 + eps) * (n - i - 0.5 + eps))
    return vals / vals.sum()


def _product_power_distribution(n: int, alpha: float) -> np.ndarray:
    """Shape a_i ∝ ((i+½)(n‑i‑½))^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-alpha)
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, p: float, q: float) -> np.ndarray:
    """Beta(p,q) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    w = np.where(np.isfinite(w), w, 0.0)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    return vals / vals.sum()


def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Γ–Dirichlet, α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()

def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 150_000, eta0: float = 0.8):
    """Mirror‑descent (exponentiated‑gradient) on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        a = a / s if s != 0 else np.full_like(a, 1.0 / a.size)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 150_000, step0: float = 0.2,
                  beta1: float = 0.9, beta2: float = 0.999):
    """Adam‑style adaptive gradient optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 400_000, eta0: float = 0.6):
    """Plain sub‑gradient descent with diminishing step size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _smooth_gradient_descent(a0: np.ndarray, deadline: float,
                             max_iters: int = 300_000,
                             eta0: float = 0.7,
                             gamma0: float = 5.0,
                             gamma_factor: float = 1.001):
    """Continuation method: start with a smooth max (γ≈5) and increase γ."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    gamma = gamma0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _ = _smooth_gradient(a, gamma)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score, best_a = cur_score, a.copy()
        gamma *= gamma_factor
    return best_a, best_score


def _heavy_ball_subgradient_descent(a0: np.ndarray, deadline: float,
                                   max_iters: int = 300_000,
                                   eta0: float = 0.6, momentum: float = 0.6):
    """Sub‑gradient descent with a heavy‑ball momentum term."""
    a_prev = a0.copy()
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * grad + momentum * (a - a_prev)
        a_new = _project_onto_simplex(a_new, s=1.0)
        a_prev, a = a, a_new
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

    # Base length – use the previous construction if present
    try:
        base_n = len(height_sequence_1)   # type: ignore[name-defined]
        if base_n < 500:
            base_n = 500
    except Exception:
        base_n = 3500

    # Candidate lengths (still affordable for FFT‑based evaluation)
    factors = (0.8, 0.9, 1.0, 1.1, 1.2, 1.5, 2.0, 2.5)
    lengths = [max(500, min(15000, int(base_n * f))) for f in factors]
    lengths = sorted(set(lengths))

    # --------------------------------------------------------------
    # 1️⃣  Coarse sweep over analytic families
    best_seq = None
    best_score = float('inf')

    # Parameter grids
    arcsine_eps = [0.0] + list(np.geomspace(0.01, 5.0, 7))
    product_alpha = list(np.arange(0.20, 1.01, 0.02))
    beta_p = list(np.arange(0.10, 2.01, 0.07))
    beta_asym = [(p, q) for p in [0.12, 0.30, 0.50, 0.80, 1.20] for q in [0.12, 0.30, 0.50, 0.80, 1.20] if p != q]
    exp_lam = [0.0005, 0.001, 0.002, 0.003, 0.005, 0.010, 0.020]

    for n in lengths:
        if time.time() >= deadline:
            break

        # arcsine & smoothed arcsine
        for eps in arcsine_eps:
            a = _arcsine_distribution(n) if eps == 0.0 else _smoothed_arcsine_distribution(n, eps)
            sc = _max_convolution_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()

        # product‑power family
        for alpha in product_alpha:
            a = _product_power_distribution(n, alpha)
            sc = _max_convolution_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()

        # symmetric Beta
        for p in beta_p:
            a = _beta_symmetric_distribution(n, p)
            sc = _max_convolution_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()

        # asymmetric Beta
        for p, q in beta_asym:
            a = _beta_asymmetric_distribution(n, p, q)
            sc = _max_convolution_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()

        # exponential tails
        for lam in exp_lam:
            a = _exponential_distribution(n, lam)
            sc = _max_convolution_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()

        # triangular distribution
        a = _triangular_distribution(n)
        sc = _max_convolution_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()

        # sine distribution
        a = _sine_distribution(n)
        sc = _max_convolution_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()

        # a few random Dirichlet draws
        for _ in range(8):
            a = _random_dirichlet(n)
            sc = _max_convolution_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()

    # incorporate the previous best, if it beats the current seed
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score, best_seq = sc, prev.copy()
    except Exception:
        pass

    print(f"[{time.time():.1f}] after coarse sweep best cheap={best_score:.9f}")

    # --------------------------------------------------------------
    # 2️⃣  Multi‑start refinement – loop until only a few seconds remain
    while True:
        remaining = deadline - time.time()
        if remaining < 12.0:   # keep a small buffer for final hill‑climb
            break

        # allocate a modest slice of the remaining budget to a single method
        sub_budget = min(30.0, remaining * 0.25)
        sub_deadline = time.time() + sub_budget

        method = np.random.choice(['exp', 'adam', 'subgrad', 'smooth', 'heavy'])
        if method == 'exp':
            seq, sc = _exp_gradient_descent(best_seq, sub_deadline,
                                            max_iters=200_000, eta0=1.0)
        elif method == 'adam':
            seq, sc = _adam_descent(best_seq, sub_deadline,
                                    max_iters=200_000, step0=0.3)
        elif method == 'subgrad':
            seq, sc = _subgradient_descent(best_seq, sub_deadline,
                                           max_iters=300_000, eta0=0.6)
        elif method == 'smooth':
            seq, sc = _smooth_gradient_descent(best_seq, sub_deadline,
                                               max_iters=300_000,
                                               eta0=0.8, gamma0=2.0,
                                               gamma_factor=1.005)
        else:  # heavy‑ball
            seq, sc = _heavy_ball_subgradient_descent(best_seq, sub_deadline,
                                                      max_iters=300_000,
                                                      eta0=0.6, momentum=0.6)

        if sc < best_score - 1e-12:
            best_seq, best_score = seq, sc
            print(f"[{time.time():.1f}] {method} refinement improved to {best_score:.9f}")

    # --------------------------------------------------------------
    # 3️⃣  Simulated‑annealing hill‑climber (tiny random mass transfers)
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    it = 0
    # geometric decay of the transferred amount (starts at ~1 % of total mass)
    max_transfer = 0.01
    # temperature schedule
    T0 = 1e-3
    while time.time() < deadline:
        it += 1
        # temperature slowly decays
        T = T0 * (0.9995 ** it)

        # random source and destination indices
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # propose a small transfer from i to j
        transfer = max_transfer * (0.5 ** (it / 8000.0))
        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score:
            accept = True
        else:
            # probabilistic acceptance of worse moves
            accept = np.random.rand() < np.exp(-(new_score - cur_score) / max(T, 1e-12))

        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_seq, best_score = a_new.copy(), new_score
                print(f"[{time.time():.1f}] SA hill‑climb iter={it} new best {best_score:.9f}")

        # occasional jitter to shake out plateaus
        if it % 5000 == 0 and np.random.rand() < 0.15:
            jitter = np.random.randn(n) * 5e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    print(f"Final cheap score: {best_score:.9f} (length={len(best_seq)})")
    return best_seq.tolist()
