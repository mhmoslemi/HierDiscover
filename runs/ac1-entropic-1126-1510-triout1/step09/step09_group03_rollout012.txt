"""
Search for a non‑negative step function that minimizes ``evaluate_sequence``.

Algorithm
---------
* Choose a handful of lengths (≈2000‑12000) and generate a large, diverse
  pool of candidate sequences from analytic families (exponential,
  power‑law, asymmetric beta, Gaussian‑like, etc.) together with many
  random Dirichlet draws (including very sparse draws).
* Keep the best K seeds (K≈15) according to the exact surrogate
  ``2·n·max(conv(a,a))`` (identical to ``evaluate_sequence`` for ∑a=1).
* Refine every seed with a suite of fast first‑order methods:
  – Adam on the true sub‑gradient (exact max‑convolution).
  – Exponentiated‑gradient (KL‑mirror) descent.
  – Adam on a smooth Lp‑approximation (p=6).
* After the gradient phases, apply a *pairwise mass‑transfer* polishing
  step (a Frank‑Wolfe style line‑search that moves probability mass from
  the index with largest gradient to the one with smallest).
* Finally run a light simulated‑annealing / jitter phase to hop out of
  tiny plateaus.
* All loops respect the overall 1000 s budget; the best sequence found
  (as a plain list) is returned.
"""

# --------------------------------------------------------------
# Basic linear‑algebra utilities
# --------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# FFT‑based convolution helpers (real, zero‑padded)
# --------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# --------------------------------------------------------------
# Objective utilities (exact surrogate)
# --------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Surrogate 2·n·max(conv(a,a)) – identical to ``evaluate_sequence`` for ∑a=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) on the simplex (∑a=1).
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    # weight vector w that encodes the active set
    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of a with w (reversed a) to get gradient
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices (self‑product terms)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# --------------------------------------------------------------
# Optimisers (projected onto the simplex)
# --------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.5,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 400_000):
    """Adam on the true sub‑gradient (max‑convolution)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          eta0: float = 0.7,
                          max_iters: int = 350_000):
    """Exponentiated‑gradient (KL‑mirror) descent."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


def _smooth_obj_and_grad(a: np.ndarray, p: float = 6.0):
    """
    Smooth Lp‑approximation of max(conv(a,a)) and its gradient.
    Returns (value, grad) where value = (∑ conv^p)^{1/p}.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)          # length 2n‑1
    b_pow = b ** p
    S = b_pow.sum()
    if S == 0.0:
        return 0.0, np.zeros_like(a)
    val = S ** (1.0 / p)
    factor = S ** (1.0 / p - 1.0)      # = S^{(1‑p)/p}
    w = factor * (b ** (p - 1))
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return val, grad


def _adam_smooth_descent(a0: np.ndarray, deadline: float,
                         p: float = 6.0,
                         step0: float = 0.5,
                         beta1: float = 0.9,
                         beta2: float = 0.999,
                         max_iters: int = 300_000):
    """Adam on the smoothed Lp objective."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        _, grad = _smooth_obj_and_grad(a, p)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score

def _exponential_asymmetric_distribution(n: int, lam: float = 0.01) -> np.ndarray:
    """a_i ∝ exp(-lam·i)  (mass concentrated on the left)."""
    idx = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * idx)
    vals /= vals.sum()
    return vals


def _gaussian_like_distribution(n: int, sigma: float = 0.2) -> np.ndarray:
    """Discrete Gaussian centred at 0 (strongly left‑biased)."""
    idx = np.arange(n, dtype=np.float64)
    mu = 0.0
    vals = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
    vals /= vals.sum()
    return vals


def _logarithmic_distribution(n: int) -> np.ndarray:
    """a_i ∝ 1/(i+1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = 1.0 / i
    vals /= vals.sum()
    return vals


def _power_law_reverse_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n-i)^{-p}  (decreasing from left to right)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = (n - i + 1) ** (-p)
    vals /= vals.sum()
    return vals


def _mixture_exponential_distribution(n: int, lam1: float, lam2: float, w: float) -> np.ndarray:
    """Mixture w·exp(-lam1·i) + (1‑w)·exp(-lam2·i)."""
    idx = np.arange(n, dtype=np.float64)
    vals = w * np.exp(-lam1 * idx) + (1.0 - w) * np.exp(-lam2 * idx)
    vals /= vals.sum()
    return vals

def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with concentration ``alpha``."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Pairwise mass‑transfer polishing (Frank‑Wolfe style)
# --------------------------------------------------------------
def _pairwise_mass_transfer(a: np.ndarray, cur_score: float,
                           deadline: float, max_iters: int = 30_000):
    """Move mass from the index with largest gradient to the smallest."""
    best_a = a.copy()
    best_score = cur_score
    a_cur = a.copy()
    score_cur = cur_score

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a_cur)
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        max_transfer = min(a_cur[i_src], 1.0 - a_cur[i_dst])
        if max_transfer < 1e-15:
            break
        # try a few fractions of the maximal transfer
        improved = False
        for frac in (1.0, 0.5, 0.25):
            delta = max_transfer * frac
            a_new = a_cur.copy()
            a_new[i_src] -= delta
            a_new[i_dst] += delta
            sc = _max_convolution_score(a_new)
            if sc < score_cur - 1e-13:
                a_cur = a_new
                score_cur = sc
                improved = True
                break
        if improved and score_cur < best_score - 1e-14:
            best_score = score_cur
            best_a = a_cur.copy()
    return best_a, best_score

# Light simulated‑annealing / jitter phase
# --------------------------------------------------------------
def _annealing_phase(a: np.ndarray, cur_score: float,
                     deadline: float, max_iters: int = 200_000):
    """Small temperature‑controlled random mass transfers."""
    a_cur = a.copy()
    score_cur = cur_score
    best_a = a_cur.copy()
    best_score = cur_score

    start = time.time()
    while time.time() < deadline and max_iters > 0:
        max_iters -= 1
        # linear temperature schedule based on remaining time
        remaining = deadline - time.time()
        total = deadline - start
        t_frac = remaining / total
        temp = 0.02 * t_frac + 0.001

        i = np.random.randint(a_cur.size)
        j = np.random.randint(a_cur.size - 1)
        if j >= i:
            j += 1
        max_delta = min(a_cur[i], 1.0 - a_cur[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * temp

        a_new = a_cur.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < score_cur - 1e-15:
            a_cur = a_new
            score_cur = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = a_new.copy()
        else:
            # Metropolis accept
            prob = np.exp(-(new_score - score_cur) / (temp + 1e-12))
            if np.random.rand() < prob:
                a_cur = a_new
                score_cur = new_score
    return best_a, best_score

    sequence found as a plain ``list[float]``.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin

    # ----------------------------------------------------------
    # 0️⃣ Base length – try to beat the previous construction (3268)
    # ----------------------------------------------------------
    base_n = 5000
    prev_seq = globals().get('height_sequence_1')
    if prev_seq is not None:
        try:
            tmp = np.asarray(prev_seq, dtype=np.float64)
            if tmp.ndim == 1 and tmp.size >= 2000:
                base_n = int(tmp.size)
        except Exception:
            pass

    # ----------------------------------------------------------
    # 1️⃣ Lengths to explore (moderate diversity)
    # ----------------------------------------------------------
    lengths = set()
    for scale in (0.8, 0.9, 1.0, 1.1, 1.2):
        n = int(base_n * scale)
        if 2000 <= n <= 12000:
            lengths.add(n)
    # extra random lengths
    while len(lengths) < 9 and time.time() < deadline:
        n = np.random.randint(2000, 12001)
        lengths.add(n)
    lengths = sorted(lengths)

    print(f"[{time.time():.1f}] Exploring {len(lengths)} lengths around base≈{base_n}")

    # ----------------------------------------------------------
    # 2️⃣ Broad candidate generation
    # ----------------------------------------------------------
    candidates = []   # list of (array, surrogate_score)

    # Grids for families
    beta_ps = np.concatenate([
        np.arange(0.05, 0.26, 0.05),
        np.arange(0.3, 0.81, 0.1),
        np.arange(0.9, 2.1, 0.3),
    ])
    power_ps = [0.5, 0.8, 1.0, 1.3, 1.6, 2.0, 3.0]
    exp_lams = [5e-4, 1e-3, 5e-3, 1e-2, 2e-2, 5e-2]
    dirichlet_alphas = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]

    for n in lengths:
        if time.time() >= deadline:
            break

        # ------- deterministic families -------
        fams = [
            (_arcsine_distribution, "arcsine"),
            (_triangular_distribution, "triangular"),
            (_raised_cosine_distribution, "raised_cos1"),
            (_sinusoidal_distribution, "sinus1"),
            (_sinusoidal_distribution, "sinus2"),
            (lambda m: _exponential_distribution(m, lam=0.005), "exp_center"),
        ]
        for gen, name in fams:
            a = gen(n)
            candidates.append((a, _max_convolution_score(a)))
            print(f"[{time.time():.1f}] n={n:<5d} fam={name}")

        # ------- asymmetric exponential -------
        for lam in exp_lams:
            a = _exponential_asymmetric_distribution(n, lam=lam)
            candidates.append((a, _max_convolution_score(a)))
            print(f"[{time.time():.1f}] n={n:<5d} exp_asym(lam={lam:.5f})")

        # ------- power‑law (decreasing) -------
        for p in power_ps:
            a = _power_law_reverse_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))

        # ------- symmetric beta -------
        for p in beta_ps:
            a = _beta_symmetric_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))

        # ------- asymmetric beta (hand‑picked) -------
        for a_alpha, a_beta in [(0.2, 1.5), (0.3, 1.2), (0.5, 2.0), (0.8, 1.0),
                                (1.0, 0.5), (1.2, 0.3)]:
            a = _beta_asymmetric_distribution(n, a_alpha, a_beta)
            candidates.append((a, _max_convolution_score(a)))

        # ------- random asymmetric beta samplings -------
        for _ in range(3):
            a_alpha = 10 ** np.random.uniform(-2, 1)   # 0.01 … 10
            a_beta  = 10 ** np.random.uniform(-2, 1)
            a = _beta_asymmetric_distribution(n, a_alpha, a_beta)
            candidates.append((a, _max_convolution_score(a)))

        # ------- logarithmic & Gaussian‑like -------
        a = _logarithmic_distribution(n)
        candidates.append((a, _max_convolution_score(a)))
        a = _gaussian_like_distribution(n, sigma=0.12)
        candidates.append((a, _max_convolution_score(a)))

        # ------- mixture of two exponentials -------
        for _ in range(4):
            lam1 = 10 ** np.random.uniform(-4, -1)   # 1e‑4 … 0.1
            lam2 = 10 ** np.random.uniform(-4, -1)
            w = np.random.rand()
            a = _mixture_exponential_distribution(n, lam1, lam2, w)
            candidates.append((a, _max_convolution_score(a)))

        # ------- random Dirichlet draws -------
        for alpha in dirichlet_alphas:
            a = _random_dirichlet_alpha(n, alpha)
            candidates.append((a, _max_convolution_score(a)))

    # ------------------------------------------------------------------
    # 3️⃣ Add the previous best if available
    # ------------------------------------------------------------------
    if prev_seq is not None:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.ndim == 1 and arr.size > 0:
                arr = arr / arr.sum()
                candidates.append((arr, _max_convolution_score(arr)))
                print(f"[{time.time():.1f}] added previous best, score={candidates[-1][1]:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 4️⃣ Keep the top K seeds for intensive refinement
    # ------------------------------------------------------------------
    candidates.sort(key=lambda x: x[1])
    top_k = min(20, len(candidates))
    top_seeds = candidates[:top_k]
    print(f"Generated {len(candidates)} candidates, keeping top {top_k} for refinement.")

    best_seq, best_score = top_seeds[0][0].copy(), top_seeds[0][1]
    print(f"[{time.time():.1f}] Initial best surrogate score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ Intensive refinement of each seed
    # ------------------------------------------------------------------
    for idx, (seed_seq, _) in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        n = seed_seq.shape[0]

        # time slice for this seed (≈ equal share of remaining budget)
        remaining = deadline - time.time()
        per_seed_budget = remaining / (top_k - idx) * 0.9   # keep a little spare for polishing
        seed_deadline = time.time() + per_seed_budget

        # ---- Adam on exact sub‑gradient ----
        seq_adam, sc_adam = _adam_descent(seed_seq, seed_deadline,
                                          step0=0.5, max_iters=200_000)
        if sc_adam < best_score - 1e-14:
            best_seq, best_score = seq_adam.copy(), sc_adam
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} Adam improved to {best_score:.6f}")

        # ---- Exponentiated‑gradient ----
        seq_eg, sc_eg = _exp_gradient_descent(seed_seq, seed_deadline,
                                              eta0=0.7, max_iters=150_000)
        if sc_eg < best_score - 1e-14:
            best_seq, best_score = seq_eg.copy(), sc_eg
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} ExpGrad improved to {best_score:.6f}")

        # ---- Adam on smooth Lp (p=6) ----
        seq_smooth, sc_smooth = _adam_smooth_descent(seed_seq, seed_deadline,
                                                      p=6.0, step0=0.5,
                                                      max_iters=120_000)
        if sc_smooth < best_score - 1e-14:
            best_seq, best_score = seq_smooth.copy(), sc_smooth
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} SmoothAdam improved to {best_score:.6f}")

        # ---- Pairwise mass‑transfer (Frank‑Wolfe style) ----
        a = best_seq.copy()
        cur_score = best_score
        a, cur_score = _pairwise_mass_transfer(a, cur_score, seed_deadline,
                                               max_iters=25_000)
        if cur_score < best_score - 1e-14:
            best_seq, best_score = a.copy(), cur_score
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} FW polishing improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣ Global polishing (FW + annealing) on the overall best
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cur_score = best_score

    # Frank‑Wolfe mass‑transfer on the global best
    a, cur_score = _pairwise_mass_transfer(a, cur_score, deadline,
                                           max_iters=80_000)
    if cur_score < best_score - 1e-14:
        best_seq, best_score = a.copy(), cur_score
        print(f"[{time.time():.1f}] Global FW improved to {best_score:.6f}")

    # Final simulated‑annealing / jitter phase
    a, cur_score = _annealing_phase(best_seq, best_score, deadline,
                                    max_iters=180_000)
    if cur_score < best_score - 1e-14:
        best_seq, best_score = a.copy(), cur_score
        print(f"[{time.time():.1f}] Final annealing improved to {best_score:.6f}")

    print(f"Final best surrogate score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
