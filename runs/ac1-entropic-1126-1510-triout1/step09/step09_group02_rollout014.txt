"""
Improved step‑function optimizer.

The idea is to explore a richer family of analytic shapes (e.g. Hann,
Hamming, Blackman, Gaussian and cosine families), generate a moderate
set of random Dirichlet draws and keep the very best seeds.  Each seed
is then refined by a long Adam run on the true sub‑gradient of the
max‑convolution, followed by a continuation Adam run in which the
max‑operator is smoothed with a soft‑max (parameter β grows during the
run).  Finally a Frank–Wolfe style mass‑transfer polishing and a short
simulated‑annealing jitter phase are applied.

All convolutions are performed with real FFTs, the probability vector
is kept on the simplex (projection after each update) and the routine
stops automatically when the 1000 s budget expires.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a: np.ndarray) -> float:
    """
    Exact objective for a simplex point (Σa = 1):
    2·n·max(conv(a,a))  == evaluate_sequence(a).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) under Σa = 1.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # correct the diagonal terms (they appear only once in the max)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active

def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.30,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 1_500_000):
    """Projected Adam on the simplex using the exact sub‑gradient."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


def _softmax_adam_descent(a0: np.ndarray, deadline: float,
                          step0: float = 0.30,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          max_iters: int = 500_000,
                          beta_start: float = 0.5,
                          beta_end: float = 25.0):
    """Adam on a smooth soft‑max approximation of the max‑convolution."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    n = a.shape[0]
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        # linearly increase β
        beta = beta_start + (beta_end - beta_start) * (t / max_iters)

        b = _fft_self_convolve(a)
        max_b = b.max()
        # stable soft‑max
        exp_w = np.exp(beta * (b - max_b))
        sum_exp = exp_w.sum()
        if sum_exp == 0:
            w = np.full_like(b, 1.0 / b.size)
        else:
            w = exp_w / sum_exp

        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score

    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    w /= w.sum()
    return w

            (_triangular_distribution, "triangular"),
            (_hann_window, "hann"),
            (_hamming_window, "hamming"),
            (_blackman_window, "blackman"),
        ]

        for gen, name in fams:
            a = gen(n)
            candidates.append((a, _max_convolution_score(a)))
            print(f"[{time.time():.1f}] n={n:<5d} {name:10s}")

        # exponential tails (already in original but keep more lambdas)
        for lam in lam_vals:
            a = _exponential_distribution(n, lam=lam)
            candidates.append((a, _max_convolution_score(a)))
            print(f"[{time.time():.1f}] n={n:<5d} exp(lam={lam})")

        # Gaussian windows
        for sigma in sigma_fracs:
            a = _gaussian_distribution(n, sigma_frac=sigma)
            candidates.append((a, _max_convolution_score(a)))

        # Cosine families (plus and minus)
        for k in cosine_ks:
            a = _cosine_plus_family(n, k)
            candidates.append((a, _max_convolution_score(a)))
            a = _cosine_minus_family(n, k)
            candidates.append((a, _max_convolution_score(a)))

        # double‑bump with several hyper‑parameters
        for block_frac in (0.15, 0.25, 0.35):
            for gap_frac in (0.05, 0.12, 0.25):
                a = _double_bump_distribution(n, block_frac, gap_frac)
                candidates.append((a, _max_convolution_score(a)))

        # random Dirichlet draws
        for alpha in dir_alpha:
            a = _random_dirichlet_alpha(n, alpha)
            candidates.append((a, _max_convolution_score(a)))

    # ------------------------------------------------------------------
    # Possibly add the previously known best sequence as a seed
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            if prev.sum() == 0:
                prev = np.full_like(prev, 1.0 / prev.size)
            else:
                prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
            print(f"[{time.time():.1f}] added previous best, score={candidates[-1][1]:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Keep the very best seeds for intensive refinement
    candidates.sort(key=lambda x: x[1])
    top_k = min(12, len(candidates))
    top_seeds = candidates[:top_k]
    print(f"Generated {len(candidates)} candidates, keeping top {top_k} for refinement.")

    best_seq = top_seeds[0][0].copy()
    best_score = top_seeds[0][1]
    print(f"Initial best surrogate score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Intensive refinement: Adam + soft‑max Adam
    for idx, (seed_seq, seed_score) in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        n = seed_seq.shape[0]

        # Adam on exact sub‑gradient
        seq_adam, sc_adam = _adam_descent(seed_seq, deadline,
                                          max_iters=800_000)
        if sc_adam < best_score - 1e-14:
            best_score, best_seq = sc_adam, seq_adam.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} Adam improved to {best_score:.6f}")

        # Soft‑max continuation Adam
        seq_sam, sc_sam = _softmax_adam_descent(seq_adam, deadline,
                                                max_iters=300_000,
                                                beta_start=0.5,
                                                beta_end=25.0)
        if sc_sam < best_score - 1e-14:
            best_score, best_seq = sc_sam, seq_sam.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} Soft‑max Adam improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Frank‑Wolfe style mass‑transfer polishing
    a = best_seq.copy()
    cur_score = best_score
    fw_iter = 0
    while time.time() < deadline and fw_iter < 150_000:
        fw_iter += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))          # mass to move away
        i_dst = int(np.argmin(grad))          # mass to add

        if i_src == i_dst:
            break

        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-15:
            break

        # try a few fractions (full, half, quarter)
        improved = False
        for frac in (1.0, 0.5, 0.25):
            delta = max_transfer * frac
            a_new = a.copy()
            a_new[i_src] -= delta
            a_new[i_dst] += delta
            sc = _max_convolution_score(a_new)
            if sc < cur_score - 1e-13:
                a = a_new
                cur_score = sc
                improved = True
                break
        if improved and cur_score < best_score - 1e-14:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time():.1f}] FW iter={fw_iter} new best {best_score:.6f}")

        # occasional random jitter to avoid tiny plateaus
        if fw_iter % 6000 == 0 and np.random.rand() < 0.13:
            jitter = np.random.randn(a.size) * 2e-5
            a = _project_onto_simplex(a + jitter, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    # Simulated‑annealing / jitter phase
    a = best_seq.copy()
    cur_score = best_score
    temp_start = 0.030
    temp_end = 0.0015
    anneal_iter = 0
    while time.time() < deadline and anneal_iter < 250_000:
        anneal_iter += 1
        # linear temperature schedule
        t_rem = deadline - time.time()
        fract = max(0.0, t_rem / (budget_s - 0.5))
        temp = temp_end + (temp_start - temp_end) * fract

        i = np.random.randint(a.size)
        j = np.random.randint(a.size - 1)
        if j >= i:
            j += 1

        max_delta = min(a[i], 1.0 - a[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * temp
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if cur_score < best_score - 1e-14:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time():.1f}] Anneal iter={anneal_iter} new best {best_score:.6f}")
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_score - cur_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                a = a_new
                cur_score = new_score

    # ------------------------------------------------------------------
    print(f"Final best surrogate score: {best_score:.6f} (length={len(best_seq)})")
    # Return a Python list; evaluate_sequence will handle scaling/clipping.
    return best_seq.tolist()
