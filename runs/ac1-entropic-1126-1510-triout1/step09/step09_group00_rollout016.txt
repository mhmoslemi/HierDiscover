"""
Optimiser for the “step‑function” evaluation problem.

The code builds a large pool of analytic seed vectors (uniform,
arcsine, symmetric/asymmetric Beta, power‑law, exponential,
two‑block spikes and random sparse vectors) on several modest
lengths.  For each seed we compute the exact objective
    2·n·max(conv(a,a))      (with ∑a = 1)
and keep the best few seeds.

These seeds are then refined by an accelerated sub‑gradient
descent (exponentiated‑gradient mirror descent) with a larger
initial learning rate and a generous schedule; the remaining
time is split among the top seeds.  Finally a cheap greedy
pairwise‑mass‑transfer walk (simulated‑annealing style) is run
to polish the best solution.

All operations respect the time budget (default 1000 s) and
the result is returned as a plain ``list[float]`` ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (real FFT, length 2·n−1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) for a vector with Σa = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) on the discrete grid."""
    eps = 1e-15
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    x = np.clip(x, eps, 1 - eps)
    w = np.exp((alpha - 1) * np.log(x) + (alpha - 1) * np.log(1 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    eps = 1e-15
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    x = np.clip(x, eps, 1 - eps)
    w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^p  (monotone increasing, p may be negative)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    """a_i ∝ exp(lam·i)  (lam>0 → increasing, lam<0 → decreasing)."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    """Two equal blocks of mass at the two ends."""
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _subgradient_descent_enhanced(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 1.0,
                                 max_iter: int = 20000,
                                 eval_interval: int = 200) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient mirror descent on the true max‑convolution
    objective.  Uses a larger initial learning rate (lr0) than the
    baseline and stops early if the time budget is exceeded.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0

    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        idx = int(conv.argmax())

        # sub‑gradient: 2·a shifted by idx
        ind = idx - np.arange(n)
        mask = (ind >= 0) & (ind < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[ind[mask]]

        # step size η_t = lr0 / sqrt(t)
        eta = lr0 / math.sqrt(t)

        # multiplicative (exponentiated) update & renormalise
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] subgrad it={t} new best {best_score:.6f}")
            else:
                no_improve += 1
            # occasional early‑stop if no improvement for long
            if no_improve >= 5 * eval_interval:
                break

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Randomly move a small amount of mass; keep move only if objective improves."""
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # transfer up to 30 % of the source mass
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")

    return best_seq, best_score

                candidates.append(_distribution_power_increasing(n, p))
            else:
                inc = _distribution_power_increasing(n, -p)   # decreasing = reversed
                candidates.append(inc[::-1])

        # exponential (both signs)
        for lam in [0.0005, 0.001, 0.002, 0.005, 0.01]:
            candidates.append(_distribution_exponential_increasing(n, lam))
            # decreasing counterpart appears automatically after reversal

        # two‑block spikes
        for block_frac in [0.01, 0.03, 0.05, 0.07, 0.10]:
            candidates.append(_distribution_two_blocks(n, block_frac))

        # sparse random vectors
        for dens in [0.005, 0.01, 0.02]:
            candidates.append(_distribution_random_sparse(n, rng, density=dens))

        # evaluate all candidates
        for cand in candidates:
            if time.time() >= deadline:
                break
            cand = cand / cand.sum()
            sc = _objective(cand)

            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

            top_seeds.append((sc, cand.copy()))

    # keep a handful of the best seeds for refinement
    top_seeds.sort(key=lambda x: x[0])
    top_seeds = top_seeds[:10]

    # ------------------------------------------------------------------
    # 2️⃣  Refine the best seeds with accelerated sub‑gradient descent
    remaining = deadline - time.time()
    if remaining > 60 and top_seeds:
        per_seed = remaining / len(top_seeds)
        for _, seed_vec in top_seeds:
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            overall_best_seq, overall_best_score = _subgradient_descent_enhanced(
                seed_vec.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=1.0,
                max_iter=30000,
                eval_interval=400,
            )
            print(f"[{time.time():.1f}] after descent best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Incorporate a previously stored solution if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            overall_best_seq, overall_best_score = _subgradient_descent_enhanced(
                prev.copy(),
                deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.7,
                max_iter=15000,
                eval_interval=300,
            )
            print(f"[{time.time():.1f}] refined previous best to {overall_best_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4️⃣  Final cheap greedy polishing (pairwise swaps)
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
