The goal is to minimise  

\[
\text{score}(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2},
\qquad a_i\ge 0 .
\]

Because the score is homogeneous of degree 0 we may (and do) normalise
\(\sum a_i = 1\).  
For a vector of length \(n\) the score simplifies to  

\[
\text{score}(a)=2n\;\max\bigl((a*a)\bigr),
\]

where “\(*\)” denotes the full (linear‑time) convolution.

The programme below follows three ideas:

1. **Broad analytic seed pool** – a large collection of families
   (uniform, Beta‑distributions, power‑laws, two‑block spikes,
   sparse Dirichlet draws, …) is generated for a variety of lengths.
   Every seed is evaluated with the exact objective and the *K* best
   (currently 15) are kept for further optimisation.

2. **Two flavours of mirror/gradient descent** –  
   from each seed we run
   *exponentiated* (multiplicative) mirror descent and
   *additive* projected gradient descent.
   The true maximal convolution index is used to form the exact
   sub‑gradient

   \[
   g_j = 2\,a_{\,\text{idx}-j}\quad\text{if }0\le\text{idx}-j<n,
   \]

   and the step‑size follows a slowly decreasing schedule  
   \(\eta_t = \eta_0 / t^{0.5}\).  
   After every 200 iterations the current objective is evaluated and,
   if it improves the global best, the vector is stored.

3. **Final polishing** – a cheap greedy pairwise mass transfer
   (the “swap” routine from the reference implementation) is applied
   until the time budget is exhausted.

The whole pipeline respects a global deadline (`budget_s`).  All
operations are vectorised and use FFT‑based convolution, so the method
comfortably fits into the 1000 s time limit even for lengths up to
\(n=64\,000\).

The returned list contains non‑negative numbers; the evaluator will
clip them to \([0,1000]\) and normalise the sum automatically, so the
output is ready for direct use.

"""
Propose a step‑function (list of non‑negative heights) that minimises
evaluate_sequence().

Algorithm
---------

1.  Build a large pool of analytic “seed” distributions for several
    lengths (2000 … 64000).  Families include uniform, symmetric/asymmetric
    Beta, power‑laws, two‑block spikes and sparse Dirichlet draws.
    Each seed is evaluated exactly (2·n·max(conv)) and the best few are
    retained.

2.  For each retained seed run two simple first‑order schemes:
    * exponentiated mirror‑descent  a←a·exp(‑η·g)
    * additive projected gradient    a←Π_simplex(a‑η·g)

    In both cases g is the exact sub‑gradient obtained from the current
    maximal convolution index.  The step‑size η decays like 1/√t.
    Periodically the objective is recomputed and the global best vector
    is updated.

3.  When the remaining time is small a greedy pairwise mass‑transfer
    walk (move a small δ from a source index to a destination index)
    is performed; moves are kept only if the true objective improves.

The whole search respects the 1000 s deadline and returns the best
sequence found.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(alpha,beta)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0))
               + (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_decreasing(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} , monotone decreasing."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _power_increasing(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{p} , monotone increasing."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _two_blocks(n: int, block_frac: float) -> np.ndarray:
    """Equal spikes at the two ends, each occupying block_frac·n entries."""
    k = max(1, int(round(block_frac * n)))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _sparse_dirichlet(n: int, rng: np.random.Generator, concentration: float) -> np.ndarray:
    """Dirichlet draw with small concentration → sparse vector."""
    g = rng.gamma(concentration, 1.0, size=n)
    if g.sum() == 0.0:
        g[0] = 1.0
    g /= g.sum()
    return g


def _candidate_pool(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Generate a broad collection of seeds for a given length."""
    candidates = []

    # classic families
    candidates.append(_uniform(n))
    candidates.append(_beta_sym(n, 0.5))                # arcsine (p=0.5)

    # symmetric beta – log‑spaced p
    for p in np.logspace(-2, 0.5, 12):
        candidates.append(_beta_sym(n, float(p)))

    # asymmetric beta – a few random draws
    for _ in range(6):
        alpha = 10 ** rng.uniform(-2, 2)   # 0.01 … 100
        beta = 10 ** rng.uniform(-2, 2)
        candidates.append(_beta_asym(n, alpha, beta))

    # monotone power‑law, both decreasing and increasing
    for p in (0.5, 1.0, 2.0, 3.0):
        candidates.append(_power_decreasing(n, p))
        candidates.append(_power_increasing(n, p))

    # two‑block spikes – several block fractions
    for bf in (0.05, 0.10, 0.20, 0.30):
        candidates.append(_two_blocks(n, bf))

    # sparse Dirichlet draws
    for conc in (0.01, 0.05, 0.1):
        candidates.append(_sparse_dirichlet(n, rng, conc))

    return candidates

# Sub‑gradient / mirror‑descent helpers ---------------------------------
def _subgrad_mirror(a: np.ndarray, deadline: float,
                    best_seq: np.ndarray, best_score: float,
                    eta0: float = 1.0,
                    max_iter: int = 20000,
                    eval_every: int = 250) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror) descent on the true max‑convolution."""
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        idx = int(conv.argmax())

        # sub‑gradient: 2·a shifted by idx
        offsets = idx - np.arange(n)
        mask = (offsets >= 0) & (offsets < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[offsets[mask]]

        # step‑size η_t = η0 / sqrt(t)
        eta = eta0 / math.sqrt(t)

        # multiplicative update, then re‑normalise
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] mirror it={t} new best {best_score:.6f}")

    return best_seq, best_score


def _subgrad_additive(a: np.ndarray, deadline: float,
                      best_seq: np.ndarray, best_score: float,
                      eta0: float = 1.0,
                      max_iter: int = 20000,
                      eval_every: int = 250) -> tuple[np.ndarray, float]:
    """Additive projected gradient descent on the exact max‑convolution."""
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        idx = int(conv.argmax())

        offsets = idx - np.arange(n)
        mask = (offsets >= 0) & (offsets < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[offsets[mask]]

        eta = eta0 / math.sqrt(t)
        a = a - eta * grad
        a = _project_onto_simplex(a)

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] additive it={t} new best {best_score:.6f}")

    return best_seq, best_score

# Greedy pairwise mass transfer (final polishing) ----------------------
def _pairwise_local(a: np.ndarray, deadline: float,
                    best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Randomly move a small amount of mass; keep only improving moves."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.25   # ≤ 25 % of source mass
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score

                eta0=0.8,
                max_iter=15000,
                eval_every=300,
            )
            overall_best_seq = best_seq.copy()

    # ------------------------------------------------------------------
    # 6.  Final greedy local swaps (tiny simulated‑annealing walk)
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    print(f"FINAL best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # The evaluator will clip values to [0,1000] and re‑normalise, so no
    # further post‑processing is required.
    return overall_best_seq.tolist()
