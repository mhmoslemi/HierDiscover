   ``max_k (a*a)[k]``.
   The loss is   L(a)=log‑sum‑exp(α·(a*a−max)), whose gradient can be
   written in closed form via a convolution.  The update
   ``a ← a·exp(‑η·grad)`` stays on the positive orthant and we finish
   each step by projecting back onto the simplex {x≥0, Σx=1}.
   The parameters α (smoothness) and η (step‑size) are increased /
   decreased geometrically over thousands of iterations.
   This stage is run for a long time (up to a few hundred seconds) on
   fairly large vectors (5000–20000 entries) – the FFT‑based convolutions
   are cheap (≈10⁻³ s per iteration).

3. **Simple hill‑climber** – After the gradient phase we perform a cheap
   random‑mass‑transfer local search: move a tiny amount of probability
   from the larger participant of the currently maximal convolution entry
   to a random location, accepting the move only if the maximal
   convolution value decreases.

All intermediate results are printed; the best vector found before the
time budget expires is returned as a list of plain Python floats.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full linear convolution a*b using FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _score_of(a):
    """Return the objective 2·n·max(a*a) for a vector with sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families – each returns a non‑negative vector that sums to 1.
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The implementation follows the three‑stage
    plan described in the module docstring.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0   # tiny safety buffer

    eps = 1e-14                     # to avoid exact zeros
    best_vec = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1.  Scan a few analytic families on several lengths.
    candidate_lengths = [2000, 4000, 8000, 16000]
    for n in candidate_lengths:
        if time.time() >= deadline:
            break

        # build a list of initial vectors for this length
        inits = []

        inits.append(_distribution_uniform(n))
        inits.append(_distribution_arcsine(n))
        inits.append(_distribution_sine_power(n, 0.5))
        inits.append(_distribution_sine_power(n, 2.0))
        inits.append(_distribution_gaussian(n, 0.25))

        for p in (0.3, 0.6, 1.0, 2.0, 4.0):
            inits.append(_distribution_beta_sym(n, p))

        for _ in range(4):
            alpha = random.choice([0.2, 0.5, 1.0, 2.0])
            inits.append(np.random.dirichlet(np.full(n, alpha)))

        # evaluate cheap initially
        for vec in inits:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _score_of(vec)
            if sc < best_score:
                best_score = sc
                best_vec = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] init n={n}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  Smooth‑max exponentiated‑gradient (mirror descent) on the best vec.
    if best_vec is None:
        # fallback (should never happen)
        best_n = 4000
        best_vec = _distribution_uniform(best_n)

    a = best_vec.copy()          # working copy
    n = a.shape[0]

    # schedule parameters – they work well for a wide range of n
    total_steps = 60000                     # you can increase if time permits
    a0, a1 = 1.0, 1e5                       # α from 1 to 1e5 (geometric)
    eta0, eta1 = 0.12, 0.0008               # learning rate from 0.12 down to 8e‑4

    for step in range(total_steps):
        if time.time() >= deadline:
            break

        # interpolation factor
        t = step / total_steps
        alpha = a0 * (a1 / a0) ** t
        eta = eta0 * (eta1 / eta0) ** t

        # self‑convolution of the current vector
        conv = _fft_self_convolve(a)
        maxc = conv.max()

        # soft‑max weights focusing on the current maximal entries
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth max (see analysis)
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        # exponentiated‑gradient update on the simplex
        a = a * np.exp(-eta * grad)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)     # keep away from exact zero
        a /= a.sum()

        # occasional full evaluation (every ~500 steps)
        if step % 500 == 0 or step == total_steps - 1:
            sc = _score_of(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_vec = a.copy()
                best_n = n
                print(f"[{time.time():.1f}] mirror step={step}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3.  Simple hill‑climber fine‑tuning (mass transfers).
    a = best_vec.copy()
    conv = _fft_self_convolve(a)
    cur_max = conv.max()
    hill_it = 0
    while time.time() < deadline:
        hill_it += 1

        # locate the index k where the max occurs
        k = int(conv.argmax())

        # find the contributing pair (i,j) with the largest product a[i]*a[j]
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        best_i, best_j = i_low, k - i_low
        max_prod = a[best_i] * a[best_j]
        for i in range(i_low, i_high + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        # move a small chunk from the larger contributor to a random slot
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a_snapshot = a.copy()
        a[source] -= delta
        a[target] += delta
        a = np.maximum(a, eps)
        a /= a.sum()
        conv = _fft_self_convolve(a)
        new_max = conv.max()

        if new_max < cur_max - 1e-12:
            cur_max = new_max
            sc = 2.0 * n * cur_max
            if sc < best_score - 1e-12:
                best_score = sc
                best_vec = a.copy()
                print(f"[{time.time():.1f}] hill it={hill_it}  score={sc:.6f}")
        else:
            a = a_snapshot
            conv = _fft_self_convolve(a)

        # occasional jitter to escape plateaus
        if hill_it % 4000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            cur_max = conv.max()
            print(f"[{time.time():.1f}] jitter applied at hill it={hill_it}")

    final_score = _score_of(best_vec)
    print(f"Final best score ≈ {final_score:.6f} (length={best_n})")
    return best_vec.tolist()
