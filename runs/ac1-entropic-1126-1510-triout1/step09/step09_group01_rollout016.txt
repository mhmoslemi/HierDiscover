def _distribution_mixture_gaussian(n, means, sigmas, weights):
    x = np.arange(n, dtype=np.float64)
    xs = (x)/(n-1)   # normalized to [0,1]
    vals = np.zeros_like(x, dtype=np.float64)
    for mu, sigma, wgt in zip(means, sigmas, weights):
        vals += wgt * np.exp(-0.5 * ((xs - mu)/(sigma))**2)
    vals = np.maximum(vals, 0)
    if vals.sum()==0:
        vals = np.full_like(vals, 1.0/n)
    else:
        vals /= vals.sum()
    return vals

"""
Propose a step‑function (non‑negative vector) that minimises the
`evaluate_sequence` score.

The search works in four stages:

1️⃣ Broad family exploration – many analytic families (uniform,
   arcsine, triangular, exponential, symmetric & skewed beta,
   geometric, mixtures of Gaussians) together with a few random
   Dirichlet draws are generated for several lengths.  The best
   normalised candidate is kept.

2️⃣ Smooth‑max mirror‑descent – a softened version of the max‑
   convolution (log‑sum‑exp) is minimised by an exponentiated‑gradient
   (mirror‑descent) update with an annealing schedule of the
   smoothing parameter `α` (up to 5·10⁵) and a decaying learning‑rate.
   The best iterate is retained.

3️⃣ Stochastic mass‑transfer (simulated‑annealing) – from the current
   best sequence we repeatedly move a tiny amount of mass from a
   high‑probability index to a random low‑probability one.  Moves are
   accepted if they improve the objective, otherwise they are accepted
   with a small temperature‑dependent probability.  This explores
   neighbourhoods that gradient descent may miss and usually yields a
   small but useful improvement.

4️⃣ Polished projected sub‑gradient descent – a true sub‑gradient of
   the max‑convolution is computed on the active set (indices attaining
   the current maximum) and an Adam‑style adaptive step is taken with
   back‑tracking line‑search.  The iterate with the smallest score is
   finally returned.

All operations respect the time budget (default 1000 s).  The returned
list can be fed directly to `evaluate_sequence`.  The algorithm works
for any length; larger lengths give a lower theoretical bound but also
more computation, so a moderate length (≈ 3000–8000) is used.

# Simplex projection (Euclidean)
def _project_onto_simplex(v, s=1.0):
    """Project vector v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# FFT‑based convolution utilities
def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a):
    """Full linear self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full linear convolution of two vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a):
    """Score 2·n·max(a*a) for a normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_skew(n, a, b):
    """Beta(a,b) – possibly skewed."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_geometric(n, decay):
    """Monotonically decreasing geometric: a_i ∝ decay**i."""
    i = np.arange(n, dtype=np.float64)
    weights = decay ** i
    total = weights.sum()
    return weights / total


def _distribution_geometric_rev(n, decay):
    """Monotonically increasing geometric (reverse of above)."""
    i = np.arange(n, dtype=np.float64)
    weights = decay ** (n - 1 - i)
    total = weights.sum()
    return weights / total


def _distribution_mixture_gaussian(n, means, sigmas, weights):
    """Mixture of (continuous) Gaussians sampled on the discrete grid."""
    xs = np.arange(n, dtype=np.float64) / (n - 1)   # in [0,1]
    vals = np.zeros_like(xs)
    for mu, sigma, w in zip(means, sigmas, weights):
        vals += w * np.exp(-0.5 * ((xs - mu) / sigma) ** 2)
    vals = np.maximum(vals, 0)
    if vals.sum() == 0:
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step function that minimises evaluate_sequence."""
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin
    eps = 1e-12

    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1. Broad exploration over several lengths.
    lengths = [2000, 4000, 8000]               # a few increasing sizes
    for n in lengths:
        if time.time() >= deadline:
            break
        cand = []

        # Analytic families
        cand.append(_distribution_uniform(n))
        cand.append(_distribution_arcsine(n))
        cand.append(_distribution_triangular(n))
        cand.append(_distribution_exponential(n, lam=0.001))
        cand.append(_distribution_exponential(n, lam=0.02))
        cand.append(_distribution_sine_power(n, power=2.0))
        cand.append(_distribution_cosine_window(n, coeff=0.5))
        cand.append(_distribution_gaussian(n, sigma=0.2))

        # Symmetric Beta with a fine grid
        for p in (0.02, 0.05, 0.08, 0.12, 0.2, 0.5, 1.0, 2.0):
            cand.append(_distribution_beta_sym(n, p))

        # Skewed Beta (a small, b larger) – creates a U‑shape
        for a, b in ((0.02, 0.5), (0.02, 1.0), (0.05, 0.5), (0.1, 2.0)):
            cand.append(_distribution_beta_skew(n, a, b))

        # Geometric families (both directions)
        for decay in (0.998, 0.999, 0.9995):
            cand.append(_distribution_geometric(n, decay))
            cand.append(_distribution_geometric_rev(n, decay))

        # Two‑Gaussian mixture
        for sigma1, sigma2 in ((0.08, 0.3), (0.12, 0.4)):
            cand.append(_distribution_mixture_gaussian(
                n, means=[0.3, 0.7],
                sigmas=[sigma1, sigma2],
                weights=[0.5, 0.5]))

        # Random Dirichlet draws with various concentrations
        for alpha in (0.01, 0.05, 0.2, 0.5, 1.0):
            vec = np.random.dirichlet(np.full(n, alpha))
            cand.append(vec)

        # Sparse random start (few non‑zeros)
        sparse = np.zeros(n, dtype=np.float64)
        k = max(2, n // 500)                # a few hundred non‑zeros at most
        idx = np.random.choice(n, k, replace=False)
        sparse[idx] = np.random.rand(k)
        sparse /= sparse.sum()
        cand.append(sparse)

        # Evaluate all candidates
        for vec in cand:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] n={n} new init best {sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Incorporate a previously discovered sequence if available.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = len(prev)
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback – should never happen.
    if best_seq is None:
        best_n = 4000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Smooth‑max (log‑sum‑exp) mirror descent – aggressive schedule.
    a = best_seq.copy()
    n = a.shape[0]
    schedule = [
        # (α_start, α_end, steps, lr_start, lr_end)
        (1.0, 500.0, 3000, 0.5, 0.02),
        (500.0, 5000.0, 3000, 0.02, 0.002),
        (5000.0, 50000.0, 3500, 0.002, 0.0002),
        (50000.0, 5e5, 3500, 0.0002, 2e-5),
    ]

    for α0, α1, steps, lr0, lr1 in schedule:
        if time.time() >= deadline:
            break
        for it in range(steps):
            if time.time() >= deadline:
                break
            t = it / steps
            α = α0 * (α1 / α0) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            mx = conv.max()
            # soft‑max weights
            w = np.exp(α * (conv - mx))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of the smooth max
            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

            # exponentiated‑gradient update (mirror descent)
            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if (it % 500 == 0) or (it == steps - 1):
                cur_sc = _max_convolution_score(a)
                if cur_sc < best_score - 1e-12:
                    best_score = cur_sc
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] mirror it={it} new best {cur_sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Stochastic mass‑transfer (simulated‑annealing style).
    a = best_seq.copy()
    n = a.shape[0]
    cur_best = best_score
    iteration = 0
    # initial temperature – controls probability of uphill moves
    T0 = 0.01

    while time.time() < deadline:
        iteration += 1
        # Choose a source index with probability proportional to its mass
        i = np.random.choice(n, p=a)
        # Choose a distinct target uniformly
        j = random.randrange(n - 1)
        if j >= i:
            j += 1

        # Step magnitude shrinks as we approach the deadline
        time_frac = (deadline - time.time()) / (deadline - start)
        max_delta = min(a[i] * 0.5, 0.02)
        delta = max_delta * random.random() * (T0 * time_frac + 1e-9)
        if delta < eps:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.maximum(a_new, eps)
        a_new /= a_new.sum()
        new_sc = _max_convolution_score(a_new)

        # Accept if improvement, otherwise accept with a tiny probability
        if new_sc < cur_best - 1e-12:
            cur_best = new_sc
            best_seq = a_new.copy()
            best_score = new_sc
            a = a_new
            print(f"[{time.time():.1f}] SA it={iteration} new best {new_sc:.6f}")
        else:
            # uphill acceptance
            prob = math.exp(-(new_sc - cur_best) / max(1e-6, T0))
            if random.random() < prob:
                a = a_new

        # occasional jitter to shake the system
        if iteration % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

    # ------------------------------------------------------------------
    # 6. Final projected sub‑gradient polishing (Adam‑style).
    a = best_seq.copy()
    velocity = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam‑style first‑moment update
        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** it)

        # simple back‑tracking line search
        step = step0
        while True:
            cand = a - step * v_hat
            cand = _project_onto_simplex(cand, 1.0)
            cand = np.maximum(cand, eps)
            cand /= cand.sum()
            cand_sc = _max_convolution_score(cand)
            if cand_sc < best_score - 1e-12 or step < 1e-8:
                break
            step *= 0.5

        a = cand
        if cand_sc < best_score - 1e-12:
            best_score = cand_sc
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {cand_sc:.6f}")

        # occasional jitter
        if it % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    # ------------------------------------------------------------------
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
