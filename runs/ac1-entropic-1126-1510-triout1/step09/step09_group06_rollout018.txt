"""
Propose a step‑function that (approximately) minimises `evaluate_sequence`.

Strategy
========
1. **Broad sampling** – generate many candidate multisets of non‑negative
   numbers (Dirichlet draws, exponential, Gaussian, arcsine, etc.) for a
   handful of lengths.  For each multiset we create a few deterministic
   reorderings (alternating extremes, centre‑outwards, binary‑tree) and a
   handful of completely random permutations.  All vectors are ℓ₁‑normalised
   and scored with the cheap surrogate `2·n·max(conv(a,a))`.

2. **Keep the best few** – a min‑heap stores the top‑k candidates (k≈12).

3. **Intensive refinement** – each top seed is polished with three
   first‑order methods (Adam, exponentiated‑gradient, plain SGD).  The
   optimisation respects the global deadline and stops as soon as the time
   budget is exhausted.

4. **Frank‑Wolfe mass transfer** – on the current global best we repeatedly
   move a fraction of mass from the index with the largest sub‑gradient to
   the index with the smallest.  This is a provably descent direction for
   the ``max``‑objective.

5. **Local permutation search** – finally we try a cheap simulated‑annealing
   on the ordering: random swaps are accepted only when they improve the
   surrogate.  Because the objective depends strongly on the ordering,
   this often yields a small but systematic gain.

All helper routines are top‑level, use only NumPy and the standard library,
and respect the 1000 s time budget.  The returned list is ready for
`evaluate_sequence`.
"""

import time
import heapq
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate for `evaluate_sequence` when Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of the max‑convolution functional on the simplex.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0
    conv_rev = _fft_self_convolve(a[::-1] * w)  # use convolution with weight mask
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # correct diagonal terms (i=j) which appear only once in the sum
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _gaussian_distribution(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    w /= w.sum()
    return w


def _cauchy_distribution(n: int, scale_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    scale = scale_factor * n
    w = 1.0 / (1.0 + ((idx - centre) / scale) ** 2)
    w /= w.sum()
    return w


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Unsymmetric Beta(α,β) on a discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide='ignore', invalid='ignore'):
        w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    if np.isnan(w).any() or np.isinf(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex from Dirichlet(α,…,α)."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Re‑ordering heuristics -----------------------------------------------
def order_alternate_extremes(desc_vals: np.ndarray) -> np.ndarray:
    """Place the largest, second‑largest, … alternately at the left‑most
    and right‑most positions."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    left, right = 0, n - 1
    toggle = True
    for v in desc_vals:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    return out


def order_center_outwards(desc_vals: np.ndarray) -> np.ndarray:
    """Largest entry at centre, then fill alternatingly outwards."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    mid = (n - 1) // 2
    positions = [mid]
    for d in range(1, n):
        left = mid - d
        if left >= 0:
            positions.append(left)
        right = mid + d
        if right < n:
            positions.append(right)
    for i, pos in enumerate(positions[:n]):
        out[pos] = desc_vals[i]
    return out


def order_binary_tree(desc_vals: np.ndarray) -> np.ndarray:
    """Fill the interval recursively at its midpoint (binary‑tree layout)."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)

    def recurse(l: int, r: int, it: list[int]):
        if l > r:
            return
        m = (l + r) // 2
        it.append(m)
        recurse(l, m - 1, it)
        recurse(m + 1, r, it)

    positions = []
    recurse(0, n - 1, positions)
    out[positions] = desc_vals
    return out


# ----------------------------------------------------------------------
# Mirror‑descent variants -----------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 300_000,
                  step0: float = 0.2,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-14:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 200_000,
                          eta0: float = 0.8) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-14:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _sgd_subgrad(a0: np.ndarray, deadline: float,
                 max_iters: int = 300_000,
                 step0: float = 0.5) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        step = step0 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-14:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Local permutation search ---------------------------------------------
def _local_swap_opt(a0: np.ndarray, deadline: float,
                    max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Hill‑climbing on the ordering: swap two entries if the surrogate
    improves.  Very cheap because each evaluation uses the FFT surrogate."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    n = len(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]
        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score, best_a = cur, a.copy()
        else:
            # reject the swap
            a[i], a[j] = a[j], a[i]
        it += 1
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects a ~1000 s time budget and returns the best vector
    discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # a small safety margin

    # ------------------------------------------------------------------
    # Base length (use previous best if available)
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        base_n = 4000

    # ------------------------------------------------------------------
    # Lengths to explore
    lengths = set()
    for factor in (0.5, 0.75, 1.0, 1.25, 1.5, 2.0):
        n = int(base_n * factor)
        if 500 <= n <= 12000:
            lengths.add(n)
    for _ in range(5):
        n = np.random.randint(500, min(12000, base_n * 3) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # Heap that holds the current best candidates (min‑heap on the score)
    TOP_K = 12
    heap = []  # entries are (-score, array)

    def consider(vec: np.ndarray):
        """Normalise, evaluate and possibly push into the top‑K heap."""
        if vec.ndim != 1 or vec.size == 0:
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s < 1e-12:
            return
        vec = vec / s
        sc = _max_convolution_score(vec)
        if len(heap) < TOP_K:
            heapq.heappush(heap, (-sc, vec))
        else:
            if sc < -heap[0][0]:
                heapq.heapreplace(heap, (-sc, vec))

    # ------------------------------------------------------------------
    # 1. Broad sampling – families + a few random permutations
    for n in lengths:
        if time.time() >= deadline:
            break

        # simple analytic families (no re‑ordering yet)
        fams = [
            (_arcsine_distribution, "arcsine"),
            (_triangular_distribution, "triangular"),
            (_exponential_distribution, "exponential"),
            (_gaussian_distribution, "gaussian"),
            (_cauchy_distribution, "cauchy")
        ]
        for gen, _name in fams:
            a = gen(n)
            consider(a)
            # deterministic re‑orderings
            desc = np.sort(a)[::-1]
            consider(order_alternate_extremes(desc))
            consider(order_center_outwards(desc))
            consider(order_binary_tree(desc))
            # a few random permutations
            for _ in range(2):
                consider(np.random.permutation(a))

        # symmetric beta with many p‑values
        beta_ps = np.concatenate([
            np.arange(0.02, 0.21, 0.02),
            np.arange(0.25, 0.71, 0.05),
            np.arange(0.80, 1.51, 0.10),
            np.array([2.0])
        ])
        for p in beta_ps:
            if time.time() >= deadline:
                break
            a = _beta_symmetric_distribution(n, p)
            consider(a)
            desc = np.sort(a)[::-1]
            consider(order_alternate_extremes(desc))
            consider(order_center_outwards(desc))

        # unsymmetric beta (select a few pairs)
        betas = [0.5, 1.0, 1.5, 2.0, 3.0]
        for alpha in betas:
            for beta in betas:
                if time.time() >= deadline:
                    break
                a = _beta_distribution(n, alpha, beta)
                consider(a)
                desc = np.sort(a)[::-1]
                consider(order_alternate_extremes(desc))

        # Dirichlet draws with several concentration parameters
        alphas = [0.1, 0.2, 0.5, 0.8, 1.0, 2.0, 5.0]
        for a_alpha in alphas:
            if time.time() >= deadline:
                break
            for _ in range(3):
                vec = _random_dirichlet_alpha(n, a_alpha)
                consider(vec)
                # a couple of random permutations of the same multiset
                for __ in range(2):
                    consider(np.random.permutation(vec))

    # ------------------------------------------------------------------
    # 2. Bring forward the previous best if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.sum() > 0:
            consider(prev)
            print(f"Injected previous best candidate.")
    except Exception:
        pass

    # Transform heap into a sorted list (best first)
    top_candidates = sorted([(-sc, arr) for sc, arr in heap], key=lambda x: x[0])
    top_candidates = [(arr, sc) for sc, arr in top_candidates]
    print(f"Collected {len(top_candidates)} top seeds (k={TOP_K}).")

    # ------------------------------------------------------------------
    # 3. Intensive refinement of each top seed
    best_seq = None
    best_score = float('inf')
    for idx, (seed_seq, _) in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        # Adam
        seq, sc = _adam_descent(seed_seq, deadline, max_iters=400_000, step0=0.2)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{idx+1}/{len(top_candidates)}] Adam → {best_score:.6f}")

        # Exponentiated gradient
        seq, sc = _exp_gradient_descent(seed_seq, deadline,
                                        max_iters=250_000, eta0=0.8)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{idx+1}/{len(top_candidates)}] ExpGrad → {best_score:.6f}")

        # Plain SGD
        seq, sc = _sgd_subgrad(seed_seq, deadline,
                               max_iters=300_000, step0=0.5)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{idx+1}/{len(top_candidates)}] SGD → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Frank‑Wolfe style mass transfer on the current global best
    if best_seq is None:
        # This should never happen, but safeguard
        raise RuntimeError("No candidate generated.")
    a = best_seq.copy()
    it_fw = 0
    while time.time() < deadline - 1.0 and it_fw < 150_000:

        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:

        if delta <= 0:
            break
        step = 0.7 * delta
        a[i_src] -= step
        a[i_dst] += step
        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score, best_seq = cur, a.copy()
            print(f"FW improved best to {best_score:.6f}")
        it_fw += 1
        # occasional tiny random jitter to escape plateaus
        if it_fw % 4000 == 0 and np.random.rand() < 0.1:
            a = _project_onto_simplex(a + np.random.randn(a.size) * 1e-5, s=1.0)

    # ------------------------------------------------------------------
    # 5. Local permutation search on the best vector found so far
    if time.time() < deadline - 1.0:
        seq, sc = _local_swap_opt(best_seq, deadline, max_iters=200_000)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"Local‑swap improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
