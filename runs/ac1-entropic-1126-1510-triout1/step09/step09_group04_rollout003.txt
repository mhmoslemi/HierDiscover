"""
propose_candidate – search for a non‑negative step function with a small
self‑convolution peak.

The algorithm consists of three stages

1.  **Broad analytic scan** – a collection of parametric families
    (uniform, triangular, arcsine, symmetric Beta, sine‑power,
    Gaussian, exponential) is evaluated on several lengths (500‑5000).
    For the Beta and sine‑power families a dense grid of exponents
    down to 0.01 is tried – the optimum of these cheap families is a
    very good starting point.

2.  **Large‑scale smooth‑max mirror descent** – the current best vector
    is refined on the simplex by minimizing a smooth approximation of
    `max(a*a)`.  The smoothness parameter `α` is increased from 1 to
    1e7 while the learning rate is simultaneously decreased.
    Each iteration performs one FFT‑based convolution, evaluates a
    soft‑max weight vector and takes an exponentiated‑gradient step,
    followed by a Euclidean projection onto the simplex.
    The loop runs as long as time permits (typically a few‑thousand
    iterations).

3.  **Targeted hill‑climber** – the convolution is examined for the
    index achieving the current maximum.  The pair of coefficients
    responsible for this entry is identified and a small amount of
    mass is moved from the larger coefficient to a random location.
    Moves that strictly lower the peak are kept; otherwise they are
    discarded.  Occasionally a tiny random jitter is added to escape
    plateaus.

All operations are O(n log n) thanks to the FFT; the code stays well
within the 1000 s wall‑clock budget and avoids the large dense LP
matrix that caused memory errors in the reference implementation.
The function returns the best sequence (as a plain list of floats) found
so far.
"""

import time
import random
import math
from typing import List

import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Given a vector with sum = 1, return the value of ``evaluate_sequence``.
    The formula reduces to 2·n·max(a*a) because sum(a)=1.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families – each returns a vector whose entries sum to 1.
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.01) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0) by clipping
    eps = np.finfo(np.float64).tiny
    x = np.clip(x, eps, 1 - eps)
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    total = w.sum()
    if total == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= total
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects a wall‑clock budget (default 1000 s) and returns the
    best vector it discovered.
    """
    # ------------------------------------------------------------------
    # Basic setup
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5   # margin for finalisation
    eps = 1e-15

    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1.  Analytic scan on a handful of lengths
    lengths = [600, 1000, 1500, 2000, 2500, 3000, 3500, 4000, 4500, 5000]
    beta_ps = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.25, 0.3,
               0.4, 0.5, 0.6, 0.8, 1.0, 1.5, 2.0, 3.0, 4.0]
    sine_ps = [0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0]
    gauss_sigmas = [0.15, 0.20, 0.25, 0.30, 0.35]
    exp_lams = [0.004, 0.006, 0.008, 0.01, 0.015]

    for n in lengths:
        if time.time() >= deadline:
            break

        # assemble candidates for this length
        cand = []

        # classic families
        cand.append(_distribution_uniform(n))
        cand.append(_distribution_triangular(n))
        cand.append(_distribution_arcsine(n))

        # exponential family (centre‑peaked)
        for lam in exp_lams:
            cand.append(_distribution_exponential(n, lam=lam))

        # symmetric Beta with many exponents (including very heavy tails)
        for p in beta_ps:
            cand.append(_distribution_beta_sym(n, p))

        # sine‑power (p < 1 pushes mass to the edges)
        for pw in sine_ps:
            cand.append(_distribution_sine_power(n, pw))

        # discrete Gaussian (moderate width)
        for sigma in gauss_sigmas:
            cand.append(_distribution_gaussian(n, sigma))

        # a few random Dirichlet draws (different concentration)
        for alpha in (0.05, 0.1, 0.3, 0.5, 1.0):
            cand.append(np.random.dirichlet(np.full(n, alpha)))

        # evaluate all candidates
        for vec in cand:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] init n={n}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  Initialise from a previously supplied global if it is better
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_seq = prev.copy()
                best_n = prev.size
                best_score = sc
                print(f"[{time.time():.1f}] global best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback to uniform if nothing was found (should never happen)
    if best_seq is None:
        best_n = 2000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Large‑scale smooth‑max mirror descent (exponentiated gradient)
    a = best_seq.copy()
    n = a.shape[0]

    # total number of gradient steps – we adapt to the remaining time
    # (aim for a few thousand; each step costs ≈ O(n log n))
    max_steps = 12000
    step = 0
    while time.time() < deadline and step < max_steps:
        # schedule for smoothness α and learning rate η
        t = step / max_steps
        alpha = 1.0 * (1e7 / 1.0) ** t               # 1 → 1e7
        eta   = 0.25 * (0.001 / 0.25) ** t          # 0.25 → 0.001

        conv = _fft_self_convolve(a)
        m = conv.max()
        # soft‑max weights (numerically stable)
        w = np.exp(alpha * (conv - m))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth max
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        # exponentiated‑gradient update on the simplex
        a = a * np.exp(-eta * grad)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)          # avoid exact zeros which hurt the FFT

        # occasional reporting and best‑score update
        if step % 300 == 0 or step == max_steps - 1:
            sc = _max_convolution_score(a)
            if sc < best_score - 1e-12:
                best_seq = a.copy()
                best_score = sc
                best_n = n
                print(f"[{time.time():.1f}] mirror step={step}  score={sc:.6f}")

        step += 1

    # ------------------------------------------------------------------
    # 5.  Targeted hill‑climber (mass‑transfer moves)
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    while time.time() < deadline:
        it += 1
        # index where the convolution is maximal
        k = int(conv.argmax())

        # locate the pair (i,j) giving the maximal product for this k
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        # vectorised search for the maximal product
        idxs = np.arange(i_low, i_high + 1, dtype=np.int64)
        prods = a[idxs] * a[k - idxs]
        max_idx = int(np.argmax(prods))
        i_star = int(idxs[max_idx])
        j_star = k - i_star

        # move a small chunk from the larger coefficient towards a random place
        src = i_star if a[i_star] >= a[j_star] else j_star
        tgt = random.randint(0, n - 1)
        if tgt == src:
            continue

        max_delta = min(a[src] * 0.5, 0.02)          # do not move too much
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta
        a_new = np.maximum(a_new, eps)
        a_new /= a_new.sum()
        conv_new = _fft_self_convolve(a_new)
        M_new = conv_new.max()

        if M_new < M - 1e-12:               # strict improvement
            a = a_new
            conv = conv_new
            M = M_new
            sc = 2.0 * n * M
            if sc < best_score - 1e-12:
                best_seq = a.copy()
                best_score = sc
                if it % 500 == 0:
                    print(f"[{time.time():.1f}] hill it={it}  score={sc:.6f}")
        else:
            # with a tiny probability we still accept to escape plateaus
            if random.random() < 0.001:
                a = a_new
                conv = conv_new
                M = M_new

        # every few thousand iterations add a microscopic jitter
        if it % 4000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] jitter added @ hill it={it}")

    # ------------------------------------------------------------------
    final_score = 2.0 * best_n * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={best_n})")
    return best_seq.tolist()
