"""
Hybrid search for a non‑negative step function that minimises
`evaluate_sequence`.  The pipeline combines:

1. Analytic families (Beta, arcsine, power‑transformed arcsine,
   exponential, Gaussian, Cauchy) on several lengths.
2. Deterministic and random re‑orderings of the generated value set.
3. Three local‑optimisers:
   • Projected Adam on the exact sub‑gradient of the max‑convolution.
   • Projected gradient on a smoothed “soft‑max’’ surrogate of the max.
   • Simple hill‑climbing by random mass transfers.
4. A final Frank‑Wolfe pairwise mass‑transfer sweep.
All steps respect a 1000 s wall‑clock budget and the best vector (ℓ₁‑normalised)
is returned as a plain ``list[float]``.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Geometry / optimisation helpers
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = v.astype(np.float64, copy=False)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Exact surrogate for `evaluate_sequence` when Σa = 1:
        score = 2·n·max(a∗a)
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of the max‑convolution functional on the simplex.
    Returns (grad, cur_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with the weight vector gives raw gradient
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for terms where i=j (even shift)
    even = (active % 2 == 0)
    diag_idx = (active[even] // 2)
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

def order_alternate_extremes(desc_vals: np.ndarray) -> np.ndarray:
    """Spread the largest values to the left‑most and right‑most ends."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    left, right = 0, n - 1
    toggle = True
    for v in desc_vals:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    return out


def order_center_outwards(desc_vals: np.ndarray) -> np.ndarray:
    """Largest in the centre, then filling outwards."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    positions = []
    mid = (n - 1) // 2
    positions.append(mid)
    for d in range(1, n):
        left = mid - d
        if left >= 0:
            positions.append(left)
        right = mid + d
        if right < n:
            positions.append(right)
    for i, p in enumerate(positions[:n]):
        out[p] = desc_vals[i]
    return out


def order_binary_tree(desc_vals: np.ndarray) -> np.ndarray:
    """Recursively fill the midpoint of each remaining interval."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)

    def recurse(l: int, r: int, pos_list):
        if l > r:
            return
        m = (l + r) // 2
        pos_list.append(m)
        recurse(l, m - 1, pos_list)
        recurse(m + 1, r, pos_list)

    positions = []
    recurse(0, n - 1, positions)
    out[positions] = desc_vals
    return out

def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size))
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _gaussian_distribution(n: int, sigma_factor: float = 0.15) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    w /= w.sum()
    return w


def _cauchy_distribution(n: int, scale_factor: float = 0.15) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    scale = scale_factor * n
    w = 1.0 / (1.0 + ((idx - centre) / scale) ** 2)
    w /= w.sum()
    return w


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    total = w.sum()
    if total == 0 or np.isnan(total):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= total
    return w


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """General Beta(α,β)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide="ignore", invalid="ignore"):
        w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x))
    if np.isnan(w).any() or np.isinf(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _arcsine_power_distribution(n: int, power: float) -> np.ndarray:
    """Take the arcsine density and raise it to `power` (power>1 accentuates edges)."""
    base = _arcsine_distribution(n)
    vals = np.power(base, power)
    vals /= vals.sum()
    return vals


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet(α,…,α) point."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 300_000,
                  step0: float = 0.2,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        g, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a -= eta * m_hat
        a = _project_onto_simplex(a)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-14:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


def _softmax_descent(a0: np.ndarray, deadline: float,
                     lambda_start: float = 1.0,
                     lambda_end: float = 30.0,
                     lambda_steps: int = 6,
                     it_per_step: int = 20_000) -> tuple[np.ndarray, float]:
    """Gradient descent on a soft‑max approximation of the max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    for step in range(lambda_steps):
        lam = lambda_start * (lambda_end / lambda_start) ** (step / (lambda_steps - 1))
        for it in range(it_per_step):
            if time.time() >= deadline:
                return best_a, best_score
            b = _fft_self_convolve(a)
            bmax = b.max()
            # soft‑max weights (stable)
            shifted = b - bmax
            w = np.exp(lam * shifted)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            # gradient of the soft‑max surrogate
            grad = 2.0 * _fft_convolve(a[::-1], w)[len(a) - 1 : len(a) - 1 + len(a)]
            step_size = 0.6 / np.sqrt(it + 1)
            a -= step_size * grad
            a = _project_onto_simplex(a)
            cur = _max_convolution_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()
    return best_a, best_score


def _hillclimb_moves(a0: np.ndarray, deadline: float,
                     max_iters: int = 150_000,
                     init_temp: float = 0.01,
                     cooling: float = 0.9999) -> tuple[np.ndarray, float]:
    """
    Very simple stochastic hill‑climbing on the simplex:
      pick i→j mass transfer, accept if improvement or according to a
      decreasing temperature.
    """
    a = a0.copy()
    cur_score = _max_convolution_score(a)
    best_a = a.copy()
    best_score = cur_score
    T = init_temp
    n = a.shape[0]
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        i, j = np.random.choice(n, 2, replace=False)
        if a[i] <= 1e-15:
            continue
        eps = np.random.rand() * a[i]            # move a fraction of a[i]
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # projection is not needed because the move keeps sum and positivity
        new_score = _max_convolution_score(a_new)
        delta = new_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        T *= cooling
    return best_a, best_score


def _frank_wolfe_pairwise(a0: np.ndarray, deadline: float,
                          max_iters: int = 120_000) -> tuple[np.ndarray, float]:
    """Mass transfer from the index with largest sub‑gradient to the smallest."""
    a = a0.copy()
    best_score = _max_convolution_score(a)
    best = a.copy()
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        delta = min(a[i_src], 1.0 - a[i_dst])
        if delta <= 0:
            break
        step = 0.65 * delta          # a modest transfer fraction
        a[i_src] -= step
        a[i_dst] += step
        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score = cur
            best = a.copy()
        # occasional tiny random jitter to escape potential plateaus
        if it % 4000 == 0 and np.random.rand() < 0.1:
            a = _project_onto_simplex(a + np.random.randn(a.size) * 1e-5)
    return best, best_score

def _add_candidate(vec: np.ndarray, store: list):
    """Normalise, guard against degenerate vectors and store with its score."""
    if vec.ndim != 1 or vec.size == 0:
        return
    v = np.maximum(vec, 0.0)
    s = v.sum()
    if s < 1e-12:
        return
    v = v / s
    store.append((v, _max_convolution_score(v)))


def _random_permutations_of(desc: np.ndarray, n_perm: int, store: list):
    """Generate a few random reorderings of the sorted values."""
    for _ in range(n_perm):
        perm = np.random.permutation(desc)
        _add_candidate(perm, store)

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function that minimises `evaluate_sequence`.
    Respects the supplied time budget and returns the best vector discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Choose a sensible base length (use previous best if present)
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        base_n = 4000

    # ------------------------------------------------------------------
    # Lengths to explore (moderate to large, but never too huge)
    lengths = set()
    for factor in (0.75, 1.0, 1.25, 1.5, 2.0):
        n = int(base_n * factor)
        if 800 <= n <= 12000:
            lengths.add(n)
    # a few random lengths
    for _ in range(6):
        n = np.random.randint(800, min(12000, base_n * 3) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # Containers for all generated seeds
    all_candidates = []          # list of (array, score)

    # ------------------------------------------------------------------
    # ------------------------------------------------------------------
    # 1. Broad sampling of analytic families
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # ---- Simple families ------------------------------------------------
        families = [
            (_arcsine_distribution, "arcsine"),
            (_triangular_distribution, "triangular"),
            (_exponential_distribution, "exponential"),
            (_gaussian_distribution, "gaussian"),
            (_cauchy_distribution, "cauchy")
        ]
        for gen, tag in families:
            a = gen(n)
            _add_candidate(a, all_candidates)

            desc = np.sort(a)[::-1]
            _add_candidate(order_alternate_extremes(desc), all_candidates)
            _add_candidate(order_center_outwards(desc), all_candidates)
            _add_candidate(order_binary_tree(desc), all_candidates)
            _random_permutations_of(desc, 3, all_candidates)

        # ---- Symmetric Beta -------------------------------------------------
        beta_ps = list(np.arange(0.05, 1.01, 0.05)) + list(np.arange(1.2, 2.01, 0.2))
        for p in beta_ps:
            if time.time() >= deadline:
                break
            a = _beta_symmetric_distribution(n, p)
            _add_candidate(a, all_candidates)
            desc = np.sort(a)[::-1]
            _add_candidate(order_alternate_extremes(desc), all_candidates)
            _add_candidate(order_center_outwards(desc), all_candidates)
            _add_candidate(order_binary_tree(desc), all_candidates)

        # ---- Unsymmetric Beta (edge‑heavy & centre‑heavy) -------------------
        small_params = [0.05, 0.1, 0.2, 0.4, 0.8]
        large_params = [2.0, 3.0, 5.0, 8.0]
        for alpha in small_params:
            for beta in large_params:
                if time.time() >= deadline:
                    break
                a = _beta_distribution(n, alpha, beta)
                _add_candidate(a, all_candidates)
                desc = np.sort(a)[::-1]
                _add_candidate(order_alternate_extremes(desc), all_candidates)

        # ---- Power‑transformed arcsine (edge accentuation) -----------------
        for power in [1.2, 1.5, 1.8, 2.0, 2.5]:
            if time.time() >= deadline:
                break
            a = _arcsine_power_distribution(n, power)
            _add_candidate(a, all_candidates)
            desc = np.sort(a)[::-1]
            _add_candidate(order_alternate_extremes(desc), all_candidates)

        # ---- Random Dirichlet draws (wide diversity) -----------------------
        for alpha in [0.01, 0.05, 0.1, 0.2, 0.5]:
            if time.time() >= deadline:
                break
            a = _random_dirichlet_alpha(n, alpha)
            _add_candidate(a, all_candidates)

    # ------------------------------------------------------------------
    # 2. Include the previous best if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        if prev.sum() > 0:
            prev = prev / prev.sum()
            _add_candidate(prev, all_candidates)
            print(f"Added previous best candidate, score={all_candidates[-1][1]:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Keep a handful of the most promising seeds
    all_candidates.sort(key=lambda x: x[1])
    top_k = min(15, len(all_candidates))
    seeds = [vec for vec, _ in all_candidates[:top_k]]
    print(f"Generated {len(all_candidates)} candidates, keeping top {top_k}.")
    print(f"Best seed score = {all_candidates[0][1]:.6f}")

    # ------------------------------------------------------------------
    # 4. Intensive refinement of each seed
    best_seq = seeds[0].copy()
    best_score = _max_convolution_score(best_seq)

    for idx, seed in enumerate(seeds):
        if time.time() >= deadline:
            break
        # ----- Adam on exact sub‑gradient -----
        a, sc = _adam_descent(seed, deadline, max_iters=300_000, step0=0.25)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} Adam -> {best_score:.6f}")

        # ----- Soft‑max (smoothed) gradient descent -----
        a, sc = _softmax_descent(seed, deadline,
                                 lambda_start=0.5, lambda_end=40.0,
                                 lambda_steps=5, it_per_step=15_000)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} SoftMax -> {best_score:.6f}")

        # ----- Simple hill‑climbing moves -----
        a, sc = _hillclimb_moves(seed, deadline, max_iters=120_000)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} HillClimb -> {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Global Frank‑Wolfe pairwise mass transfer on the current best
    a, sc = _frank_wolfe_pairwise(best_seq, deadline, max_iters=140_000)
    if sc < best_score:
        best_score, best_seq = sc, a.copy()
        print(f"Frank–Wolfe sweep improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final random re‑ordering sweep (tiny budget)
    remaining = deadline - time.time()
    if remaining > 5.0:
        desc = np.sort(best_seq)[::-1]
        trials = 2000
        for t in range(trials):
            if time.time() >= deadline:
                break
            perm = np.random.permutation(desc)
            cur = _max_convolution_score(perm)
            if cur < best_score:
                best_score = cur
                best_seq = perm.copy()
                print(f"Random reorder iteration {t+1}/{trials} new best {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
