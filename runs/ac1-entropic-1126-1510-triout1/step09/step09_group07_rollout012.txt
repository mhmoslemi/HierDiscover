#!/usr/bin/env python
"""
Propose a step‑function that minimises ``evaluate_sequence``.

The search consists of three stages:

1️⃣ **Initialisation & cheap seeding** – a handful of analytic families
   (arcsine, smoothed arcsine, power‑product, symmetric/ asymmetric Beta,
   Hann/Hamming/Blackman windows and an exponential tail) are evaluated on a
   modest length (≈ 3000).  The best cheap score (2·n·max(conv)) is kept as a
   seed.

2️⃣ **Convex LP refinement** – given a seed a (∑a=1) compute M=‖a*a‖∞.
   Solve the linear programme

        maximise   1ᵀg
        subject to  a∗g ≤ M   (element‑wise)
                    g ≥ 0

   for g.  The optimum gives a larger ℓ₁‑mass under the same max‑convolution
   bound.  After normalising g to unit sum we move a towards g:
   a ← (1‑t)·a + t·g (t≈0.2).  Repeating this step monotonically reduces M.
   Only a few (≤ 15) LP solves are needed; each LP has 2n‑1 ≤ 6500 constraints,
   and we build its matrix from a using a sparse Toeplitz construction.

3️⃣ **Fine‑grained optimisation** – on the final a we run a long
   sub‑gradient descent (exact sub‑gradient of max‑convolution) and a tiny
   random‑mass‑transfer hill‑climber.  Both respect the simplex via Euclidean
   projection.  The whole routine respects a 1000 s global deadline.

All returned numbers are non‑negative and sum to one, i.e. they can be fed
directly to ``evaluate_sequence``.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a∗a (length 2·n‑1) via real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a∗b via real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap surrogate = 2·n·max(a∗a) (a assumed to sum to 1)."""
    n = a.size
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
#  Analytic families (all return a normalised probability vector)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _smoothed_arcsine_distribution(n: int, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5 + eps) * (n - i - 0.5 + eps))
    return vals / vals.sum()


def _product_power_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-alpha)
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, p: float, q: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    w = np.where(np.isfinite(w), w, 0.0)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _hann_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2 * np.pi * i / (n - 1))
    return w / w.sum()


def _hamming_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    return w / w.sum()


def _blackman_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    return w / w.sum()


def _random_dirichlet(n: int) -> np.ndarray:
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()

def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a∗a) on the simplex."""
    n = a.size
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (k = 2*i)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 300_000, eta0: float = 0.8):
    """Plain sub‑gradient descent with diminishing step size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _final_hill_climb(a: np.ndarray, deadline: float) -> np.ndarray:
    """Tiny random mass transfers (accept‑only‑improvements)."""
    cur_score = _max_convolution_score(a)
    n = a.size
    it = 0
    max_transfer = 0.10
    while time.time() < deadline:
        it += 1
        transfer = max_transfer * (0.5 ** (it / 15000.0))
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)
        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
    return a


# ----------------------------------------------------------------------
def _conv_constraint_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the sparse matrix A (shape (2n‑1, n)) such that
            (A @ g) = a ∗ g
    for any non‑negative vector g.
    Row k corresponds to shift k (0 ≤ k < 2n‑1).
    """
    n = a.shape[0]
    # column index repeats 0..n‑1 for each diagonal
    cols = np.tile(np.arange(n, dtype=np.int64), n)
    # row index = i + j where i is diagonal offset (0..n‑1)
    rows = np.repeat(np.arange(n, dtype=np.int64), n) + np.tile(np.arange(n, dtype=np.int64), n)
    data = np.repeat(a, n)
    shape = (2 * n - 1, n)
    return sp.csr_matrix((data, (rows, cols)), shape=shape)


def _lp_refinement(a: np.ndarray, deadline: float,
                   max_iters: int = 12, t: float = 0.2) -> np.ndarray:
    """
    Perform a small number of LP‑based refinement steps.
    Each step solves:
        maximise 1ᵀg   s.t. a∗g ≤ M, g ≥ 0
    where M = max(a∗a).  The resulting g is renormalised and blended
    with the current a.
    """
    n = a.shape[0]
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        M = _fft_self_convolve(a).max()
        A = _conv_constraint_matrix(a)          # (2n‑1) × n   sparse
        c = -np.ones(n)                         # maximise sum(g)  → minimise -sum(g)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0, None)] * n

        # Solve LP – HiGHS is the default for scipy ≥1.6 and handles sparse mats well.
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'time_limit': max(0.0, deadline - time.time())})
        if not res.success:
            # If LP fails, abort refinement.
            break

        g = np.maximum(res.x, 0.0)          # numeric safety
        sum_g = g.sum()
        # Trivial case – cannot increase ℓ₁‑mass ⇒ stop.
        if sum_g <= 1.0 + 1e-12:
            break

        # Scale to unit sum and blend towards it.
        g = g / sum_g
        a_new = (1.0 - t) * a + t * g
        a_new = _project_onto_simplex(a_new, s=1.0)

        # Accept only if we actually improve the cheap objective.
        if _max_convolution_score(a_new) < _max_convolution_score(a) - 1e-14:
            a = a_new
        else:
            # try a smaller blend; if still no improvement stop.
            t_small = t * 0.5
            a_new = (1.0 - t_small) * a + t_small * g
            a_new = _project_onto_simplex(a_new, s=1.0)
            if _max_convolution_score(a_new) < _max_convolution_score(a) - 1e-14:
                a = a_new
            else:
                break
    return a
