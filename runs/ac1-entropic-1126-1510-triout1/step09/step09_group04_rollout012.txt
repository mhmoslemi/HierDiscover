"""
Propose a step‑function (non‑negative list) that minimises
``evaluate_sequence``.

Algorithm
---------
1. Scan a handful of analytic families (uniform, arcsine, triangular,
   exponential, beta, sine‑power, Gaussian) on several lengths and keep
   the best normalised vector (∑a_i = 1).
2. From the best vectors run a *smooth‑max* optimisation:
   minimise   (1/α)·log∑_k exp(α·(a∗a)_k)   by L‑BFGS‑B on the log‑softmax
   parametrisation a_i = exp(z_i)/∑_j exp(z_j).  The smoothing parameter
   α is increased (5→15→30→60) – a continuation scheme that drives the
   surrogate towards the true max‑convolution.
3. Perform a short stochastic hill‑climber that moves a tiny amount of
   mass from a pair contributing to the current maximum toward a random
   index; accept only if the true max‑convolution decreases.
4. Return the best vector found (as plain Python ``float`` objects).

All steps respect a global wall‑clock deadline (default 1000 s).  The
implementation uses only NumPy and SciPy and works purely in memory.
"""

import time

from scipy.optimize import minimize

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Score 2·n·max(a*a) for a vector already normalised to sum = 1.
    This is exactly the value returned by ``evaluate_sequence`` for such a
    vector (because the denominator (∑a)^2 = 1).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families (all return a vector with sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.01) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_power_law(n: int, exponent: float) -> np.ndarray:
    """p_i ∝ (i+1)^{-exponent} (after a small shift to avoid zero)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-exponent)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Smooth‑max surrogate and its gradient (used by L‑BFGS‑B)
# ----------------------------------------------------------------------
def _smooth_max_obj_and_grad(z: np.ndarray, n: int, alpha: float):
    """
    Returns (obj, grad) where
        obj  = (1/α)·log∑_k exp(α·(a∗a)_k)
        grad = ∇_z obj   (z are the log‑softmax parameters)
    """
    # a = softmax(z)
    a = np.exp(z)
    a_sum = a.sum()
    a /= a_sum

    # convolution a∗a
    conv = _fft_self_convolve(a)

    # log‑sum‑exp in a numerically stable way
    max_c = conv.max()
    exp_term = np.exp(alpha * (conv - max_c))
    sum_exp = exp_term.sum()
    obj = (np.log(sum_exp) + alpha * max_c) / alpha

    # softmax weights w_k = exp(α·conv_k) / ∑exp
    w = exp_term / sum_exp

    # gradient w.r.t. a (see derivation in the analysis)
    grad_a = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

    # chain rule ∂obj/∂z = (∂obj/∂a)·∂a/∂z  ;  ∂a/∂z = diag(a) - a a^T
    dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot)

    return obj, grad_z


def _refine_by_lbfgs(a0: np.ndarray, n: int, deadline: float,
                    alphas=(5.0, 15.0, 30.0, 60.0),
                    maxiter_per_stage: int = 80) -> np.ndarray:
    """
    Run a continuation of L‑BFGS‑B optimisations on the smooth‑max surrogate.
    ``a0`` must already be normalised (∑a = 1).  Returns a (potentially)
    improved vector, still on the simplex.
    """
    a = a0.copy()
    # start from log‑softmax parameters
    z = np.log(a + 1e-16)

    for alpha in alphas:
        if time.time() >= deadline:
            break
        # minimise the smooth‑max; L‑BFGS‑B returns both value and gradient
        res = minimize(_smooth_max_obj_and_grad,
                       z,
                       args=(n, alpha),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': maxiter_per_stage,
                                'disp': False})
        # update parameters (even if optimiser did not converge)
        z = res.x
        # recompute a from the latest z
        a = np.exp(z)
        a /= a.sum()
    return a

# Small stochastic hill‑climber (mass‑transfer moves)
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple random hill‑climber that repeatedly moves a tiny amount of mass
    from the index most responsible for the current maximum towards a
    random index.  Stops when the wall‑clock is exhausted.
    """
    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0

    while time.time() < deadline:
        it += 1
        # locate the convolution index attaining the current maximum
        k = int(conv.argmax())

        # find the pair (i,j) with i+j=k giving the largest product a_i*a_j
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        best_i, best_j = i_low, k - i_low
        best_prod = a[best_i] * a[best_j]
        for i in range(i_low, i_high + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > best_prod:
                best_prod = prod
                best_i, best_j = i, j

        # move a small chunk from the larger contributor to a random slot
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a_snapshot = a.copy()
        a[source] -= delta
        a[target] += delta
        a = np.maximum(a, eps)
        a /= a.sum()
        conv = _fft_self_convolve(a)
        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
            if it % 500 == 0:
                print(f"[{time.time():.1f}] hill‑climb it={it}  max={M:.6g}")
        else:
            a = a_snapshot
            conv = _fft_self_convolve(a)

        # occasional jitter to avoid stagnation
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] jitter applied at hill‑climb it={it}")

    return a
