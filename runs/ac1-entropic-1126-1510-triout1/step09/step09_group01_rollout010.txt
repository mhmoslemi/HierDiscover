"""
Propose a non‑negative step‑function (vector) that minimises
``evaluate_sequence``.  The method consists of three stages:

1. **Broad sweep** – generate a palette of analytic families (uniform,
   arcsine, beta, sine‑power, cosine‑window, Gaussian) and a few random
   Dirichlet draws for several lengths.  The best few candidates are kept.

2. **Adam‑optimisation of a smooth max surrogate** – the objective
   `max(conv(a,a))` is approximated by a temperature‑controlled
   soft‑max.  The variables are parametrised by logits (softmax) so that
   the simplex constraint is always satisfied.  A standard Adam optimiser
   with geometric schedules for the temperature (α) and the learning
   rate is applied.  Several restarts with a small random jitter are
   performed.

3. **Simple hill‑climber (mass‑transfer) polish** – using the current
   maximum convolution index, a tiny amount of mass is moved from the
   pair that contributes most to that entry to a random location; the
   move is accepted only if it reduces the true objective.

All work respects the 1000 s CPU budget and prints progress.  The final
vector (list of floats) is returned.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of a and b (real) via FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families – each returns a vector with sum = 1
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    if n == 1:
        return np.array([1.0])
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 5000,
    alpha_start: float = 1.0,
    alpha_end: float = 5e4,
    lr_start: float = 0.2,
    lr_end: float = 1e-3,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a smooth‑max surrogate for `max(conv(a,a))`.
    The vector is kept on the simplex via a soft‑max parametrisation.
    Returns the best vector seen and its (non‑smooth) score.
    """
    eps = 1e-12
    n = a0.shape[0]
    # initialise logits (stable soft‑max)
    logits = np.log(np.maximum(a0, eps))
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # current best
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # schedules (geometric interpolation)
        t = (it - 1) / max_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t

        # soft‑max to obtain current a
        logits_shift = logits - np.max(logits)   # numerical stabilisation
        a = np.exp(logits_shift)
        a = np.maximum(a, eps)
        a /= a.sum()

        # convolution
        conv = _fft_self_convolve(a)
        maxc = conv.max()

        # smooth‑max weights
        if alpha < 1e5:
            w = np.exp(alpha * (conv - maxc))
            ws = w.sum()
            if ws == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w = w / ws
        else:
            # true max – uniform over the (tiny) active set
            tol = 1e-12 * (abs(maxc) + 1.0)
            active = np.flatnonzero(conv >= maxc - tol)
            w = np.zeros_like(conv)
            w[active] = 1.0 / max(active.size, 1)

        # gradient w.r.t. a (see derivation in analysis)
        grad_a = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # convert to gradient w.r.t. logits (soft‑max Jacobian)
        dot = np.dot(a, grad_a)
        grad_log = a * (grad_a - dot)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # parameter update in logit space
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # every few iterations evaluate the true (non‑smoothed) score
        if it % 250 == 0 or it == max_iters:
            cur_score = _max_convolution_score(a)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(
                    f"[{time.time():.1f}] adam iter={it} new best {cur_score:.6f}"
                )
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:

    best_score = float("inf")
    best_n = None

    lengths = [2000, 4000, 8000, 15000, 25000, 40000]

        if time.time() >= deadline:
            break

        # analytic families (already normalised)
        candidates.append(_distribution_uniform(n))
        candidates.append(_distribution_arcsine(n))
        candidates.append(_distribution_triangular(n))
        candidates.append(_distribution_exponential(n, lam=0.003))
        candidates.append(_distribution_exponential(n, lam=0.02))

        # symmetric beta (edge‑heavy for small p)
        for p in (0.02, 0.04, 0.06, 0.08, 0.12, 0.15, 0.2,
                  0.25, 0.3, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
            candidates.append(_distribution_beta_sym(n, p))

        # sine‑power family
        for pw in (0.5, 1.0, 2.0, 4.0, 8.0, 12.0):
            candidates.append(_distribution_sine_power(n, pw))

        # cosine windows
        for coeff in (0.3, 0.5, 0.7, 0.9):
            candidates.append(_distribution_cosine_window(n, coeff))

        # Gaussians
        for sigma in (0.12, 0.18, 0.25, 0.35, 0.45):
            candidates.append(_distribution_gaussian(n, sigma))

        # Dirichlet draws of various concentrations
        for alpha in (0.2, 0.5, 1.0, 2.0):
            vec = np.random.dirichlet(np.full(n, alpha))
            candidates.append(vec)

        # sparse random start (few non‑zeros)
        sparse = np.zeros(n, dtype=np.float64)
        k = max(2, n // 300)
        idx = np.random.choice(n, k, replace=False)
        sparse[idx] = np.random.rand(k)
        sparse /= sparse.sum()
        candidates.append(sparse)

        # evaluate all candidates, keep the top few for this n
        top_local = []      # list of (score, vec)
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] n={n} new global best {sc:.6f}")
            # maintain a small local heap of the best 5 for this length
            if len(top_local) < 5 or sc < top_local[-1][0]:
                top_local.append((sc, vec.copy()))
                top_local.sort(key=lambda x: x[0])
                if len(top_local) > 5:
                    top_local.pop()

        # add the local bests to the global pool
        global_pool.extend(top_local)

    # ------------------------------------------------------------------
    # 2. Optionally pull in a previously discovered sequence
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = prev.size
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
            global_pool.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback in the extremely unlikely case that we have nothing
    if best_seq is None:
        best_n = 4000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # sort pool and keep only a handful of promising seeds
    global_pool.sort(key=lambda x: x[0])
    seed_pool = [vec for _, vec in global_pool[:10]]

    # ------------------------------------------------------------------
    # 4. Adam optimisation on the smooth‑max surrogate (multiple restarts)
    for init_vec in seed_pool:
        if time.time() >= deadline:
            break
        n = init_vec.shape[0]

        # first run (no jitter)
        a_opt, sc_opt = _adam_optimize(init_vec, deadline)
        if sc_opt < best_score - 1e-12:
            best_score = sc_opt
            best_seq = a_opt.copy()
            best_n = n
            print(f"[{time.time():.1f}] adam seed n={n} improved to {sc_opt:.6f}")

        # a couple of jittered restarts from the best found so far
        for _ in range(2):
            if time.time() >= deadline:
                break
            jitter = np.random.randn(n) * 1e-4
            a_jit = np.maximum(a_opt + jitter, eps)
            a_jit /= a_jit.sum()
            a_opt2, sc_opt2 = _adam_optimize(a_jit, deadline)
            if sc_opt2 < best_score - 1e-12:
                best_score = sc_opt2
                best_seq = a_opt2.copy()
                best_n = n
                print(
                    f"[{time.time():.1f}] adam jitter n={n} improved to {sc_opt2:.6f}"

    # 5. Final hill‑climber (mass‑transfer) to polish the objective
    a = best_seq.copy()
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    while time.time() < deadline:

        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        jdx = k - idx
        prod = a[idx] * a[jdx]
        max_idx = prod.argmax()
        i_star, j_star = int(idx[max_idx]), int(jdx[max_idx])
        source = i_star if a[i_star] >= a[j_star] else j_star
        target = random.randint(0, n - 1)
        if target == source:
            continue
        # transfer a small amount of mass
        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M
            best_seq = a.copy()
            best_score = 2.0 * n * M
            print(f"[{time.time():.1f}] hill‑climb new best {best_score:.6f}")

        # occasional tiny random jitter to escape plateaus
        if random.random() < 0.001:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

    final_score = 2.0 * n * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={best_n})")
    return best_seq.tolist()
