def _smooth_subgrad(a, tau):
    n = a.shape[0]
    b = _fft_self_convolve(a)
    maxb = b.max()
    # compute weights w_i = exp(tau*(b_i - maxb))
    # we may use scaling to avoid overflow
    exp_w = np.exp(tau * (b - maxb))
    w = exp_w / exp_w.sum()
    # gradient
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2 * conv_rev[n - 1 : n - 1 + n]
    # diagonal correction
    even_mask = (np.arange(2 * n - 1) % 2 == 0)
    diag_i = np.flatnonzero(even_mask)  # even indices
    # diag_idx = diag_i // 2
    diag_idx = diag_i // 2
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx] * w[diag_i]
    return grad, b.max(), w

"""
Propose a non‑negative step‑function that minimises ``evaluate_sequence``.

The algorithm works in three stages:

1. **Broad sampling.**  
   A collection of analytic families (beta, exponential,
   arcsine, triangular, Hann, Gaussian, double‑bump, …) together
   with many random Dirichlet / sparse draws are generated for a
   handful of lengths (≈2000–18000).  The best ~40 candidates (by the
   exact surrogate score 2·n·max(conv)) are kept.

2. **Mixing & intensive local optimisation.**  
   The top candidates are randomly mixed (convex combinations) – this
   often straightens the autocorrelation.  The current best seed is
   then refined by a long Adam run, a smooth‑gradient descent (a
   differentiable soft‑max approximation of the max‑convolution) and
   a Frank‑Wolfe style mass‑transfer phase.  Each step projects back
   onto the probability simplex.

3. **Polishing.**  
   A quick simulated‑annealing jitter finishes the search.

All heavy work uses FFT‑based convolutions; the routine respects the
1000 s budget and prints progress messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(a,a)) (a is assumed to sum to 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) under Σa = 1.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for i=j (even active indices)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


def _smooth_subgrad(a: np.ndarray, tau: float):
    """
    Gradient of a smooth approximation of max(conv) using the
    Log‑Sum‑Exp (soft‑max) with temperature ``tau``.
    Returns (gradient, current_max, weight_vector).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    maxb = b.max()
    # stable soft‑max weights
    w = np.exp(tau * (b - maxb))
    w = w / w.sum()

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction with the same weights
    even_mask = (np.arange(2 * n - 1) % 2 == 0)
    diag = np.flatnonzero(even_mask)
    diag_idx = diag // 2
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx] * w[diag]

    return grad, b.max(), w


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.20,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 1_500_000):
    """Projected Adam on the simplex. Returns best point seen."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


def _smooth_gradient_descent(a0: np.ndarray, deadline: float,
                             tau_start: float = 0.5,
                             tau_end: float = 10.0,
                             phases: int = 5,
                             iters_per_phase: int = 200_000):
    """
    Gradient descent on the soft‑max surrogate.  The temperature ``tau``
    is slowly increased;  each phase runs a simple SGD with decreasing
    step size.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    taus = np.geomspace(tau_start, tau_end, num=phases)
    for tau in taus:
        step = 0.5
        it = 0
        while time.time() < deadline and it < iters_per_phase:
            it += 1
            grad, _, _ = _smooth_subgrad(a, tau)
            a = a - step * grad
            a = _project_onto_simplex(a, 1.0)
            step *= 0.9999          # very gentle decay
            cur = _max_convolution_score(a)
            if cur < best_score - 1e-15:
                best_score = cur
                best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with concentration ``alpha``."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


def _random_dirichlet_flexible(n: int) -> np.ndarray:
    """Dirichlet where each coordinate gets its own random concentration."""
    alphas = np.exp(np.random.uniform(-2.0, 2.0, size=n))
    x = np.random.gamma(alphas, 1.0)
    return x / x.sum()


def _random_sparse(n: int, k: int) -> np.ndarray:
    """Sparse vector with exactly ``k`` non‑zero entries."""
    idx = np.random.choice(n, size=k, replace=False)
    vals = np.random.rand(k)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    a /= a.sum()
    return a

# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _double_bump_distribution(n: int,
                              block_frac: float = 0.25,
                              gap_frac: float = 0.1) -> np.ndarray:
    """Two equal uniform blocks separated by a gap."""
    total_block = max(2, int(block_frac * n))
    m = total_block // 2
    g = int(gap_frac * n)
    a = np.zeros(n, dtype=np.float64)
    a[:m] = 1.0 / (2 * m)
    start2 = m + g
    if start2 + m > n:
        start2 = n - m
    a[start2:start2 + m] = 1.0 / (2 * m)
    return a


def _hann_distribution(n: int) -> np.ndarray:
    """Hann (raised‑cosine) window."""
    i = np.arange(n, dtype=np.float64)
    vals = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    vals /= vals.sum()
    return vals


def _gaussian_distribution(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    vals /= vals.sum()
    return vals

            if cur_score < best_score - 1e-14:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time():.1f}] Anneal iter={anneal_iter} new best {best_score:.6f}")
        else:
            prob = np.exp(-(new_score - cur_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                a = a_new
                cur_score = new_score

    # ------------------------------------------------------------------
    print(f"Final best surrogate score: {best_score:.6f} (length={len(best_seq)})")
    # Return plain Python list (evaluate_sequence will handle any further scaling)
    return best_seq.tolist()
