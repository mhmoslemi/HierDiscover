    scaled = b / tau

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact objective when Σa = 1: 2·n·max(conv(a,a))."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _smooth_grad(a: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of the τ‑smoothed objective
        f_τ(a) = 2·n·τ·log ∑_m exp(conv(a,a)[m]/τ)
    with respect to a (under the simplex constraint).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)               # full self‑convolution
    # stable softmax weights for the log‑sum‑exp
    scaled = b / tau
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w /= w.sum()                            # w_i = ∂f_τ/∂b_i
    # gradient of Σ_i w_i·b_i w.r.t. a is 2·conv_rev(w)
    conv_rev = _fft_convolve(a[::-1], w)
    return 2.0 * conv_rev[n - 1 : n - 1 + n]   # central n entries


def _adam_smooth_descent(a0: np.ndarray,
                         deadline: float,
                         step0: float = 0.30,
                         beta1: float = 0.9,
                         beta2: float = 0.999,
                         max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the τ‑smoothed objective.
    Returns the best point (according to the exact objective) seen.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    # initialise τ‑schedule – start with a fraction of the current max‑conv
    M0 = _fft_self_convolve(a).max()
    taus = [M0 * f for f in (0.20, 0.10, 0.05, 0.02, 0.01, 0.005, 0.001)]

    t = 0
    for tau in taus:
        # reset Adam's moments when τ changes
        m = np.zeros_like(a)
        v = np.zeros_like(a)

        while time.time() < deadline and t < max_iters:
            t += 1
            grad = _smooth_grad(a, tau)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            eta = step0 / (np.sqrt(v_hat) + 1e-8)
            a = a - eta * m_hat
            a = _project_onto_simplex(a, s=1.0)

            # check exact objective every few iterations (cheaper than each step)
            if t % 1000 == 0:
                cur = _max_convolution_score(a)
                if cur < best_score - 1e-15:
                    best_score = cur
                    best_a = a.copy()
        # go to next τ
    return best_a, best_score


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) under Σa = 1.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # correct the diagonal (i=j) terms where index is even
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with concentration ``alpha``."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _double_bump_distribution(n: int,
                              block_frac: float = 0.25,
                              gap_frac: float = 0.1) -> np.ndarray:
    """Two equal uniform blocks separated by a gap."""
    total_block = max(2, int(block_frac * n))
    m = total_block // 2
    g = int(gap_frac * n)
    a = np.zeros(n, dtype=np.float64)
    a[:m] = 1.0 / (2 * m)
    start2 = m + g
    if start2 + m > n:
        start2 = n - m
    a[start2:start2 + m] = 1.0 / (2 * m)
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a Python list; the caller will apply the same pre‑processing
    (clipping, normalisation) as ``evaluate_sequence``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose a sensible base length (respect any previously found sequence)
    try:
        base_n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        base_n = 6000
    base_n = max(1500, min(base_n, 18000))

    # A handful of lengths to explore
    lengths = set()
    for f in (0.8, 0.9, 1.0, 1.1, 1.25, 1.5, 2.0):
        n = int(base_n * f)
        if 1500 <= n <= 18000:
            lengths.add(n)
    # a few random lengths for diversity
    while len(lengths) < 12:
        lengths.add(np.random.randint(1500, min(20000, base_n * 2) + 1))
    lengths = sorted(lengths)

    print(f"[{time.time():.1f}] Exploring {len(lengths)} lengths (base≈{base_n})")

    # --------------------------------------------------------------
    # Generate candidate shapes
    candidates: list[tuple[np.ndarray, float]] = []
    beta_ps = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0]
    asym_grid = [(0.3, 0.7), (0.7, 0.3), (0.2, 0.8), (0.8, 0.2), (0.5, 1.2), (1.2, 0.5)]
    power_ps = [0.5, 0.8, 1.0, 1.5, 2.0]
    lam_vals = [0.002, 0.005, 0.01]
    dirichlet_alphas = [0.2, 0.5, 1.0, 2.0]

    for n in lengths:
        if time.time() >= deadline:
            break

        # deterministic analytic families --------------------------------
        a = _arcsine_distribution(n)
        candidates.append((a, _max_convolution_score(a)))
        a = _triangular_distribution(n)
        candidates.append((a, _max_convolution_score(a)))

        for lam in lam_vals:
            a = _exponential_distribution(n, lam=lam)
            candidates.append((a, _max_convolution_score(a)))

        for p in beta_ps:
            a = _beta_symmetric_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))

        for a_alpha, a_beta in asym_grid:
            a = _beta_asymmetric_distribution(n, a_alpha, a_beta)
            candidates.append((a, _max_convolution_score(a)))

        for p in power_ps:
            a = _power_law_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))

        for block_frac in (0.15, 0.20, 0.25):
            for gap_frac in (0.05, 0.10, 0.20):
                a = _double_bump_distribution(n, block_frac, gap_frac)
                candidates.append((a, _max_convolution_score(a)))

        for alpha in dirichlet_alphas:
            a = _random_dirichlet_alpha(n, alpha)
            candidates.append((a, _max_convolution_score(a)))

    # ------------------------------------------------------------------
    # Add previously best (if it exists)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            if prev.sum() == 0:
                prev = np.full_like(prev, 1.0 / prev.size)
            else:
                prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
            print(f"[{time.time():.1f}] added previous best, score={candidates[-1][1]:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Keep the best seeds for intensive refinement
    candidates.sort(key=lambda x: x[1])
    top_k = min(12, len(candidates))
    top_seeds = candidates[:top_k]
    print(f"Generated {len(candidates)} candidates, keeping top {top_k} for refinement.")

    best_seq = top_seeds[0][0].copy()
    best_score = top_seeds[0][1]
    print(f"Initial best surrogate score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Intensive refinement of each seed with smooth‑Adam
    for idx, (seed_seq, _) in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        n = seed_seq.shape[0]
        seq_opt, sc_opt = _adam_smooth_descent(seed_seq, deadline,
                                                step0=0.30,
                                                max_iters=300_000)
        if sc_opt < best_score - 1e-14:
            best_score, best_seq = sc_opt, seq_opt.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} "
                  f"smooth‑Adam improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Frank‑Wolfe style mass‑transfer polishing
    a = best_seq.copy()
    cur_score = best_score
    fw_iter = 0
    while time.time() < deadline and fw_iter < 130_000:
        fw_iter += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-15:
            break
        improved = False
        for frac in (1.0, 0.5, 0.25):
            delta = max_transfer * frac
            a_new = a.copy()
            a_new[i_src] -= delta
            a_new[i_dst] += delta
            sc = _max_convolution_score(a_new)
            if sc < cur_score - 1e-13:
                a = a_new
                cur_score = sc
                improved = True
                break
        if improved and cur_score < best_score - 1e-14:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time():.1f}] FW iter={fw_iter} new best {best_score:.6f}")

        # occasional jitter to escape tiny plateaus
        if fw_iter % 6000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(a.size) * 3e-5
            a = _project_onto_simplex(a + jitter, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    # Simulated‑annealing / jitter phase
    a = best_seq.copy()
    cur_score = best_score
    temp_start = 0.025
    temp_end = 0.001
    anneal_iter = 0
    while time.time() < deadline and anneal_iter < 200_000:
        anneal_iter += 1
        # linear schedule (hot → cold)
        t_rem = deadline - time.time()
        fraction = max(0.0, t_rem / (budget_s - 0.5))
        temp = temp_end + (temp_start - temp_end) * fraction

        i = np.random.randint(a.size)
        j = np.random.randint(a.size - 1)
        if j >= i:
            j += 1
        max_delta = min(a[i], 1.0 - a[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * temp
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if cur_score < best_score - 1e-14:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time():.1f}] Anneal iter={anneal_iter} new best {best_score:.6f}")
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_score - cur_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                a = a_new
                cur_score = new_score

    # ------------------------------------------------------------------
    print(f"Final best surrogate score: {best_score:.6f} (length={len(best_seq)})")
    # Return a plain list – ``evaluate_sequence`` will take care of clipping/scaling
    return best_seq.tolist()
