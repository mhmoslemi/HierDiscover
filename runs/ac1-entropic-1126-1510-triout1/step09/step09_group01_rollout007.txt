"""
Propose a non‑negative step function (list of floats) that minimizes the
``evaluate_sequence`` objective.

Algorithm
---------

1. **Broad analytic + random init** – generate a few hundred candidates
   (uniform, arcsine, triangular, exponential, symmetric beta, sine‑power,
   cosine‑window, Gaussian, random Dirichlet) for a chosen length *n*.
   Keep the one with the smallest ``2·n·max(conv)`` (the objective for a
   unit‑sum vector).

2. **Conditional‑gradient (Frank‑Wolfe) style refinement** – for the current
   vector ``a`` compute its maximal self‑convolution ``M``.
   Solve the linear program

        max   Σ g_i
        s.t.  (a * g)_k ≤ M   for all k,
              g_i ≥ 0.

   (The convolution ``a*g`` is linear in ``g``.)  The optimum ``g`` is a
   promising descent direction.  The LP is solved with ``scipy.optimize.linprog``
   (dense Toeplitz matrix, method='highs').

   The new vector is taken as a convex combination
        a(t) = (1‑t) a + t g ,   t ∈ [0,1] .
   For a given ``t`` the convolution of ``a(t)`` with itself can be written
   analytically and its maximum evaluated in O(n) time.  A ternary search on
   ``t`` finds the value that minimizes the maximal convolution.
   If the new maximum is strictly smaller we accept the step, otherwise a
   small random jitter is added and the process repeats.  Only a modest
   number of LP iterations (≈10–12) are performed – each iteration costs
   O(n²) to build the Toeplitz matrix and O(n·log n) for the FFTs, which
   comfortably fits the 1000 s budget.

3. **Optional hill‑climbing polish** – with any remaining time a simple
   coordinate‑wise mass‑transfer local search is performed: move a tiny amount
   of probability from the index that contributes the most to the current
   maximal convolution towards a random index.  Accept the move only if the
   maximal convolution decreases.

All vectors are kept non‑negative, projected onto the probability simplex,
and normalised to sum = 1 (the objective is scale‑invariant).  The final
vector is returned as a plain Python list of floats.
"""

import time
import random
import math
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz
from scipy.interpolate import interp1d

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σ x = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using a real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families – each returns a non‑negative vector with sum = 1
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid 1…n."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0)
    eps = np.finfo(np.float64).tiny
    x = np.clip(x, eps, 1 - eps)
    logw = (p - 1) * (np.log(x) + np.log(1 - x))
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n: int, coeff: float = 0.5) -> np.ndarray:
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return the best of several analytic families and random draws for size n."""
    eps = 1e-12
    candidates = []

    # Analytic families
    candidates.append(_distribution_uniform(n))
    candidates.append(_distribution_arcsine(n))
    candidates.append(_distribution_triangular(n))
    candidates.append(_distribution_exponential(n, lam=0.003))
    candidates.append(_distribution_exponential(n, lam=0.02))

    for p in (0.02, 0.04, 0.06, 0.08, 0.12, 0.15, 0.2,
              0.25, 0.3, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        candidates.append(_distribution_beta_sym(n, p))

    for pw in (0.5, 1.0, 2.0, 4.0, 8.0, 12.0):
        candidates.append(_distribution_sine_power(n, pw))

    for coeff in (0.3, 0.5, 0.7, 0.9):
        candidates.append(_distribution_cosine_window(n, coeff))

    for sigma in (0.12, 0.18, 0.25, 0.35, 0.45):
        candidates.append(_distribution_gaussian(n, sigma))

    # Dirichlet draws
    for alpha in (0.2, 0.5, 1.0, 2.0):
        candidates.append(rng.dirichlet(np.full(n, alpha)))

    # Sparse random start
    sparse = np.zeros(n, dtype=np.float64)
    k = max(2, n // 300)
    idx = rng.choice(n, k, replace=False)
    sparse[idx] = rng.random(k)
    sparse /= sparse.sum()
    candidates.append(sparse)

    best = None
    best_sc = float("inf")
    for vec in candidates:
        vec = np.maximum(vec, eps)
        vec /= vec.sum()
        sc = _max_convolution_score(vec)
        if sc < best_sc:
            best_sc = sc
            best = vec.copy()
    return best


# ----------------------------------------------------------------------
def _line_search_optimal_t(a: np.ndarray,
                           g: np.ndarray,
                           conv_aa: np.ndarray,
                           conv_ag: np.ndarray,
                           conv_gg: np.ndarray) -> tuple[float, float]:
    """
    Ternary search on t∈[0,1] to minimise max convolution of
    a(t) = (1‑t)·a + t·g.
    Returns (t_opt, M_opt) where M_opt = max((a(t) * a(t))).
    """
    lo, hi = 0.0, 1.0
    for _ in range(30):
        t1 = lo + (hi - lo) / 3.0
        t2 = hi - (hi - lo) / 3.0

        w1 = (1 - t1) ** 2
        w2 = 2 * t1 * (1 - t1)
        w3 = t1 ** 2
        conv_t1 = w1 * conv_aa + w2 * conv_ag + w3 * conv_gg
        M1 = conv_t1.max()

        w1 = (1 - t2) ** 2
        w2 = 2 * t2 * (1 - t2)
        w3 = t2 ** 2
        conv_t2 = w1 * conv_aa + w2 * conv_ag + w3 * conv_gg
        M2 = conv_t2.max()

        if M1 < M2:
            hi = t2
        else:
            lo = t1

    t_opt = (lo + hi) / 2.0
    w1 = (1 - t_opt) ** 2
    w2 = 2 * t_opt * (1 - t_opt)
    w3 = t_opt ** 2
    conv_opt = w1 * conv_aa + w2 * conv_ag + w3 * conv_gg
    M_opt = conv_opt.max()
    return t_opt, M_opt


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function (non‑negative vector) that minimises
    ``evaluate_sequence``.  The routine respects a CPU‑time budget of
    ``budget_s`` seconds and returns the best sequence found as a plain
    Python list of floats.
    """
    rng = np.random.default_rng(seed)
    random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1. Broad initial search (single length – large enough to be competitive)
    n = 2500
    if time.time() < deadline:
        a = _initial_candidate(n, rng)
        sc = _max_convolution_score(a)
        best_seq = a.copy()
        best_score = sc
        best_n = n
        print(f"[{time.time():.1f}] initial length={n} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Include a previously found sequence if the global variable exists.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            # Resample to current length if necessary (linear interpolation)
            if prev.size != best_n:
                x_old = np.linspace(0.0, 1.0, prev.size)
                x_new = np.linspace(0.0, 1.0, best_n)
                f_int = interp1d(x_old, prev, kind="linear", fill_value="extrapolate")
                prev = f_int(x_new)
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous sequence improved score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Conditional‑gradient (LP) refinement
    a = best_seq.copy()
    n = a.shape[0]

    # pre‑compute current convolution
    conv_aa = _fft_self_convolve(a)
    M = conv_aa.max()

    max_lp_iters = 12
    lp_iter = 0
    while time.time() < deadline and lp_iter < max_lp_iters:
        lp_iter += 1

        # Build Toeplitz matrix for constraints (a * g) ≤ M
        # First column: a followed by n‑1 zeros; first row: a[0] then zeros
        col = np.concatenate([a, np.zeros(n - 1, dtype=np.float64)])
        row = np.concatenate([a[:1], np.zeros(n - 1, dtype=np.float64)])
        A_ub = toeplitz(col, row)                     # shape (2n‑1, n)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        c = -np.ones(n, dtype=np.float64)           # maximise Σ g_i

        # Solve LP
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True})
        if not res.success:
            print(f"[{time.time():.1f}] LP #{lp_iter} failed: {res.message}")
            break

        g = np.maximum(res.x, 0.0)
        if g.sum() == 0:
            print(f"[{time.time():.1f}] LP #{lp_iter} produced zero vector")
            break
        g /= g.sum()                                 # enforce Σ g = 1

        # Convolutions needed for line search
        conv_ag = _fft_convolve(a, g)
        conv_gg = _fft_self_convolve(g)

        # optimal convex‑combination parameter
        t_opt, M_new = _line_search_optimal_t(a, g, conv_aa, conv_ag, conv_gg)

        if M_new < M - 1e-14:                        # genuine improvement
            a = (1 - t_opt) * a + t_opt * g
            a = np.maximum(a, eps)
            a /= a.sum()
            conv_aa = (1 - t_opt) ** 2 * conv_aa + \
                      2 * t_opt * (1 - t_opt) * conv_ag + \
                      t_opt ** 2 * conv_gg
            M = M_new
            cur_score = 2.0 * n * M
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] LP #{lp_iter} t={t_opt:.4f} new best {cur_score:.6f}")
        else:
            # No descent – inject a tiny random jitter and recompute
            jitter = rng.standard_normal(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv_aa = _fft_self_convolve(a)
            M = conv_aa.max()
            print(f"[{time.time():.1f}] LP #{lp_iter} no improvement – jitter applied")

        if time.time() + 0.5 >= deadline:
            break

    # ------------------------------------------------------------------
    # 4. Optional cheap hill‑climber (mass‑transfer) if time permits
    a = best_seq.copy()
    conv_a = _fft_self_convolve(a)
    M = conv_a.max()
    hill_iter = 0
    while time.time() < deadline:
        hill_iter += 1
        # locate the index that contributes the current maximal entry
        k = int(conv_a.argmax())
        # the pair (i,j) that gave the maximal product a_i*a_j for this k
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        best_i, best_j = i_start, k - i_start
        best_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > best_prod:
                best_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = rng.integers(0, n)
        if target == source:
            continue

        # transfer a small amount from source → target
        delta = min(a[source] * 0.5, 0.02) * rng.random()
        if delta < eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        M2 = _fft_self_convolve(a2).max()
        if M2 < M - 1e-12:
            a = a2
            conv_a = _fft_self_convolve(a)
            M = M2
            cur_score = 2.0 * n * M
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hill‑climb it={hill_iter} new best {cur_score:.6f}")

        if hill_iter % 5000 == 0:
            # occasional jitter to avoid stagnation
            jitter = rng.standard_normal(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv_a = _fft_self_convolve(a)
            M = conv_a.max()
            print(f"[{time.time():.1f}] hill‑climb jitter at it={hill_iter}")

        if time.time() + 0.5 >= deadline:
            break

    final_score = 2.0 * best_seq.shape[0] * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
