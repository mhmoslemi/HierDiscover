def _smooth_adam_extend(a0, deadline, max_iters=500_000, eps0=1e-3, step0=0.2):
    a = a0.copy()
    m = np.zeros_like(a); v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    eps = eps0
    eps_min = 1e-8
    while time.time() < deadline and t < max_iters:
        t += 1
        grad = _smooth_grad(a, eps)
        m = 0.9*m + 0.1*grad
        v = 0.999*v + 0.001*(grad**2)
        m_hat = m/(1-0.9**t)
        v_hat = v/(1-0.999**t)
        eta = step0/(np.sqrt(v_hat)+1e-8)
        a = a - eta*m_hat
        a = _project_onto_simplex(a)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
        # Update epsilon
        if t % 20000 == 0:
            eps = max(eps*0.5, eps_min)
            # optionally also reduce step0?
        # occasional jitter
        if t % 60000 == 0:
            jitter = np.random.randn(a.size) * 5e-5
            a = best_a + jitter
            a = _project_onto_simplex(a)
    return best_a, best_score

   iterations and with an aggressively decreasing smoothing parameter
   `ε`.  Afterwards a plain sub‑gradient descent finishes the local optimisation.

3. **Long‑run polishing** – a cheap hill‑climbing phase moves tiny amounts of
   mass between indices guided by the exact sub‑gradient.  Random jitter
   escapes flat plateaus.

Only the *cheap* score is used during the search because it equals the true
objective for a sequence with `sum=1`.  The final list is returned; the
evaluation script will call `evaluate_sequence` on it.

The code respects the 1000 s budget, prints progress, and works for any
available previous solution `height_sequence_1`.

```python
"""
Search for a non‑negative step‑function that minimises

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The objective is scale‑invariant, therefore all candidates are kept on the
probability simplex (sum = 1).  The routine consists of:

1️⃣  Broad sampling of many analytic families (arcsine, Beta, power‑law,
    exponential, Gaussian, uniform, Dirichlet) for several lengths.

2️⃣  Heavy refinement of the best few seeds by an extended *smooth‑Adam*
    optimiser (log‑sum‑exp smoothing of the max‑convolution) with a very
    aggressive ε‑schedule, followed by plain sub‑gradient descent.

3️⃣  A final hill‑climbing phase that transfers tiny amounts of mass
    between indices guided by the exact sub‑gradient.

The whole procedure stops after the supplied time budget (default 1000 s)
and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of a and b using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap objective = 2·n·max(a*a) for a normalized (sum=1) sequence."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.
    Returns (grad, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with the active‑mask
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even positions (each appears twice)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _smooth_grad(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smoothed max using a log‑sum‑exp surrogate:

        f_eps(a) = eps * log Σ_i exp(conv_i/eps)
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)                    # length 2n‑1
    maxc = conv.max()
    w = np.exp((conv - maxc) / eps)                # unnormalised softmax
    w_sum = w.sum()
    w /= w_sum

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even positions
    diag = np.arange(conv.size)[(np.arange(conv.size) % 2) == 0] // 2
    grad[diag] -= w[2 * diag] * a[diag]

    return grad


# ----------------------------------------------------------------------
def _smooth_adam_extend(a0: np.ndarray, deadline: float,
                        max_iters: int = 800_000,
                        eps0: float = 1e-3,
                        step0: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp smoothed objective.
    ε is halved every 20 000 iterations down to 1e‑8.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    t = 0
    eps = eps0
    eps_min = 1e-8
    while time.time() < deadline and t < max_iters:
        t += 1
        grad = _smooth_grad(a, eps)

        # Adam updates
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad ** 2)
        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()

        # ε schedule
        if t % 20_000 == 0:
            eps = max(eps * 0.5, eps_min)

        # occasional jitter to leave shallow plateaus
        if t % 60_000 == 0:
            jitter = np.random.randn(a.size) * 5e-5
            a = best_a + jitter
            a = _project_onto_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 2_000_000,
                         eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Plain sub‑gradient descent with a diminishing step‑size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Analytic families -------------------------------------------------------

def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_w = (p - 1) * (np.log(x) + np.log(1.0 - x))
    log_w = np.clip(log_w, -700, 700)
    w = np.exp(log_w)
    if w.sum() == 0.0:
        return np.full_like(w, 1.0 / n)
    return w / w.sum()


def _powerlaw_distribution(n: int, exponent: float) -> np.ndarray:
    """Weight ∝ ((i+0.5)*(n-i-0.5))^{-exponent/2}."""
    i = np.arange(n, dtype=np.float64)
    base = (i + 0.5) * (n - i - 0.5)
    vals = np.power(base, -exponent / 2.0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _gaussian_distribution(n: int, sigma_frac: float = 0.30) -> np.ndarray:
    sigma = sigma_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if vals.sum() == 0.0:
        return np.full_like(vals, 1.0 / n)
    return vals / vals.sum()


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    """Dirichlet draw with concentration α."""
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function (non‑negative, sum ≥ 0.01) that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget,
    prints coarse progress information, and finally returns the best list
    of floats found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Base length (use previous solution if available)
    try:
        base_n = int(len(height_sequence_1))        # type: ignore[name-defined]
    except Exception:
        base_n = 3000

    # ------------------------------------------------------------------
    # Lengths to explore (moderate neighbourhood + a few random ones)
    lengths = set()
    for f in (0.9, 1.0, 1.1, 1.2, 1.4):
        n = max(500, int(base_n * f))
        lengths.add(min(n, 6000))
    for _ in range(5):
        n = np.random.randint(500, min(6000, base_n * 3) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # 1️⃣  Broad sampling
    candidates = []   # list of (array, cheap_score)

    beta_ps = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0]
    power_exps = [0.5, 0.7, 0.9, 1.1, 1.3]
    exp_lams = [0.001, 0.003, 0.006, 0.01]
    sigma_fracs = [0.18, 0.24, 0.30, 0.36]
    dir_alphas = [0.02, 0.05, 0.1, 0.2]

    for n in lengths:
        if time.time() >= deadline:
            break

        # arcsine (baseline)
        a = _arcsine_distribution(n)
        candidates.append((a, _max_convolution_score(a)))

        # symmetric beta
        for p in beta_ps:
            a = _beta_symmetric_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))

        # power‑law
        for e in power_exps:
            a = _powerlaw_distribution(n, e)
            candidates.append((a, _max_convolution_score(a)))

        # exponential
        for lam in exp_lams:
            a = _exponential_distribution(n, lam=lam)
            candidates.append((a, _max_convolution_score(a)))

        # Gaussian
        for sf in sigma_fracs:
            a = _gaussian_distribution(n, sigma_frac=sf)
            candidates.append((a, _max_convolution_score(a)))

        # uniform
        a = _uniform_distribution(n)
        candidates.append((a, _max_convolution_score(a)))

        # random Dirichlet draws
        for alpha in dir_alphas:
            for _ in range(2):
                a = _random_dirichlet(n, alpha=alpha)
                candidates.append((a, _max_convolution_score(a)))

        # jittered arcsine (helps escaping plateaus)
        a = _arcsine_distribution(n)
        jitter = np.random.randn(n) * 1e-4
        a = np.clip(a + jitter, 0, None)
        a = _project_onto_simplex(a, s=1.0)
        candidates.append((a, _max_convolution_score(a)))

    # ------------------------------------------------------------------
    # Insert the previous best solution, if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _uniform_distribution(prev.size)
            else:
                prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Keep a few most promising seeds
    candidates.sort(key=lambda x: x[1])
    top_k = 5
    seeds = [c[0] for c in candidates[:top_k]]
    best_seq = seeds[0].copy()
    best_score = _max_convolution_score(best_seq)
    print(f"[{time.time():.1f}] Selected {len(seeds)} seeds, best cheap={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Heavy refinement (smooth‑Adam → sub‑gradient)
    for idx, seed_seq in enumerate(seeds):
        if time.time() >= deadline:
            break

        # smooth‑Adam with aggressive ε‑schedule
        seq, sc = _smooth_adam_extend(seed_seq, deadline,
                                      max_iters=800_000,
                                      eps0=1e-3,
                                      step0=0.25)
        if sc < best_score - 1e-14:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time():.1f}] seed {idx+1}/{len(seeds)} smooth‑Adam → {sc:.6f}")

        # short sub‑gradient phase (if time left)
        sub_deadline = deadline - 20.0
        if time.time() < sub_deadline:
            seq2, sc2 = _subgradient_descent(best_seq, sub_deadline,
                                            max_iters=1_500_000,
                                            eta0=0.6)
            if sc2 < best_score - 1e-14:
                best_seq, best_score = seq2.copy(), sc2
                print(f"[{time.time():.1f}] sub‑gradient after seed {idx+1} → {sc2:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Long‑run polishing – gradient‑guided hill‑climbing
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    it = 0
    max_transfer = 0.03  # start with up to 3 % of the total mass

    while time.time() < deadline:
        it += 1
        # geometric decay of the transfer magnitude
        transfer = max_transfer * (0.5 ** (it / 10_000.0))

        # donor / receiver choice (every few steps we use the sub‑gradient)
        if it % 4 == 0:
            grad, _, _ = _subgrad(a)
            i = int(np.argmax(grad))
            j = int(np.argmin(grad))
            if i == j:
                i = np.random.randint(n)
                j = np.random.randint(n - 1)
                if j >= i:
                    j += 1
        else:
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_seq = a.copy()
                best_score = new_score
                print(f"[{time.time():.1f}] hill‑climb iter={it} new best {best_score:.6f}")

        # occasional jitter to jump out of flat regions
        if it % 5_000 == 0 and np.random.rand() < 0.2:
            jitter = np.random.randn(n) * 5e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a)
            cur_score = _max_convolution_score(a)

    print(f"Final cheap score {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
```
