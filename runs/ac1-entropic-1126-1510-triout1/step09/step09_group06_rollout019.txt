"""
search for step function minimizing evaluate_sequence.

Algorithm:
1. Sample analytic families (arcsine, triangular, exponential, gaussian, cauchy)
   and random Dirichlet draws for a few lengths (4000‑8000).
2. For each seed run a custom Frank‑Wolfe‑style mass‑transfer optimizer that
   moves probability mass from the index with the largest sub‑gradient of the
   max‑convolution to the index with the smallest, using an exact line‑search
   on the piecewise‑linear objective.
   The sub‑gradient is computed exactly (no erroneous diagonal correction)
   via FFT.  The convolution is updated incrementally and periodically
   recomputed for numerical stability.
3. Keep the best vector seen across all seeds and return it.
The routine respects a 1000 s budget, uses only NumPy and the supplied
`evaluate_sequence` function, and never writes to disk.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate for `evaluate_sequence` when Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _gaussian_distribution(n: int, sigma_factor: float = 0.2) -> np.ndarray:

    centre = (n - 1) / 2.0
    sigma = sigma_factor * n

def _subgrad_vector(a: np.ndarray, b: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of the max‑convolution functional.
    Returns a vector of length n (the gradient of max(b) on the simplex).
    The active indices are supplied to avoid recomputing them.
    """
    n = a.shape[0]
    if active.size == 0:
        return np.zeros_like(a)
    w = np.zeros_like(b)
    w[active] = 1.0 / active.size
    conv = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv[len(b) - 1: len(b) - 1 + n]
    return grad

def _flatten_mass_transfer(a0: np.ndarray,
                          deadline: float,
                          max_iters: int = 20_000,
                          recompute_interval: int = 500) -> np.ndarray:
    """
    Frank‑Wolfe‑style pairwise mass transfer that directly minimises the
    max‑convolution.  The method moves mass from the index with the largest
    sub‑gradient to the index with the smallest, using an exact line search on
    the piecewise‑linear objective.  The convolution is kept incrementally
    (O(n) per step) and recomputed periodically to remove accumulated error.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _fft_self_convolve(a)            # full self‑convolution
    cur_max = b.max()
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        eps = cur_max * 1e-12 + 1e-15
        active = np.flatnonzero(b >= cur_max - eps)

        grad = _subgrad_vector(a, b, active)
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))

        if i_src == i_dst:
            break

        # construct the linear change of the convolution when moving δ from src to dst
        diff = np.zeros_like(b)
        diff[i_dst:i_dst + n] = a
        diff[i_src:i_src + n] -= a
        diff = 2.0 * diff                     # derivative factor 2

        max_active_delta = diff[active].max()
        # we need the move to *decrease* the current maximum
        if max_active_delta >= -1e-12:
            # try the opposite direction
            i_src, i_dst = i_dst, i_src
            diff = np.zeros_like(b)
            diff[i_dst:i_dst + n] = a
            diff[i_src:i_src + n] -= a
            diff = 2.0 * diff
            max_active_delta = diff[active].max()
            if max_active_delta >= -1e-12:
                break

        delta = min(a[i_src], 1.0 - a[i_dst])   # feasible amount of mass to move

        # line‑search: smallest positive s where some inactive index catches up
        mask = np.ones(b.shape[0], dtype=bool)
        mask[active] = False
        inactive = np.nonzero(mask)[0]

        if inactive.size > 0:
            denom = diff[inactive] - max_active_delta
            pos = denom > 1e-12
            if np.any(pos):
                s_candidates = (cur_max - b[inactive[pos]]) / denom[pos]
                s_opt = s_candidates.min()
                step = min(delta, s_opt * 0.99)   # stay a little away from the breakpoint
            else:
                step = delta
        else:
            step = delta

        if step <= 1e-15:
            break

        # update the coefficient vector
        a[i_src] -= step
        a[i_dst] += step

        # update the convolution (exact linear part)
        b += step * diff
        cur_max = b.max()

        # periodic exact recomputation to keep numerical errors bounded
        if it % recompute_interval == 0:
            b = _fft_self_convolve(a)
            cur_max = b.max()
    # final exact convolution (ensures consistency)
    _ = _fft_self_convolve(a)  # discard – only to force a final recompute if needed
    return a

"""
Search for a step‑function that minimises `evaluate_sequence`.

The algorithm works as follows:

1. **Generate seeds** – for a few moderately large lengths (4000‑8000)
   we create analytic families (arcsine, triangular, exponential,
   Gaussian, Cauchy) and a handful of random Dirichlet draws.  All vectors
   are ℓ₁‑normalised.

2. **Short projected sub‑gradient descent** – each seed is briefly refined
   with a simple projected sub‑gradient method that uses the *exact*
   sub‑gradient of `max(a*a)` (the previous implementation had an
   erroneous diagonal correction).

3. **Frank‑Wolfe–style mass transfer** – from the refined vector we run a
   dedicated optimizer that repeatedly moves probability mass from the
   index with the largest sub‑gradient to the index with the smallest,
   using an exact line‑search on the piecewise‑linear objective
   `max(conv(a,a))`.  The convolution is kept incrementally (O(n) per
   iteration) and recomputed periodically for numerical stability.

4. The best vector over all lengths and seeds is returned.
The whole procedure respects the 1000 s time budget, uses only NumPy,
and works directly on the simplex (non‑negative with unit sum), which is
exactly the quantity minimised by `evaluate_sequence`.

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate for `evaluate_sequence` when Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _gaussian_distribution(n: int, sigma_factor: float = 0.2) -> np.ndarray:

    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    w /= w.sum()
    return w


def _cauchy_distribution(n: int, scale_factor: float = 0.2) -> np.ndarray:

    scale = scale_factor * n
    w = 1.0 / (1.0 + ((idx - centre) / scale) ** 2)
    w /= w.sum()

# ----------------------------------------------------------------------
def _subgrad_vector(a: np.ndarray, b: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv(a,a)).  `active` are the indices where
    `b` attains its maximum.  The gradient is a convex combination of the
    individual gradients of those active convolution entries.
    """
    n = a.shape[0]
    if active.size == 0:
        return np.zeros_like(a)
    w = np.zeros_like(b)
    w[active] = 1.0 / active.size
    conv = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv[len(b) - 1 : len(b) - 1 + n]
    return grad

def _flatten_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 20_000,
    recompute_interval: int = 500,
) -> np.ndarray:
    """
    Frank‑Wolfe‑style pairwise mass transfer that directly minimises
    `max(conv(a,a))`.  The algorithm moves mass from the index with the
    largest sub‑gradient to the index with the smallest, performing an
    exact line‑search on the piecewise‑linear objective.
    The convolution is updated incrementally (O(n) per iteration) and
    periodically recomputed exactly to keep numerical error bounded.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _fft_self_convolve(a)            # exact convolution
    cur_max = b.max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        eps = cur_max * 1e-12 + 1e-15
        active = np.flatnonzero(b >= cur_max - eps)

        grad = _subgrad_vector(a, b, active)
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))

        if i_src == i_dst:
            break

        # linear change of convolution when moving δ from i_src to i_dst
        diff = np.zeros_like(b)
        diff[i_dst : i_dst + n] = a
        diff[i_src : i_src + n] -= a
        diff = 2.0 * diff                     # factor 2 from derivative

        max_active_delta = diff[active].max()
        if max_active_delta >= -1e-12:
            # try the opposite direction
            i_src, i_dst = i_dst, i_src
            diff = np.zeros_like(b)
            diff[i_dst : i_dst + n] = a
            diff[i_src : i_src + n] -= a
            diff = 2.0 * diff
            max_active_delta = diff[active].max()
            if max_active_delta >= -1e-12:
                break

        delta = min(a[i_src], 1.0 - a[i_dst])   # feasible amount to transfer

        # line‑search: smallest positive step where an inactive index catches up
        mask = np.ones(b.shape[0], dtype=bool)
        mask[active] = False
        inactive = np.nonzero(mask)[0]

        if inactive.size > 0:
            denom = diff[inactive] - max_active_delta
            pos = denom > 1e-12
            if np.any(pos):
                s_cand = (cur_max - b[inactive[pos]]) / denom[pos]
                s_opt = s_cand.min()
                step = min(delta, s_opt * 0.99)   # stay a little away from the breakpoint
            else:
                step = delta
        else:
            step = delta

        if step <= 1e-15:
            break

        # apply the transfer
        a[i_src] -= step
        a[i_dst] += step
        b += step * diff
        cur_max = b.max()

        # periodic exact recomputation
        if it % recompute_interval == 0:
            b = _fft_self_convolve(a)
            cur_max = b.max()

    # final exact recomputation (ensures consistency)
    _ = _fft_self_convolve(a)
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Entry‑point for the grader.  Generates a few candidate vectors,
    refines each with the mass‑transfer optimizer and returns the best
    one found before the time limit expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5    # safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # Lengths to explore – larger lengths give a lower theoretical bound.
    lengths = [4000, 6000, 8000]

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----------------------------------------------------------------
        # Build a small pool of diverse seeds
        seeds = [
            _arcsine_distribution(n),
            _triangular_distribution(n),
            _exponential_distribution(n, lam=0.005),
            _gaussian_distribution(n, sigma_factor=0.15),
            _cauchy_distribution(n, scale_factor=0.15),
        ]

        # random Dirichlet draws
        for alpha in (0.2, 0.5, 1.0, 2.0):
            seeds.append(_random_dirichlet_alpha(n, alpha))

        # previous best sequence, if available and long enough
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.size >= n:
                a = prev[:n].copy()
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 0:
                    a = a / s
                    seeds.append(a)
        except Exception:
            pass

        # ----------------------------------------------------------------
        # Normalise and filter seeds
        candidates = []
        for s_vec in seeds:
            a = np.asarray(s_vec, dtype=np.float64)
            if a.ndim != 1 or a.size == 0:
                continue
            a = np.maximum(a, 0.0)
            tot = a.sum()
            if tot < 1e-12:
                continue
            a = a / tot
            candidates.append(a)

        # Keep the three seeds with the lowest surrogate score
        scored = [(a, _max_convolution_score(a)) for a in candidates]
        scored.sort(key=lambda x: x[1])
        top_seeds = [a for a, _ in scored[:3]]

        # ----------------------------------------------------------------
        # Optimise each seed
        for seed_vec in top_seeds:
            if time.time() >= deadline:
                break

            # short projected sub‑gradient descent (few hundred iterations)
            a = seed_vec.copy()
            for it in range(300):
                b = _fft_self_convolve(a)
                cur_max = b.max()
                eps = cur_max * 1e-12 + 1e-15
                active = np.flatnonzero(b >= cur_max - eps)
                grad = _subgrad_vector(a, b, active)
                step = 0.5 / np.sqrt(it + 1)
                a = a - step * grad
                a = _project_onto_simplex(a, s=1.0)

            # Frank‑Wolfe mass‑transfer refinement
            a = _flatten_mass_transfer(a, deadline)

            score = _max_convolution_score(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] length={n} new best score={best_score:.6f}")

    # --------------------------------------------------------------------
    # Fallback if nothing was produced (should never happen)
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _max_convolution_score(best_seq)

    # Final safe projection onto the simplex
    best_seq = _project_onto_simplex(best_seq, s=1.0)

    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
