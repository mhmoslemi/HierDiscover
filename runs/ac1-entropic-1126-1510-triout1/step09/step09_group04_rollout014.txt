"""
propose_candidate
-----------------

Search for a non‑negative step‑function (list of floats) that minimises
``evaluate_sequence``.  The method is a lightweight but fairly
aggressive meta‑heuristic:

1.  Scan a collection of analytic families (uniform, arcsine,
    triangular, exponential, Beta, sine‑power, cosine‑power,
    Kaiser, DPSS) for a few lengths and keep the best normalised vector.
2.  Run a smooth‑max mirror‑descent (exponentiated‑gradient) on the
    simplex to push the self‑convolution down.
3.  Apply a few sub‑gradient (momentum) steps for polishing.
4.  Perform a **sparse‑LP refinement**: solve a small LP that
    maximises Σg_i under the active self‑convolution constraints,
    then mix the solution back into the current vector.
5.  Finish with a cheap hill‑climber that repeatedly moves a tiny
    amount of mass away from the pair that dominates the current
    maximum.
6.  Return the best vector found (as a plain ``list[float]``).

All phases respect the global wall‑clock budget (default 1000 s) and
periodically print the current best objective value.
"""

import time
import random
import math
from typing import List

import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog
from scipy.signal import windows

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Returns the quantity 2·n·max(a*a) for a vector that has already
    been normalised to sum = 1.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.01) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    # cosine runs from 1 (at i=0) down to -1 (at i=n‑1); shift to non‑negative
    vals = (np.cos(np.pi * idx / (n - 1)) + 1) / 2.0
    vals = vals ** power
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float = 0.3) -> np.ndarray:
    """Discrete Gaussian centred at (n‑1)/2."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_kaiser(n: int, beta: float = 5.0) -> np.ndarray:
    vals = windows.kaiser(n, beta, sym=False).astype(np.float64)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_dpss(n: int, NW: float = 2.5) -> np.ndarray:
    # SciPy returns a (n,) array; enforce non‑negativity (the first DPSS is >=0)
    vals = windows.dpss(n, NW, Kmax=1, sym=False).astype(np.float64).ravel()
    vals = np.abs(vals)                 # safeguard against tiny negative round‑off
    vals /= vals.sum()
    return vals

def _lp_refine_sparse(f: np.ndarray, mix_factor: float,
                      deadline: float, max_constraints: int = 200) -> tuple[np.ndarray, float]:
    """
    One LP‑based refinement step using only the most critical convolution
    constraints (those with largest values).  Returns a potentially
    improved vector (still normalised to sum = 1) and its max‑convolution.
    """
    n = f.shape[0]
    conv_f = _fft_self_convolve(f)
    M = conv_f.max()

    # pick the largest entries (up to max_constraints)
    ind = np.argpartition(-conv_f, min(max_constraints, conv_f.size - 1))[:max_constraints]

    # Build the (small) constraint matrix T such that (T @ g)[t] = (f * g)[t]
    rows = len(ind)
    T = np.zeros((rows, n), dtype=np.float64)
    for r, t in enumerate(ind):
        lo = max(0, t - (n - 1))
        hi = min(n - 1, t)
        # for i in [lo, hi] we have g_i multiplied by f[t-i]
        T[r, lo:hi + 1] = f[t - np.arange(lo, hi + 1)]

    # Linear program: maximise Σ g_i  <=> minimise –Σ g_i
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(rows, M, dtype=np.float64)

    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=(0, None), method='highs')
    except Exception as e:
        # LP failed – just return the original vector
        return f, M

    if not res.success:
        return f, M

    g = res.x
    gsum = g.sum()
    if gsum < 1e-12:
        return f, M

    g_norm = g / gsum

    # try a few mixing ratios
    for t in (mix_factor, mix_factor * 0.5, mix_factor * 0.25,
              mix_factor * 0.1, mix_factor * 0.05):
        new_f = (1 - t) * f + t * g_norm
        new_f = np.maximum(new_f, 1e-12)
        new_f /= new_f.sum()
        new_M = _fft_self_convolve(new_f).max()
        if new_M < M - 1e-12:
            return new_f, new_M

    return f, M

    Search for a step‑function (list of non‑negative floats) that minimises

    returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    eps = 1e-12
    best_seq = None
    best_score = float('inf')
    best_n = None

    # --------------------------------------------------------------
    # 1.  Scan analytic families for a handful of lengths.
    lengths = [1800, 2400, 3000, 3600, 4200, 4800]   # moderate sizes
    for n in lengths:
        if time.time() >= deadline:
            break

        candidates = []

        # --- families -------------------------------------------------
        candidates.append(_distribution_uniform(n))
        candidates.append(_distribution_arcsine(n))
        candidates.append(_distribution_triangular(n))
        candidates.append(_distribution_exponential(n, lam=0.004))
        candidates.append(_distribution_exponential(n, lam=0.02))

        # beta with a wide span (including very U‑shaped)
        for p in (0.05, 0.1, 0.2, 0.4, 0.8, 1.5, 3.0):
            candidates.append(_distribution_beta_sym(n, p))

        # sine / cosine powers
        for pw in (0.5, 1.0, 2.0, 4.0):
            candidates.append(_distribution_sine_power(n, pw))
            candidates.append(_distribution_cosine_power(n, pw))

        # Gaussian / Kaiser / DPSS
        for sigma in (0.15, 0.25, 0.35):
            candidates.append(_distribution_gaussian(n, sigma))
        for beta_kaiser in (3.0, 5.0, 8.0):
            candidates.append(_distribution_kaiser(n, beta_kaiser))
        for NW in (1.5, 2.0, 2.5, 3.0):
            candidates.append(_distribution_dpss(n, NW))

        # random Dirichlet draws – a few alphas
        for alpha in (0.05, 0.2, 0.5, 1.0, 2.0):
            candidates.append(np.random.dirichlet(np.full(n, alpha)))

        # -----------------------------------------------------------
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] init n={n}  score {sc:.6f}")

    # --------------------------------------------------------------
    # 2.  Bring forward any externally supplied sequence.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_seq = prev.copy()
                best_n = prev.size
                best_score = sc
                print(f"[{time.time():.1f}] external best {sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3.  Fallback – uniform vector.
    if best_seq is None:
        best_n = 2400
        best_seq = _distribution_uniform(best_n)
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # --------------------------------------------------------------
    # 4.  Smooth‑max mirror descent (exponentiated gradient)
    a = best_seq.copy()
    n = a.shape[0]

    # schedule – slowly decay the smoothness (α) and learning rate (lr)
    schedule = [
        (1.0, 200.0, 2000, 0.15, 0.015),   # (α_start, α_end, steps, lr_start, lr_end)
        (200.0, 5000.0, 3000, 0.015, 0.0015),
    ]
    for a0, a1, steps, lr0, lr1 in schedule:
        if time.time() >= deadline:
            break
        for it in range(steps):
            if time.time() >= deadline:
                break
            t = it / steps
            alpha = a0 * (a1 / a0) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            # smooth‑max weights
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of the smoothed max w.r.t a (see derivation in original code)
            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if it % 500 == 0 or it == steps - 1:
                sc = _max_convolution_score(a)
                if sc < best_score - 1e-12:
                    best_seq = a.copy()
                    best_score = sc
                    best_n = n
                    print(f"[{time.time():.1f}] mirror it={it}  best {sc:.6f}")

    # --------------------------------------------------------------
    # 5.  Sub‑gradient descent with momentum (polishing)
    a = best_seq.copy()
    n = a.shape[0]
    velocity = np.zeros_like(a)
    beta = 0.85
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        eps_rel = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - eps_rel)
        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = 0.20 / np.sqrt(it)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()

        sc = _max_convolution_score(a)
        if sc < best_score - 1e-12:
            best_seq = a.copy()
            best_score = sc
            print(f"[{time.time():.1f}] subgrad it={it}  best {sc:.6f}")

        if time.time() + 0.4 > deadline:
            break

    # --------------------------------------------------------------
    # 6.  Sparse LP refinement (few iterations)
    a = best_seq.copy()
    n = a.shape[0]
    mix = 0.3
    lp_iters = 5
    while time.time() < deadline and lp_iters > 0:
        a, M = _lp_refine_sparse(a, mix, deadline, max_constraints=150)
        sc = 2.0 * n * M
        if sc < best_score - 1e-12:
            best_seq = a.copy()
            best_score = sc
            print(f"[{time.time():.1f}] LP refine  best {sc:.6f}")
        lp_iters -= 1
        mix = max(0.01, mix * 0.85)

    # --------------------------------------------------------------
    # 7.  Final cheap hill‑climber (mass‑transfer moves)
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    while time.time() < deadline:
        it += 1

        # identify the index achieving the current maximum
        k = int(conv.argmax())
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        # find the pair (i,j) with largest product in this slice
        best_i, best_j = i_low, k - i_low
        max_prod = a[best_i] * a[best_j]
        for i in range(i_low, i_high + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a_snapshot = a.copy()
        a[source] -= delta
        a[target] += delta
        a = np.maximum(a, eps)
        a /= a.sum()
        conv = _fft_self_convolve(a)
        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
            sc = 2.0 * n * M
            if sc < best_score - 1e-12:
                best_seq = a.copy()
                best_score = sc
                if it % 500 == 0:
                    print(f"[{time.time():.1f}] hill it={it}  best {sc:.6f}")
        else:
            a = a_snapshot
            conv = _fft_self_convolve(a)

        if it % 4000 == 0:
            # occasional jitter to escape plateaus
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] jitter applied at hill iter {it}")

    final_score = 2.0 * n * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
