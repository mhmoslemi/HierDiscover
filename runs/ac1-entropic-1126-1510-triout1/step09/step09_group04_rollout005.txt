"""
Propose a step‑function (non‑negative list) that minimises ``evaluate_sequence``.

Algorithm
---------
1. Initialise a pool of candidate vectors coming from several analytic families
   (uniform, arcsine, triangular, sine‑power, symmetric Beta, exponential,
   Gaussian) and a few random Dirichlet draws for several lengths.
   Keep the best one (lowest 2·n·max(conv)).

2. Refine the best candidate with a *soft‑max* mirror‑descent on the simplex.
   A smooth approximation of the max (soft‑max with temperature α) is
   differentiable; we update the vector with an Adam‑style exponentiated‑
   gradient (multiplicative update) while projecting back onto the simplex.

3. Perform a cheap hill‑climber that repeatedly moves a small amount of mass
   from the pair that contributes most to the current maximal convolution
   entry toward a random position.  Improvements are kept; occasional jitter
   helps escape plateaus.

The whole procedure respects a wall‑clock budget (default 1000 s).  All
intermediate best scores are printed; the final best vector (as plain Python
floats) is returned.
"""

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families – all return a vector with sum = 1
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.01) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at (n‑1)/2."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

        for pw in (0.5, 1.0, 2.0, 3.0, 4.0):
            candidates.append(_distribution_sine_power(n, pw))
        for p in (0.2, 0.4, 0.6, 1.0, 2.0, 4.0):
            candidates.append(_distribution_beta_sym(n, p))
        for lam in (0.005, 0.010, 0.020):
            candidates.append(_distribution_exponential(n, lam))
        for sigma in (0.2, 0.35, 0.5):
            candidates.append(_distribution_gaussian(n, sigma))

        # few random Dirichlet draws
        for alpha in (0.2, 0.5, 1.0, 2.0):
            for _ in range(3):
                candidates.append(np.random.dirichlet(np.full(n, alpha)))

        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_score = sc
                best_n = n
                print(f"[{time.time() - start_time:.1f}s] init n={n} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2.  Fallback uniform if nothing found yet.
    if best_seq is None:
        best_n = 2000
        best_seq = _distribution_uniform(best_n)
        best_seq = np.maximum(best_seq, eps)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time() - start_time:.1f}s] fallback uniform n={best_n} score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3.  Adam‑based exponentiated‑gradient (soft‑max) refinement.
    a = best_seq.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # scheduling parameters (exponential interpolation)
    total_iters = 20000
    # bound by remaining time (roughly 0.003 s per iteration)
    max_possible = max(0, int((deadline - time.time()) / 0.003))
    total_iters = min(total_iters, max_possible)
    if total_iters < 500:
        total_iters = max_possible

    if total_iters > 0:
        alpha_start = 0.5
        alpha_end = 5000.0
        lr_start = 0.8
        lr_end = 0.001

        log_alpha_start = math.log(alpha_start)
        log_alpha_end = math.log(alpha_end)
        log_lr_start = math.log(lr_start)
        log_lr_end = math.log(lr_end)

        eval_every = max(200, total_iters // 200)

        for t in range(1, total_iters + 1):
            if time.time() >= deadline:
                break

            frac = t / total_iters
            alpha = math.exp(log_alpha_start + frac * (log_alpha_end - log_alpha_start))
            lr = math.exp(log_lr_start + frac * (log_lr_end - log_lr_start))

            conv = _fft_self_convolve(a)
            maxc = conv.max()

            # soft‑max weights over convolution entries
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0 or np.isnan(w_sum):
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of the smooth max w.r.t. a
            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

            # Adam bias‑corrected moments
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # exponentiated‑gradient update
            a = a * np.exp(-step)
            a = np.maximum(a, eps)
            a = _project_onto_simplex(a, 1.0)

            if t % eval_every == 0:
                score = 2.0 * n * conv.max()
                if score < best_score:
                    best_seq = a.copy()
                    best_score = score
                    print(f"[{time.time() - start_time:.1f}s] Adam it={t} score={score:.6f}")

    # --------------------------------------------------------------
    # 4.  Simple hill‑climber (mass‑transfer moves).
    a = best_seq.copy()
    n = a.shape[0]

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        k = int(conv.argmax())

        # Find the pair (i,j) attaining this max entry.
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        best_i, best_j = i_low, k - i_low
        best_prod = a[best_i] * a[best_j]
        for i in range(i_low, i_high + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > best_prod:
                best_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        # move a small chunk
        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a_old = a.copy()
        a[source] -= delta
        a[target] += delta
        a = np.maximum(a, eps)
        a /= a.sum()

        new_max = _fft_self_convolve(a).max()
        if new_max < cur_max - 1e-12:
            # improvement
            score = 2.0 * n * new_max
            if score < best_score:
                best_seq = a.copy()
                best_score = score
                print(f"[{time.time() - start_time:.1f}s] hill it={it} new best {score:.6f}")
        else:
            # revert
            a = a_old

        # occasional jitter to escape flat plateaus
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            print(f"[{time.time() - start_time:.1f}s] jitter applied at hill iter {it}")

    final_score = 2.0 * n * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
