def _guided_hill_climb(a0, deadline, max_iters=200_000):
    a = a0.copy()
    cur_score = _max_convolution_score(a)
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # compute subgradient to guide direction
        grad, _, _ = _subgrad(a)
        # find i with max positive grad, j with min negative grad
        pos_idx = np.argmax(grad)
        neg_idx = np.argmin(grad)
        # Determine max transferable mass from pos idx (must stay >=0)
        max_transfer = min(0.01 * a[pos_idx], a[pos_idx])  # 1% of mass
        if max_transfer <= 0:
            # fallback random move
            i = np.random.randint(n)
            j = np.random.randint(n-1)
            if j >= i: j+=1
            delta = np.random.rand() * a[i] * 0.01
            i, j = i, j
        else:
            delta = max_transfer * np.random.rand()
            i, j = pos_idx, neg_idx
        if i == j or delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)
        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = a_new.copy()
                # print progress
                if it % 2000 == 0:
                    print(f"[{time.time():.1f}] guided hill iter={it} best={best_score:.6f}")
        # occasional random jitter
        if it % 5000 == 0 and np.random.rand() < 0.2:
            jitter = np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a + jitter, s=1.0)
            cur_score = _max_convolution_score(a)
    return best_a, best_score

def _lp_refine(a0, deadline, max_iters=20, step=0.5):
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # compute current M
        b_conv = _fft_self_convolve(a)
        M = b_conv.max()
        # solve LP
        n = a.shape[0]
        offsets = np.arange(n, dtype=int)
        # create list of data arrays
        data = [np.full(n, a[k], dtype=np.float64) for k in range(n)]
        A_ub = sparse.diags(data, offsets, shape=(2*n-1, n), format='csr')
        b_ub = np.full(2*n-1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs', options={'presolve': True, 'time_limit': max(0.5, (deadline - time.time())/5)})
        if not res.success:
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-12:
            # no improvement possible
            break
        g = b / sum_b
        # try replace a with g
        new_a = g
        new_score = _max_convolution_score(new_a)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = new_a.copy()
            a = new_a
            print(f"[{time.time():.1f}] LP iter {it} new best cheap {best_score:.6f}")

        # else combine with convex combo

            best_score = cur_score
            best_a = a.copy()

#!/usr/bin/env python
"""
A multi‑phase optimiser for ``evaluate_sequence``.

1️⃣  Coarse sweep over a few analytically motivated families
    (arcsine, smoothed arcsine, product‑power, symmetric / asymmetric Beta,
    exponential tails, random Dirichlet) on several nearby lengths.
    The best cheap candidates (2·n·max(conv)) are kept.

2️⃣  Heavy refinement of the retained seeds:
    * line‑search sub‑gradient descent (projected onto the simplex)
    * Adam and exponentiated‑gradient are omitted – the line‑search variant
      adaptively chooses a step that actually lowers the cheap objective.

3️⃣  Linear‑programming refinement:  given the current vector ``a`` and its
    maximal self‑convolution ``M``, solve

        maximise   Σ b_j
        subject to (a * b)_i ≤ M ,   b ≥ 0

    (a * b) is linear in ``b``; the LP is built with a sparse Toeplitz matrix.
    The solution ``b`` is normalised to sum 1 and used as a new iterate.
    A small convex combination is performed if the replacement does not improve.

4️⃣  Guided hill‑climbing – the sub‑gradient tells from which index mass should
    be moved to which other index.  Tiny transfers are accepted only when they
    lower the objective.

5️⃣  Random restarts with a short descent when time permits.

All steps respect the global deadline (≈1000 s) and keep the vector non‑negative
with unit sum (the objective is scale‑invariant, therefore the exact magnitude
does not matter).  The returned list can be fed directly into ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap surrogate = 2·n·max(a*a) assuming Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.
    Returns (grad, cur_max, active_set).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (k = 2*i)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
# families of probability vectors
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _smoothed_arcsine_distribution(n: int, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5 + eps) * (n - i - 0.5 + eps))
    return vals / vals.sum()


def _product_power_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-alpha)
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, p: float, q: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    w = np.where(np.isfinite(w), w, 0.0)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Γ–Dirichlet, α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()

# optimisation primitives
def _line_search_subgrad_descent(a0: np.ndarray,
                                 deadline: float,
                                 max_iters: int = 200_000,
                                 eta0: float = 1.0):
    """
    Projected sub‑gradient descent with a simple back‑tracking line search.
    Returns the best vector found and its cheap score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        direction = -grad
        step = eta0 / np.sqrt(it)

        # up to five back‑tracking attempts
        for _ in range(5):
            cand = a + step * direction
            cand = _project_onto_simplex(cand, s=1.0)
            score = _max_convolution_score(cand)
            if score < best_score - 1e-15:
                a = cand
                best_a = cand.copy()
                best_score = score
                break
            step *= 0.5
        # if none of the attempts improved, keep the current a
    return best_a, best_score


def _guided_hill_climb(a0: np.ndarray,
                       deadline: float,
                       max_iters: int = 150_000):
    """
    Small mass transfers guided by the sub‑gradient.
    Accept only strictly improving moves.
    """
    a = a0.copy()
    cur_score = _max_convolution_score(a)
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)

        # indices with biggest positive / negative gradient components
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            # fall back to random pair
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # move a tiny fraction of the mass from i to j
        max_transfer = min(0.02 * a[i], a[i])  # at most 2 % of the donor
        if max_transfer <= 0.0:
            continue
        delta = np.random.rand() * max_transfer

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = _max_convolution_score(a_new)
        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = a_new.copy()
                if it % 2000 == 0:
                    print(f"[{time.time():.1f}] guided hill iter={it} best={best_score:.6f}")

        # occasional jitter to escape flat plateaux
        if it % 5000 == 0 and np.random.rand() < 0.2:
            jitter = np.random.randn(n) * 5e-5
            a = _project_onto_simplex(best_a + jitter, s=1.0)
            cur_score = _max_convolution_score(a)

    return best_a, best_score


def _lp_refine(a0: np.ndarray,
               deadline: float,
               max_iters: int = 12,
               mix_step: float = 0.5):
    """
    Linear‑programming refinement.
    Repeatedly:
        * compute M = max(a*a)
        * solve maximise Σ b  subject to (a*b) ≤ M , b ≥ 0
        * set g = b / Σb   (normalise)
        * either replace a by g (if it improves) or move partway towards g.
    Returns best vector found and its cheap score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    n = a.shape[0]
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # current bound M
        M = _fft_self_convolve(a).max()

        # build the Toeplitz constraint matrix A (size (2n‑1) × n)
        offsets = np.arange(n, dtype=int)
        # each diagonal k contains the constant a[k]
        diag_data = [np.full(n, a[k], dtype=np.float64) for k in range(n)]
        A_ub = sparse.diags(diag_data, offsets, shape=(2 * n - 1, n), format='csr')
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # maximise Σ b  → minimise -Σ b
        c = -np.ones(n, dtype=np.float64)

        # solve LP
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=(0, None), method='highs')

        if not res.success:
            # LP failed – abort refinement
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-12:
            # cannot improve further
            break

        g = b / sum_b          # normalised candidate

        # try pure replacement
        cand_score = _max_convolution_score(g)
        if cand_score < best_score - 1e-12:
            a = g
            best_score = cand_score
            best_a = g.copy()
            print(f"[{time.time():.1f}] LP iter {it} pure replace cheap {best_score:.6f}")
            continue

        # otherwise convex blend
        a = (1 - mix_step) * a + mix_step * g
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] LP iter {it} blended cheap {best_score:.6f}")

    return best_a, best_score
