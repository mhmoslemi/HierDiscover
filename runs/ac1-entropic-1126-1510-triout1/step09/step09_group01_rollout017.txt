def smooth_max_optimize(a, deadline, best_seq, best_score):
    n = a.shape[0]
    eps = 1e-12
    # schedule steps: total 5000 or based on time left.
    # Determine allowed steps: (deadline - time.time()) / estimated_time_per_step.
    # Rough per step cost: conv + weight + gradient (several FFTs). Assume 0.005 sec.
    est_step = 0.008
    max_steps = int((deadline - time.time()) / est_step)
    max_steps = max(0, min(max_steps, 8000))
    # define schedule of alphas and lr.
    alpha_start = 1.0
    alpha_end = 1e7
    lr_start = 0.2
    lr_end = 1e-4
    for step in range(max_steps):
        if time.time() >= deadline: break
        t = step / max_steps
        alpha = alpha_start * ((alpha_end / alpha_start) ** t)
        lr = lr_start * ((lr_end / lr_start) ** t)
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # compute softmax weights
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient
        grad = 2.0 * _fft_convolve(a[::-1], w)[n-1:n-1+n]
        # exponentiated update
        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()
        if step % 400 == 0:
            cur_sc = _max_convolution_score(a)
            if cur_sc < best_score - 1e-12:
                best_score = cur_sc
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth_max step={step} new best {cur_sc:.6f}")
    return best_seq, best_score

def subgradient_optimize(a, deadline, best_seq, best_score):
    n = a.shape[0]
    eps = 1e-12
    velocity = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999  # not used maybe
    step0 = 0.5
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)
        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n-1:n-1+n]
        # Adam momentum
        velocity = beta1 * velocity + (1 - beta1) * grad
        v_hat = velocity / (1 - beta1 ** it)
        # backtracking line search
        step = step0
        while True:
            a_candidate = a - step * v_hat
            a_candidate = _project_onto_simplex(a_candidate, 1.0)
            a_candidate = np.maximum(a_candidate, eps)
            a_candidate /= a_candidate.sum()
            cand_sc = _max_convolution_score(a_candidate)
            if cand_sc < best_score - 1e-12 or step < 1e-8:
                break
            step *= 0.5
        a = a_candidate
        if cand_sc < best_score - 1e-12:
            best_score = cand_sc
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {cand_sc:.6f}")
        if it % 500 == 0:
            # jitter to escape plateaus
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            print(f"[{time.time():.1f}] jitter applied at subgrad it {it}")
        if time.time() + 0.5 > deadline:
            break
    return best_seq, best_score

def hill_climb_optimize(a, deadline, best_seq, best_score):
    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    while time.time() < deadline:
        it += 1
        # pick index that contributed to max M
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # find pair (i,j) in range [i_start,i_end] maximizing product
        # but we can approximate by picking i = argmax a[i] within range.
        # For speed, we can just pick i = np.argmax(a[i_start:i_end+1]) + i_start, j=k-i.
        idx_range = slice(i_start, i_end+1)
        sub_a = a[idx_range]
        i_local = np.argmax(sub_a)
        i = i_start + i_local
        j = k - i
        # choose source = i (if a[i] larger) else j
        source = i if a[i] >= a[j] else j
        target = random.randint(0, n-1)
        if target == source:
            continue
        max_delta = min(a[source] * 0.4, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue
        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M
            best_seq = a.copy()
            best_score = 2.0 * n * M
            if it % 200 == 0:
                print(f"[{time.time():.1f}] hill it={it} new best {best_score:.6f}")
        if it % 2000 == 0:
            # jitter
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] hill jitter at it {it}")
    return best_seq, best_score
```

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) for a vector already normalised to sum = 1.
    This is the value returned by ``evaluate_sequence`` (ignoring the
    clipping that never fires for our candidates).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families (all return a vector that sums to 1) ----------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_exponential_left(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * idx)
    vals /= vals.sum()
    return vals


def _distribution_powerlaw(n: int, exponent: float) -> np.ndarray:
    # decreasing power‑law (i+1)^{-exponent}
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = idx ** (-exponent)
    vals /= vals.sum()
    return vals


def _distribution_beta_asym(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # work in log‑space to avoid overflow/underflow
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    w = np.exp(logw)
    w[~np.isfinite(w)] = 0.0
    if w.sum() == 0:
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_logistic(n: int, midpoint: float, steep: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    vals = 1.0 / (1.0 + np.exp(-steep * (x - midpoint)))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian_offset(n: int, sigma: float, offset: float) -> np.ndarray:
    """Discrete Gaussian of relative width sigma, centred at offset∈[0,1]."""
    x = np.arange(n, dtype=np.float64)
    centre = offset * (n - 1)
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n: int, coeff: float = 0.5) -> np.ndarray:
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_triangular_left(n: int) -> np.ndarray:
    """Linear decrease from the left edge to zero."""
    vals = np.linspace(1.0, 0.0, n, dtype=np.float64)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

# Optimisation primitives -----------------------------------------------
def smooth_max_optimize(a: np.ndarray, deadline: float,
                        best_seq: np.ndarray, best_score: float):
    """Mirror‑descent on the log‑sum‑exp approximation of the max‑convolution."""
    eps = 1e-12
    n = a.shape[0]

    # Rough estimate of how many iterations we can afford.
    # One iteration costs a few FFTs (~0.005‑0.015 s for n≈2e4).
    max_iters = int((deadline - time.time()) / 0.008)
    max_iters = max(0, min(max_iters, 8000))

    if max_iters == 0:
        return best_seq, best_score

    # Schedule for temperature (α) and learning‑rate (lr)
    alpha_start, alpha_end = 1.0, 1e7
    lr_start, lr_end = 0.2, 1e-4

    momentum = np.zeros_like(a)
    beta = 0.9

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        t = it / max_iters
        alpha = alpha_start * ((alpha_end / alpha_start) ** t)
        lr = lr_start * ((lr_end / lr_start) ** t)

        conv = _fft_self_convolve(a)
        maxc = conv.max()
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # momentum (exponential moving average) for smoother updates
        momentum = beta * momentum + (1.0 - beta) * grad
        grad_hat = momentum / (1.0 - beta ** (it + 1))

        a = a * np.exp(-lr * grad_hat)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()

        if it % 500 == 0 or it == max_iters - 1:
            cur_sc = _max_convolution_score(a)
            if cur_sc < best_score - 1e-12:
                best_score = cur_sc
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth_max it={it} new best {cur_sc:.6f}")

    return best_seq, best_score


def subgradient_optimize(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float):
    """Projected Adam‑style sub‑gradient descent on the true max‑convolution."""
    eps = 1e-12
    n = a.shape[0]
    velocity = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0

        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam‑style momentum
        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** it)

        # simple back‑tracking line‑search
        step = step0
        while True:
            a_candidate = a - step * v_hat
            a_candidate = _project_onto_simplex(a_candidate, 1.0)
            a_candidate = np.maximum(a_candidate, eps)
            a_candidate /= a_candidate.sum()
            cand_sc = _max_convolution_score(a_candidate)
            if cand_sc < best_score - 1e-12 or step < 1e-8:
                break
            step *= 0.5

        a = a_candidate
        if cand_sc < best_score - 1e-12:
            best_score = cand_sc
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {cand_sc:.6f}")

        # occasional jitter to escape tiny plateaus
        if it % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            print(f"[{time.time():.1f}] jitter applied at subgrad it {it}")

        if time.time() + 0.5 > deadline:
            break

    return best_seq, best_score


def hill_climb_optimize(a: np.ndarray, deadline: float,
                        best_seq: np.ndarray, best_score: float):
    """Simple random mass‑transfer hill‑climber for a final polish."""
    eps = 1e-12
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0

    while time.time() < deadline:
        it += 1
        # Pick the lag that currently realises the maximum
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # Locate the pair (i,j) on this lag with largest product a[i]*a[j]
        sub = a[i_start : i_end + 1]
        i_local = np.argmax(sub)
        i = i_start + i_local
        j = k - i
        source = i if a[i] >= a[j] else j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        max_delta = min(a[source] * 0.4, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M
            best_seq = a.copy()
            best_score = 2.0 * n * M
            if it % 200 == 0:
                print(f"[{time.time():.1f}] hill it={it} new best {best_score:.6f}")

        if it % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] hill jitter at it {it}")

    return best_seq, best_score

# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a step‑function that minimises `evaluate_sequence`.

    The routine respects a CPU‑time budget of ``budget_s`` seconds.
    It returns a list of non‑negative floats (the best vector found).
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5          # small safety margin
    eps = 1e-12

    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1. Broad stochastic exploration
    lengths = [2000, 4000, 8000, 12000, 20000, 30000]

    # deterministic families for each length
    for n in lengths:
        if time.time() >= deadline:
            break

        # list of candidate generators (functions that return a vector)
        generators = []

        # a small curated palette
        generators.append(lambda n=n: _distribution_uniform(n))
        generators.append(lambda n=n: _distribution_triangular_left(n))
        generators.append(lambda n=n: _distribution_exponential_left(n, lam=0.001))
        generators.append(lambda n=n: _distribution_exponential_left(n, lam=0.01))
        generators.append(lambda n=n: _distribution_exponential_left(n, lam=0.05))
        generators.append(lambda n=n: _distribution_powerlaw(n, exponent=0.3))
        generators.append(lambda n=n: _distribution_powerlaw(n, exponent=0.6))
        generators.append(lambda n=n: _distribution_powerlaw(n, exponent=1.0))

        # random asymmetric beta families
        for _ in range(8):
            p = 10 ** np.random.uniform(-3, 1)     # 0.001 … 10
            q = 10 ** np.random.uniform(-3, 1)
            generators.append(lambda n=n, p=p, q=q: _distribution_beta_asym(n, p, q))

        # logistic families
        for _ in range(6):
            midpoint = random.random()
            steep = 10 ** np.random.uniform(0, 2)  # 1 … 100
            generators.append(lambda n=n, m=midpoint, s=steep: _distribution_logistic(n, m, s))

        # Gaussian families (different centres and widths)
        for _ in range(6):
            sigma = random.uniform(0.04, 0.30)
            offset = random.uniform(0.0, 0.6)
            generators.append(lambda n=n, sig=sigma, off=offset:
                              _distribution_gaussian_offset(n, sig, off))

        # sine‑power and cosine windows
        for pw in (0.5, 1.0, 2.0, 4.0, 8.0):
            generators.append(lambda n=n, p=pw: _distribution_sine_power(n, p))
        for coeff in (0.2, 0.4, 0.6, 0.8):
            generators.append(lambda n=n, c=coeff: _distribution_cosine_window(n, c))

        # random Dirichlet draws (different concentration)
        for alpha in (0.2, 0.5, 1.0, 2.0, 5.0):
            generators.append(lambda n=n, a=alpha: np.random.dirichlet(np.full(n, a)))

        # sparse random starts
        def sparse_vec(n=n):
            k = max(2, n // 400)               # a few hundred non‑zeros at most
            idx = np.random.choice(n, k, replace=False)
            vec = np.zeros(n, dtype=np.float64)
            vec[idx] = np.random.rand(k)
            vec /= vec.sum()
            return vec
        generators.append(sparse_vec)

        # evaluate all generators for this length
        for gen in generators:
            if time.time() >= deadline:
                break
            vec = gen()
            vec = np.maximum(vec, eps)          # guard against exact zeros
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] n={n} new init best {sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Incorporate a previously discovered sequence if it exists
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, np.ndarray)):
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.ndim == 1 and prev_arr.size > 0:
                prev_arr = np.maximum(prev_arr, eps)
                prev_arr /= prev_arr.sum()
                sc = _max_convolution_score(prev_arr)
                if sc < best_score:
                    best_score = sc
                    best_seq = prev_arr.copy()
                    best_n = prev_arr.size
                    print(f"[{time.time():.1f}] previous best {sc:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3. Fallback in the unlikely case no candidate was generated.
    if best_seq is None:
        best_n = 4000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Smooth‑max mirror‑descent refinement
    best_seq, best_score = smooth_max_optimize(best_seq, deadline, best_seq, best_score)

    # ------------------------------------------------------------------
    # 5. Projected sub‑gradient polishing (active‑set, Adam‑style)
    best_seq, best_score = subgradient_optimize(best_seq, deadline, best_seq, best_score)

    # ------------------------------------------------------------------
    # 6. Final hill‑climber polishing
    best_seq, best_score = hill_climb_optimize(best_seq, deadline, best_seq, best_score)

    # ------------------------------------------------------------------
    final_score = 2.0 * best_seq.shape[0] * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
