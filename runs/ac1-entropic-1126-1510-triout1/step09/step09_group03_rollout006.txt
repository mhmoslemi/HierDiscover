"""
propose_candidate – Large‑scale smooth‑max Adam + Frank‑Wolfe polishing.

The search works on a single (large) length `n` (≈20 000).  The
starting point is the discrete arcsine law, which already has an
almost‑constant self‑convolution.  We then minimise a smooth
approximation of the quantity

    2·n·max(conv(a,a))

using Adam on the soft‑max of the convolution entries.
The smoothness parameter β is increased in three stages so that the
optimizer first sees a very smooth landscape and later focuses on the
true maximum.

After the Adam phase the current point is refined by a cheap
pair‑wise mass‑transfer (Frank‑Wolfe) move and a short simulated‑annealing
jitter phase.  All vectors are kept on the simplex (non‑negative,
unit ℓ₁‑norm) by Euclidean projection.  The best sequence seen (as a
list of Python floats) is returned; its exact score can be obtained
with ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate 2·n·max(conv(a,a)) (sum(a)=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) on the simplex (∑a=1).
    Returns (gradient, cur_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices (self‑product terms)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
def _smoothmax_obj_and_grad(a: np.ndarray, beta: float = 5.0):
    """
    Smooth “soft‑max” approximation of max(conv(a,a)).
    Returns objective value (≈ max) and its gradient.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)                     # length 2n‑1
    m = b.max()
    # stable soft‑max weights
    exp_term = np.exp(beta * (b - m))
    sum_exp = exp_term.sum()
    w = exp_term / sum_exp                       # probabilities, sum to 1
    # gradient of weighted sum: ∇a  = 2 * conv_rev
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    # smooth approximation of the max value
    obj = m + np.log(sum_exp) / beta
    return obj, grad


# ----------------------------------------------------------------------
def _adam_smooth_max(a0: np.ndarray,
                     deadline: float,
                     beta: float,
                     step0: float = 0.12,
                     beta1: float = 0.9,
                     beta2: float = 0.999,
                     max_iters: int = 10**9):
    """
    Adam on the smooth‑max objective with a fixed smoothing parameter β.
    Returns the best point (ℓ₁‑normalised) seen before ``deadline``.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        _, grad = _smoothmax_obj_and_grad(a, beta)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best = a.copy()
    return best, best_score


# ----------------------------------------------------------------------
def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with concentration ``alpha``."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The routine respects the 1000 s budget and returns the best list found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 0️⃣ Choose a large length (the larger, the closer to the asymptotic 1.5)
    #    20 000 is safe for the 1 000 s budget (FFT size 65536).
    n = 20000

    # ------------------------------------------------------------------
    # 1️⃣ Build a small collection of diverse seeds (all ℓ₁‑normalised)
    seeds = []

    # classic arcsine law – already near‑optimal
    seeds.append(_arcsine_distribution(n))

    # a couple of symmetric beta laws (different shapes)
    seeds.append(_beta_symmetric_distribution(n, p=0.5))
    seeds.append(_beta_symmetric_distribution(n, p=1.5))

    # random Dirichlet draws (moderate concentration)
    seeds.append(_random_dirichlet_alpha(n, alpha=0.5))
    seeds.append(_random_dirichlet_alpha(n, alpha=1.0))

    # if a previous construction exists, up‑sample it to the current n
    prev_seq = globals().get('height_sequence_1')
    if prev_seq is not None:
        try:
            old = np.asarray(prev_seq, dtype=np.float64)
            if old.ndim == 1 and old.size > 0:
                # linear interpolation onto the new grid
                x_old = np.linspace(0.0, 1.0, old.size, endpoint=False)
                x_new = np.linspace(0.0, 1.0, n, endpoint=False)
                up = np.interp(x_new, x_old, old)
                up = np.maximum(up, 0.0)
                up /= up.sum()
                seeds.append(up)
                print(f"[{time.time():.1f}] added up‑sampled previous best")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2️⃣ Refine each seed using a three‑stage Adam on the smooth‑max
    best_seq = None
    best_score = np.inf

    for idx, seed_vec in enumerate(seeds):
        if time.time() >= deadline:
            break

        print(f"[{time.time():.1f}] seed {idx+1}/{len(seeds)} – start refinement")

        # Stage A – very smooth (β=4)
        sub_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.35)
        a_a, s_a = _adam_smooth_max(seed_vec, sub_deadline, beta=4.0)
        if s_a < best_score - 1e-14:
            best_seq, best_score = a_a.copy(), s_a
            print(f"[{time.time():.1f}]   after β=4  score={best_score:.6f}")

        # Stage B – moderate smoothness (β=12)
        sub_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.35)
        a_b, s_b = _adam_smooth_max(a_a, sub_deadline, beta=12.0)
        if s_b < best_score - 1e-14:
            best_seq, best_score = a_b.copy(), s_b
            print(f"[{time.time():.1f}]   after β=12 score={best_score:.6f}")

        # Stage C – rather sharp (β=30)
        sub_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.30)
        a_c, s_c = _adam_smooth_max(a_b, sub_deadline, beta=30.0)
        if s_c < best_score - 1e-14:
            best_seq, best_score = a_c.copy(), s_c
            print(f"[{time.time():.1f}]   after β=30 score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Frank‑Wolfe style pairwise mass‑transfer polishing
    if best_seq is None:
        # should never happen, but fall back to a simple arcsine
        best_seq = _arcsine_distribution(n)

    a = best_seq.copy()
    cur_score = best_score
    fw_iter = 0
    while time.time() < deadline and fw_iter < 60_000:
        fw_iter += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))   # mass to remove
        i_dst = int(np.argmin(grad))   # mass to add
        if i_src == i_dst:
            break
        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-15:
            break

        # try a modest transfer (≈30 % of the possible amount)
        delta = max_transfer * 0.3
        a[i_src] -= delta
        a[i_dst] += delta

        new_score = _max_convolution_score(a)
        if new_score < cur_score - 1e-14:
            cur_score = new_score
            if cur_score < best_score - 1e-14:
                best_seq, best_score = a.copy(), cur_score
                print(f"[{time.time():.1f}] FW iter={fw_iter} new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Light simulated‑annealing / jitter phase
    a = best_seq.copy()
    cur_score = best_score
    temp_start = 0.02
    temp_end = 0.001
    anneal_iter = 0
    while time.time() < deadline and anneal_iter < 80_000:
        anneal_iter += 1
        # linear schedule based on remaining time
        remaining = deadline - time.time()
        total = budget_s - 0.5
        t_frac = remaining / total
        temp = temp_end + (temp_start - temp_end) * t_frac

        i = np.random.randint(a.size)
        j = np.random.randint(a.size - 1)
        if j >= i:
            j += 1
        max_delta = min(a[i], 1.0 - a[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * temp

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if cur_score < best_score - 1e-14:
                best_seq, best_score = a.copy(), cur_score
                print(f"[{time.time():.1f}] Anneal iter={anneal_iter} new best {best_score:.6f}")
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_score - cur_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                a = a_new
                cur_score = new_score

    # ------------------------------------------------------------------
    # 5️⃣ Final evaluation (the surrogate equals the true value because Σa=1)
    final_score = _max_convolution_score(best_seq)  # same as evaluate_sequence
    print(f"Final best surrogate score: {final_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
