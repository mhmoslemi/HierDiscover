def beta_distribution(n, p, q):
    x = (np.arange(n)+0.5)/n
    logw = (p-1) * np.log(x) + (q-1) * np.log(1-x)
    logw = logw - np.max(logw) # shift for stability
    w = np.exp(logw)
    w_sum = w.sum()
    if w_sum == 0 or not np.isfinite(w_sum):
        return np.full(n, 1.0/n)
    return w / w_sum

def _distribution_beta(n, p, q):
    x = (np.arange(n) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    max_log = np.max(logw)
    w = np.exp(logw - max_log)  # subtract to avoid overflow
    w_sum = w.sum()
    if w_sum == 0 or not np.isfinite(w_sum):
        return np.full(n, 1.0 / n)
    w /= w_sum
    return w

"""
Propose a non‑negative step function (list of floats) that minimizes
`evaluate_sequence`.  The search proceeds in four stages:

1. **Broad grid search** over several lengths (up to 50 k) and a rich
   family of U‑shaped probability vectors (symmetric/asymmetric Beta,
   arcsine, edge‑biased Dirichlet draws, etc.).  The best candidate
   (already normalised to sum = 1) is kept.

2. **Smooth‑max mirror descent** (exponentiated‑gradient) that minimises
   a softmax approximation of the max‑convolution.  The temperature is
   increased geometrically while the learning rate decays, allowing a
   transition from exploration to precise optimisation.

3. **Projected sub‑gradient descent with Adam‑style momentum** on the
   true max‑convolution (active‑set gradient) and a back‑tracking line
   search.

4. **Targeted mass‑transfer hill‑climbing** to polish the final vector.

All operations respect the supplied time budget (default 1000 s) and
the constraints of `evaluate_sequence` (non‑negative entries,
∑a ≥ 0.01, aᵢ ≤ 1000).  The returned vector is a plain Python list.

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a):
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a):
    """
    For a vector already normalised to sum = 1 returns
    2·n·max(a*a) – exactly the value produced by `evaluate_sequence`.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    logw -= np.max(logw)               # shift for stability
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        return np.full_like(w, 1.0 / n)
    w /= w.sum()
    return w


def _distribution_beta(n, p, q):
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        return np.full_like(w, 1.0 / n)
    w /= w.sum()
    return w


def _distribution_edge_dirichlet(n, beta_exp):
    """
    Dirichlet draw with concentration concentrated at the edges:
    c_i ∝ ((i+0.5)*(n-i-0.5))^{-beta_exp}  + 0.1 .
    """
    idx = np.arange(n, dtype=np.float64)
    conc = ((idx + 0.5) * (n - idx - 0.5)) ** (-beta_exp) + 0.1
    vec = np.random.dirichlet(conc)
    return vec
