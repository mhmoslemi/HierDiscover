    # Use sparse matrix from scipy.sparse
    rows = []
    cols = []
    data = []

        k_end = min(n-1, i)

        # for k in k_start..k_end: coefficient = a[i - k]

        # fallback to zeros
        g = np.zeros(n)
    return g

"""
Propose a non‑negative step function (list of floats) that minimises
evaluate_sequence().  The search proceeds in three stages:

1.  Broad exploration – for several lengths we generate many analytic families
    (uniform, arcsine, triangular, exponential, symmetric beta, sine‑power,
    cosine window, Gaussian) together with a few random Dirichlet draws and a
    sparse random start.  The best candidate (normalised to sum = 1) is kept.

2.  Refinement – a two‑phase optimisation.
    * Smooth‑max mirror‑descent with an annealed temperature (α up to 2·10⁵)
      and an exponentially decaying learning‑rate.
    * Projected sub‑gradient descent (with a simple 1/√k step‑schedule) and
      an occasional Adam‑style momentum update.
    * A single linear‑programming “mass‑addition” step (maximise ∑gⱼ
      under the current max‑convolution bound) and a short mixing phase.

3.  Final polish – a cheap hill‑climber that repeatedly moves a tiny amount of
    mass from a large coefficient to a random one, only accepting moves that
    reduce the current peak of the self‑convolution.

All operations respect a hard CPU‑time budget (default 1000 s).  The
function returns the best sequence found as a plain list of floats.
"""

import time
import random
import math
import numpy as np
import scipy.optimize as spopt
import scipy.sparse as sp

# Helper utilities
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a):
    """Full linear self–convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full linear convolution of two real vectors using real FFT."""
    n = a.shape[0] + b.shape[0] - 1
    size = _next_pow2(n)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[:n]


def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families – each returns a vector with sum = 1
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Linear‑programming step (maximise total mass under current max‑convolution bound)
def _lp_max_sum(a, M):
    """
    Solve
        maximise   ∑ g_j
        subject to a * g ≤ M   (element‑wise, self‑convolution with a)
                    g ≥ 0
    Returns g (may be all‑zero if LP fails).
    """
    n = a.shape[0]
    # Build the (2n‑1)×n convolution matrix using banded diagonals.
    # Row i, column j gets a[i‑j] when i‑j∈[0,n‑1]; 0 otherwise.
    diags_data = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
    offsets = [-d for d in range(n)]
    A = sp.diags(diags_data, offsets, shape=(2 * n - 1, n), format='csr')
    b = np.full(2 * n - 1, M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)               # maximise → minimise –∑g
    bounds = [(0, None)] * n

    # Use the HiGHS interior‑point method; it handles sparse matrices efficiently.
    res = spopt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                        method='highs',
                        options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x
    # fall back – return a zero vector (no improvement)
    return np.zeros_like(a)

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step function that minimises evaluate_sequence."""
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin
    eps = 1e-12

    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1. Broad exploration over several lengths
    lengths = [2000, 4000, 6000, 8000, 12000]   # a few increasing sizes
    for n in lengths:
        if time.time() >= deadline:
            break

        candidates = []

        # Analytic families
        candidates.append(_distribution_uniform(n))
        candidates.append(_distribution_arcsine(n))
        candidates.append(_distribution_triangular(n))
        candidates.append(_distribution_exponential(n, lam=0.003))
        candidates.append(_distribution_exponential(n, lam=0.02))

        for p in (0.02, 0.04, 0.06, 0.08, 0.12, 0.15, 0.2,
                  0.25, 0.3, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
            candidates.append(_distribution_beta_sym(n, p))

        for pw in (0.5, 1.0, 2.0, 4.0, 8.0, 12.0):
            candidates.append(_distribution_sine_power(n, pw))

        for coeff in (0.3, 0.5, 0.7, 0.9):
            candidates.append(_distribution_cosine_window(n, coeff))

        for sigma in (0.12, 0.18, 0.25, 0.35, 0.45):
            candidates.append(_distribution_gaussian(n, sigma))

        # Random Dirichlet draws
        for alpha in (0.2, 0.5, 1.0, 2.0):
            vec = np.random.dirichlet(np.full(n, alpha))
            candidates.append(vec)

        # Sparse random start (few non‑zeros)
        sparse = np.zeros(n, dtype=np.float64)
        k = max(2, n // 300)
        idx = np.random.choice(n, k, replace=False)
        sparse[idx] = np.random.rand(k)
        sparse /= sparse.sum()
        candidates.append(sparse)

        # Evaluate all candidates (they already sum to 1)
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] n={n} new init best {sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Incorporate a previously discovered sequence if available.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = len(prev)
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback in the unlikely case no candidate was generated.
    if best_seq is None:
        best_n = 4000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Smooth‑max mirror‑descent refinement
    a = best_seq.copy()
    n = a.shape[0]

    schedule = [
        (1.0, 100.0, 2500, 0.30, 0.03),
        (100.0, 1000.0, 2500, 0.03, 0.003),
        (1000.0, 10000.0, 3000, 0.003, 0.0003),
        (10000.0, 200000.0, 4000, 0.0003, 0.00003),
    ]

    for (alpha0, alpha1, steps, lr0, lr1) in schedule:
        if time.time() >= deadline:
            break
        for it in range(steps):
            if time.time() >= deadline:
                break
            t = it / steps
            alpha = alpha0 * (alpha1 / alpha0) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

            a *= np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if (it % 800 == 0) or (it == steps - 1):
                cur_sc = _max_convolution_score(a)
                if cur_sc < best_score - 1e-12:
                    best_score = cur_sc
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] mirror it={it} new best {cur_sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Projected sub‑gradient descent with simple step‑schedule
    a = best_seq.copy()
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # diminishing step size 0.5 / sqrt(it)
        step = 0.5 / math.sqrt(it)
        a_candidate = a - step * grad
        a_candidate = _project_onto_simplex(a_candidate, 1.0)
        a_candidate = np.maximum(a_candidate, eps)
        a_candidate /= a_candidate.sum()
        cand_sc = _max_convolution_score(a_candidate)

        if cand_sc < best_score - 1e-12:
            best_score = cand_sc
            best_seq = a_candidate.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {cand_sc:.6f}")

        a = a_candidate
        if time.time() + 0.5 > deadline:
            break

    # ------------------------------------------------------------------
    # 6. LP‑based mass addition & mixing (usually a single useful pass)
    a = best_seq.copy()
    for lp_iter in range(3):
        if time.time() >= deadline:
            break
        conv = _fft_self_convolve(a)
        M = conv.max()
        g = _lp_max_sum(a, M)
        if g.sum() == 0.0:
            break
        # normalise g to the simplex (the LP may return a sum >1)
        g = np.maximum(g, eps)
        g /= g.sum()

        t = 0.07                         # modest mixing factor
        a_new = (1.0 - t) * a + t * g
        a_new = _project_onto_simplex(a_new, 1.0)
        a_new = np.maximum(a_new, eps)
        a_new /= a_new.sum()
        new_sc = _max_convolution_score(a_new)

        if new_sc < best_score - 1e-12:
            best_score = new_sc
            best_seq = a_new.copy()
            a = a_new
            print(f"[{time.time():.1f}] LP mix {lp_iter} new best {new_sc:.6f}")
        else:
            # no improvement – stop LP attempts
            break

    # ------------------------------------------------------------------
    # 7. Final cheap hill‑climber (mass‑transfer) polishing
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    while time.time() < deadline:
        it += 1
        # pick a source from the top–5% of coefficients (likely responsible for the peak)
        top_k = max(2, n // 200)
        source = np.random.choice(np.argsort(-a)[:top_k])
        target = np.random.randint(0, n)
        if target == source:
            continue

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * np.random.rand()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()

        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-13:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M
            best_seq = a.copy()
            best_score = 2.0 * n * M
            if it % 200 == 0:
                print(f"[{time.time():.1f}] hill it={it} new best {best_score:.6f}")

        if it % 2000 == 0:
            # occasional jitter to get off tiny plateaus
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

        if time.time() + 0.5 > deadline:
            break

    final_score = 2.0 * n * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
