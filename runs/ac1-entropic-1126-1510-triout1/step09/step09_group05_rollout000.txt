"""
A fast heuristic for the “minimal maximal self‑convolution” problem.

The objective is                2·n·max(conv(a,a))/(sum(a)²)
and is scale‑invariant, therefore we restrict the search to the simplex
(sum a = 1, a_i ≥ 0).  The algorithm has three phases:

1️⃣ Broad sampling – many analytic families (arcsine, beta, Gaussian,…)
   and random Dirichlet draws are generated for several lengths.
   They are scored cheaply by 2·n·max(a*a) (the sum is 1).

2️⃣ Heavy refinement – each of the best seeds is fed to an L‑BFGS
   optimizer that minimises a smooth surrogate
       f_ε(a)=ε·log Σ exp(conv(a,a)/ε) .
   The variables are the unconstrained logits y (a = softmax(y)),
   which automatically enforce the simplex constraints.
   ε is gradually reduced so that the final solution is close to the
   true max‑convolution objective.

3️⃣ Polish – a short projected sub‑gradient descent followed by a
   simple hill‑climbing that moves a tiny amount of mass between two
   indices.  The best sequence found before the deadline is returned.

All heavy loops are guarded by the supplied time budget (default 1000 s).
"""
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap surrogate = 2·n·max(a*a) (sum(a)=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.
    Returns (grad, cur_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _smooth_grad(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smoothed max using a log‑sum‑exp surrogate.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)  # length 2n‑1
    maxc = conv.max()
    w = np.exp((conv - maxc) / eps)
    w /= w.sum()

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    diag_indices = np.arange(conv.size)[(np.arange(conv.size) % 2) == 0] // 2
    grad[diag_indices] -= w[2 * diag_indices] * a[diag_indices]

    return grad


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(y)
    exp_y = np.exp(y - m)
    return exp_y / exp_y.sum()


# ----------------------------------------------------------------------
class _SmoothObjective:
    """
    Callable for scipy.optimize.minimize.

    Implements f_ε(a)=ε·log Σ exp(conv(a,a)/ε) together with its gradient,
    where a = softmax(y).  The gradient with respect to y is obtained by
    chain‑rule using the Jacobian of the softmax.
    """

    def __init__(self, eps: float, n: int):
        self.eps = eps
        self.n = n

    def __call__(self, y):
        a = _softmax(y)

        conv = _fft_self_convolve(a)
        maxc = conv.max()

        # stable evaluation of the smoothed max
        obj = maxc + self.eps * np.log(np.sum(np.exp((conv - maxc) / self.eps)))

        # weights for the gradient of the log‑sum‑exp
        w = np.exp((conv - maxc) / self.eps)
        w /= w.sum()

        conv_rev = _fft_convolve(a[::-1], w)
        grad_a = 2.0 * conv_rev[self.n - 1 : self.n - 1 + self.n]

        # diagonal correction (each even‑index appears twice)
        diag_idx = np.arange(conv.size)[(np.arange(conv.size) % 2) == 0] // 2
        if diag_idx.size:
            grad_a[diag_idx] -= w[2 * diag_idx] * a[diag_idx]

        # chain rule through softmax
        dot = np.dot(a, grad_a)
        grad_y = a * (grad_a - dot)

        return obj, grad_y


# ----------------------------------------------------------------------
def _lbfgs_optimize_on_simplex(a0: np.ndarray,
                               deadline: float,
                               eps_list=None,
                               max_iter_per=400) -> tuple[np.ndarray, float]:
    """
    L‑BFGS refinement on the simplex using a sequence of smoothing parameters.
    Returns the best sequence (still summing to 1) and its cheap score.
    """
    if eps_list is None:
        eps_list = [1e-2, 5e-3, 2e-3, 1e-3, 5e-4, 2e-4, 1e-4]

    import scipy.optimize

    n = a0.shape[0]
    # initialise logits from a0 (add tiny epsilon to avoid log(0))
    y = np.log(np.maximum(a0, 1e-12))

    best_a = a0.copy()
    best_score = _max_convolution_score(best_a)

    for eps in eps_list:
        if time.time() >= deadline:
            break
        obj_fun = _SmoothObjective(eps, n)
        try:
            res = scipy.optimize.minimize(obj_fun,
                                          y,
                                          method='L-BFGS-B',
                                          jac=True,
                                          options={'maxiter': max_iter_per,
                                                   'disp': False})
        except Exception as e:          # pragma: no cover
            print(f"[{time.time():.1f}] L‑BFGS failed: {e}")
            break
        y = res.x
        a = _softmax(y)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
        # keep the current logits for the next (smaller) eps
    return best_a, best_score

def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         max_iters: int = 500_000,
                         eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Plain projected sub‑gradient descent (simple, cheap)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_w = (p - 1) * (np.log(x) + np.log(1.0 - x))
    log_w = np.clip(log_w, -700, 700)
    w = np.exp(log_w)
    if w.sum() == 0.0:
        return np.full_like(w, 1.0 / n)
    return w / w.sum()


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _gaussian_distribution(n: int, sigma_frac: float = 0.30) -> np.ndarray:
    sigma = sigma_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if vals.sum() == 0.0:
        return np.full_like(vals, 1.0 / n)
    return vals / vals.sum()


def _cauchy_distribution(n: int, scale_frac: float = 0.30) -> np.ndarray:
    scale = scale_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = 1.0 / (1.0 + ((i - centre) / scale) ** 2)
    return vals / vals.sum()


def _semicircle_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    vals = np.sqrt(np.clip(1.0 - x * x, 0.0, None))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    return vals / vals.sum()


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return x / x.sum()

                          (_semicircle_distribution, "semicircle")]:
            a = gen(n)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} {name:10s} score={score:.6f}")

        # Exponential
        for lam in exp_lams:
            a = _exponential_distribution(n, lam=lam)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} exponential(lam={lam:.3f}) score={score:.6f}")

        # Gaussian
        for sf in sigma_fracs:
            a = _gaussian_distribution(n, sigma_frac=sf)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} gaussian(sig={sf:.2f}) score={score:.6f}")

        # Cauchy
        for sf in scale_fracs:
            a = _cauchy_distribution(n, scale_frac=sf)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} cauchy(scale={sf:.2f}) score={score:.6f}")

        # Symmetric beta
        for p in beta_ps:
            if time.time() >= deadline:
                break
            a = _beta_symmetric_distribution(n, p)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} beta(p={p:.3f}) score={score:.6f}")

        # Random Dirichlet (very sparse draws are useful)
        for alpha in dirichlet_alphas:
            for _ in range(2):
                if time.time() >= deadline:
                    break
                a = _random_dirichlet(n, alpha=alpha)
                score = _max_convolution_score(a)
                candidates.append((a, score))
                print(f"[{time.time():.1f}] n={n:5d} Dirichlet(α={alpha:.2f}) score={score:.6f}")

        # Small jittered arcsine (break symmetry a bit)
        a = _arcsine_distribution(n)
        jitter = np.random.randn(n) * 2e-4
        a = a + jitter
        a = np.clip(a, 0, None)
        a = _project_onto_simplex(a, s=1.0)
        score = _max_convolution_score(a)
        candidates.append((a, score))
        print(f"[{time.time():.1f}] n={n:5d} arcsine‑jitter score={score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Insert previous best if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _uniform_distribution(prev.size)
            else:
                prev = prev / prev.sum()
            score = _max_convolution_score(prev)
            candidates.append((prev, score))
            print(f"[{time.time():.1f}] previous best score={score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3️⃣  Keep the most promising seeds
    candidates.sort(key=lambda x: x[1])
    top_k = 20
    top_candidates = candidates[:top_k]
    print(f"=== Selected {len(top_candidates)} seeds (best cheap={top_candidates[0][1]:.6f}) ===")

    # ------------------------------------------------------------------
    # 4️⃣  Heavy refinement – L‑BFGS on a smooth surrogate
    best_seq = top_candidates[0][0].copy()
    best_score = top_candidates[0][1]

    for idx, (seed_seq, _) in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        # L‑BFGS refinement
        seq, sc = _lbfgs_optimize_on_simplex(seed_seq, deadline)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} L‑BFGS → {sc:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Short sub‑gradient polishing (uses the exact max)
    subgrad_deadline = deadline - 8.0   # keep a few seconds for hill‑climb
    if time.time() < subgrad_deadline:
        seq, sc = _subgradient_descent(best_seq, subgrad_deadline,
                                       max_iters=800_000,
                                       eta0=0.7)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] sub‑gradient polishing → {sc:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Final hill‑climbing (tiny mass transfers)
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    iteration = 0
    max_transfer = 0.030   # up to 3 % of total mass initially

    while time.time() < deadline:
        iteration += 1
        # exponential decay of the transfer magnitude
        transfer = max_transfer * (0.5 ** (iteration / 9000.0))

        # pick donor / receiver
        if iteration % 7 == 0:
            g, _, _ = _subgrad(a)
            i = int(np.argmax(g))
            j = int(np.argmin(g))
            if i == j:
                i = np.random.randint(n)
                j = (i + np.random.randint(1, n)) % n
        else:
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} new best {best_score:.6f}")

        # occasional jitter to escape plateaus
        if iteration % 6500 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 6e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    # convert back to plain Python list (scale does not matter)
    return best_seq.tolist()
