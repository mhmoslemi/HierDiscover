def smooth_max_descent(a, steps, alpha_start, alpha_end, lr_start, lr_end):
    n = a.shape[0]
    eps = 1e-12
    a = a.copy()
    for it in range(steps):
        t = it / steps
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t

        conv = _fft_self_convolve(a)
        maxc = conv.max()
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]
        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a, 1.0)
        # maintain positivity and normalization
        a = np.maximum(a, eps)
        a /= a.sum()
    return a
```

def subgrad_descent(a, max_iters, eta0):
    n = a.shape[0]
    eps = 1e-12
    vel = np.zeros_like(a)
    beta = 0.9
    for it in range(1, max_iters+1):
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        eps_rel = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - eps_rel)
        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        vel = beta * vel + (1-beta) * grad
        eta = eta0 / np.sqrt(it)
        a = a - eta * vel
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()
    return a
```

def hill_climb(a, max_iters, deadline):
    n = a.shape[0]
    eps = 1e-12
    a = a.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # find k of max
        k = int(conv.argmax())
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        best_i, best_j = i_low, k - i_low
        max_prod = a[best_i] * a[best_j]
        for i in range(i_low, i_high + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)

"""
Propose a non‑negative step‑function (list of floats) that minimises
``evaluate_sequence``.  The algorithm works in three phases:

1.  **Broad scan** – evaluate a collection of analytic families
    (uniform, arcsine, triangular, exponential, symmetric Beta,
    Gaussian) on several lengths (1500…5000).  The five best
    candidates (lowest 2·n·max(a*a)) are kept.

2.  **Local optimisation** – each of the retained candidates is refined
    by (a) a long smooth‑max mirror‑descent (multiplicative updates
    with a temperature that grows to ≈5 000) and (b) a sub‑gradient
    descent with momentum (Adam‑style step‑size decay).  The best
    result is retained.

3.  **Exploration & polishing** – while time permits the current best
    vector is jittered, run a short mirror‑descent + sub‑gradient run,
    and finally a cheap hill‑climber that shifts a small amount of mass
    away from the pair responsible for the current maximal convolution
    entry.

All updates keep the vector on the simplex (non‑negative, sum = 1)
using Euclidean projection.  The code respects a 1000 s wall‑clock
budget; intermediate results are printed for monitoring.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) for a vector with Σa = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.01) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _smooth_max_descent(a: np.ndarray,
                        steps: int,
                        alpha_start: float,
                        alpha_end: float,
                        lr_start: float,
                        lr_end: float,
                        deadline: float) -> np.ndarray:
    """Multiplicative mirror‑descent on the smooth‑max of a*a."""
    n = a.shape[0]
    eps = 1e-12
    a = a.copy()
    for it in range(steps):
        if time.time() >= deadline:
            break
        t = it / steps
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t

        conv = _fft_self_convolve(a)
        maxc = conv.max()
        w = np.exp(alpha * (conv - maxc))          # stable: max exponent = 0
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]
        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()
    return a


def _subgrad_descent(a: np.ndarray,
                     max_iters: int,
                     eta0: float,
                     deadline: float) -> np.ndarray:
    """Sub‑gradient descent with momentum (Adam‑style decay)."""
    n = a.shape[0]
    eps = 1e-12
    a = a.copy()
    vel = np.zeros_like(a)
    beta = 0.9
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        eps_rel = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - eps_rel)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        vel = beta * vel + (1.0 - beta) * grad
        eta = eta0 / np.sqrt(it)
        a = a - eta * vel
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()
    return a


def _hill_climb(a: np.ndarray,
                max_iters: int,
                deadline: float) -> np.ndarray:
    """Very cheap mass‑transfer hill‑climber."""
    n = a.shape[0]
    eps = 1e-12
    a = a.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        k = int(conv.argmax())
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        best_i, best_j = i_low, k - i_low
        max_prod = a[best_i] * a[best_j]
        for i in range(i_low, i_high + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue
        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        snapshot = a.copy()
        a[source] -= delta
        a[target] += delta
        a = np.maximum(a, eps)
        a /= a.sum()
        conv = _fft_self_convolve(a)
        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
        else:
            a = snapshot
            conv = _fft_self_convolve(a)

        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
    return a

            fams.append(_distribution_gaussian(n, sigma))

        # a few random Dirichlet draws
        for alpha in (0.5, 1.0, 2.0):
            for _ in range(3):
                fams.append(np.random.dirichlet(np.full(n, alpha)))

        # evaluate all
        for vec in fams:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, 1e-12)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            best_candidates.append((sc, vec))

        # keep only the top few
        best_candidates.sort(key=lambda x: x[0])
        best_candidates = best_candidates[:5]

    # ------------------------------------------------------------------
    # 2.  Pull in a previously found global candidate, if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 1e-12)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            best_candidates.append((sc, prev))
            best_candidates.sort(key=lambda x: x[0])
            best_candidates = best_candidates[:5]
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback to uniform if nothing was found
    if not best_candidates:
        n = 3000
        vec = _distribution_uniform(n)
        sc = _max_convolution_score(vec)
        best_candidates = [(sc, vec)]

    # initialise global best
    best_seq, best_score = best_candidates[0][1], best_candidates[0][0]
    print(f"[{time.time():.1f}] initial best score {best_score:.6f} (n={best_seq.shape[0]})")

    # ------------------------------------------------------------------
    # 4.  Refine each top candidate with smooth‑max descent + sub‑grad
    for idx, (sc, vec) in enumerate(best_candidates):
        if time.time() >= deadline:
            break
        a = vec.copy()
        n = a.shape[0]

        # smooth‑max mirror descent: longer runs for larger n
        steps = int(min(20000, max(5000, n // 2)))
        a = _smooth_max_descent(a,
                                steps=steps,
                                alpha_start=0.5,
                                alpha_end=5000.0,
                                lr_start=0.2,
                                lr_end=0.001,
                                deadline=deadline)
        sc2 = _max_convolution_score(a)
        if sc2 < best_score:
            best_seq, best_score = a, sc2
            print(f"[{time.time():.1f}] after smooth descent from cand {idx} score {sc2:.6f}")

        # sub‑gradient polishing
        a = _subgrad_descent(a,
                             max_iters=200_000,
                             eta0=0.2,
                             deadline=deadline)
        sc3 = _max_convolution_score(a)
        if sc3 < best_score:
            best_seq, best_score = a, sc3
            print(f"[{time.time():.1f}] after sub‑grad from cand {idx} score {sc3:.6f}")

    # ------------------------------------------------------------------
    # 5.  Random restarts with jitter while time permits
    while time.time() < deadline - 15.0:
        a = best_seq.copy()
        n = a.shape[0]
        # jitter the current best
        a += np.random.randn(n) * 1e-4
        a = np.maximum(a, 1e-12)
        a /= a.sum()

        a = _smooth_max_descent(a,
                                steps=3000,
                                alpha_start=1.0,
                                alpha_end=2000.0,
                                lr_start=0.15,
                                lr_end=0.001,
                                deadline=deadline)
        a = _subgrad_descent(a,
                             max_iters=60_000,
                             eta0=0.15,
                             deadline=deadline)
        sc = _max_convolution_score(a)
        if sc < best_score:
            best_seq, best_score = a, sc
            print(f"[{time.time():.1f}] random restart improved score {sc:.6f}")

    # ------------------------------------------------------------------
    # 6.  Final cheap hill‑climber
    if time.time() < deadline - 5.0:
        a = best_seq.copy()
        a = _hill_climb(a, max_iters=300_000, deadline=deadline)
        sc = _max_convolution_score(a)
        if sc < best_score:
            best_seq, best_score = a, sc
            print(f"[{time.time():.1f}] hill‑climb final score {sc:.6f}")

    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
