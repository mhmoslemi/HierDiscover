"""
propose_candidate
-----------------

Search for a non‑negative step function (list of heights) that minimizes the
score returned by ``evaluate_sequence``.

The algorithm is deliberately different from the previous approach:

1. **Parameterised exponential family** –  
   The sequence is taken as a (truncated) exponential
   `a_i = C·exp(-α·i)`.  For a given length `n` the normalising constant `C`
   is chosen so that the ℓ¹‑norm equals `√(2·n)`.  The objective depends only
   on the shape, not on the absolute scale, therefore every `α ≥ 0` defines
   a valid candidate.

2. **Fast one‑dimensional search** –  
   A golden‑section search over `α∈[0,α_max]` (with `α_max = 0.01`) quickly
   finds the value that gives the smallest score.  The search is limited
   by the global time budget (default 1000 s).

3. **Optional linear family** –  
   A very simple decreasing linear shape `a_i = max(0, 1−β·i)` is also
   examined (again with a golden‑section search) – it sometimes beats the
   exponential family for the same length.

4. **Permutation heuristics** –  
   After the best shape is known, a few deterministic permutations are
   tried (descending order, big‑small interleaving, alternating high/low)
   because the convolution is sensitive to the ordering of the coefficients.

5. **Local stochastic hill‑climbing** –  
   A few thousand random “δ‑mass transfer” moves are performed, each move
   being kept only if it strictly improves the exact score.  The move is
   projected back onto the ℓ¹‑simplex to preserve feasibility.

The routine respects the user‑supplied time budget, prints progress messages,
and finally returns the best list of heights it has found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z <= 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _make_exponential(alpha: float, n: int, scale: float) -> np.ndarray:
    """Return a normalized exponential sequence of length n."""
    i = np.arange(n, dtype=np.float64)
    # for α = 0 we fall back to a uniform array
    if alpha == 0.0:
        raw = np.ones_like(i)
    else:
        raw = np.exp(-alpha * i)
    total = raw.sum()
    return scale * raw / total


# ----------------------------------------------------------------------
def _make_linear(beta: float, n: int, scale: float) -> np.ndarray:
    """Return a normalized linearly decreasing sequence (clipped at 0)."""
    i = np.arange(n, dtype=np.float64)
    raw = np.maximum(0.0, 1.0 - beta * i)
    total = raw.sum()
    if total == 0.0:
        # degenerate case – fall back to uniform
        raw = np.ones_like(i)
        total = raw.sum()
    return scale * raw / total


# ----------------------------------------------------------------------
def _interleave_descending(arr: np.ndarray) -> np.ndarray:
    """Create a big‑small interleaving from a descending array."""
    n = arr.shape[0]
    vals = np.sort(arr)[::-1]                     # descending
    half = n // 2
    left = vals[:half][::-1]                      # smallest half, reversed
    right = vals[half:][::-1]                     # largest half, reversed
    inter = np.empty_like(vals)
    inter[0::2] = left
    inter[1::2] = right
    return inter


# ----------------------------------------------------------------------
def _alternating_high_low(arr: np.ndarray) -> np.ndarray:
    """Place the largest, smallest, second‑largest, second‑smallest, …."""
    sorted_desc = np.sort(arr)[::-1]
    lo = 0
    hi = len(sorted_desc) - 1
    pos = 0
    out = np.empty_like(sorted_desc)
    while lo <= hi:
        out[pos] = sorted_desc[lo]
        lo += 1
        pos += 1
        if lo <= hi:
            out[pos] = sorted_desc[hi]
            hi -= 1
            pos += 1
    return out


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Find a step function with a low evaluate_sequence score.
    The routine stops no later than ``budget_s`` seconds and returns the
    best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # a small safety margin

    # --------------------------------------------------------------
    # problem size
    try:
        # use the reference length if available
        n = len(height_sequence_1)  # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 9000
        have_ref = False

    # target ℓ¹‑norm (scale‑invariant)
    S = math.sqrt(2.0 * n)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # utility: evaluate a candidate and keep it if it improves
    def _maybe_update(candidate: np.ndarray) -> None:
        nonlocal best_seq, best_score
        if time.time() > deadline:
            return
        sc = evaluate_sequence(candidate.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"[improve] new best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 0) start from the reference sequence (if any) and from uniform
    if have_ref:
        ref = np.asarray(height_sequence_1, dtype=float)
        ref_proj = _project_to_simplex(ref, S)
        _maybe_update(ref_proj)

    uniform = np.full(n, S / n, dtype=np.float64)
    _maybe_update(uniform)

    # --------------------------------------------------------------
    # 1) exponential family – golden‑section search over α
    # search interval
    lo_alpha = 0.0
    hi_alpha = 0.01               # generous upper bound
    phi = (np.sqrt(5.0) - 1.0) / 2.0

    # initial interior points
    c_alpha = hi_alpha - phi * (hi_alpha - lo_alpha)
    d_alpha = lo_alpha + phi * (hi_alpha - lo_alpha)

    seq_c = _make_exponential(c_alpha, n, S)
    seq_d = _make_exponential(d_alpha, n, S)
    sc_c = evaluate_sequence(seq_c.tolist())
    sc_d = evaluate_sequence(seq_d.tolist())
    _maybe_update(seq_c)
    _maybe_update(seq_d)

    # keep the current best α
    best_alpha = c_alpha if sc_c < sc_d else d_alpha

    # iterative refinement
    max_iter = 30
    it = 0
    while it < max_iter and (hi_alpha - lo_alpha) > 1e-9 and time.time() < deadline:
        if sc_c < sc_d:
            hi_alpha = d_alpha
            d_alpha = c_alpha
            sc_d = sc_c
            c_alpha = hi_alpha - phi * (hi_alpha - lo_alpha)
            seq_c = _make_exponential(c_alpha, n, S)
            sc_c = evaluate_sequence(seq_c.tolist())
            _maybe_update(seq_c)
        else:
            lo_alpha = c_alpha
            c_alpha = d_alpha
            sc_c = sc_d
            d_alpha = lo_alpha + phi * (hi_alpha - lo_alpha)
            seq_d = _make_exponential(d_alpha, n, S)
            sc_d = evaluate_sequence(seq_d.tolist())
            _maybe_update(seq_d)
        it += 1
        # keep the best α seen so far
        if sc_c < best_score:
            best_alpha = c_alpha
        if sc_d < best_score:
            best_alpha = d_alpha

    # --------------------------------------------------------------
    # 2) small random refinements around the best exponential α
    for _ in range(2000):
        if time.time() > deadline:
            break
        # Gaussian perturbation (relative)
        alpha = best_alpha * (1.0 + rng.normal(scale=0.15))
        alpha = max(0.0, min(hi_alpha, alpha))
        cand = _make_exponential(alpha, n, S)
        _maybe_update(cand)
        if best_seq is not None and np.allclose(alpha, best_alpha, atol=1e-12):
            best_alpha = alpha

    # --------------------------------------------------------------
    # 3) linear decreasing family (optional – cheap golden‑section)
    lo_beta = 0.0
    hi_beta = 2.0 / (n - 1)   # ensures the line hits zero at the last index
    c_beta = hi_beta - phi * (hi_beta - lo_beta)
    d_beta = lo_beta + phi * (hi_beta - lo_beta)

    seq_c = _make_linear(c_beta, n, S)
    seq_d = _make_linear(d_beta, n, S)
    sc_c = evaluate_sequence(seq_c.tolist())
    sc_d = evaluate_sequence(seq_d.tolist())
    _maybe_update(seq_c)
    _maybe_update(seq_d)

    best_beta = c_beta if sc_c < sc_d else d_beta
    it = 0
    while it < 25 and (hi_beta - lo_beta) > 1e-12 and time.time() < deadline:
        if sc_c < sc_d:
            hi_beta = d_beta
            d_beta = c_beta
            sc_d = sc_c
            c_beta = hi_beta - phi * (hi_beta - lo_beta)
            seq_c = _make_linear(c_beta, n, S)
            sc_c = evaluate_sequence(seq_c.tolist())
            _maybe_update(seq_c)
        else:
            lo_beta = c_beta
            c_beta = d_beta
            sc_c = sc_d
            d_beta = lo_beta + phi * (hi_beta - lo_beta)
            seq_d = _make_linear(d_beta, n, S)
            sc_d = evaluate_sequence(seq_d.tolist())
            _maybe_update(seq_d)
        it += 1
        if sc_c < best_score:
            best_beta = c_beta
        if sc_d < best_score:
            best_beta = d_beta

    # --------------------------------------------------------------
    # 4) deterministic permutations of the current best sequence
    if best_seq is not None:
        # descending order
        desc = np.sort(best_seq)[::-1]
        _maybe_update(_project_to_simplex(desc, S))

        # big‑small interleaving (as in the previous code)
        inter = _interleave_descending(best_seq)
        _maybe_update(_project_to_simplex(inter, S))

        # alternating high‑low
        alt = _alternating_high_low(best_seq)
        _maybe_update(_project_to_simplex(alt, S))

    # --------------------------------------------------------------
    # 5) stochastic hill‑climbing (δ‑mass transfers)
    if best_seq is not None:
        a = best_seq.copy()
        step_trials = int(min(30000, (deadline - time.time()) * 30))
        print(f"[hill‑climb] attempting up to {step_trials} random transfers")
        for trial in range(step_trials):
            if time.time() > deadline:
                break
            # pick donor and receiver
            donor = rng.integers(n)
            receiver = rng.integers(n)
            if donor == receiver:
                continue
            if a[donor] <= 0.0:
                continue
            # move a small fraction (up to 5 % of donor)
            delta = rng.random() * a[donor] * 0.05
            cand = a.copy()
            cand[donor] -= delta
            cand[receiver] += delta
            # keep feasibility (non‑negative and same ℓ¹‑norm)
            cand = _project_to_simplex(cand, S)
            sc = evaluate_sequence(cand.tolist())
            if sc < best_score - 1e-12:
                best_score = sc
                best_seq = cand.copy()
                a = cand
                print(f"[hill‑climb] trial {trial:5d} → {best_score:.6f}")

    # --------------------------------------------------------------
    # final result
    if best_seq is None:
        # fallback – uniform (should never happen)
        best_seq = uniform
        best_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best score = {best_score:.6f} (length {n})")
    return best_seq.tolist()
