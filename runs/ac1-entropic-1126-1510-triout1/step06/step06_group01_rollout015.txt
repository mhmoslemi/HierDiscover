def simulated_annealing(a_init, deadline):
    a = a_init.copy()
    cur_score = compute_score(a)
    best = a.copy()
    best_score = cur_score
    T0 = cur_score * 0.01 (maybe)
    T_end = cur_score * 1e-4
    total_iters = 200000 (or time-bounded)
    iteration = 0
    while time.time() < deadline and iteration < total_iters:
        iteration+=1
        T = T0 * (T_end/T0) ** (iteration/total_iters)  # exponential cooling
        # choose i,j
        i = np.random.randint(len(a))
        j = np.random.randint(len(a)-1)
        if j >= i: j += 1
        # compute max transfer based on a[i]
        max_delta = min(0.1*a[i], 0.1)  # 10% of a[i] maybe
        delta = np.random.rand() * max_delta
        # propose moving delta from i to j
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # ensure non-negatives (a_new[i]>=0)
        if a_new[i] < 0:
            continue
        new_score = compute_score(a_new)
        delta_score = new_score - cur_score
        if delta_score < 0 or np.random.rand() < np.exp(-delta_score / max(T,1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a_new.copy()
                print(f"[{time.time():.1f}] SA iter={iteration} new best {best_score:.6f}")
    return best, best_score

"""
search – a lightweight multi‑stage optimiser for the step‑function objective
--------------------------------------------------------------------------

The function ``propose_candidate`` builds a non‑negative sequence satisfying
∑a = 1 (the scale of the problem does not matter) and tries to minimise

    score(a) = 2·n·max_k (a * a)[k]          (n = len(a))

The optimiser proceeds in four phases

1. **Analytic seeding** – a short sweep over a few classic families
   (uniform, arcsine, triangular, exponential and a finely sampled
   symmetric beta) for several lengths.  The best vector becomes the seed.

2. **Smoothed Adam‑descent** – minimise a smooth approximation of the
   ``max`` using the soft‑max
   ``μ_α(b) = (1/α)·log Σ exp(α·b_i)``.
   The schedule increases ``α`` up to 2000, followed by an optional
   symmetry projection (a ← (a + a[::-1]) / 2).  The best iterate is kept.

3. **Simulated annealing** – a temperature‑controlled random‑transfer
   Markov chain that respects the simplex constraints
   (mass is moved from one index to another).  The chain is kept
   symmetric and accepts uphill moves with probability exp(−Δ/T).

4. **Exact max‑subgradient refinement + local hill‑climb** – the classic
   sub‑gradient method from the reference code finishes the optimisation,
   and a cheap random‑transfer hill‑climber cleans up any remaining
   plateaux.

All heavy work is performed with real FFTs, giving O(n log n) per
convolution.  The routine respects the 1000 s budget and prints progress
messages.  The returned list of floats is ready for ``evaluate_sequence``.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a*b using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0) when p<1
    with np.errstate(divide='ignore'):
        w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0 or np.isnan(w).any():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _initial_candidates(n: int) -> list[np.ndarray]:
    """Return a handful of plausible seeds for a given length."""
    cand = [
        _distribution_uniform(n),
        _distribution_arcsine(n),
        _distribution_triangular(n),
        _distribution_exponential(n, lam=0.003),
        _distribution_exponential(n, lam=0.01),
    ]
    # coarse beta sweep
    for p in (0.2, 0.5, 1.0, 2.0):
        cand.append(_distribution_beta_sym(n, p))
    # a sparse random start
    rnd = np.zeros(n, dtype=np.float64)
    k = max(2, n // 200)
    idx = np.random.choice(n, k, replace=False)
    rnd[idx] = np.random.rand(k)
    rnd /= rnd.sum()
    cand.append(rnd)
    return cand

def _smooth_adam_descent(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max approximation of the max.
    Schedules α up to 2000 and projects onto the probability simplex.
    The iterate is forced to stay symmetric (a = (a+rev)/2) to exploit
    the known symmetry of the optimum.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_score(a)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # (α, steps) schedule – the final stage (α≈2000) is very close to the true max
    schedule = [
        (0.5, 1500),
        (1.0, 1500),
        (5.0, 1500),
        (20.0, 1500),
        (100.0, 1500),
        (500.0, 1500),
        (2000.0, 2000),
    ]

    for alpha, steps in schedule:
        if time.time() >= deadline:
            break
        for _ in range(steps):
            if time.time() >= deadline:
                break
            t += 1

            b = _fft_self_convolve(a)
            maxb = b.max()
            exp_terms = np.exp(alpha * (b - maxb))
            w = exp_terms / exp_terms.sum()                # soft‑max weights

            # gradient = 2 * (a_rev * w)  (only the central n entries are needed)
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # simplex & symmetry projection
            a = _project_onto_simplex(a, 1.0)
            a = (a + a[::-1]) * 0.5   # enforce symmetry (keeps sum unchanged)

            if t % 200 == 0:
                cur = _max_convolution_score(a)
                if cur < best_score - 1e-12:
                    best_score, best = cur, a.copy()
                    print(f"[{time.time():.1f}] Adam α={alpha:.1f} iter={t} new best {best_score:.6f}")
                if best_score <= 1.503:
                    return best, best_score
    return best, best_score

def _simulated_annealing(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Temperature‑controlled random mass transfer on the simplex.
    Moves are always symmetric (a ← (a+rev)/2) so the search stays within
    the symmetric sub‑space that contains the optimum.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _max_convolution_score(a)
    best, best_score = a.copy(), cur_score

    # initial / final temperature – a few percent of the current score works well
    T0 = cur_score * 0.02
    Tf = cur_score * 1e-4

    # exponential cooling factor – tuned so that we get ~2·10⁵ iterations on a 5 k vector
    cooling_rate = 1e-5
    it = 0

    while time.time() < deadline:
        it += 1
        T = max(Tf, T0 * math.exp(-cooling_rate * it))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # transfer a fraction of the source mass (never more than 10 %)
        max_delta = min(0.1 * a[i], 0.05)
        if max_delta <= 0:
            continue
        delta = np.random.rand() * max_delta

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # keep symmetry
        a_new = (a_new + a_new[::-1]) * 0.5

        new_score = _max_convolution_score(a_new)
        dscore = new_score - cur_score

        if dscore < 0 or np.random.rand() < math.exp(-dscore / max(T, 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best, best_score = a_new.copy(), new_score
                print(f"[{time.time():.1f}] SA iter={it} new best {best_score:.6f}")

        # occasional jitter to avoid getting stuck
        if it % 5000 == 0 and np.random.rand() < 0.2:
            jitter = np.random.randn(n) * 1e-5
            a = best + jitter
            a = _project_onto_simplex(a, 1.0)
            a = (a + a[::-1]) * 0.5
            cur_score = _max_convolution_score(a)

    return best, best_score

def _exact_subgradient_refine(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Exact max‑subgradient descent with momentum (identical to the reference)."""
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_score(a)

    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0

    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps)

        w = np.zeros_like(b)
        w[active] = 1.0
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-12:
            best_score, best = cur, a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {best_score:.6f}")
            if best_score <= 1.503:
                break

        if time.time() + 0.5 > deadline:
            break
    return best, best_score

    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_score(a)
    cur_score = best_score

    it = 0

    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # 1.  Coarse analytic sweep (few lengths, a handful of families)
    lengths = [2000, 3000, 4000, 5000, 6000, 8000]
    overall_best = None
    overall_best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        for cand in _initial_candidates(n):
            cand = cand / cand.sum()
            sc = _max_convolution_score(cand)
            if sc < overall_best_score:
                overall_best_score, overall_best = sc, cand.copy()
                print(f"[{time.time():.1f}] len={n} init cand score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  Refined beta‑parameter sweep around the current best length
    if overall_best is not None:
        n_best = overall_best.shape[0]
    else:
        n_best = lengths[0]

    # we also try the neighbour lengths because the optimum may shift slightly
    refine_lengths = [max(2000, n_best - 2000), n_best, min(8000, n_best + 2000)]
    p_grid = np.concatenate([np.linspace(0.05, 0.5, 9), np.linspace(0.6, 3.0, 12)])

    for n in refine_lengths:
        if time.time() >= deadline:
            break
        for p in p_grid:
            cand = _distribution_beta_sym(n, p)
            cand = cand / cand.sum()
            sc = _max_convolution_score(cand)
            if sc < overall_best_score:
                overall_best_score, overall_best = sc, cand.copy()
                print(f"[{time.time():.1f}] len={n} beta(p={p:.3f}) score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3.  If a previously found sequence exists, try to improve it.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # Fallback – never happens but keeps the code safe.
    if overall_best is None:
        n = lengths[0]
        overall_best = _distribution_uniform(n)
        overall_best_score = _max_convolution_score(overall_best)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Smoothed Adam descent (extended schedule)
    a, best_score = _smooth_adam_descent(overall_best, deadline)

    # ------------------------------------------------------------------
    # 5.  Simulated annealing – global exploration from the Adam result
    a, best_score = _simulated_annealing(a, deadline)

    # ------------------------------------------------------------------
    # 6.  Exact max‑subgradient refinement (as in the reference)
    a, best_score = _exact_subgradient_refine(a, deadline)

    # ------------------------------------------------------------------
    # 7.  Final cheap hill‑climb for a polished finish
    a, best_score = _local_hill_climb(a, deadline)

    print(f"Final best score: {best_score:.6f} (length={len(a)})")
    return a.tolist()
