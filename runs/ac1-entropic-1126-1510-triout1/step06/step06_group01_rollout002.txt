def _annealing_refine(a, deadline, init_temp=0.1, final_temp=1e-5, steps=50000):
    n = a.shape[0]
    cur = a.copy()
    cur_score = _max_convolution_score(cur)
    best = cur.copy()
    best_score = cur_score
    temp = init_temp
    t = 0
    total_steps = steps
    while time.time() < deadline and t < total_steps:
        # linear temperature schedule
        frac = t / total_steps
        temp = init_temp * (1 - frac) + final_temp * frac
        # propose move: transfer small amount between random i,j
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j >= i:
            j+=1
        delta = (np.random.rand() * 0.02) * cur[i]  # up to 2% of value
        if delta <= 0:
            continue
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new_score = _max_convolution_score(new)
        diff = new_score - cur_score
        if diff < 0 or np.random.rand() < np.exp(-diff / (temp+1e-12)):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = new.copy()
                # maybe print
                if t % 1000 ==0:
                    print(f"[{time.time():.1f}] SA iter {t} new best {best_score:.6f}")
        t += 1
    return best, best_score

The objective is  

\[
\min_{a\ge 0,\;\sum a_i = 1}\;2\,n\;\max_k (a*a)_k ,
\]

i.e. minimise the peak of the self‑convolution of a non‑negative step‑function.
The code below implements a *hybrid* optimisation:

1. **Initial families** – a handful of analytic shapes (uniform, arcsine, β‑distribution,
   triangular, exponential and edge‑block) for several lengths.  
   The best one becomes the first seed.

2. **Multi‑length *warm‑starting*** – the best seed is interpolated to other lengths
   (up/down‑sample) to keep the search space large without paying the full cost
   of a fresh random start each time.

3. **Smoothed soft‑max descent** – minimise the smooth surrogate  

   \[
   \mu_\alpha(a)=\frac{1}{\alpha}\log\!\Bigl(\sum_k e^{\alpha\,(a*a)_k}\Bigr)
   \]

   using Adam.  The smoothing parameter α is increased geometrically (1 → 5 → 20 → 100 → 500 → 2000)
   and several thousand steps are taken for each stage.

4. **Exact max‑sub‑gradient refinement** – after the continuation phase a true
   sub‑gradient of the max‑function is used (as in the reference code) together
   with momentum.

5. **Simulated‑annealing hill‑climber** – random transfers of mass between two
   positions are accepted with a Metropolis rule; the temperature decays
   exponentially.  This helps to escape tiny plateaux that gradient‑based
   methods may miss.

6. **Local greedy hill‑climber** – a fast deterministic move that swaps a tiny
   amount of mass from a higher‑peak region to a lower‑peak one.

All steps project onto the probability simplex after each update
(`_project_onto_simplex`).  The whole routine lives inside a 1000 s time‑budget,
periodically printing the current best bound.  The final list of floats is
returned; the external `evaluate_sequence` will be called by the judge.

```python
"""
Hybrid optimisation for the min‑max self‑convolution problem.

* generate analytic candidates for several lengths,
* up‑sample / down‑sample the current best to explore larger n,
* smooth soft‑max (log‑sum‑exp) descent with Adam and a geometric α‑schedule,
* exact max‑sub‑gradient refinement (momentum),
* simulated‑annealing hill‑climber,
* final greedy hill‑climber.

All vectors are kept on the simplex (non‑negative, sum = 1).  The routine
stops after the allotted time (default 1000 s) and returns the best sequence
found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_edge_block(n: int, frac: float = 0.1) -> np.ndarray:
    """
    Two uniform blocks at the left and right ends.
    `frac` is the total fraction of the vector occupied (split equally).
    """
    block_len = max(1, int(frac * n / 2))
    a = np.zeros(n, dtype=np.float64)
    a[:block_len] = 1.0
    a[-block_len:] = 1.0
    if a.sum() == 0:
        return _distribution_uniform(n)
    a /= a.sum()
    return a

def _initial_candidates(n: int) -> list[np.ndarray]:
    """Gather a small but diverse set of starts for a given length."""
    cand = [
        _distribution_uniform(n),
        _distribution_arcsine(n),
        _distribution_triangular(n),
        _distribution_exponential(n, lam=0.003),
        _distribution_exponential(n, lam=0.01),
    ]
    for p in (0.2, 0.4, 0.6, 0.8):
        cand.append(_distribution_beta_sym(n, p))
    for f in (0.05, 0.10, 0.15):
        cand.append(_distribution_edge_block(n, f))
    # a few random sparse starts
    for _ in range(3):
        k = max(2, n // 200)
        idx = np.random.choice(n, k, replace=False)
        a = np.zeros(n, dtype=np.float64)
        a[idx] = np.random.rand(k)
        a /= a.sum()
        cand.append(a)
    return cand

def _smooth_descent(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate of max.
    Returns the best vector (simplex‑projected) and its exact score.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # (α, steps) schedule – we go up to α=2000 for a very tight max‑approx
    schedule = [
        (1.0, 1500),
        (5.0, 1500),
        (20.0, 1500),
        (100.0, 1500),
        (500.0, 1500),
        (2000.0, 2000),
    ]

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score
            t += 1

            b = _fft_self_convolve(a)            # full self‑conv
            maxb = b.max()
            # stable soft‑max
            exp_terms = np.exp(alpha * (b - maxb))
            w = exp_terms / exp_terms.sum()      # weights sum to 1

            # gradient = 2 * a_rev * w   (first n entries)
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam update + simplex projection
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a, 1.0)

            # occasional exact evaluation
            if t % 200 == 0:
                cur_score = _max_convolution_score(a)
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_seq = a.copy()
    return best_seq, best_score

def _exact_subgradient_refine(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Exact max‑subgradient descent with momentum and a diminishing step size.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)

    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps)

        w = np.zeros_like(b)
        w[active] = 1.0

        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / np.sqrt(iteration + 1)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad it={iteration} new best {best_score:.6f}")

        if best_score <= 1.503:
            break
        if time.time() + 0.4 >= deadline:
            break
    return best_seq, best_score

def _annealing_refine(a: np.ndarray, deadline: float,
                      init_temp: float = 0.08,
                      final_temp: float = 1e-5,
                      max_steps: int = 40000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climber.
    Randomly moves a tiny amount of mass between two positions,
    accepting worsening moves with Metropolis probability.
    """
    n = a.shape[0]
    cur = a.copy()
    cur_score = _max_convolution_score(cur)
    best = cur.copy()
    best_score = cur_score

    step = 0
    while time.time() < deadline and step < max_steps:
        # linear temperature decay
        frac = step / max_steps
        temp = init_temp * (1.0 - frac) + final_temp * frac

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # transfer up to 3 % of the source mass
        delta = np.random.rand() * 0.03 * cur[i]
        if delta <= 0:
            step += 1
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        new_score = _max_convolution_score(cand)
        diff = new_score - cur_score

        if diff < 0 or np.random.rand() < math.exp(-diff / (temp + 1e-12)):
            cur = cand
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best = cand.copy()
                if step % 2000 == 0:
                    print(f"[{time.time():.1f}] SA step={step} best={best_score:.6f}")

        step += 1
    return best, best_score

def _local_hill_climb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Deterministic greedy hill climbing:
    move a small amount of probability mass from a random i to a random j
    only if it strictly improves the exact max‑score.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)
    cur_score = best_score

    max_transfer = 0.07
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        transfer = max_transfer * (0.5 ** (iteration / 5000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[{time.time():.1f}] hill iter={iteration} new best {best_score:.6f}")

        if iteration % 5000 == 0 and np.random.rand() < 0.2:
            # occasional jitter to shake out plateaux
            pert = np.random.randn(n) * 1e-4
            a = best_seq + pert
            a = _project_onto_simplex(a, 1.0)
            cur_score = _max_convolution_score(a)

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence found (Python list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1.  Coarse sampling from analytic families for several lengths
    lengths = [800, 1600, 3200, 5000, 8000, 12000, 16000]
    overall_best_seq = None
    overall_best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        for cand in _initial_candidates(n):
            if time.time() >= deadline:
                break
            cand = cand / cand.sum()               # normalise to sum = 1
            score = _max_convolution_score(cand)
            if score < overall_best_score:
                overall_best_score = score
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] n={n} init score={score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Include a previous construction if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # Fallback – should never happen
    if overall_best_seq is None:
        n0 = max(800, lengths[0])
        overall_best_seq = _distribution_uniform(n0)
        overall_best_score = _max_convolution_score(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Main optimisation loop – we repeatedly pick a start,
    #     possibly rescaled to a new length, and run the full pipeline.
    def run_full_pipeline(start_vec: np.ndarray, dl: float) -> tuple[np.ndarray, float]:
        a, sc = _smooth_descent(start_vec, dl)
        a, sc = _exact_subgradient_refine(a, dl)
        a, sc = _annealing_refine(a, dl)
        a, sc = _local_hill_climb(a, dl)
        return a, sc

    # Keep a copy of the best vector we have so far
    best_vec = overall_best_seq.copy()
    best_score = overall_best_score

    while time.time() < deadline:
        # Decide whether to change length (geometric up/down‑sample)
        if np.random.rand() < 0.25:
            factor = np.random.choice([0.7, 0.85, 1.15, 1.4])
            new_len = max(500, int(best_vec.shape[0] * factor))
            # Linear interpolation to the new length
            x_old = np.linspace(0, 1, best_vec.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, new_len, endpoint=False)
            a_interp = np.interp(x_new, x_old, best_vec)
            a_interp = np.maximum(a_interp, 0.0)
            if a_interp.sum() == 0:
                a_interp = _distribution_uniform(new_len)
            else:
                a_interp = a_interp / a_interp.sum()
            start = a_interp
        else:
            # Small random jitter around the current best
            jitter = np.random.randn(best_vec.shape[0]) * 1e-3
            start = best_vec + jitter
            start = np.maximum(start, 0.0)
            if start.sum() == 0:
                start = best_vec.copy()
            else:
                start = start / start.sum()

        # Run the whole optimisation pipeline on this start
        a_opt, sc_opt = run_full_pipeline(start, deadline)

        if sc_opt < best_score - 1e-12:
            best_score = sc_opt
            best_vec = a_opt.copy()
            print(f"[{time.time():.1f}] NEW OVERALL BEST {best_score:.6f} (len={len(best_vec)})")
            if best_score <= 1.503:
                break

    # ------------------------------------------------------------------
    # 4.  Final sanity normalisation and return
    best_vec = np.maximum(best_vec, 0.0)
    if best_vec.sum() == 0:
        best_vec = _distribution_uniform(best_vec.size)
    else:
        best_vec = best_vec / best_vec.sum()
    print(f"Final best score: {best_score:.6f} (length={len(best_vec)})")
    return best_vec.tolist()
```
