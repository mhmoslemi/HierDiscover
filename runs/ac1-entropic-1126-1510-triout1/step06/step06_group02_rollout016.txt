"""
Propose a non‑negative step function (list of heights) that minimizes
`evaluate_sequence`.  The algorithm works in three stages:

1.  **Candidate generation** – for a handful of problem sizes we
    synthesize many analytical shapes (Beta, exponential, power‑law,
    triangular, random Dirichlet, previous best, …) and keep the
    cheapest few according to the cheap objective
    ``2·n·max(conv(a,a))`` (the exact `evaluate_sequence` value when the
    vector sums to 1).

2.  **Projected Adam refinement** – each keeper is optimised on the
    probability simplex using Adam on the soft‑max parametrisation.
    A smooth soft‑max surrogate of the `max` is used early on,
    switching to the exact sub‑gradient later.  The learning rate,
    temperature schedule and jitter are tuned for fast convergence.

3.  **Selection** – the best refined vector is finally evaluated with the
    real black‑box `evaluate_sequence`.  The globally best sequence
    (lowest score) is returned.  The search respects the 1000 s budget
    and aborts early as soon as a score ≤ 1.503 is found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float = 1.0) -> np.ndarray:
    """Geometric‑like decreasing shape: a_i ∝ exp(-lam·i/(n‑1))."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * i / max(1, n - 1))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _power_law_distribution(n: int, p: float, symmetric: bool = False) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} (or U‑shaped if symmetric=True)."""
    i = np.arange(n, dtype=np.float64) + 1.0
    if symmetric:
        vals = i ** (-p) + (n + 1 - i) ** (-p)
    else:
        vals = i ** (-p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _triangular_distribution(n: int) -> np.ndarray:
    """Mount‑like shape that rises then falls linearly."""
    k = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    vals = np.minimum(k, n - 1 - k) + 1.0
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate of max(conv).
    a : current probability vector (Σa=1)
    conv : self‑convolution of a
    alpha : temperature multiplier (larger → sharper)
    """
    n = a.shape[0]
    cur_max = conv.max()
    exp_term = np.exp(alpha * (conv - cur_max))
    w = exp_term / exp_term.sum()
    # convolution of reversed a with the weighting vector w
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the true max: average over all active indices.
    """
    n = a.shape[0]
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (remove one copy of a_i*a_i for even shifts)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad


# ----------------------------------------------------------------------
def _adam_step(z: np.ndarray,
               g: np.ndarray,
               m: np.ndarray,
               v: np.ndarray,
               t: int,
               lr: float = 0.5,
               beta1: float = 0.9,
               beta2: float = 0.999,
               eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Adam update on the unconstrained variable z."""
    m = beta1 * m + (1.0 - beta1) * g
    v = beta2 * v + (1.0 - beta2) * (g * g)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z, m, v


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 1234, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimizes
    ``evaluate_sequence`` within the allotted time.
    Returns the best list of heights found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # decide a base length – use the length of the previously known
    # construction if it exists, otherwise fall back to 8000.
    try:
        base_n = len(height_sequence_1)      # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # Breadth of lengths we will inspect.
    length_factors = (0.5, 0.75, 1.0, 1.25, 1.5, 2.0)
    lengths = sorted(
        max(2000, int(base_n * f))
        for f in length_factors
        if int(base_n * f) >= 2000
    )
    # guarantee at least one length
    if not lengths:
        lengths = [base_n]

    # ------------------------------------------------------------------
    def cheap_score(a: np.ndarray) -> float:
        """Score = 2·n·max(conv) for a vector that already sums to 1."""
        return 2.0 * a.shape[0] * _fft_self_convolve(a).max()

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    target_score = 1.503

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        remaining = max(0.0, deadline - time.time())
        slice_deadline = time.time() + min(remaining * 0.35, 180.0)
        print(f"[{time.time():.1f}] ── Exploring length n={n}")

        # --------------------------------------------------------------
        # 1. Build a diverse pool of analytic candidates.
        pool: list[np.ndarray] = []

        # uniform
        pool.append(np.full(n, 1.0 / n, dtype=np.float64))

        # beta grid
        alphas = np.logspace(-1, 1, num=9)   # 0.1 … 10
        betas = np.logspace(-1, 1, num=9)
        for a in alphas:
            for b in betas:
                pool.append(_beta_distribution(n, a, b))

        # exponentials
        for lam in (0.2, 0.5, 1.0, 2.0):
            pool.append(_exponential_distribution(n, lam))

        # power‑law shapes (both decreasing and U‑shaped)
        for p in (0.5, 1.0, 1.5, 2.0):
            pool.append(_power_law_distribution(n, p, symmetric=False))
            pool.append(_power_law_distribution(n, p, symmetric=True))

        # triangular
        pool.append(_triangular_distribution(n))

        # a few random Dirichlet draws (low concentration → spikier)
        for _ in range(4):
            g = np.random.gamma(0.3, 1.0, size=n)
            g /= g.sum()
            pool.append(g)

        # previous best construction, interpolated if needed
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.size != n:
                x_old = np.linspace(0.0, 1.0, prev.size)
                x_new = np.linspace(0.0, 1.0, n)
                pool.append(np.interp(x_new, x_old, prev))
            else:
                pool.append(prev.copy())
        except Exception:
            pass

        # --------------------------------------------------------------
        # 2. Keep the cheapest few candidates.
        scored = [(cheap_score(a), a) for a in pool]
        scored.sort(key=lambda x: x[0])
        top_k = 8
        starters = [a.copy() for _, a in scored[:top_k]]
        print(f"   top cheap scores (first {top_k}): {[s for s, _ in scored[:top_k]]}")

        # --------------------------------------------------------------
        # 3. Refine each starter with projected Adam on the simplex.
        for sid, a0 in enumerate(starters):
            if time.time() >= slice_deadline:
                break

            # allocate a sub‑deadline for this starter
            now = time.time()
            sub_deadline = now + min(80.0, slice_deadline - now)
            if sub_deadline - now < 5.0:
                continue

            # initialise log‑odds
            a = np.maximum(a0, 1e-15)
            z = np.log(a)
            m = np.zeros_like(z)
            v = np.zeros_like(z)

            # optimisation hyper‑parameters
            lr0 = 0.6
            beta1, beta2 = 0.9, 0.999
            eps = 1e-8

            temperature = 0.5
            temp_factor = 1.015
            temp_max = 5000.0

            best_local_a = a.copy()
            best_local_score = cheap_score(a)
            it = 0

            while time.time() < sub_deadline:
                it += 1
                a = np.exp(z)
                a /= a.sum()
                conv = _fft_self_convolve(a)

                cur_score = 2.0 * n * conv.max()
                if cur_score < best_local_score:
                    best_local_score = cur_score
                    best_local_a = a.copy()
                    print(f"[{time.time():.1f}] n={n} starter={sid} it={it} score={cur_score:.6f}")

                    if cur_score <= target_score:
                        break

                # gradient: start with soft‑max, later switch to exact sub‑gradient
                if temperature < temp_max:
                    grad_a = _softmax_gradient(a, conv, temperature)
                    temperature *= temp_factor
                else:
                    grad_a = _exact_gradient(a, conv)

                # convert to gradient w.r.t. logits z
                dot = np.dot(a, grad_a)
                grad_z = a * (grad_a - dot)

                # Adam step
                z, m, v = _adam_step(z, grad_z, m, v, it,
                                     lr=lr0, beta1=beta1, beta2=beta2, eps=eps)

                # occasional jitter to escape tiny plateaus
                if it % 3000 == 0 and np.random.rand() < 0.15:
                    z += np.random.randn(n) * 1e-7

            # ----------------------------------------------------------
            # Evaluate the best local candidate with the true black‑box
            try:
                final_score = evaluate_sequence(best_local_a.tolist())
            except Exception:
                final_score = best_local_score   # fall back to cheap estimate

            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = best_local_a.copy()
                print(f"*** Global improvement → n={n} starter={sid} "
                      f"score={overall_best_score:.6f}")
                if overall_best_score <= target_score:
                    print("Target reached – exiting early.")
                    return overall_best_seq.tolist()

        # global time check
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Fallback – should never be needed, but guarantees a valid return.
    if overall_best_seq is None:
        n_fallback = 4000
        overall_best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())
        print("Fallback used.")

    print(f"Final best score: {overall_best_score:.6f} (length = {len(overall_best_seq)})")
    return overall_best_seq.tolist()
