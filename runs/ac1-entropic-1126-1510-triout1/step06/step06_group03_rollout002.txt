"""
Propose a step function using a smoothed‐max gradient descent.

The objective to minimise is 2·n·max(a*a)  (a is the non‑negative
sequence with Σa = 1).  The maximum of the self‑convolution is approximated
by a soft‑max (log‑sum‑exp) with a temperature β that is slowly increased.
For a given β the gradient of the smooth approximation is
    ∇f(a) = 2·conv_rev(a, w)      with   w_k ∝ exp(β·conv_k) .
After each gradient step a is projected onto the simplex {x≥0, Σx=1}
using Euclidean projection.  The method is run on a schedule of
increasing lengths (2 k, …, 40 k) – each length is warm‑started by
linearly interpolating the best sequence found so far.  A final
projected sub‑gradient refinement (exact max) is applied before returning
the result.

The routine respects the 1000 s budget, uses only NumPy, and prints
periodic progress reports.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Returns (grad, max_conv) where grad is the gradient of the
    smoothed max (log‑sum‑exp) of the self‑convolution.
    """
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    # softmax weights (stable)
    exp_weights = np.exp(beta * (conv - cur_max))
    w = exp_weights / exp_weights.sum()
    # gradient: 2 * conv_rev(a, w)
    grad_full = _fft_convolve(a[::-1], w)
    n = a.shape[0]
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad, cur_max

def _gradient_descent_smooth(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 5000,
    step0: float = 0.5,
    beta_start: float = 0.3,
    beta_end: float = 30.0,
) -> tuple[np.ndarray, float]:
    """
    Smoothed‑max gradient descent with Euclidean projection onto the simplex.
    Returns the best sequence encountered and its raw score 2·n·max(conv).
    """
    a = a0.copy()
    best_seq = a.copy()
    best_raw = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        # current objective (exact max, for monitoring)
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        raw = 2.0 * n * cur_max
        if raw < best_raw:
            best_raw = raw
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] n={n} it={it:05d} raw={best_raw:.6f} beta≈{beta_start + (beta_end - beta_start) * min(it / max_iters, 1.0):.2f}"
            )

        # schedule smoothing parameter β
        beta = beta_start + (beta_end - beta_start) * min(it / max_iters, 1.0)

        # gradient of smoothed max
        grad, _ = _smooth_max_gradient(a, beta)

        # step size (diminishing)
        step = step0 / math.sqrt(it)

    final_conv = _fft_self_convolve_full(best_seq)
    best_raw = 2.0 * n * final_conv.max()
    return best_seq, best_raw
