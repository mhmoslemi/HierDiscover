def _adam_optimize_with_variance(x0, n, deadline, max_iters=6000, lr0=0.12, gamma=30.0, lam=0.2):
    x = x0.copy()
    m = np.zeros_like(x); v = np.zeros_like(x)
    beta1=0.9; beta2=0.999; eps=1e-12
    best_a = None; best_score = np.inf
    it = 0
    inv_two_n_minus_one = 1.0 / (2 * n - 1)

    while it < max_iters and time.time() < deadline:
        it += 1
        # softmax
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()
        # compute conv
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        raw_score = 2.0 * n * cur_max
        if raw_score < best_score:
            best_score = raw_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it} raw={best_score:.6f}")

        # compute loss components
        avg = inv_two_n_minus_one  # constant 1/(2n-1)
        # smooth max via softmax weights
        # Use stable softmax on gamma*conv
        tmp = gamma * conv
        max_tmp = np.max(tmp)
        exp_tmp = np.exp(tmp - max_tmp)
        w = exp_tmp / exp_tmp.sum()  # sum to 1, approximates argmax
        # gradient from smooth max
        a_rev = a[::-1]
        grad_max = 2.0 * _fft_convolve(a_rev, w)[n-1 : n-1+n]

"""
optimiser_with_variance.py

A step‑function optimiser that directly minimises a smoothed version of the
objective 2·n·max(a*a) while penalising the variance of the full
autoconvolution.  The loss is

    L(a) = (1/γ)·log Σ exp(γ·(a*a))          # smooth max
           + λ· Σ((a*a) - μ)²                # variance, μ = 1/(2n‑1)

Gradient of L is obtained analytically via FFTs, the simplex constraint
(Σa = 1, a ≥ 0) is handled by a soft‑max parametrisation of the logits
followed by an Adam optimiser.  Several warm‑starts (interpolated previous
best, uniform, random Dirichlet draws) are tried per length, and the best
candidate is kept.  After the main phase a short projected sub‑gradient
refinement (as in the reference solution) is applied.

The function ``propose_candidate`` follows the required signature and
returns a plain Python list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    v = v.astype(np.float64, copy=True)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of max_k (a*a)_k.
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(conv >= cur_max - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


# ----------------------------------------------------------------------
def _adam_optimize_with_variance(
    x0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 6000,
    lr0: float = 0.12,
    gamma: float = 30.0,
    lam: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑it parametrisation, minimising a smoothed
    max + λ·variance loss.
    Returns the best vector (a) found and its raw score 2·n·max(a*a).
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    best_a = None
    best_score = float("inf")
    inv_mu = 1.0 / (2 * n - 1)          # μ = 1/(2n‑1)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # --- soft‑max (stable) ---
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()          # Σa = 1, a ≥ 0

        # --- current convolution ---
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        raw_score = 2.0 * n * cur_max
        if raw_score < best_score:
            best_score = raw_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it} raw={best_score:.6f}")

        # --- smooth max (log‑sum‑exp) ---
        tmp = gamma * conv
        max_tmp = np.max(tmp)
        exp_tmp = np.exp(tmp - max_tmp)
        w = exp_tmp / exp_tmp.sum()      # soft‑max weights (Σw = 1)

        a_rev = a[::-1]

        # gradient of the smooth max term
        grad_max = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]

        # gradient of the variance term λ·Σ(c‑μ)²
        delta = conv - inv_mu
        grad_var = 4.0 * lam * _fft_convolve(a_rev, delta)[n - 1 : n - 1 + n]

        grad_a = grad_max + grad_var

        # --- chain rule through the soft‑max ---
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to help escaping plateaus
        if it % 1000 == 0:
            x += np.random.randn(*x.shape) * 0.01

    # fallback (should never happen)
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_max(best_a)

    return best_a, best_score


# ----------------------------------------------------------------------
def _optimise_length_with_variance(
    n: int,
    deadline: float,
    prev_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length n using several starts and the variance‑augmented
    Adam optimiser.  Returns the best sequence (normalised to sum=1) and its raw score.
    """
    init_logits = []

    # warm‑start from an interpolated predecessor, if available
    if prev_seq is not None:
        cand = _interpolate_to_length(prev_seq, n)
        eps = 1e-12
        init_logits.append(np.log(cand + eps))

    # uniform start
    init_logits.append(np.zeros(n, dtype=np.float64))

    # a few random Dirichlet draws
    for _ in range(3):
        alpha = np.random.rand() * 4.0 + 0.1      # α ∈ (0.1, 4.1)
        dir_sample = np.random.gamma(alpha, 1.0, size=n)
        dir_sample /= dir_sample.sum()
        init_logits.append(np.log(dir_sample + 1e-12))

    best_seq = None
    best_score = float("inf")

    for ix, x0 in enumerate(init_logits):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] length {n}, start {ix+1}/{len(init_logits)}")
        seq, score = _adam_optimize_with_variance(
            x0, n, deadline, max_iters=6000, lr0=0.12, gamma=30.0, lam=0.20
        )
        if score < best_score:
            best_score = score
            best_seq = seq.copy()

    # safety fallback
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_max(best_seq)

    return best_seq, best_score
