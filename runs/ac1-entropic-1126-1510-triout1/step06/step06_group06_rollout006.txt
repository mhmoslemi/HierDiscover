```
    start = time.time()
    deadline = start + budget_s - 0.5

    except:
        base_seq = None
        base_n = 4000

    # Phase0: generate initial candidates per length and per family.
    candidates = []  # list of (seq, fast_score)
    for n in lengths:
        # families
        fams = {}
        # arcsine
        i = np.arange(n, dtype=np.float64)
        arcs = 1.0/np.sqrt((i+0.5)*(n-i-0.5))

        fams["arcsine"] = arcs
        # uniform

        fams["uniform"] = uni
        # power-law p in [0.5, 2], maybe several.
        for p in [0.5,0.8,1.0,1.2,1.5,2.0]:
            pl = 1.0/((i+1)**p)
            pl = pl/pl.sum()
            fams[f"pow_{p}"] = pl
        # exponential decays lambda
        for lam in [0.001,0.002,0.005,0.01]:
            expd = np.exp(-lam * i)
            expd = expd/expd.sum()
            fams[f"exp_{lam}"] = expd
        # quadratic (tent shape)
        tent = 1.0 - np.abs((i - (n-1)/2))/( (n-1)/2)
        tent = np.maximum(tent,0)
        tent = tent/tent.sum()

        # maybe mixture arcsine+tent
        mix = 0.5*arcs + 0.5*tent
        mix = mix/mix.sum()

        # previous best interpolation if exists

            vec = _project_onto_simplex(vec,1.0)
```

The routine searches for a non‑negative step‑function `a = (a₀,…,aₙ₋₁)` that minimises  

\[
\text{score}(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 ,
\]

where `(a*a)` is the linear convolution of `a` with itself.  
Because the score is homogeneous, we may restrict the vectors to the simplex
`Σ a_i = 1`.  

The algorithm consists of four stages

1. **Broad family sampling** – a few deterministic families (arcsine, uniform,
   power‑law, exponential, triangular, two‑level blocks, …) are generated for a
   small set of lengths.  The best few candidates (by the cheap *fast score*
   `2·n·max(conv)`) are kept.

2. **Simulated‑annealing refinement** – each kept candidate is improved by a
   Metropolis‑type local search that moves a small amount of mass from an index
   that participates heavily in the current maximal convolution term to a
   random destination.  The temperature is decreased linearly, allowing the
   search to escape shallow local minima.

3. **Fine‑grained optimisation** – the best candidate from stage 2 is
   processed by the Adam optimiser on a smooth‑max surrogate, followed by an
   exact sub‑gradient descent and a directed greedy mass‑transfer search (the
   same building blocks that were used in the previous state‑of‑the‑art code).

4. **Final evaluation** – the black‑box `evaluate_sequence` is called once for
   the final vector and the best vector found is returned.

All heavy work uses FFT‑based convolutions (`O(n log n)`) and every vector is
kept on the probability simplex (`Σa_i = 1`).  The programme respects the
1000 s budget, prints progress and never writes to the file system.

```python
"""
Propose a low‑score non‑negative step function.

Algorithm
---------

1.  Generate a handful of simple families (arcsine, uniform, power‑law,
    exponential, triangular, two‑level blocks, …) for several lengths
    around the size of the previously known best construction.  Keep the
    best few vectors according to the cheap “fast score”
    `2·n·max(conv)`.

2.  For each kept vector run a simulated‑annealing search that repeatedly
    moves a small fraction of mass from an index that contributes strongly
    to the current maximal convolution term to a random destination.
    The move is accepted if it reduces the maximal convolution or with a
    Metropolis probability `exp(-(Δmax)/T)`.  Temperature `T` is lowered
    linearly to 0.  The best vector found by this stage becomes the new
    seed.

3.  Fine‑tune the seed with the Adam optimiser on a smooth‑max surrogate,
    then perform exact sub‑gradient descent followed by the greedy
    “directed local search” used in the reference implementation.

4.  Evaluate the final candidate with the black‑box `evaluate_sequence`
    and return the best vector (as a plain Python list).

The implementation is self‑contained, uses only NumPy, and respects the
1000 s time budget.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # soft‑max weights (numerically stable)
    exp_term = np.exp(alpha * (b - cur_max))
    w_sum = exp_term.sum()
    w = exp_term / w_sum
    # convolution of reversed a with the weight vector
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.01,
                   alpha_max: float = 5000.0,
                   lr0: float = 0.3,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best probability vector encountered.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size schedule (slow decay)
        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # keep the best exact peak seen so far
        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        # continuation on the smoothness parameter
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny jitter
        if t % 400 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    One sub‑gradient step for the exact ℓ∞ objective.
    Uses an equal weight on all indices attaining the current maximum.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Sub‑gradient descent on the exact max‑convolution objective.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.02 / np.sqrt(it)  # diminishing step
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        # occasional jitter to avoid stagnation
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


# ----------------------------------------------------------------------
def _directed_local_search(a0: np.ndarray, deadline: float):
    """
    Greedy mass‑transfer search that always targets the current peak.
    Only improving moves are accepted.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        max_idx = np.random.choice(np.where(b >= cur_max - 1e-12)[0])
        i_low = max(0, max_idx - (n - 1))
        i_high = min(max_idx, n - 1)
        i_vals = np.arange(i_low, i_high + 1, dtype=int)
        prod = a[i_vals] * a[max_idx - i_vals]
        if prod.sum() == 0.0:
            src = np.random.randint(n)
        else:
            prob = prod / prod.sum()
            src = np.random.choice(i_vals, p=prob)

        dst = np.random.randint(n)
        if dst == src:
            continue

        max_delta = min(a[src], 0.05)          # up to 5 % of src
        delta = np.random.rand() * max_delta

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        b_new = _fft_self_convolve(a_new)
        new_max = b_new.max()

        if new_max < cur_max - 1e-15:         # accept only if improvement
            a = a_new
            b = b_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()

        # occasional random jitter to explore neighbourhood
        if np.random.rand() < 0.005:
            a += np.random.randn(n) * 1e-7
            a = _project_onto_simplex(a, s=1.0)
            b = _fft_self_convolve(a)
            cur_max = b.max()

    return best_a


# ----------------------------------------------------------------------
def _two_level_family(n: int, deadline: float):
    """
    Search a simple two‑level family:
        a_i = p   for i < L   (front block)
        a_i = q   otherwise   (back block)  with p = r·q , r ≥ 1.
    Returns the best vector and its fast score.
    """
    best_seq = None
    best_score = float('inf')
    step_L = max(1, n // 100)          # ≈1 % of the length
    r_grid = np.logspace(0.0, 2.3, num=25)   # from 1 to ≈200

    for L in range(step_L, n, step_L):
        if time.time() > deadline:
            break
        denom_base = n - L
        for r in r_grid:
            denom = L * r + denom_base
            q = 1.0 / denom
            p = r * q
            seq = np.empty(n, dtype=np.float64)
            seq[:L] = p
            seq[L:] = q
            peak = _fft_self_convolve(seq).max()
            score = 2.0 * n * peak
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
    return best_seq, best_score


# ----------------------------------------------------------------------
def _annealing_optimize(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simulated‑annealing refinement.
    Moves a small fraction of mass from an index that contributes strongly
    to the current peak to a random destination.
    Accepts worsening moves with a Metropolis probability that decays linearly.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    start = time.time()
    total = max(deadline - start, 1e-12)
    init_T = cur_max * 0.10   # initial temperature ≈10 % of the peak

    while time.time() < deadline:
        # linear cooling schedule
        elapsed = time.time() - start
        T = init_T * max(0.0, 1.0 - elapsed / total)

        # pick an index belonging to a maximal convolution term
        max_idx = np.random.choice(np.where(b >= cur_max - 1e-12)[0])
        i_low = max(0, max_idx - (n - 1))
        i_high = min(max_idx, n - 1)
        i_vals = np.arange(i_low, i_high + 1, dtype=int)
        contributions = a[i_vals] * a[max_idx - i_vals]
        s = contributions.sum()
        if s > 0:
            probs = contributions / s
            src = np.random.choice(i_vals, p=probs)
        else:
            src = np.random.randint(n)

        dst = np.random.randint(n)
        if dst == src:
            continue

        # move up to 20 % of the source mass
        max_delta = min(a[src], 0.20)
        if max_delta <= 1e-12:
            continue
        delta = np.random.rand() * max_delta

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        # keep exact sum (numerical drift is negligible, we keep the simplex only
        # when the move is accepted)
        b_new = _fft_self_convolve(a_new)
        new_max = b_new.max()

        if new_max < cur_max:
            a, b, cur_max = a_new, b_new, new_max
            if new_max < best_max:
                best_max, best_a = new_max, a_new.copy()
        else:
            # Metropolis accept/reject
            if T > 0 and np.random.rand() < np.exp(-(new_max - cur_max) / T):
                a, b, cur_max = a_new, b_new, new_max

        # tiny occasional jitter to avoid stagnation
        if np.random.rand() < 0.001:
            a += np.random.randn(n) * 1e-7
            a = _project_onto_simplex(a, s=1.0)
            b = _fft_self_convolve(a)
            cur_max = b.max()

    return best_a

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Main entry point called by the competition framework.
    Returns a plain Python list of non‑negative numbers.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Try to reuse a previously discovered construction if it exists.
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_seq = _project_onto_simplex(base_seq, s=1.0)
        base_n = len(base_seq)
    except Exception:
        base_seq = None
        base_n = 4000

    # ------------------------------------------------------------------
    # Lengths to explore (±30 % around the previous best, at least 500)
    lengths = [max(500, int(base_n * f))
               for f in (0.7, 0.85, 1.0, 1.15, 1.3)
               if int(base_n * f) >= 500]
    if not lengths:
        lengths = [max(500, base_n)]

    # ------------------------------------------------------------------
    # Stage 0 – broad sampling of simple families
    print("[stage 0] broad family sampling")
    initial_candidates = []   # list of (vector, fast_score)

    for n in lengths:
        if time.time() >= deadline:
            break

        families = {}

        # arcsine (classic)
        i = np.arange(n, dtype=np.float64)
        arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
        families["arcsine"] = arcs / arcs.sum()

        # uniform
        families["uniform"] = np.full(n, 1.0 / n, dtype=np.float64)

        # power‑law a_i ∝ (i+1)^{-p}
        for p in (0.5, 0.8, 1.0, 1.2, 1.5, 2.0):
            pw = 1.0 / (i + 1) ** p
            families[f"pow_{p:.1f}"] = pw / pw.sum()

        # exponential decay
        for lam in (0.0005, 0.001, 0.002, 0.005):
            expd = np.exp(-lam * i)
            families[f"exp_{lam:.4f}"] = expd / expd.sum()

        # triangular (tent) shape
        tent = 1.0 - np.abs(i - (n - 1) / 2) / ((n - 1) / 2)
        tent = np.maximum(tent, 0.0)
        families["tent"] = tent / tent.sum()

        # mixture of arcsine and uniform
        for w in (0.2, 0.4, 0.6):
            mix = (1.0 - w) * families["arcsine"] + w * families["uniform"]
            families[f"mix_{w:.1f}"] = mix / mix.sum()

        # two‑level parametric family (fast search)
        seq2, score2 = _two_level_family(n, deadline)
        if seq2 is not None:
            families["two_level"] = seq2

        # interpolation of the previous best if available
        if base_seq is not None:
            old_n = len(base_seq)
            x_old = np.linspace(0.0, 1.0, old_n)
            x_new = np.linspace(0.0, 1.0, n)
            interp = np.interp(x_new, x_old, base_seq)
            interp = np.maximum(0.0, interp)
            families["prev"] = interp / interp.sum()

        # evaluate fast scores
        for tag, vec in families.items():
            vec = _project_onto_simplex(vec, s=1.0)
            peak = _fft_self_convolve(vec).max()
            fast_score = 2.0 * n * peak
            initial_candidates.append((vec, fast_score))
            if fast_score < 1.6:  # just a loose filter for debug output
                print(f"[init] n={n:4d} {tag:<12s} fast_score={fast_score:.6f}")

    # keep the best few (up to 3) by fast_score
    initial_candidates.sort(key=lambda x: x[1])
    top_k = min(3, len(initial_candidates))
    candidates = initial_candidates[:top_k]

    best_seq, best_score = candidates[0]
    print(f"[stage 0] best fast_score={best_score:.6f} (len={len(best_seq)})")

    # ------------------------------------------------------------------
    # Stage 1 – simulated‑annealing refinement for each top candidate
    time_left = deadline - time.time()
    if time_left > 30.0 and candidates:
        per_candidate = min(200.0, time_left * 0.5 / len(candidates))
        print("[stage 1] simulated‑annealing refinement")
        for idx, (seq, _) in enumerate(candidates):
            if time.time() >= deadline:
                break
            cand_deadline = time.time() + per_candidate
            seq_opt = _annealing_optimize(seq, cand_deadline)
            cur_score = 2.0 * len(seq_opt) * _fft_self_convolve(seq_opt).max()
            if cur_score < best_score:
                best_score, best_seq = cur_score, seq_opt.copy()
                print(f"[anneal] new best fast_score={best_score:.6f} (len={len(best_seq)})")
            else:
                # keep the previous best if this run did not improve
                pass

    # ------------------------------------------------------------------
    # Stage 2 – fine‑grained optimisation (Adam + sub‑gradient + greedy)
    time_left = deadline - time.time()
    if time_left > 10.0:
        n = len(best_seq)
        print(f"[stage 2] Adam optimisation (n={n}) – time left {time_left:.1f}s")
        phase_end = time.time() + time_left * 0.55
        a_opt, _ = _adam_optimize(best_seq, phase_end)
        best_seq = a_opt.copy()
        best_score = 2.0 * n * _fft_self_convolve(best_seq).max()
        print(f"[stage 2] after Adam fast_score={best_score:.6f}")

        # sub‑gradient + directed greedy search
        time_left = deadline - time.time()
        if time_left > 5.0:
            print(f"[stage 2] sub‑gradient & greedy search (n={n}) – time left {time_left:.1f}s")
            sub_end = time.time() + time_left * 0.55
            a_ref = _refine_max(best_seq, sub_end)
            a_final = _directed_local_search(a_ref, deadline)
            best_seq = a_final.copy()
            best_score = 2.0 * n * _fft_self_convolve(best_seq).max()
            print(f"[stage 2] after greedy fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final black‑box evaluation
    try:
        true_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence score={true_score:.6f}")
        best_score = min(best_score, true_score)
    except Exception as exc:
        print("[final] evaluate_sequence raised:", exc)

    print(f"Best approx. score {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
```
