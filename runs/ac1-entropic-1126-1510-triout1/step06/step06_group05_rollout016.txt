"""
Propose a step‑function with a low evaluation score.

The algorithm consists of four stages:

1. **Broad exploration** – generate many candidate probability vectors of
   various lengths using analytically‑motivated families (arcsine, power‑law,
   Beta, Dirichlet, triangular, mixtures).  The cheap proxy
   `2·n·max(conv)` is used to keep the best few candidates.

2. **Heavy continuous optimisation** – on the best candidate we run a
   long Adam optimisation of the smooth‑max surrogate
   `Lα(a)=1/α·log Σ exp(α·(a*a)_k)` with a large final α (up to 10 000).
   Afterwards a short exact‑max sub‑gradient descent refines the solution.

3. **Optional LP‑based widening** – for moderate sizes (≤3000) a
   linear‑program is solved (via cvxpy) to enlarge the ℓ₁‑norm under the same
   convolution bound, and the result is mixed back.

4. **Stochastic polishing** – a light simulated‑annealing / hill‑climbing
   phase that repeatedly moves a small amount of mass between two positions,
   always accepting improvements and occasionally uphill moves.

All steps respect the simplex constraint Σaᵢ=1, aᵢ≥0 and stop as soon as the
global time budget is exhausted.  The final sequence (as a plain Python list)
is returned ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# optional LP support

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())   # next power of two ≥ 2n‑1
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 2).bit_length())
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution of length `n`."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) on a uniform grid, normalised to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_vals = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    vals[~np.isfinite(vals)] = 0.0
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _powerlaw_boundary_distribution(n: int, p: float) -> np.ndarray:
    """Distribution ∝ ((i+0.5)*(n‑i‑0.5))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5) + 1e-12) ** (-p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _triangular_distribution(n: int) -> np.ndarray:
    """Symmetric triangular shape: high at the ends, low in the centre."""
    i = np.arange(n, dtype=np.float64)
    vals = (i + 1) * (n - i)
    vals = vals.astype(np.float64)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """Gradient of Lα(a) = (1/α)·log Σ exp(α·(a*a)_k)."""
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # prevent overflow – shift by cur_max
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.02,
                   alpha_max: float = 10000.0,
                   lr0: float = 0.3,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """Adam optimisation of the smooth‑max surrogate."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # keep the best (smallest max convolution) seen so far
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        # increase the smoothness parameter
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny jitter to escape tiny plateaus
        if t % 500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """Exact ℓ∞ sub‑gradient step for max (a*a)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)

    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size

    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max

def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Short exact‑max refinement using sub‑gradient descent."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        # a slightly larger step than the original script
        step = 0.04 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        # occasional jitter to avoid stagnation
        if it % 400 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a

def _lp_improve(a: np.ndarray,
                max_iters: int = 2,
                mix_t: float = 0.5,
                inner_deadline: float = None) -> np.ndarray:
    """LP‑based improvement (dual widening) for moderate sized vectors."""
    if not _cvxpy_available:
        return a

    n = a.shape[0]
    for _ in range(max_iters):
        if inner_deadline is not None and time.time() > inner_deadline:
            break
        M = _fft_self_convolve(a).max()
        # build convolution matrix A (2n‑1)×n
        col = np.r_[a, np.zeros(n - 1)]
        row = np.r_[a[0], np.zeros(n - 1)]
        A = scipy.linalg.toeplitz(col, row)  # dense but acceptable for n≤3000

        b_var = cp.Variable(n, nonneg=True)
        constraints = [A @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception:  # pragma: no cover
            break
        if b_var.value is None:
            break
        b_vec = np.maximum(b_var.value, 0.0)
        s = b_vec.sum()
        if s <= 1e-12:
            break
        g = b_vec / s
        a = (1.0 - mix_t) * a + mix_t * g
        a = _project_onto_simplex(a, s=1.0)
    return a

def _simulated_annealing(a0: np.ndarray, deadline: float):
    """Light SA / hill‑climbing polish."""
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_peak = _fft_self_convolve(cur_a).max()
    best_a = cur_a.copy()
    best_peak = cur_peak

    start = time.time()
    total_time = max(deadline - start, 1e-6)

    start_temp = max(cur_peak, 1e-3) * 0.5
    final_temp = start_temp * 1e-5
    if final_temp <= 0.0:
        final_temp = 1e-9
    cooling_rate = -math.log(final_temp / start_temp) / total_time

    while time.time() < deadline:
        src = np.random.randint(n)
        if cur_a[src] <= 1e-12:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue
        # move a random fraction of the source mass (up to 5 %)
        max_delta = min(cur_a[src], 0.05)
        delta = np.random.rand() * max_delta

        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_peak = _fft_self_convolve(a_new).max()
        delta_obj = new_peak - cur_peak

        elapsed = time.time() - start
        temperature = start_temp * math.exp(-cooling_rate * elapsed)

        # Metropolis acceptance
        if delta_obj < 0.0:
            accept = True
        else:
            prob = math.exp(-delta_obj / max(temperature, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            cur_a = a_new
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_a = cur_a.copy()

        # occasional reset to the best found so far
        if np.random.rand() < 0.0005:
            cur_a = best_a.copy()
            cur_peak = best_peak

    return best_a, best_peak


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a non‑negative sequence to a new length, then renormalise."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals

        for conc in (0.1, 0.2, 0.5, 1.0, 2.0):
            candidates[f"dirichlet_{conc:g}"] = np.random.dirichlet(np.full(n, conc))

        # Symmetric beta families
        for a in (0.3, 0.4, 0.45, 0.55, 0.6, 0.7):
            candidates[f"beta_{a:g}"] = _beta_distribution(n, a, a)

        # Power‑law boundary families (including exponents below 0.5)
        for p in (0.30, 0.35, 0.40, 0.45, 0.50, 0.55):
            candidates[f"power_{p:g}"] = _powerlaw_boundary_distribution(n, p)

        # mixtures of arcsine and uniform
        for w in (0.2, 0.4, 0.6, 0.8):
            candidates[f"mix_{w:g}"] = (1 - w) * _arcsine_distribution(n) + w * np.full(n, 1.0 / n)

        # interpolation of the known good construction (if any)
        if base_seq is not None:
            candidates["prev_interp"] = _interpolate_to_length(base_seq, n)

        # evaluate cheap proxy and keep the best few across all lengths
        for tag, vec in candidates.items():
            vec = _project_onto_simplex(vec, s=1.0)
            peak = _fft_self_convolve(vec).max()
            fast_score = 2.0 * n * peak
            if len(best_seqs) < K_KEEP:
                best_seqs.append((fast_score, vec.copy()))
                best_seqs.sort(key=lambda x: x[0])
            else:
                if fast_score < best_seqs[-1][0]:
                    best_seqs[-1] = (fast_score, vec.copy())
                    best_seqs.sort(key=lambda x: x[0])
            # progress print (optional)
            if fast_score < best_seqs[0][0] + 1e-6:
                print(f"[init] n={n:4d} {tag:<12s} fast_score={fast_score:.6f}")

    if not best_seqs:
        # fallback – uniform of a reasonable size
        n = max(500, base_n)
        best_vec = np.full(n, 1.0 / n)
        best_seqs = [(2.0 * n * _fft_self_convolve(best_vec).max(), best_vec)]

    # take the absolute best from the shortlist as starting point
    overall_best_seq = best_seqs[0][1].copy()
    n = len(overall_best_seq)
    overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
    print(f"[phase 1] selected length {n}, initial fast_score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Adam optimisation + exact sub‑gradient refinement on the best candidate.
    time_left = deadline - time.time()
    if time_left > 5.0:
        print(f"[phase 2] heavy continuous optimisation – {time_left:.1f}s left")
        # allocate ~70 % of the remaining time to this phase
        phase_deadline = time.time() + time_left * 0.70
        a_opt, _ = _adam_optimize(overall_best_seq, phase_deadline,
                                   alpha_max=10000.0)  # stronger smoothing
        # sub‑gradient refinement gets the remaining 20 % of the original left time
        sub_deadline = time.time() + time_left * 0.20
        a_opt = _refine_max(a_opt, sub_deadline)
        overall_best_seq = a_opt.copy()
        n = len(overall_best_seq)
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print(f"[phase 2] post‑refinement fast_score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Optional LP improvement (only for moderate n to keep memory reasonable)
    time_left = deadline - time.time()
    if time_left > 2.0 and n <= 3000 and _cvxpy_available:
        print(f"[phase 3] LP widening – n={n}, {time_left:.1f}s left")
        lp_deadline = time.time() + time_left * 0.6
        a_lp = _lp_improve(overall_best_seq, max_iters=3,
                           mix_t=0.5, inner_deadline=lp_deadline)
        overall_best_seq = a_lp.copy()
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print(f"[phase 3] after LP fast_score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Light simulated‑annealing polishing (uses the exact max objective)
    time_left = deadline - time.time()
    if time_left > 1.0:
        print(f"[phase 4] simulated‑annealing polish – {time_left:.1f}s left")
        a_final, final_peak = _simulated_annealing(overall_best_seq, deadline)
        overall_best_seq = a_final.copy()
        overall_best_score = 2.0 * n * final_peak
        print(f"[phase 4] final SA fast_score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final exact evaluation (the only call to the hidden evaluate_sequence)
    try:
        true_score = evaluate_sequence(overall_best_seq.tolist())
        print(f"[final] evaluate_sequence returned {true_score:.6f}")
        if true_score < overall_best_score:
            overall_best_score = true_score
    except Exception as exc:  # pragma: no cover
        print("[final] evaluation error:", exc)

    print(f"Best sequence length: {len(overall_best_seq)}; approx. fast_score={overall_best_score:.6f}")
    return overall_best_seq.tolist()
