"""
propose_candidate
=================

Search for a low‑value step function for the ``evaluate_sequence`` black‑box.
The algorithm combines three ideas:

1. **Smooth max relaxation** – the non‑differentiable objective
   ``max_k (a*a)_k`` is approximated by a temperature‑scaled log‑sum‑exp.
   For a fixed temperature τ the objective is smooth and its gradient can
   be written in closed form using FFT‑based convolutions.  An Adam optimiser
   follows that gradient while τ is slowly decreased (cooling schedule).
   This gives a more stable descent than directly using a sub‑gradient of
   the hard max.

2. **Multi‑scale warm‑starts** – we optimise a sequence of increasing
   lengths (e.g. 4000 → 8000 → 16000 → 20000).  The best solution for a
   smaller length is interpolated to initialise the next larger problem,
   allowing the optimiser to inherit good structure.

3. **Final projected sub‑gradient refinement** – after the smooth phase
   τ is essentially zero, so we fall back to a classic projected sub‑gradient
   descent (as in the reference solution) to polish the final answer.

All arithmetic stays on the probability simplex (non‑negative, sum = 1) by
working with a soft‑max parametrisation ``x`` (logits).  The code respects
the 1000 s time budget, prints intermediate scores and finally returns the
best sequence as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of ``old_seq`` (already normalised) onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


# ----------------------------------------------------------------------
def _smooth_adam_optimize(
    x0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 25000,
    lr0: float = 0.18,
    tau0_factor: float = 0.8,
    tau_decay: float = 0.96,
    tau_min: float = 1e-9,
    epoch_len: int = 1500,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth approximation of the max‑convolution.
    Returns the best sequence (on the simplex) found and its raw score
    ``2·n·max(a*a)``.
    """
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_score = math.inf

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        # ------- soft‑max parametrisation -------
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)          # stable
        a = exp_x / exp_x.sum()            # Σa = 1, a ≥ 0

        # ------- convolution and current raw score -------
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it}  raw score={best_score:.6f}")

        # ------- temperature (τ) schedule -------
        # τ decreases stepwise every ``epoch_len`` iterations
        epoch = (it - 1) // epoch_len
        tau = max(tau_min, tau0_factor * cur_max * (tau_decay ** epoch))

        # ------- smooth max weights -------
        scaled = conv / tau
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)   # avoid overflow/underflow
        w = exp_scaled / exp_scaled.sum()          # sums to 1

        # ------- gradient of the smooth objective -------
        grad_a = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # ------- chain rule through soft‑max -------
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape flat plateaus
        if it % 2000 == 0:
            x += np.random.randn(*x.shape) * 0.02

    # fallback (should never happen)
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_full(best_a).max()

    return best_a, best_score


# ----------------------------------------------------------------------
def _final_subgradient_refine(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent (as in the reference solution)
    to polish the result after the smooth phase.
    Returns the refined sequence and its raw score.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = 2.0 * n * _fft_self_convolve_full(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.07 / math.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a)
        # occasional tiny random perturbation
        if it % 800 == 0 and np.random.rand() < 0.06:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a)
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] post‑refine it={it}  raw score={best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Compute a sub‑gradient of the function ``max_k (a*a)_k``.
    Returns (grad, cur_max) where ``grad`` has the same shape as ``a``.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    # active set (tolerance scales with current maximum)
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(conv >= cur_max - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    return grad, cur_max

        else:
            warm_start = np.full(n, 1.0 / n, dtype=np.float64)

        # Build a small set of initial logits
        init_logits = []

        # 1) warm‑start (log‑odds)
        eps = 1e-12
        init_logits.append(np.log(warm_start + eps))

        # 2) uniform
        init_logits.append(np.zeros(n, dtype=np.float64))

        # 3) few random Dirichlet draws
        for _ in range(3):
            alpha = np.random.rand() * 3.0 + 0.3   # α ∈ (0.3,3.3)
            sample = np.random.gamma(alpha, 1.0, size=n)
            sample /= sample.sum()
            init_logits.append(np.log(sample + eps))

        # run Adam​ with smooth max from each start, keep best
        local_best_seq = None
        local_best_score = math.inf

        for idx, logits in enumerate(init_logits):
            if time.time() >= deadline:
                break
            print(f"[{time.time():.1f}] length {n}, start {idx + 1}/{len(init_logits)}")
            seq, score = _smooth_adam_optimize(
                logits,
                n,
                deadline,
                max_iters=20000,
                lr0=0.18,
                tau0_factor=0.8,
                tau_decay=0.96,
                tau_min=1e-12,
                epoch_len=1500,
            )
            if score < local_best_score:
                local_best_score = score
                local_best_seq = seq.copy()

        # safety fallback
        if local_best_seq is None:
            local_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
            local_best_score = 2.0 * n * _fft_self_convolve_full(local_best_seq).max()

        print(f"  finished length {n}: best raw score = {local_best_score:.6f}")

        # keep globally best
        if local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_seq.copy()

        # the best sequence becomes the warm‑start for the next length
        # (interpolation happens automatically at the start of the loop)

    # ------------------------------------------------------------------
    # final polishing with classic projected sub‑gradient descent
    if best_seq is not None:
        print("\n--- Final projected sub‑gradient refinement ---")
        refine_deadline = deadline
        best_seq, best_score = _final_subgradient_refine(best_seq, refine_deadline)

    # ------------------------------------------------------------------
    # final evaluation by the provided black‑box
    final_list = best_seq.tolist()
    final_score = evaluate_sequence(final_list)
    print(f"\nFinal evaluated score: {final_score:.6f} (length={len(final_list)})")
    return final_list
