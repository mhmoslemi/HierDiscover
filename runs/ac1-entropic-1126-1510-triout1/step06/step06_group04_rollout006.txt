"""
Propose a low‑score step function.

The algorithm consists of two phases:

1️⃣ **Broad stochastic search** – for several candidate lengths we generate
   many distributions (uniform, arcsine, symmetric / asymmetric Beta,
   Dirichlet, and mixtures thereof).  All candidates are projected onto the
   simplex Σ a_i = 1.  The cheap proxy 2·n·max(a*a) is used to keep the best
   sequence found.

2️⃣ **Smooth gradient refinement** – starting from the best candidate we
   minimise the true objective 2·n·max(conv(a,a)) by approximating the max
   with a soft‑max (log‑sum‑exp).  This yields a smooth gradient that can be
   handled with projected (momentum) gradient descent.  The temperature of
   the soft‑max is increased gradually so the method first follows a smooth
   landscape and later focuses on the true maximum.

The whole routine respects the 1000 s budget, prints progress and finally
returns the sequence (as a plain list) that gives the lowest evaluated
score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution of length n, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf (stable)
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # avoid overflow
    logpdf = np.minimum(logpdf, 700.0)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """Proxy score assuming Σa = 1 :  2·n·max(a*a)."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of the smoothed objective
        f(a) = 2·n·softmax_τ(conv(a,a))
    where softmax_τ(x) = (1/τ)·log Σ exp(τ·x).
    The returned gradient has the same shape as `a`.
    """
    n = a.shape[0]
    # full self‑convolution
    b = _self_convolution_full(a)
    maxb = b.max()
    # softmax weights (stable)
    w = np.exp(tau * (b - maxb))
    w /= w.sum()
    # gradient of the soft‑max part:
    #   ∂f/∂a = 2·n· 2·conv(a_rev, w) = 4·n·conv(a_rev, w)
    a_rev = a[::-1]
    conv = _fft_convolve(a_rev, w)[n - 1: n - 1 + n]
    return 4.0 * n * conv

    # ------------------------------------------------------------------
    # candidate lengths – a mix of powers of two and a few intermediate values
    lengths = [1024, 2048, 4096, 8192, 12000]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Exploring length {n}")

        # ----- deterministic baselines -----------------------------------
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        sc = _raw_score(uniform)
        if sc < best_score:
            best_score, best_seq, best_len = sc, uniform.copy(), n
            print(f"   uniform best={best_score:.6f}")

        arcsine = _beta_distribution(n, 0.5, 0.5)
        sc = _raw_score(arcsine)
        if sc < best_score:
            best_score, best_seq, best_len = sc, arcsine.copy(), n
            print(f"   arcsine best={best_score:.6f}")

        # sinusoidal prototype (just for curiosity)
        sin_sq = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n) ** 2
        sin_sq /= sin_sq.sum()
        sc = _raw_score(sin_sq)
        if sc < best_score:
            best_score, best_seq, best_len = sc, sin_sq.copy(), n
            print(f"   sin² best={best_score:.6f}")

        if base_seq is not None:
            cand = _interpolate_to_length(base_seq, n)
            sc = _raw_score(cand)
            if sc < best_score:
                best_score, best_seq, best_len = sc, cand.copy(), n
                print(f"   from base best={best_score:.6f}")

        # ----- symmetric Beta grid (including very light tails) ----------
        alphas = np.concatenate((
            np.linspace(0.02, 0.12, 8, endpoint=False),
            np.linspace(0.12, 0.6, 12, endpoint=False),
            np.linspace(0.6, 2.0, 15),
            np.linspace(2.0, 5.0, 8),
        ))
        for a in alphas:
            if time.time() >= deadline:
                break
            vec = _beta_distribution(n, a, a)
            sc = _raw_score(vec)
            if sc < best_score:
                best_score, best_seq, best_len = sc, vec.copy(), n
                print(f"   symBeta a={a:.3f} best={best_score:.6f}")

        # ----- random (possibly extreme) Beta draws -----------------------
        for _ in range(800):
            if time.time() >= deadline:
                break
            alpha = 10 ** rng.uniform(-3, 1)   # 0.001 … 10
            beta = 10 ** rng.uniform(-3, 1)
            vec = _beta_distribution(n, alpha, beta)
            sc = _raw_score(vec)
            if sc < best_score:
                best_score, best_seq, best_len = sc, vec.copy(), n
                print(f"   randBeta a={alpha:.4g} b={beta:.4g} best={best_score:.6f}")

        # ----- Dirichlet draws (including very spiky) --------------------
        for _ in range(800):
            if time.time() >= deadline:
                break
            conc = 10 ** rng.uniform(-3, 1)   # 0.001 … 10
            vec = rng.gamma(conc, 1.0, size=n)
            vec /= vec.sum()
            sc = _raw_score(vec)
            if sc < best_score:
                best_score, best_seq, best_len = sc, vec.copy(), n
                print(f"   dirichlet conc={conc:.4g} best={best_score:.6f}")

        # ----- mixture of two Betas ---------------------------------------
        for _ in range(500):
            if time.time() >= deadline:
                break
            a1, b1 = 10 ** rng.uniform(-3, 1), 10 ** rng.uniform(-3, 1)
            a2, b2 = 10 ** rng.uniform(-3, 1), 10 ** rng.uniform(-3, 1)
            lam = rng.random()
            p1 = _beta_distribution(n, a1, b1)
            p2 = _beta_distribution(n, a2, b2)
            vec = lam * p1 + (1.0 - lam) * p2
            vec = _project_onto_simplex(vec)
            sc = _raw_score(vec)
            if sc < best_score:
                best_score, best_seq, best_len = sc, vec.copy(), n
                print(f"   mix2 λ={lam:.3f} best={best_score:.6f}")

        # ----- mixture of three Betas ------------------------------------
        for _ in range(300):
            if time.time() >= deadline:
                break
            a1, b1 = 10 ** rng.uniform(-3, 1), 10 ** rng.uniform(-3, 1)
            a2, b2 = 10 ** rng.uniform(-3, 1), 10 ** rng.uniform(-3, 1)
            a3, b3 = 10 ** rng.uniform(-3, 1), 10 ** rng.uniform(-3, 1)
            weights = rng.dirichlet([1.0, 1.0, 1.0])
            p1 = _beta_distribution(n, a1, b1)
            p2 = _beta_distribution(n, a2, b2)
            p3 = _beta_distribution(n, a3, b3)
            vec = weights[0] * p1 + weights[1] * p2 + weights[2] * p3
            vec = _project_onto_simplex(vec)
            sc = _raw_score(vec)
            if sc < best_score:
                best_score, best_seq, best_len = sc, vec.copy(), n
                print(f"   mix3 best={best_score:.6f}")

        # ----- uniform + Beta mixture ------------------------------------
        for _ in range(300):
            if time.time() >= deadline:
                break
            alpha = 10 ** rng.uniform(-3, 1)
            beta = 10 ** rng.uniform(-3, 1)
            beta_vec = _beta_distribution(n, alpha, beta)
            lam = rng.random()
            vec = lam * beta_vec + (1.0 - lam) * uniform
            vec = _project_onto_simplex(vec)
            sc = _raw_score(vec)
            if sc < best_score:
                best_score, best_seq, best_len = sc, vec.copy(), n
                print(f"   mixU λ={lam:.3f} best={best_score:.6f}")

    # ------------------------------------------------------------------
    if best_seq is None:
        # Ultra‑fallback (should never trigger)
        n = 2048
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_len = n
        best_score = _raw_score(best_seq)
        print("Fallback uniform used.")

    print(f"Best raw score after exploration: {best_score:.6f} (len={best_len})")

    # ------------------------------------------------------------------
    # 2) Local refinement with smoothed‑max gradient descent
    a = best_seq.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = best_score

    momentum = np.zeros_like(a)
    mu = 0.9                     # momentum coefficient
    it = 0
    # soft‑max temperature schedule
    tau = 0.5
    tau_inc = 0.001
    tau_max = 30.0

    while time.time() < deadline:
        it += 1
        grad = _smooth_gradient(a, tau)          # gradient of the true objective
        step = 0.25 / math.sqrt(it)               # diminishing step size
        momentum = mu * momentum + (1.0 - mu) * grad
        a = a - step * momentum
        a = _project_onto_simplex(a)

        # occasional tiny jitter to escape flat plateaus
        if it % 1000 == 0 and rng.random() < 0.05:
            a += rng.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        cur_val = _raw_score(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] refine it={it} τ={tau:.2f} score={best_val:.6f}")

        # increase the temperature slowly – we end up with almost the true max
        if tau < tau_max:
            tau = min(tau_max, tau + tau_inc)

    # ------------------------------------------------------------------
    final_seq = best_a
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Final evaluated score: {final_score:.6f} (len={len(final_seq)})")
    return final_seq.tolist()
