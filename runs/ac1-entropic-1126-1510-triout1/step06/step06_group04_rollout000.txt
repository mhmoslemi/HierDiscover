   – mixtures of two / three Betas,
   – uniform+Beta mixtures.
   Every candidate is projected onto the probability simplex (∑a = 1)
   and evaluated by the fast proxy  2·n·max(a*a).  The best candidate
   (shape only, sum = 1) is kept.

2. **Smooth optimisation** of the best candidate.
   We minimise a smooth approximation of the max‑convolution
   ``ϕτ(a) = (1/τ)·log∑exp(τ·(a*a))``.
   Its gradient is
        ∇ϕτ(a) = 2·(a_rev  *  w)   (restricted to n entries)
   where ``w_k = exp(τ·(a*a)_k) / ∑exp(τ·(a*a))``.
   The optimisation uses Adam with a learning‑rate schedule and
   projects after each step onto the simplex.
   τ is increased slowly from 0.5 to 8.0 so that the early steps are
   very smooth and later steps focus on the true max.

3. The best sequence found during the refinement (according to the
   exact proxy) is finally evaluated with ``evaluate_sequence`` and
   returned.

The whole routine respects a user‑provided time budget (default 1000 s)
and never exceeds it.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on n bins, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf for numerical stability
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logpdf = np.minimum(logpdf, 700.0)               # avoid overflow in exp
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf

# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals

# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

# ----------------------------------------------------------------------
def _raw_score(a: np.ndarray) -> float:
    """Proxy assuming Σa = 1 :  2·n·max(a*a)."""
    n = a.shape[0]
    return 2.0 * n * _self_convolution_full(a).max()

# ----------------------------------------------------------------------
def _smooth_objective_and_grad(a: np.ndarray, tau: float):
    """
    Returns
        obj  – smooth approximation of max(a*a),
        grad – gradient of obj,
        conv – the exact self‑convolution (needed for raw score).
    """
    n = a.shape[0]
    conv = _self_convolution_full(a)                      # length 2n‑1
    # log‑sum‑exp (stable)
    scaled = tau * conv
    m = scaled.max()
    exp_shifted = np.exp(scaled - m)
    sum_exp = exp_shifted.sum()
    obj = (m + np.log(sum_exp)) / tau                     # (1/τ)·log∑exp(τ·conv)
    w = exp_shifted / sum_exp                             # softmax weights, sum to 1
    # gradient = 2 * (a_rev * w)[n-1 : n-1+n]
    a_rev = a[::-1]
    grad_full = _fft_convolve(a_rev, w)                  # length 3n‑2
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return obj, grad, conv

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step function with a low evaluate_sequence score.
    The routine respects the provided time budget (default 1000 s).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered sequence if available.
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_seq = base_seq / base_seq.sum()
    except Exception:
        base_seq = None

    best_seq = None
    best_score = float("inf")
    best_len = None

    # ------------------------------------------------------------------
    # Lengths to explore – a compromise between expressiveness and speed.
    lengths = [2000, 4000, 8000, 12000]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Exploring length {n}")

        # ----- deterministic baselines ---------------------------------
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        sc = _raw_score(uniform)
        if sc < best_score:
            best_score, best_seq, best_len = sc, uniform.copy(), n
            print(f"   uniform score={best_score:.6f}")

        arcsine = _beta_distribution(n, 0.5, 0.5)
        sc = _raw_score(arcsine)
        if sc < best_score:
            best_score, best_seq, best_len = sc, arcsine.copy(), n
            print(f"   arcsine score={best_score:.6f}")

        if base_seq is not None:
            cand = _interpolate_to_length(base_seq, n)
            sc = _raw_score(cand)
            if sc < best_score:
                best_score, best_seq, best_len = sc, cand.copy(), n
                print(f"   from base score={best_score:.6f}")

        # ----- symmetric Beta grid ---------------------------------------
        alphas = np.concatenate((
            np.linspace(0.01, 0.12, 12, endpoint=False),
            np.linspace(0.12, 0.6, 12, endpoint=False),
            np.linspace(0.6, 2.0, 15),
            np.linspace(2.0, 5.0, 10)
        ))
        for a in alphas:
            if time.time() >= deadline:
                break
            vec = _beta_distribution(n, a, a)
            sc = _raw_score(vec)
            if sc < best_score:
                best_score, best_seq, best_len = sc, vec.copy(), n
                # comment out heavy printing if too noisy
                print(f"   symBeta a={a:.3f} score={best_score:.6f}")

        # ----- random Beta draws -----------------------------------------
        for _ in range(400):
            if time.time() >= deadline:
                break
            alpha = 10 ** rng.uniform(-3, 1)   # 0.001 … 10
            beta = 10 ** rng.uniform(-3, 1)
            vec = _beta_distribution(n, alpha, beta)
            sc = _raw_score(vec)
            if sc < best_score:
                best_score, best_seq, best_len = sc, vec.copy(), n
                print(f"   randBeta a={alpha:.3g} b={beta:.3g} score={best_score:.6f}")

        # ----- Dirichlet draws -------------------------------------------
        for _ in range(400):
            if time.time() >= deadline:
                break
            conc = 10 ** rng.uniform(-3, 1)   # 0.001 … 10
            vec = rng.gamma(conc, 1.0, size=n)
            vec /= vec.sum()
            sc = _raw_score(vec)
            if sc < best_score:
                best_score, best_seq, best_len = sc, vec.copy(), n
                print(f"   dirichlet conc={conc:.3g} score={best_score:.6f}")

        # ----- mixture of two Betas ---------------------------------------
        for _ in range(250):
            if time.time() >= deadline:
                break
            a1, b1 = 10 ** rng.uniform(-3, 1), 10 ** rng.uniform(-3, 1)
            a2, b2 = 10 ** rng.uniform(-3, 1), 10 ** rng.uniform(-3, 1)
            lam = rng.random()
            p1 = _beta_distribution(n, a1, b1)
            p2 = _beta_distribution(n, a2, b2)
            vec = lam * p1 + (1.0 - lam) * p2
            vec = _project_onto_simplex(vec)
            sc = _raw_score(vec)
            if sc < best_score:
                best_score, best_seq, best_len = sc, vec.copy(), n
                print(f"   mix2 λ={lam:.3f} score={best_score:.6f}")

        # ----- mixture of three Betas -------------------------------------
        for _ in range(150):
            if time.time() >= deadline:
                break
            a1, b1 = 10 ** rng.uniform(-3, 1), 10 ** rng.uniform(-3, 1)
            a2, b2 = 10 ** rng.uniform(-3, 1), 10 ** rng.uniform(-3, 1)
            a3, b3 = 10 ** rng.uniform(-3, 1), 10 ** rng.uniform(-3, 1)
            weights = rng.dirichlet([1, 1, 1])
            p1 = _beta_distribution(n, a1, b1)
            p2 = _beta_distribution(n, a2, b2)
            p3 = _beta_distribution(n, a3, b3)
            vec = weights[0] * p1 + weights[1] * p2 + weights[2] * p3
            vec = _project_onto_simplex(vec)
            sc = _raw_score(vec)
            if sc < best_score:
                best_score, best_seq, best_len = sc, vec.copy(), n
                print(f"   mix3 score={best_score:.6f}")

        # ----- uniform + Beta mixture ------------------------------------
        for _ in range(200):
            if time.time() >= deadline:
                break
            alpha = 10 ** rng.uniform(-3, 1)
            beta = 10 ** rng.uniform(-3, 1)
            beta_vec = _beta_distribution(n, alpha, beta)
            lam = rng.random()
            vec = lam * beta_vec + (1.0 - lam) * uniform
            vec = _project_onto_simplex(vec)
            sc = _raw_score(vec)
            if sc < best_score:
                best_score, best_seq, best_len = sc, vec.copy(), n
                print(f"   mixU λ={lam:.3f} score={best_score:.6f}")

    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – should never be reached
        n = 4000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_len = n
        best_score = _raw_score(best_seq)
        print("Fallback uniform used.")

    print(f"Best raw score after exploration: {best_score:.6f} (len={best_len})")

    # ------------------------------------------------------------------
    # 2) Smooth optimisation (Adam) starting from the best candidate.
    a = best_seq.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.12

    # τ schedule (smooth → sharp)
    tau_start = 0.5
    tau_end = 8.0
    ramp_iters = 8000  # after this τ stays at tau_end

    best_a = a.copy()
    best_raw = _raw_score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # τ grows linearly for the first *ramp_iters* iterations
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / ramp_iters)

        # smooth objective, gradient, and also obtain the exact convolution
        obj, grad, conv = _smooth_objective_and_grad(a, tau)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = base_lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_onto_simplex(a)

        # occasional tiny jitter to hop out of plateaus
        if it % 2000 == 0 and rng.random() < 0.05:
            a += rng.normal(scale=1e-7, size=n)
            a = _project_onto_simplex(a)

        cur_raw = 2.0 * n * conv.max()
        if cur_raw < best_raw - 1e-12:
            best_raw = cur_raw
            best_a = a.copy()
            print(f"[{time.time():.1f}] it={it}  raw_score={best_raw:.6f}  τ={tau:.2f}")

    # ------------------------------------------------------------------
    final_seq = best_a
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Final evaluated score: {final_score:.6f} (len={len(final_seq)})")
    return final_seq.tolist()
