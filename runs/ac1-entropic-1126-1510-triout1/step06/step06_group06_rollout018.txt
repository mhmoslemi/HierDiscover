def _beta_family(n, alpha_grid, beta_grid, deadline):
    best_seq = None
    best_score = inf
    i = np.arange(n, dtype=np.float64)
    term1 = i + 0.5
    term2 = n - i - 0.5
    # precompute logs for speed
    log_term1 = np.log(term1)
    log_term2 = np.log(term2)
    for a in alpha_grid:
        for b in beta_grid:
            if time.time() > deadline:
                return best_seq, best_score
            # compute log raw
            raw_log = (a - 1) * log_term1 + (b - 1) * log_term2
            # avoid overflow
            raw = np.exp(raw_log)
            seq = raw / raw.sum()
            peak = _fft_self_convolve(seq).max()
            score = 2.0 * n * peak
            if score < best_score:

"""
A fast‐search + fine‑tuning pipeline for the step‑function challenge.

* Phase 0 – enumerate a wide range of parametric families:
  – uniform, arcsine, exponential, random.
  – a two‑parameter Beta family   a_i ∝ (i+½)^{α‑1}·(n‑i‑½)^{β‑1} .
    (α,β>0, covering symmetric & skewed shapes).

  The best vector (lowest fast score = 2·n·max(convolution)) is kept.

* Phase 1 – cheap two‑level block search (front‑block/high, back‑block/low).

* Phase 2 – Adam optimisation of a smooth‑max surrogate of the ℓ∞
  convolution objective (continuous‐time continuation on the smoothness
  parameter α).

* Phase 3 – exact sub‑gradient descent on the true max‑convolution,
  followed by a greedy directed local‑search that moves a small amount
  of mass from a “source” contributing heavily to the current peak to a
  random “destination”.

All updates are projected onto the simplex Σa_i = 1, keeping the shape
scale‑invariant (the evaluation function 2·n·max(conv)/sum² is homogeneous).
The routine respects a 1000 s budget and prints progress.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.02,
                   alpha_max: float = 20000.0,
                   lr0: float = 0.5,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best probability vector seen and its exact peak.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule (slow decay)
        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        # continuation on the smoothness parameter
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny jitter to escape plateaus
        if t % 500 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 3e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    One sub‑gradient step for the exact ℓ∞ objective.
    Uses an equal weight on every convolution index attaining the current max.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Sub‑gradient descent on the exact max‑convolution objective.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.018 / np.sqrt(it)        # slightly smaller step for stability
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        if it % 600 == 0 and np.random.rand() < 0.09:
            a += np.random.randn(n) * 4e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


# ----------------------------------------------------------------------
def _directed_local_search(a0: np.ndarray, deadline: float):
    """
    Greedy mass‑transfer local search that always tries to reduce the current peak.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        # pick a convolution index close to the peak
        max_idx = np.random.choice(np.where(b >= cur_max - 1e-12)[0])

        # indices i that contribute to (a*a)[max_idx]
        i_low = max(0, max_idx - (n - 1))
        i_high = min(max_idx, n - 1)
        i_vals = np.arange(i_low, i_high + 1, dtype=int)
        prod = a[i_vals] * a[max_idx - i_vals]
        if prod.sum() == 0.0:
            src = np.random.randint(n)
        else:
            prob = prod / prod.sum()
            src = np.random.choice(i_vals, p=prob)

        dst = np.random.randint(n)
        if dst == src:
            continue

        max_delta = min(a[src], 0.06)          # move up to 6 % of the source
        delta = np.random.rand() * max_delta

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        b_new = _fft_self_convolve(a_new)
        new_max = b_new.max()

        if new_max < cur_max - 1e-15:
            a = a_new
            b = b_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()

        # very occasional random jitter
        if np.random.rand() < 0.004:
            a += np.random.randn(n) * 9e-8
            a = _project_onto_simplex(a, s=1.0)
            b = _fft_self_convolve(a)
            cur_max = b.max()

    return best_a


# ----------------------------------------------------------------------
def _two_level_family(n: int, deadline: float):
    """
    Search a simple two‑level family:
        a_i = p   for i < L   (front block)
        a_i = q   otherwise   (back block)
    with p = r·q, r≥1.
    Returns best vector and its fast score.
    """
    best_seq = None
    best_score = float('inf')
    step_L = max(1, n // 120)               # ≈0.8 % steps
    r_grid = np.logspace(0.0, 2.2, num=28)   # ratios 1 … ≈160

    for L in range(step_L, n, step_L):
        if time.time() > deadline:
            break
        denom_base = n - L
        for r in r_grid:
            denom = L * r + denom_base
            q = 1.0 / denom
            p = r * q
            seq = np.empty(n, dtype=np.float64)
            seq[:L] = p
            seq[L:] = q
            peak = _fft_self_convolve(seq).max()
            score = 2.0 * n * peak
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
    return best_seq, best_score


# ----------------------------------------------------------------------
def _beta_family(n: int, alpha_grid, beta_grid, deadline: float):
    """
    Exhaustive search over the 2‑parameter Beta‑like family:
        a_i ∝ (i+½)^{α‑1}·(n‑i‑½)^{β‑1},   α,β>0.
    Returns the best normalized vector and its fast score.
    """
    best_seq = None
    best_score = float('inf')

    i = np.arange(n, dtype=np.float64)
    x = i + 0.5
    y = n - i - 0.5
    log_x = np.log(x)
    log_y = np.log(y)

    for a in alpha_grid:
        for b in beta_grid:
            if time.time() > deadline:
                return best_seq, best_score
            raw_log = (a - 1.0) * log_x + (b - 1.0) * log_y
            raw = np.exp(raw_log)
            seq = raw / raw.sum()
            peak = _fft_self_convolve(seq).max()
            score = 2.0 * n * peak
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
    return best_seq, best_score
