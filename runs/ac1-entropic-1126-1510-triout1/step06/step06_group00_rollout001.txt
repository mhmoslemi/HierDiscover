def _projected_gradient_descent(a0, deadline, step0=0.2):
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline:
        t += 1
        grad, cur_max, active = _subgrad(a)
        # Subgradient for max of quadratic might be non-unique; we have one.
        step = step0 / np.sqrt(t)
        a = a - step * grad
        # projection onto simplex
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score
```

"""
Propose a low‑score non‑negative step function.

Algorithm
---------

1. **Broad sampling** – generate many candidate sequences of several
   lengths using analytic families (arcsine, symmetric beta, triangular,
   exponential) and random Dirichlet draws.  All candidates are normalised
   to ℓ₁‑norm 1 and scored with the cheap fast‑FFT objective
   ``score = 2·n·max(a*a)`` (identical to ``evaluate_sequence`` for sum = 1).

2. **Refinement** – keep the best few candidates and run three independent
   optimisation schemes on each of them:
      * exponentiated‑gradient (mirror descent w.r.t. KL)
      * projected‑gradient descent on the simplex using the exact
        sub‑gradient of the max‑convolution,
      * a second pass of exponentiated‑gradient with slightly different
        hyper‑parameters.
   The best sequence among all runs is kept.

3. **Targeted transfer / simulated‑annealing** – starting from the best
   sequence, repeatedly move a small amount of mass from an index with
   large sub‑gradient (i.e. that contributes heavily to the current
   maximal convolution entry) to an index with small sub‑gradient.
   The move is accepted only if the objective improves; otherwise it is
   discarded.  The transferred amount decays geometrically and an
   occasional tiny Gaussian jitter helps to escape plateaus.

The routine respects the 1000 s budget and always returns a list of
non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Fast surrogate: 2·n·max(a*a) assuming Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (the self‑pair contributes only once)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 20000,
                          eta0: float = 0.5, decay: float = 0.9999):
    """Mirror descent with KL‑divergence (exponentiated‑gradient)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _projected_gradient_descent(a0: np.ndarray, deadline: float,
                               step0: float = 0.3):
    """Projected (Euclidean) sub‑gradient descent on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _, _ = _subgrad(a)
        step = step0 / np.sqrt(t)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a point on the simplex (Gamma‑Dirichlet, α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _targeted_transfer(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer guided by the sub‑gradient.
    Move a small amount from an index with large positive gradient
    (contributing heavily to the current maximal convolution entry)
    to an index with small (or negative) gradient.
    The transferred mass decays geometrically; occasional tiny jitter
    is added to escape plateaus.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    cur_score = best_score
    n = a.shape[0]
    iteration = 0
    # maximum fraction of total mass that may be moved in a single trial
    max_transfer = 0.04

    while time.time() < deadline:
        iteration += 1
        # compute gradient for the current maximal entry
        grad, _, _ = _subgrad(a)

        i = int(np.argmax(grad))        # donor (large gradient)
        j = int(np.argmin(grad))        # receiver (small gradient)
        if i == j:
            # degenerate case – fall back to random pair
            i, j = np.random.choice(n, size=2, replace=False)

        # amount to move, respecting positivity and a geometric decay
        delta_cap = min(a[i], max_transfer)          # never exceed current mass
        if delta_cap <= 0:
            continue
        delta = delta_cap * (0.5 ** (iteration / 18000.0))

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # re‑project onto the simplex to correct tiny numerical drift
        a_new = _project_onto_simplex(a_new, s=1.0)

        new_score = _max_convolution_score(a_new)
        if new_score < cur_score - 1e-14:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] targeted transfer iter={iteration} "
                      f"new best {best_score:.6f}")

        # every few thousand iterations try a tiny Gaussian jitter
        if iteration % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a + jitter, s=1.0)
            cur_score = _max_convolution_score(a)

    return best_a, best_score
