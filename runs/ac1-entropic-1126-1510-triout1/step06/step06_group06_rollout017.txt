def evaluate_sequence(sequence):

"""
Propose a low‑score step‑function.

The search consists of three stages:

1) Broad sampling of many parametric families (arcsine, Beta, a
   distance‑based power‑law shape, random Dirichlet and a simple two‑level
   family).  For each length we keep the best few candidates according to
   the exact objective 2·n·max(a∗a).

2) Adam optimisation of the smooth‑max surrogate of the ℓ∞‑convolution
   starting from the top candidates.  Only a modest amount of time is spent
   on each candidate so that several restarts can be tried.

3) Exact sub‑gradient descent combined with a greedy mass‑transfer local
   search on the overall best candidate for the rest of the budget.  The
   final vector is returned as a plain list of non‑negative numbers.

All helper functions are written to be side‑effect free and the whole
routine respects the supplied time budget (default 1000 s).
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())   # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Exact value of the evaluation function for a sequence with sum=1."""
    n = seq.shape[0]
    peak = _fft_self_convolve(seq).max()
    return 2.0 * n * peak

# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # soft‑max weights (numerically stable)
    exp_term = np.exp(alpha * (b - cur_max))
    w_sum = exp_term.sum()
    w = exp_term / w_sum
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.01,
                   alpha_max: float = 5000.0,
                   lr0: float = 0.3,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best probability vector encountered.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size schedule (slow decay)
        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # keep the best exact peak seen so far
        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        # continuation on the smoothness parameter
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional jitter to escape shallow plateaus
        if t % 400 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    One sub‑gradient step for the exact ℓ∞ objective.
    Uses an equal weight on all indices attaining the current maximum.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Sub‑gradient descent on the exact max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.02 / np.sqrt(it)  # diminishing step
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        # occasional jitter
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


def _directed_local_search(a0: np.ndarray, deadline: float):
    """Greedy mass‑transfer local search focusing on the active peak."""
    a = a0.copy()
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        # pick a convolution index that is (almost) maximal
        max_idx = np.random.choice(np.where(b >= cur_max - 1e-12)[0])
        # indices i that contribute to (a*a)[max_idx]
        i_low = max(0, max_idx - (n - 1))
        i_high = min(max_idx, n - 1)
        i_vals = np.arange(i_low, i_high + 1, dtype=int)
        prod = a[i_vals] * a[max_idx - i_vals]
        if prod.sum() == 0.0:
            src = np.random.randint(n)
        else:
            prob = prod / prod.sum()
            src = np.random.choice(i_vals, p=prob)

        dst = np.random.randint(n)
        if dst == src:
            continue

        max_delta = min(a[src], 0.05)          # move at most 5 % of source mass
        delta = np.random.rand() * max_delta

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        b_new = _fft_self_convolve(a_new)
        new_max = b_new.max()

        if new_max < cur_max - 1e-15:         # accept only genuine improvements
            a = a_new
            b = b_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()

        # tiny random jitter now and then
        if np.random.rand() < 0.005:
            a += np.random.randn(n) * 1e-7
            a = _project_onto_simplex(a, s=1.0)
            b = _fft_self_convolve(a)
            cur_max = b.max()

    return best_a


# ----------------------------------------------------------------------
def _two_level_family(n: int, deadline: float):
    """
    Search a simple two‑level family:
        a_i = p   for i < L   (front block)
        a_i = q   otherwise   (back block)
    with p = r·q, r≥1.
    Returns the best vector and its exact fast score.
    """
    best_seq = None
    best_score = float('inf')
    step_L = max(1, n // 100)          # about 1 % steps
    r_grid = np.logspace(0.0, 2.3, num=25)   # from 1 to ≈200

    for L in range(step_L, n, step_L):
        if time.time() > deadline:
            break
        denom_base = n - L
        for r in r_grid:
            denom = L * r + denom_base
            q = 1.0 / denom
            p = r * q
            seq = np.empty(n, dtype=np.float64)
            seq[:L] = p
            seq[L:] = q
            score = _fast_score(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
    return best_seq, best_score


# ----------------------------------------------------------------------
def _beta_seq(n: int, a: float):
    """Symmetric Beta(a,a) density on a discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_vals = (a - 1.0) * np.log(x) + (a - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    vals[~np.isfinite(vals)] = 0.0
    s = vals.sum()
    if s == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / s
    return vals


def _arcsine_seq(n: int):
    """Discrete analogue of the arcsine law."""
    seq = 1.0 / np.sqrt((np.arange(n) + 0.5) * (n - np.arange(n) - 0.5))
    return seq / seq.sum()


def _distance_seq(n: int, beta: float, eps: float):
    """Shape proportional to (distance_to_edge + eps)^{-beta}."""
    d = np.minimum(np.arange(n), np.arange(n - 1, -1, -1)).astype(np.float64)
    eps_val = eps if eps > 0 else 1e-12
    seq = 1.0 / ((d + eps_val) ** beta)
    return seq / seq.sum()

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())   # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Exact value of the evaluation function for a sequence with sum=1."""
    n = seq.shape[0]
    peak = _fft_self_convolve(seq).max()
    return 2.0 * n * peak


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.01,
                   alpha_max: float = 5000.0,
                   lr0: float = 0.3,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best probability vector encountered.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size schedule (slow decay)
        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        if t % 400 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    One sub‑gradient step for the exact ℓ∞ objective.
    Uses an equal weight on all indices attaining the current maximum.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Sub‑gradient descent on the exact max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.02 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


def _directed_local_search(a0: np.ndarray, deadline: float):
    """Greedy mass‑transfer local search focusing on the active peak."""
    a = a0.copy()
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        max_idx = np.random.choice(np.where(b >= cur_max - 1e-12)[0])
        i_low = max(0, max_idx - (n - 1))
        i_high = min(max_idx, n - 1)
        i_vals = np.arange(i_low, i_high + 1, dtype=int)
        prod = a[i_vals] * a[max_idx - i_vals]
        if prod.sum() == 0.0:
            src = np.random.randint(n)
        else:
            prob = prod / prod.sum()
            src = np.random.choice(i_vals, p=prob)

        dst = np.random.randint(n)
        if dst == src:
            continue

        max_delta = min(a[src], 0.05)
        delta = np.random.rand() * max_delta

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        b_new = _fft_self_convolve(a_new)
        new_max = b_new.max()

        if new_max < cur_max - 1e-15:
            a = a_new
            b = b_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()

        if np.random.rand() < 0.005:
            a += np.random.randn(n) * 1e-7
            a = _project_onto_simplex(a, s=1.0)
            b = _fft_self_convolve(a)
            cur_max = b.max()

    return best_a


# ----------------------------------------------------------------------
def _two_level_family(n: int, deadline: float):
    """
    Search a simple two‑level family:
        a_i = p   for i < L   (front block)
        a_i = q   otherwise   (back block)
    with p = r·q, r≥1.
    Returns the best vector and its exact fast score.
    """
    best_seq = None
    best_score = float('inf')
    step_L = max(1, n // 100)          # about 1 % steps
    r_grid = np.logspace(0.0, 2.3, num=25)   # from 1 to ≈200

    for L in range(step_L, n, step_L):
        if time.time() > deadline:
            break
        denom_base = n - L
        for r in r_grid:
            denom = L * r + denom_base
            q = 1.0 / denom
            p = r * q
            seq = np.empty(n, dtype=np.float64)
            seq[:L] = p
            seq[L:] = q
            score = _fast_score(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
    return best_seq, best_score


def _beta_seq(n: int, a: float):
    """Symmetric Beta(a,a) density on a discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_vals = (a - 1.0) * np.log(x) + (a - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    vals[~np.isfinite(vals)] = 0.0
    s = vals.sum()
    if s == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / s
    return vals


def _arcsine_seq(n: int):
    """Discrete analogue of the arcsine law."""
    seq = 1.0 / np.sqrt((np.arange(n) + 0.5) * (n - np.arange(n) - 0.5))
    return seq / seq.sum()


def _distance_seq(n: int, beta: float, eps: float):
    """Shape ∝ (distance_to_edge + eps)^{-beta}."""
    d = np.minimum(np.arange(n), np.arange(n - 1, -1, -1)).astype(np.float64)
    eps_val = eps if eps > 0 else 1e-12
    seq = 1.0 / ((d + eps_val) ** beta)
    return seq / seq.sum()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """Main entry point – returns a list of non‑negative heights."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Load a previously discovered construction if it exists.
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_seq = _project_onto_simplex(base_seq, s=1.0)
        base_n = len(base_seq)
    except Exception:
        base_seq = None
        base_n = 4000

    # ------------------------------------------------------------------
    # Lengths to explore (around the previous best, capped at 8 000).
    cand_lengths = [max(500, int(base_n * f))
                    for f in (0.6, 0.8, 1.0, 1.2, 1.4, 1.6, 2.0)
                    if int(base_n * f) >= 500]
    cand_lengths = [min(l, 8000) for l in cand_lengths]
    cand_lengths = sorted(set(cand_lengths))

    # ------------------------------------------------------------------
    # Phase 0 – broad family sampling.
    print("[phase 0] broad family sampling")
    TOP_K = 12               # keep the best few seeds
    candidates = []          # list of (score, seq)

    def consider(seq: np.ndarray):
        if time.time() >= deadline:
            return
        seq = _project_onto_simplex(seq, s=1.0)
        sc = _fast_score(seq)
        candidates.append((sc, seq.copy()))
        candidates.sort(key=lambda x: x[0])
        if len(candidates) > TOP_K:
            candidates.pop()

    for n in cand_lengths:
        if time.time() >= deadline:
            break

        # simple families
        consider(_arcsine_seq(n))
        consider(np.full(n, 1.0 / n, dtype=np.float64))
        for _ in range(2):
            consider(np.random.rand(n))

        # symmetric Beta densities
        for a in np.linspace(0.05, 0.95, 19):
            consider(_beta_seq(n, a))

        # distance‑based power‑law shapes
        for beta in np.linspace(0.42, 0.58, 9):
            for eps in (0.0, 0.5, 1.0, 2.0, 4.0, 8.0):
                consider(_distance_seq(n, beta, eps))

        # two‑level family
        seq2, sc2 = _two_level_family(n, deadline)
        if seq2 is not None:
            consider(seq2)

        # interpolation of the previous best (if any)
        if base_seq is not None:
            x_old = np.linspace(0.0, 1.0, base_n)
            x_new = np.linspace(0.0, 1.0, n)
            interp = np.interp(x_new, x_old, base_seq)
            interp = np.maximum(0.0, interp)
            interp = interp / interp.sum()
            consider(interp)

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation from the top seeds.
    print(f"[phase 1] Adam optimisation on {len(candidates)} seeds")
    remaining = deadline - time.time()
    if remaining > 5.0:
        # reserve about 40 % of the remaining budget for Adam runs
        time_for_adam = remaining * 0.40
        per_candidate = max(5.0, time_for_adam / len(candidates))
        improved = []
        for score, seq in list(candidates):      # copy because we extend later
            if time.time() >= deadline:
                break
            end = time.time() + per_candidate
            opt_seq, _ = _adam_optimize(seq, end)
            opt_score = _fast_score(opt_seq)
            improved.append((opt_score, opt_seq))
        for sc, s in improved:
            candidates.append((sc, s))
        candidates.sort(key=lambda x: x[0])
        candidates = candidates[:TOP_K]

    # ------------------------------------------------------------------
    # Phase 2 – exact sub‑gradient + greedy local search on the best seed.
    print("[phase 2] final exact refinement")
    best_seq = candidates[0][1]
    best_score = candidates[0][0]

    while time.time() < deadline:
        time_left = deadline - time.time()
        sub_end = time.time() + time_left * 0.55
        refined = _refine_max(best_seq, sub_end)
        best_seq = _directed_local_search(refined, deadline)
        best_score = _fast_score(best_seq)

        # small random jitter before a possible next pass
        if time.time() + 0.5 < deadline:
            best_seq += np.random.randn(best_seq.shape[0]) * 1e-6
            best_seq = _project_onto_simplex(best_seq, s=1.0)
        else:
            break

    # ------------------------------------------------------------------
    # Final black‑box evaluation.
    try:
        true_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence score = {true_score:.6f}")
        if true_score < best_score:
            best_score = true_score
    except Exception as exc:
        print("[final] evaluate_sequence raised:", exc)

    print(f"Best score {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
