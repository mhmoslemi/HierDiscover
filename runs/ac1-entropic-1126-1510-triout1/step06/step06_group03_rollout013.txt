def _lp_refine(a, deadline, max_iter=3):
    import scipy.optimize as opt
    import scipy.sparse as sp
    n = len(a)
    a = a.copy()
    for it in range(max_iter):
        if time.time()>=deadline:
            break
        M = _fft_self_convolve_max(a)
        # build A_ub
        # Use banded Toeplitz: diags list
        diags = []
        offsets = []
        for offset in range(n):
            # diagonal with value a[offset], length n
            diags.append(np.full(n, a[offset]))
            offsets.append(offset)
        A_ub = sp.diags(diags, offsets, shape=(2*n-1, n), format='csr')
        b_ub = np.full(2*n-1, M)
        c = -np.ones(n)  # minimize -sum(g)
        bounds = [(0, None)] * n
        try:
            res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
            if not res.success:
                print(f"LP failed: {res.message}")
                break
            g = res.x
        except Exception as e:
            print(f"LP exception: {e}")
            break
        # Normalize g to sum 1 (if possible)
        s = g.sum()
        if s <= 0:
            g = np.full(n, 1.0/n)
        else:
            g = g / s
        # Mix with current a
        t = 0.5  # mixing factor; can decrease over iterations
        a = (1-t) * a + t * g
        # Project onto simplex to correct numeric drift
        a = _project_onto_simplex(a)
    return a

"""
Hybrid multi‑scale optimiser for the step‑function attack.

The routine works in three phases:

1.  Small‑scale search (n≈2000) using Adam on a soft‑max
    parametrisation together with a few random restarts.
    Optionally a tiny linear‑programming (LP) refinement
    (SciPy ‘highs’) is applied – the LP solves
    max ∑g_j  s.t.  (f*g)_k ≤ M   (f fixed, M = max f*f).

2.  The best small‑scale shape is interpolated to a sequence of larger
    length (several doublings).  Each size is optimised again with Adam;
    warm‑starting from the previous size keeps the number of iterations
    modest.

3.  A long projected sub‑gradient descent (on the max‑convolution) finishes
    the optimisation.  Random jitter occasionally helps to escape plateaus.

All steps respect the global time budget (default 1000 s).  The final
sequence is returned as a plain Python list, ready for the supplied
``evaluate_sequence`` black‑box.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# -------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of max_k (a*a)_k.
    Returns (grad, cur_max), where grad has the same shape as a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    # active set – tolerance scales with the current maximum
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(conv >= cur_max - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    # gradient of the max (still contains a factor 2)
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# -------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


# -------------------------------------------------------------
def _adam_optimize_from_logits(
    x0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 3000,
    lr0: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the soft‑max (log‑it) parametrisation.
    Returns best a (sum = 1) found and its raw score 2·n·max(a*a).
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    best_a = None
    best_score = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1

        # --- soft‑max (stable) ---
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()          # Σa = 1, a ≥ 0

        # --- objective (raw score) ---
        cur_max = _fft_self_convolve_max(a)
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it}  raw={best_score:.6f}")

        # --- sub‑gradient of the max ---
        grad_a, _ = _max_subgradient_step(a)

        # --- chain rule through the soft‑max ---
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape flat plateaus
        if it % 1000 == 0:
            x += np.random.randn(*x.shape) * 0.01

    # fallback (should never happen)
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_max(best_a)

    return best_a, best_score


# -------------------------------------------------------------
def _lp_refine(a: np.ndarray, deadline: float, max_iter: int = 3) -> np.ndarray:
    """
    Optional tiny linear‑programming refinement.
    Solve   max ∑g_j   s.t.   (a * g)_k ≤ M   (for all k)
    and mix the result back into a.
    Requires SciPy.
    """
    try:
        import scipy.optimize as opt
        import scipy.sparse as sp
    except Exception:          # SciPy not available – nothing to do
        return a

    n = len(a)
    a = a.copy()
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        M = _fft_self_convolve_max(a)

        # Build the (2n‑1)×n convolution matrix of a (lower‑triangular Toeplitz)
        diags = []
        offsets = []
        for offset in range(n):
            diags.append(np.full(n, a[offset]))
            offsets.append(offset)
        A_ub = sp.diags(diags, offsets, shape=(2 * n - 1, n), format="csr")
        b_ub = np.full(2 * n - 1, M)
        c = -np.ones(n)          # minimise –∑g_j  → maximise ∑g_j
        bounds = [(0, None)] * n

        try:
            res = opt.linprog(
                c,
                A_ub=A_ub,
                b_ub=b_ub,
                bounds=bounds,
                method="highs",
                options={"presolve": True},
            )
        except Exception as e:
            print(f"LP exception: {e}")
            break

        if not res.success:
            # LP failed – bail out
            print(f"LP failed (iter {it}): {res.message}")
            break

        g = res.x
        s = g.sum()
        if s > 0:
            g = g / s
        else:
            g = np.full(n, 1.0 / n)

        # convex mixture – mixing factor 0.5 is a safe choice
        t = 0.5
        a = (1 - t) * a + t * g
        a = _project_onto_simplex(a)      # small numerical drift correction

    return a


# -------------------------------------------------------------
def _subgradient_refine(a: np.ndarray, deadline: float) -> np.ndarray:
    """Long projected sub‑gradient descent on max‑convolution."""
    n = a.shape[0]
    best = a.copy()
    best_raw = 2.0 * n * _fft_self_convolve_max(a)
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.07 / math.sqrt(it)          # diminishing step size
        a = a - step * grad
        a = _project_onto_simplex(a)

        # occasional tiny random jitter
        if it % 800 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a)

        cur_raw = 2.0 * n * cur_max
        if cur_raw < best_raw - 1e-12:
            best_raw = cur_raw
            best = a.copy()
            print(f"[{time.time():.1f}] sub‑grad it={it} raw={best_raw:.6f}")

    return best

def _optimise_length(
    n: int,
    deadline: float,
    prev_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length n.
    Several start points (warm‑start + uniform + Dirichlet + sine) are tried.
    Returns the best (a, raw_score) pair.
    """
    init_logits = []

    # warm‑start from previous sequence (interpolated)
    if prev_seq is not None:
        cand = _interpolate_to_length(prev_seq, n)
        eps = 1e-12
        init_logits.append(np.log(cand + eps))

    # uniform start
    init_logits.append(np.zeros(n, dtype=np.float64))

    # a handful of random Dirichlet draws
    for _ in range(4):
        alpha = np.random.rand() * 4.0 + 0.1   # α ∈ (0.1,4.1)
        sample = np.random.gamma(alpha, 1.0, size=n)
        sample /= sample.sum()
        init_logits.append(np.log(sample + 1e-12))

    # sine‑shaped start (positive half‑wave)
    x = (np.arange(n, dtype=float) + 0.5) / n
    sine = np.sin(np.pi * x)
    sine = np.clip(sine, 0, None)
    if sine.sum() > 0:
        sine /= sine.sum()
        init_logits.append(np.log(sine + 1e-12))

    best_seq = None
    best_raw = float("inf")

    for ix, logit in enumerate(init_logits):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] length {n}, start {ix+1}/{len(init_logits)}")
        seq, raw = _adam_optimize_from_logits(
            logit, n, deadline, max_iters=3000, lr0=0.12
        )
        # optional LP refinement (cheap for the modest n we use here)
        if time.time() < deadline:
            seq = _lp_refine(seq, deadline, max_iter=2)
            raw = 2.0 * n * _fft_self_convolve_max(seq)

        if raw < best_raw:
            best_raw = raw
            best_seq = seq.copy()

    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_raw = 2.0 * n * _fft_self_convolve_max(best_seq)

    return best_seq, best_raw

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Entry point for the competition harness.
    Returns a list of non‑negative heights that (hopefully) achieves a low
    ``evaluate_sequence`` score.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin – the platform kills the process exactly at the deadline
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Try to reuse a previously discovered construction if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        s = prev.sum()
        if s > 0.0:
            prev = prev / s
        else:
            prev = None
    except Exception:
        prev = None

    # ------------------------------------------------------------------
    # First stage: modest size optimisation (n≈2000).  This finds a good
    # shape that can later be up‑sampled.
    small_n = 2000
    print(f"\n=== Small‑scale optimisation (n={small_n}) ===")
    seq_small, raw_small = _optimise_length(small_n, deadline, prev_seq=prev)
    print(f"Small scale raw = {raw_small:.6f}")

    # ------------------------------------------------------------------
    # Second stage: progressive up‑sampling to larger lengths.
    # The schedule doubles the size a few times; each stage receives a
    # warm start from the previous best sequence.
    lengths = [4000, 8000, 12000, 20000]   # final length around 20 k
    best_seq = seq_small
    best_raw = raw_small
    best_n = small_n

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Upsampling to length {n} ===")
        seq, raw = _optimise_length(n, deadline, prev_seq=best_seq)
        print(f"Length {n}: best raw = {raw:.6f}")
        if raw < best_raw:
            best_raw = raw
            best_seq = seq.copy()
            best_n = n

    # ------------------------------------------------------------------
    # Final fine‑tuning with a long projected sub‑gradient descent.
    if best_seq is not None:
        print(f"\n--- Final sub‑gradient refinement (n={best_n}) ---")
        best_seq = _subgradient_refine(best_seq, deadline)

    # ------------------------------------------------------------------
    # Convert to plain list and evaluate via the black‑box.
    final_list = best_seq.tolist()
    final_score = evaluate_sequence(final_list)
    print(f"\nFinal evaluated score: {final_score:.6f} (length={len(final_list)})")
    return final_list
