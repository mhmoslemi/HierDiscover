def _simulated_annealing_local_search(a0, deadline):
    a = a0.copy()
    n = len(a)
    cur_max = _fft_self_convolve(a).max()
    best_a = a.copy()
    best_max = cur_max
    start = time.time()
    T0 = 0.01  # initial temperature relative to cur_max
    while time.time() < deadline:
        elapsed = time.time() - start
        T = T0 * (1 - elapsed / (deadline - start))  # linear cooling
        # choose source: weighted by contribution to max (product)
        # compute active indices (b near max)
        b = _fft_self_convolve(a)
        cur_max = b.max()
        active_idx = np.where(b >= cur_max - 1e-12)[0]
        # Choose some convolution index k from active set uniformly
        k = np.random.choice(active_idx)
        # Choose i in [max(0, k-(n-1)), min(k, n-1)] weighted by product a_i a_{k-i}
        i_low = max(0, k - (n-1))
        i_high = min(k, n-1)
        i_vals = np.arange(i_low, i_high+1)
        probs = a[i_vals] * a[k - i_vals]
        prob_sum = probs.sum()
        if prob_sum > 0:
            i_vals = i_vals[np.random.choice(len(i_vals), p=probs/prob_sum)]
        else:
            i_vals = np.random.randint(n)
        src = i_vals
        dst = np.random.randint(n)
        while dst == src:
            dst = np.random.randint(n)
        # Determine delta
        max_delta = min(a[src], 0.1)
        delta = np.random.rand() * max_delta
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        new_max = _fft_self_convolve(a_new).max()
        # Accept if new_max lower, else with prob exp(-(new_max - cur_max) / T)
        if new_max < cur_max - 1e-15 or np.random.rand() < np.exp(-(new_max - cur_max) / T):
            a = a_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
        # occasional small jitter
        if np.random.rand() < 0.001:
            a += np.random.randn(n)*1e-7
            a = _project_onto_simplex(a)
            cur_max = _fft_self_convolve(a).max()
    return best_a
```

        if time.time() >= deadline:
            break

        M = b.max()
        # active constraints
        eps_active = max(1e-12, M * 1e-8)
        active_idx = np.where(b >= M - eps_active)[0]
        if active_idx.size == 0:
            active_idx = np.array([np.argmax(b)])
        # Build constraint matrix A of shape (len(active_idx), n)
        rows = []
        for k in active_idx:
            row = np.zeros(n, dtype=np.float64)
            i_low = max(0, k - (n - 1))
            i_high = min(k, n - 1)
            i_vals = np.arange(i_low, i_high + 1, dtype=int)
            j_vals = k - i_vals

"""
Propose a step‑function (non‑negative sequence) with a low
evaluate_sequence() score.

Algorithm
---------
1. Initialise a set of candidate vectors using several analytic families
   (uniform, arcsine, Beta, exponential, mixtures) for a handful of
   lengths around a previously discovered construction (if available).

2. Search a cheap two‑level parametrisation (front block of height p,
   back block of height q) on the same lengths.

3. Run a number of random restarts:
   * initialise a Dirichlet vector (different concentration parameters)
   * optimise the smooth‑max surrogate of the ℓ∞‑convolution with Adam
   * refine the exact max with sub‑gradient descent
   The best result over all restarts updates the global best.

4. Linear‑programming refinement (optional, requires SciPy).  For the
   current best vector f, solve the LP
        maximise   Σ g_i
        subject to  f * g ≤ M   (elementwise),   g ≥ 0,
   where M = max (f*f).  After normalising g, replace f by a convex
   combination (1‑t)f + t g.  Repeat a few times – this mimics the
   classical “alternating LP” improvement and often drops the peak
   noticeably.

5. Exact max sub‑gradient descent followed by two greedy local‑search
   stages:
   * a directed mass‑transfer search that moves a small fraction of mass
     from a source heavily participating in the active peak to a random
     destination,
   * a simulated‑annealing search that occasionally accepts uphill moves
     to escape shallow basins.

6. Final evaluation with the black‑box ``evaluate_sequence`` and return
   the best vector found (as a plain Python list).

All vectors are projected onto the ℓ1‑simplex Σa_i = 1 after every
update, guaranteeing non‑negativity and scale‑invariance of the score.
The routine respects the supplied time budget (default 1000 s) and
prints progress information.

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth‑max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # stable soft‑max weights
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.01,
                   alpha_max: float = 5000.0,
                   lr0: float = 0.3,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best vector encountered (projected onto the simplex)
    and its exact peak value.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size schedule (slow decay)
        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        # continuation on the smoothness parameter
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny jitter
        if t % 400 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    One sub‑gradient step for the exact ℓ∞ objective.
    Uses an equal weight on all indices attaining the current maximum.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Sub‑gradient descent on the exact max‑convolution objective.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.02 / np.sqrt(it)  # diminishing step
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        # occasional jitter
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


# ----------------------------------------------------------------------
def _directed_local_search(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Greedy mass‑transfer local search that always targets the current peak.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        # pick a convolution index from the active peak
        max_idx = np.random.choice(np.where(b >= cur_max - 1e-12)[0])
        i_low = max(0, max_idx - (n - 1))
        i_high = min(max_idx, n - 1)
        i_vals = np.arange(i_low, i_high + 1, dtype=int)
        prod = a[i_vals] * a[max_idx - i_vals]
        if prod.sum() == 0.0:
            src = np.random.randint(n)
        else:
            prob = prod / prod.sum()
            src = np.random.choice(i_vals, p=prob)

        dst = np.random.randint(n)
        if dst == src:
            continue

        max_delta = min(a[src], 0.05)          # at most 5 % of source mass
        delta = np.random.rand() * max_delta

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        b_new = _fft_self_convolve(a_new)
        new_max = b_new.max()

        if new_max < cur_max - 1e-15:
            a = a_new
            b = b_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()

        # tiny random jitter
        if np.random.rand() < 0.005:
            a += np.random.randn(n) * 1e-7
            a = _project_onto_simplex(a, s=1.0)
            b = _fft_self_convolve(a)
            cur_max = b.max()
    return best_a


# ----------------------------------------------------------------------
def _simulated_annealing_search(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simulated‑annealing style local search: moves a fraction of mass from a
    source (chosen proportionally to its contribution to the active peak)
    to a random destination and accepts uphill moves with a linearly
    decreasing temperature.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_max = _fft_self_convolve(a).max()
    start = time.time()
    # initial temperature proportional to current max
    T0 = max(1e-4, best_max * 0.02)

    while time.time() < deadline:
        elapsed = time.time() - start
        remaining = deadline - start
        T = T0 * max(0.0, 1.0 - elapsed / remaining)  # linear cooling

        b = _fft_self_convolve(a)
        cur_max = b.max()
        # pick a convolution index that is (almost) maximal
        k = np.random.choice(np.where(b >= cur_max - 1e-12)[0])
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        i_vals = np.arange(i_low, i_high + 1, dtype=int)
        prod = a[i_vals] * a[k - i_vals]
        if prod.sum() == 0.0:
            src = np.random.randint(n)
        else:
            prob = prod / prod.sum()
            src = np.random.choice(i_vals, p=prob)

        dst = np.random.randint(n)
        if dst == src:
            continue

        max_delta = min(a[src], 0.08)          # up to 8 % of source mass
        delta = np.random.rand() * max_delta

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_max = _fft_self_convolve(a_new).max()

        if new_max < cur_max - 1e-15:
            a = a_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
        else:
            # Metropolis acceptance
            if T > 0 and np.random.rand() < math.exp(-(new_max - cur_max) / T):
                a = a_new
                cur_max = new_max

        # occasional tiny jitter
        if np.random.rand() < 0.001:
            a += np.random.randn(n) * 1e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a


# ----------------------------------------------------------------------
def _lp_improve(f: np.ndarray,
                max_iters: int,
                t: float,
                deadline: float):
    """
    Linear‑programming based improvement.
    Repeatedly solves:
        maximise Σ g_i
        s.t.   f * g ≤ M   (M = max(f*f)),   g ≥ 0
    and mixes the solution back into f with weight t.
    Only the currently active constraints (those attaining M) are kept,
    making the LP tiny.
    Returns the best f encountered and its fast score.
    """
    n = f.shape[0]
    best_f = f.copy()
    best_score = 2.0 * n * _fft_self_convolve(best_f).max()

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv_ff = _fft_self_convolve(f)
        M = conv_ff.max()
        eps_active = max(1e-12, M * 1e-8)
        active_idx = np.where(conv_ff >= M - eps_active)[0]
        if active_idx.size == 0:
            active_idx = np.array([np.argmax(conv_ff)])

        # Build the (small) inequality matrix A·g ≤ M
        rows = []
        for k in active_idx:
            row = np.zeros(n, dtype=np.float64)
            i_low = max(0, k - (n - 1))
            i_high = min(k, n - 1)
            i_vals = np.arange(i_low, i_high + 1, dtype=int)
            j_vals = k - i_vals
            row[j_vals] = f[i_vals]
            rows.append(row)
        A = np.vstack(rows)
        b_ub = np.full(A.shape[0], M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)   # maximise Σg_i  → minimise -Σg_i

        if not _SCIPY_LP:          # pragma: no cover
            break

        try:
            res = linprog(c,
                          A_ub=A,
                          b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs')
        except Exception:
            break

        if not res.success:
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g = g / sum_g                    # re‑normalise to Σg = 1

        # convex mixing
        f = (1.0 - t) * f + t * g
        f = _project_onto_simplex(f, s=1.0)

        cur_score = 2.0 * n * _fft_self_convolve(f).max()
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
    return best_f, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Main entry point.  Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin of 1 s for the final black‑box call
    deadline = start_time + budget_s - 1.0

    # --------------------------------------------------------------
    # Retrieve a previously discovered construction if present.
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_seq = _project_onto_simplex(base_seq, s=1.0)
        base_n = len(base_seq)
    except Exception:
        base_seq = None
        base_n = 4000

    # --------------------------------------------------------------
    # Lengths to explore – a mix around the previous best.
    lengths = [max(500, int(base_n * f))
               for f in (0.6, 0.8, 1.0, 1.2, 1.4)
               if int(base_n * f) >= 500]
    if not lengths:
        lengths = [max(500, base_n)]

    overall_best_seq = None
    overall_best_score = float('inf')

    # --------------------------------------------------------------
    # Phase 0 – broad sampling of simple analytic families.
    print("[phase 0] broad family sampling")
    for n in lengths:
        if time.time() >= deadline:
            break
        starters = {}

        # arcsine (the classic continuous optimum)
        starters["arcsine"] = (1.0 / np.sqrt((np.arange(n) + 0.5) *
                                            (n - np.arange(n) - 0.5)))
        starters["arcsine"] /= starters["arcsine"].sum()

        # uniform
        starters["uniform"] = np.full(n, 1.0 / n, dtype=np.float64)

        # a couple of random draws
        starters["random1"] = np.random.rand(n)
        starters["random1"] /= starters["random1"].sum()
        starters["random2"] = np.random.rand(n)
        starters["random2"] /= starters["random2"].sum()

        # symmetric Beta(a,a) densities
        for a in (0.25, 0.35, 0.45, 0.55, 0.65, 0.75):
            x = (np.arange(n, dtype=np.float64) + 0.5) / n
            log_vals = (a - 1.0) * np.log(x) + (a - 1.0) * np.log(1.0 - x)
            vals = np.exp(log_vals)
            vals[~np.isfinite(vals)] = 0.0
            if vals.sum() == 0.0:
                vals = np.full(n, 1.0 / n)
            else:
                vals /= vals.sum()
            starters[f"beta_{a:.2f}"] = vals

        # exponential decays
        for lam in (0.0005, 0.001, 0.002):
            vec = np.exp(-lam * np.arange(n))
            vec /= vec.sum()
            starters[f"exp_{lam:.4f}"] = vec

        # mixes of arcsine and uniform
        for w in (0.2, 0.4, 0.6):
            mix = (1.0 - w) * starters["arcsine"] + w * starters["uniform"]
            starters[f"mix_{w:.1f}"] = mix

        if base_seq is not None:
            # linear interpolation of previous best to the new length
            old_n = len(base_seq)
            x_old = np.linspace(0.0, 1.0, old_n)
            x_new = np.linspace(0.0, 1.0, n)
            interp = np.interp(x_new, x_old, base_seq)
            interp = np.maximum(0.0, interp)
            interp /= interp.sum()
            starters["prev"] = interp

        for tag, vec in starters.items():
            vec = _project_onto_simplex(vec, s=1.0)
            peak = _fft_self_convolve(vec).max()
            fast_score = 2.0 * n * peak
            if fast_score < overall_best_score:
                overall_best_score = fast_score
                overall_best_seq = vec.copy()
                print(f"[init] n={n:4d} {tag:<12s} fast_score={fast_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – cheap two‑level parametric search.
    print("[phase 1] two‑level parametric search")
    for n in lengths:
        if time.time() >= deadline:
            break
        seq, score = _two_level_family(n, deadline)
        if seq is not None and score < overall_best_score:
            overall_best_score = score
            overall_best_seq = seq.copy()
            print(f"[two‑level] n={n:4d} fast_score={score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – random restarts with Adam + sub‑gradient refinement.
    time_left = deadline - time.time()
    if time_left > 15.0:
        print("[phase 2] random restarts")
        # allocate roughly 30 % of remaining time to this phase
        phase_end = time.time() + 0.30 * time_left
        restart_cnt = 0
        while time.time() < phase_end:
            restart_cnt += 1
            # pick a length (prefer the best length seen so far)
            n = np.random.choice(lengths)
            # Dirichlet (Gamma) concentration – try a few values
            conc = np.random.choice([0.1, 0.3, 0.6, 1.0, 2.0, 5.0])
            vec = np.random.gamma(conc, 1.0, size=n)
            vec /= vec.sum()

            # Adam optimisation (limited time per restart)
            adam_dead = time.time() + min(8.0, max(2.0, (phase_end - time.time()) * 0.4))
            a_opt, _ = _adam_optimize(vec, adam_dead)

            # Sub‑gradient refinement
            sub_dead = time.time() + min(6.0, max(2.0, (phase_end - time.time()) * 0.4))
            a_ref = _refine_max(a_opt, sub_dead)

            fast_score = 2.0 * n * _fft_self_convolve(a_ref).max()
            if fast_score < overall_best_score:
                overall_best_score = fast_score
                overall_best_seq = a_ref.copy()
                print(f"[restart {restart_cnt:02d}] n={n:4d} fast_score={fast_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – LP‑based improvement (if SciPy is available).
    if _SCIPY_LP and overall_best_seq is not None:
        time_left = deadline - time.time()
        if time_left > 8.0:
            print("[phase 3] LP refinement")
            # Use a modest number of iterations – each LP is cheap because
            # we only keep the active constraints.
            lp_dead = time.time() + min(6.0, time_left * 0.6)
            best_lp_seq, best_lp_score = _lp_improve(
                overall_best_seq.copy(),
                max_iters=7,
                t=0.28,
                deadline=lp_dead)
            if best_lp_score < overall_best_score:
                overall_best_score = best_lp_score
                overall_best_seq = best_lp_seq.copy()
                print(f"[LP] improved fast_score={best_lp_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – exact max sub‑gradient descent + greedy local searches.
    time_left = deadline - time.time()
    if overall_best_seq is not None and time_left > 8.0:
        n = len(overall_best_seq)
        print(f"[phase 4] sub‑gradient + local search (n={n}) – time left {time_left:.1f}s")
        # Sub‑gradient descent (half of the remaining budget)
        sub_end = time.time() + time_left * 0.45
        a_refined = _refine_max(overall_best_seq, sub_end)

        # Directed greedy mass‑transfer (10 % of remaining time)
        gre_end = time.time() + (deadline - time.time()) * 0.10
        a_directed = _directed_local_search(a_refined, gre_end)

        # Simulated‑annealing search (use whatever time is left)
        a_final = _simulated_annealing_search(a_directed, deadline)

        overall_best_seq = a_final.copy()
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print(f"[phase 4] after local searches fast_score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # Final black‑box evaluation
    try:
        true_score = evaluate_sequence(overall_best_seq.tolist())
        print(f"[final] evaluate_sequence score={true_score:.6f}")
        if true_score < overall_best_score:
            overall_best_score = true_score
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    print(f"Best score {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()


# ----------------------------------------------------------------------
def _two_level_family(n: int, deadline: float):
    """
    Exhaustive search over a simple two‑level family:
        a_i = p for i < L,   a_i = q otherwise,
    with ratio r = p/q ≥ 1.  Returns the best vector found and its fast score.
    """
    best_seq = None
    best_score = float('inf')
    step_L = max(1, n // 100)          # ≈1 % of the length
    r_grid = np.logspace(0.0, 2.3, num=25)   # ratios from 1 to ≈200

    for L in range(step_L, n, step_L):
        if time.time() > deadline:
            break
        denom_base = n - L
        for r in r_grid:
            denom = L * r + denom_base
            q = 1.0 / denom
            p = r * q
            seq = np.empty(n, dtype=np.float64)
            seq[:L] = p
            seq[L:] = q
            peak = _fft_self_convolve(seq).max()
            score = 2.0 * n * peak
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
    return best_seq, best_score
