   # a is original before move.

   n = a.shape[0]
   c = _fft_self_convolve(a)  # compute full conv
   cur_max = c.max()
   cur_score = 2.0 * n * cur_max
   best_a = a.copy()
   best_score = cur_score
   # parameters
   max_transfer = 0.1   (maybe 0.05)
   T = cur_score * 0.02  # maybe

       if a[i] <= 1e-15:
           continue

def _annealing_optimize(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    c = _fft_self_convolve(a)  # shape 2n-1
    cur_max = c.max()
    cur_score = 2.0 * n * cur_max
    best_a = a.copy()
    best_score = cur_score

    max_transfer = 0.1
    T = cur_score * 0.05
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # decayed transfer amount
        transfer = max_transfer * (0.5 ** (iteration / 25000.0))
        i = np.random.randint(n)
        if a[i] < 1e-12:
            continue
        j = np.random.randint(n-1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        c_old = c.copy()
        a_i_old, a_j_old = a[i], a[j]

        _apply_delta(a, c, i, -delta, j, delta)

        new_max = c.max()
        new_score = 2.0 * n * new_max

        if new_score < cur_score:
            accept = True
        else:
            # acceptance probability
            accept_prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            accept = np.random.rand() < accept_prob

        if accept:
            cur_score = new_score
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] annealing iter={iteration} new best {best_score:.6f}")
            # cool down (slowly)
            T *= 0.99995
        else:
            # revert
            a[i] = a_i_old
            a[j] = a_j_old
            c[:] = c_old

        # occasional jitter to avoid stagnation
        if iteration % 5000 == 0 and np.random.rand() < 0.1:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, 0.0)

            if a_sum > 0:
                a /= a_sum
            c = _fft_self_convolve(a)
            cur_max = c.max()
            cur_score = 2.0 * n * cur_max
    return best_a, best_score

"""
Multi‑stage search for a low‑score non‑negative step function.

1. **Candidate generation** – produce a set of sequences using analytic
   families (arcsine, symmetric β, exponential) together with a few
   random Dirichlet draws.  All candidates are normalised to ℓ₁‑norm 1
   and scored with the cheap FFT‑based objective
   ``2·n·max(a*a)`` (identical to ``evaluate_sequence`` for sum = 1).

2. **Mirror‑descent refinement** – the best few seeds are optimised by
   exponentiated‑gradient (KL‑mirror descent) and by Adam‑style adaptive
   moment descent.  These use the exact sub‑gradient of the max‑convolution
   and keep the iterate on the simplex.

3. **Simulated annealing** – starting from the current best sequence we
   perform a large number of *mass‑transfer* moves while maintaining the
   full self‑convolution vector with an O(n) incremental update.  Moves
   that increase the score are occasionally accepted according to a
   temperature schedule, allowing the search to escape local minima.
   Periodic re‑normalisation and full recomputation of the convolution
   keep numerical drift in check.

The routine respects the global time budget (≈ 1000 s) and always
returns a list of non‑negative floats that can be fed directly to
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 15000,
                          eta0: float = 0.6, decay: float = 0.9999):
    """Exponentiated‑gradient (KL‑mirror descent)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / a.size)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score, best_a = cur_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 15000,
                  step0: float = 0.12,
                  beta1: float = 0.9,
                  beta2: float = 0.999):
    """Adam‑style adaptive‑moment descent on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score, best_a = cur_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _apply_delta(a: np.ndarray, c: np.ndarray,
                 i: int, delta_i: float, j: int, delta_j: float):
    """
    Incrementally update the full convolution vector ``c`` after moving
    ``delta_i`` from position ``i`` and ``delta_j`` from position ``j``.
    The array ``a`` is the *old* vector; it is updated in‑place after
    the convolution update.
    """
    n = a.shape[0]

    # linear contributions from the two positions
    c[i : i + n] += 2.0 * delta_i * a
    c[j : j + n] += 2.0 * delta_j * a

    # quadratic diagonal terms
    c[2 * i] += delta_i * delta_i
    c[2 * j] += delta_j * delta_j

    # cross term (i,j)
    c[i + j] += 2.0 * delta_i * delta_j

    # finally update the coefficients themselves
    a[i] = a[i] + delta_i
    a[j] = a[j] + delta_j


# ----------------------------------------------------------------------
def _annealing_optimize(a0: np.ndarray, deadline: float):
    """
    Simulated‑annealing style optimisation that preserves the ℓ₁‑norm.
    It works with an exact O(n) incremental convolution update, allowing
    many cheap trials.
    """
    a = a0.copy()
    n = a.shape[0]
    c = _fft_self_convolve(a)                # full self‑convolution
    cur_score = 2.0 * n * c.max()
    best_a = a.copy()
    best_score = cur_score

    # annealing parameters
    max_transfer = 0.08          # up‑to 8 % of total mass in a single trial
    temperature = cur_score * 0.04
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # geometric decay of the transferred amount
        transfer = max_transfer * (0.5 ** (iteration / 25000.0))

        i = np.random.randint(n)
        if a[i] <= 1e-16:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        # keep a copy of the convolution vector for a cheap revert
        c_old = c.copy()
        a_i_old, a_j_old = a[i], a[j]

        _apply_delta(a, c, i, -delta, j, delta)

        new_score = 2.0 * n * c.max()

        if new_score < cur_score:
            accept = True
        else:
            # Metropolis acceptance rule
            prob = math.exp(-(new_score - cur_score) / temperature) if temperature > 0 else 0.0
            accept = np.random.rand() < prob

        if accept:
            cur_score = new_score
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] anneal iter={iteration} new best {best_score:.6f}")
            temperature *= 0.99995   # slow cooling
        else:
            # revert the move
            a[i], a[j] = a_i_old, a_j_old
            c[:] = c_old

        # occasional full recomputation to avoid numerical drift
        if iteration % 5000 == 0:
            c = _fft_self_convolve(a)
            cur_score = 2.0 * n * c.max()
            # tiny jitter to shake out plateaus
            if np.random.rand() < 0.12:
                jitter = np.random.randn(n) * 1e-5
                a += jitter
                a = np.maximum(a, 0.0)
                a_sum = a.sum()
                if a_sum > 0:
                    a /= a_sum
                c = _fft_self_convolve(a)
                cur_score = 2.0 * n * c.max()

    return best_a, best_score
