def _exact_gradient(a, conv):
    n = a.shape[0]
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)
    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    # subtract diagonal correction for even active indices.
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad
```

   one) is used to keep the few most promising candidates.

2. **Local optimisation** – every kept candidate is refined by Adam
   optimisation on the simplex, using the *exact* sub‑gradient of the
   max‑convolution (no temperature schedule).  The optimisation runs
   until the time budget for the current length is exhausted.

3. **Selection** – after refinement the true `evaluate_sequence` is
   called; the globally best vector found so far is returned (or a
   uniform fallback if nothing succeeded).

The search respects the 1000 s limit, prints progress, and makes
extensive use of FFT‑based convolutions for speed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float = 1.0) -> np.ndarray:
    """Geometric‑like decreasing shape a_i ∝ exp(-lam·i/(n‑1))."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * i / max(1, n - 1))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    a must sum to 1, conv = a*a (full linear convolution).
    """
    n = a.shape[0]
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even shifts (i=j)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad


# ----------------------------------------------------------------------
def _adam_step(z: np.ndarray,
               g: np.ndarray,
               m: np.ndarray,
               v: np.ndarray,
               t: int,
               lr: float = 0.5,
               beta1: float = 0.9,
               beta2: float = 0.999,
               eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Adam update on the unconstrained variable z."""
    m = beta1 * m + (1.0 - beta1) * g
    v = beta2 * v + (1.0 - beta2) * (g * g)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z, m, v


# ----------------------------------------------------------------------
def _initial_vector(kind: str, n: int) -> np.ndarray:
    """Create a feasible probability vector of length n."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif kind == "arcsine":
        a = _beta_distribution(n, 0.5, 0.5)
    elif kind == "exp":
        a = _exponential_distribution(n, lam=1.0)
    elif kind == "prev":
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.size != n:
                x_old = np.linspace(0.0, 1.0, prev.size)
                x_new = np.linspace(0.0, 1.0, n)
                a = np.interp(x_new, x_old, prev)
            else:
                a = prev.copy()
        except Exception:
            a = _beta_distribution(n, 0.5, 0.5)
    else:
        a = _beta_distribution(n, 0.5, 0.5)

    a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 1234, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 0. base length (use previous result if present)
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # 1. lengths to explore (a handful around the base length)
    factors = (0.75, 1.0, 1.25, 1.5, 2.0)
    lengths = sorted(
        max(2000, int(base_n * f)) for f in factors if int(base_n * f) >= 2000
    )
    if not lengths:
        lengths = [base_n]

    # helper cheap scoring (exact objective for sum‑to‑one vectors)
    def cheap_score(a: np.ndarray) -> float:
        return 2.0 * a.shape[0] * _fft_self_convolve(a).max()

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        time_left = deadline - time.time()
        slice_deadline = time.time() + min(time_left * 0.4, 240.0)   # at most 4 min per length
        print(f"[{time.time():.1f}] Exploring length n={n}")

        # --------------------------------------------------------------
        # 2. build a diverse pool of analytic candidates
        pool: list[np.ndarray] = []

        # exhaustive grid on (α,β) in log‑space
        alphas = np.logspace(np.log10(0.05), np.log10(10.0), num=12)
        betas = np.logspace(np.log10(0.05), np.log10(10.0), num=12)
        for a_val in alphas:
            for b_val in betas:
                pool.append(_beta_distribution(n, a_val, b_val))

        # extra manually‑chosen shapes
        pool.append(_beta_distribution(n, 0.5, 0.5))          # arcsine‑like
        pool.append(_exponential_distribution(n, lam=0.5))
        pool.append(_exponential_distribution(n, lam=2.0))
        pool.append(_initial_vector("uniform", n))
        pool.append(_initial_vector("random", n))
        pool.append(_initial_vector("arcsine", n))
        pool.append(_initial_vector("exp", n))

        # random Dirichlet draws with various concentrations
        for conc in (0.1, 0.5, 1.0, 2.0):
            a = np.random.dirichlet(np.full(n, conc))
            pool.append(a)

        # previous best if available
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.size != n:
                x_old = np.linspace(0.0, 1.0, prev.size)
                x_new = np.linspace(0.0, 1.0, n)
                a = np.interp(x_new, x_old, prev)
                a = _project_onto_simplex(a)
            else:
                a = _project_onto_simplex(prev)
            pool.append(a)
        except Exception:
            pass

        # cheap scores + keep the best few
        scored = [(cheap_score(vec), vec) for vec in pool]
        scored.sort(key=lambda x: x[0])
        starters = [vec.copy() for _, vec in scored[:4]]
        print(f"  top cheap scores: {[s for s, _ in scored[:4]]}")

        # --------------------------------------------------------------
        # 3. refine each starter with Adam on the simplex (exact gradient)
        for sid, a0 in enumerate(starters):
            now = time.time()
            cand_deadline = min(slice_deadline, now + 80.0)  # max 80 s per candidate
            if cand_deadline - now < 5.0:
                continue

            # initialise Adam on the log‑odds (softmax) variables
            a = a0.copy()
            a = np.maximum(a, 1e-15)          # avoid log(0)
            z = np.log(a)                     # unconstrained
            m = np.zeros_like(z)
            v = np.zeros_like(z)

            best_local_a = a.copy()
            best_local_score = cheap_score(a)

            it = 0
            while time.time() < cand_deadline:
                it += 1

                # current probability vector
                a = np.exp(z)
                s = a.sum()
                if s == 0:
                    break
                a /= s

                conv = _fft_self_convolve(a)
                cur_score = 2.0 * n * conv.max()

                # keep the best point seen during this run
                if cur_score < best_local_score:
                    best_local_score = cur_score
                    best_local_a = a.copy()
                    print(
                        f"[{time.time():.1f}] n={n} cand={sid} it={it} score={cur_score:.6f}"
                    )
                    if best_local_score <= 1.503:          # early exit trigger
                        break

                # exact sub‑gradient of the max‑convolution
                grad_a = _exact_gradient(a, conv)

                # convert to gradient w.r.t. z (softmax parametrisation)
                dot = np.dot(a, grad_a)
                grad_z = a * (grad_a - dot)

                # Adam update
                z, m, v = _adam_step(z, grad_z, m, v, it,
                                     lr=0.6, beta1=0.9, beta2=0.999, eps=1e-8)

                # occasional jitter to escape plateaus
                if it % 3000 == 0 and np.random.rand() < 0.12:
                    z += np.random.randn(n) * 1e-7

            # ----------------------------------------------------------
            # evaluate the best local candidate with the true black‑box
            try:
                final_score = evaluate_sequence(best_local_a.tolist())
            except Exception:
                final_score = best_local_score

            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = best_local_a.copy()
                print(
                    f"*** Global improvement: n={n} cand={sid} → "
                    f"score={overall_best_score:.6f}"
                )
                if overall_best_score <= 1.503:
                    print("Target reached – exiting early.")
                    return overall_best_seq.tolist()

            if time.time() >= slice_deadline:
                break

        # end of length loop – global time guard
        if time.time() >= deadline:
            break
        if overall_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # fallback (should never be needed)
    if overall_best_seq is None:
        n_fallback = 4000
        overall_best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())
        print("Fallback used.")

    print(
        f"Final best score: {overall_best_score:.6f} "
        f"(length = {len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
