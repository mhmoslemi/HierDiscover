"""
Propose a low‑score step‑function.

The search proceeds in three stages:

1. **Broad sampling** – many simple families (arcsine, Beta, power‑law,
   Dirichlet, mixtures) are evaluated on several lengths (≈5k‑20k). The
   best according to the cheap proxy `2·n·max(conv)` is kept.

2. **Local refinement** – the best candidate is improved with a few
   restarts of Adam optimisation on the smooth‑max surrogate, followed
   by a short exact ℓ∞ sub‑gradient descent.

3. **Polish** – a light simulated‑annealing pass accepts improving
   moves (and occasional uphill moves) until the time budget expires.

All vectors are kept on the probability simplex (Σaᵢ=1, aᵢ≥0), which
makes the final score `evaluate_sequence` independent of scaling.
The routine respects the 1000‑second budget and returns a plain list.
"""

import math
import time
import numpy as np
from numpy.fft import rfft, irfft


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_vals = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    vals[~np.isfinite(vals)] = 0.0
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


def _powerlaw_boundary_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5) + 1e-12) ** (-p)
    vals /= vals.sum()
    return vals


def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a non‑negative sequence to a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """Gradient of Lα(a) = (1/α)·log Σ exp(α·(a*a)_k)."""
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # log‑sum‑exp trick
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.01,
                   alpha_max: float = 2000.0,
                   lr0: float = 0.25,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):

    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional jitter to escape flat regions
        if t % 500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


def _max_subgradient_step(a: np.ndarray):
    """Exact ℓ∞ sub‑gradient step for the self‑convolution peak."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Short exact‑max refinement using sub‑gradient descent."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.02 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        if it % 400 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


def _simulated_annealing(a0: np.ndarray, deadline: float):
    """Light simulated‑annealing polish."""
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_peak = _fft_self_convolve(cur_a).max()
    best_a = cur_a.copy()
    best_peak = cur_peak

    start = time.time()
    total_time = max(deadline - start, 1e-6)

    start_temp = max(cur_peak, 1e-3) * 0.5
    final_temp = start_temp * 1e-5
    if final_temp <= 0.0:
        final_temp = 1e-9
    cooling_rate = -math.log(final_temp / start_temp) / total_time

    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1
        src = np.random.randint(n)
        if cur_a[src] <= 1e-12:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue
        max_delta = min(cur_a[src], 0.05)
        delta = np.random.rand() * max_delta

        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_peak = _fft_self_convolve(a_new).max()
        delta_obj = new_peak - cur_peak

        elapsed = time.time() - start
        temperature = start_temp * math.exp(-cooling_rate * elapsed)

        accept = False
        if delta_obj < 0.0:
            accept = True
        else:
            prob = math.exp(-delta_obj / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_a = a_new
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_a = cur_a.copy()

        if iter_cnt % 20000 == 0 and np.random.rand() < 0.05:
            cur_a = best_a.copy()
            cur_peak = best_peak

    return best_a, best_peak


# ----------------------------------------------------------------------
def _fast_score(vec: np.ndarray) -> float:
    """Cheap proxy 2·n·max(conv) – identical to evaluate_sequence for sum=1."""
    n = vec.shape[0]
    return 2.0 * n * _fft_self_convolve(vec).max()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a step‑function with a low score.
    Returns a plain Python list suitable for `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Optional warm‑start from a previously discovered construction.
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_seq = _project_onto_simplex(base_seq, s=1.0)
    except Exception:
        base_seq = None

    # ------------------------------------------------------------------
    # Lengths to explore – a few large values give a better asymptotic constant.
    lengths = [5000, 8000, 12000, 15000, 18000, 20000]
    overall_best_seq = None
    overall_best_score = float("inf")

    print("[phase 1] broad sampling over several lengths")
    for n in lengths:
        if time.time() >= deadline:
            break

        candidates = {}

        # Classic families
        candidates["arcsine"] = _arcsine_distribution(n)
        candidates["uniform"] = np.full(n, 1.0 / n, dtype=np.float64)

        # Symmetric Beta with a wide range of alphas (including extreme ends)
        for a in (0.1, 0.15, 0.2, 0.25, 0.3, 0.35, 0.4, 0.45, 0.5, 0.6):
            candidates[f"beta_{a:.2f}"] = _beta_distribution(n, a, a)

        # Power‑law boundary families
        for p in (0.3, 0.35, 0.4, 0.45, 0.5, 0.55):
            candidates[f"power_{p:.2f}"] = _powerlaw_boundary_distribution(n, p)

        # Random Dirichlet draws (few samples)
        for conc in (0.05, 0.1, 0.2):
            candidates[f"dirich_{conc:.2f}"] = np.random.dirichlet(np.full(n, conc))

        # Mixtures of arcsine and uniform
        for w in (0.2, 0.5, 0.8):
            candidates[f"mix_{w:.1f}"] = (1.0 - w) * _arcsine_distribution(n) + w * np.full(n, 1.0 / n)

        # Interpolated previous best (if available)
        if base_seq is not None:
            candidates["prev_interp"] = _interpolate_to_length(base_seq, n)

        # ------------------------------------------------------------------
        # Evaluate cheap proxy and keep the best
        for tag, vec in candidates.items():
            if time.time() >= deadline:
                break
            vec = _project_onto_simplex(vec, s=1.0)
            sc = _fast_score(vec)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = vec.copy()
                print(f"[init] n={n:5d} {tag:<12s} fast_score={sc:.6f}")

    # Fallback: uniform distribution if nothing was generated
    if overall_best_seq is None:
        n = 5000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = _fast_score(overall_best_seq)
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # 2.  Local refinement with Adam + sub‑gradient (a few restarts)
    time_left = deadline - time.time()
    if time_left > 20.0:
        n = overall_best_seq.shape[0]
        print(f"[phase 2] refinement (n={n}) – {time_left:.1f}s left")
        # number of independent restarts; share time evenly
        restarts = 3
        per_restart = max(5.0, time_left / (restarts + 1))
        best_seq = overall_best_seq.copy()
        best_score = overall_best_score

        for r in range(restarts):
            if time.time() >= deadline:
                break
            # jitter the current best to obtain a fresh start
            start_vec = best_seq + np.random.randn(n) * 1e-3
            start_vec = np.clip(start_vec, 0, None)
            start_vec = _project_onto_simplex(start_vec, s=1.0)

            adam_deadline = time.time() + per_restart * 0.6
            a_opt, _ = _adam_optimize(start_vec, adam_deadline,
                                       init_alpha=1.0,
                                       alpha_factor=1.02,
                                       alpha_max=3000.0,
                                       lr0=0.4)

            sub_deadline = time.time() + per_restart * 0.35
            a_opt = _refine_max(a_opt, sub_deadline)

            sc = _fast_score(a_opt)
            if sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"[phase 2] restart {r} improved to {sc:.6f}")

        overall_best_seq = best_seq
        overall_best_score = best_score

    # ------------------------------------------------------------------
    # 3.  Light simulated‑annealing polishing
    time_left = deadline - time.time()
    if time_left > 2.0:
        n = overall_best_seq.shape[0]
        print(f"[phase 3] simulated annealing (n={n}) – {time_left:.1f}s left")
        a_final, final_peak = _simulated_annealing(overall_best_seq, deadline)
        overall_best_seq = a_final
        overall_best_score = 2.0 * n * final_peak
        print(f"[phase 3] SA final fast_score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # Final exact evaluation (might be slightly larger than the proxy)
    try:
        true_val = evaluate_sequence(overall_best_seq.tolist())
        print(f"[final] evaluate_sequence score={true_val:.6f}")
        if true_val < overall_best_score:
            overall_best_score = true_val
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    print(f"Best score (approx.): {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
