"""
Optimisation of a non‑negative step function to minimise the
maximum self‑convolution value (the score returned by
``evaluate_sequence``).  The algorithm proceeds in five stages:

1.  Several multi‑restart smooth‑max L‑BFGS‑B optimisations
    (gradients are obtained analytically).  The temperature μ is
    cooled geometrically, ending with the true max‑objective.

2.  Projected sub‑gradient descent on the exact max‑objective,
    with a modest back‑tracking line‑search.

3.  A large random permutation search – values are permuted
    (full shuffles or cyclic steps with a coprime stride) and
    accepted only if they improve the score (fast FFT based
    max‑convolution is used as a cheap filter).

4.  Aggressive stochastic hill‑climbing (position swaps,
    δ‑mass transfers, and targeted transfers that remove mass
    from indices participating in the current maximal convolution
    shift).

5.  A final monotone‑interleaving heuristic (big‑small‑big‑small,
    pure ascending and descending orders) as a sanity check.

The routine respects a user‑supplied time budget (default 1000 s) and
returns the best sequence found.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _compute_smooth_obj_and_grad(y: np.ndarray, mu: float,
                                 n: int, S: float) -> tuple[float, np.ndarray]:
    """Value and gradient of the smooth‑max surrogate."""
    p = _softmax(y)            # probabilities, Σp = 1, p ≥ 0
    a = S * p                  # actual heights, Σa = S

    conv = np.convolve(a, a)   # full auto‑convolution, length 2·n‑1

    max_c = conv.max()
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)          # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:                       # hard max (μ = 0)
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. a  (same as in hard case)
    conv_wr = np.convolve(w, a[::-1])               # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]            # slice length n

    # chain rule back to y (soft‑max Jacobian)
    inner = np.dot(g_a, p)
    grad_y = S * p * (g_a - inner)
    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable for scipy.optimize that returns (value, gradient)."""

    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_smooth_obj_and_grad(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
def _hard_subgradient(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient of max(conv) at point a.
    Returns (g, max_conv) where g has length n.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_c = conv.max()
    mask = conv == max_c
    cnt = np.count_nonzero(mask)
    if cnt == 0:
        w = np.zeros_like(conv)
    else:
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
    conv_wr = np.convolve(w, a[::-1])
    g = 2.0 * conv_wr[n - 1:n - 1 + n]
    return g, max_c


# ----------------------------------------------------------------------
def _max_conv_fft(a: np.ndarray) -> float:
    """Fast (FFT) computation of max(a * a) for a non‑negative vector."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return float(conv.max())

        print(f"[init] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Multi‑restart smooth‑max optimisation
    mu_factor = 0.75
    mu_min = 1e-12
    maxiter_per_mu = 500
    init_ys = []

    # (0) uniform
    init_ys.append(np.zeros(n, dtype=np.float64))

    # (1) decreasing exponential
    alpha = rng.uniform(0.001, 0.004) * (n / 2000.0)
    init_ys.append(-alpha * np.arange(n, dtype=np.float64))

    # (2) increasing exponential
    init_ys.append(-alpha * np.arange(n - 1, -1, -1, dtype=np.float64))

    # (3) random Gaussian
    init_ys.append(rng.standard_normal(n) * 0.3)

    # (4‑7) sparse Dirichlet draws
    for _ in range(4):
        dir_alpha = rng.uniform(0.05, 0.7)
        p = rng.dirichlet([dir_alpha] * n)
        init_ys.append(np.log(p + 1e-12))

    # (8) reference (if available)
    if have_ref:
        a_ref = _project_to_simplex(a_ref, S)
        init_ys.append(np.log(a_ref + 1e-12))

    n_restart = len(init_ys)

    for rid, y0 in enumerate(init_ys):
        if time.time() > deadline:
            break
        print(f"[restart {rid + 1}/{n_restart}] L‑BFGS start")
        y = y0.copy()
        a_cur = S * _softmax(y)
        mu = float(np.convolve(a_cur, a_cur).max())
        mu = max(mu, 1.0)

        while mu > mu_min and time.time() < deadline:
            obj = _SmoothMaxObjective(mu, n, S)
            try:
                res = opt.minimize(
                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": maxiter_per_mu,
                             "gtol": 1e-12,
                             "disp": False})
            except Exception as exc:
                print(f"[error] L‑BFGS failed: {exc}")
                break

            y = res.x
            a = S * _softmax(y)
            sc = evaluate_sequence(a.tolist())
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
                print(f"[improve] new best score = {best_score:.6f}")

            mu *= mu_factor

    # ------------------------------------------------------------------
    # 2️⃣  Projected sub‑gradient descent on the hard max
    if time.time() < deadline:
        a = best_a.copy()
        step0 = S * 0.5 / n
        it = 0
        max_local_iters = 30000
        while time.time() < deadline and it < max_local_iters:
            g, _ = _hard_subgradient(a)
            step = step0 / (1.0 + 1e-6 * it)
            improved = False
            for _ in range(12):
                a_new = a - step * g
                a_new = _project_to_simplex(a_new, S)
                sc = evaluate_sequence(a_new.tolist())
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new
                    improved = True
                    print(f"[subgrad] iter {it:5d} step {step:.2e} → {best_score:.6f}")
                    break
                step *= 0.5
            if not improved:
                # occasional jitter to escape plateaux
                nz = np.where(a > 1e-12)[0]
                if nz.size >= 2:
                    i, j = rng.choice(nz, size=2, replace=False)
                    delta = rng.random() * a[i] * 0.04
                    a[i] -= delta
                    a[j] += delta
                    a = _project_to_simplex(a, S)
            it += 1

    # ------------------------------------------------------------------
    # 3️⃣  Random permutation search (fast FFT filter)
    if time.time() < deadline:
        remaining = deadline - time.time()
        perm_trials = int(min(2000, remaining * 5))
        print(f"[perm] {perm_trials} permutation trials")
        vals = best_a.copy()
        for t in range(perm_trials):
            if rng.random() < 0.7:
                # cyclic permutation with coprime stride
                step = rng.integers(1, n)
                while math.gcd(step, n) != 1:
                    step = rng.integers(1, n)
                perm = (np.arange(n) * step) % n
                cand = vals[perm]
            else:
                perm = rng.permutation(n)
                cand = vals[perm]

            approx = _max_conv_fft(cand)
            if approx < best_score - 1e-12:
                sc = evaluate_sequence(cand.tolist())
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = cand.copy()
                    vals = cand
                    print(f"[perm] trial {t:4d} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Large‑scale stochastic hill‑climbing
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_trials = int(min(50000, remaining * 35))
        print(f"[hill] {max_trials} hill‑climbing trials")
        a = best_a.copy()
        for trial in range(max_trials):
            move_type = rng.integers(0, 3)
            cand = a.copy()
            if move_type == 0:                 # swap two positions
                i, j = rng.choice(n, size=2, replace=False)
                cand[i], cand[j] = cand[j], cand[i]
            elif move_type == 1:               # small δ‑mass transfer
                i = rng.choice(np.where(cand > 1e-12)[0])
                j = rng.choice(n)
                delta = rng.random() * cand[i] * 0.06
                cand[i] -= delta
                cand[j] += delta
            else:                               # targeted transfer from a max‑conv contributor
                conv_full = np.convolve(cand, cand)
                q = int(conv_full.argmax())
                lo = max(0, q - (n - 1))
                hi = min(q, n - 1)
                i = rng.integers(lo, hi + 1)
                if cand[i] > 1e-12:
                    delta = rng.random() * cand[i] * 0.04
                    j = rng.integers(0, n)
                    cand[i] -= delta
                    cand[j] += delta
            # quick FFT filter
            approx = _max_conv_fft(cand)
            if approx < best_score - 1e-12:
                sc = evaluate_sequence(cand.tolist())
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = cand.copy()
                    a = cand
                    print(f"[hill] trial {trial:5d} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final monotone‑interleaving heuristics
    if time.time() < deadline:
        # descending order, then interleave big‑small
        vals_desc = np.sort(best_a)[::-1]
        interleaved = np.empty_like(vals_desc)
        half = len(vals_desc) // 2
        left = vals_desc[:half][::-1]
        right = vals_desc[half:][::-1]
        interleaved[0::2] = left
        interleaved[1::2] = right
        cand = _project_to_simplex(interleaved, S)
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = cand.copy()
            print(f"[final] interleaved → {best_score:.6f}")

        # pure ascending
        asc = np.sort(best_a)
        cand = _project_to_simplex(asc, S)
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = cand.copy()
            print(f"[final] ascending → {best_score:.6f}")

        # pure descending
        cand = _project_to_simplex(vals_desc, S)
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = cand.copy()
            print(f"[final] descending → {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
