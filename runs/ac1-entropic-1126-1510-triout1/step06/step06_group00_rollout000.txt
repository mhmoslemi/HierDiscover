def _softmax_gradient(a, beta):

    b = _fft_self_convolve(a) # length 2n-1

    maxb = b.max()

    # subtract maxb for numerical stability
    exponent = beta * (b - maxb)
    # clip exponent maybe to avoid overflow
    np.clip(exponent, -700, 700, out=exponent) # np.exp can handle up to 709 for double.
    w = np.exp(exponent)
    w_sum = w.sum()
    if w_sum == 0 or np.isnan(w_sum):
        # fallback to uniform w (maybe if overflow)
        w = np.ones_like(b) / len(b)
    else:
        w /= w_sum

    # compute conv_rev = conv(a_rev, w)
    conv_rev = _fft_convolve(a[::-1], w)  # length n + len(w) - 1
    # slice central n entries
    start = a.size - 1
    grad = 2 * conv_rev[start:start + a.size]

    # subtract diagonal term
    # w_even: w[2*j] for j where 2j < len(b)
    w_even = np.zeros_like(a)
    idx_even = np.arange(a.size) * 2
    mask = idx_even < w.shape[0]
    w_even[mask] = w[idx_even[mask]]
    grad -= a * w_even

    return grad, maxb

def _softmax_adam_descent(a0, deadline, max_iters=20000, step0=0.05, beta0=5.0, beta_max=200.0, beta_inc=1.001):

    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    beta = beta0
    t = 0

    while time.time() < deadline and t < max_iters:
        t += 1
        grad, cur_max = _softmax_gradient(a, beta)
        # Adam updates
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad ** 2)
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        # project onto simplex
        a = _project_onto_simplex(a, s=1.0)
        # update current score (exact)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] softmax Adam improvement: {cur_score:.6f}")
        # update beta
        if beta < beta_max:
            beta = min(beta_max, beta * beta_inc)
    return best_a, best_score

"""
Propose a non‑negative step‑function that (hopefully) beats the previous
best score ≈ 1.5101.

Algorithm
---------

1. **Broad sampling** – generate a family of candidate sequences of several
   lengths (≈ 500 … 8000).  Families include the arcsine law, symmetric beta
   distributions with a range of shape parameters, and a few random Dirichlet
   draws.  All candidates are normalised to ℓ₁‑norm 1 and scored with the fast
   “max‑convolution” proxy `2·n·max(a*a)`.

2. **Refinement** – for the best few seeds we run a *soft‑max Adam* optimiser.
   The exact max in the objective is replaced by a smooth log‑sum‑exp
   approximation

        L_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k),

   whose gradient can be computed analytically by convolving the current
   distribution with the soft‑max weights.  Adam with a continuation on β
   (β grows from 5 to 200) gives a smooth descent that typically finds a lower
   exact max than the binary sub‑gradient used before.

3. **Local hill‑climbing** – a cheap random‑transfer move (move a small amount
   of mass from one index to another) is applied repeatedly to the current
   best sequence.  Any improvement is kept.

All steps respect the 1000 s time budget; the routine always returns a list
of non‑negative floats that can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact score for a normalized sequence (sum(a)=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _softmax_weights(b: np.ndarray, beta: float) -> np.ndarray:
    """
    Soft‑max (log‑sum‑exp) weights for the entries of ``b``.
    Numerical stability is ensured by subtracting ``max(b)``.
    """
    maxb = b.max()
    # exponent may become large → clip
    exp_arg = beta * (b - maxb)
    np.clip(exp_arg, -700, 700, out=exp_arg)
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0 or not np.isfinite(w_sum):
        w = np.full_like(b, 1.0 / b.size)
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the smoothed objective L_β(a) = (1/β)·log Σ exp(β·(a*a)_k)
    with respect to ``a`` (assumes sum(a)=1).

    Returns:
        grad  – ndarray of shape a.shape
        cur_max – exact max of the (un‑smoothed) convolution (for logging)
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)                     # length 2n‑1
    cur_max = b.max()
    w = _softmax_weights(b, beta)                # length 2n‑1, sum=1

    # convolution of reversed a with the weight vector
    conv_rev = _fft_convolve(a[::-1], w)         # length n + (2n‑1) - 1
    start = n - 1
    grad = 2.0 * conv_rev[start:start + n]        # central n entries

    # diagonal correction for terms a_j·a_j
    w_even = np.zeros_like(a)
    idx_even = np.arange(n) * 2
    mask = idx_even < w.shape[0]
    w_even[mask] = w[idx_even[mask]]
    grad -= a * w_even
    return grad, cur_max


# ----------------------------------------------------------------------
def _softmax_adam_descent(a0: np.ndarray,
                          deadline: float,
                          max_iters: int = 20000,
                          step0: float = 0.05,
                          beta0: float = 5.0,
                          beta_max: float = 200.0,
                          beta_inc: float = 1.001) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed objective L_β.
    β starts at *beta0* and is multiplied by *beta_inc* each iteration
    (capped at *beta_max*).  The iterate is always projected onto the simplex
    {x ≥ 0, sum(x)=1}.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    beta = beta0
    t = 0
    beta_inc = max(beta_inc, 1.0)   # safety

    while time.time() < deadline and t < max_iters:
        t += 1
        grad, cur_max = _softmax_gradient(a, beta)

        # Adam update (β1=0.9, β2=0.999)
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad ** 2)

        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        # exact score for tracking (expensive but cheap enough for few‑thousand iters)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] soft‑max Adam new best {cur_score:.6f}")

        # increase β (cooling)
        if beta < beta_max:
            beta = min(beta_max, beta * beta_inc)

    return best_a, best_score


# ----------------------------------------------------------------------
def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Gamma‑Dirichlet, α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) density sampled on the grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑density to avoid overflow for extreme p
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))
    w_sum = w.sum()
    if w_sum == 0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals
