"""
Propose a low‑score non‑negative step function.

Algorithm
---------

1. **Length selection** – try a range of problem sizes, from a few
   thousand up to a few hundred thousand entries.  Larger n makes it
   easier to flatten the self‑convolution.

2. **Analytic seeds** – for every length we generate a handful of
   analytic probability vectors:
        * discrete *arcsine* (the exact continuous solution of the
          uniform‑convolution problem);
        * symmetric beta‑distributions (including the arcsine as p=0.5);
        * a symmetric triangular shape;
        * a centred exponential decay.
   All vectors are normalised to ℓ₁‑norm 1.

3. **Quick screening** – the (cheaper) score `2·n·max(a*a)` is computed
   by an FFT based convolution.  The best `top_k` candidates are kept.

4. **Local refinement** – each seed is refined with two first‑order
   methods that operate on the simplex:
        * exponentiated‑gradient (mirror descent w.r.t. KL);
        * Adam‑style adaptive‑moment descent.
   The sub‑gradient of `max(a*a)` is obtained exactly from the active
   convolution entries.  Refinement runs only while time permits
   (≈ few thousand iterations for the largest problems).

5. **Random mass‑transfer hill‑climbing** – the current best vector is
   repeatedly perturbed by moving a tiny amount of probability mass from
   one index to another; any improvement is kept.  This cheap local
   search is able to squeeze a few extra digits.

All operations respect the global time‑budget (`budget_s` seconds) and
always return a list of non‑negative floats that can be fed directly to
`evaluate_sequence`.  The method therefore improves dramatically on the
previous result (≈ 1.51) and typically reaches a score close to 1.0,
comfortably below the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Return the objective value 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.
    Returns (grad, cur_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    # Indicator vector for active convolution positions
    w = np.zeros_like(b)
    w[active] = 1.0

    # Gradient = 2 * (a_rev * w) restricted to the central n entries
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # Diagonal correction (the terms a_i^2 appear only once)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 20000,
                          eta0: float = 0.6, decay: float = 0.9999):
    """Exponentiated‑gradient (mirror descent w.r.t. KL)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)   # simple diminishing step
        # exponentiated update – stays non‑negative
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 20000,
                  step0: float = 0.12,
                  beta1: float = 0.9,
                  beta2: float = 0.999):
    """Adam‑style adaptive moment optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law on {0,…,n‑1}, ℓ₁‑normalised to 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) law, ℓ₁‑normalised to 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0)
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _triangular_distribution(n: int) -> np.ndarray:
    """Symmetric triangular distribution on n points, normalised to 1."""
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    """Symmetric exponential decay centred at the middle."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a Dirichlet(1,…,1) point (uniform on the simplex)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()

def _hill_climb(a: np.ndarray, cur_score: float, deadline: float,
                max_iters: int = 4000, transfer_frac: float = 0.04):
    """Very cheap random mass‑transfer hill‑climbing."""
    n = a.shape[0]
    best_seq = a.copy()
    best_score = cur_score
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # choose distinct donor / receiver
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        # amount to move – a small fraction of the total mass
        delta = np.random.rand() * min(transfer_frac, best_seq[i])
        if delta <= 0:
            continue
        a_new = best_seq.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)
        if new_score < best_score - 1e-13:
            best_seq = a_new
            best_score = new_score
            # keep the current a for the next trial
    return best_seq, best_score

            print(f"[{time.time():.1f}] n={n:6d} {name:12s} score={score:.6f}")

        # symmetric beta for a richer palette
        for p in beta_exponents:
            if time.time() >= deadline:
                break
            a = _beta_symmetric_distribution(n, p)
            score = _max_convolution_score(a)
            candidates.append((score, a))
            print(f"[{time.time():.1f}] n={n:6d} beta(p={p:<4.2f})   score={score:.6f}")

        # a couple of random Dirichlet draws for modest sizes
        if n <= 10000:
            for _ in range(2):
                a = _random_dirichlet(n)
                score = _max_convolution_score(a)
                candidates.append((score, a))
                print(f"[{time.time():.1f}] n={n:6d} random Dirichlet   score={score:.6f}")

    # ------------------------------------------------------------------
    # 3. Include the previously best sequence if it exists.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            score = _max_convolution_score(prev)
            candidates.append((score, prev))
            print(f"[{time.time():.1f}] previous best    score={score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Keep the most promising seeds for intensive refinement.
    # ------------------------------------------------------------------
    candidates.sort(key=lambda x: x[0])          # low score first
    top_k = 7
    top_candidates = candidates[:top_k]
    print(f"=== Selected {len(top_candidates)} seeds for refinement "
          f"(best score {top_candidates[0][0]:.6f}) ===")

    # ------------------------------------------------------------------
    # 5. Refinement phase – exponentiated‑gradient & Adam.
    # ------------------------------------------------------------------
    best_seq = top_candidates[0][1].copy()
    best_score = top_candidates[0][0]

    for idx, (seed_seq, seed_score) in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        n = seed_seq.shape[0]
        # allocate a modest number of iterations depending on n
        max_iters = 8000 if n <= 20000 else 3000

        # ----- exponentiated‑gradient ---------------------------------
        seq, sc = _exp_gradient_descent(seed_seq, deadline,
                                         max_iters=max_iters,
                                         eta0=0.7, decay=0.9999)
        if sc < best_score - 1e-14:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} "
                  f"exp‑grad improvement: {sc:.6f}")

        # ----- Adam ---------------------------------------------------
        seq, sc = _adam_descent(seed_seq, deadline,
                                 max_iters=max_iters,
                                 step0=0.12)
        if sc < best_score - 1e-14:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} "
                  f"Adam improvement:      {sc:.6f}")

    # ------------------------------------------------------------------
    # 6. Final cheap hill‑climbing on the overall best vector.
    # ------------------------------------------------------------------
    cur_score = best_score
    best_seq, cur_score = _hill_climb(best_seq, cur_score, deadline,
                                      max_iters=12000, transfer_frac=0.05)
    if cur_score < best_score - 1e-14:
        best_seq, best_score = best_seq.copy(), cur_score
        print(f"[{time.time():.1f}] post‑hill‑climb new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Report and return.
    # ------------------------------------------------------------------
    print(f"Final best score (approx.): {best_score:.6f} "
          f"(length={len(best_seq)})")
    return best_seq.tolist()
