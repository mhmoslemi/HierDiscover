"""
Multi‑stage optimizer for the min‑max convolution problem.

The pipeline:
1.   Generate a diverse pool of starter probability vectors (arcsine, Beta,
     power‑law, Dirichlet, mixtures, random draws) for several lengths.
2.   Keep the few best (according to the cheap proxy 2·n·max(conv)).
3.   For each survivor run an extended refinement:
       – Adam optimisation on a smooth‑max surrogate (α grows up to 4000).
       – Exact ℓ∞ sub‑gradient descent (longer horizon, larger step).
       – Optional LP improvement (only for moderate lengths ≤ 1500).
       – Prolonged simulated‑annealing polishing.
       – One final sub‑gradient sweep.
4.   Evaluate the true score with ``evaluate_sequence`` and return the best
     sequence found within the time budget.

The implementation respects the simplex (Σaᵢ=1, aᵢ≥0) and uses fast FFT
convolutions.  It works with or without ``cvxpy`` (LP step is simply skipped
if unavailable).  The whole search stops safely before the supplied budget
expires.
"""

import math
import time
import numpy as np
from numpy.fft import rfft, irfft
import scipy.linalg

# optional LP support
try:
    import cvxpy as cp
    _cvxpy_available = True
except Exception:  # pragma: no cover
    _cvxpy_available = False


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a∗a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_distribution(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_vals = (a - 1.0) * np.log(x) + (b - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    vals[~np.isfinite(vals)] = 0.0
    if vals.sum() == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= vals.sum()
    return vals


def _powerlaw_boundary_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5) + 1e-12) ** (-p)
    vals /= vals.sum()
    return vals


def _random_dirichlet(n: int, conc: float) -> np.ndarray:
    return np.random.dirichlet(np.full(n, conc))


def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals

def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """Gradient of Lα(a) = (1/α)·log Σ exp(α·(a∗a)_k)."""
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # shift for numerical stability
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.02,
                   alpha_max: float = 4000.0,
                   lr0: float = 0.5,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """Adam optimisation of the smooth‑max surrogate."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)    # slower decay than in the baseline

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional jitter to escape flat regions
        if t % 700 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 8e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


def _max_subgradient_step(a: np.ndarray):
    """Exact ℓ∞ sub‑gradient step (active‑set weight = uniform)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Longer exact‑max refinement (sub‑gradient descent)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.04 / np.sqrt(it)          # a bit more aggressive than baseline
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        # small random perturbations to avoid stagnation
        if it % 500 == 0 and np.random.rand() < 0.08:
            a += np.random.randn(n) * 7e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


def _simulated_annealing(a0: np.ndarray, deadline: float):
    """Heavier simulated‑annealing polish."""
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_peak = _fft_self_convolve(cur_a).max()
    best_a = cur_a.copy()
    best_peak = cur_peak

    start = time.time()
    total_time = max(deadline - start, 1e-6)

    start_temp = max(cur_peak, 1e-3) * 0.8
    final_temp = start_temp * 5e-6
    cooling_rate = -math.log(final_temp / start_temp) / total_time

    it = 0
    while time.time() < deadline:
        it += 1
        src = np.random.randint(n)
        if cur_a[src] <= 1e-14:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue
        # allow relatively larger moves (up to 10 % of src mass)
        max_delta = min(cur_a[src], 0.10)
        delta = np.random.rand() * max_delta

        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_peak = _fft_self_convolve(a_new).max()
        delta_obj = new_peak - cur_peak
        elapsed = time.time() - start
        temperature = start_temp * math.exp(-cooling_rate * elapsed)

        accept = False
        if delta_obj < 0.0:
            accept = True
        else:
            prob = math.exp(-delta_obj / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_a = a_new
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_a = cur_a.copy()

        # every now and then restart from the current best to dump noise
        if it % 25000 == 0 and np.random.rand() < 0.07:
            cur_a = best_a.copy()
            cur_peak = best_peak

    return best_a, best_peak


def _lp_improve(a: np.ndarray,
                max_iters: int = 3,
                mix_t: float = 0.5,
                inner_deadline: float = None) -> np.ndarray:
    """LP‑based dual‑widening (only for modest n)."""
    if not _cvxpy_available:
        return a

    n = a.shape[0]
    for _ in range(max_iters):
        if inner_deadline is not None and time.time() > inner_deadline:
            break
        M = _fft_self_convolve(a).max()

        # Build Toeplitz convolution matrix A (shape (2n‑1, n))
        col = np.r_[a, np.zeros(n - 1)]
        row = np.r_[a[0], np.zeros(n - 1)]
        A = scipy.linalg.toeplitz(col, row)  # dense but okay for n ≤ 1500

        b_var = cp.Variable(n, nonneg=True)
        constraints = [A @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception:  # pragma: no cover
            break
        if b_var.value is None:
            break
        b_vec = np.maximum(b_var.value, 0.0)
        s = b_vec.sum()
        if s <= 1e-12:
            break
        g = b_vec / s
        a = (1.0 - mix_t) * a + mix_t * g
        a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
def _run_full_refinement(seq: np.ndarray, deadline: float) -> np.ndarray:
    """Apply the whole multi‑phase refinement to one candidate."""
    n = len(seq)

    # 1) Adam optimisation (≈55 % of the remaining budget)
    t_mid = time.time() + 0.55 * (deadline - time.time())
    seq, _ = _adam_optimize(seq, t_mid)

    # 2) Exact ℓ∞ sub‑gradient descent (≈25 %)
    t_mid2 = time.time() + 0.25 * (deadline - time.time())
    seq = _refine_max(seq, t_mid2)

    # 3) Optional LP step (only for modest n)
    if n <= 1500 and _cvxpy_available:
        lp_dead = time.time() + 0.07 * (deadline - time.time())
        seq = _lp_improve(seq, max_iters=3, mix_t=0.5, inner_deadline=lp_dead)

    # 4) Simulated annealing (≈10 %)
    if time.time() < deadline:
        sa_dead = time.time() + 0.10 * (deadline - time.time())
        seq, _ = _simulated_annealing(seq, sa_dead)

    # 5) Final short sub‑gradient sweep (use whatever time left)
    if time.time() < deadline:
        seq = _refine_max(seq, deadline)

    return seq

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a low‑score step‑function.  Returns a plain Python list ready for
    ``evaluate_sequence``.
    """

    deadline = start_time + budget_s - 0.5   # safety margin

        base_n = len(base_seq)
    except Exception:
        base_seq = None
        base_n = 4000

    # ------------------------------------------------------------------
    # 1) Decide which lengths to investigate.
    scale_factors = [0.5, 0.75, 1.0, 1.25, 1.5, 2.0]
    lengths = sorted(
        set(
            max(500, int(base_n * f))
            for f in scale_factors
            if int(base_n * f) >= 500 and int(base_n * f) <= 8000
        )
    )
    if not lengths:
        lengths = [max(500, base_n)]

    global_best_seq = None
    global_best_proxy = float("inf")
    candidates = []  # (proxy_score, vector)

    print("[stage 0] generate initial candidates")
    # ------------------------------------------------------------------
    # For each length, generate a bunch of starters and keep the few best.
    for n in lengths:
        if time.time() >= deadline:
            break
        starters = {}

        starters["arcsine"] = _arcsine_distribution(n)
        starters["uniform"] = np.full(n, 1.0 / n, dtype=np.float64)

        # Dirichlet draws (different concentrations)
        for conc in (0.15, 0.3, 0.6, 1.0, 2.0):
            starters[f"dirichlet_{conc:.2f}"] = _random_dirichlet(n, conc)

        # Symmetric Beta family
        for a in (0.3, 0.4, 0.5, 0.6, 0.7):
            starters[f"beta_{a:.2f}"] = _beta_distribution(n, a, a)

        # Power‑law boundary family
        for p in (0.35, 0.40, 0.45, 0.50):
            starters[f"power_{p:.2f}"] = _powerlaw_boundary_distribution(n, p)

        # Mixtures of arcsine & uniform
        for w in (0.1, 0.3, 0.5, 0.7, 0.9):
            starters[f"mix_{w:.1f}"] = (1.0 - w) * _arcsine_distribution(n) + w * np.full(n,
                                                                                           1.0 / n)

        # Random jittered uniforms (helps escape symmetric basins)
        for _ in range(3):
            rnd = np.random.rand(n)
            rnd = _project_onto_simplex(rnd, s=1.0)
            starters[f"rnd_{np.random.randint(1e6)}"] = rnd

        if base_seq is not None:
            starters["prev"] = _interpolate_to_length(base_seq, n)

        # ------------------------------------------------------------------
        # Quick proxy: 2·n·max(conv). Keep the top 3 for this length.
        top_local = []
        for tag, vec in starters.items():
            vec = _project_onto_simplex(vec, s=1.0)
            maxconv = _fft_self_convolve(vec).max()
            proxy = 2.0 * n * maxconv
            if proxy < global_best_proxy:
                global_best_proxy = proxy
                global_best_seq = vec.copy()
                print(f"[init] n={n:4d} {tag:<12s} proxy={proxy:.6f}")
            # keep a few local bests
            top_local.append((proxy, vec.copy()))
        top_local.sort(key=lambda x: x[0])
        candidates.extend(top_local[:3])   # keep three per length

    # ------------------------------------------------------------------
    # If nothing was generated (should not happen) fall back to uniform.
    if not candidates:
        n = max(500, base_n)
        uni = np.full(n, 1.0 / n, dtype=np.float64)
        candidates = [(2.0 * n * _fft_self_convolve(uni).max(), uni)]

    # ------------------------------------------------------------------
    # 2) Sort all collected candidates and keep the global top‑K.
    candidates.sort(key=lambda x: x[0])
    K = min(6, len(candidates))
    top_candidates = [vec for _, vec in candidates[:K]]
    print(f"[stage 1] kept {K} promising starters for deep refinement")

    # ------------------------------------------------------------------
    # 3) Deep optimisation for each kept starter.
    for idx, vec in enumerate(top_candidates, 1):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # allocate a slice of the remaining budget (roughly equal for each)
        slice_deadline = time.time() + remaining * (1.0 / (K - idx + 1))
        print(f"[stage 2] refining candidate #{idx} (length={len(vec)})")
        refined = _run_full_refinement(vec, slice_deadline)
        # evaluate the true score
        try:
            true_score = evaluate_sequence(refined.tolist())
            print(f"          true_score={true_score:.6f}")
        except Exception as exc:
            print(f"          evaluate_sequence raised: {exc}")
            true_score = float("inf")
        if true_score < global_best_proxy:
            global_best_proxy = true_score
            global_best_seq = refined.copy()
            print(f"[update] new global best = {true_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final safety check – evaluate the best candidate once more.
    if global_best_seq is None:   # very unlikely
        raise RuntimeError("No candidate produced")
    try:
        final_score = evaluate_sequence(global_best_seq.tolist())
        print(f"[final] best score after full run = {final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence raised:", exc)
        final_score = None

    # Return the best sequence we have (as plain Python list)
    return global_best_seq.tolist()
