"""
Propose a step function using an alternating linear‑program / sub‑gradient scheme.

The algorithm:
1.  Start from a warm‑started shape (previous best if available,
    otherwise a uniform distribution) and normalise it to Σa = 1.
2.  Repeatedly
    • compute the current self‑convolution peak M,
    • solve the linear program  max Σ g_j  s.t.  (a∗g)_k ≤ M  (g ≥ 0),
      i.e. maximise the total mass that can be added without violating the
      current maximum autocorrelation,
    • mix the new vector g into a with a diminishing step‑size and renormalise.
    The LP is built as a sparse Toeplitz matrix; solving is done with
    SciPy's HiGHS interior‑point solver.
3.  After the LP phase a short Euclidean‑projected sub‑gradient descent
    (as used in the reference solution) refines the best point found.
4.  Return the best sequence (as a plain list) – the black‑box
    ``evaluate_sequence`` will compute the final score.

The method follows the classical “alternating LP” construction described in
the problem statement and usually improves the bound obtained by pure
gradient descent.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a∗a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a∗a."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of max_k (a∗a)_k.
    Returns (grad, cur_max), where grad has the same shape as a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    # active set – tolerance scales with the current maximum
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(conv >= cur_max - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad_full = 2.0 * _fft_convolve(a_rev, w)
    grad = grad_full[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


# ----------------------------------------------------------------------
def _lp_max_sum_g(f: np.ndarray, M: float):
    """
    Solve   max Σ g_j   s.t.   (f ∗ g)_k ≤ M  ∀k ,  g ≥ 0.
    Returns a feasible g (not normalised).
    """
    n = f.shape[0]
    # Build the convolution matrix A (size (2n‑1) × n) – each diagonal i
    # (i = 0 … n‑1) contains the constant f[i].
    offsets = -np.arange(n, dtype=int)                     # negative offsets
    data = [np.full(n, f[i]) for i in range(n)]           # each diagonal as a constant vector
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csc")

    c = -np.ones(n)                                        # minimise –Σ g  ⇔ maximise Σ g
    b = np.full(2 * n - 1, M)

    # HiGHS is fast for medium‑sized sparse LPs.
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=(0, None),
                      method="highs", options={"presolve": True})
    if not res.success:
        # In the unlikely event of failure fall back to the current f.
        return f.copy()
    g = np.maximum(res.x, 0.0)
    return g

        M = cur_max

        # solve the LP
        g_raw = _lp_max_sum_g(a, M)

        # normalise the new candidate
        sum_g = g_raw.sum()
        if sum_g > 0.0:
            g = g_raw / sum_g
        else:
            g = np.full_like(a, 1.0 / n)

        # diminishing mixing factor (larger steps early on)
        t = 0.5 / math.sqrt(it_lp)      # 0.5, 0.35, 0.28, …

        a = (1.0 - t) * a + t * g
        a = a / a.sum()                  # stay on the simplex

        raw = 2.0 * n * _fft_self_convolve_max(a)
        if raw < best_raw:
            best_raw = raw
            best_a = a.copy()
            print(f"[{time.time():.1f}] LP it {it_lp:4d} raw={raw:.6f}")

        # stop early if progress stalls
        if it_lp >= 30:          # generous cap – more would waste time
            break

    # ------------------------------------------------------------------
    # ---------- Sub‑gradient refinement (as in the reference) -------
    a = best_a.copy()
    sub_it = 0
    while time.time() < deadline and sub_it < 1_200_000:
        sub_it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.07 / math.sqrt(sub_it)
        a = a - step * grad
        a = _project_onto_simplex(a)
        raw = 2.0 * n * cur_max
        if raw < best_raw:
            best_raw = raw
            best_a = a.copy()
            print(f"[{time.time():.1f}] sub‑grad it {sub_it:7d} raw={raw:.6f}")

        # occasional tiny jitter to avoid stagnation
        if sub_it % 800 == 0 and np.random.rand() < 0.06:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a)

    # ------------------------------------------------------------------
    final_list = best_a.tolist()
    final_score = evaluate_sequence(final_list)          # provided by the harness
    print(f"\nFinal evaluated score: {final_score:.6f} (length={len(final_list)})")
    return final_list
