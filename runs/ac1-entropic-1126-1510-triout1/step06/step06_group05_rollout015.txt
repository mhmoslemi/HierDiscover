def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np, math
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1
    # candidate lengths list based on budget. Start with base lengths maybe 800,1200,... up to 10000.
    lengths = [800,1200,2000,3000,5000,8000]  # ensure >=500
    # candidate forms
    def init_vectors(n):
        # produce a dict of raw vectors before projection.
        d={}
        d['uniform'] = np.full(n, 1.0/n)
        d['arcsine'] = _arcsine_distribution(n)
        d['beta_0.5'] = _beta_distribution(n,0.5,0.5)
        d['beta_0.6'] = _beta_distribution(n,0.6,0.6)
        for p in (0.45,0.5,0.55):
            d[f'power_{p}'] = _powerlaw_boundary_distribution(n,p)
        # sin shape:
        i = np.arange(n)
        sin_vals = np.sin(np.pi*(i+0.5)/n)
        sin_vals = np.maximum(sin_vals,0)
        sin_vals /= sin_vals.sum()
        d['sin'] = sin_vals
        # exp decay
        exp_vals = np.exp(-i/(n/5))
        exp_vals /= exp_vals.sum()
        d['exp'] = exp_vals
        # random Dirichlet draws:

def subgradient_opt(a, time_limit):
    n=len(a)
    a = a.copy()
    best_a=a.copy()
    best_max = _fft_self_convolve(a).max()
    v = np.zeros_like(a)
    momentum=0.9
    it=0
    while time.time()<time_limit:
        it+=1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.1/np.sqrt(it)  # tune step
        v = momentum*v - step*grad
        a = a + v
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
        if it % 1000 ==0:
            # add random jitter
            a = best_a.copy()
            a += np.random.randn(n)*1e-6
            a = _project_onto_simplex(a, s=1.0)
    return best_a, best_max
```

"""
Propose a low‑score step‑function.

The routine follows four loosely coupled phases inside the allotted time
budget:

1. **Broad generation** – many families of probability vectors (arcsine,
   beta, power‑law, sinusoid, exponential, Dirichlet draws and random
   mixtures) are created for several lengths.  The cheap proxy
   ``2·n·max(conv)`` is evaluated and the best few candidates are kept.

2. **Adam optimisation** – each retained candidate is refined with Adam on
   a smooth‑max surrogate of the ℓ∞ objective.  The optimisation is
   limited by a per‑candidate deadline so that several restarts can be
   afforded.

3. **Long sub‑gradient descent** – the current best vector is then improved
   by an extended exact ℓ∞ sub‑gradient method (the same routine used in
   the reference implementation, but run for a much larger fraction of the
   remaining time).

4. **Simulated‑annealing polish** – a light SA pass performs stochastic
   mass‑exchange moves that accept downhill improvements and occasionally
   accept uphill moves, providing a final touch‑up.

All vectors are projected onto the probability simplex (non‑negative,
unit‑sum) after each update, guaranteeing that the evaluation function
is scale‑invariant.  The function returns a plain Python list ready for
``evaluate_sequence``.
"""

import math
import time
import numpy as np
from numpy.fft import rfft, irfft
import scipy.linalg

# optional LP support – not used in this version
try:
    import cvxpy as cp  # noqa: F401
    _cvxpy_available = True
except Exception:  # pragma: no cover
    _cvxpy_available = False


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_vals = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    vals[~np.isfinite(vals)] = 0.0
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


def _powerlaw_boundary_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5) + 1e-12) ** (-p)
    vals /= vals.sum()
    return vals


def _sinusoid_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-i / (n / 5.0))
    vals /= vals.sum()
    return vals

def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """Gradient of the smooth‑max surrogate Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)."""
    b = _fft_self_convolve(a)
    cur_max = b.max()
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.02,
                   alpha_max: float = 5000.0,
                   lr0: float = 0.4,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """Adam optimisation of the smooth‑max surrogate."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny jitter to escape plateaus
        if t % 400 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """One sub‑gradient step for the exact ℓ∞ objective."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Extended exact‑max refinement using sub‑gradient descent."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    # momentum variables – simple heavy‑ball style
    v = np.zeros_like(a)
    momentum = 0.85

    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)

        step = 0.03 / np.sqrt(it)          # a bit larger than the reference
        v = momentum * v - step * grad
        a = a + v
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        # occasional stochastic refresh to avoid stagnation
        if it % 1200 == 0 and np.random.rand() < 0.08:
            a = best_a + np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a

def _simulated_annealing(a0: np.ndarray, deadline: float):
    """Light simulated‑annealing polish."""
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_peak = _fft_self_convolve(cur_a).max()
    best_a = cur_a.copy()
    best_peak = cur_peak

    start = time.time()
    total_time = max(deadline - start, 1e-6)

    start_temp = max(cur_peak, 1e-3) * 0.5
    final_temp = start_temp * 1e-5
    if final_temp <= 0.0:
        final_temp = 1e-9
    cooling_rate = -math.log(final_temp / start_temp) / total_time

    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1
        src = np.random.randint(n)
        if cur_a[src] <= 1e-12:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue
        max_delta = min(cur_a[src], 0.07)               # allow slightly larger jumps
        delta = np.random.rand() * max_delta

        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_peak = _fft_self_convolve(a_new).max()
        delta_obj = new_peak - cur_peak

        elapsed = time.time() - start
        temperature = start_temp * math.exp(-cooling_rate * elapsed)

        accept = False
        if delta_obj < 0.0:
            accept = True
        else:
            prob = math.exp(-delta_obj / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_a = a_new
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_a = cur_a.copy()

        if iter_cnt % 25000 == 0 and np.random.rand() < 0.06:
            cur_a = best_a.copy()
            cur_peak = best_peak

    return best_a, best_peak

            )

        if base_seq is not None:
            starters["prev_interp"] = _project_onto_simplex(
                np.interp(
                    np.linspace(0, 1, n),
                    np.linspace(0, 1, len(base_seq)),
                    base_seq,
                ),
                s=1.0,
            )

        # cheap proxy evaluation
        for tag, vec in starters.items():
            vec = _project_onto_simplex(vec, s=1.0)
            peak = _fft_self_convolve(vec).max()
            fast_score = 2.0 * n * peak
            candidate_pool.append((fast_score, vec, n))
            if fast_score < 2.0 * n * (1.0 / n):  # just a tiny sanity filter
                print(f"   [init] n={n:4d} {tag:<14s} fast_score={fast_score:.6f}")

    # keep the best few (they will receive the heavy optimisation)
    candidate_pool.sort(key=lambda x: x[0])
    TOP_K = 5
    top_candidates = candidate_pool[:TOP_K]

    # --------------------------------------------------------------
    # 2️⃣ Adam optimisation on the survivors
    time_left = deadline - time.time()
    if time_left > 5.0 and top_candidates:
        per_candidate = max(0.5, time_left * 0.55 / len(top_candidates))
        print(f"[phase 2] Adam + short refinement – {time_left:.1f}s left")
        refined = []
        for score, a, n in top_candidates:
            if time.time() >= deadline:
                break
            phase_deadline = time.time() + per_candidate * 0.6
            a_opt, _ = _adam_optimize(a, phase_deadline)
            # a quick sub‑gradient pass to clean up the Adam output
            sub_deadline = time.time() + per_candidate * 0.3
            a_opt = _refine_max(a_opt, sub_deadline)
            new_score = 2.0 * n * _fft_self_convolve(a_opt).max()
            refined.append((new_score, a_opt, n))
            print(f"   [adam] n={n:4d} score={new_score:.6f}")
        refined.sort(key=lambda x: x[0])
        best_seq, best_n = refined[0][1], refined[0][2]
    else:
        # fallback to the absolute best from phase 1
        best_seq, best_n = top_candidates[0][1], top_candidates[0][2]

    # --------------------------------------------------------------
    # 3️⃣ Long sub‑gradient refinement (the main workhorse)
    time_left = deadline - time.time()
    if time_left > 3.0:
        print(f"[phase 3] extended sub‑gradient refinement – {time_left:.1f}s left")
        refine_deadline = time.time() + time_left * 0.85
        best_seq = _refine_max(best_seq, refine_deadline)

    # --------------------------------------------------------------
    # 4️⃣ Light simulated‑annealing polishing (if time permits)
    time_left = deadline - time.time()
    if time_left > 1.0:
        print(f"[phase 4] final simulated annealing – {time_left:.1f}s left")
        best_seq, _ = _simulated_annealing(best_seq, deadline)

    # --------------------------------------------------------------
    # 5️⃣ Exact evaluation
    try:
        true_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence score={true_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)
        true_score = float("inf")

    print(f"Best score (approx.): {true_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
