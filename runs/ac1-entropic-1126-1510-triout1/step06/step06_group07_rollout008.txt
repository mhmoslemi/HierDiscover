"""
Propose a low‑score step‑function.

Algorithm overview
-------------------
1. **Parameter sweep** – generate a set of simple analytic shapes
   (uniform, exponentials, symmetric exponentials, power‑law, etc.),
   normalize each to the required ℓ¹‑norm and keep the best one.

2. **Smoothed max optimisation** – minimise a μ‑smoothed surrogate of
   the hard max‑objective using L‑BFGS‑B.  Several random restarts and a
   slow geometric cooling schedule (μ←0.85·μ) are employed.

3. **Simulated annealing** – starting from the best vector so far,
   repeatedly apply cheap stochastic moves (swap, δ‑mass transfer,
   block averaging).  Moves are accepted according to a temperature
   schedule, allowing uphill steps early on.

4. **Permutation refinement** – the multiset of heights is fixed;
   a hill‑climbing on swaps is performed to find a better ordering
   (large values are spread apart).  The best ordering replaces the
   incumbent if it improves the true score.

All helpers are top‑level functions; no closures or lambdas are used.
The routine respects a user‑supplied time budget (default 1000 s) and
returns the best height sequence found.
"""

import time

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector (sum = 1)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _compute_smooth_obj_and_grad(y: np.ndarray, mu: float,
                                 n: int, S: float) -> tuple[float, np.ndarray]:
    """
    Value and gradient of the smoothed max‑objective

        f_μ(y) = max(conv) + μ·log Σ exp((conv‑max)/μ)

    Parameters
    ----------
    y : ndarray (n,)
        Unconstrained parameters (log‑probabilities on the simplex).
    mu : float > 0 ( μ = 0 ⇒ hard max, handled separately )
    n, S : problem size and target ℓ¹‑norm.

    Returns
    -------
    (f_val, grad_y) – smooth‑max value and gradient w.r.t. y.
    """
    # probabilities on the simplex, actual heights
    p = _softmax(y)          # Σ p_i = 1, p_i ≥ 0
    a = S * p                # Σ a_i = S

    conv = np.convolve(a, a)        # length 2·n‑1

    max_c = conv.max()
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # prevent overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        if cnt > 0:
            w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. a (identical for smoothed / hard case)
    conv_wr = np.convolve(w, a[::-1])          # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]       # slice length n

    inner = np.dot(g_a, p)                     # scalar
    grad_y = S * p * (g_a - inner)             # chain rule of soft‑max

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable object for scipy.optimize: returns (value, gradient)."""

    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_smooth_obj_and_grad(y, self.mu, self.n, self.S)

def _max_convolution(a: np.ndarray) -> float:
    """Return max(conv(a,a)).  Faster than evaluate_sequence for internal use."""
    return np.convolve(a, a).max()


def _evaluate(a: np.ndarray) -> float:
    """Exact evaluation using the provided evaluate_sequence."""
    # The competition supplies `evaluate_sequence` in the global namespace.
    return evaluate_sequence(a.tolist())
