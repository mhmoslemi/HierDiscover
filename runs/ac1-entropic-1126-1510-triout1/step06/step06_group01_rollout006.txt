    # hyperparameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

   for each schedule:
       for step in range(steps):
           if time.time() >= deadline: return a,best_score
           t += 1
           b = _fft_self_convolve(a)
           maxb = b.max()
           # compute softmax weights

           w = exp / exp.sum()

"""
Propose a non‑negative step‑function (probability vector) that
minimises the evaluation function:

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm works in several stages:

1. **Rich initialisation** – a wide family of analytic shapes
   (uniform, arcsine, symmetric beta, exponential, alternating,
   block‑spike, sparse random) is generated for several lengths.
   The best of those is kept as a seed.

2. **Smoothed Adam descent** – minimise a smooth approximation of the
   max‑convolution using the log‑sum‑exp trick and Adam optimisation.
   The smoothing parameter α is increased progressively (1 → 5 → 20 →
   100 → 500 → 2000) and many optimisation steps are taken.  The
   simplex constraint (non‑negative, sum = 1) is enforced by Euclidean
   projection after each update.

3. **Exact sub‑gradient descent** – after the continuation phase the
   true sub‑gradient of the max‑convolution is used (active set of
   indices attaining the maximum).  A momentum term and a diminishing
   stepsize accelerate convergence.

4. **Greedy active‑set swaps** – using the current active set we compute
   the sub‑gradient g.  Mass is repeatedly moved from the index with
   largest g (where the max would increase most) to the index with
   smallest g (where it would increase least).  Each swap is kept only
   if it improves the objective.

5. **Simulated‑annealing fine‑tuning** – a cheap Metropolis‑type random
   walk (mass transfers between random positions) with an exponential
   cooling schedule helps escape shallow plateaux that the deterministic
   steps miss.

All stages respect a global time budget (default 1000 s).  The best
vector found is returned as a plain Python list of floats.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    # symmetric beta with shape parameter p (p>0)
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    """Uniform mass on every `stride`‑th coordinate (starting at 0)."""
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    """Uniform mass on the first and last `block_frac` fraction of indices."""
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a

def _smooth_adam_descent(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp smooth approximation of max‑convolution."""
    n = a.shape[0]

    # hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # increasing smoothness schedule
    schedule = [
        (1.0, 2000),
        (5.0, 2000),
        (20.0, 2000),
        (100.0, 2000),
        (500.0, 2000),
        (2000.0, 2000),
    ]

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score

            t += 1
            b = _fft_self_convolve(a)
            maxb = b.max()
            # soft‑max weights
            w = np.exp(alpha * (b - maxb))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(b, 1.0 / b.size)
            else:
                w /= w_sum

            # gradient of the smooth max (size n)
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam update + simplex projection
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a, 1.0)

            # occasional exact evaluation
            if t % 150 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-13:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] smooth α={alpha:.0f} step={t} new best {best_score:.6f}")

    return best_seq, best_score

def _exact_subgradient_descent(a: np.ndarray, deadline: float,
                               best_seq: np.ndarray,
                               best_score: float) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the genuine max‑convolution objective."""
    n = a.shape[0]
    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.30
    it = 0

    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = (b >= cur_max - eps).astype(float)

        # sub‑gradient from active set
        conv_rev = _fft_convolve(a[::-1], active)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # momentum + diminishing stepsize
        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)

        cur = _objective(a)
        if cur < best_score - 1e-13:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {best_score:.6f}")

        # early exit if we have already crossed the target
        if best_score <= 1.503:
            break

        # cheap time check
        if time.time() + 0.5 >= deadline:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 3 – greedy active‑set swapping ---------------------------------
def _greedy_swap_opt(a: np.ndarray, deadline: float,
                     best_seq: np.ndarray,
                     best_score: float) -> tuple[np.ndarray, float]:
    """Move mass from high‑gradient to low‑gradient indices while improving."""
    n = a.shape[0]
    while time.time() < deadline:
        b = _fft_self_convolve(a)
        maxb = b.max()
        eps = maxb * 1e-12 + 1e-12
        active = (b >= maxb - eps).astype(float)

        # sub‑gradient for the active set
        conv_rev = _fft_convolve(a[::-1], active)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # source: index with large gradient that actually carries mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = src_candidates[np.argmax(grad[src_candidates])]

        # destination: index with smallest gradient (could be zero)
        j = int(np.argmin(grad))
        if i == j:
            break

        # amount to move – a fraction of the source mass, not too large
        delta = min(a[i] * 0.5, 0.005)
        if delta <= 0:
            break

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        cur = _objective(a_new)

        if cur < best_score - 1e-13:
            a = a_new
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] greedy swap improved best {best_score:.6f}")
        else:
            # try a smaller transfer, otherwise give up this round
            delta = delta * 0.25
            if delta < 1e-7:
                break
            a_new[i] = a[i] - delta
            a_new[j] = a[j] + delta
            cur = _objective(a_new)
            if cur < best_score - 1e-13:
                a = a_new
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] tiny swap improved best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 4 – simulated annealing -----------------------------------------
def _simulated_annealing(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Metropolis‑type random walk with exponential cooling."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = _objective(cur_a)

    T = 0.04                     # initial temperature (in objective units)
    iter_cnt = 0

    while time.time() < deadline and T > 1e-4:
        iter_cnt += 1

        i = np.random.randint(n)
        if cur_a[i] <= 1e-12:
            continue
        # pick a different destination
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # random transfer, up to 10 % of source mass
        delta = np.random.rand() * min(cur_a[i], 0.1 * cur_a[i])
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = _objective(new_a)

        dE = new_score - cur_score
        if dE < 0 or np.random.rand() < math.exp(-dE / T):
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score - 1e-13:
                best_score = new_score
                best_seq = cur_a.copy()
                print(f"[{time.time():.1f}] SA iter={iter_cnt} new best {best_score:.6f}")

        # exponential cooling
        if iter_cnt % 2500 == 0:
            T *= 0.85

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best vector found (as a plain Python list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a set of lengths (more lengths give more opportunities).
    lengths = [800, 1200, 2000, 3000, 5000, 8000]

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 2. Scan analytic families for each length.
    for n in lengths:
        if time.time() >= deadline:
            break
        for cand in _initial_candidates_extended(n):
            if time.time() >= deadline:
                break
            cand = cand / cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] init n={n} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3. If a previous solution exists, reuse it as a seed.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Fallback – should never fire, but keep a safe default.
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Main optimisation pipeline (all respect the global deadline).
    a = overall_best_seq.copy()

    # 5a – smoothed Adam descent
    a, overall_best_score = _smooth_adam_descent(a, deadline,
                                                 best_seq=a,
                                                 best_score=overall_best_score)

    # 5b – exact sub‑gradient refinement
    a, overall_best_score = _exact_subgradient_descent(a, deadline,
                                                       best_seq=a,
                                                       best_score=overall_best_score)

    # 5c – greedy active‑set swapping
    a, overall_best_score = _greedy_swap_opt(a, deadline,
                                             best_seq=a,
                                             best_score=overall_best_score)

    # 5d – simulated annealing fine‑tuning
    a, overall_best_score = _simulated_annealing(a, deadline,
                                                 best_seq=a,
                                                 best_score=overall_best_score)

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(a)})")
    return a.tolist()
