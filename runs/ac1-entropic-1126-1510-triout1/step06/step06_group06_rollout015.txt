    cur_max = b.max()
    best_a = a.copy()

def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0

    # Attempt to load prior best sequence if exists
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore
        base_seq = _project_onto_simplex(base_seq, s=1.0)
        base_n = len(base_seq)
    except Exception:
        base_seq = None
        base_n = 4000

    # define lengths to explore
    lengths = [max(500, int(base_n * f)) for f in (0.6, 0.8, 1.0, 1.2, 1.4) if int(base_n * f) >= 500]
    if not lengths:
        lengths = [max(500, base_n)]

    overall_best_seq = None
    overall_best_score = float('inf')
    # Phase 0: families
    print("[phase 0] sampling families")
    for n in lengths:
        if time.time() >= deadline: break
        # generate families
        factory_map = {}
        # arcsine:
        arcsine = 1.0 / np.sqrt((np.arange(n) + 0.5) * (n - np.arange(n) - 0.5))
        arcsine = arcsine / arcsine.sum()
        factory_map["arcsine"] = arcsine

        factory_map["uniform"] = np.full(n, 1.0/n)

        factory_map["random1"] = np.random.rand(n)
        factory_map["random1"] /= factory_map["random1"].sum()
        factory_map["random2"] = np.random.rand(n)
        factory_map["random2"] /= factory_map["random2"].sum()

        # Beta symmetric
        for a in (0.25, 0.35, 0.45, 0.55, 0.65, 0.75):
            x = (np.arange(n, dtype=np.float64) + 0.5) / n
            log_vals = (a-1.0)*np.log(x) + (a-1.0)*np.log(1.0 - x)
            vals = np.exp(log_vals)
            vals[~np.isfinite(vals)] = 0.0
            if vals.sum() == 0.0:
                vals = np.full(n, 1.0/n)
            else:
                vals /= vals.sum()
            factory_map[f"beta_{a:.2f}"] = vals

        # exponential
        for lam in (0.0005, 0.001, 0.002):
            vec = np.exp(-lam * np.arange(n))
            vec /= vec.sum()
            factory_map[f"exp_{lam:.4f}"] = vec

        # mix arcsine + uniform
        for w in (0.2, 0.4, 0.6):
            mix = (1.0 - w) * arcsine + w * factory_map["uniform"]
            factory_map[f"mix_{w:.1f}"] = mix

        if base_seq is not None:
            # interpolate base_seq to length n
            old_n = len(base_seq)
            x_old = np.linspace(0.0, 1.0, old_n)
            x_new = np.linspace(0.0, 1.0, n)
            interp = np.interp(x_new, x_old, base_seq)
            interp = np.maximum(interp, 0.0)
            interp /= interp.sum()
            factory_map["prev"] = interp

        for tag, vec in factory_map.items():
            vec = _project_onto_simplex(vec, s=1.0)
            peak = _fft_self_convolve(vec).max()
            fast_score = 2.0 * n * peak

"""
Propose a low‑score step‑function.

The search consists of several stages:

0.  Broad sampling of simple families (arcsine, uniform, beta,
    exponential, random mixes) for a few lengths.  The best fast
    score (2·n·max(conv)) is retained.

1.  Exhaustive two‑level piecewise‑constant family (front block / back
    block) with a logarithmic grid of height ratios.

2.  Adam optimisation of the smooth‑max surrogate of the ℓ∞
    autocorrelation.  The schedule for the smoothness parameter α
    starts at 1.0 and is multiplied by 1.01 each step up to 5000.
    The best exact peak seen is kept.

3.  Exact sub‑gradient descent on the true max‑convolution objective.

4.  An improved greedy local search that moves a small amount of mass
    from a source index (chosen proportionally to its contribution to
    the current peak) to a destination that most reduces the peak.
    The update of the full convolution is performed analytically,
    avoiding repeated FFTs.

5.  (optional) a short final Adam run if time permits.

All vectors stay on the probability simplex (non‑negative, sum = 1)
by Euclidean projection after every update.  The routine respects a
user‑provided time budget (default 1000 s) and returns the best
sequence found as a plain Python list.
"""

import time

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth‑max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    exp_term = np.exp(alpha * (b - cur_max))          # stable soft‑max
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)                # length = 3n‑2
    n = a.shape[0]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.01,
                   alpha_max: float = 5000.0,
                   lr0: float = 0.3,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best probability vector witnessed and its exact peak.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # tiny random jitter to help escape plateaus
        if t % 400 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    One exact sub‑gradient step for the ℓ∞‑objective.
    Returns (grad, cur_max); the gradient points towards increasing the
    current peak, so the descent direction is -grad.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Diminishing‑step sub‑gradient descent on the exact max‑convolution.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.02 / np.sqrt(it)          # diminishing step length
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


# ----------------------------------------------------------------------
def _two_level_family(n: int, deadline: float):
    """
    Search a simple two‑level family:
        a_i = p   for i < L   (front block)
        a_i = q   otherwise   (back block)
    with ratio r = p/q ≥ 1.
    Returns the best vector found and its fast score.
    """
    best_seq = None
    best_score = float('inf')
    step_L = max(1, n // 100)          # roughly 1 % of n
    r_grid = np.logspace(0.0, 2.3, num=25)   # from 1 to ≈200

    for L in range(step_L, n, step_L):
        if time.time() > deadline:
            break
        denom_base = n - L
        for r in r_grid:
            denom = L * r + denom_base
            q = 1.0 / denom
            p = r * q
            seq = np.empty(n, dtype=np.float64)
            seq[:L] = p
            seq[L:] = q
            peak = _fft_self_convolve(seq).max()
            score = 2.0 * n * peak
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
    return best_seq, best_score


# ----------------------------------------------------------------------
def _directed_local_search_improved(a0: np.ndarray, deadline: float):
    """
    Greedy mass‑transfer search that always attacks the current peak.
    For a chosen source index (probability proportional to its
    contribution to the peak) a small amount of mass is moved toward a
    destination that yields the largest reduction of the peak.
    The convolution update is performed analytically (O(n) per
    candidate) instead of recomputing a full FFT each time.
    Only improving moves are accepted.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max
    it = 0

    while time.time() < deadline:
        it += 1

        # --- select a convolution index that attains the peak ---
        active_idx = np.where(b >= cur_max - 1e-12)[0]
        if active_idx.size == 0:
            break
        k = np.random.choice(active_idx)

        # --- choose a source index i that contributes heavily to (a*a)[k] ---
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        i_vals = np.arange(i_low, i_high + 1, dtype=int)
        prod = a[i_vals] * a[k - i_vals]
        if prod.sum() == 0.0:
            src = np.random.randint(n)
        else:
            prob = prod / prod.sum()
            src = np.random.choice(i_vals, p=prob)

        # amount we are allowed to move (up to 5 % of the source mass)
        max_delta = min(a[src], 0.05)
        if max_delta <= 0.0:
            continue

        # pre‑compute the shifted version of a for the source
        src_shift = np.zeros(2 * n - 1, dtype=a.dtype)
        src_shift[src : src + n] = a

        # sample a few destination candidates (excluding src)
        dst_candidates = np.random.choice(
            np.delete(np.arange(n), src), size=min(5, n - 1), replace=False
        )

        # try a handful of delta values for each destination
        deltas = (max_delta, max_delta * 0.5, max_delta * 0.2)
        improved = False

        for dst in dst_candidates:
            # shifted version for destination
            dst_shift = np.zeros(2 * n - 1, dtype=a.dtype)
            dst_shift[dst : dst + n] = a

            for delta in deltas:
                # linear part of the update
                new_b = b + 2.0 * delta * (dst_shift - src_shift)

                # quadratic corrections for the three special indices
                idx2src = 2 * src
                idx2dst = 2 * dst
                idxsum = src + dst
                if idx2src < new_b.size:
                    new_b[idx2src] += delta * delta
                if idx2dst < new_b.size:
                    new_b[idx2dst] += delta * delta
                if idxsum < new_b.size:
                    new_b[idxsum] -= 2.0 * delta * delta

                new_max = new_b.max()
                if new_max < cur_max - 1e-15:
                    # accept move
                    a[src] -= delta
                    a[dst] += delta
                    b = new_b
                    cur_max = new_max
                    if cur_max < best_max:
                        best_max = cur_max
                        best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # occasional random jitter to avoid stagnation
        if it % 200 == 0 and np.random.rand() < 0.02:
            a += np.random.randn(n) * 1e-7
            a = _project_onto_simplex(a, s=1.0)
            b = _fft_self_convolve(a)
            cur_max = b.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

    return best_a

        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print(f"[phase 2] after Adam fast_score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – exact sub‑gradient descent followed by improved local search
    time_left = deadline - time.time()
    if time_left > 4.0:
        n = len(overall_best_seq)
        print(f"[phase 3] sub‑gradient descent (n={n}) – time left {time_left:.1f}s")
        sub_end = time.time() + time_left * 0.45
        a_refined = _refine_max(overall_best_seq, sub_end)

        # Greedy directed local search with analytic updates
        local_end = deadline
        a_final = _directed_local_search_improved(a_refined, local_end)

        overall_best_seq = a_final.copy()
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print(f"[phase 3] after refined local fast_score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # Optional tiny Adam run if any time is still left (helps smoothing)
    time_left = deadline - time.time()
    if time_left > 2.0:
        n = len(overall_best_seq)
        print(f"[phase 4] final short Adam run – time left {time_left:.1f}s")
        a_opt, _ = _adam_optimize(overall_best_seq, deadline)
        overall_best_seq = a_opt.copy()
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print(f"[phase 4] after final Adam fast_score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # Final black‑box evaluation (actual score)
    try:
        true_score = evaluate_sequence(overall_best_seq.tolist())
        print(f"[final] evaluate_sequence score={true_score:.6f}")
        if true_score < overall_best_score:
            overall_best_score = true_score
    except Exception as exc:
        print("[final] evaluate_sequence raised:", exc)

    # Store the best sequence globally for possible reuse in a later run
    global height_sequence_1   # type: ignore
    height_sequence_1 = overall_best_seq.tolist()   # type: ignore

    print(f"Best approx. score {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
