def _lp_refine_iterative(a, deadline):
    best_a = a.copy()
    best_M = _fft_self_convolve(a).max()
    while time.time() < deadline:
        n = a.shape[0]
        M = best_M
        # Build Toeplitz matrix A
        col = np.r_[a, np.zeros(n-1)]
        row = np.r_[a[0], np.zeros(n-1)]
        A = scipy.linalg.toeplitz(col, row)
        # Solve LP using cvxpy
        g_var = cp.Variable(n, nonneg=True)
        constraints = [A @ g_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception:
            break
        if g_var.value is None:
            break
        g = np.maximum(g_var.value, 0.0)
        if g.sum() == 0:
            break
        g = g / g.sum()
        # Mix
        t = 0.5
        improved = False
        for _ in range(5):
            new_a = (1-t)*a + t*g
            # optional small projection for numerical errors
            new_a = _project_onto_simplex(new_a, s=1.0)
            new_M = _fft_self_convolve(new_a).max()
            if new_M < best_M - 1e-12:
                a = new_a
                best_M = new_M
                best_a = new_a.copy()
                improved = True
                print(f"[LP] iter improved M to {new_M:.6f}")
                break
            t *= 0.5
        if not improved:
            print("[LP] no improvement, break")
            break
        # continue loop
    return best_a

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_vals = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    vals[~np.isfinite(vals)] = 0.0
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


def _powerlaw_boundary_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5) + 1e-12) ** (-p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth‑max surrogate
    Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k).
    Returns (grad, current_max).
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # stability shift
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


def _adam_optimize(a0: np.ndarray, deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.01,
                   alpha_max: float = 5000.0,
                   lr0: float = 0.25,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-9):
    """Adam optimisation of the smooth‑max surrogate."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny jitter to avoid stagnation
        if t % 1000 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """One sub‑gradient step for the exact ℓ∞ objective."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Short exact‑max refinement using sub‑gradient descent."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.02 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        if it % 400 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, deadline: float):
    """Light simulated‑annealing polish."""
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_peak = _fft_self_convolve(cur_a).max()
    best_a = cur_a.copy()
    best_peak = cur_peak

    start = time.time()
    total_time = max(deadline - start, 1e-6)

    start_temp = max(cur_peak, 1e-3) * 0.5
    final_temp = start_temp * 1e-5
    if final_temp <= 0.0:
        final_temp = 1e-9
    cooling_rate = -math.log(final_temp / start_temp) / total_time

    while time.time() < deadline:
        src = np.random.randint(n)
        if cur_a[src] <= 1e-12:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue
        max_delta = min(cur_a[src], 0.05)
        delta = np.random.rand() * max_delta

        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_peak = _fft_self_convolve(a_new).max()
        delta_obj = new_peak - cur_peak

        elapsed = time.time() - start
        temperature = start_temp * math.exp(-cooling_rate * elapsed)

        if delta_obj < 0.0 or np.random.rand() < math.exp(-delta_obj / max(temperature, 1e-12)):
            cur_a = a_new
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_a = cur_a.copy()

    return best_a, best_peak


# ----------------------------------------------------------------------
def _lp_iterative_refine(a: np.ndarray, deadline: float,
                        max_outer_iters: int = 20,
                        mix_factor: float = 0.5) -> np.ndarray:
    """
    Repeated LP improvement (dual‑widening) as described in the
    statement.  Stops when no improvement is seen or time expires.
    """
    if not _cvxpy_available:
        return a

    n = a.shape[0]
    best_a = a.copy()
    best_M = _fft_self_convolve(best_a).max()

    outer = 0
    while time.time() < deadline and outer < max_outer_iters:
        outer += 1
        M = best_M

        # Build convolution matrix A (2n‑1)×n : A @ g = f * g
        col = np.r_[best_a, np.zeros(n - 1)]
        row = np.r_[best_a[0], np.zeros(n - 1)]
        A = scipy.linalg.toeplitz(col, row)          # dense, shape (2n‑1, n)

        # Linear program: maximise Σ g_i  subject to A g ≤ M , g ≥ 0
        g_var = cp.Variable(n, nonneg=True)
        constraints = [A @ g_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, abstol=1e-9, reltol=1e-9)
        except Exception:                               # pragma: no cover
            break
        if g_var.value is None:
            break
        g = np.maximum(g_var.value, 0.0)
        s = g.sum()
        if s <= 1e-12:
            break
        g = g / s                                   # enforce Σ g = 1

        # Back‑track line search on mixing factor
        t = mix_factor
        improved = False
        for _ in range(5):
            new_a = (1.0 - t) * best_a + t * g
            new_a = _project_onto_simplex(new_a, s=1.0)
            new_M = _fft_self_convolve(new_a).max()
            if new_M < best_M - 1e-12:
                best_a = new_a
                best_M = new_M
                improved = True
                print(f"[LP] iter {outer} accepted t={t:.3f}  M={best_M:.6f}")
                break
            t *= 0.5
        if not improved:
            # no progress – stop the outer loop
            print(f"[LP] iter {outer} no improvement, terminating.")
            break
    return best_a


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a non‑negative sequence to a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a step‑function with a low score.
    Returns a plain Python list ready for `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose a working length (large enough for a good constant, small enough
    # for the LP to be tractable).  We adapt to the remaining time.
    # 2000 is a sweet spot for most budgets.
    n = 2000
    # if we have a lot of time we can afford a bigger n
    if budget_s > 300:
        n = 3000
    if budget_s > 600:
        n = 4000

    print(f"[init] chosen length n = {n}")

    # ------------------------------------------------------------------
    # 1.  Build a diverse collection of starter vectors.
    starters = {}

    starters["arcsine"] = _arcsine_distribution(n)
    starters["uniform"] = np.full(n, 1.0 / n, dtype=np.float64)

    for conc in (0.2, 0.4, 0.8, 1.2):
        starters[f"dirichlet_{conc:.1f}"] = np.random.dirichlet(np.full(n, conc))

    for a in (0.3, 0.4, 0.5, 0.6, 0.7):
        starters[f"beta_{a:.2f}"] = _beta_distribution(n, a, a)

    for p in (0.35, 0.45, 0.5, 0.55, 0.65):
        starters[f"power_{p:.2f}"] = _powerlaw_boundary_distribution(n, p)

    for w in (0.2, 0.4, 0.6, 0.8):
        starters[f"mix_{w:.1f}"] = (1.0 - w) * _arcsine_distribution(n) + w * np.full(n, 1.0 / n)

    # try to reuse a previous construction if available
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        base_seq = _project_onto_simplex(base_seq, s=1.0)
        starters["prev"] = _interpolate_to_length(base_seq, n)
    except Exception:                                            # pragma: no cover
        pass

    # ------------------------------------------------------------------
    # 2. Pick the best starter using the cheap bound 2·n·max(conv).
    best_seq = None
    best_fast_score = float("inf")
    for tag, vec in starters.items():
        vec = _project_onto_simplex(vec, s=1.0)
        peak = _fft_self_convolve(vec).max()
        fast_score = 2.0 * n * peak
        if fast_score < best_fast_score:
            best_fast_score = fast_score
            best_seq = vec.copy()
            print(f"[seed] {tag:<12s} fast_score={fast_score:.6f}")

    if best_seq is None:                                      # pragma: no cover
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_fast_score = 2.0 * n * _fft_self_convolve(best_seq).max()
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # 3. Adam optimisation (smooth‑max) – allocate roughly half of remaining time.
    time_left = deadline - time.time()
    if time_left > 5.0:
        phase_deadline = time.time() + time_left * 0.55
        print(f"[phase 1] Adam optimisation – {time_left:.1f}s budget")
        best_seq, _ = _adam_optimize(best_seq, phase_deadline)

    # ------------------------------------------------------------------
    # 4. Exact max sub‑gradient refinement – short budget.
    time_left = deadline - time.time()
    if time_left > 2.0:
        phase_deadline = time.time() + time_left * 0.30
        print(f"[phase 2] Sub‑gradient refinement – {time_left:.1f}s budget")
        best_seq = _refine_max(best_seq, phase_deadline)

    # ------------------------------------------------------------------
    # 5. Iterated LP improvement – the core of the new contribution.
    time_left = deadline - time.time()
    if time_left > 4.0 and _cvxpy_available:
        lp_deadline = time.time() + time_left * 0.70
        print(f"[phase 3] Iterated LP improvement – {time_left:.1f}s budget")
        best_seq = _lp_iterative_refine(best_seq, lp_deadline,
                                        max_outer_iters=12,
                                        mix_factor=0.5)

    # ------------------------------------------------------------------
    # 6. Simulated‑annealing polish – use whatever time is left.
    time_left = deadline - time.time()
    if time_left > 1.0:
        print(f"[phase 4] Simulated annealing – {time_left:.1f}s left")
        best_seq, final_peak = _simulated_annealing(best_seq, deadline)
        best_fast_score = 2.0 * n * final_peak
    else:
        best_fast_score = 2.0 * n * _fft_self_convolve(best_seq).max()

    # ------------------------------------------------------------------
    # 7. Final exact evaluation.
    try:
        true_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence score = {true_score:.6f}")
        best_fast_score = min(best_fast_score, true_score)
    except Exception as exc:                               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    print(f"Best (approx.) score = {best_fast_score:.6f}  (len={len(best_seq)})")
    return best_seq.tolist()
