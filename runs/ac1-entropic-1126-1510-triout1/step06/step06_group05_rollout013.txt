def _search_at_length(n, deadline, base_seq=None):
    # create starters dictionary
    starters = {}
    starters["arcsine"] = _arcsine_distribution(n)
    # uniform
    starters["uniform"] = np.full(n, 1.0/n)
    # Dirichlet draws
    for conc in (0.2, 0.5, 1.0, 2.0):
        starters[f"dirichlet_{conc}"] = np.random.dirichlet(np.full(n, conc))
    # Beta
    for a in np.linspace(0.3, 0.8, 6):
        starters[f"beta_{a:.2f}"] = _beta_distribution(n, a, a)
    # Powerlaw
    for p in np.linspace(0.4, 1.0, 7):
        starters[f"power_{p:.2f}"] = _powerlaw_boundary_distribution(n, p)
    # Mixtures arcsine+uniform
    for w in (0.2, 0.5, 0.8):
        starters[f"mix_{w:.1f}"] = (1-w)*_arcsine_distribution(n) + w*np.full(n,1.0/n)
    if base_seq is not None:
        starters["prev"] = _interpolate_to_length(base_seq, n)

    # Evaluate cheap
    best_vec = None
    best_fast = float('inf')
    for tag, vec in starters.items():
        vec = _project_onto_simplex(vec, s=1.0)
        peak = _fft_self_convolve(vec).max()
        fast = 2.0 * n * peak
        if fast < best_fast:
            best_fast = fast
            best_vec = vec.copy()
    a = best_vec

    # allocate phases
    now = time.time()
    if now >= deadline:
        return a, _fft_self_convolve(a).max()
    time_left = deadline - now
    # Adam
    adam_dead = now + time_left * 0.5
    a_opt, _ = _adam_optimize(a, adam_dead, init_alpha=1.0, alpha_max=4000.0)
    # subgrad
    now = time.time()
    if now >= deadline:
        return a_opt, _fft_self_convolve(a_opt).max()
    time_left = deadline - now
    sub_dead = now + time_left * 0.3
    a_opt = _refine_max(a_opt, sub_dead)

    # LP if small n and cvxpy
    now = time.time()
    if now < deadline and len(a_opt) <= 2500 and _cvxpy_available:

        lp_dead = now + time_left * 0.6

    now = time.time()
    if now < deadline:
        a_opt, final_peak = _simulated_annealing(a_opt, deadline)
        return a_opt, final_peak
    else:
        return a_opt, _fft_self_convolve(a_opt).max()

def _lp_improve(a, max_iters=10, mix_t=0.3, inner_deadline=None):
    if not _cvxpy_available:
        return a
    n = len(a)
    for i in range(max_iters):
        if inner_deadline is not None and time.time() > inner_deadline:
            break
        M = _fft_self_convolve(a).max()
        # Build convolution matrix A
        col = np.r_[a, np.zeros(n - 1)]
        row = np.r_[a[0], np.zeros(n - 1)]
        A = scipy.linalg.toeplitz(col, row)
        b_var = cp.Variable(n, nonneg=True)
        constraints = [A @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
        if b_var.value is None:
            break
        b_vec = np.maximum(b_var.value, 0.0)
        s = b_vec.sum()
        if s <= 1e-12:
            break
        b_norm = b_vec / s
        a = (1 - mix_t) * a + mix_t * b_norm
        a = _project_onto_simplex(a, s=1.0)
        # Optionally reduce mix_t gradually
        mix_t = max(0.05, mix_t * 0.9)
    return a
```

"""
Enhanced step‑function search.

Goal: minimise 2·n·max(conv(a,a))  (equivalently the exact
evaluate_sequence score, because the objective is scale‑invariant).

Strategy
--------
1.  Initialise a collection of candidate probability vectors.
    Families: arcsine, uniform, Dirichlet, symmetric Beta, power‑law
    on the boundary and simple mixtures.  The best candidate by the cheap
    proxy 2·n·max(conv) is kept.

2.  Adam optimisation of a smooth‑max surrogate Lα(a)= (1/α)·log
    Σ exp(α·(a*a)k).  α is increased gradually (up to 4000) so that the
    surrogate approaches the true max.

3.  Exact ℓ∞ sub‑gradient descent – a short but aggressive descent
    using the true max‑convolution gradient.

4.  LP‑based “dual‑widening” (optional, only for moderate n).  Repeatedly
    solve
        maximise Σ bj   subject to   (a∗b)k ≤ M  (M = max(a∗a))
    and mix the normalised solution back into a with a small convex step.
    Up to ten iterations are performed, with a decreasing mixing factor.

5.  Light simulated‑annealing polishing.

6.  Final ℓ∞ sub‑gradient refinement using any remaining time.

All steps respect the simplex Σ aj = 1, aj ≥ 0; the objective is
scale‑invariant, therefore the simplex projection does not affect the
final score.  The routine runs until the supplied deadline and returns
the best sequence found.  The code works without any external files and
uses only NumPy, SciPy and (if available) CVXPY.
"""

import math
import time
import numpy as np
from numpy.fft import rfft, irfft
import scipy.linalg

# optional LP support --------------------------------------------------------
try:
    import cvxpy as cp
    _cvxpy_available = True
except Exception:   # pragma: no cover
    _cvxpy_available = False


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length `n`."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric Beta(α,β) on a uniform grid, normalised to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_vals = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    vals[~np.isfinite(vals)] = 0.0
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _powerlaw_boundary_distribution(n: int, p: float) -> np.ndarray:
    """Distribution ∝ ((i+0.5)*(n‑i‑0.5))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5) + 1e-12) ** (-p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a non‑negative sequence to a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """Gradient of the smooth‑max surrogate."""
    b = _fft_self_convolve(a)
    cur_max = b.max()
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.01,
                   alpha_max: float = 4000.0,
                   lr0: float = 0.5,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """Adam optimisation of the smooth‑max surrogate."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional jitter to escape shallow plateaux
        if t % 500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """One sub‑gradient step for the exact ℓ∞ objective."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Exact ℓ∞ refinement using sub‑gradient descent."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.03 / np.sqrt(it)            # slightly larger initial steps
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        if it % 400 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, deadline: float):
    """Light simulated‑annealing polish."""
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_peak = _fft_self_convolve(cur_a).max()
    best_a = cur_a.copy()
    best_peak = cur_peak

    start = time.time()
    total_time = max(deadline - start, 1e-6)

    start_temp = max(cur_peak, 1e-3) * 0.6
    final_temp = start_temp * 1e-5
    if final_temp <= 0.0:
        final_temp = 1e-9
    cooling_rate = -math.log(final_temp / start_temp) / total_time

    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1
        src = np.random.randint(n)
        if cur_a[src] <= 1e-12:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue
        max_delta = min(cur_a[src], 0.05)
        delta = np.random.rand() * max_delta

        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_peak = _fft_self_convolve(a_new).max()
        delta_obj = new_peak - cur_peak

        elapsed = time.time() - start
        temperature = start_temp * math.exp(-cooling_rate * elapsed)

        accept = False
        if delta_obj < 0.0:
            accept = True
        else:
            prob = math.exp(-delta_obj / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_a = a_new
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_a = cur_a.copy()

        if iter_cnt % 20000 == 0 and np.random.rand() < 0.05:
            cur_a = best_a.copy()
            cur_peak = best_peak

    return best_a, best_peak

def _lp_improve(a: np.ndarray,
                max_iters: int = 8,
                mix_t: float = 0.3,
                inner_deadline: float = None) -> np.ndarray:
    """
    LP‑based improvement (dual widening).

    Repeatedly solves:
        maximise Σ bj   subject to   (a∗b)k ≤ M,
    where M = max(a∗a).  The normalised solution is mixed back into a.
    """
    if not _cvxpy_available:
        return a

    n = a.shape[0]
    for _ in range(max_iters):
        if inner_deadline is not None and time.time() > inner_deadline:
            break

        M = _fft_self_convolve(a).max()
        # Build convolution matrix A (2n‑1)×n : columns are shifted copies of a
        col = np.r_[a, np.zeros(n - 1)]
        row = np.r_[a[0], np.zeros(n - 1)]
        A = scipy.linalg.toeplitz(col, row)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [A @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                       abstol=1e-8, reltol=1e-8, feastol=1e-8)
        except Exception:   # pragma: no cover
            break
        if b_var.value is None:
            break
        b_vec = np.maximum(b_var.value, 0.0)
        s = b_vec.sum()
        if s <= 1e-12:
            break
        g = b_vec / s
        a = (1.0 - mix_t) * a + mix_t * g
        a = _project_onto_simplex(a, s=1.0)

        # gradually shrink the mixing weight to stabilise the iteration
        mix_t = max(0.05, mix_t * 0.85)

    return a

def _search_at_length(n: int,
                     deadline: float,
                     base_seq: np.ndarray = None) -> tuple[np.ndarray, float]:
    """
    Full optimisation pipeline for a fixed length `n`.
    Returns the best vector (sum = 1) and its peak convolution value.
    """
    # ---- 1. diverse starters -------------------------------------------------
    starters = {}
    starters["arcsine"] = _arcsine_distribution(n)
    starters["uniform"] = np.full(n, 1.0 / n, dtype=np.float64)

    # Dirichlet draws
    for conc in (0.2, 0.5, 1.0, 2.0):
        starters[f"dirichlet_{conc:.1f}"] = np.random.dirichlet(np.full(n, conc))

    # Symmetric Beta family
    for a in np.linspace(0.32, 0.78, 8):
        starters[f"beta_{a:.2f}"] = _beta_distribution(n, a, a)

    # Power‑law boundary family
    for p in np.linspace(0.4, 0.9, 9):
        starters[f"power_{p:.2f}"] = _powerlaw_boundary_distribution(n, p)

    # Mixtures arcsine+uniform
    for w in (0.2, 0.4, 0.6, 0.8):
        starters[f"mix_{w:.1f}"] = (1.0 - w) * _arcsine_distribution(n) + w * np.full(n,
                                                                                       1.0 / n)

    if base_seq is not None:
        starters["prev"] = _interpolate_to_length(base_seq, n)

    best_vec = None
    best_fast = float("inf")
    for tag, vec in starters.items():
        vec = _project_onto_simplex(vec, s=1.0)
        peak = _fft_self_convolve(vec).max()
        fast = 2.0 * n * peak
        if fast < best_fast:
            best_fast = fast
            best_vec = vec.copy()
    a = best_vec

    # ---- 2. Adam optimisation ------------------------------------------------
    now = time.time()
    if now < deadline:
        adam_dead = now + 0.45 * (deadline - now)
        a, _ = _adam_optimize(a, adam_dead, init_alpha=1.0, alpha_max=4000.0,
                              lr0=0.5)

    # ---- 3. Exact ℓ∞ sub‑gradient descent ------------------------------------
    now = time.time()
    if now < deadline:
        sub_dead = now + 0.35 * (deadline - now)
        a = _refine_max(a, sub_dead)

    # ---- 4. LP‑based improvement (only for moderate n) -----------------------
    now = time.time()
    if now < deadline and len(a) <= 2500 and _cvxpy_available:
        lp_dead = now + 0.15 * (deadline - now)
        a = _lp_improve(a, max_iters=10, mix_t=0.3, inner_deadline=lp_dead)

    # ---- 5. Simulated annealing polishing ------------------------------------
    now = time.time()
    if now < deadline:
        a, final_peak = _simulated_annealing(a, deadline)
    else:
        final_peak = _fft_self_convolve(a).max()

    # ---- 6. Final ℓ∞ sub‑gradient refinement (use any leftover time) ---------
    now = time.time()
    if now < deadline:
        a = _refine_max(a, deadline)
        final_peak = _fft_self_convolve(a).max()

    return a, final_peak

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Entry point used by the competition harness.
    Returns a list of non‑negative heights (probability vector, sum≈1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Try to initialise from a previously discovered construction.
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_seq = _project_onto_simplex(base_seq, s=1.0)
        base_n = len(base_seq)
    except Exception:
        base_seq = None
        base_n = 2000

    # --------------------------------------------------------------
    # Choose a working length.  We deliberately stay ≤2500 so that the LP
    # step can be used without excessive memory.
    target_n = max(800, min(2500, int(base_n * 0.9)))
    print(f"[setup] working length = {target_n}")

    # --------------------------------------------------------------
    # Run the optimisation pipeline.
    best_vec, best_peak = _search_at_length(target_n, deadline, base_seq)
    best_fast = 2.0 * target_n * best_peak
    print(f"[result] best fast_score = {best_fast:.6f} (n={target_n})")

    # --------------------------------------------------------------
    # If time permits, upscale to a larger length (preserving the shape)
    # and perform a very light polish.
    time_left = deadline - time.time()
    if time_left > 20.0:
        new_n = max(target_n, int(target_n * 1.8))
        print(f"[upscale] interpolating to n={new_n}")
        a_big = _interpolate_to_length(best_vec, new_n)
        # a brief SA on the larger vector
        a_big, big_peak = _simulated_annealing(a_big, deadline)
        best_vec = a_big
        best_peak = big_peak
        best_fast = 2.0 * new_n * best_peak
        print(f"[upscale] post‑upscale fast_score = {best_fast:.6f}")

    # --------------------------------------------------------------
    # Final exact evaluation (scale‑invariant, so we can keep sum=1).
    try:
        true_score = evaluate_sequence(best_vec.tolist())
        print(f"[final] evaluate_sequence score = {true_score:.6f}")
    except Exception as exc:   # pragma: no cover
        true_score = float("inf")
        print("[final] evaluate_sequence raised:", exc)

    return best_vec.tolist()
