def _lp_optimize(a: np.ndarray, deadline: float, max_iter: int = 200) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_score(a)
    t = 0.3
    iteration = 0
    while time.time() < deadline and iteration < max_iter:
        iteration += 1
        b = _fft_self_convolve(a)
        M = b.max()
        # active indices: choose top K values of b, where K = min(5, len(b))
        K = min(5, b.size)
        k_inds = np.argpartition(-b, K-1)[:K]   # indices of K largest values
        # Build A_ub
        rows = []
        cols = []
        data = []
        for row_idx, k in enumerate(k_inds):
            start = max(0, k - (n - 1))
            end = min(k, n - 1)
            js = np.arange(start, end + 1, dtype=int)
            i_vals = k - js
            rows.extend([row_idx] * js.size)
            cols.extend(js.tolist())
            data.extend(a[i_vals].tolist())
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(k_inds), n))
        # solve linear program: max sum(g) <=...
        c = -np.ones(n)  # minimize -sum = maximize sum
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(len(k_inds), M), bounds=(0, None), method='highs')
        if not res.success:
            break
        g = res.x
        s = g.sum()
        if s <= 0:
            break
        g = g / s
        # update
        a = (1 - t) * a + t * g
        a = _project_onto_simplex(a, 1.0)  # ensure sum=1, nonnegative
        # compute score
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
        # reduce t gradually:
        t = max(0.05, t * 0.99)  # slowly decay
    return best, best_score

"""
Optimise a non‑negative step function (probability vector) to minimise
evaluate_sequence.  The objective is 2·n·max(a*a)/sum(a)^2, which is
scale‑invariant: we normalise the vector to sum = 1 and minimise
2·n·max(a*a).  The optimiser proceeds in four phases for each chosen
length:

1. **Initial analytic families** – uniform, arcsine, triangular,
   exponential and (asymmetric) Beta distributions give a set of seeds.

2. **Linear‑programming (active‑set) update** – for a fixed vector f we
   solve a tiny LP that maximises the total mass of a new vector g
   under the constraint ‖f∗g‖∞ ≤ M (M is the current max‑convolution).
   The LP has only a few active constraints, so it is solved in a
   fraction of a millisecond.  A convex combination (1‑t)·f + t·g
   (projected onto the simplex) strictly reduces the objective.

3. **Smoothed Adam descent** – a continuation method that minimises a
   smooth soft‑max approximation of the max‑convolution, using FFT‑based
   gradients and projection onto the simplex.

4. **Exact sub‑gradient refinement** – the true max‑sub‑gradient with
   momentum, followed by a cheap random‑transfer hill‑climber.

The whole search is timed (default 1000 s).  Multiple lengths (800 … 8000)
are tried, the best vector ever seen is kept and finally returned as a
plain Python list of floats.
"""

import time

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, p: float, q: float) -> np.ndarray:
    """Beta(p,q) – p,q>0 – discretised on the grid (i+0.5)/n."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.power(x, p - 1) * np.power(1 - x, q - 1)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _smooth_descent(a: np.ndarray, deadline: float) -> Tuple[np.ndarray, float]:
    """Adam optimiser on the soft‑max smoothed max‑convolution."""
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # more aggressive schedule than the reference implementation
    schedule = [(1.0, 2000), (5.0, 2000), (20.0, 2000),
                (100.0, 2000), (500.0, 2000)]

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score
            t += 1

            b = _fft_self_convolve(a)

            # stable softmax over b
            maxb = b.max()
            exp_terms = np.exp(alpha * (b - maxb))
            w = exp_terms / exp_terms.sum()

            # gradient = 2 * conv_rev(a[::-1], w) (first n entries)
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam step + simplex projection
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a, 1.0)

            if t % 250 == 0:
                cur_score = _max_convolution_score(a)
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_seq = a.copy()
    return best_seq, best_score


# ----------------------------------------------------------------------
def _exact_subgradient_refine(a: np.ndarray, deadline: float) -> Tuple[np.ndarray, float]:
    """Exact max‑subgradient descent with momentum."""
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)

    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps)

        w = np.zeros_like(b)
        w[active] = 1.0

        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / np.sqrt(iteration)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad iter={iteration} new best {best_score:.6f}")

        if best_score <= 1.503:
            break
        if time.time() + 0.5 > deadline:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_hill_climb(a: np.ndarray, deadline: float) -> Tuple[np.ndarray, float]:
    """Random‑transfer hill climbing on the simplex."""
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)
    cur_score = best_score

    max_transfer = 0.07
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        transfer = max_transfer * (0.5 ** (iteration / 4000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[{time.time():.1f}] hill iter={iteration} improved best {best_score:.6f}")

        if iteration % 5000 == 0 and np.random.rand() < 0.2:
            pert = np.random.randn(n) * 1e-4
            a = best_seq + pert
            a = _project_onto_simplex(a, 1.0)
            cur_score = _max_convolution_score(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _lp_optimize(a: np.ndarray,
                 deadline: float,
                 max_iter: int = 200,
                 t0: float = 0.3) -> Tuple[np.ndarray, float]:
    """
    Active‑set linear‑programming step.
    For the current vector f (=a) we solve:
        maximise   sum(g)
        subject to f * g ≤ M   (elementwise)
                   g ≥ 0
    where M = max(f*f).  Only the constraints that are (almost) active
    (the K largest entries of f*f) are kept – this makes the LP tiny.
    The result g is normalised to sum = 1 and blended with f.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)

    t = t0
    iteration = 0
    while time.time() < deadline and iteration < max_iter:
        iteration += 1
        b = _fft_self_convolve(a)
        M = b.max()
        # pick up to 5 most active convolution indices
        K = min(5, b.size)
        active_idx = np.argpartition(-b, K - 1)[:K]

        # build the (small) constraint matrix A g ≤ M
        A = np.zeros((len(active_idx), n), dtype=np.float64)
        for row, k in enumerate(active_idx):
            j_start = max(0, k - (n - 1))
            j_end = min(k, n - 1)
            js = np.arange(j_start, j_end + 1, dtype=int)
            A[row, js] = a[k - js]

        # linear programme: maximise sum(g)  → minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)
        res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=np.full(A.shape[0], M, dtype=np.float64),
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            # LP failed – fall back to gradient descent
            break
        g = res.x
        g_sum = g.sum()
        if g_sum <= 0:
            break
        g = g / g_sum

        # convex combination and projection
        a = (1.0 - t) * a + t * g
        a = _project_onto_simplex(a, 1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] LP iter={iteration} new best {best_score:.6f}")

        # slowly decay step size
        t = max(0.05, t * 0.97)

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Main entry point required by the evaluation harness.
    Returns the best step‑function (list of non‑negative floats) found
    within the allotted time.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 3.0   # small safety margin

    # lengths to explore – a mix of medium and large sizes
    lengths = [800, 1200, 2000, 3000, 5000, 8000]

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. Scan initial analytic families
    for n in lengths:
        if time.time() >= deadline:
            break
        for cand in _initial_candidates(n):
            if time.time() >= deadline:
                break
            cand = _project_onto_simplex(cand, 1.0)
            score = _max_convolution_score(cand)
            if score < overall_best_score:
                overall_best_score = score
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] init n={n} score={score:.6f}")

    # ------------------------------------------------------------------
    # 2. If a previous construction exists, try it as a seed
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = _project_onto_simplex(prev, 1.0)
            sc = _max_convolution_score(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # If nothing survived (should not happen) fall back to uniform
    if overall_best_seq is None:
        overall_best_seq = _distribution_uniform(1200)
        overall_best_score = _max_convolution_score(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Run the three optimisation phases for each promising length
    for n in lengths:
        if time.time() >= deadline:
            break
        # take the best vector of exactly this length
        if overall_best_seq.shape[0] != n:
            a = _distribution_uniform(n)
        else:
            a = overall_best_seq.copy()

        # LP‑based active‑set optimisation
        a, sc = _lp_optimize(a, deadline, max_iter=150, t0=0.4)
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, a.copy()
            print(f"[{time.time():.1f}] after LP n={n} best={sc:.6f}")

        # Smoothed Adam descent
        a, sc = _smooth_descent(a, deadline)
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, a.copy()
            print(f"[{time.time():.1f}] after smooth n={n} best={sc:.6f}")

        # Exact max‑subgradient refinement
        a, sc = _exact_subgradient_refine(a, deadline)
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, a.copy()
            print(f"[{time.time():.1f}] after subgrad n={n} best={sc:.6f}")

        # Local hill‑climbing
        a, sc = _local_hill_climb(a, deadline)
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, a.copy()
            print(f"[{time.time():.1f}] after hill n={n} best={sc:.6f}")

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # ensure a plain Python list of floats is returned
    return overall_best_seq.tolist()
