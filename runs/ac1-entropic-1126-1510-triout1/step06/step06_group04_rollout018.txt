"""
Propose a step‑function sequence with a low evaluate_sequence score.

The method consists of three phases:

1. **Broad exploration** – try several families of distributions
   (uniform, arcsine, symmetric Beta, random asymmetric Beta) on a few
   lengths (4000, 8000, 12000).  The best candidate (lowest raw score
   2·n·max(a*a) ) is retained.

2. **Fine symmetric‑Beta sweep** – a dense linear sweep of the
   concentration parameter α ∈ (0.51, 3] for the symmetric Beta(α,α)
   distribution is performed on the current best length.  This refines
   the shape that gives the smallest L∞‑norm of the self‑convolution.

3. **Projected Adam refinement** – starting from the best discrete
   distribution, we optimise the smooth approximation
   `softmax_β(b) = exp(β·b)/∑exp(β·b)` of the max, with β slowly
   increased (continuation).  The variables are re‑parameterised by
   logits `u` (so that a = softmax(u) automatically satisfies the
   simplex constraints).  An Adam optimiser is used; the best iterate
   (according to the exact raw score) is kept.

The whole routine respects the 1000 s budget and returns the final
sequence as a plain Python list of floats.

import time

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) of length n, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n                # mid‑points
    # log‑normaliser
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # guard against overflow / underflow
    logpdf = np.minimum(logpdf, 700.0)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a via real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of a and b via real FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _raw_score(a: np.ndarray) -> float:
    """Proxy score for a normalised to Σa = 1 :  2·n·max(a*a)."""
    n = a.shape[0]
    return 2.0 * n * _self_convolution_full(a).max()


# ----------------------------------------------------------------------
def _softmax(u: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (sum to 1, entries ≥0)."""
    shift = u - np.max(u)
    e = np.exp(shift)
    return e / e.sum()


def _smooth_max_weights(b: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute the soft‑max weights w_k = exp(β·b_k) / Σ exp(β·b_k).
    The argument b is the full self‑convolution (length 2·n‑1).
    """
    # shift for numerical stability
    shift = beta * (b - b.max())
    w = np.exp(shift)
    s = w.sum()
    if s == 0.0:
        w[:] = 1.0 / w.size
    else:
        w /= s
    return w


def _smooth_max_grad(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth max (soft‑max) of the self‑convolution.
    Returns ∂/∂a (softmax_β( a*a )).
    """
    n = a.shape[0]
    b = _self_convolution_full(a)                     # length 2n‑1
    w = _smooth_max_weights(b, beta)                  # same length
    # gradient = 2 * (a_rev * w) restricted to n entries
    a_rev = a[::-1]
    grad_full = _fft_convolve(a_rev, w)               # length n‑1 + (2n‑1)
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]         # central n entries
    return grad

            if time.time() >= deadline:
                break
            vec = _beta_distribution(n, a, a)
            sc = _raw_score(vec)
            if sc < best_score:
                best_score, best_seq, best_len = sc, vec.copy(), n
                print(f"   sym‑Beta a={a:.3f} score={best_score:.6f}")

        # ---- random asymmetric Beta draws ------------------------------------------------
        for _ in range(400):
            if time.time() >= deadline:
                break
            alpha = 10 ** rng.uniform(-2, 1)   # 0.01 … 10
            beta = 10 ** rng.uniform(-2, 1)
            vec = _beta_distribution(n, alpha, beta)
            sc = _raw_score(vec)
            if sc < best_score:
                best_score, best_seq, best_len = sc, vec.copy(), n
                print(f"   rand‑Beta a={alpha:.3f} b={beta:.3f} score={best_score:.6f}")

    # ------------------------------------------------------------------
    if best_seq is None:
        n = 8000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_len = n
        best_score = _raw_score(best_seq)
        print("Fallback uniform used.")

    print(f"Best after coarse search: score={best_score:.6f} (len={best_len})")

    # ------------------------------------------------------------------
    # 2) Fine symmetric‑Beta sweep on the best length
    n = best_len
    fine_alphas = np.linspace(0.51, 3.0, 250)   # dense sweep
    for a in fine_alphas:
        if time.time() >= deadline:
            break
        vec = _beta_distribution(n, a, a)
        sc = _raw_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.copy()
            print(f"   fine‑Beta a={a:.4f} improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Local Adam‑refinement with smooth‑max continuation
    # initialise logits u so that softmax(u) = best_seq (sum already 1)
    a = best_seq.copy()
    a = a / a.sum()
    u = np.log(a + 1e-12)           # logits, unconstrained
    m = np.zeros_like(u)           # Adam first moment
    v = np.zeros_like(u)           # Adam second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.05                      # base learning rate
    step = 1
    best_a = a.copy()
    best_val = best_score
    # smoothing parameter for soft‑max (continuation)
    beta_smooth = 5.0

    while time.time() < deadline:
        # ---- compute current a and its convolution ---------------------------------
        a = _softmax(u)                         # guaranteed simplex point
        b = _self_convolution_full(a)           # length 2n‑1
        cur_max = b.max()

        # ---- compute smooth‑max gradient (Adam works on logits) --------------------
        w = _smooth_max_weights(b, beta_smooth)                     # soft‑max weights
        # gradient wrt a
        a_rev = a[::-1]
        grad_a = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]   # shape (n,)

        # chain rule to obtain gradient wrt logits u (softmax Jacobian)
        grad_u = a * (grad_a - np.dot(grad_a, a))

        # ---- Adam update -----------------------------------------------------------
        step += 1
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        u -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- continuity: slowly increase the smoothing ----------------------------
        if step % 5000 == 0:
            beta_smooth = min(5000.0, beta_smooth * 1.5)

        # ---- occasional jitter to escape plateaus ---------------------------------
        if step % 2000 == 0 and rng.random() < 0.03:
            jitter = rng.normal(scale=1e-7, size=n)
            u += jitter

        # ---- evaluate exact raw score ------------------------------------------------
        cur_val = _raw_score(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] refinement it={step}  score={best_val:.6f}")

        # early exit if we have already achieved the target
        if best_val * 2 <= 1.503:   # 2·C ≤ target → raw score ≤ target
            # the exact `evaluate_sequence` would give the same value,
            # but we stop a little earlier to leave time for final evaluation.
            break

    # ------------------------------------------------------------------
    final_seq = best_a
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Final evaluated score: {final_score:.6f} (len={len(final_seq)})")
    return final_seq.tolist()
