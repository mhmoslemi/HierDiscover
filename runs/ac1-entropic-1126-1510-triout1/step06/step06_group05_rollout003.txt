"""
Search for a low‑score non‑negative step function.

The algorithm consists of the following stages (all respecting a hard
deadline supplied by the caller).

1. **Broad seeding** – generate a variety of probability vectors for several
   lengths (arcsine, Beta, power‑law, Dirichlet, simple mixtures).  The best
   according to the cheap proxy `2·n·max(conv)` is kept.

2. **Adam on a smooth‑max surrogate** – minimise  
   Lα(a)= (1/α)·log Σexp(α·(a∗a)k) while keeping a on the simplex.  The
   schedule for the smoothness parameter α is very gradual (α grows only
   0.5 % per iteration) which yields a very accurate approximation of the
   ℓ∞‑norm.

3. **Exact ℓ∞ sub‑gradient descent** – a short but aggressive routine that
   follows the true max‑objective.

4. **Optional LP improvement** – solves the dual linear programme
   “maximise Σb_j subject to (a∗b)_k ≤ M” a few times (only for moderate
   n to keep memory low).

5. **Simulated‑annealing polishing** – stochastic moves with a slowly
   decreasing temperature; occasionally large mass transfers are allowed.

6. **Fine‑tuning** – a final Adam run with a *fixed* large α and a tiny
   learning‑rate to push the solution past the plateau that often remains
   after the previous stages.

All steps constantly re‑project onto the simplex Σa_i = 1, a_i ≥ 0.
The best sequence found before the deadline is returned as a plain Python
list, ready for ``evaluate_sequence``.
"""

import math
import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if ind.size == 0:
        theta = 0.0
    else:
        rho = ind[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]

# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length `n`."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) distribution on a uniform grid, normalised to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_vals = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    vals[~np.isfinite(vals)] = 0.0
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


def _powerlaw_boundary_distribution(n: int, p: float) -> np.ndarray:
    """Distribution ∝ ((i+0.5)*(n‑i‑0.5))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5) + 1e-12) ** (-p)
    vals /= vals.sum()
    return vals

# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """Gradient of Lα(a) = (1/α)·log Σ exp(α·(a*a)_k)."""
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # use log‑sum‑exp trick for numerical stability
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()          # weights sum to 1
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)    # length 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.005,
                   alpha_max: float = 2000.0,
                   lr0: float = 0.15,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """Adam optimisation of the smooth‑max surrogate."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny random perturbation to escape flat regions
        if t % 400 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """Exact ℓ∞ sub‑gradient step (uniform weight on active constraints)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-14, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Short exact‑max refinement using sub‑gradient descent."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.04 / np.sqrt(it)   # a bit more aggressive than the original
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()
        if it % 800 == 0 and np.random.rand() < 0.08:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)
    return best_a

def _lp_improve(a: np.ndarray,
                max_iters: int = 3,
                mix_t: float = 0.5,
                inner_deadline: float = None) -> np.ndarray:
    """
    Dual LP improvement – maximise Σb_j under (a∗b)_k ≤ M.
    The resulting b is mixed back into a.
    """
    if not _cvxpy_available:
        return a

    n = a.shape[0]
    for _ in range(max_iters):
        if inner_deadline is not None and time.time() > inner_deadline:
            break
        M = _fft_self_convolve(a).max()
        # Toeplitz matrix A of size (2n‑1)×n  (dense, but n is kept moderate)
        col = np.r_[a, np.zeros(n - 1)]
        row = np.r_[a[0], np.zeros(n - 1)]
        A = scipy.linalg.toeplitz(col, row)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [A @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception:   # pragma: no cover
            break
        if b_var.value is None:
            break
        b_vec = np.maximum(b_var.value, 0.0)
        s = b_vec.sum()
        if s <= 1e-12:
            break
        g = b_vec / s
        a = (1.0 - mix_t) * a + mix_t * g
        a = _project_onto_simplex(a, s=1.0)
    return a

def _simulated_annealing(a0: np.ndarray, deadline: float):
    """Light simulated‑annealing polish."""
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_peak = _fft_self_convolve(cur_a).max()
    best_a = cur_a.copy()
    best_peak = cur_peak

    start = time.time()
    total_time = max(deadline - start, 1e-6)

    # initialise a temperature proportional to the current peak
    start_temp = max(cur_peak, 1e-4) * 0.6
    final_temp = start_temp * 1e-5
    if final_temp <= 0.0:
        final_temp = 1e-9
    cooling_rate = -math.log(final_temp / start_temp) / total_time

    it = 0
    while time.time() < deadline:
        it += 1
        src = np.random.randint(n)
        if cur_a[src] <= 1e-14:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue

        # allow moderately larger jumps later on
        max_delta = min(cur_a[src], 0.08)
        delta = np.random.rand() * max_delta

        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_peak = _fft_self_convolve(a_new).max()
        delta_obj = new_peak - cur_peak

        elapsed = time.time() - start
        temperature = start_temp * math.exp(-cooling_rate * elapsed)

        accept = False
        if delta_obj < 0.0:
            accept = True
        else:
            prob = math.exp(-delta_obj / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_a = a_new
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_a = cur_a.copy()

        # occasional restart to the currently best solution
        if it % 25000 == 0 and np.random.rand() < 0.06:
            cur_a = best_a.copy()
            cur_peak = best_peak

    return best_a, best_peak

def _final_smooth_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Very fine Adam descent on the smooth‑max surrogate with a *fixed* large
    α.  This is meant to push the solution past the flat plateau that often
    remains after the other stages.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    # fixed, very large α → almost the true max
    alpha = 2000.0
    lr0 = 0.02

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam update with a much slower learning‑rate schedule
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad * grad)
        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)

        lr = lr0 / (t ** 0.7)

        a = a - lr * m_hat / (np.sqrt(v_hat) + 1e-9)
        a = _project_onto_simplex(a, s=1.0)

    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """Return a list of non‑negative heights with a low evaluation score."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # ---- 1.  Broad seeding -------------------------------------------------
    print("[phase 1] broad seeding")
    lengths = [500, 800, 1200, 2000, 3000, 4000, 5000]
    overall_best_seq = None
    overall_best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break

        starters = {}

        starters["arcsine"] = _arcsine_distribution(n)
        starters["uniform"] = np.full(n, 1.0 / n, dtype=np.float64)

        # Dirichlet draws (different concentration values)
        for conc in (0.2, 0.5, 1.0, 2.0):
            starters[f"dirichlet_{conc:.1f}"] = np.random.dirichlet(
                np.full(n, conc)
            )

        # Symmetric beta families
        for a in (0.35, 0.45, 0.55, 0.65, 0.75):
            starters[f"beta_{a:.2f}"] = _beta_distribution(n, a, a)

        # Power‑law at the boundary
        for p in (0.4, 0.45, 0.5, 0.55, 0.6):
            starters[f"power_{p:.2f}"] = _powerlaw_boundary_distribution(n, p)

        # Mix arcsine & uniform
        for w in (0.2, 0.4, 0.6, 0.8):
            starters[f"mix_{w:.1f}"] = (1.0 - w) * _arcsine_distribution(n) + w * np.full(
                n, 1.0 / n
            )

        # evaluate cheap proxy
        for tag, vec in starters.items():
            vec = _project_onto_simplex(vec, s=1.0)
            peak = _fft_self_convolve(vec).max()
            fast_score = 2.0 * n * peak
            if fast_score < overall_best_score:
                overall_best_score = fast_score
                overall_best_seq = vec.copy()
                print(f"[init] n={n:4d} {tag:<12s} fast_score={fast_score:.6f}")

    if overall_best_seq is None:
        # very unlikely, but keep a safe fallback
        n = 800
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # ---- 2.  Adam optimisation (smooth‑max) -------------------------------
    time_left = deadline - time.time()
    if time_left > 6.0:
        n = len(overall_best_seq)
        print(f"[phase 2] Adam optimisation – {time_left:.1f}s left (n={n})")
        adam_deadline = time.time() + time_left * 0.45
        a_opt, _ = _adam_optimize(overall_best_seq, adam_deadline)
        # a few extra Adam restarts with tiny random perturbations
        restarts = 3
        for r in range(restarts):
            if time.time() >= adam_deadline:
                break
            pert = a_opt + np.random.randn(n) * 5e-5
            pert = _project_onto_simplex(pert, s=1.0)
            sub_dead = time.time() + (time_left * 0.02)
            a_opt, _ = _adam_optimize(pert, sub_dead)
        overall_best_seq = a_opt.copy()
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print(f"[phase 2] post‑Adam fast_score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # ---- 3.  Exact max sub‑gradient refinement ----------------------------
    time_left = deadline - time.time()
    if time_left > 5.0:
        n = len(overall_best_seq)
        print(f"[phase 3] sub‑gradient refinement – {time_left:.1f}s left")
        sub_deadline = time.time() + time_left * 0.55
        overall_best_seq = _refine_max(overall_best_seq, sub_deadline)
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print(f"[phase 3] after sub‑grad fast_score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # ---- 4.  Optional LP improvement --------------------------------------
    time_left = deadline - time.time()
    if time_left > 3.0 and len(overall_best_seq) <= 3000 and _cvxpy_available:
        n = len(overall_best_seq)
        print(f"[phase 4] LP improvement – {time_left:.1f}s left (n={n})")
        lp_deadline = time.time() + time_left * 0.6
        overall_best_seq = _lp_improve(
            overall_best_seq, max_iters=4, mix_t=0.5, inner_deadline=lp_deadline
        )
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print(f"[phase 4] after LP fast_score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # ---- 5.  Simulated‑annealing polishing ---------------------------------
    time_left = deadline - time.time()
    if time_left > 2.0:
        n = len(overall_best_seq)
        print(f"[phase 5] simulated annealing – {time_left:.1f}s left")
        sa_deadline = deadline   # run until the global deadline
        a_sa, final_peak = _simulated_annealing(overall_best_seq, sa_deadline)
        overall_best_seq = a_sa.copy()
        overall_best_score = 2.0 * n * final_peak
        print(f"[phase 5] SA fast_score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # ---- 6.  Final fine‑tuning (high‑α Adam) -------------------------------
    time_left = deadline - time.time()
    if time_left > 1.0:
        n = len(overall_best_seq)
        print(f"[phase 6] final smooth‑max fine‑tune – {time_left:.1f}s left")
        fin_dead = time.time() + time_left
        overall_best_seq = _final_smooth_descent(overall_best_seq, fin_dead)
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print(f"[phase 6] final fast_score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # ---- 7.  Exact evaluation ---------------------------------------------
    try:
        true_score = evaluate_sequence(overall_best_seq.tolist())
        print(f"[final] evaluate_sequence score={true_score:.6f}")
        if true_score < overall_best_score:
            overall_best_score = true_score
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    print(f"Best score (approx.): {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
