def compute_smooth_obj_and_grad(y, mu, n, S):
    p = _softmax(y)
    a = S * p
    # conv via FFT:
    N = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, N)
    conv = np.fft.irfft(A * A, N)[:2*n-1]
    max_c = conv.max()
    if mu > 0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700, 700)
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv)
        if cnt > 0:
            w[mask] = 1.0 / cnt
        f_val = max_c
    # compute grad_a via convolution w * a_rev:
    a_rev = a[::-1]
    # length needed: len(w)+len(a_rev)-1 = (2n-1)+n-1 = 3n-2
    L = 1 << ((3*n - 2).bit_length())
    W = np.fft.rfft(w, L)
    A_rev = np.fft.rfft(a_rev, L)
    conv_wr = np.fft.irfft(W * A_rev, L)[:3*n - 2]
    g_a = 2.0 * conv_wr[n-1:n-1+n]
    # gradient w.r.t. y:
    inner = np.dot(g_a, p)
    grad_y = S * p * (g_a - inner)
    return float(f_val), grad_y.astype(np.float64)
```

"""
Optimise a non‑negative step function for the provided evaluate_sequence()
score.

The search consists of three stages:

1️⃣  Smooth‑max optimisation (L‑BFGS‑B) on the log‑softmax parameters.
    The gradient and the smooth max are evaluated with FFT based
    convolutions, which is far faster than the direct np.convolve used
    inside evaluate_sequence.  Several random restarts (uniform, exponential,
    Dirichlet draws, and the reference sequence when available) are tried.

2️⃣  Projected sub‑gradient descent on the true max‑objective.
    The sub‑gradient of max(conv) is obtained analytically (again with FFT)
    and we perform a long diminishing‑step descent.

3️⃣  Fast stochastic hill‑climbing.
    Random swaps, small δ‑mass transfers and block‑averaging moves are
    evaluated via a custom max‑convolution routine (FFT).  The move is
    accepted only if it strictly improves the exact score.
    A simple simulated‑annealing acceptance rule is also used to escape
    plateaus.

All helper functions are top‑level (no closures or lambdas) and the
routine respects the user supplied time budget (default 1000 s).  The
best sequence found is returned as a list of floats.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """
    Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0).
    """
    if z == 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """
    Compute max_i (a * a)[i] using FFT based convolution.
    All entries of a are assumed non‑negative and not larger than 1000.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, L)
    conv = np.fft.irfft(A * A, L)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
def _compute_score(a: np.ndarray) -> float:
    """
    Exact score according to evaluate_sequence().
    Since the ℓ¹‑norm is fixed to √(2n), the score equals max(convolution).
    """
    max_c = _max_convolution_fft(a)
    # for safety – use the same formula as evaluate_sequence()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_c / (sum_a * sum_a))


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(y: np.ndarray, mu: float, n: int, S: float
                        ) -> tuple[float, np.ndarray]:
    """
    Smoothed max‑objective and its gradient w.r.t. the log‑softmax
    parametrisation `y`.  The convolution and the gradient are evaluated
    with FFTs for speed.

    Parameters
    ----------
    y : ndarray (n,)
        Unconstrained parameters (log‑probabilities).
    mu : float (>0)
        Smoothing parameter (μ → 0 gives the hard max).
    n, S : problem size and target ℓ¹‑norm.

    Returns
    -------
    f_val : float – smooth‑max value
    grad_y : ndarray (n,) – gradient w.r.t. `y`.
    """
    # probabilities on the simplex, actual heights
    p = _softmax(y)          # Σ p = 1, p ≥ 0
    a = S * p                # Σ a = S

    # ---- convolution (FFT) --------------------------------------------
    Lc = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, Lc)
    conv = np.fft.irfft(A * A, Lc)[: 2 * n - 1]
    max_c = conv.max()

    # ---- smooth‑max weighting -----------------------------------------
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # guard overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z                       # soft‑max weights (sum to 1)
        f_val = max_c + mu * math.log(Z)
    else:
        # hard max – uniform weight over all active indices
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        if cnt > 0:
            w[mask] = 1.0 / cnt
        f_val = max_c

    # ---- gradient w.r.t. a -------------------------------------------
    a_rev = a[::-1]
    Lg = _next_pow2((2 * n - 1) + n - 1)          # length needed for w * a_rev
    W = np.fft.rfft(w, Lg)
    A_rev = np.fft.rfft(a_rev, Lg)
    conv_wr = np.fft.irfft(W * A_rev, Lg)[: 3 * n - 2]
    g_a = 2.0 * conv_wr[n - 1 : n - 1 + n]

    # ---- chain rule to y (soft‑max Jacobian) -------------------------
    inner = np.dot(g_a, p)                # scalar Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)
    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable for scipy.optimize that returns (value, gradient)."""
    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _smooth_obj_and_grad(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
def _hard_subgradient(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient of the true max‑objective max(conv) at point a.
    Returns (g, max_conv) where g has length n.
    """
    n = a.shape[0]
    Lc = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, Lc)
    conv = np.fft.irfft(A * A, Lc)[: 2 * n - 1]
    max_c = conv.max()
    mask = conv == max_c
    cnt = np.count_nonzero(mask)
    w = np.zeros_like(conv, dtype=float)
    if cnt > 0:
        w[mask] = 1.0 / cnt

    a_rev = a[::-1]
    Lg = _next_pow2((2 * n - 1) + n - 1)
    W = np.fft.rfft(w, Lg)
    A_rev = np.fft.rfft(a_rev, Lg)
    conv_wr = np.fft.irfft(W * A_rev, Lg)[: 3 * n - 2]
    g = 2.0 * conv_wr[n - 1 : n - 1 + n]
    return g, float(max_c)

        eps = 1e-12
        init_ys.append(np.log(p_dir + eps))

    # if we already have a decent point, add its log‑softmax representation
    if best_score < float("inf"):
        p_best = best_a / max(best_a.sum(), 1e-12)
        eps = 1e-12
        init_ys.append(np.log(p_best + eps))

    mu_factor = 0.70                # cooling schedule
    mu_min = 1e-8
    maxiter_per_mu = 600           # L‑BFGS inner iterations per temperature
    n_restart = len(init_ys)

    for rid, y0 in enumerate(init_ys):
        if time.time() > deadline:
            break
        print(f"[restart {rid + 1}/{n_restart}] initialising")
        y = y0.copy()

        # initialise temperature from the current max‑convolution
        a_cur = S * _softmax(y)
        mu = max(_max_convolution_fft(a_cur), 1.0)

        while mu > mu_min and time.time() < deadline:
            obj = _SmoothMaxObjective(mu, n, S)
            try:
                res = opt.minimize(
                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": maxiter_per_mu, "gtol": 1e-12, "disp": False},
                )
            except Exception as exc:
                print(f"[error] L‑BFGS failed: {exc}")
                break

            y = res.x
            a = S * _softmax(y)
            _maybe_update(a)

            mu *= mu_factor

    # ------------------------------------------------------------------
    # 2️⃣  Projected sub‑gradient descent on the exact max
    if best_score < float("inf") and time.time() < deadline:
        a = best_a.copy()
        step0 = S * 0.5 / n            # initial step size heuristic
        it = 0
        max_local_iters = 80000
        while time.time() < deadline and it < max_local_iters:
            g, max_c = _hard_subgradient(a)
            step = step0 / math.sqrt(it + 1.0)
            a_new = a - step * g
            a_new = _project_to_simplex(a_new, S)
            sc = _compute_score(a_new)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                print(f"[subgrad] iter {it:5d} → {best_score:.6f}")
            else:
                # occasional jitter to escape flat regions
                if rng.random() < 0.01:
                    i, j = rng.choice(n, size=2, replace=False)
                    delta = rng.random() * a[i] * 0.03
                    a[i] -= delta
                    a[j] += delta
                    a = _project_to_simplex(a, S)
            it += 1

    # ------------------------------------------------------------------
    # 3️⃣  Large‑scale stochastic hill‑climbing (FFT‑based evaluation)
    if best_score < float("inf") and time.time() < deadline:
        a = best_a.copy()
        trial = 0
        # determine roughly how many trials we can still afford
        remaining = deadline - time.time()
        max_trials = int(min(300000, remaining * 120.0))
        print(f"[hill‑climb] planning up to {max_trials} trials")
        while time.time() < deadline and trial < max_trials:
            move_type = rng.integers(0, 3)      # 0: swap, 1: δ‑transfer, 2: block‑average
            cand = a.copy()
            if move_type == 0:
                i, j = rng.choice(n, size=2, replace=False)
                cand[i], cand[j] = cand[j], cand[i]
            elif move_type == 1:
                i = rng.choice(np.where(cand > 1e-12)[0])
                j = rng.choice(n)
                max_delta = cand[i] * 0.08
                delta = rng.random() * max_delta
                cand[i] -= delta
                cand[j] += delta
            else:  # block‑average
                start = rng.integers(0, n - 1)
                length = rng.integers(2, min(120, n - start))
                block = slice(start, start + length)
                avg = cand[block].mean()
                cand[block] = avg
            # projection only needed for the transfer move (sum may drift)
            cand = _project_to_simplex(cand, S)

            sc = _compute_score(cand)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = cand.copy()
                a = cand
                print(f"[hill] trial {trial:6d} → {best_score:.6f}")

            trial += 1

    # ------------------------------------------------------------------
    # Final evaluation with the provided black‑box (to avoid any hidden
    # discrepancy between our fast routine and the official scorer).
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] evaluate_sequence = {final_score:.6f}")

    return best_a.tolist()
