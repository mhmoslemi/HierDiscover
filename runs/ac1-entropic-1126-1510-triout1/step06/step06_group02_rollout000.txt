def propose_candidate(seed=1234, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2

    # get base_n
    try:
        base_n = len(height_sequence_1)
    except NameError:
        base_n = 8000

    # lengths
    factors = [0.8, 0.95, 1.0, 1.1, 1.3]
    lengths = sorted({max(2000, int(base_n * f)) for f in factors})

    top_global = None
    top_score = float('inf')

    # Phase 1: generate candidates
    candidates = []   # tuples (score, a, n)
    for n in lengths:
        # generate shapes
        # Beta grid
        alphas = np.logspace(-0.7, 0.7, num=11)   # 0.2..5
        betas = alphas
        for a in alphas:
            for b in betas:
                a_vec = _beta_distribution(n, a, b)
                score = cheap_score(a_vec)
                candidates.append((score, a_vec, n))
        # arcsine
        a_vec = _beta_distribution(n, 0.5, 0.5)
        candidates.append((cheap_score(a_vec), a_vec, n))
        # exponential lam values
        for lam in [0.3, 0.6, 1.0, 2.0]:
            a_vec = _exponential_distribution(n, lam)
            candidates.append((cheap_score(a_vec), a_vec, n))
        # triangular shape
        idx = np.arange(n)
        mid = n//2
        tri = np.minimum(idx, n-1-idx) + 1
        tri = tri / tri.sum()
        candidates.append((cheap_score(tri), tri, n))
        # random
        a_rand = np.random.rand(n)
        a_rand /= a_rand.sum()
        candidates.append((cheap_score(a_rand), a_rand, n))

    # sort and keep best per length maybe limited
    candidates.sort(key=lambda x: x[0])
    # keep best K overall, say 30
    K = 30
    selected = candidates[:K]

    # Phase 2: gradient refinement
    for idx, (init_score, a0, n) in enumerate(selected):
        if time.time() > deadline:
            break
        # run Adam for up to max_iters_time = maybe 20 sec per candidate
        # we compute #iterations based on time left
        max_iters = 5000
        # logit variable
        a = np.maximum(a0, 1e-12)
        z = np.log(a)   # unconstrained
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        t = 0
        best_local_a = a.copy()
        best_local_score = init_score
        temperature = 1.0
        temp_factor = 1.005
        temp_max = 5000

        while t < max_iters and time.time() < deadline:
            t += 1
            a = np.exp(z); a /= a.sum()
            conv = _fft_self_convolve(a)
            score = 2.0 * n * conv.max()
            if score < best_local_score:
                best_local_score = score
                best_local_a = a.copy()
                # optional print
                #print(...)

            # gradient
            if temperature < temp_max:
                grad_a = _softmax_gradient(a, conv, temperature)
                temperature *= temp_factor
            else:
                grad_a = _exact_gradient(a, conv)

            # convert to grad_z
            dot = np.dot(a, grad_a)
            grad_z = a * (grad_a - dot)

            # Adam step
            lr = 0.1  # maybe decayed?
            # compute hydrogen
            m = 0.9 * m + (1 - 0.9) * grad_z
            v = 0.999 * v + (1 - 0.999) * (grad_z**2)
            m_hat = m / (1 - 0.9**t)
            v_hat = v / (1 - 0.999**t)
            z = z - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # evaluate best local candidate with true evaluate_sequence
        final_val = evaluate_sequence(best_local_a.tolist())
        if final_val < top_score:
            top_score = final_val
            top_global = best_local_a.copy()
            print(f"New global best: {final_val:.6f} at length {n}")
            if top_score <= 1.503:
                return top_global.tolist()

    # Phase 3: Simulated annealing starting from top_global
    if top_global is None:
        # fallback uniform
        n = lengths[0]
        top_global = np.full(n, 1.0/n)
        top_score = evaluate_sequence(top_global.tolist())

    a = top_global.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = cheap_score(a)

    # SA parameters
    T0 = best_score * 0.1
    T_end = best_score * 1e-4
    total_sa_time = max(0.0, deadline - time.time())
    # set annealing schedule based on time
    start_time = time.time()
    while time.time() < deadline:
        # fraction of time elapsed
        elapsed = time.time() - start_time
        fraction = elapsed / total_sa_time if total_sa_time>0 else 1
        T = T0 * (T_end / T0) ** fraction   # exponential cooling
        # propose move
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # amount to transfer
        max_transfer = a[i]
        if max_transfer == 0:
            continue
        delta = max_transfer * np.random.rand() * 0.2   # at most 20% of source
        # apply move
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # compute cheap score
        cur_score = cheap_score(a_new)   # compute conv each time
        # acceptance
        if cur_score < best_score or np.random.rand() < np.exp(-(cur_score - best_score) / max(T, 1e-12)):
            a = a_new
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_new.copy()
                # optionally print
                #print(f"SA improve: {cur_score:.6f}")

    # final evaluate using real evaluate_sequence
    final_val = evaluate_sequence(best_a.tolist())
    if final_val < top_score:
        top_score = final_val
        top_global = best_a.copy()
    print(f"Final best: {top_score:.6f} length {len(top_global)}")
    return top_global.tolist()

"""
Propose a step‑function (non‑negative vector) that minimizes the objective
used by ``evaluate_sequence``.

Algorithm
---------
1. Choose a handful of lengths (≈ 3 k–12 k).  Larger n gives a better
   asymptotic constant but costs more per iteration.
2. For each length generate a *large* pool of analytic candidates:
   * Beta(α,β) densities on a grid of (α,β)
   * exponential decay, cosine, triangular shapes
   * a few Dirichlet draws (sparse random vectors)
   * the previous best vector (if ``height_sequence_1`` exists)
   * a simple random vector
   All candidates are normalised to sum = 1, i.e. they lie on the simplex.
3. Compute a cheap exact score ``2·n·max(conv)`` (the objective when the
   sum equals 1) via an FFT based self‑convolution and keep the best few
   (≤ 30) candidates overall.
4. Refine every kept candidate with a projected Adam optimiser on the
   simplex:
   * Parameterise a by logits ``z`` (softmax) so the sum constraint is
     automatically satisfied.
   * Use a smoothed surrogate of the max (soft‑max with a temperature
     that grows geometrically).  The gradient of the surrogate is
     obtained analytically via a second FFT.  After the temperature
     becomes large we switch to the exact sub‑gradient.
   * Adam provides adaptive step‑sizes and momentum.
   * Keep the best vector seen during the optimisation.
5. Starting from the best refined vector, perform a lightweight simulated
   annealing (random mass transfers between two coordinates) to escape
   possible local minima.
6. Return the best vector found.  The vector is already on the simplex,
   has only non‑negative entries and therefore yields a finite value from
   ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float = 1.0) -> np.ndarray:
    """Geometric‑like decreasing shape: a_i ∝ exp(-lam·i/(n‑1))."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * i / max(1, n - 1))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _cosine_distribution(n: int) -> np.ndarray:
    """Half‑cosine shape (positive on (0,1))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    vals = np.sin(np.pi * x)
    vals = np.clip(vals, 0.0, None)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _triangular_distribution(n: int) -> np.ndarray:
    """Symmetric triangular shape, peak in the middle."""
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    vals = 1.0 - np.abs(idx - mid) / (mid + 1)
    vals = np.clip(vals, 0.0, None)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _dirichlet_distribution(n: int, concentration: float) -> np.ndarray:
    """Random Dirichlet draw with common concentration."""
    g = np.random.gamma(concentration, 1.0, size=n)
    if g.sum() == 0:
        g = np.ones_like(g)
    return g / g.sum()


# ----------------------------------------------------------------------
def _cheap_score(a: np.ndarray) -> float:
    """Exact objective when sum(a)=1: 2·n·max(a*a)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate of max(conv).
    a   : current probability vector (sum=1)
    conv: self‑convolution of a (length 2n‑1)
    alpha: temperature multiplier (larger → sharper)
    """
    n = a.shape[0]
    cur_max = conv.max()
    exp_term = np.exp(alpha * (conv - cur_max))
    w = exp_term / exp_term.sum()
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv): average of the gradients at all
    active (max‑attaining) shift indices.
    """
    n = a.shape[0]
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (remove double counting a_i * a_i for even shifts)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad
