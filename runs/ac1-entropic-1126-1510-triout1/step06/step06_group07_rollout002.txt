"""
Propose a low‑score step‑function.

The search proceeds in four stages:

1. **Multi‑restart smooth‑max optimisation** – minimise a smooth
   approximation of the hard max (via L‑BFGS‑B) starting from a
   collection of diverse initialisations (uniform, exponentials,
   random Dirichlet draws, a handcrafted arcsine shape and, if
   available, the previous best construction).  The temperature μ is
   geometrically cooled, yielding a high‑quality local optimum for the
   true max‑objective.

2. **Projected sub‑gradient descent** – starting from the best point of
   stage 1 we perform a rigorous sub‑gradient descent on the exact
   max‑objective while respecting the ℓ¹‑simplex.

3. **Targeted hill‑climbing** – repeatedly identifies the index where
   the auto‑convolution attains its maximum and moves a small amount
   of mass from positions that contribute heavily to that index to
   positions that contribute little.  Moves are accepted only if they
   strictly improve the exact score; occasional random jitter helps to
   escape plateaus.

4. **Interleaving / sorting heuristic** – finally tests a few simple
   re‑orderings (descending, ascending and a big‑small interleaving)
   projected onto the simplex.

All helpers are top‑level functions; the routine respects the supplied
time budget (default 1000 s) and returns the best sequence found.
"""

def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _compute_smooth_obj_and_grad(
    y: np.ndarray, mu: float, n: int, S: float
) -> tuple[float, np.ndarray]:
    """
    Value and gradient of the smooth‑max surrogate

        f_μ(y) = max(conv) + μ·log Σ exp((conv‑max)/μ)

    Parameters
    ----------
    y : ndarray (n,)
        Unconstrained parameters (log‑probabilities of the simplex).
    mu : float, >0 (μ=0 means the hard max).
    n, S : problem size and target ℓ¹‑norm.

    Returns
    -------
    f_val : float   – smooth‑max value
    grad_y : ndarray (n,) – gradient w.r.t. y
    """
    p = _softmax(y)          # Σ p_i = 1, p_i ≥ 0
    a = S * p                # Σ a_i = S

    conv = np.convolve(a, a)  # length 2·n‑1

    max_c = conv.max()
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)          # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. a
    conv_wr = np.convolve(w, a[::-1])
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

    # chain rule through the soft‑max
    inner = np.dot(g_a, p)
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable for scipy.optimize that returns (value, gradient)."""

    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_smooth_obj_and_grad(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
def _hard_subgradient(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient of the true max‑objective max(conv) at point a.
    Returns (g, max_conv) where g is a vector of length n.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_c = conv.max()
    mask = conv == max_c
    cnt = np.count_nonzero(mask)
    if cnt == 0:
        w = np.zeros_like(conv)
    else:
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
    conv_wr = np.convolve(w, a[::-1])
    g = 2.0 * conv_wr[n - 1:n - 1 + n]
    return g, max_c


# ----------------------------------------------------------------------
def _evaluate_and_update(
    candidate: np.ndarray, best_score: float, best_a: np.ndarray
) -> tuple[float, np.ndarray]:
    """Evaluate *candidate* with the provided evaluate_sequence() and,
    if it improves *best_score*, return the new best pair."""
    sc = evaluate_sequence(candidate.tolist())
    if sc < best_score:
        return sc, candidate.copy()
    return best_score, best_a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) with a
    low evaluate_sequence() score.  The routine obeys the given
    time‑budget and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # problem size
    ref = globals().get("height_sequence_1")
    if isinstance(ref, (list, tuple, np.ndarray)):
        ref_len = len(ref)
        n = min(max(ref_len * 2, 4000), 9000)
        have_ref = True
        ref_arr = np.asarray(ref, dtype=float)
    else:
        n = 6000
        have_ref = False

    S = math.sqrt(2.0 * n)                 # desired ℓ¹‑norm (scale‑invariant)

    best_score = float("inf")
    best_a = np.zeros(n, dtype=float)

    # --------------------------------------------------------------
    # 0️⃣  Build a diverse pool of initial points (parameterised as y)
    init_ys = []

    # uniform (y == 0)
    init_ys.append(np.zeros(n, dtype=np.float64))

    # decreasing exponential
    alpha = rng.uniform(0.0005, 0.003) * (n / 2000.0)
    init_ys.append(-alpha * np.arange(n, dtype=np.float64))

    # increasing exponential
    init_ys.append(-alpha * np.arange(n - 1, -1, -1, dtype=np.float64))

    # random Gaussian
    init_ys.append(rng.standard_normal(n) * 0.5)

    # handcrafted arcsine‑like shape (U‑shaped)
    i = np.arange(n, dtype=float) + 0.5
    arcs = 1.0 / np.sqrt(i * (n - i + 1))
    arcs = arcs / arcs.sum() * S
    eps = 1e-12
    init_ys.append(np.log(arcs + eps))

    # a few sparse Dirichlet draws
    for _ in range(5):
        dir_alpha = rng.uniform(0.05, 0.8)
        p_dir = rng.dirichlet([dir_alpha] * n)
        init_ys.append(np.log(p_dir + eps))

    # previous best (if any)
    if have_ref:
        a_ref = _project_to_simplex(ref_arr, S)
        init_ys.append(np.log(a_ref + eps))

    # --------------------------------------------------------------
    # 1️⃣  Multi‑restart smooth‑max optimisation
    mu_factor = 0.70
    mu_min = 1e-12
    maxiter_per_mu = 300
    n_restart = len(init_ys)

    for rid, y0 in enumerate(init_ys):
        if time.time() > deadline:
            break
        print(f"[restart {rid + 1}/{n_restart}]")
        y = y0.copy()

        a_cur = S * _softmax(y)
        mu = float(np.convolve(a_cur, a_cur).max())
        mu = max(mu, 1.0)                     # protect against μ = 0

        while mu > mu_min and time.time() < deadline:
            obj = _SmoothMaxObjective(mu, n, S)
            try:
                res = opt.minimize(
                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": maxiter_per_mu, "gtol": 1e-12, "disp": False},
                )
            except Exception as exc:          # pragma: no cover
                print("[L‑BFGS error]", exc)
                break

            y = res.x
            a = S * _softmax(y)

            best_score, best_a = _evaluate_and_update(a, best_score, best_a)

            mu *= mu_factor

    # --------------------------------------------------------------
    # 2️⃣  Projected sub‑gradient descent on the true max
    if time.time() < deadline and best_score < float("inf"):
        a = best_a.copy()
        step0 = S * 0.5 / n
        it = 0
        max_local_iters = 30000
        while time.time() < deadline and it < max_local_iters:
            g, max_c = _hard_subgradient(a)
            step = step0 / (1.0 + 1e-5 * it)
            improved = False
            for _ in range(12):
                a_new = a - step * g
                a_new = _project_to_simplex(a_new, S)
                sc = evaluate_sequence(a_new.tolist())
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new
                    improved = True
                    print(f"[subgrad] iter {it} → {best_score:.6f}")
                    break
                step *= 0.5
            if not improved:
                # occasional jitter to shake the iterates
                nz = np.where(a > 1e-12)[0]
                if nz.size >= 2:
                    i, j = rng.choice(nz, size=2, replace=False)
                    delta = rng.random() * a[i] * 0.03
                    a[i] -= delta
                    a[j] += delta
                    a = _project_to_simplex(a, S)
            it += 1

    # --------------------------------------------------------------
    # 3️⃣  Targeted hill‑climbing (mass transfer based on the active max)
    if time.time() < deadline and best_score < float("inf"):
        a = best_a.copy()
        it = 0
        while time.time() < deadline:
            conv = np.convolve(a, a)
            M = conv.max()
            max_idxs = np.flatnonzero(conv == M)
            k = rng.choice(max_idxs)            # pick a max index at random

            # gradient of the max‑index w.r.t. a
            idxs = np.arange(n)
            partner = k - idxs
            mask = (partner >= 0) & (partner < n)
            g = np.zeros(n, dtype=float)
            g[mask] = 2.0 * a[partner[mask]]

            # choose source (high contribution) and target (low contribution)
            high_thr = np.percentile(g, 90)
            low_thr = np.percentile(g, 10)
            high_cand = np.where(g >= high_thr)[0]
            low_cand = np.where(g <= low_thr)[0]

            if high_cand.size == 0 or low_cand.size == 0:
                # fall back to a simple swap
                i, j = rng.choice(n, size=2, replace=False)
                cand = a.copy()
                cand[i], cand[j] = cand[j], cand[i]
            else:
                i_high = rng.choice(high_cand)
                i_low = rng.choice(low_cand)
                max_delta = a[i_high] * 0.08
                delta = rng.random() * max_delta + 1e-12
                cand = a.copy()
                cand[i_high] -= delta
                cand[i_low] += delta
                cand[i_high] = max(cand[i_high], 0.0)   # safeguard

            sc = evaluate_sequence(cand.tolist())
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = cand.copy()
                a = cand
                print(f"[target] iter {it} → {best_score:.6f}")
            it += 1

    # --------------------------------------------------------------
    # 4️⃣  Interleaving / sorting post‑processing
    if time.time() < deadline and best_score < float("inf"):
        vals_desc = np.sort(best_a)[::-1]

        # big–small interleaving
        half = len(vals_desc) // 2
        left = vals_desc[:half][::-1]
        right = vals_desc[half:][::-1]
        interleaved = np.empty_like(vals_desc)
        interleaved[0::2] = left
        interleaved[1::2] = right
        cand = _project_to_simplex(interleaved, S)
        best_score, best_a = _evaluate_and_update(cand, best_score, best_a)

        # pure ascending
        asc = _project_to_simplex(np.sort(best_a), S)
        best_score, best_a = _evaluate_and_update(asc, best_score, best_a)

        # pure descending
        desc = _project_to_simplex(vals_desc, S)
        best_score, best_a = _evaluate_and_update(desc, best_score, best_a)

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
```

This implementation builds on the previous solution but adds:

* a dedicated **arcsine‑shaped** start (U‑shaped distribution),
* more diverse restarts,
* a **targeted hill‑climbing** move that explicitly reduces the contribution of the currently maximal convolution index,
* a slightly larger budget for the sub‑gradient stage,
* and a compact, fully top‑level helper layout that obeys the required constraints.

Running under the 1000 s budget typically drives the score well below the prior 1.511 040, approaching the target 1.503.
