    best_a = a.copy()

        # choose source randomly weighted by contribution to peak
        max_idx = np.random.choice(np.where(b >= cur_max - 1e-12)[0])

"""
Propose a step function via extensive parametric search, multi‑restart
smooth‑max Adam optimisation and a final simulated‑annealing local
search.  The routine explores several families (arcsine with variable
exponent, symmetric beta, exponential decay, two‑level blocks,
mixtures) on a handful of lengths, keeps the best candidate, refines it
with Adam (large smoothness parameter) and exact sub‑gradient descent,
then performs a greedy+annealing mass‑transfer search.  All vectors are
projected onto the simplex after each update.  The total run time is
kept below the provided budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth‑max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # soft‑max weights (numerically stable)
    exp_term = np.exp(alpha * (b - cur_max))
    w_sum = exp_term.sum()
    w = exp_term / w_sum
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.01,
                   alpha_max: float = 5000.0,
                   lr0: float = 0.3,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best probability vector encountered and its exact peak.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size schedule (slow decay)
        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # keep the best exact peak seen so far
        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        # continuation on the smoothness parameter
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny jitter
        if t % 400 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    One sub‑gradient step for the exact ℓ∞ objective.
    Uses a uniform weight over all indices attaining the current maximum.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Sub‑gradient descent on the exact max‑convolution objective.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        # Polyak‑like step (using diminishing factor for stability)
        step = 0.02 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        # occasional jitter
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


# ----------------------------------------------------------------------
def _two_level_family(n: int, deadline: float):
    """
    Search a simple two‑level family:
        a_i = p   for i < L   (front block)
        a_i = q   otherwise   (back block)
    with p = r·q, r≥1.  Returns the best vector found and its fast score.
    """
    best_seq = None
    best_score = float('inf')
    step_L = max(1, n // 100)          # ≈1 % steps
    # ratios (log‑space) – extended range for more freedom
    r_grid = np.logspace(0.0, 3.0, num=35)   # 1 … 1000

    for L in range(step_L, n, step_L):
        if time.time() > deadline:
            break
        denom_base = n - L
        for r in r_grid:
            denom = L * r + denom_base
            q = 1.0 / denom
            p = r * q
            seq = np.empty(n, dtype=np.float64)
            seq[:L] = p
            seq[L:] = q
            peak = _fft_self_convolve(seq).max()
            score = 2.0 * n * peak
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
    return best_seq, best_score


# ----------------------------------------------------------------------
def _simulated_annealing_search(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Greedy + simulated‑annealing mass‑transfer local search.
    Moves a small amount of mass from a source (chosen from active
    convolution terms) to a random destination.  Accepts worsening
    moves with probability exp(-Δ/T) where the temperature T cools
    linearly to zero.
    """
    a = a0.copy()
    n = len(a)
    b = _fft_self_convolve(a)
    cur_max = b.max()
    best_a = a.copy()
    best_val = cur_max
    start = time.time()
    T0 = cur_max * 0.2 + 1e-12      # initial temperature scale

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / max(deadline - start, 1e-12)
        T = T0 * (1.0 - frac)       # linear cooling

        # pick a convolution index that participates in the current peak
        max_idx = np.random.choice(np.where(b >= cur_max - 1e-12)[0])

        # indices i that can contribute to (a*a)[max_idx]
        i_low = max(0, max_idx - (n - 1))
        i_high = min(max_idx, n - 1)
        i_vals = np.arange(i_low, i_high + 1, dtype=int)
        prod = a[i_vals] * a[max_idx - i_vals]
        if prod.sum() == 0.0:
            src = np.random.randint(n)
        else:
            prob = prod / prod.sum()
            src = np.random.choice(i_vals, p=prob)

        dst = np.random.randint(n)
        if dst == src:
            continue

        max_delta = min(a[src], 0.05)       # up to 5 % of src mass
        delta = np.random.rand() * max_delta

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        b_new = _fft_self_convolve(a_new)
        new_max = b_new.max()
        dobj = new_max - cur_max

        if dobj < 0 or np.random.rand() < math.exp(-dobj / max(T, 1e-12)):
            a = a_new
            b = b_new
            cur_max = new_max
            if new_max < best_val:
                best_val = new_max
                best_a = a.copy()

        # occasional random jitter to avoid stagnation
        if np.random.rand() < 0.005:
            a += np.random.randn(n) * 1e-7
            a = _project_onto_simplex(a, s=1.0)
            b = _fft_self_convolve(a)
            cur_max = b.max()

    return best_a


# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """2·n·max(conv) – the quantity minimised before the final black‑box call."""
    return 2.0 * len(a) * _fft_self_convolve(a).max()

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Entry point.  Explores several parametric families, refines the best
    candidate with Adam, exact sub‑gradient descent and a simulated‑annealing
    local search, then returns the final list suitable for evaluate_sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Try to reuse a previously discovered construction if present.
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_seq = _project_onto_simplex(base_seq, s=1.0)
        base_n = len(base_seq)
    except Exception:
        base_seq = None
        base_n = 4000

    # ------------------------------------------------------------------
    # Lengths to scan – a handful around the current best.
    lengths = sorted(
        set(
            max(500, int(base_n * f))
            for f in (0.6, 0.8, 1.0, 1.2, 1.5, 2.0)
            if int(base_n * f) >= 500
        )
    )
    if not lengths:
        lengths = [max(500, base_n)]

    overall_best_seq = None
    overall_best_score = float('inf')

    # ------------------------------------------------------------------
    # Phase 0 – exhaustive parametric sweep (cheap fast score)
    print("[phase 0] parametric sweep")
    for n in lengths:
        if time.time() > deadline:
            break
        # ----- arcsine with variable exponent ---------------------------------
        i = np.arange(n)
        base_term = (i + 0.5) * (n - 0.5 - i) + 1e-12
        for p in np.arange(0.30, 1.20, 0.05):
            a = base_term ** (-p)
            a /= a.sum()
            sc = _fast_score(a)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[init] n={n:4d} arcsine_p={p:.2f}  fast={sc:.6f}")

        # ----- symmetric beta densities ---------------------------------------
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        for a_param in np.arange(0.10, 0.95, 0.10):
            # log‑form to avoid overflow/underflow
            log_vals = (a_param - 1.0) * np.log(x) + (a_param - 1.0) * np.log(1.0 - x)
            vals = np.exp(log_vals)
            vals[~np.isfinite(vals)] = 0.0
            if vals.sum() == 0.0:
                continue
            a = vals / vals.sum()
            sc = _fast_score(a)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[init] n={n:4d} beta_a={a_param:.2f} fast={sc:.6f}")

        # ----- exponential decays --------------------------------------------
        for lam in (0.0001, 0.00025, 0.0005, 0.001, 0.002):
            a = np.exp(-lam * np.arange(n))
            a /= a.sum()
            sc = _fast_score(a)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[init] n={n:4d} exp_λ={lam:.5f}  fast={sc:.6f}")

        # ----- mixtures of arcsine and uniform --------------------------------
        arcs = base_term ** (-0.5)
        arcs /= arcs.sum()
        uni = np.full(n, 1.0 / n, dtype=np.float64)
        for w in (0.2, 0.4, 0.6):
            a = (1.0 - w) * arcs + w * uni
            a /= a.sum()
            sc = _fast_score(a)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[init] n={n:4d} mix_{w:.1f}      fast={sc:.6f}")

        # ----- two‑level parametric family ------------------------------------
        seq, sc = _two_level_family(n, deadline)
        if seq is not None and sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = seq.copy()
            print(f"[two‑level] n={n:4d} fast={sc:.6f}")

        # ----- random vectors -------------------------------------------------
        for _ in range(3):
            a = np.random.rand(n)
            a /= a.sum()
            sc = _fast_score(a)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[rand] n={n:4d}                 fast={sc:.6f}")

        # ----- reuse previous best if same length -----------------------------
        if base_seq is not None and len(base_seq) == n:
            a = base_seq.copy()
            sc = _fast_score(a)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[prev] n={n:4d}                 fast={sc:.6f}")

    # If the sweep failed (should not happen) fall back to uniform.
    if overall_best_seq is None:
        n = max(500, base_n)
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = _fast_score(overall_best_seq)
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation on the smooth‑max surrogate.
    time_left = deadline - time.time()
    if time_left > 15.0:
        phase_end = time.time() + time_left * 0.55
        a_start = overall_best_seq.copy()
        # several restarts around the current best
        for r in range(3):
            if time.time() > phase_end:
                break
            a0 = a_start + np.random.randn(len(a_start)) * 1e-3
            a0 = np.maximum(a0, 0.0)
            a0 = _project_onto_simplex(a0)
            a_opt, _ = _adam_optimize(
                a0,
                phase_end,
                init_alpha=1.0,
                alpha_factor=1.02,
                alpha_max=20000.0,
                lr0=0.4,
            )
            sc = _fast_score(a_opt)
            if sc < overall_best_score - 1e-12:
                overall_best_score = sc
                overall_best_seq = a_opt.copy()
                print(f"[adam] restart {r} improved fast_score={sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact max sub‑gradient descent.
    time_left = deadline - time.time()
    if time_left > 8.0:
        sub_end = time.time() + time_left * 0.55
        a_refined = _refine_max(overall_best_seq, sub_end)
        sc = _fast_score(a_refined)
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = a_refined.copy()
            print(f"[subgrad] improved fast_score={sc:.6f}")

        # ----------------------------------------------------------------
        # Phase 3 – simulated‑annealing directed local search.
        phase3_end = deadline
        a_final = _simulated_annealing_search(overall_best_seq, phase3_end)
        sc = _fast_score(a_final)
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = a_final.copy()
            print(f"[anneal] improved fast_score={sc:.6f}")

    # ------------------------------------------------------------------
    # Final black‑box evaluation (the true score).
    try:
        true_score = evaluate_sequence(overall_best_seq.tolist())
        print(f"[final] evaluate_sequence score={true_score:.6f}")
        if true_score < overall_best_score:
            overall_best_score = true_score
    except Exception as exc:
        print("[final] evaluate_sequence raised:", exc)

    print(
        f"Best approx. fast_score {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
