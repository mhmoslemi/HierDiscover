    deadline = start_time + budget_s - 0.5

    # attempt to load previous height_sequence_1
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            prev = (prev + prev[::-1]) / 2
            prev = prev / prev.sum()
        else:
            prev = None
    except Exception:
        prev = None

    # Choose base length
    base_n = 1200  # moderate
    lengths = [base_n]
    # Extend lengths doubling up to max_n if time permits
    max_n = 8000
    cur_n = base_n
    while cur_n * 2 <= max_n and time.time() < deadline - 30:
        cur_n *= 2
        lengths.append(cur_n)

    overall_best = None
    overall_best_score = float('inf')

    # start from previous if available and choose appropriate length
    # We could downsample/upsample previous to each length to use as init.
    # But easier: use prev as seed for base length by resizing.
    # For each length, get init candidate from one of: previous (resized), arcsine shape, random known shapes.

    for n in lengths:
        if time.time() >= deadline: break
        # generate candidate list
        candidates = _initial_candidates(n)
        if prev is not None:
            # resize prev to n
            cand_prev = _upscale(prev, n)
            candidates.append(cand_prev)
        # Evaluate and keep best for this n
        best_local = None
        best_local_score = float('inf')
        for cand in candidates:
            cand = cand / cand.sum()
            cand = (cand + cand[::-1]) / 2
            cand = cand / cand.sum()
            sc = _max_convolution_score(cand)
            if sc < best_local_score:
                best_local_score = sc
                best_local = cand.copy()
        # Optimize with smooth descent
        a, sc = _smooth_descent_sym(best_local, deadline)
        if sc < best_local_score:
            best_local_score = sc
            a = a.copy()
        # subgradient refine
        a, sc = _exact_subgradient_refine_sym(a, deadline)
        if sc < best_local_score:
            best_local_score = sc
        # hill climb
        a, sc = _hill_climb_sym(a, deadline)
        if sc < best_local_score:
            best_local_score = sc
        # optionally annealing
        a, sc = _simulated_annealing(a, deadline)
        if sc < best_local_score:
            best_local_score = sc
        # power transform search
        a, sc = _power_transform_search(a, deadline)
        if sc < best_local_score:
            best_local_score = sc

        # keep overall best
        if best_local_score < overall_best_score:
            overall_best_score = best_local_score
            overall_best = a.copy()
            print(f"[{time.time():.1f}] New overall best length={n} score={overall_best_score:.6f}")

    # final symmetrize and scale to sum (any) for output
    if overall_best is None:
        overall_best = _distribution_uniform(800)
        overall_best_score = _max_convolution_score(overall_best)

    # ensure non-negative and finite
    overall_best = np.maximum(overall_best, 0.0)
    # scale maybe to sum = 1 (any scale fine)
    overall_best = overall_best / overall_best.sum()
    print(f"Final best score approx: {overall_best_score:.6f}, length={len(overall_best)}")
    return overall_best.tolist()

def _upscale(vec, new_n):
    old_n = len(vec)
    if old_n == new_n:
        return vec.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    new_vec = np.interp(new_x, old_x, vec)
    new_vec = np.maximum(new_vec, 0.0)
    # Normalize
    total = new_vec.sum()
    if total == 0:
        new_vec = np.full(new_n, 1.0 / new_n)
    else:
        new_vec = new_vec / total
    # Symmetrize
    new_vec = (new_vec + new_vec[::-1]) / 2.0
    new_vec = new_vec / new_vec.sum()
    return new_vec

"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.

Algorithm
---------
1. Generate a handful of analytic shapes (uniform, arcsine, beta,
   triangular, exponential) for a modest length `n0`.
2. Pick the best shape, enforce exact symmetry `a_i = a_{n‑1‑i}`
   and normalise to sum = 1.
3. Apply a smoothed log‑sum‑exp descent (Adam optimiser) with a
   schedule of increasing smoothness `α`.  After each Adam step the
   vector is projected onto the simplex and symmetrised.
4. Refine with an exact max‑sub‑gradient method (momentum,
   diminishing step size) – again keeping the vector symmetric.
5. Perform a cheap random‑transfer hill‑climber that respects the
   simplex and the symmetry.
6. Optionally run a short simulated‑annealing phase to escape shallow
   plateaux.
7. Upscale the current distribution to a larger length (doubling each
   time) by linear interpolation, symmetrise and re‑optimise from
   step 3 onward.  Continue while time permits (up to ~8000 points).
8. After the final length, try a small family of power‑transform
   perturbations `a_i ← a_i^p` (p∈[0.5,2]) to possibly lower the
   maximum of the self‑convolution.
9. Return the best vector found as a plain Python list of floats.

All helper functions are top‑level, use only NumPy, `time` and `math`,
and respect the global time budget (default 1000 s).  The code works
without any file I/O and will be called with `evaluate_sequence`
already imported.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _project_and_symmetrize(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project onto simplex and enforce a_i = a_{n‑1‑i}."""
    w = _project_onto_simplex(v, s)
    w = (w + w[::-1]) / 2.0          # symmetry
    # the averaging preserves the sum, but project once more for safety
    w = _project_onto_simplex(w, s)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) assuming `a` is normalised to sum = 1.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    """Uniform distribution (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _initial_candidates(n: int) -> list[np.ndarray]:
    """Generate a handful of plausible start vectors for a given length."""
    cand = [
        _distribution_uniform(n),
        _distribution_arcsine(n),
        _distribution_triangular(n),
        _distribution_exponential(n, lam=0.003),
        _distribution_exponential(n, lam=0.01),
    ]
    for p in (0.2, 0.5, 1.0, 2.0):
        cand.append(_distribution_beta_sym(n, p))
    # a random sparse start (few non‑zeros)
    rnd = np.zeros(n, dtype=np.float64)
    k = max(2, n // 200)                # about 0.5 % non‑zeros
    idx = np.random.choice(n, k, replace=False)
    rnd[idx] = np.random.rand(k)
    if rnd.sum() > 0:
        rnd /= rnd.sum()
    cand.append(rnd)
    return cand


# ----------------------------------------------------------------------
def _smooth_descent_sym(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth log‑sum‑exp approximation of the max
    convolution.  Symmetry is enforced after each projection.
    Returns (best_vector, best_score).
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.15
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # smoothing schedule (α, number of steps)
    schedule = [
        (1.0, 2000),
        (5.0, 2000),
        (20.0, 2000),
        (100.0, 2000),
        (500.0, 2000),
    ]

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score
            t += 1

            # self‑convolution
            b = _fft_self_convolve(a)

            # softmax weights (log‑sum‑exp trick)
            maxb = b.max()
            exp_terms = np.exp(alpha * (b - maxb))
            w = exp_terms / exp_terms.sum()

            # gradient = 2 * conv_rev(a_rev, w) (first n entries)
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # project onto simplex and enforce symmetry
            a = _project_and_symmetrize(a, 1.0)

            if t % 100 == 0:
                cur_score = _max_convolution_score(a)
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_seq = a.copy()
    return best_seq, best_score


def _exact_subgradient_refine_sym(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Exact max‑subgradient refinement with momentum, keeping symmetry.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)

    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps)

        w = np.zeros_like(b)
        w[active] = 1.0
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # momentum / diminishing step
        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / np.sqrt(iteration)
        a = a - eta * velocity
        a = _project_and_symmetrize(a, 1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            # print(f"[{time.time():.1f}] subgrad iter={iteration} new best {best_score:.6f}")

        # safety break – we don’t want endless loops if time is huge
        if iteration >= 30000:
            break
    return best_seq, best_score


def _hill_climb_sym(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple random‑transfer hill climbing, keeping the distribution symmetric.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)
    cur_score = best_score

    max_transfer = 0.07          # ≤ 7 % of total mass per trial
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        # exponential decay of the transfer magnitude
        transfer = max_transfer * (0.5 ** (iteration / 4000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # enforce symmetry and simplex
        a_new = _project_and_symmetrize(a_new, 1.0)

        new_score = _max_convolution_score(a_new)
        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a_new.copy()
                # print(f"[{time.time():.1f}] hill iter={iteration} improved best {best_score:.6f}")

        # occasional jitter to escape plateaux
        if iteration % 5000 == 0 and np.random.rand() < 0.2:
            pert = np.random.randn(n) * 1e-4
            a = best_seq + pert
            a = _project_and_symmetrize(a, 1.0)
            cur_score = _max_convolution_score(a)

    return best_seq, best_score


def _simulated_annealing(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    A lightweight simulated‑annealing phase (still respecting symmetry).
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)
    cur_seq = a.copy()
    cur_score = best_score

    start = time.time()
    T0 = 0.01                       # initial temperature

    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        if total <= 0:
            break
        T = T0 * (1 - elapsed / total)   # linear cooling

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = np.random.rand() * min(0.05, cur_seq[i])

        a_new = cur_seq.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_and_symmetrize(a_new, 1.0)

        new_score = _max_convolution_score(a_new)
        if new_score < cur_score or (T > 0 and np.random.rand() < math.exp(-(new_score - cur_score) / T)):
            cur_seq = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                # print(f"[{time.time():.1f}] SA improved best {best_score:.6f}")

    return best_seq, best_score


def _upscale(vec: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from length len(vec) to `new_n`, followed by
    normalisation and symmetry enforcement.
    """
    old_n = vec.shape[0]
    if new_n == old_n:
        return vec.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_vec = np.interp(new_x, old_x, vec)
    new_vec = np.maximum(new_vec, 0.0)

    if new_vec.sum() == 0:
        new_vec = np.full(new_n, 1.0 / new_n, dtype=np.float64)
    else:
        new_vec = new_vec / new_vec.sum()

    new_vec = (new_vec + new_vec[::-1]) / 2.0
    new_vec = new_vec / new_vec.sum()
    return new_vec


def _power_transform_search(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Scan a small family of power transforms a_i ← a_i^p (p∈[0.5,2])
    to see if the objective can be lowered further.
    """
    best = a.copy()
    best_score = _max_convolution_score(best)

    p_vals = np.arange(0.5, 2.01, 0.1)
    for p in p_vals:
        if time.time() >= deadline:
            break
        transformed = np.power(best, p)
        if transformed.sum() == 0:
            continue
        transformed = transformed / transformed.sum()
        transformed = (transformed + transformed[::-1]) / 2.0
        transformed = transformed / transformed.sum()
        score = _max_convolution_score(transformed)
        if score < best_score:
            best_score = score
            best = transformed.copy()
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence found (as a Python list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Try to reuse a previous construction if available.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            prev = (prev + prev[::-1]) / 2.0
            prev = prev / prev.sum()
        else:
            prev = None
    except Exception:
        prev = None

    # ------------------------------------------------------------------
    # Choose a modest base length and grow it geometrically while time permits.
    base_n = 1200
    max_n = 8000
    lengths = [base_n]
    cur_n = base_n
    while cur_n * 2 <= max_n and time.time() < deadline - 30:
        cur_n *= 2
        lengths.append(cur_n)

    overall_best_seq = None
    overall_best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break

        # ---- generate initial candidates for the current length ----
        candidates = _initial_candidates(n)
        if prev is not None:
            cand_prev = _upscale(prev, n)
            candidates.append(cand_prev)

        # keep the best candidate after a cheap evaluation
        local_best = None
        local_best_score = float("inf")
        for cand in candidates:
            cand = cand / cand.sum()
            cand = (cand + cand[::-1]) / 2.0
            cand = cand / cand.sum()
            sc = _max_convolution_score(cand)
            if sc < local_best_score:
                local_best_score = sc
                local_best = cand.copy()

        # ---- optimisation pipeline on the chosen length ----
        a, sc = _smooth_descent_sym(local_best, deadline)
        if sc < local_best_score:
            local_best_score = sc
            a = a.copy()

        a, sc = _exact_subgradient_refine_sym(a, deadline)
        if sc < local_best_score:
            local_best_score = sc

        a, sc = _hill_climb_sym(a, deadline)
        if sc < local_best_score:
            local_best_score = sc

        a, sc = _simulated_annealing(a, deadline)
        if sc < local_best_score:
            local_best_score = sc

        a, sc = _power_transform_search(a, deadline)
        if sc < local_best_score:
            local_best_score = sc

        # ---- record overall best ----
        if local_best_score < overall_best_score:
            overall_best_score = local_best_score
            overall_best_seq = a.copy()
            print(f"[{time.time():.1f}] NEW OVERALL BEST length={n} score={overall_best_score:.6f}")

        # prepare `prev` for the next (larger) length
        prev = a.copy()

    # ------------------------------------------------------------------
    # Fallback in the very unlikely case that nothing was found.
    if overall_best_seq is None:
        overall_best_seq = _distribution_uniform(800)
        overall_best_score = _max_convolution_score(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # final safety normalisation
    overall_best_seq = np.maximum(overall_best_seq, 0.0)
    overall_best_seq = overall_best_seq / overall_best_seq.sum()

    print(f"FINAL BEST score≈{overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
