def _directed_local_search_improved(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max
    while time.time()<deadline:
        # compute gradient for active set
        grad, cur_max = _max_subgradient_step(a) # grad from subgradient step
        # probability for src based on grad magnitude
        if grad.sum()==0:
            src = np.random.randint(n)
        else:
            prob = grad / grad.sum()
            src = np.random.choice(n, p=prob)
        # choose dst from few lowest gradient indices
        # Exclude src
        # sort indices by grad
        sorted_idx = np.argsort(grad)
        # pick random among lowest 10% maybe
        low_idxs = sorted_idx[: max(1, n//50)]
        if src in low_idxs:
            # ensure different
            low_idxs = low_idxs[low_idxs != src]
        if low_idxs.size == 0:
            dst = (src + np.random.randint(1, n)) % n
        else:
            dst = np.random.choice(low_idxs)
        # delta as fraction of src mass up to max 5% or maybe 2%
        max_delta = min(a[src], 0.05)
        delta = np.random.rand() * max_delta
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        b_new = _fft_self_convolve(a_new)
        new_max = b_new.max()
        if new_max < best_max - 1e-15:
            best_max = new_max
            best_a = a_new.copy()
        if new_max < cur_max - 1e-15:
            a = a_new
            b = b_new
            cur_max = new_max
        # occasional jitter
        if np.random.rand() < 0.003:
            a += np.random.randn(n) * 1e-7
            a = _project_onto_simplex(a, s=1.0)
            b = _fft_self_convolve(a)
            cur_max = b.max()
    return best_a

"""
Propose a low‑score non‑negative step function.

Algorithm overview
------------------
1. **Shape sampling** – generate a set of candidate vectors from simple
   analytic families (arcsine, symmetric and asymmetric Beta densities,
   random Dirichlet).  For each length we evaluate the cheap proxy
   `fast_score = 2·n·max(conv(a,a))` and keep the best vector.

2. **Local refinement** – starting from the best few seeds we run
   projected sub‑gradient descent on the exact ℓ∞‑objective
   (`_refine_max`).  Afterward a *directed* mass‑transfer search
   (`_directed_local_search_improved`) actively moves mass from indices
   that contribute most to the current peak toward indices that contribute
   little.

3. **Optional Adam fine‑tuning** – if time permits we run the smooth‑max
   Adam optimiser from the current best vector and finish with a short
   sub‑gradient + directed search.

All vectors are kept on the probability simplex (sum = 1) and are
projected after each update.  The routine respects a 1000 s budget and
returns the best sequence found, ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth‑max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # numerically stable soft‑max weights
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.01,
                   alpha_max: float = 5000.0,
                   lr0: float = 0.3,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best vector (on the simplex) and its exact peak.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size (slow decay)
        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        if t % 400 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    One sub‑gradient step for the exact ℓ∞ objective.
    Returns (grad, cur_max) where grad is a sub‑gradient of the max‑convolution.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max

    """
    Sub‑gradient descent on the exact max‑convolution objective.

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.02 / np.sqrt(it)               # diminishing step size
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a

def _directed_local_search_improved(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Greedy mass‑transfer search that always moves mass from a high‑gradient
    index (large contribution to the current peak) to a low‑gradient one.
    Accepts only improving moves.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        # obtain sub‑gradient for the active set
        grad, cur_max = _max_subgradient_step(a)

        # source = index with probability proportional to gradient magnitude
        if grad.sum() == 0:
            src = np.random.randint(n)
        else:
            prob = grad / grad.sum()
            src = np.random.choice(n, p=prob)

        # destination = one of the indices with smallest gradient values
        low_candidates = np.argsort(grad)[: max(1, n // 50)]
        if src in low_candidates:
            low_candidates = low_candidates[low_candidates != src]
        if low_candidates.size == 0:
            dst = (src + np.random.randint(1, n)) % n
        else:
            dst = np.random.choice(low_candidates)

        # move a modest fraction of the source mass (up to 5 %)
        max_delta = min(a[src], 0.05)
        delta = np.random.rand() * max_delta

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        b_new = _fft_self_convolve(a_new)
        new_max = b_new.max()

        if new_max < best_max - 1e-15:
            best_max = new_max
            best_a = a_new.copy()

        if new_max < cur_max - 1e-15:
            a = a_new
            b = b_new
            cur_max = new_max

        if np.random.rand() < 0.003:
            a += np.random.randn(n) * 1e-7
            a = _project_onto_simplex(a, s=1.0)
            b = _fft_self_convolve(a)
            cur_max = b.max()

    return best_a

def _generate_arcsine_shape(n: int) -> np.ndarray:
    """Arcsine‑type shape a_i ∝ 1/√((i+½)(n−i−½)), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑density shape on the discrete grid, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # work in log‑space for stability
    log_vals = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    # replace any NaNs/Infs (can happen near the boundaries)
    vals[~np.isfinite(vals)] = 0.0
    s = vals.sum()
    if s == 0.0:
        # fall back to uniform if the density collapses
        return np.full(n, 1.0 / n, dtype=np.float64)
    vals /= s
    return vals
