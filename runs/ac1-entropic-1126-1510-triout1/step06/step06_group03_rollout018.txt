"""
Propose a step‑function (non‑negative height sequence) with a low
evaluate_sequence score.

The core is a projected Adam optimiser on the soft‑max (logit)
parameterisation – the same idea as the reference solution but with
more iterations and a few extra restarts.  After a good candidate is
found we improve it by adding a uniform offset (all entries +ε).  This
increases the total ℓ₁‑norm while only modestly raising the maximal
convolution value, thus lowering the final objective

    2·n·max(a*a) / (sum(a)**2).

A cheap 1‑D golden‑section search finds the optimal total offset.
Finally a few sub‑gradient steps (projected onto the simplex) polish
the result.

The routine respects the supplied time budget (default 1000 s) and
returns a plain Python list ready for evaluate_sequence().
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a*a (exact, using FFT)."""
    return float(_fft_self_convolve_full(a).max())


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of max_k (a*a)_k.
    Returns (grad, cur_max), where grad has the same shape as a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    # active set – tolerance scales with cur_max
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(conv >= cur_max - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    # gradient (still contains factor 2)
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals

def _adam_optimize_from_logits(
    x0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 8000,
    lr0: float = 0.13,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the logit parametrisation of the simplex.
    Returns the best probability vector (sum = 1) found and its
    raw score 2·n·max(a*a).
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    best_a = None
    best_score = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        # soft‑max (stable)
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()          # Σa = 1, a ≥ 0

        cur_max = _fft_self_convolve_max(a)
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it}  raw={best_score:.6f}")

        grad_a, _ = _max_subgradient_step(a)

        # chain rule through the soft‑max
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % 1200 == 0:
            x += np.random.randn(*x.shape) * 0.008

    # fallback
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_max(best_a)

    return best_a, best_score

def _optimise_length(
    n: int,
    deadline: float,
    prev_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """Optimise a sequence of length n with several random restarts."""
    init_logits = []

    # warm‑start (interpolation of previous best)
    if prev_seq is not None:
        cand = _interpolate_to_length(prev_seq, n)
        init_logits.append(np.log(cand + 1e-12))

    # uniform start
    init_logits.append(np.zeros(n, dtype=np.float64))

    # a few Dirichlet draws
    for _ in range(4):
        alpha = np.random.rand() * 4.0 + 0.1
        sample = np.random.gamma(alpha, 1.0, size=n)
        sample /= sample.sum()
        init_logits.append(np.log(sample + 1e-12))

    best_local_seq = None
    best_local_score = float("inf")
    for ix, x0 in enumerate(init_logits):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] length {n}, start {ix+1}/{len(init_logits)}")
        seq, score = _adam_optimize_from_logits(x0, n, deadline)
        if score < best_local_score:
            best_local_score = score
            best_local_seq = seq.copy()

    if best_local_seq is None:
        best_local_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_local_score = 2.0 * n * _fft_self_convolve_max(best_local_seq)

    return best_local_seq, best_local_score


# ----------------------------------------------------------------------
def _raw_score(seq: np.ndarray) -> float:
    """Exact objective value (without clamping) for a given sequence."""
    n = len(seq)
    sum_a = float(seq.sum())
    if sum_a < 1e-12:
        return float("inf")
    max_b = _fft_self_convolve_max(seq)
    return 2.0 * n * max_b / (sum_a * sum_a)

def _optimal_uniform_offset(seq: np.ndarray, max_t: float = 3.0) -> np.ndarray:
    """
    Find the optimal total uniform mass t (added evenly to every entry)
    that minimises the raw score.  Returns a new sequence = seq + t/n.
    """
    n = len(seq)
    base_score = _raw_score(seq)
    best_seq = seq.copy()
    best_score = base_score

    # golden‑section search on t ∈ [0, max_t]
    phi = (math.sqrt(5) - 1) / 2  # 0.618...
    lo, hi = 0.0, max_t
    # 30 iterations → sufficient precision for our purpose
    for _ in range(30):
        t1 = lo + (1 - phi) * (hi - lo)
        t2 = lo + phi * (hi - lo)
        eps1 = t1 / n
        eps2 = t2 / n
        seq1 = seq + eps1
        seq2 = seq + eps2
        s1 = _raw_score(seq1)
        s2 = _raw_score(seq2)

        if s1 < s2:
            hi = t2
            if s1 < best_score:
                best_score, best_seq = s1, seq1
        else:
            lo = t1
            if s2 < best_score:
                best_score, best_seq = s2, seq2

    return best_seq

        best_seq = _optimal_uniform_offset(best_seq, max_t=2.0)

    # ------------------------------------------------------------------
    final_list = best_seq.tolist()
    final_score = evaluate_sequence(final_list)
    print(f"\nFinal evaluated score: {final_score:.6f} (length={len(final_list)})")
    return final_list
