def _nesterov_subgradient(a0, deadline):

    a = a0.copy()
    a_prev = a.copy()
    t_cur = 0
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        # compute momentum parameter
        t_next = (1 + np.sqrt(1+4*t_cur**2))/2
        beta = (t_cur-1)/t_next
        # compute extrapolated point y = a + beta*(a - a_prev)
        y = a + beta * (a - a_prev)
        # project y onto simplex (should be pushing out of feasible set)
        y = _project_onto_simplex(y, s=1.0)

        # compute subgradient at y
        grad, cur_max = _max_subgradient_step(y)  # return grad and max (though grad from earlier implementation) Actually _max_subgradient_step returns both gradient and cur_max. So we can call that function.

        # step size schedule
        step = 0.02 / np.sqrt(iteration)

        # update a_prev, a
        a_prev = a
        a = y - step * grad
        a = _project_onto_simplex(a, s=1.0)
        a = np.maximum(a, 0)  # already enforced

        # track best
        if cur_max < best_peak:
            best_peak = cur_max
            best_a = a.copy()
        t_cur = t_next

        # periodic jitter

    return best_a

def _annealed_local_search(a0, deadline):

    n = len(a)
    b = _fft_self_convolve(a)
    cur_max = b.max()
    best_a = a.copy()

            src = np.random.choice(i_vals, p=probs)

        # choose dest biased away from src; e.g., probability inversely proportional to a + epsilon
        dest_probs = 1.0 / (a + 1e-12)
        dest_probs[src] = 0.0
        dest_probs /= dest_probs.sum()
        dst = np.random.choice(n, p=dest_probs)

        # move delta
        max_delta = min(a[src], 0.05)  # up to 5% mass
        delta = np.random.rand() * max_delta

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        b_new = _fft_self_convolve(a_new)
        new_max = b_new.max()

        # accept with Metropolis
        if new_max < cur_max:
            accept = True
        else:

            accept = np.random.rand() < np.exp(-(new_max - cur_max) / T)

def _annealed_local_search(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max
    # temperature schedule
    T0 = cur_max * 0.01  # maybe adjust
    start = time.time()
    while time.time() < deadline:
        # compute gradually decreasing temperature
        elapsed = time.time() - start
        fraction = max(0.0, 1.0 - elapsed / (deadline - start))
        T = T0 * fraction
        # pick active index near peak
        active = np.where(b >= cur_max - 1e-12)[0]
        max_idx = np.random.choice(active)
        i_low = max(0, max_idx - (n - 1))
        i_high = min(max_idx, n - 1)
        i_vals = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[i_vals] * a[max_idx - i_vals]
        if contrib.sum() == 0:
            src = np.random.randint(n)
        else:
            prob = contrib / contrib.sum()
            src = np.random.choice(i_vals, p=prob)

        # destination selection: low a probability
        inv = 1.0 / (a + 1e-12)
        inv[src] = 0.0
        prob_dst = inv / inv.sum()
        dst = np.random.choice(n, p=prob_dst)

        # delta random up to fraction of source
        max_delta = min(a[src], 0.05)
        delta = np.random.rand() * max_delta

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        b_new = _fft_self_convolve(a_new)
        new_max = b_new.max()

        # Metropolis acceptance
        if new_max < cur_max:
            accept = True
        else:
            if T > 0:
                accept = np.random.rand() < np.exp(-(new_max - cur_max) / T)
            else:
                accept = False

        if accept:
            a = a_new
            b = b_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()

        # occasional jitter
        if np.random.rand() < 0.001:
            a += np.random.randn(n) * 1e-7
            a = _project_onto_simplex(a, s=1.0)
            b = _fft_self_convolve(a)
            cur_max = b.max()
    return best_a

"""
Propose a low‑score step‑function.

Approach
--------
1. **Broad family sampling** – For several lengths we generate many
   simple families (arcsine, Beta, exponential, uniform, random,
   two‑level blocks, mixes).  Fast score = 2·n·max(conv) is computed and
   the best ~15 candidates are kept.

2. **Refinement pipeline** – Each retained candidate is refined independently
   using the remaining time budget:
        * Adam optimisation of a smooth‑max surrogate of the ℓ∞‑convolution.
        * Nesterov‑accelerated sub‑gradient descent on the exact max‑convolution.
        * Simulated‑annealing style local mass‑transfer search that accepts
          occasional uphill moves.

3. **Final selection** – The refined candidate with the smallest exact
   `evaluate_sequence` value is returned.

All vectors are projected onto the simplex Σaᵢ = 1 after every update,
hence the denominator of `evaluate_sequence` stays equal to 1 and the
objective reduces to `2·n·max(a*a)`.  The routine respects a 1000 s wall‑clock
budget and prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth‑max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)

    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # soft‑max weights (numerically stable)
    exp_term = np.exp(alpha * (b - cur_max))
    w_sum = exp_term.sum()
    w = exp_term / w_sum
    # convolution of reversed a with the weight vector
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3·n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 10.0,
                   alpha_factor: float = 1.02,
                   alpha_max: float = 5000.0,
                   lr0: float = 0.5,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best vector encountered (projected onto the simplex) and its exact peak.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size (slow decay)
        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # keep the best exact peak seen so far
        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        # continuation on the smoothness parameter
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional jitter to escape plateaus
        if t % 400 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of the exact ℓ∞‑convolution objective at a.
    Returns (grad, cur_max).  The gradient is an equal weight on all
    indices attaining the current maximum.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


def _nesterov_refine(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Accelerated sub‑gradient descent (Nesterov) on the exact max‑convolution.
    """
    a = a0.copy()
    a_prev = a.copy()
    t_curr = 0.0
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0

    while time.time() < deadline:
        it += 1

        # Nesterov momentum parameters
        t_next = (1.0 + math.sqrt(1.0 + 4.0 * t_curr * t_curr)) / 2.0
        beta = (t_curr - 1.0) / t_next if t_curr > 0.0 else 0.0
        t_curr = t_next

        # extrapolation
        y = a + beta * (a - a_prev)
        y = _project_onto_simplex(y, s=1.0)

        # sub‑gradient at extrapolated point
        grad, cur_max = _max_subgradient_step(y)

        # diminishing step size
        step = 0.02 / np.sqrt(it)

        a_next = y - step * grad
        a_next = _project_onto_simplex(a_next, s=1.0)

        # track best
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a_next.copy()

        # update
        a_prev = a
        a = a_next

        # occasional jitter
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(len(a)) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a

def _annealed_local_search(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Greedy mass‑transfer search with a Metropolis acceptance rule (simulated annealing).
    Moves a small fraction of mass from an index heavily participating in the current peak
    towards an index with low probability weight.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    # temperature schedule: start proportional to the current peak
    T0 = max(1e-8, cur_max * 0.02)
    start = time.time()
    total = max(deadline - start, 1e-9)

    while time.time() < deadline:
        # linear cooling
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / total)

        # pick a peak index
        active_idxs = np.where(b >= cur_max - 1e-12)[0]
        max_idx = np.random.choice(active_idxs)

        # indices that contribute to this convolution term
        i_low = max(0, max_idx - (n - 1))
        i_high = min(max_idx, n - 1)
        i_vals = np.arange(i_low, i_high + 1, dtype=int)

        prod = a[i_vals] * a[max_idx - i_vals]
        if prod.sum() == 0.0:
            src = np.random.randint(n)
        else:
            probs = prod / prod.sum()
            src = np.random.choice(i_vals, p=probs)

        # choose a destination biased towards low‑value entries
        inv = 1.0 / (a + 1e-12)
        inv[src] = 0.0
        dst = np.random.choice(n, p=inv / inv.sum())

        # move a modest fraction of mass
        max_delta = min(a[src], 0.05)          # up to 5 % of source mass
        delta = np.random.rand() * max_delta

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        b_new = _fft_self_convolve(a_new)
        new_max = b_new.max()

        # Metropolis acceptance
        if new_max <= cur_max:
            accept = True
        else:
            accept = (T > 0.0) and (np.random.rand() < math.exp(-(new_max - cur_max) / T))

        if accept:
            a = a_new
            b = b_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()

        # occasional tiny jitter to keep the walk alive
        if np.random.rand() < 0.001:
            a += np.random.randn(n) * 1e-7
            a = _project_onto_simplex(a, s=1.0)
            b = _fft_self_convolve(a)
            cur_max = b.max()

    return best_a

def _refine_candidate(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Full refinement pipeline for a single candidate.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0

    # 35 % Adam, 35 % Nesterov sub‑gradient, rest local search
    adam_deadline = now + remaining * 0.35
    a_adam, _ = _adam_optimize(
        a0,
        adam_deadline,
        init_alpha=10.0,
        alpha_factor=1.02,
        alpha_max=5000.0,
        lr0=0.5,
    )

    now = time.time()
    if now >= deadline:
        return a_adam
    sub_deadline = now + (deadline - now) * 0.35
    a_sub = _nesterov_refine(a_adam, sub_deadline)

    # final annealed local search
    a_final = _annealed_local_search(a_sub, deadline)
    return a_final

def _two_level_family(n: int, deadline: float):
    """
    Simple two‑level parametric family:
        a_i = p   for i < L,   a_i = q otherwise,
    with ratio r = p/q ≥ 1.
    """
    best_seq = None
    best_score = float('inf')
    step_L = max(1, n // 100)          # roughly 1 % step
    r_grid = np.logspace(0.0, 2.3, num=25)   # 1 … ≈200

    for L in range(step_L, n, step_L):
        if time.time() > deadline:
            break
        denom_base = n - L
        for r in r_grid:
            denom = L * r + denom_base
            q = 1.0 / denom
            p = r * q
            seq = np.empty(n, dtype=np.float64)
            seq[:L] = p
            seq[L:] = q
            peak = _fft_self_convolve(seq).max()
            score = 2.0 * n * peak
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
    return best_seq, best_score
