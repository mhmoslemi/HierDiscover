"""
Propose a low‑score step‑function.

The search proceeds in several stages:

1. **Very coarse multi‑start** on a small size (n≈512).  A wide
   family of random and parametric distributions (Dirichlet,
   Beta, arcsine, power‑law, exponential, mixtures) is sampled.
   The cheapest “fast” score 2·n·max(conv) is used to keep the best
   candidate.

2. **Continuous refinement** of that candidate with the Adam optimiser
   on a smooth‑max surrogate, followed by an exact ℓ∞ sub‑gradient
   descent.

3. **Multi‑scale up‑sampling** – the refined vector is interpolated to
   larger lengths (≈2 k and finally ≈4 k–8 k) and each scale is again
   refined with Adam + sub‑gradient.  Upsampling lets the optimisation
   escape local minima of a coarse discretisation.

4. **Optional LP boost** (if cvxpy is available and the vector is not too
   large) – the classic “dual‑widening’’ step that maximises the ℓ₁‑norm
   of a partner vector under the current convolution bound.

5. **Simulated‑annealing polish** – cheap stochastic moves that
   occasionally accept uphill steps, terminating exactly at the overall
   deadline.

All steps respect the simplex (non‑negative entries, Σ aᵢ = 1).  The
final vector is returned as a plain list ready for ``evaluate_sequence``.
"""

import math
import time
import numpy as np
from numpy.fft import rfft, irfft
import scipy.linalg

# optional LP support
try:
    import cvxpy as cp
    _cvxpy_available = True
except Exception:  # pragma: no cover
    _cvxpy_available = False


def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_vals = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    vals[~np.isfinite(vals)] = 0.0
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


def _powerlaw_boundary_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5) + 1e-12) ** (-p)
    vals /= vals.sum()
    return vals


def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """Gradient of Lα(a) = (1/α)·log Σ exp(α·(a*a)_k)."""
    b = _fft_self_convolve(a)
    cur_max = b.max()
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.01,
                   alpha_max: float = 2000.0,
                   lr0: float = 0.25,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """Adam optimisation of the smooth‑max surrogate."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional jitter to escape plateaus
        if t % 500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


def _max_subgradient_step(a: np.ndarray):
    """Exact ℓ∞ sub‑gradient step."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Short exact‑max refinement using sub‑gradient descent."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.02 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        if it % 400 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


def _simulated_annealing(a0: np.ndarray, deadline: float):
    """Light simulated‑annealing polishing."""
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_peak = _fft_self_convolve(cur_a).max()
    best_a = cur_a.copy()
    best_peak = cur_peak

    start = time.time()
    total_time = max(deadline - start, 1e-6)

    start_temp = max(cur_peak, 1e-3) * 0.5
    final_temp = start_temp * 1e-5
    if final_temp <= 0.0:
        final_temp = 1e-9
    cooling_rate = -math.log(final_temp / start_temp) / total_time

    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1
        src = np.random.randint(n)
        if cur_a[src] <= 1e-12:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue
        max_delta = min(cur_a[src], 0.05)
        delta = np.random.rand() * max_delta

        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_peak = _fft_self_convolve(a_new).max()
        delta_obj = new_peak - cur_peak

        elapsed = time.time() - start
        temperature = start_temp * math.exp(-cooling_rate * elapsed)

        accept = False
        if delta_obj < 0.0:
            accept = True
        else:
            prob = math.exp(-delta_obj / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_a = a_new
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_a = cur_a.copy()

        if iter_cnt % 20000 == 0 and np.random.rand() < 0.05:
            cur_a = best_a.copy()
            cur_peak = best_peak

    return best_a, best_peak


def _lp_improve(a: np.ndarray,
                max_iters: int = 2,
                mix_t: float = 0.5,
                inner_deadline: float = None) -> np.ndarray:
    """LP‑based dual‑widening improvement (only for moderate n)."""
    if not _cvxpy_available:
        return a

    n = a.shape[0]
    for _ in range(max_iters):
        if inner_deadline is not None and time.time() > inner_deadline:
            break
        M = _fft_self_convolve(a).max()
        col = np.r_[a, np.zeros(n - 1)]
        row = np.r_[a[0], np.zeros(n - 1)]
        A = scipy.linalg.toeplitz(col, row)  # dense (2n‑1)×n matrix

        b_var = cp.Variable(n, nonneg=True)
        constraints = [A @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception:
            break
        if b_var.value is None:
            break
        b_vec = np.maximum(b_var.value, 0.0)
        s = b_vec.sum()
        if s <= 1e-12:
            break
        g = b_vec / s
        a = (1.0 - mix_t) * a + mix_t * g
        a = _project_onto_simplex(a, s=1.0)
    return a


def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a non‑negative sequence to a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """Search for a step‑function with low evaluate_sequence score.

    The function respects the 1000 s budget and returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Possibly start from a known construction.
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_seq = _project_onto_simplex(base_seq, s=1.0)
    except Exception:
        base_seq = None

    # ------------------------------------------------------------------
    # Phase 0 – coarse exploration on a tiny grid (n≈512).
    n0 = 512
    candidates = []

    # a few deterministic families
    candidates.append(_arcsine_distribution(n0))
    candidates.append(np.full(n0, 1.0 / n0, dtype=np.float64))
    candidates.append(_powerlaw_boundary_distribution(n0, p=0.5))

    # random draws
    for _ in range(12):
        conc = 10 ** np.random.uniform(-2, 1)          # 0.01 .. 10
        candidates.append(np.random.dirichlet(np.full(n0, conc)))
        a = np.random.uniform(0.1, 5.0)
        b = np.random.uniform(0.1, 5.0)
        candidates.append(_beta_distribution(n0, a, b))
        p = np.random.uniform(0.3, 0.8)
        candidates.append(_powerlaw_boundary_distribution(n0, p))
        lam = np.random.uniform(0.1, 5.0)
        exp_vec = np.exp(-lam * np.arange(n0))
        exp_vec /= exp_vec.sum()
        candidates.append(exp_vec)

    if base_seq is not None:
        candidates.append(_interpolate_to_length(base_seq, n0))

    # pick the best according to cheap proxy
    best_vec = None
    best_score = float("inf")
    for vec in candidates:
        vec = _project_onto_simplex(vec, s=1.0)
        peak = _fft_self_convolve(vec).max()
        fast_score = 2.0 * n0 * peak
        if fast_score < best_score:
            best_score = fast_score
            best_vec = vec.copy()
    print(f"[stage 0] best coarse fast_score={best_score:.6f} (n={n0})")
    if best_vec is None:
        best_vec = np.full(n0, 1.0 / n0)

    # ------------------------------------------------------------------
    # Phase 1 – deep refinement on the tiny grid.
    time_left = deadline - time.time()
    if time_left > 5.0:
        adam_deadline = time.time() + time_left * 0.4
        best_vec, _ = _adam_optimize(best_vec, adam_deadline)
        sub_deadline = time.time() + time_left * 0.2
        best_vec = _refine_max(best_vec, sub_deadline)
        print("[stage 1] refined at n0")

    # ------------------------------------------------------------------
    # Phase 2 – upsample to medium size and optimise.
    n1 = 2048
    best_vec = _interpolate_to_length(best_vec, n1)
    time_left = deadline - time.time()
    if time_left > 5.0:
        adam_deadline = time.time() + time_left * 0.45
        best_vec, _ = _adam_optimize(best_vec, adam_deadline)
        sub_deadline = time.time() + time_left * 0.25
        best_vec = _refine_max(best_vec, sub_deadline)
        print(f"[stage 2] medium‑scale optimisation (n={n1})")

    # ------------------------------------------------------------------
    # Phase 3 – upsample to a large final length.
    n_final = max(4000, n1 * 2)
    if n_final > 8000:
        n_final = 8000
    best_vec = _interpolate_to_length(best_vec, n_final)
    print(f"[stage 3] upsampled to final length n={n_final}")

    # Phase 3‑a – long Adam + sub‑gradient on the full size.
    time_left = deadline - time.time()
    if time_left > 5.0:
        adam_deadline = time.time() + time_left * 0.5
        best_vec, _ = _adam_optimize(best_vec, adam_deadline)
        sub_deadline = time.time() + time_left * 0.3
        best_vec = _refine_max(best_vec, sub_deadline)
        print("[stage 3‑a] deep Adam + sub‑gradient on final size")

    # ------------------------------------------------------------------
    # Optional LP boost (skip for very large vectors to keep memory low).
    time_left = deadline - time.time()
    if time_left > 2.5 and len(best_vec) <= 3000 and _cvxpy_available:
        lp_deadline = time.time() + time_left * 0.6
        best_vec = _lp_improve(best_vec, max_iters=2,
                               mix_t=0.5, inner_deadline=lp_deadline)
        print("[stage 3‑b] LP improvement")

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing polish for the remaining budget.
    time_left = deadline - time.time()
    if time_left > 1.0:
        best_vec, final_peak = _simulated_annealing(best_vec, deadline)
        print("[stage 4] simulated annealing finished")
    else:
        final_peak = _fft_self_convolve(best_vec).max()

    # Symmetrise (often helpful for convolution‑type objectives)
    best_vec = (best_vec + best_vec[::-1]) / 2.0
    best_vec = _project_onto_simplex(best_vec, s=1.0)
    final_peak = _fft_self_convolve(best_vec).max()

    # ------------------------------------------------------------------
    # Final exact evaluation (for reporting only – not used by the judge).
    try:
        true_score = evaluate_sequence(best_vec.tolist())
        print(f"[final] evaluate_sequence score={true_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence raised:", exc)

    approx_score = 2.0 * len(best_vec) * final_peak
    print(f"[final] approximate fast_score={approx_score:.6f} (length={len(best_vec)})")
    return best_vec.tolist()
