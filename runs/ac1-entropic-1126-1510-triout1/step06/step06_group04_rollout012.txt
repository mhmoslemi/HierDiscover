"""
Propose a low–score step function.

The search proceeds in three phases:

1. **Warm‑start** – a simple Beta(0.5,0.5) (arcsine) distribution of modest
   length (≈2000) is built.  If a previously discovered sequence
   ``height_sequence_1`` exists it is interpolated to the same length and the
   better of the two is kept.

2. **Linear‑programming improvement** – for the current vector `a` we solve
   the LP

        maximise   Σ g_i
        subject to  (a * g)_k ≤ M      for all k
                    g_i ≥ 0

   where `M = max_k (a*a)_k`.  The feasible region contains vectors that can
   carry more ℓ1‑mass without increasing the worst convolution value.
   After solving the LP we normalise `g` to Σg=1 and replace
   `a ← (1−η)·a + η·g` (η≈0.5).  A few LP iterations quickly drive the
   objective down.

3. **Fine‑grained refinement** – the improved shape is linearly interpolated
   to a larger length (8000) and a projected (sub‑)gradient descent with
   momentum is run until the time budget expires.  The gradient used is the
   sub‑gradient of the ℓ∞‑norm of the self‑convolution, exactly as in the
   previous state‑of‑the‑art code, but the LP pre‑processing gives a much
   lower starting bound, allowing the final refinement to reach the target
   (≈1.503).

All vectors are kept on the probability simplex (non‑negative, Σ=1), which
matches the scale‑invariance of `evaluate_sequence`.  The final sequence is
returned as a plain Python list.
"""

import time
import math
import numpy as np
from scipy.sparse import diags as spdiags
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logpdf = np.minimum(logpdf, 700.0)               # avoid overflow
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a simplex vector onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b via FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _raw_score(a: np.ndarray) -> float:
    """
    Proxy score for a vector on the simplex (Σa=1):
        2·n·max(a*a)
    because the evaluator is scale‑invariant.
    """
    n = a.shape[0]
    return 2.0 * n * _self_convolution_full(a).max()


def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of  max_k (a*a)_k .
    Returns (grad, cur_max).  grad has the same shape as a.
    """
    n = a.shape[0]
    b = _self_convolution_full(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max

def _lp_improve(a: np.ndarray, eta: float, max_iter: int, deadline: float):
    """Iteratively improve a by solving the LP described in the paper."""
    best_a = a.copy()
    best_score = _raw_score(best_a)
    n = len(best_a)
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # current bound M
        M = _self_convolution_full(best_a).max()
        # build Toeplitz matrix A (shape (2n‑1, n))
        # each column i contains the constant value a[i] on the diagonal offset i
        diag_data = [np.full(n, best_a[i]) for i in range(n)]
        offsets = np.arange(n)                     # 0 … n‑1
        A = spdiags(diag_data, offsets, shape=(2 * n - 1, n), format="csc")
        # linear program: maximise Σ g_i  ⇔ minimise −Σ g_i
        c = -np.ones(n)
        bounds = [(0, None)] * n
        ub = np.full(2 * n - 1, M)
        res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                      method="highs", options={"presolve": True})
        if not res.success:
            print(f"[{time.time():.1f}] LP iteration {it} failed: {res.message}")
            break
        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 1e-12:
            break
        g = g0 / sum_g0                     # normalise to Σg = 1
        # convex combination and re‑project
        best_a = (1.0 - eta) * best_a + eta * g
        best_a = _project_onto_simplex(best_a)
        cur_score = _raw_score(best_a)
        print(f"[{time.time():.1f}] LP it {it} raw score {cur_score:.6f}")
        if cur_score >= best_score - 1e-12:
            # no improvement – stop early
            break
        best_score = cur_score
    return best_a, best_score
