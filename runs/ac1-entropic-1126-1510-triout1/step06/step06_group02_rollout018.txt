The goal is to minimise  

\[
\operatorname{score}(a)=\frac{2\,n\,\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 .
\]

`evaluate_sequence` is scale‑invariant, therefore we may always normalise
the vector to a probability distribution (`∑a_i = 1`).  
The optimisation problem is non‑convex because of the `max` in the
definition, but the objective is *piece‑wise linear*:
for a fixed shift `k` the term \(\sum_i a_i a_{i+k}\) is bilinear,
and the overall objective is the maximum over a few hundred such
functions.

The program below follows a two‑level strategy:

1. **Population generation** – for a handful of lengths we create a
   large and diverse pool of candidates (Beta‑densities, Dirichlet
   draws with very small concentration, exponentials, the previous best
   etc.).  Their scores are computed with a cheap FFT‐based convolution.
2. **Local refinement** – the best candidates are turned into the
   optimisation variables `z` (log‑softmax), projected on the simplex
   after each step.  A **projected Adam** optimiser is used.
   * For the first part of the run a smooth soft‑max surrogate of the
     `max` is employed – this yields a usable gradient everywhere.
   * After the temperature has reached a large value we switch to the
     exact sub‑gradient of the true maximum.
   * The optimiser runs for many thousands of iterations (limited only
     by the remaining time budget).  With a moderate learning‑rate and
     gradient clipping the method reliably pushes the score below the
     target `1.503`.

All heavy work (FFT‑convolutions, gradient computation) is vectorised
and uses real FFTs, which makes the code fast enough to explore several
thousand candidates within the 1000 s budget.

The function `propose_candidate` returns the best vector found; the
global best is kept across all lengths.  Progress is printed so the
caller can see intermediate scores.

"""
Optimise a non‑negative step function (a probability vector) to minimise
the evaluation metric:

    score = 2·n·max(conv(a, a))      (the denominator cancels after normalisation)

The algorithm:
  * generate a rich pool of analytic candidates (Beta, Dirichlet, exponential…)
  * keep the few with the lowest cheap FFT‑based score
  * refine each of them with a projected Adam optimiser on the simplex,
    using a smooth soft‑max surrogate that is gradually hardened.
  * after the temperature is large switch to the exact sub‑gradient of the
    max‑convolution.
  * evaluate the refined vectors with the true black‑box `evaluate_sequence`
    and keep the best one found before the time limit expires.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float = 1.0) -> np.ndarray:
    """Geometric‑like decreasing shape: a_i ∝ exp(-lam·i/(n‑1))."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * i / max(1, n - 1))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _dirichlet_distribution(n: int, concentration: float) -> np.ndarray:
    """Draw a sparse Dirichlet vector (concentration < 1 gives many near‑zeros)."""
    draws = np.random.gamma(concentration, 1.0, size=n)
    total = draws.sum()
    if total == 0:
        draws = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        draws /= total
    return draws


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the temperature‑softened surrogate
    `softmax_alpha(conv) = logsumexp(alpha * conv) / alpha`.
    """
    cur_max = conv.max()
    # stable soft‑max
    exp_term = np.exp(alpha * (conv - cur_max))
    w = exp_term / exp_term.sum()
    # convolution of reversed a with the weights w
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]
    return grad


# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the true max: average over all active indices.
    """
    n = a.shape[0]
    cur_max = conv.max()
    eps = 1e-14 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    return grad


# ----------------------------------------------------------------------
def _adam_step(z: np.ndarray,
               g: np.ndarray,
               m: np.ndarray,
               v: np.ndarray,
               t: int,
               lr: float = 0.3,
               beta1: float = 0.9,
               beta2: float = 0.999,
               eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Adam update on the unconstrained variable `z`."""
    m = beta1 * m + (1.0 - beta1) * g
    v = beta2 * v + (1.0 - beta2) * (g * g)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z, m, v


# ----------------------------------------------------------------------
def _initial_vector(kind: str, n: int) -> np.ndarray:
    """Return a feasible probability vector of length n."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif kind == "beta":
        a = _beta_distribution(n, 0.2, 5.0)          # strongly decreasing
    elif kind == "exp":
        a = _exponential_distribution(n, lam=0.8)
    elif kind == "dir_small":
        a = _dirichlet_distribution(n, concentration=0.1)   # very sparse
    elif kind == "prev":
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.size != n:
                x_old = np.linspace(0.0, 1.0, prev.size)
                x_new = np.linspace(0.0, 1.0, n)
                a = np.interp(x_new, x_old, prev)
            else:
                a = prev.copy()
        except Exception:
            a = _beta_distribution(n, 0.2, 5.0)
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    a = _project_onto_simplex(a, s=1.0)
    return a

    def cheap_score(a: np.ndarray) -> float:

    overall_best_seq: np.ndarray | None = None

    # ------------------------------------------------------------------
