"""
Propose a low‑score step function.

Algorithm
---------
1.  Choose a problem size n (≈ 6000‑9000) and the target ℓ¹‑norm
    S = sqrt(2 n).  All candidates are projected onto the simplex
    {a ≥ 0, Σa = S}.

2.  Initialise many diverse start points (uniform, exponentials,
    Gaussian, Dirichlet draws and the previous best if available).

3.  For each start run a **smooth‑max Adam** optimiser.
    The smooth surrogate of max(conv) is
        f_μ(a) = max(conv) + μ·log Σ exp((conv‑max)/μ) .
    Gradients are obtained analytically via a single FFT‑based
    convolution.  μ is cooled geometrically (×0.7) down to 1e‑12.

4.  Refine the best solution with a **hard‑max sub‑gradient** method
    (exact max of the convolution) using a diminishing step‑size.

5.  Finish with a **targeted hill‑climber**:
    * locate the index k* where the convolution attains its maximum,
    * compute the contribution a[i]·a[k*‑i] of each coordinate,
    * move a small amount of mass from the biggest contributor to a
      low‑mass coordinate, keep the ℓ¹‑norm by projection,
    * accept the move only if the exact score improves.

6.  A final interleaving‑sorting heuristic is tried.

The routine respects a user‑supplied time budget (default 1000 s) and
returns the best height list found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full auto‑convolution using FFT (real input)."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, mu: float,
                         n: int, fft_len: int) -> tuple[float, np.ndarray]:
    """
    Value and gradient (w.r.t. a) of the smooth‑max surrogate:
        f_μ(a) = max(conv) + μ·log Σ exp((conv‑max)/μ)
    Parameters
    ----------
    a      : ndarray (n,)  – non‑negative heights (Σa = S)
    mu     : float        – smoothing parameter (μ>0, μ=0 → hard max)
    n, fft_len : problem size and FFT length (power of two ≥ 2n‑1)
    Returns
    -------
    f_val : float
    g_a   : ndarray (n,) – gradient w.r.t. a
    """
    # convolution
    A_fft = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A_fft * A_fft, fft_len)[: 2 * n - 1]

    max_c = conv.max()
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)          # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:                                            # hard max
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            w = np.zeros_like(conv)
        else:
            w = np.zeros_like(conv, dtype=float)
            w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. a
    w_fft = np.fft.rfft(w, fft_len)
    a_rev_fft = np.fft.rfft(a[::-1], fft_len)
    grad = 2.0 * np.fft.irfft(w_fft * a_rev_fft, fft_len)[:n]
    return float(f_val), grad.astype(np.float64)


# ----------------------------------------------------------------------
def _hard_subgradient(a: np.ndarray, n: int, fft_len: int) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient of the true max‑objective max(conv) at point a.
    Returns (g, max_conv) where g is a gradient vector (length n).
    """
    A_fft = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A_fft * A_fft, fft_len)[: 2 * n - 1]
    max_c = conv.max()
    mask = conv == max_c
    cnt = np.count_nonzero(mask)
    if cnt == 0:
        w = np.zeros_like(conv)
    else:
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
    w_fft = np.fft.rfft(w, fft_len)
    a_rev_fft = np.fft.rfft(a[::-1], fft_len)
    g = 2.0 * np.fft.irfft(w_fft * a_rev_fft, fft_len)[:n]
    return g, float(max_c)


# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray, n: int, fft_len: int) -> float:
    """Convenient wrapper returning only the maximal convolution entry."""
    A_fft = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A_fft * A_fft, fft_len)[: 2 * n - 1]
    return float(conv.max())

            best_score = sc
            best_a = a_vec.copy()
            print(f"[improve] new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Build a pool of diverse initial points ----------------------
    init_pool = []

    # uniform
    init_pool.append(np.full(n, S / n))

    # decreasing / increasing exponential
    alpha = rng.uniform(0.001, 0.008) * (n / 2000.0)
    init_pool.append(np.exp(-alpha * np.arange(n)))
    init_pool.append(np.exp(-alpha * np.arange(n - 1, -1, -1)))

    # Gaussian bump
    sigma = rng.uniform(0.08, 0.25) * n
    x = np.arange(n)
    gauss = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
    init_pool.append(gauss)

    # Dirichlet draws with various concentrations
    for _ in range(8):
        conc = rng.uniform(0.2, 5.0)
        a = rng.dirichlet([conc] * n)
        init_pool.append(a)

    # reference (if present)
    try:
        ref = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        ref = _project_to_simplex(ref, S)
        init_pool.append(ref)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2️⃣  Adam optimisation on the smooth‑max surrogate ----------------
    # hyper‑parameters
    mu_factor = 0.7
    mu_min = 1e-12
    iters_per_mu = 250               # L‑BFGS used 400 – we keep comparable
    lr = 0.1
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    for start_id, a0 in enumerate(init_pool):
        if time.time() > deadline:
            break
        # initialise
        a = np.clip(a0, 0.0, 1000.0)
        a = _project_to_simplex(a, S)
        _maybe_update(a)

        # initialise Adam moments
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        t = 0

        # initial temperature based on current max‑convolution
        mu = max(_conv_max(a, n, fft_len), 1.0)

        while mu > mu_min and time.time() < deadline:
            for _ in range(iters_per_mu):
                t += 1
                fval, grad = _smooth_max_and_grad(a, mu, n, fft_len)

                # Adam update
                m = beta1 * m + (1.0 - beta1) * grad
                v = beta2 * v + (1.0 - beta2) * (grad ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                a = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

                # maintain feasibility
                a = np.clip(a, 0.0, 1000.0)
                a = _project_to_simplex(a, S)

                # cheap progress check every 100 steps
                if t % 100 == 0:
                    sc = evaluate_sequence(a.tolist())
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_a = a.copy()
                        print(f"[adam] iter {t:5d}  score {best_score:.6f}")

            mu *= mu_factor

        # ------------------------------------------------------------------
        # 3️⃣  Hard‑max sub‑gradient descent (more precise) ---------------
        max_sub_iters = 15000
        step0 = S * 0.5 / n
        a_sub = a.copy()
        for it in range(max_sub_iters):
            if time.time() > deadline:
                break
            g, max_c = _hard_subgradient(a_sub, n, fft_len)
            step = step0 / (1.0 + 1e-5 * it)
            improved = False
            for _ in range(12):
                a_new = a_sub - step * g
                a_new = np.clip(a_new, 0.0, 1000.0)
                a_new = _project_to_simplex(a_new, S)

                sc = evaluate_sequence(a_new.tolist())
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_new.copy()
                    a_sub = a_new
                    improved = True
                    print(f"[sub] it {it:5d}  score {best_score:.6f}")
                    break
                step *= 0.5
            if not improved:
                # occasional jitter to escape plateaus
                i, j = rng.choice(n, size=2, replace=False)
                delta = rng.random() * a_sub[i] * 0.03
                a_sub[i] -= delta
                a_sub[j] += delta
                a_sub = np.clip(a_sub, 0.0, 1000.0)
                a_sub = _project_to_simplex(a_sub, S)

        # ------------------------------------------------------------------
        # 4️⃣  Targeted hill‑climbing ------------------------------------
        remaining_trials = int(min(30000, (deadline - time.time()) * 35.0))
        a_hc = a_sub.copy()
        for trial in range(remaining_trials):
            if time.time() > deadline:
                break

            # locate the current worst (max) convolution index
            A_fft = np.fft.rfft(a_hc, fft_len)
            conv = np.fft.irfft(A_fft * A_fft, fft_len)[: 2 * n - 1]
            k_star = int(np.argmax(conv))
            max_val = conv[k_star]

            # contributions of each coordinate to this peak
            i_min = max(0, k_star - (n - 1))
            i_max = min(n - 1, k_star)
            idx = np.arange(i_min, i_max + 1, dtype=int)
            jdx = k_star - idx
            contrib = a_hc[idx] * a_hc[jdx]

            donor = idx[np.argmax(contrib)]
            # choose a receiver among low‑mass indices
            low_candidates = np.where(a_hc < a_hc.mean())[0]
            if low_candidates.size == 0:
                receiver = rng.integers(n)
            else:
                receiver = rng.choice(low_candidates)

            delta = min(a_hc[donor] * 0.08, (a_hc.mean() - a_hc[receiver]) * 0.6 + 1e-12)
            if delta <= 0:
                continue
            cand = a_hc.copy()
            cand[donor] -= delta
            cand[receiver] += delta
            cand = np.clip(cand, 0.0, 1000.0)
            cand = _project_to_simplex(cand, S)

            new_max = _conv_max(cand, n, fft_len)
            if new_max < max_val - 1e-12:
                a_hc = cand
                sc = evaluate_sequence(a_hc.tolist())
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_hc.copy()
                    print(f"[hill] trial {trial:5d}  score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final interleaving / sorting heuristic ----------------------
    if time.time() < deadline and best_score < float("inf"):
        # descending order then interleaved big‑small pattern
        vals_desc = np.sort(best_a)[::-1]
        interleaved = np.empty_like(vals_desc)
        half = len(vals_desc) // 2
        left = vals_desc[:half][::-1]
        right = vals_desc[half:][::-1]
        interleaved[0::2] = left
        interleaved[1::2] = right
        interleaved = _project_to_simplex(interleaved, S)
        _maybe_update(interleaved)

        # pure ascending / descending checks
        asc = np.sort(best_a)
        _maybe_update(_project_to_simplex(asc, S))

        desc = vals_desc
        _maybe_update(_project_to_simplex(desc, S))

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
