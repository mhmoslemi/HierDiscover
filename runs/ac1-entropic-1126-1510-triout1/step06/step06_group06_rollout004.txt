def _gradient_mass_transfer(a, deadline):
    while time left:
        b = _fft_self_convolve(a)
        cur_max = b.max()
        # compute subgradient weights uniform on active set
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(b >= cur_max - tol)[0]
        w = np.zeros_like(b)
        if active.size > 0:
            w[active] = 1.0 / active.size
        # compute gradient = 2 * conv(a_rev, w)
        grad = 2.0 * _fft_convolve(a[::-1], w)[len(a)-1:len(a)-1+len(a)]

        # choose src = argmax(grad), dst = argmin(grad)
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if grad[src] <= grad[dst] + 1e-15:
            # gradient flat, nothing to improve
            break

        # move fraction of src mass to dst
        max_delta = min(a[src], 0.1)  # up to 10% or absolute 0.1
        if max_delta <= 0:
            break
        # might use binary search to find best delta that reduces max
        # simple: try a few deltas
        best_delta = 0.0
        best_max = cur_max
        for factor in (0.2, 0.5, 0.8, 1.0):
            delta = factor * max_delta
            a_new = a.copy()
            a_new[src] -= delta
            a_new[dst] += delta
            # compute new max quickly using FFT
            b_new = _fft_self_convolve(a_new)
            new_max = b_new.max()
            if new_max < best_max:
                best_max = new_max
                best_delta = delta
        if best_delta > 0:
            a[src] -= best_delta
            a[dst] += best_delta
            # ensure numeric within bounds
            # (no need to projection because sum preserved)
        else:
            # fallback to jitter
            a += np.random.randn(len(a))*1e-7
            a = _project_onto_simplex(a,s=1.0)
    return a

"""
Propose a low‑score non‑negative step function.

The algorithm improves on the reference pipeline by:

* Exploring a richer family of analytic shapes (power‑law, Beta with
  very small parameters, skewed Beta, etc.) to obtain a better initial
  guess.
* Running the Adam optimiser with a larger smoothness range and a
  higher learning rate.
* Adding a second short Adam run after a small random jitter (multiple
  restarts).
* Performing a dedicated “gradient‑mass‑transfer’’ step that moves mass
  from the index with the largest sub‑gradient (most harmful for the
  current peak) to the index with the smallest sub‑gradient, accepting
  only improving moves (a greedy descent on the exact ℓ∞ objective).

All updates keep the vector on the probability simplex Σa_i = 1, which
does not affect the final score because the evaluation function is
homogeneous of degree 0.  The code respects the 1000 s time budget and
prints intermediate scores for monitoring.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth‑max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # numerically stable soft‑max
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max

def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 5.0,
                   alpha_factor: float = 1.02,
                   alpha_max: float = 10000.0,
                   lr0: float = 0.5,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best vector seen (on the exact max) and its peak value.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size schedule (slow decay)
        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # keep the best exact peak seen so far
        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        # continuation on the smoothness parameter
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny jitter to escape plateaus
        if t % 400 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak

def _max_subgradient_step(a: np.ndarray):
    """
    One sub‑gradient step for the exact ℓ∞ convolution objective.
    Returns (grad, cur_max) where grad is the sub‑gradient.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max

def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Sub‑gradient descent on the exact max‑convolution objective.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.02 / np.sqrt(it)  # diminishing step
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        # occasional jitter
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a

def _gradient_mass_transfer(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Greedy mass‑transfer guided by the sub‑gradient.
    Moves mass from the index with the largest gradient to the one with the
    smallest gradient, accepting only improving steps.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_max = _fft_self_convolve(a).max()

    while time.time() < deadline:
        b = _fft_self_convolve(a)
        cur_max = b.max()
        # sub‑gradient (uniform weight on active set)
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(b >= cur_max - tol)[0]
        w = np.zeros_like(b)
        if active.size > 0:
            w[active] = 1.0 / active.size
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))

        # If gradients are essentially flat, stop.
        if grad[src] <= grad[dst] + 1e-15:
            break

        max_delta = min(a[src], 0.2)          # move at most 20 % of src mass
        if max_delta <= 0:
            break

        # try a few candidate deltas
        best_delta = 0.0
        best_new_max = cur_max
        for frac in (0.1, 0.25, 0.5, 1.0):
            delta = frac * max_delta
            a_new = a.copy()
            a_new[src] -= delta
            a_new[dst] += delta
            new_max = _fft_self_convolve(a_new).max()
            if new_max < best_new_max - 1e-15:
                best_new_max = new_max
                best_delta = delta

        if best_delta > 0.0:
            a[src] -= best_delta
            a[dst] += best_delta
            cur_max = best_new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
        else:
            # small random perturbation to escape possible stagnation
            a += np.random.randn(n) * 1e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a

def _two_level_family(n: int, deadline: float):
    """
    Search a simple two‑level family:
        a_i = p   for i < L   (front block)
        a_i = q   otherwise   (back block)
    with p = r·q, r≥1.
    Returns the best vector found and its fast score.
    """
    best_seq = None
    best_score = float('inf')
    step_L = max(1, n // 100)          # ~1 % of length
    r_grid = np.logspace(0.0, 2.3, num=25)   # 1 … ≈200

    for L in range(step_L, n, step_L):
        if time.time() > deadline:
            break
        denom_base = n - L
        for r in r_grid:
            denom = L * r + denom_base
            q = 1.0 / denom
            p = r * q
            seq = np.empty(n, dtype=np.float64)
            seq[:L] = p
            seq[L:] = q
            peak = _fft_self_convolve(seq).max()
            score = 2.0 * n * peak
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
    return best_seq, best_score

        for w in (0.2, 0.4, 0.6):
            mix = (1.0 - w) * candidates["arcsine"] + w * candidates["uniform"]
            candidates[f"mix_{w:.1f}"] = mix

        # incorporate previously best if available
        if base_seq is not None:
            old_n = len(base_seq)
            x_old = np.linspace(0.0, 1.0, old_n)
            x_new = np.linspace(0.0, 1.0, n)
            interp = np.interp(x_new, x_old, base_seq)
            interp = np.maximum(0.0, interp)
            interp /= interp.sum()
            candidates["prev"] = interp

        # two‑level parametric family (cheap evaluation)
        seq_tw, score_tw = _two_level_family(n, deadline)
        if seq_tw is not None:
            candidates["two_level"] = seq_tw

        # evaluate fast scores for all candidates
        for tag, vec in candidates.items():
            vec = _project_onto_simplex(vec, s=1.0)
            peak = _fft_self_convolve(vec).max()
            fast_score = 2.0 * n * peak
            if fast_score < overall_best_score:
                overall_best_score = fast_score
                overall_best_seq = vec.copy()
                print(f"[init] n={n:4d} {tag:<16s} fast_score={fast_score:.6f}")

    # --------------------------------------------------------------
    # Safety net – fall back to uniform if nothing was found.
    if overall_best_seq is None:
        n = max(500, base_n)
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print("[fallback] using uniform distribution")

    # --------------------------------------------------------------
    # Phase 1 – Adam optimisation (first long run)
    time_left = deadline - time.time()
    if time_left > 5.0:
        n = len(overall_best_seq)
        phase_end = time.time() + time_left * 0.55
        print(f"[phase 1] Adam optimisation (n={n}) – time left {time_left:.1f}s")
        a_opt, _ = _adam_optimize(overall_best_seq, phase_end,
                                   init_alpha=5.0, alpha_max=10000.0, lr0=0.5)
        overall_best_seq = a_opt.copy()
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print(f"[phase 1] after Adam fast_score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – a short second Adam run after jitter (multiple restart)
    time_left = deadline - time.time()
    if time_left > 4.0:
        n = len(overall_best_seq)
        # small random perturbation
        a_jitter = overall_best_seq.copy()
        a_jitter += np.random.randn(n) * 1e-4
        a_jitter = _project_onto_simplex(a_jitter, s=1.0)
        phase_end = time.time() + time_left * 0.45
        print(f"[phase 2] second Adam run (n={n}) – time left {time_left:.1f}s")
        a_opt2, _ = _adam_optimize(a_jitter, phase_end,
                                   init_alpha=5.0, alpha_max=10000.0, lr0=0.5)
        # keep whichever is better on the exact max
        if _fft_self_convolve(a_opt2).max() < _fft_self_convolve(overall_best_seq).max():
            overall_best_seq = a_opt2.copy()
            overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
            print(f"[phase 2] improved after second Adam fast_score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – exact sub‑gradient descent
    time_left = deadline - time.time()
    if time_left > 3.0:
        n = len(overall_best_seq)
        sub_end = time.time() + time_left * 0.55
        print(f"[phase 3] sub‑gradient refinement (n={n}) – time left {time_left:.1f}s")
        a_ref = _refine_max(overall_best_seq, sub_end)
        overall_best_seq = a_ref.copy()
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print(f"[phase 3] after sub‑gradient fast_score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – directed gradient‑mass‑transfer search (greedy improvement)
    time_left = deadline - time.time()
    if time_left > 2.0:
        n = len(overall_best_seq)
        print(f"[phase 4] gradient‑mass‑transfer (n={n}) – time left {time_left:.1f}s")
        a_final = _gradient_mass_transfer(overall_best_seq, deadline)
        overall_best_seq = a_final.copy()
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print(f"[phase 4] after mass transfer fast_score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # Final black‑box evaluation
    try:
        true_score = evaluate_sequence(overall_best_seq.tolist())
        print(f"[final] evaluate_sequence score={true_score:.6f}")
        if true_score < overall_best_score:
            overall_best_score = true_score
    except Exception as exc:
        print("[final] evaluate_sequence raised:", exc)

    print(f"Best approx. score {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
