def gradient_y(y, beta):
    a = softmax(y)  # define function
    b = conv(a, a)
    bmax = b.max()
    # compute softmax over b
    w = np.exp(beta * (b - bmax))
    w /= w.sum()
    # gradient w.r.t a
    grad_a = 2.0 * _fft_convolve(a[::-1], w)[len(a)-1:len(a)-1+len(a)]
    # compute dot
    dot = np.dot(grad_a, a)
    grad_y = a * (grad_a - dot)
    return grad_y

def _grad_and_score(y, beta):
    a = _softmax(y)
    # compute full self-convolution via FFT
    b = _self_convolution_full(a)
    bmax = b.max()
    # compute w = softmax(beta * (b - bmax))
    # subtract bmax to maintain stability.
    w = np.exp(beta * (b - bmax))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(b, 1.0 / b.size)
    else:
        w /= w_sum
    # gradient w.r.t a
    a_rev = a[::-1]
    grad_a_full = 2.0 * _fft_convolve(a_rev, w)  # full conv length a_rev + w -1
    n = a.shape[0]
    # central part:
    grad_a = grad_a_full[n-1:n-1+n]
    # gradient w.r.t y
    dot = np.dot(grad_a, a)
    grad_y = a * (grad_a - dot)
    # raw score
    raw_score = 2.0 * n * bmax
    # return gradient, raw_score, a
    return grad_y, raw_score, a

Propose a low‑score step function.

This implementation mixes three ideas:

1️⃣  A quick, diverse random search (Beta / Dirichlet families) to obtain a
    decent starting sequence.
2️⃣  A smooth–max approximation with a soft‑max over the self‑convolution
    followed by Adam optimisation on the parametrisation a_i = softmax(y_i).
    The temperature (β) is increased gradually, so the optimisation first
    sees a smoothed objective and finally concentrates on the true peak.
3️⃣  A final projected sub‑gradient refinement (the same scheme that was
    used previously) to polish the result.

All heavy work is done with FFT‑based convolutions (O(n log n)) and the
procedure respects the sum‑to‑1 simplex constraint via the soft‑max
modelling.  The total running time is limited to the supplied budget
(e.g. 1000 s).  The returned list is ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution of length n, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf – use log‑gamma for stability
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logpdf = np.minimum(logpdf, 700.0)          # avoid overflow in exp
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a * a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """
    Proxy score assuming Σa = 1 :  2·n·max(a*a).
    This equals the true objective because the evaluator is scale‑invariant.
    """
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of max_k (a*a)_k.
    Returns (grad, cur_max).  grad has the same shape as a.
    """
    n = a.shape[0]
    b = _self_convolution_full(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _grad_and_score(y: np.ndarray, beta: float):
    """
    For a given log‑softmax vector y and temperature beta, return:
      * gradient of the smoothed max (w.r.t. y)
      * raw score   2·n·max(a*a)
      * current a (softmax of y)
    """
    # a = softmax(y)
    max_y = np.max(y)
    e = np.exp(y - max_y)
    a = e / e.sum()

    # self‑convolution and its maximum
    b = _self_convolution_full(a)
    bmax = b.max()
    raw_score = 2.0 * a.shape[0] * bmax

    # smoothed weights w = softmax(beta * (b - bmax))
    w = np.exp(beta * (b - bmax))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(b, 1.0 / b.size)
    else:
        w /= w_sum

    # gradient w.r.t. a
    a_rev = a[::-1]
    grad_a_full = 2.0 * _fft_convolve(a_rev, w)
    n = a.shape[0]
    grad_a = grad_a_full[n - 1 : n - 1 + n]

    # chain rule through softmax: ∂f/∂y_i = a_i (∂f/∂a_i - Σ_j a_j ∂f/∂a_j)
    dot = np.dot(grad_a, a)
    grad_y = a * (grad_a - dot)

    return grad_y, raw_score, a
