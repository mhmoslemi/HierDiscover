def improve_sequence(f, max_iter, t, time_budget):
  compute sum_target = sqrt(2*n)
  while time left and improvement:
    M = max(conv(f,f))
    # solve LP to get g
    g = cvxpy.Variable(n)
    conv_fg = cvxpy.conv(f, g)  # size (2n-1)
    constraints = [g >= 0, conv_fg <= M]
    obj = cvxpy.Maximize(cvxpy.sum(g))
    prob = cvxpy.Problem(obj, constraints)
    result = prob.solve(solver='ECOS', warm_start=True, verbose=False)
    if prob.status != cvxpy.OPTIMAL:
      break
    g_vec = np.array(g.value).flatten()
    sum_g = g_vec.sum()
    if sum_g == 0:
      break
    g_scaled = g_vec * (sum_target / sum_g)
    # update f via convex combination
    f = (1 - t) * f + t * g_scaled
    # maybe update M and continue. t can be fixed (e.g., 0.5?) 
  return f

"""
Propose a step‑function with a lower ∥f∗f∥∞ using a Frank‑Wolfe style
procedure.

The objective (after normalising Σa_i=√(2n)) is exactly
    max_k (f∗f)_k .
We iteratively improve a feasible vector f of length n:

1.  Compute M = max(f∗f).
2.  Solve a linear programme
        maximise   Σ g_i
        s.t.       g_i ≥ 0
                   (f∗g)_k ≤ M   for all “active’’ shifts k
   The LP only includes those shifts k for which (f∗f)_k is within a tiny
   tolerance of M (the active set).  The LP is solved with
   ``scipy.optimize.linprog`` (HiGHS).  The solution g is a direction that
   tries to spread the mass while keeping the convolution below the current
   bound.
3.  Rescale g to the required normalisation (Σg = √(2n)).
4.  Take a convex combination f←(1‑t)f + t·g with a modest step size t.
   The bound M never increases, and in practice it decreases rapidly.

The whole routine is applied to a few different lengths (≈ 3000‑8000) and
different initialisations (previous best, beta‑shapes, uniform, random).
The best vector found (according to the real ``evaluate_sequence``) is
returned.  The algorithm respects the 1000 s time budget.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of a."""
    return _fft_self_convolve(a).max()


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


def _exponential_distribution(n: int, lam: float = 1.0) -> np.ndarray:
    """Geometric‑like decreasing shape: a_i ∝ exp(-lam·i/(n‑1))."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * i / max(1, n - 1))
    vals /= vals.sum()
    return vals


def _initial_vector(kind: str, n: int, reference: np.ndarray | None) -> np.ndarray:
    """Return a feasible non‑negative vector of length n."""
    if kind == "prev" and reference is not None:
        if reference.size != n:
            # linear interpolation to required size
            old_x = np.linspace(0.0, 1.0, reference.size)
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, reference)
        else:
            a = reference.copy()
    elif kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif kind == "beta":
        a = _beta_distribution(n, 0.5, 0.5)
    elif kind == "beta_asym1":
        a = _beta_distribution(n, 0.2, 0.8)
    elif kind == "beta_asym2":
        a = _beta_distribution(n, 0.8, 0.2)
    elif kind == "exp_small":
        a = _exponential_distribution(n, lam=0.5)
    elif kind == "exp_big":
        a = _exponential_distribution(n, lam=1.5)
    else:
        a = _beta_distribution(n, 0.5, 0.5)

    # Guard against exact zeros – the LP needs every coefficient to be
    # strictly positive to keep it bounded.
    eps = 1e-12
    a = np.maximum(a, eps)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def _build_active_matrix(f: np.ndarray, active_idx: list[int]) -> sp.csr_matrix:
    """
    Build the sparse matrix A (|active| × n) where each row corresponds to
    the linear constraint (f∗g)_k ≤ M for k∈active_idx.
    Row i contains coefficients f_j for variable g_{k−j}.
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    for row, k in enumerate(active_idx):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # loop over i where f_i contributes
        for i in range(i_min, i_max + 1):
            j = k - i               # index of g
            rows.append(row)
            cols.append(j)
            data.append(f[i])
    return sp.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))


def _solve_lp_direction(f: np.ndarray, M: float, active_idx: list[int]) -> np.ndarray | None:
    """
    Linear programme:
        maximise   Σ g_i
        s.t.       g_i ≥ 0
                   (f∗g)_k ≤ M   for k ∈ active_idx
    Returns the optimal g as a dense ndarray, or None on failure.
    """
    n = f.shape[0]
    if len(active_idx) == 0:
        # No active constraints: the LP would be unbounded.
        # In that case we simply return the uniform vector.
        return np.full(n, 1.0 / n, dtype=np.float64)

    A_ub = _build_active_matrix(f, active_idx)          # shape (|A|, n)
    b_ub = np.full(A_ub.shape[0], M, dtype=np.float64)

    # objective: maximise Σ g → minimise –Σ g
    c = -np.ones(n, dtype=np.float64)

    # Bounds for each variable: g_i ≥ 0
    bounds = [(0.0, None) for _ in range(n)]

    # Use HiGHS (fast for sparse LPs)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
    except Exception as e:
        print("[LP] Exception:", e)
        return None

    if not res.success:
        # Occasionally the active set may be too weak leading to unboundedness.
        # Fallback to a simple feasible point (uniform).
        # print("[LP] failed:", res.message)
        return None

    g = res.x
    # Clip tiny negative noise (if any)
    g = np.maximum(g, 0.0)
    return g


# ----------------------------------------------------------------------
def _improve_f(f: np.ndarray,
               t: float,
               max_outer: int,
               deadline: float) -> np.ndarray:
    """
    Perform a few Frank‑Wolfe style updates.
    f is assumed to be already normalised (Σf = √(2n)).
    """
    n = f.shape[0]
    sum_target = math.sqrt(2.0 * n)

    # Ensure strict positivity for the LP (tiny epsilon)
    eps = 1e-12
    f = np.maximum(f, eps)
    f /= f.sum()
    f *= sum_target

    for outer in range(max_outer):
        if time.time() >= deadline:
            break

        # 1) current bound
        conv_f = _fft_self_convolve(f)
        M = conv_f.max()

        # 2) active set – indices where conv_f is close to its maximum
        # We keep a modest absolute tolerance (1e‑9) plus a relative part.
        tol = max(1e-12, M * 1e-6)
        active = np.where(conv_f >= M - tol)[0].tolist()

        # 3) solve LP to obtain a direction g
        g_raw = _solve_lp_direction(f, M, active)

        # if the LP failed (unlikely) we stop improving this candidate
        if g_raw is None:
            break

        # 4) rescale g to the required normalisation
        sum_g = g_raw.sum()
        if sum_g <= 0:
            break
        g = g_raw * (sum_target / sum_g)

        # 5) convex combination
        f_new = (1.0 - t) * f + t * g

        # 6) check progress – stop early if no improvement
        new_M = _fft_self_convolve(f_new).max()
        if new_M >= M - 1e-9 * M:
            # No meaningful improvement – we can shrink the step size
            t *= 0.5
            if t < 1e-3:
                break
        else:
            f = f_new
            # optionally increase step size a little
            t = min(0.6, t * 1.05)

    return f

            sub_deadline = time.time() + min(120.0, deadline - time.time())
            improved = _improve_f(a0,
                                  t=0.35,
                                  max_outer=12,
                                  deadline=sub_deadline)

            # Final evaluation with the true black‑box
            try:
                final_score = evaluate_sequence(improved.tolist())
            except Exception:
                final_score = cheap_score(improved / improved.sum())  # fallback

            print(f"--> starter {sid} final score = {final_score:.6f}")

            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = improved.copy()
                print(f"*** New global best: {overall_best_score:.6f} (n={n})")
                if overall_best_score <= 1.503:
                    print("Target achieved – terminating early.")
                    return overall_best_seq.tolist()

    # ------------------------------------------------------------------
    # fallback in the unlikely case no sequence was produced
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())
        print("Fallback sequence used.")

    print(f"\nBest score after search: {overall_best_score:.6f} (length {len(overall_best_seq)})")
    return overall_best_seq.tolist()
