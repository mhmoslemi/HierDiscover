"""
Propose a low‑score step function.

The routine combines several inexpensive ideas that were not used in the
previous reference implementation:

1.  Random Dirichlet draws (small concentration → few large spikes) are
    generated and then **interleaved** (largest, smallest, 2nd largest,
    2nd smallest, …).  This puts large values far apart and immediately
    reduces the overlap of large‑large products in the auto‑convolution.

2.  The best candidate found so far is refined with the smooth‑max
    surrogate of the true objective using L‑BFGS‑B and a slow cooling
    schedule (μ ← 0.8·μ).  Gradients are exact because the surrogate is
    differentiable.

3.  A cheap projected sub‑gradient descent (hard max) further pushes the
    score down.

4.  Finally a simulated‑annealing hill‑climber tries random swaps,
    δ‑mass transfers, block‑averaging and block‑rotations.  Moves are
    accepted only if they improve the exact score, otherwise a
    temperature‑controlled Metropolis step allows occasional uphill
    moves to escape plateaus.

All helpers are top‑level, there are no closures or lambdas, and the
function respects the user‑supplied time budget (default 1000 s).  The
returned list of heights is ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _compute_smooth_obj_and_grad(y: np.ndarray, mu: float,
                                 n: int, S: float) -> tuple[float, np.ndarray]:
    """
    Value and gradient of the smooth‑max surrogate:

        f_μ(y) = max(conv) + μ·log Σ exp((conv‑max)/μ)

    Parameters
    ----------
    y : ndarray (n,)
        Unconstrained parameters (log‑odds of the simplex).
    mu : float  (>0, μ=0 means the hard max)
    n, S : problem size and target ℓ¹‑norm.

    Returns
    -------
    f_val  : float   – smooth‑max value
    grad_y : ndarray – gradient w.r.t. y
    """
    p = _softmax(y)                     # Σ p = 1, p ≥ 0
    a = S * p                           # Σ a = S

    conv = np.convolve(a, a)            # length 2·n‑1

    max_c = conv.max()
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. a
    conv_wr = np.convolve(w, a[::-1])
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

    inner = np.dot(g_a, p)              # scalar Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)
    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable for scipy.optimize that returns (value, gradient)."""

    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_smooth_obj_and_grad(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
def _hard_subgradient(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient of the true max‑objective  max(conv)  at point a.
    Returns (g, max_conv) where g is the sub‑gradient vector.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_c = conv.max()
    mask = conv == max_c
    cnt = np.count_nonzero(mask)
    if cnt == 0:
        w = np.zeros_like(conv)
    else:
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
    conv_wr = np.convolve(w, a[::-1])
    g = 2.0 * conv_wr[n - 1:n - 1 + n]
    return g, max_c


# ----------------------------------------------------------------------
def _interleave_extremes(vals_desc: np.ndarray) -> np.ndarray:
    """
    Produce [largest, smallest, 2nd‑largest, 2nd‑smallest, …] ordering.
    ``vals_desc`` must be sorted in descending order.
    """
    n = vals_desc.shape[0]
    res = np.empty_like(vals_desc)
    i, j, k = 0, n - 1, 0
    while i <= j:
        res[k] = vals_desc[i]
        k += 1
        i += 1
        if i <= j:
            res[k] = vals_desc[j]
            k += 1
            j -= 1
    return res


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) with a low
    ``evaluate_sequence`` score.  The routine respects ``budget_s`` seconds
    and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5        # safety margin

    # ------------------------------------------------------------------
    # choose a working length ------------------------------------------------
    try:
        # if a reference exists use roughly twice its length (capped)
        ref_len = len(height_sequence_1)            # type: ignore[name-defined]
        n = min(max(ref_len * 2, 4000), 9000)
    except Exception:
        n = 6000                                     # default

    S = math.sqrt(2.0 * n)                          # ℓ¹‑norm target (scale‑invariant)

    # ------------------------------------------------------------------
    best_a = np.full(n, S / n, dtype=float)         # uniform fallback
    best_score = float("inf")

    def _maybe_update(a_vec: np.ndarray) -> None:
        """Evaluate a candidate and keep it if it improves the incumbent."""
        nonlocal best_score, best_a
        sc = evaluate_sequence(a_vec.tolist())
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_vec.copy()
            print(f"[improve] new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Random Dirichlet draws + extreme interleaving --------------------
    attempts = 0
    max_attempts = 2000  # enough while still cheap
    while time.time() < deadline and attempts < max_attempts:
        # concentration α∈[0.05,0.5] gives a few dominant spikes
        alpha = rng.uniform(0.05, 0.5)
        p = rng.dirichlet([alpha] * n)
        a = S * p
        _maybe_update(a)

        # sort descending – this is the raw multiset
        a_desc = np.sort(a)[::-1]
        _maybe_update(a_desc)

        # interleave extremes (large–small–large–small …)
        a_int = _interleave_extremes(a_desc)
        _maybe_update(a_int)

        # also try the pure ascending order (symmetry test)
        _maybe_update(a_desc[::-1])

        attempts += 1

    # ------------------------------------------------------------------
    # 2️⃣  Smooth‑max optimisation (L‑BFGS) ---------------------------------
    if time.time() < deadline:
        # start from the current best point
        y = np.log(best_a + 1e-12)               # log‑odds, always feasible
        mu = max(1.0, np.convolve(best_a, best_a).max())
        mu_factor = 0.80                         # slower cooling for quality
        mu_min = 1e-12
        maxiter_per_mu = 600                    # more LBFGS steps per temperature

        while mu > mu_min and time.time() < deadline:
            obj = _SmoothMaxObjective(mu, n, S)
            try:
                res = opt.minimize(
                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": maxiter_per_mu,
                             "gtol": 1e-12,
                             "disp": False}
                )
            except Exception as exc:
                print(f"[error] L‑BFGS failed: {exc}")
                break
            y = res.x
            a_cur = S * _softmax(y)
            _maybe_update(a_cur)
            mu *= mu_factor

    # ------------------------------------------------------------------
    # 3️⃣  Projected sub‑gradient descent on the hard max --------------------
    if time.time() < deadline:
        a = best_a.copy()
        step0 = S * 0.5 / n                       # heuristic initial step
        for it in range(30000):
            if time.time() >= deadline:
                break
            g, _ = _hard_subgradient(a)
            step = step0 / (1.0 + 1e-5 * it)
            a_new = a - step * g
            a_new = _project_to_simplex(a_new, S)
            sc = evaluate_sequence(a_new.tolist())
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                print(f"[subgrad] iter {it:5d} → {best_score:.6f}")

            # occasional jitter to avoid stagnation
            if it % 500 == 0:
                i, j = rng.choice(n, size=2, replace=False)
                delta = rng.random() * a[i] * 0.03
                a[i] -= delta
                a[j] += delta
                a = _project_to_simplex(a, S)

    # ------------------------------------------------------------------
    # 4️⃣  Simulated‑annealing hill‑climbing ---------------------------------
    if time.time() < deadline:
        a = best_a.copy()
        total_budget = budget_s - 0.5
        while time.time() < deadline:
            remaining = deadline - time.time()
            # linear temperature schedule (max ≈ 0.05)
            T = 0.05 * (remaining / total_budget)

            move_type = rng.integers(0, 4)  # 0‑swap, 1‑δ‑transfer, 2‑block‑avg, 3‑block‑rotate
            cand = a.copy()

            if move_type == 0:
                i, j = rng.choice(n, size=2, replace=False)
                cand[i], cand[j] = cand[j], cand[i]
            elif move_type == 1:
                i = rng.choice(np.where(cand > 1e-12)[0])
                j = rng.choice(n)
                max_delta = cand[i] * 0.05
                delta = rng.random() * max_delta
                cand[i] -= delta
                cand[j] += delta
            elif move_type == 2:
                start = rng.integers(0, n - 1)
                length = rng.integers(2, min(80, n - start))
                blk = slice(start, start + length)
                cand[blk] = cand[blk].mean()
            else:  # block rotation
                start = rng.integers(0, n - 1)
                length = rng.integers(2, min(80, n - start))
                blk = slice(start, start + length)
                seg = cand[blk].copy()
                shift = rng.integers(1, len(seg))
                cand[blk] = np.roll(seg, shift)

            cand = _project_to_simplex(cand, S)
            sc_new = evaluate_sequence(cand.tolist())

            if sc_new < best_score - 1e-12:
                best_score = sc_new
                best_a = cand.copy()
                a = cand
                print(f"[anneal] improved → {best_score:.6f}")
            elif T > 0.0 and rng.random() < math.exp(-(sc_new - best_score) / T):
                a = cand  # accept uphill move

    # ------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
