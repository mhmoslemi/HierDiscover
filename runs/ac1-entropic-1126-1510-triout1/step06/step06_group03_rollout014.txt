"""
Propose a step function (non‑negative heights) that minimizes the
black‑box ``evaluate_sequence`` score.

Algorithm overview
------------------
* The sequence is restricted to the simplex (non‑negative, Σa = 1) and a
  symmetric shape is enforced – this reduces the search space and the
  optimal solution is believed to be symmetric.
* The objective  f(a) = 2·n·max(conv(a,a))  is non‑smooth.
  We approximate the max by a smooth soft‑max:
        f_p(a) = (1/p)·log Σ exp(p·conv_k)
  and optimise f_p for increasing values of p (continuation).
* A soft‑max parametrisation a = softmax(x) guarantees feasibility.
  Gradients are obtained analytically via convolution with a weight
  vector derived from the soft‑max.
* Optimisation of the logits x is performed with Adam (adaptive
  moment estimation). After each Adam update the logits are symmetrised
  (x ← (x+rev(x))/2) which keeps the resulting a symmetric.
* Several initialisations are tried (warm‑start, uniform, random Dirichlet).
  The best result across all runs and across a schedule of increasing
  sequence lengths is kept.
* Finally a classic projected sub‑gradient descent (as in the reference
  solution) refines the best candidate.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of max_k (a*a)_k.
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(conv >= cur_max - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, p: float):
    """
    Gradient of the smooth approximation (1/p)·log Σ exp(p·conv_k).
    Returns (grad, cur_max) where grad is ∂/∂a of the smooth function
    and cur_max = max(conv).
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    # soft‑max weights (stable)
    z = p * conv
    z -= np.max(z)          # shift for numerical stability
    w = np.exp(z)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize_smoothing(
    x0: np.ndarray,
    n: int,
    p: float,
    deadline: float,
    max_iters: int = 3000,
    lr0: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max objective for a fixed p.
    Parameters are logits x (soft‑max representation of a).
    Returns the best a seen and its raw score 2·n·max(conv).
    """
    x = x0.copy()
    # ensure symmetry of the logits before first use
    x = (x + x[::-1]) / 2.0

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = None
    best_raw = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # enforce symmetry of logits at the start of each iteration
        x = (x + x[::-1]) / 2.0

        # stable soft‑max to obtain a on the simplex
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()   # Σa = 1, a ≥ 0

        # gradient of the smooth max
        grad_a, cur_max = _smooth_max_gradient(a, p)
        raw = 2.0 * n * cur_max
        if raw < best_raw:
            best_raw = raw
            best_a = a.copy()
            print(
                f"[{time.time():.1f}] n={n} p={p:.1f} it={it} raw={best_raw:.6f}"
            )

        # chain rule through the soft‑max
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape flat plateaus
        if it % 1000 == 0:
            x += np.random.randn(*x.shape) * 0.01

    # fallback (should never happen)
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_raw = 2.0 * n * _fft_self_convolve_max(best_a)

    return best_a, best_raw


# ----------------------------------------------------------------------
def _optimise_length_smoothing(
    n: int,
    deadline: float,
    prev_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length n using the smoothing‑Adam scheme.
    Multiple initialisations are tried; the best result is returned.
    """
    # ----- build list of start logits ---------------------------------
    init_logits = []

    # warm‑start from previous solution (interpolated to the current length)
    if prev_seq is not None:
        cand = prev_seq
        eps = 1e-12
        init_logits.append(np.log(cand + eps))

    # uniform start
    init_logits.append(np.zeros(n, dtype=np.float64))

    # a few random Dirichlet initialisations
    for _ in range(3):
        alpha = np.random.rand() * 4.0 + 0.1  # α ∈ (0.1,4.1)
        dir_sample = np.random.gamma(alpha, 1.0, size=n)
        dir_sample /= dir_sample.sum()
        init_logits.append(np.log(dir_sample + 1e-12))

    # ----- continuation over p (smooth‑max temperature) ----------------
    p_schedule = [5.0, 10.0, 20.0, 40.0, 80.0]

    best_local_seq = None
    best_local_raw = float("inf")

    for start_ix, x0 in enumerate(init_logits):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] length {n}, start {start_ix+1}/{len(init_logits)}")
        # start from the current logits
        x = x0.copy()
        seq = None
        raw = float("inf")
        for p in p_schedule:
            # allow a few thousand Adam steps for each p
            max_iters = int(2000 * (n / 5000))
            # respect the remaining time budget
            time_left = deadline - time.time()
            # if we have less than 5 seconds, break early
            if time_left < 5.0:
                break
            seq, raw = _adam_optimize_smoothing(
                x, n, p, deadline, max_iters=max_iters, lr0=0.12
            )
            # initialise next stage from the best found sequence
            x = np.log(seq + 1e-12)
            # keep symmetry of logits
            x = (x + x[::-1]) / 2.0
        # keep the best across different starts
        if raw < best_local_raw:
            best_local_raw = raw
            best_local_seq = seq.copy()

    # safety fallback
    if best_local_seq is None:
        best_local_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_local_raw = 2.0 * n * _fft_self_convolve_max(best_local_seq)

    return best_local_seq, best_local_raw


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Entry point for the competition harness.
    Returns a list of non‑negative heights that (hopefully) yields a low
    ``evaluate_sequence`` score.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a small safety margin – the platform kills the process exactly at the deadline
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # try to reuse a previously discovered construction if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        s = prev.sum()
        if s > 0.0:
            prev = prev / s
        else:
            prev = None
    except Exception:
        prev = None

    # ------------------------------------------------------------------
    # list of lengths – larger n gives a finer discretisation but costs more time
    lengths = [2000, 4000, 8000, 12000, 16000]

    best_seq = None
    best_raw = float("inf")

    # optimise each length in increasing order – each optimisation can reuse the previous best
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length {n} ===")
        warm = None
        if best_seq is not None:
            warm = _interpolate_to_length(best_seq, n)
        seq, raw = _optimise_length_smoothing(n, deadline, prev_seq=warm)
        print(f"  finished length {n}: best raw = {raw:.6f}")
        if raw < best_raw:
            best_raw = raw
            best_seq = seq.copy()
        # prepare warm‑start for the next length
        best_seq = best_seq  # already the best for this n

    # ------------------------------------------------------------------
    # final fine‑tuning with a classic projected sub‑gradient method
    if best_seq is not None:
        a = best_seq.copy()
        n = a.shape[0]
        it = 0
        while time.time() < deadline:
            it += 1
            grad, cur_max = _max_subgradient_step(a)
            step = 0.07 / math.sqrt(it)               # diminishing steps
            a = a - step * grad
            a = _project_onto_simplex(a)               # keep on the simplex
            # occasional tiny random perturbation
            if it % 800 == 0 and np.random.rand() < 0.06:
                a += np.random.randn(n) * 5e-7
                a = _project_onto_simplex(a)
            cur_score = 2.0 * n * cur_max
            if cur_score < best_raw - 1e-12:
                best_raw = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] post‑refinement it={it} raw={best_raw:.6f}")

    # ------------------------------------------------------------------
    # final evaluation through the provided black‑box
    final_list = best_seq.tolist()
    final_score = evaluate_sequence(final_list)
    print(
        f"\nFinal evaluated score: {final_score:.6f} (length={len(final_list)})"
    )
    return final_list
