def _lp_refinement(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    # compute convolution max M
    b = _fft_self_convolve(a)
    M = b.max()
    # Build matrix A: each row corresponds to conv index k.
    # Represent A as dense matrix of shape (2n-1, n)
    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        # convolution: A[k,i] = a[i] if k - i in [0,n-1]
        # That is, for k from i to i + n - 1, set A[k,i] = a[i]
        A[i:i+n, i] = a[i]  # This sets rows i..i+n-1
    # Right-hand side: M
    b_rhs = np.full(rows, M, dtype=np.float64)
    # objective c = ones
    c = -np.ones(n)  # because linprog does min c^T x
    # bounds: g_i >= 0
    bounds = [(0, None)] * n
    # Solve LP
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A, b_ub=b_rhs, bounds=bounds, method='highs')
    if not res.success:
        # fallback: return a unchanged
        return a, _max_convolution_score(a)
    g = res.x
    # Scale to match sum=1 (or to sum sqrt(2n)?) We'll keep normalized to sum=1.
    sg = g.sum()
    if sg <= 0:
        return a, _max_convolution_score(a)
    g = g / sg
    # return g and its score
    return g, _max_convolution_score(g)

   All methods work on the simplex (ℓ₁‑norm = 1) and use the exact sub‑gradient
   of the max‑convolution.

3. **Hill climbing** – a cheap random‐transfer move is repeatedly applied to
   the current best sequence until the time budget expires.  Small jitters are
   added periodically to escape plateaus.

The routine respects a 1000 s budget (a 0.5 s safety margin) and always
returns a list of non‑negative floats that can be fed directly to
``evaluate_sequence``.
"""

import time
import numpy as np

# -------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Score  = 2·n·max(a*a)  under the assumption sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with the indicator of active entries
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # correction for the diagonal terms (i=j) that appear only once
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# -------------------------------------------------------------
def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Gamma‑Dirichlet, α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


# -------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 20000,
                          eta0: float = 0.6, decay: float = 0.9995):
    """Mirror descent with KL‑divergence (exponentiated‑gradient)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _projected_gradient_descent(a0: np.ndarray, deadline: float,
                               max_iters: int = 20000,
                               step0: float = 0.5):
    """Plain projected gradient descent on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        step = step0 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _lp_refinement(a0: np.ndarray, deadline: float):
    """
    One linear‑programming refinement step for n ≤ 2000:
    maximise Σg subject to (a0 * g) ≤ M and g ≥ 0.
    Returned vector is normalised to sum = 1.
    """
    a = a0.copy()
    n = a.shape[0]
    if n > 2000:
        # not worth the memory / time
        return a, _max_convolution_score(a)

    # current bound
    conv = _fft_self_convolve(a)
    M = conv.max()

    # Build the Toeplitz matrix A (shape (2n-1, n))
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column i contributes a[i] to rows i … i+n-1
    for i in range(n):
        A[i:i + n, i] = a[i]

    b_rhs = np.full(rows, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # linprog minimises cᵀx

    try:
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A, b_ub=b_rhs, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})
        if not res.success:
            return a, _max_convolution_score(a)
        g = res.x
    except Exception:  # SciPy missing or failure
        return a, _max_convolution_score(a)

    sg = g.sum()
    if sg == 0:
        return a, _max_convolution_score(a)

    g_norm = g / sg        # force Σg = 1 (same normalisation as a)
    return g_norm, _max_convolution_score(g_norm)

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step function minimising evaluate_sequence."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Choose lengths to explore (including some very large ones)
    try:
        base_n = len(height_sequence_1)      # type: ignore[name-defined]
    except Exception:
        base_n = 4000

    lengths = set()
    for f in (0.5, 0.75, 1.0, 1.5, 2.0, 3.0, 4.0):
        n = int(base_n * f)
        if 500 <= n <= 50000:
            lengths.add(n)
    for _ in range(6):
        n = np.random.randint(500, min(50000, base_n * 5) + 1)
        lengths.add(n)
    # also make sure we test a few modest sizes (helpful for the LP step)
    for n in (800, 1200, 1600):
        if 500 <= n <= 50000:
            lengths.add(n)
    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # 2. Broad sampling
    candidates = []                     # (sequence, score)

    # dense log‑grid for the beta shape parameter
    beta_ps = np.logspace(np.log10(0.01), np.log10(5.0), num=30)

    for n in lengths:
        if time.time() >= deadline:
            break

        # arcsine
        a = _arcsine_distribution(n)
        candidates.append((a, _max_convolution_score(a)))

        # triangular
        a = _triangular_distribution(n)
        candidates.append((a, _max_convolution_score(a)))

        # exponential (slightly flatter than before)
        a = _exponential_distribution(n, lam=0.003)
        candidates.append((a, _max_convolution_score(a)))

        # symmetric beta family on a fine grid
        for p in beta_ps:
            if time.time() >= deadline:
                break
            a = _beta_symmetric_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))

        # a couple of random Dirichlet draws
        for _ in range(2):
            if time.time() >= deadline:
                break
            a = _random_dirichlet(n)
            candidates.append((a, _max_convolution_score(a)))

    # ------------------------------------------------------------------
    # 3. Add the previously best known construction (if any)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Keep the most promising seeds
    candidates.sort(key=lambda x: x[1])
    top_k = 10
    top_candidates = candidates[:top_k]
    print(f"Selected {len(top_candidates)} seeds for refinement; best score {top_candidates[0][1]:.6f}")

    # ------------------------------------------------------------------
    # 5. Refinement phase
    best_seq = top_candidates[0][0].copy()
    best_score = top_candidates[0][1]

    for idx, (seed_seq, seed_score) in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        n = seed_seq.shape[0]

        # (a) LP refinement for modest sizes
        if n <= 2000:
            try:
                seq_lp, sc_lp = _lp_refinement(seed_seq, deadline)
                if sc_lp < best_score - 1e-14:
                    best_score, best_seq = sc_lp, seq_lp.copy()
                    print(f"[{time.time():.1f}] LP refinement seed {idx+1}/{len(top_candidates)} → {sc_lp:.6f}")
            except Exception:
                pass

        # (b) exponentiated‑gradient mirror descent
        seq_exp, sc_exp = _exp_gradient_descent(seed_seq, deadline,
                                                max_iters=8000,
                                                eta0=0.8, decay=0.9995)
        if sc_exp < best_score - 1e-14:
            best_score, best_seq = sc_exp, seq_exp.copy()
            print(f"[{time.time():.1f}] exp‑grad seed {idx+1}/{len(top_candidates)} → {sc_exp:.6f}")

        # (c) plain projected‑gradient descent
        seq_pg, sc_pg = _projected_gradient_descent(seed_seq, deadline,
                                                    max_iters=8000,
                                                    step0=0.7)
        if sc_pg < best_score - 1e-14:
            best_score, best_seq = sc_pg, seq_pg.copy()
            print(f"[{time.time():.1f}] proj‑grad seed {idx+1}/{len(top_candidates)} → {sc_pg:.6f}")

    # ------------------------------------------------------------------
    # 6. Local random‑transfer hill climbing (pure exploitation)
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    it = 0
    max_transfer = 0.07   # up to 7 % of total mass per trial

    while time.time() < deadline:
        it += 1
        # geometrically decaying transfer amount
        transfer = max_transfer * (0.5 ** (it / 12000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={it} new best {best_score:.6f}")

        # occasional jitter to jump out of tiny plateaus
        if it % 5000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 1e-4
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
