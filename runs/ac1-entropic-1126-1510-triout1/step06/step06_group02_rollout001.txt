"""
Propose a step‑function with a low value of evaluate_sequence.

Algorithm (high‑level)
----------------------
* Work on a probability vector a (sum = 1, a_i ≥ 0).  The objective reduces to
  `score = 2·n·max(a * a)` (the scaling of a does not matter).
* For several problem sizes n (around the size of the previous best) we
  generate a diverse pool of feasible start vectors:
    – Beta(α,β) densities on a grid of parameters,
    – exponential and edge‑spike shapes,
    – random Dirichlet draws,
    – the previously found best vector (interpolated if necessary).
* Each start is refined by *Adam* on the simplex using the exact sub‑gradient
  of `max(conv)`.  To stay on the simplex we work in the log‑odds domain
  (`z` with a = softmax(z)).  After each update we enforce symmetry
  a_i = a_{n‑1‑i} (the optimal solution is known to be symmetric) by
  averaging the logits with their reversed copy.
* The cheap objective `cheap_score = 2·n·max(conv)` is monitored; when it
  improves we keep the best local vector.  If it drops below the target
  (≈ 1.503) the search stops immediately.
* After the per‑length search we perform a final polishing run on the
  globally best vector with many more Adam steps.
* The routine respects a user‑provided time budget (default 1000 s) and
  returns the best sequence found as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, v.size + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    Returned gradient corresponds to d(max(conv))/da (no extra scaling).
    """
    n = a.shape[0]
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)

    # Average over all active shifts (sub‑gradient of max)
    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    # Convolution of reversed a with weights w
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for the central shift (avoid double counting a_i*a_i)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad


# ----------------------------------------------------------------------
def _adam_step(z: np.ndarray,
               g: np.ndarray,
               m: np.ndarray,
               v: np.ndarray,
               t: int,
               lr: float = 0.5,
               beta1: float = 0.9,
               beta2: float = 0.999,
               eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Adam update on the unconstrained variable z."""
    m = beta1 * m + (1.0 - beta1) * g
    v = beta2 * v + (1.0 - beta2) * (g * g)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z, m, v


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float = 1.0) -> np.ndarray:
    """Geometric‑like decreasing shape: a_i ∝ exp(-lam·i/(n‑1))."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * i / max(1, n - 1))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _edge_spike_distribution(n: int, p: float) -> np.ndarray:
    """
    Two‑point spike at the ends with weight p each, uniform interior.
    Returns a probability vector (sum = 1).  p must be ≤ 0.5.
    """
    if n < 2:
        return np.full(1, 1.0, dtype=np.float64)
    p = min(max(p, 0.0), 0.5)
    interior = max(n - 2, 0)
    if interior == 0:
        # all mass at the two ends
        vals = np.array([0.5, 0.5], dtype=np.float64) if n == 2 else np.full(n, 1.0 / n)
        return vals
    rest = 1.0 - 2.0 * p
    interior_val = rest / interior
    vals = np.full(n, interior_val, dtype=np.float64)
    vals[0] = p
    vals[-1] = p
    # Numerical safety
    vals = _project_onto_simplex(vals, s=1.0)
    return vals


# ----------------------------------------------------------------------
def _dirichlet_distribution(n: int, alpha: float = 1.0) -> np.ndarray:
    """Draw a random point from Dirichlet(α,…,α) and return as a probability vector."""
    if alpha <= 0:
        raise ValueError("Dirichlet concentration must be positive")
    sample = np.random.gamma(alpha, 1.0, size=n)
    sample /= sample.sum()
    return sample


# ----------------------------------------------------------------------
def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Force symmetry a_i = a_{n-1-i} (preserves sum)."""
    return (a + a[::-1]) / 2.0


# ----------------------------------------------------------------------
def cheap_score(a: np.ndarray) -> float:
    """Fast bound equal to the true objective when sum(a)=1."""
    n = a.shape[0]
    return 2.0 * n * _fft_self_convolve(a).max()
