def _beta_distribution(n, alpha, beta):
    x = (np.arange(n) + 0.5)/ n
    # if alphas <1, values blow up near edges; keep numeric
    w = np.power(x, alpha-1) * np.power(1-x, beta-1)

    a = a0.copy()

    best_score = _max_convolution_score(a)

Propose a low‑score non‑negative step function.

The search proceeds in three stages:

1. **Exploratory sampling** – for several lengths we generate candidates from
   many analytic families (arcsine, power‑edge, symmetric/asymmetric beta,
   Hann/Hamming/Blackman, Kaiser, Gaussian, exponential, triangular) and

   ``score = 2·n·max(a*a)`` (exact for ``sum(a)=1``).

2. **Refinement** – the best few seeds are improved by three first‑order
   optimisation schemes that respect the simplex constraint:
   * exponentiated‑gradient (mirror descent w.r.t. KL)

   by transferring a small amount of mass between two random positions, by
   swapping two entries, and by adding a tiny random jitter.  Any accepted
   improvement is kept.

The algorithm respects the supplied time budget (≈ 1000 s) and always returns

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Score 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.
    Returns (grad, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives raw gradient
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active positions (i=j)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active

# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _power_edge_distribution(n: int, p: float) -> np.ndarray:
    """w_i ∝ ((i+0.5)*(n-i-0.5))^{-p}  (p≥0)."""
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - i - 0.5)
    # avoid overflow for very small denom when p large
    with np.errstate(divide='ignore'):
        vals = denom ** (-p)
    vals[np.isnan(vals)] = 0.0
    vals /= vals.sum()
    return vals


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric or asymmetric Beta(α,β) density on a discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # Guard against α or β being too close to 0
    eps = 1e-12
    a = max(alpha, eps)
    b = max(beta, eps)
    vals = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    # numerical safety
    if not np.all(np.isfinite(vals)):
        vals = np.ones_like(vals)
    if vals.sum() == 0:
        vals = np.ones_like(vals)
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    vals /= vals.sum()
    return vals


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    vals /= vals.sum()
    return vals


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    vals /= vals.sum()
    return vals


def _kaiser_window(n: int, beta_param: float = 5.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (n - 1) / 2.0
    val = i0(beta_param * np.sqrt(1.0 - ((i - a) / a) ** 2)) / i0(beta_param)
    val /= val.sum()
    return val


def _gaussian_window(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if vals.sum() == 0:
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    """Draw a Dirichlet point with concentration α (scalar)."""
    x = np.random.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    return x / s

# Optimisation primitives -----------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 50000,
                          eta0: float = 0.6):
    """Mirror descent with KL push (exponentiated‑gradient)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _additive_subgrad_descent(a0: np.ndarray, deadline: float,
                              max_iters: int = 50000,
                              step0: float = 0.8):
    """Projected additive sub‑gradient descent on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        step = step0 / np.sqrt(t)
        a = a - step * grad  # move opposite to sub‑gradient
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 50000,
                  step0: float = 0.12,
                  beta1: float = 0.9,
                  beta2: float = 0.999):
    """Adam‑style adaptive moment optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

    while time.time() < deadline:
        iteration += 1
        # geometric decay of the transferred amount
        transfer = max_transfer * (0.5 ** (iteration / 14000.0))

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = rng.random() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_seq, best_score = a.copy(), new_score
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} new best {best_score:.6f}")

        # occasional swap move (no net mass change)
        if iteration % 2500 == 0:
            i1 = rng.integers(n)
            i2 = rng.integers(n - 1)
            if i2 >= i1:
                i2 += 1
            a_swap = a.copy()
            a_swap[i1], a_swap[i2] = a_swap[i2], a_swap[i1]
            swap_score = _max_convolution_score(a_swap)
            if swap_score < cur_score - 1e-12:
                a = a_swap
                cur_score = swap_score
                if swap_score < best_score - 1e-14:
                    best_seq, best_score = a.copy(), swap_score
                    print(
                        f"[{time.time():.1f}] hill‑climb iter={iteration} swap improved to {best_score:.6f}"
                    )

        # occasional tiny jitter to escape plateaus
        if iteration % 4000 == 0 and rng.random() < 0.12:
            jitter = rng.normal(scale=1e-4, size=n)
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
