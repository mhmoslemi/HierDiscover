def propose_candidate(seed=12345,budget_s=1000):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5

    # decide n based on time left, maybe 9000 or larger if we have time.
    # Use n = 12000 if deadline - time() > some threshold.
    # But to keep uniform, use n = 9000.

    n = 9000
    S = math.sqrt(2.0 * n)

    # Setup FFT parameters
    L = 1 << ((2*n - 1).bit_length())  # next power of 2

    # define projection function _project_to_simplex already present.

    # define conv functions using FFT
    def conv_self(a):
        A = np.fft.rfft(a, L)
        conv = np.fft.irfft(A*A, L)[:2*n-1]
        return conv

    def conv_wr(w, a_rev):
        W = np.fft.rfft(w, L)
        AR = np.fft.rfft(a_rev, L)
        conv = np.fft.irfft(W*AR, L)[:2*n-1]
        return conv

    # initial sequences (list of start vectors)

    init_candidates = []

    # uniform
    a_uniform = np.full(n, S/n)
    init_candidates.append(a_uniform)

    # decreasing exponential (choose random decay)
    exp_decay = rng.uniform(0.0005, 0.005) * (n/2000.0)
    a_exp = np.exp(-exp_decay * np.arange(n))
    a_exp = a_exp / a_exp.sum() * S
    init_candidates.append(a_exp)

    # increasing exponential (mirror)
    a_exp_inc = np.exp(-exp_decay * np.arange(n-1, -1, -1))
    a_exp_inc = a_exp_inc / a_exp_inc.sum() * S
    init_candidates.append(a_exp_inc)

    # randomly Dirichlet draws
    for _ in range(5):
        alpha = rng.uniform(0.05, 0.8)
        p = rng.dirichlet([alpha]*n)
        a = p * S
        init_candidates.append(a)

    # maybe include reference
    try:
        a_ref = np.asarray(height_sequence_1, dtype=float)
        a_ref = _project_to_simplex(a_ref, S)
        init_candidates.append(a_ref)

        pass

    # Maybe add sorted descending version of each candidate
    for a in list(init_candidates):
        init_candidates.append(np.sort(a)[::-1])

    # Best candidate init

    best_score = float('inf')

        sc = evaluate_sequence(a.tolist())
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()

    def _softmax(y):

        y_max = np.max(y)
        e = np.exp(y - y_max)
        return e / e.sum()

    def _smooth_obj_and_grad(y, mu):

        p = _softmax(y)
        a = S * p
        conv = conv_self(a)
        max_c = conv.max()
        if mu > 0:
            diff = (conv - max_c) / mu
            diff = np.clip(diff, -700, 700)
            exp_diff = np.exp(diff)
            Z = exp_diff.sum()
            w = exp_diff / Z

        else:

            w = np.zeros_like(conv)

        grad_y = S * p * (g_a - inner)
        return float(f_val), grad_y.astype(np.float64)

            g_a = 2.0 * conv_wr_full[n-1:n-1+n]

            # gradient wrt p = a/S
            p = a / S
            inner = np.dot(g_a, p)
            grad_p = p * (g_a - inner)  # gradient of max wrt p (since a=S p, chain factor accounted later)
            # convert to gradient wrt a: grad_a = grad_p * (1/S)?? Actually a = S p, so derivative da/dp = S, but we already have g_a = ∂f/∂a. So we can use g_a directly.
            # We'll use g_a directly for update in a space.

            # Use momentum update

            momentum = beta * momentum - step * g_a

            # project to simplex
            a_new = _project_to_simplex(a_new, S)

            sc = evaluate_sequence(a_new.tolist())
            if sc < best_score - 1e-12:

                best_a = a_new.copy()

            else:
                a = a_new

    if time.time() < deadline:

                j = rng.integers(0, n)

                delta = rng.random() * max_delta
                cand[i] -= delta
                cand[j] += delta
            elif move_type == 2:  # block average

                avg = cand[block].mean()
                cand[block] = avg
            else:  # permute block

"""
Propose a low‑score step function.

The algorithm works in four stages:

1️⃣  Multiple random restarts of a smooth‑max surrogate.
    The surrogate f_µ = max(conv) + µ·log‑sum‑exp((conv‑max)/µ) is
    minimised with L‑BFGS‑B on the soft‑max (log‑odds) parametrisation.
    µ is decreased geometrically (µ←0.85·µ) until it reaches a modest
    floor (µ≈1e‑4).  This yields a good approximation of the true
    max‑objective while providing a usable gradient.

2️⃣  Projected sub‑gradient descent (with simple momentum) on the
    exact max‑objective.  The sub‑gradient is obtained analytically.
    The iteration count is large (up to 8·10⁴ steps) so that the
    sequence can escape shallow plateaus.

3️⃣  Stochastic hill‑climbing.  Random swaps, δ‑mass transfers,
    block‑averaging and block‑permutations are tried; a move is kept
    only if it strictly improves the exact score.  The number of
    trials is scaled to the remaining time budget.

4️⃣  Final monotone‑sorting heuristics (descending, ascending and an
    interleaved big‑small ordering) are evaluated for a possible
    finishing gain.

All convolutions are performed with the FFT (O(n log n)) to keep the
runtime low even for n≈9000.  The ℓ¹‑simplex projection is the
standard Euclidean projection (O(n)).  The routine respects a user‑supplied
time budget (default 1000 s) and returns the best sequence discovered.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z <= 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors via FFT."""
    size = 1 << ((len(x) + len(y) - 1).bit_length())
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    conv = np.fft.irfft(X * Y, size)
    return conv[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·n‑1)."""
    return _conv_fft(a, a)


def _conv_wr(w: np.ndarray, a_rev: np.ndarray) -> np.ndarray:
    """Convolution of w with the reversed a (length 3·n‑2)."""
    return _conv_fft(w, a_rev)


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(y: np.ndarray, mu: float, S: float):
    """
    Value and gradient of the smoothed max surrogate for a given
    log‑odds vector y and smoothing parameter mu.
    """
    p = _softmax(y)                     # Σp = 1, p ≥ 0
    a = S * p                           # Σa = S

    conv = _conv_self(a)                # length 2·n‑1
    max_c = conv.max()

    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)    # protect exp
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = mask.sum()
        w = np.zeros_like(conv)
        if cnt > 0:
            w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. a
    a_rev = a[::-1]
    conv_wr_full = _conv_wr(w, a_rev)          # length 3·n‑2
    n = len(a)
    g_a = 2.0 * conv_wr_full[n - 1 : n - 1 + n]   # slice length n

    # gradient w.r.t. y (chain rule through soft‑max)
    inner = np.dot(g_a, p)
    grad_y = S * p * (g_a - inner)
    return float(f_val), grad_y.astype(np.float64)


class _SmoothObj:
    """Callable for scipy.optimize returning (value, gradient)."""
    __slots__ = ("mu", "S")
    def __init__(self, mu: float, S: float):
        self.mu = mu
        self.S = S
    def __call__(self, y: np.ndarray):
        return _smooth_obj_and_grad(y, self.mu, self.S)

    init_candidates.append(np.full(n, S / n))
