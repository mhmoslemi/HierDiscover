# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float = 1.0) -> np.ndarray:
    """Geometric‑like decreasing shape: a_i ∝ exp(-lam·i/(n‑1))."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * i / max(1, n - 1))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _gaussian_distribution(n: int, sigma_frac: float = 0.2) -> np.ndarray:
    """Symmetric Gaussian shape centred at (n‑1)/2, sigma as a fraction of n."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    sigma = sigma_frac * n
    vals = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate of max(conv).
    a   – probability vector (Σa=1)
    conv – full self‑convolution of a (length 2·n‑1)
    alpha – temperature multiplier (larger → sharper)
    """
    n = a.shape[0]
    cur_max = conv.max()
    # stable soft‑max
    exp_term = np.exp(alpha * (conv - cur_max))
    w = exp_term / exp_term.sum()
    # convolution of reversed a with the weights w
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]   # shape (n,)
    return grad


# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the true max(conv).
    Returns ∇_a max(conv).
    """
    n = a.shape[0]
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (remove double counting of a_i * a_i for even shifts)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad


# ----------------------------------------------------------------------
def _adam_step(z: np.ndarray,
               g: np.ndarray,
               m: np.ndarray,
               v: np.ndarray,
               t: int,
               lr: float = 0.5,
               beta1: float = 0.9,
               beta2: float = 0.999,
               eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Adam update on the unconstrained variable z."""
    m = beta1 * m + (1.0 - beta1) * g
    v = beta2 * v + (1.0 - beta2) * (g * g)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z, m, v


# ----------------------------------------------------------------------
def cheap_score(a: np.ndarray) -> float:
    """Cheap score = 2·n·max(conv); equal to the true objective when Σa=1."""
    n = a.shape[0]
    return 2.0 * n * _fft_self_convolve(a).max()


# ----------------------------------------------------------------------
def _optimize_candidate(a0: np.ndarray,
                        n: int,
                        time_budget: float) -> tuple[np.ndarray, float]:
    """
    Refine one starter vector by Adam on the simplex.
    Returns the best vector found (Σa=1) and its cheap score.
    """
    start = time.time()
    # initialise logits
    a0 = np.maximum(a0, 1e-15)
    a0 = _project_onto_simplex(a0, s=1.0)
    z = np.log(a0)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    best_a = a0.copy()
    best_sc = cheap_score(a0)

    # temperature schedule for the soft‑max surrogate
    beta = 1.0
    beta_factor = 1.005
    beta_max = 2000.0

    it = 0
    while True:
        now = time.time()
        if now - start > time_budget:
            break
        it += 1

        # ---- current point on simplex ----
        a = np.exp(z)
        a /= a.sum()
        conv = _fft_self_convolve(a)
        cur_sc = 2.0 * n * conv.max()

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{now:.1f}] n={n} it={it} score={cur_sc:.6f}")
            if best_sc <= 1.503:
                # very good – we can stop early
                break

        # ---- gradient ----
        if beta < beta_max:
            grad_a = n * _softmax_gradient(a, conv, beta)
            beta *= beta_factor
        else:
            grad_a = n * _exact_gradient(a, conv)

        # gradient w.r.t logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)

        # ---- Adam step with a decreasing learning‑rate ----
        lr = 0.6 / np.sqrt(it)   # simple schedule
        z, m, v = _adam_step(z, grad_z, m, v, it, lr=lr)

        # occasional jitter to escape plateaus
        if it % 4000 == 0 and np.random.rand() < 0.12:
            z += np.random.randn(n) * 1e-7

    return best_a, best_sc


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 1234, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the supplied
    evaluation function.  The routine respects the time budget (default
    1000 s) and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # base problem size (use previously discovered length if available)
    try:
        base_n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # problem sizes to explore (all ≥ 3000)
    length_factors = (0.75, 1.0, 1.25, 1.5, 2.0)
    lengths = sorted(
        max(3000, int(base_n * f))
        for f in length_factors
        if int(base_n * f) >= 3000
    )
    if not lengths:
        lengths = [max(3000, base_n)]

    overall_best_seq = None               # type: np.ndarray | None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # time slice for this n (≈30 % of remaining budget, capped)
        remaining = max(0.0, deadline - time.time())
        slice_deadline = time.time() + min(remaining * 0.35, 250.0)
        print(f"[{time.time():.1f}] Exploring length n={n}")

        # --------------------------- 0. build initial pool ---------------------------
        pool = []

        # grid of Beta distributions
        alphas = np.logspace(-0.7, 0.7, num=7)   # ~0.2 … 5
        betas = np.logspace(-0.7, 0.7, num=7)
        for a in alphas:
            for b in betas:
                pool.append(_beta_distribution(n, a, b))

        # a few hand‑picked shapes
        pool.append(_beta_distribution(n, 0.2, 0.8))
        pool.append(_beta_distribution(n, 0.8, 0.2))
        pool.append(_exponential_distribution(n, lam=0.5))
        pool.append(_exponential_distribution(n, lam=1.5))
        for sigma in (0.08, 0.12, 0.18):
            pool.append(_gaussian_distribution(n, sigma))

        # uniform & arcsine
        pool.append(_project_onto_simplex(np.full(n, 1.0 / n)))
        pool.append(_beta_distribution(n, 0.5, 0.5))   # arcsine

        # previous construction (interpolated if needed)
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
            if prev.size != n:
                x_old = np.linspace(0.0, 1.0, prev.size)
                x_new = np.linspace(0.0, 1.0, n)
                a = np.interp(x_new, x_old, prev)
                a = _project_onto_simplex(a, s=1.0)
            else:
                a = prev.copy()
            pool.append(a)
        except Exception:
            pass

        # a handful of random Dirichlet draws
        for _ in range(12):
            a = np.random.rand(n)
            a = _project_onto_simplex(a, s=1.0)
            pool.append(a)

        # keep the best few starters (based on cheap score)
        scored = [(cheap_score(p), p) for p in pool]
        scored.sort(key=lambda x: x[0])
        starters = [p.copy() for _, p in scored[:10]]
        print(f"  top cheap scores: {[s for s, _ in scored[:5]]}")

        # --------------------------- 1. optimise each starter --------------------
        for sid, a0 in enumerate(starters):
            now = time.time()
            sub_deadline = now + min(80.0, slice_deadline - now)
            if sub_deadline - now < 5.0:
                continue

            best_local_a, best_local_score = _optimize_candidate(a0, n, sub_deadline - now)

            # final evaluation with the true black‑box (should match cheap score)
            try:
                final_score = evaluate_sequence(best_local_a.tolist())
            except Exception:
                final_score = best_local_score

            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = best_local_a.copy()
                print(
                    f"*** Global improvement: n={n} starter={sid} → "
                    f"score={overall_best_score:.6f}"
                )
                if overall_best_score <= 1.503:
                    print("Target reached – exiting early.")
                    return overall_best_seq.tolist()

            # respect per‑length deadline
            if time.time() >= slice_deadline:
                break

        # global time check
        if time.time() >= deadline:
            break
        if overall_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    if overall_best_seq is None:
        n_fallback = 4000
        overall_best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        overall_best_score = cheap_score(overall_best_seq)
        print("Fallback solution used.")

    print(
        f"Final best score: {overall_best_score:.6f} (length = {len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
