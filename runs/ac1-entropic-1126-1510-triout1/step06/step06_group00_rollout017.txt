def _subgrad_descent(a0, deadline, max_iters=None, eta0=0.2):
   a = a0.copy()
   best = a.copy()
   best_score = _max_convolution_score(a)
   t = 0
   while time.time() < deadline and (max_iters is None or t < max_iters):
       t += 1
       grad, cur_max, active = _subgrad(a)
       # compute step size
       eta = eta0 / np.sqrt(t)
       a_new = a - eta * grad
       a_new = _project_onto_simplex(a_new,1.0)
       new_score = _max_convolution_score(a_new)
       if new_score < best_score:
           best_score = new_score
           best = a_new.copy()
       a = a_new
   return best, best_score

"""
A fast, multi‑phase optimizer for the step‑function problem.

Phases
------
1. **Broad sweep** – for a handful of lengths (≈ 5 k … 20 k) we evaluate
   several analytic families (β‑distribution, power‑law, exponential,
   triangular, arcsine) on a fine grid of parameters and keep the best few
   candidates (≈ 10).

2. **Long projected sub‑gradient descent** – each seed is refined with a
   pure projected sub‑gradient method (step size ηₜ = η₀/√t).  The method is
   convex, cheap (two FFTs per iteration) and converges much deeper than the
   short Adam/exp‑gradient runs used before.

3. **Targeted mass‑transfer hill‑climb** – using the last sub‑gradient we
   repeatedly move a small fraction of mass from the index with the largest
   positive gradient to the index with the most negative gradient (the
   direction that most reduces the active convolution).  This often squeezes
   the last few digits.

All phases respect the global time budget (≈ 1000 s) and always return a
non‑negative list that can be fed directly to ``evaluate_sequence``.
"""

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a: np.ndarray) -> float:
    """
    Return the objective 2·n·max(a*a) assuming sum(a)=1.
    (Exact value used by ``evaluate_sequence`` after the normalisation.)
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.
    Returns (grad, cur_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (kept from the reference implementation)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active

def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 100_000,
                         eta0: float = 0.5, decay: float = 0.9999):
    """Exponentiated‑gradient (mirror descent w.r.t. KL)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 100_000,
                  step0: float = 0.1,
                  beta1: float = 0.9,
                  beta2: float = 0.999):
    """Adam‑style adaptive moment optimiser on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _subgrad_descent(a0: np.ndarray, deadline: float,
                     max_iters: int = 200_000,
                     eta0: float = 0.6):
    """Projected sub‑gradient descent (ηₜ = η₀/√t)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * grad
        a_new = _project_onto_simplex(a_new, s=1.0)
        cur_score = _max_convolution_score(a_new)

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score

def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (α=1 for all coordinates)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on {0,…,n‑1} (continuous version, then normalised)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    s = w.sum()
    if s == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _powerlaw_distribution(n: int, alpha: float) -> np.ndarray:
    """U‑shaped distribution a_i ∝ (i·(n‑i))^{-α} (both ends singular)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = (i * (n + 0.5 - i)) ** (-alpha)
    w /= w.sum()
    return w


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    """Exponentially decaying mass from the centre outwards."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-lam * np.abs(idx - centre))
    w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    """Symmetric triangular shape (peaked at the centre)."""
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete version of the arcsine law (β‑distribution with p=½)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step‑function minimising ``evaluate_sequence``."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # 1. Determine a set of lengths to explore.
    try:
        base_n = len(height_sequence_1)            # type: ignore[name-defined]
        if base_n < 500:
            base_n = 500
    except Exception:
        base_n = 2000

    lengths = set()
    for factor in (0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 2.5, 3.0):
        n = int(base_n * factor)
        if 500 <= n <= 20000:
            lengths.add(n)
    # a few random lengths in the same range
    for _ in range(5):
        n = np.random.randint(500, min(20000, base_n * 4) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    print(f"[{time.time():.1f}] Exploring lengths: {lengths}")

    # --------------------------------------------------------------
    # 2. Broad, but memory‑light, sweep over analytic families.
    #    For each family we keep the best parameter value for the current length.
    candidates = []   # (score, sequence) tuples

    # grids for parameters
    beta_ps = np.concatenate((
        np.linspace(0.02, 0.5, 45, endpoint=False),
        np.linspace(0.5, 2.0, 55)
    ))
    power_alphas = np.linspace(0.30, 1.50, 39)        # arcsine ≈0.5 lies inside
    expo_lams = [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]

    for n in lengths:
        if time.time() >= deadline:
            break

        # ---- arcsine (fixed) ------------------------------------
        a = _arcsine_distribution(n)
        candidates.append((_max_convolution_score(a), a.copy()))
        print(f"[{time.time():.1f}] n={n:5d} arcsine score={candidates[-1][0]:.6f}")

        # ---- beta family (grid search) -------------------------
        best_beta = None
        best_beta_score = np.inf
        for p in beta_ps:
            a = _beta_symmetric_distribution(n, p)
            sc = _max_convolution_score(a)
            if sc < best_beta_score:
                best_beta_score = sc
                best_beta = a
        candidates.append((best_beta_score, best_beta.copy()))
        print(f"[{time.time():.1f}] n={n:5d} beta best score={best_beta_score:.6f}")

        # ---- power‑law family -----------------------------------
        best_pow = None
        best_pow_score = np.inf
        for alpha in power_alphas:
            a = _powerlaw_distribution(n, alpha)
            sc = _max_convolution_score(a)
            if sc < best_pow_score:
                best_pow_score = sc
                best_pow = a
        candidates.append((best_pow_score, best_pow.copy()))
        print(f"[{time.time():.1f}] n={n:5d} power‑law best score={best_pow_score:.6f}")

        # ---- exponential family ---------------------------------
        best_exp = None
        best_exp_score = np.inf
        for lam in expo_lams:
            a = _exponential_distribution(n, lam)
            sc = _max_convolution_score(a)
            if sc < best_exp_score:
                best_exp_score = sc
                best_exp = a
        candidates.append((best_exp_score, best_exp.copy()))
        print(f"[{time.time():.1f}] n={n:5d} exponential best score={best_exp_score:.6f}")

        # ---- triangular (fixed) --------------------------------
        a = _triangular_distribution(n)
        candidates.append((_max_convolution_score(a), a.copy()))
        print(f"[{time.time():.1f}] n={n:5d} triangular score={candidates[-1][0]:.6f}")

        # ---- mixtures of the best beta with uniform -------------
        uniform = np.full(n, 1.0 / n)
        for w in (0.1, 0.2, 0.3, 0.5):
            mix = (1.0 - w) * best_beta + w * uniform
            mix /= mix.sum()
            candidates.append((_max_convolution_score(mix), mix.copy()))
            print(f"[{time.time():.1f}] n={n:5d} beta+uni w={w:.1f} score={candidates[-1][0]:.6f}")

        # ---- random Dirichlet draws -----------------------------
        for _ in range(3):
            a = _random_dirichlet(n)
            candidates.append((_max_convolution_score(a), a.copy()))
    # --------------------------------------------------------------
    # 3. Include the previous best if it exists.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.abs(prev)          # safety
            prev /= prev.sum()
            candidates.append((_max_convolution_score(prev), prev.copy()))
            print(f"[{time.time():.1f}] previous best added, score={candidates[-1][0]:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 4. Keep the top‑k seeds for intensive refinement.
    top_k = 12
    candidates.sort(key=lambda x: x[0])               # smaller score first
    top_candidates = candidates[:top_k]
    print(f"[{time.time():.1f}] Selected {len(top_candidates)} seeds for refinement, "
          f"best score {top_candidates[0][0]:.6f}")

    # --------------------------------------------------------------
    # 5. Refinement – projected sub‑gradient + Adam + exponentiated‑gradient
    best_seq = top_candidates[0][1].copy()
    best_score = top_candidates[0][0]

    for idx, (seed_seq, seed_score) in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        n = seed_seq.shape[0]

        # a) vanilla sub‑gradient descent (long run)
        seq, sc = _subgrad_descent(seed_seq, deadline,
                                   max_iters=150_000, eta0=0.6)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} "
                  f"sub‑grad improvement: {sc:.6f}")

        # b) Adam
        seq, sc = _adam_descent(seed_seq, deadline,
                                 max_iters=30_000, step0=0.12)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} "
                  f"Adam improvement: {sc:.6f}")

        # c) Exponentiated gradient
        seq, sc = _exp_gradient_descent(seed_seq, deadline,
                                         max_iters=30_000,
                                         eta0=0.5, decay=0.9999)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} "
                  f"exp‑grad improvement: {sc:.6f}")

    # --------------------------------------------------------------
    # 6. Final directed hill‑climbing on the current best.
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    iteration = 0
    max_transfer_frac = 0.05          # up to 5 % of the source mass per trial

    while time.time() < deadline:
        iteration += 1
        # recompute gradient every 500 iterations to guide moves
        if iteration % 500 == 1:
            grad, _, _ = _subgrad(a)

        src = int(np.argmax(grad))          # largest positive gradient
        sink = int(np.argmin(grad))         # most negative gradient
        if src == sink or grad[src] <= 0 or grad[sink] >= 0:
            # no clear improvement direction – fall back to a small random move
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            delta = np.random.rand() * min(max_transfer_frac * a[i], a[i])
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
        else:
            delta = min(max_transfer_frac * a[src], a[src])
            a_new = a.copy()
            a_new[src] -= delta
            a_new[sink] += delta

        new_score = _max_convolution_score(a_new)
        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} "
                      f"new best {best_score:.6f}")

        # occasional jitter to escape possible plateaus
        if iteration % 4000 == 0 and np.random.rand() < 0.1:
            jitter = np.random.randn(n) * 1e-4
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    # --------------------------------------------------------------
    print(f"[{time.time():.1f}] Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
