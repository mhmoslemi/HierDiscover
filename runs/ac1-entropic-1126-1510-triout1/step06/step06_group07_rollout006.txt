"""
Propose a low‑score step function.

The routine builds a family of symmetric “Beta‑shaped’’ sequences
(a_i ∝ (i+½)^{p}·(n‑½‑i)^{p}) – the case p=‑0.5 is the continuous
arcsine distribution whose self‑convolution is (almost) flat.
All candidates are normalised so that Σa_i = √(2·n), the scale used by
evaluate_sequence().  The best shape is kept, then a brief stochastic
refinement (mass transfers, local scaling, Gaussian perturbations) is
performed while the time budget permits.  Simple ordering heuristics
are finally tried.  The method is completely self‑contained, uses only
NumPy and the provided evaluate_sequence, and respects the 1000 s
budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _scale_to_sum(seq: np.ndarray, target: float) -> np.ndarray:
    """Rescale a non‑negative vector so that its ℓ¹‑norm equals *target*."""
    total = seq.sum()
    if total == 0.0:
        # avoid division by zero – return a uniform fallback
        return np.full_like(seq, target / seq.size)
    return seq * (target / total)


# ----------------------------------------------------------------------
def _maybe_update(seq: np.ndarray,
                  best: dict,
                  deadline: float) -> None:
    """
    Evaluate *seq* with evaluate_sequence() and store it in *best* if it
    improves the incumbent.  *best* is a dict with keys ``'score'`` and
    ``'seq'``.  No update is performed after the time limit expires.
    """
    if time.time() > deadline:
        return
    score = evaluate_sequence(seq.tolist())
    if score < best["score"]:
        best["score"] = score
        best["seq"] = seq.copy()
        print(f"[improve] score = {score:.6f}")


# ----------------------------------------------------------------------

    """
    Search for a non‑negative step‑function (list of heights) with a low
    evaluate_sequence() score.  The routine respects the given time
    budget (seconds) and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5      # small safety margin

    # --------------------------------------------------------------
    # problem size: use a reference length if available, otherwise 9000
    try:
        n = len(height_sequence_1)               # type: ignore[name-defined]
    except Exception:
        n = 9000

    S = math.sqrt(2.0 * n)                       # target ℓ¹‑norm

    # best incumbent
    best = {"score": float("inf"), "seq": np.zeros(n, dtype=float)}

    # --------------------------------------------------------------
    # 1️⃣  Parametric “Beta’’ shapes (including the arcsine case)
    i = np.arange(n, dtype=float)
    # product (i+½)(n‑½‑i) > 0 for all i∈[0,n‑1]
    prod = (i + 0.5) * (n - 0.5 - i)

    # uniform baseline (p = 0)
    uniform = np.full(n, S / n, dtype=float)
    _maybe_update(uniform, best, deadline)

    # exponents p = α‑1 with α ∈ (0,1] give edge‑heavy shapes.
    # we sweep p ∈ [‑0.9, ‑0.1] (the arcsine case is p = ‑0.5)
    exponents = np.linspace(-0.9, -0.1, 25)

    # mixture proportion λ for (λ·uniform + (1‑λ)·shape)
    lam_grid = [0.0, 0.25, 0.5, 0.75, 1.0]

    for p in exponents:
        if time.time() > deadline:
            break
        shape = np.power(prod, p, dtype=float)   # edge‑heavy for p<0
        shape = _scale_to_sum(shape, S)

        for lam in lam_grid:
            if time.time() > deadline:
                break
            if lam == 0.0:
                cand = shape
            elif lam == 1.0:
                cand = uniform
            else:

                cand = _scale_to_sum(cand, S)     # re‑normalise
            _maybe_update(cand, best, deadline)

    a = best["seq"].copy()

    max_iters = 8000

        cand = a.copy()

        it += 1

    # interleaved big‑small ordering
    vals_desc = desc
    half = len(vals_desc) // 2
    left = vals_desc[:half][::-1]
    right = vals_desc[half:][::-1]
    inter = np.empty_like(vals_desc)
    inter[0::2] = left
    inter[1::2] = right
    _maybe_update(inter, best, deadline)

    print(f"[final] best score = {best['score']:.6f}")
    return best["seq"].tolist()
