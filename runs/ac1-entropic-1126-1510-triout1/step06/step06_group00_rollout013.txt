def _smooth_gradient(a, gamma):
    b = _fft_self_convolve(a)
    # compute weights as softmax of gamma*b
    maxb = b.max()
    w = np.exp(gamma * (b - maxb))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(b) / b.size
    else:
        w = w / w_sum
    # compute conv_rev
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n-1:n-1+n]
    # diagonal correction:
    diag_idx = np.arange(n) * 2
    grad -= a * w[diag_idx]
    return grad, b

Note: For w, length is 2n-1; diag_idx goes up to 2*(n-1) = 2n-2, within range.

In answer we include code block with python.assistantfinal**Solution Overview**

The goal is to minimise  

\[
\text{score}= \frac{2\,n\;\max\big((a*a)[k]\big)}{\big(\sum_i a_i\big)^2},
\qquad a_i\ge 0 .
\]

The score is *scale‑invariant* (as long as no element is clamped to 1000), therefore we keep all candidates *ℓ₁‑normalised* (`∑a_i = 1`).  
For a normalised vector the objective reduces to  

\[
\text{score}= 2 n\;\max\big((a*a)[k]\big) .
\]

The problem is non‑convex because the convolution is quadratic.  
The previous script used a plain sub‑gradient method.  
We improve the search in three ways:

1. **A smooth surrogate for the max‑operator** – we minimise  

   \[
   f_\gamma(a)=\frac{1}{\gamma}\log\Big(\sum_k e^{\gamma\,(a*a)[k]}\Big) ,
   \]

   whose gradient is a *soft* version of the sub‑gradient.  
   When `γ` grows the surrogate approaches the true max, but for moderate `γ`
   the landscape is much smoother and Adam converges faster.

2. **Adam optimisation on the simplex** – after every Adam step the vector is
   projected onto the probability simplex; this guarantees feasibility
   (`a_i≥0`, `∑a_i=1`).  The learning‑rate schedule is the standard Adam one,
   while `γ` is increased linearly from a small value (≈ 1) to a large value
   (≈ 200) so that the optimisation progressively focuses on the true peak.

3. **More diverse starting points and longer refinement** – analytic families
   (arcsine, symmetric beta, power‑law, exponential) and a few random
   Dirichlet draws are generated for several lengths (`n≈4000…8000`).  
   The best 8 seeds are refined with the *smooth‑Adam* routine for up to
   80 000 iterations each (time‑controlled).  Finally a cheap hill‑climbing
   “mass transfer” phase squeezes a few extra digits.

All heavy work is performed with FFT‑based convolutions (`O(n log n)`) and
uses only `numpy` and `scipy` (no file or network access).  
The function `propose_candidate` returns the best sequence found before the
1000 s budget expires.

```python
"""
Propose a low‑score non‑negative step function.

The algorithm proceeds in three stages:

1. **Broad sampling** – generate many candidate sequences from analytic
   families (arcsine, symmetric beta, power‑law, exponential) and a few
   random Dirichlet draws for several lengths.  All candidates are
   ℓ₁‑normalised and evaluated with the cheap score ``2·n·max(a*a)``.

2. **Refinement** – the best few seeds are optimised with
   *smooth‑Adam* descent: a smooth log‑sum‑exp surrogate of the max‑convolution
   is minimised, the smoothing parameter γ is slowly increased, and after each
   step the vector is Euclidean‑projected onto the simplex.

3. **Local hill‑climbing** – a random transfer of a tiny amount of mass
   between two positions is accepted if it improves the exact score.
   Occasional jitter helps to escape plateaus.

The routine respects the 1000 s budget and always returns a list of
non‑negative floats suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy import sparse
from scipy.optimize import linprog  # (unused now but kept for possible extensions)

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of *v* onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # Fast projection (see Wang & Carreira‑Perpiñán 2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact score for a normalised vector (∑a=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, gamma: float):
    """
    Gradient of the smooth surrogate
        fγ(a) = (1/γ)·log Σk exp(γ·(a*a)[k])
    Returns (grad, full convolution vector b).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)                     # length 2n‑1
    # soft‑max weights (avoid overflow)
    maxb = b.max()
    w = np.exp(gamma * (b - maxb))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(b, 1.0 / b.size)
    else:
        w /= w_sum

    # convolution of reversed a with weight vector w
    conv_rev = _fft_convolve(a[::-1], w)          # length 2n‑1
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]

    # diagonal correction (the term a_i·a_i appears only once in (a*a)[2i])
    diag_idx = np.arange(n) * 2
    grad -= a * w[diag_idx]
    return grad, b


# ----------------------------------------------------------------------
def _smooth_adam_descent(a0: np.ndarray,
                         deadline: float,
                         max_iters: int = 80000,
                         step0: float = 0.12,
                         beta1: float = 0.9,
                         beta2: float = 0.999,
                         gamma_start: float = 1.0,
                         gamma_end: float = 200.0):
    """
    Adam optimisation of the smooth surrogate.
    ``gamma`` is linearly increased from ``gamma_start`` to ``gamma_end``.
    The iterate is always projected onto the simplex (∑a=1, a≥0).
    Returns the best sequence (as ndarray) and its exact score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _max_convolution_score(a)

    n = a.shape[0]
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # schedule for γ
        gamma = gamma_start + (gamma_end - gamma_start) * (it - 1) / max_iters

        grad, _ = _smooth_gradient(a, gamma)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat

        # projection onto the simplex
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _max_convolution_score(a)

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}]   smooth‑Adam iter={it}  "
                  f"γ={gamma:.1f}  score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
def _random_dirichlet(n: int) -> np.ndarray:
    """Uniform Dirichlet point (α=1) on the simplex."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}  (symmetric by mirroring)."""
    half = n // 2
    left = (np.arange(half, 0, -1) + 1) ** (-p)
    right = left[::-1] if n % 2 == 0 else np.append(left, [1.0 ** (-p)])
    if n % 2 == 1:
        middle = np.array([1.0 ** (-p)])
        vals = np.concatenate([left, middle, right])
    else:
        vals = np.concatenate([left, right])
    # In case of length mismatch because of rounding, pad / truncate
    if vals.size != n:
        vals = np.resize(vals, n)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:

    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

        n = np.random.randint(500, min(20000, base_n * 3) + 1)

    # --------------------------------------------------------------

    candidates = []   # list of (sequence ndarray, score)

    beta_ps = [0.05, 0.10, 0.15, 0.20, 0.25, 0.30, 0.40,

    power_ps = [0.30, 0.45, 0.60, 0.80, 1.00]

    for n in lengths:

        for gen, name in (

            print(f"[{time.time():.1f}] n={n:4d} {name:12s} score={score:.6f}")

        for p in beta_ps:

                break
            a = _beta_symmetric_distribution(n, p)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:4d} beta(p={p:<4.2f}) score={score:.6f}")

        for p in power_ps:
            if time.time() >= deadline:
                break
            a = _power_law_distribution(n, p)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:4d} power(p={p:<4.2f}) score={score:.6f}")

        # ---- a few random Dirichlet points ------------------------------------
        for _ in range(2):
            if time.time() >= deadline:
                break
            a = _random_dirichlet(n)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:4d} random Dirichlet score={score:.6f}")

    # --------------------------------------------------------------
    # 3. Include the previously discovered best sequence (if present).
    try:

    # --------------------------------------------------------------
    print(f"[{time.time():.1f}] Final best score: {best_score:.6f} "

    return best_seq.tolist()
```
