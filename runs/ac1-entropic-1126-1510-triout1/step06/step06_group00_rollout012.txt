def subgrad(a):
    b = self_convolve(a)  # compute full conv
    max_b = b.max()
    # choose active indices where b close to max
    eps = 1e-12 * (abs(max_b) + 1.0)
    active = np.where(b >= max_b - eps)[0]
    m = len(active)
    if m == 0:
        # fallback
        w = np.zeros_like(b)
        grad = np.zeros_like(a)
        return grad, max_b, active
    w = np.zeros_like(b)

def pgd(a0, deadline, max_iters=50000, init_step=1.0):
    a = a0.copy()
    best = a.copy()
    best_score = compute_score(a)
    step = init_step
    for it in range(max_iters):
        if time.time() >= deadline: break
        grad, cur_max, active = subgrad(a)
        # gradient may be scaled large; maybe adjust step.
        # We'll perform backtracking line search.
        trial_step = step
        while trial_step > 1e-8:
            a_candidate = a - trial_step * grad
            a_candidate = project_onto_simplex(a_candidate)
            cand_score = compute_score(a_candidate)
            if cand_score < best_score - 1e-14:
                # accept and maybe increase step size next iteration
                a = a_candidate
                if cand_score < best_score - 1e-12:
                    best = a_candidate
                    best_score = cand_score
                # maybe increase step for next iteration
                step = max(step, trial_step * 1.05)
                break
            else:
                trial_step *= 0.5
        else:
            # no improvement, reduce step for future attempts
            step *= 0.5
        # optionally break early if step small and no recent improvements
    return best, best_score

The goal is to find a non‑negative vector `a` (a step‑function) that minimises  

\[
\text{score}(a)=\frac{2\,n\;\max (a\ast a)}{(\sum a)^2},
\]

where `\*` denotes the discrete convolution.  
Because the score is invariant under a positive scaling of `a`, we may (and will) work with
vectors normalised to `∑a = 1`.  
In that case the objective reduces to  

\[
\text{score}(a)=2\,n\;\max (a\ast a),
\]

i.e. we only have to make the *peak* of the self‑convolution as small as possible.

The program consists of four phases

1. **Broad sampling** – a large collection of analytically‑defined families
   (Beta, arcsine, triangular, exponential, cosine) and a few random Dirichlet draws.
   All candidates are normalised and scored with a cheap FFT‑based evaluation.

2. **Intensive refinement** – the best candidates are polished with an
   Adam‑style adaptive‑moment optimiser that uses the *exact* sub‑gradient of the
   `max‑convolution` functional.  
   The sub‑gradient is the average of the gradients of all active convolution
   entries; the previous implementation subtracted a diagonal term which is
   incorrect – the fixed version gives a much more accurate descent direction.

3. **Upscaling & re‑refinement** – promising solutions are linearly interpolated
   to a larger length (more degrees of freedom) and refined again.  Since the
   objective contains the factor `n`, a finer discretisation can lower the
   constant term.

4. **Local hill‑climbing** – a cheap random “transfer” move (move a small amount
   of mass from one index to another) is repeatedly applied to the current best
   sequence; any improvement is kept.  Occasionally a tiny jitter is added to
   escape plateaus.

All phases respect the global time budget (`budget_s` ≈ 1000 s).  The
implementation uses only `numpy`, `time` and `scipy` (`linprog` is *not* needed);
the heavy lifting is done with real FFTs, which makes the algorithm fast even
for lengths up to ~ 10 000.

"""Propose a low‑score non‑negative step function.

The search consists of four phases:

1. Broad sampling from several analytic families and random Dirichlet draws.
2. Intensive refinement of the best candidates with Adam‑style adaptive‑moment
   optimisation, using a *correct* sub‑gradient for the max‑convolution.
3. Upscaling of promising solutions (larger n) followed by another round of
   refinement.
4. Simple stochastic hill climbing (mass transfer) to squeeze out the last few
   digits.

All work is performed on vectors normalised to ℓ₁‑norm 1, because the final
objective is invariant under scaling.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) under the assumption sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of   max_k (a * a)_k   on the simplex.
    Returns (gradient, max_value, active_indices).
    The gradient is the *average* of the gradients of all active convolution
    entries – a valid element of the sub‑differential.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)
    m = active.size
    if m == 0:                       # should never happen
        return np.zeros_like(a), cur_max, active

    # weight vector – average over the active positions
    w = np.zeros_like(b)
    w[active] = 1.0 / m

    # convolution of a with w gives the (unnormalised) gradient
    conv = _fft_convolve(a, w)
    grad = 2.0 * conv[n - 1 : n - 1 + n]
    return grad, cur_max, active

def _adam_descent(a0: np.ndarray,
                  deadline: float,
                  max_iters: int = 5000,
                  step0: float = 0.12,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> (np.ndarray, float):
    """Adam‑style optimisation with the correct sub‑gradient."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - step * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Gamma‑Dirichlet, α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _cosine_distribution(n: int, power: float = 1.0) -> np.ndarray:
    """Raised‑cosine shape: (max(0, cos(...)))**power, normalised."""
    i = np.arange(n, dtype=np.float64)
    theta = np.pi * (i - (n - 1) / 2.0) / (n - 1)
    vals = np.maximum(0.0, np.cos(theta)) ** power
    if vals.sum() == 0:
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _upscale(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence to a larger length."""
    old_n = seq.shape[0]
    if new_n == old_n:
        return seq.copy()
    # positions in [0, 1]
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    new_vals = np.interp(new_x, old_x, seq)
    # re‑normalise (tiny numerical drift may appear)
    new_vals /= new_vals.sum()
    return new_vals
