def _adam_optimize_smooth(x0, n, deadline, max_iters=8000, lr0=0.10, t0=1.0, t_mult=1.2, t_max=200.0):
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1,beta2 = 0.9,0.999
    eps = 1e-12
    best_a = None
    best_raw = float('inf')
    t = t0
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # softmax
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()
        # compute convolution
        conv = _fft_self_convolve_full(a)
        # raw objective
        cur_max = conv.max()
        cur_raw = 2.0 * n * cur_max
        if cur_raw < best_raw:
            best_raw = cur_raw
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n}, it={it}, raw={best_raw:.6f}, t={t:.2f}")
        # smoothed gradient
        shift = cur_max
        # compute softmax over conv
        # use scaling to avoid overflow: conv-shift <=0
        exp_term = np.exp(t * (conv - shift))
        w = exp_term / exp_term.sum()
        # gradient wrt a
        grad_a = 2.0 * _fft_convolve(a[::-1], w)[n-1:n-1+n]
        # chain rule through softmax
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        lr = lr0 * (np.sqrt(1 - beta2**it) / (1 - beta1**it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        # increase temperature t
        if it % 500 == 0:
            t = min(t * t_mult, t_max)
        # jitter occasionally
        if it % 2000 == 0:
            x += np.random.randn(*x.shape) * 0.005
    if best_a is None:
        best_a = np.full(n, 1.0/n)
        best_raw = 2.0 * n * _fft_self_convolve_max(best_a)
    return best_a, best_raw

def _optimise_length(n, deadline, prev_seq):

"""
Propose a step‐function that minimizes
    2·n·max(a*a) / (sum a)^2

The implementation follows a continuation method:
  * a is parametrised by a soft‑max of unconstrained logits (guarantees
    non‑negativity and Σa = 1);
  * instead of the non‑smooth max we minimise a smooth “log‑sum‑exp”
    surrogate of the max (temperature t);  t is slowly increased,
    turning the problem gradually into the original one;
  * Adam is used for optimisation of the logits;
  * several diverse initialisations are tried (uniform, Dirichlet,
    interpolated previous best and geometric shapes);
  * after the Adam passes a short projected sub‑gradient refinement
    discards any remaining bias;
  * the whole search respects the supplied time‑budget (default 1000 s).

The code is self‑contained, uses only numpy and the provided
evaluate_sequence function, and returns a plain Python list.
"""

import time

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of max_k (a*a)_k.  Returns (grad, cur_max),
    where grad has the same shape as a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    # active set – tolerance scales with the current maximum
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(conv >= cur_max - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max

def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals

def _smooth_max_and_grad(a: np.ndarray, t: float):
    """
    Computes a smooth approximation of the max( a*a ) together with its
    gradient w.r.t. a.
    Returns (grad, raw_max, smooth_obj).

    smooth_obj = (1/t) * log( Σ_k exp( t * (a*a)_k ) )
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)                 # length 2n-1
    raw_max = conv.max()
    # log‑sum‑exp in a numerically stable way
    shift = raw_max
    exp_term = np.exp(t * (conv - shift))            # all ≤ 1
    sum_exp = exp_term.sum()
    # softmax weights over the convolution entries
    w = exp_term / sum_exp
    # gradient of the smoothed max (chain rule through the convolution)
    grad_a = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    # smooth objective value (equivalent to (1/t)*log‑sum‑exp)
    smooth_obj = shift + (1.0 / t) * np.log(sum_exp)
    return grad_a, raw_max, smooth_obj

def _adam_optimize_smooth(x0: np.ndarray, n: int, deadline: float,
                         max_iters: int = 8000, lr0: float = 0.10,
                         t0: float = 1.0, t_mult: float = 1.2,
                         t_max: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed objective.
    Returns the best vector a (Σa = 1, a ≥ 0) found and its *raw* score
    2·n·max(a*a).
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = None
    best_raw = float("inf")
    t = t0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- soft‑max (stable) -----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()               # Σa = 1, a ≥ 0

        # ----- raw objective -----
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        cur_raw = 2.0 * n * cur_max
        if cur_raw < best_raw:
            best_raw = cur_raw
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it}  raw={best_raw:.6f}  t={t:.2f}")

        # ----- smooth gradient -----
        grad_a, _, _ = _smooth_max_and_grad(a, t)

        # ----- chain rule through the soft‑max -----
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- temperature schedule -----
        if it % 500 == 0:
            t = min(t * t_mult, t_max)

        # occasional jitter to escape plateaus
        if it % 2000 == 0:
            x += np.random.randn(*x.shape) * 0.005

    # fallback
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_raw = 2.0 * n * _fft_self_convolve_max(best_a)
    return best_a, best_raw
