    best_seq = None

        n_min = int(((p+1)**2) / C) + 1

        if time.time() >= deadline:

        # generate
        i = np.arange(1, n+1, dtype=np.float64)

"""Step‑function generator.

The evaluation function is
    score = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

For a monotone power‑law sequence a_i ∝ (i+1)**p one can compute the
asymptotic peak of the convolution in closed form:

    max(conv) ≈ C(p) / n   with   C(p) = ((p+1)!)**2 / (2·p+1)!

Hence the score tends to 2·C(p).  For p = 2 the score is ≈0.60,
for p = 5 it is ≈0.026 and for p = 6 it is ≈0.008 – far below the
previous best of 1.51.  We simply pick a safe exponent (p=5) and a
length large enough that the centre term dominates (n≈5000).  The
resulting vector is normalised to sum = 1, contains only non‑negative
entries and is returned as a plain Python list.

The function respects the time budget, prints a short log and falls back
to a uniform distribution only if something goes wrong.
"""

import time
import math
import numpy as np

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Return a list of non‑negative heights that gives a very low
    `evaluate_sequence` score.
    The construction is a monotone power‑law distribution
    a_i ∝ (i+1)**p, normalised to a simplex.
    """
    np.random.seed(seed)                # deterministic, not really needed
    start = time.time()
    deadline = start + budget_s - 0.5   # a tiny safety margin

    best_seq = None
    best_score = float("inf")
    best_info = None

    # Try a few exponents; higher exponents give smaller theoretical scores
    # but need larger n to keep the centre term dominant.
    for p in (2, 3, 4, 5, 6):
        if time.time() >= deadline:
            break

        # ---- choose a length -------------------------------------------------
        # C(p) = ((p+1)!)**2 / (2p+1)!
        fact = math.factorial(p + 1)
        C = (fact * fact) / math.factorial(2 * p + 1)

        # we need n > (p+1)**2 / C   (so that the centre term dominates)
        n_min = int(((p + 1) ** 2) / C) + 1
        # keep n modest (to stay fast) but safely above the bound
        n = max(500, min(30000, int(n_min * 1.5)))
        # very large n for p=6 would require >12k; cap at 20000 for safety
        if p == 6:
            n = max(n, 15000)
            n = min(n, 20000)

        # ---- build the power‑law vector -------------------------------------
        i = np.arange(1, n + 1, dtype=np.float64)
        seq = i ** p
        seq_sum = seq.sum()
        if seq_sum == 0.0:
            continue
        seq = seq / seq_sum                     # simplex projection

        # ---- evaluate --------------------------------------------------------
        try:
            score = evaluate_sequence(seq.tolist())
        except Exception as exc:                # should never happen
            print(f"[error] p={p}, n={n} → {exc}")
            continue

        print(f"[candidate] p={p:<2d}, n={n:5d}, score={score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            best_info = (p, n, score)

        # if we already beat the target by a large margin we can stop early
        if best_score < 0.5:
            break

    # ---- fallback (unlikely) -------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)
        best_info = (0, n, best_score)
        print("[fallback] uniform distribution used")

    # final log
    p_best, n_best, sc_best = best_info
    print(f"[final] best p={p_best}, n={n_best}, score={sc_best:.6f}")

    return best_seq
