"""
Propose a step function by direct optimisation on the simplex.

The algorithm works as follows:
1)  Initialise a probability vector a (Σa = 1, a_i ≥ 0) either from a warm‑start,
    a uniform vector or several random Dirichlet draws.
2)  Optimise a via Adam on the soft‑max parametrisation of a.
    The objective is 2·n·max(a*a) (the same quantity returned by evaluate_sequence
    after the internal normalisation).  The gradient of max(a*a) is obtained from
    a single active convolution index (the index where the current maximum is
    attained), which yields a cheap O(n) sub‑gradient.
3)  Increase the length of the sequence gradually (2000 → 5000 → 10000 → …).
    Each longer length is warm‑started by linear interpolation of the best
    sequence found so far.
4)  Perform a final projected sub‑gradient descent directly on the simplex
    (no soft‑max) to fine‑tune the best candidate.
5)  Return the best list found; the harness will evaluate it with
    evaluate_sequence().

The whole routine respects a user‑provided time budget (default 1000 s) and
prints occasional progress reports.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _fft_self_convolve_full(a).max()


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


# ----------------------------------------------------------------------
def _max_and_grad(a: np.ndarray):
    """
    Compute max(a*a) and one sub‑gradient of the max.
    Returns (max_val, grad) where grad has the same shape as a.
    The gradient corresponds to the derivative of the selected maximal entry.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    max_val = conv.max()
    k_max = int(conv.argmax())                     # index of the active entry

    grad = np.zeros_like(a)

    # Fill gradient according to the formula
    if k_max < n:
        # indices i = 0 .. k_max   (j = k_max - i)
        grad[: k_max + 1] = 2.0 * a[k_max::-1]
    else:
        start = k_max - n + 1                      # first i that yields a valid j
        # indices i = start .. n‑1   (j = k_max - i runs from n‑1 down to start)
        grad[start:] = 2.0 * a[n - 1 : start - 1 : -1]
    return max_val, grad

def _adam_optimize_from_logits(
    x0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 4000,
    lr0: float = 0.15,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the logit parametrisation.
    Returns (best_a, best_raw_score) where best_raw_score = 2·n·max(a*a).
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    best_a = None
    best_score = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1

        # ---------- soft‑max (stable) ----------
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()          # Σa = 1, a ≥ 0

        # ---------- objective ----------
        cur_max, _ = _max_and_grad(a)
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it}  raw={best_score:.6f}")

        # ---------- sub‑gradient of max ----------
        _, grad_a = _max_and_grad(a)

        # ---------- chain‑rule through the soft‑max ----------
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % 800 == 0:
            x += np.random.randn(*x.shape) * 0.01

    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_max(best_a)

    return best_a, best_score

def _optimise_length(
    n: int,
    deadline: float,
    prev_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length n.
    Several initialisations are tried (warm‑start, uniform, random Dirichlet).
    The best candidate is returned together with its raw score 2·n·max(a*a).
    """
    # ---- build list of start logits ---------------------------------
    init_logits = []

    # warm‑start from previous best (interpolated)
    if prev_seq is not None:
        cand = _interpolate_to_length(prev_seq, n)
        eps = 1e-12
        init_logits.append(np.log(cand + eps))

    # uniform start
    init_logits.append(np.zeros(n, dtype=np.float64))

    # a few random Dirichlet draws
    for _ in range(5):
        alpha = np.random.rand() * 4.0 + 0.1
        draw = np.random.gamma(alpha, 1.0, size=n)
        draw /= draw.sum()
        init_logits.append(np.log(draw + 1e-12))

    # ---- run Adam from each start -----------------------------------
    best_seq = None
    best_score = float("inf")

    max_iters = 5000 if n <= 10000 else 3000

    for idx, x0 in enumerate(init_logits):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}]   length {n}, start {idx + 1}/{len(init_logits)}")
        seq, score = _adam_optimize_from_logits(
            x0, n, deadline, max_iters=max_iters, lr0=0.15
        )
        if score < best_score:
            best_score = score
            best_seq = seq.copy()

    # safety fall‑back

        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_max(best_seq)

    return best_seq, best_score

    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin before the platform enforces the kill‑signal
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Try to reuse a previously discovered construction if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        s = prev.sum()
        if s > 0.0:
            prev = prev / s
        else:
            prev = None
    except Exception:
        prev = None

    best_seq = None
    best_score = float("inf")

    # baseline evaluation
    if prev is not None:
        raw = 2.0 * len(prev) * _fft_self_convolve_max(prev)
        print(f"Baseline (from height_sequence_1) raw score = {raw:.6f}")
        if raw < best_score:
            best_score = raw
            best_seq = prev.copy()

    # ------------------------------------------------------------------
    # Length schedule – start modest, grow as time permits
    lengths = [2000, 5000, 10000, 20000, 40000]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length {n} ===")
        seq, raw = _optimise_length(n, deadline, prev_seq=best_seq)
        print(f"Finished length {n}: raw score = {raw:.6f}")
        if raw < best_score:
            best_score = raw
            best_seq = seq.copy()
        # warm‑start for the next size
        prev = best_seq

    # ------------------------------------------------------------------
    # Final fine‑tuning directly on the simplex (projected sub‑gradient)
    if best_seq is not None:
        a = best_seq.copy()
        n = a.shape[0]
        it = 0
        while time.time() < deadline:
            it += 1
            cur_max, grad = _max_and_grad(a)
            step = 0.07 / math.sqrt(it)
            a = a - step * grad
            a = _project_onto_simplex(a)          # keep Σa = 1, a ≥ 0
            raw = 2.0 * n * cur_max
            if raw < best_score - 1e-12:
                best_score = raw
                best_seq = a.copy()
                print(f"[{time.time():.1f}] post‑refine it={it} raw={best_score:.6f}")
            # occasional tiny jitter to avoid stagnation
            if it % 800 == 0:
                a += np.random.randn(n) * 5e-7
                a = _project_onto_simplex(a)

    # ------------------------------------------------------------------
    final_list = best_seq.tolist()
    final_score = evaluate_sequence(final_list)
    print(f"\nFinal evaluated score: {final_score:.6f} (length={len(final_list)})")
    return final_list
