Propose a low‑score non‑negative step function.

Algorithm
---------
1. Choose a moderate length `n` (≈2‑3 k).  Initialise the distribution with a
   classic arcsine shape (U‑shaped, good known baseline).  If a previous
   best (`height_sequence_1`) exists it is mixed in.

2. Iterative *linear‑programming improvement*:
   * For the current vector `a` (∑a = 1) compute `M = max(conv(a,a))`.
   * Build the sparse convolution matrix `A` that satisfies  `A @ g = conv(a,g)`.
   * Solve the LP  
        maximise   1ᵀ g
        subject to A g ≤ M·1,   g ≥ 0

        a ← (1‑t)·a + t·g_norm   (t decays geometrically).

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Score under the normalisation ∑a = 1 (i.e. 2·n·max(conv))."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(conv(a,a)) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # correct the diagonal (the self‑terms counted only once)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active

def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 200000,
                          eta0: float = 0.6, decay: float = 0.9999):
    """Mirror descent with KL (exponentiated‑gradient)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:

        grad, _, _ = _subgrad(a)
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = a / a.sum()
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def _make_conv_matrix(f: np.ndarray):

    data = [np.full(n, f[i], dtype=np.float64) for i in range(n)]
    A = sp.diags(data, offsets, shape=(rows, cols), format='csc')
    return A

            best_score = score_new
            best_a = a_new.copy()
            print(f"[{time.time():.1f}] LP step {it}, new best {best_score:.6f}")

        # accept the move if it improves the bound
        if score_new < _max_convolution_score(a) - 1e-14:
            a = a_new
        else:
            # still replace a with a_new (it often helps later steps)
            a = a_new

    return best_a, best_score


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a plain list of floats (the heights).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose a problem size.  Use the length of a previously discovered
    # solution if it exists, otherwise fall back to a reasonable default.
    default_n = 2500
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.ndim == 1 and prev.size >= 500:
            n = int(prev.size * 1.0)          # keep the same length
        else:
            n = default_n
    except Exception:
        n = default_n

    print(f"[{time.time():.1f}] Using length n={n}")

    # ------------------------------------------------------------------
    # Initialise with a classic arcsine shape (good baseline).  If a previous
    # best is present we blend it in.
    a0 = _arcsine_distribution(n)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.ndim == 1 and prev.size == n:
            a0 = 0.6 * a0 + 0.4 * (prev / prev.sum())
    except Exception:
        pass
    a0 = np.maximum(a0, 1e-12)
    a0 = a0 / a0.sum()

    # ------------------------------------------------------------------
    # Phase 1 – LP based improvement
    if sp is not None and opt is not None:
        a_lp, score_lp = _lp_improve(a0, deadline, max_iters=60)
    else:
        print("SciPy not available – skipping LP phase.")
        a_lp, score_lp = a0, _max_convolution_score(a0)

    # ------------------------------------------------------------------
    # Phase 2 – fine‑tuning with exponentiated‑gradient mirror descent
    a_ref, score_ref = _exp_gradient_descent(a_lp, deadline,
                                             max_iters=80000,
                                             eta0=0.5, decay=0.9999)

    # keep the best of the two refinements
    if score_ref < score_lp:
        best_a, best_score = a_ref, score_ref
    else:
        best_a, best_score = a_lp, score_lp

    # ------------------------------------------------------------------
    # Final safety check: make sure there are no exact zeros (LP could create
    # them).  Tiny positive values keep the LP well‑posed if the routine is
    # called again.
    eps = 1e-12
    best_a = np.maximum(best_a, eps)
    best_a /= best_a.sum()

    final_score = _max_convolution_score(best_a)
    print(f"[{time.time():.1f}] Final best score (norm‑1) = {final_score:.6f}"
          f"   (raw evaluate = {evaluate_sequence(best_a.tolist()):.6f})")
    print(f"Length = {len(best_a)}")
    return best_a.tolist()
