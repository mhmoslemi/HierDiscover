"""
Propose a step‑function with a lower evaluation score.

Algorithm
---------
* **Idea** – The evaluation is scale‑invariant: for a vector `a` with
  `sum(a)=1` the score equals `2·n·max(conv(a,a))`.  Hence we only need to
  minimise the maximal entry of the self‑convolution under the simplex
  constraint `a≥0 , sum(a)=1`.

* **Iterated linear programme** – For a current candidate `f` we
  compute `M = max(conv(f,f))`.  The constraint `conv(f,g) ≤ M` is *linear*
  in `g`.  Solving the LP  

        maximise   Σ g_i
        subject to conv(f,g) ≤ M   (active shifts only)
                   g_i ≥ 0

  yields a feasible direction `g` that tries to increase the total mass
  while respecting the current maximal convolution value.  By mixing
  `f` and the normalised `g` we obtain a new point with a (weakly) smaller
  `max(conv)`.

* **Active‑set LP** – The LP only needs the constraints that are
  *active* for the current `f` (those shifts where `conv(f,f)` is within a
  tiny tolerance of `M`).  After solving the LP we check the full
  convolution `conv(f,g)`; any newly violated shift is added to the active
  set and the LP is resolved (a small cutting‑plane loop).

* **Monotone mixing** – The new point is taken as a convex combination
  `f←(1‑t)·f + t·ĝ` where `ĝ` is the normalised LP solution.
  The step size `t` is reduced (by a factor 0.5) until the maximal
  convolution does not increase, guaranteeing a non‑increasing score.

* **Search over lengths & initialisations** – Several problem sizes
  (≈3000–8000) are tried.  For each size a handful of analytic
  candidates (Beta‑densities, exponentials, previous best, random) are
  generated, the best few are taken as starting points and refined by the
  LP iteration described above.

* **Time budget** – The whole routine respects the 1000 s limit; it
  returns the best sequence found (as a plain list of floats).  All numbers
  are non‑negative and summed to one, which is exactly the normalisation
  required by `evaluate_sequence`.

The implementation uses only NumPy, SciPy’s `linprog` (HiGHS), and the
provided `evaluate_sequence` for the final verification.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals

def _exponential_distribution(n: int, lam: float = 1.0) -> np.ndarray:
    """Geometric‑like decreasing shape: a_i ∝ exp(-lam·i/(n-1))."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * i / max(1, n - 1))
    vals /= vals.sum()
    return vals

def _cheap_score(a: np.ndarray) -> float:
    """Score for a normalised vector: 2·n·max(conv)."""
    return 2.0 * a.shape[0] * _fft_self_convolve(a).max()

def _lp_feasible_optimize(f: np.ndarray,
                          M: float,
                          active_idxs: np.ndarray,
                          n: int,
                          max_outer: int = 8) -> np.ndarray | None:
    """
    Solve   max Σ g_i   s.t.   conv(f,g)_k ≤ M   for all k∈active_idxs,
    g ≥ 0.  If other shifts violate the bound we add them (cutting‑plane)
    up to ``max_outer`` rounds.
    """
    active = set(int(k) for k in active_idxs)

    for outer in range(max_outer):
        # build dense constraint matrix (few rows → cheap)
        rows = len(active)
        A = np.zeros((rows, n), dtype=np.float64)
        active_list = sorted(active)
        for r, k in enumerate(active_list):
            # indices j where 0 ≤ k‑j < n
            j_start = max(0, k - (n - 1))
            j_end = min(k, n - 1)
            j_range = np.arange(j_start, j_end + 1, dtype=int)
            i_range = k - j_range
            A[r, j_range] = f[i_range]

        # linear programme (maximise sum g_i)
        c = -np.ones(n, dtype=np.float64)          # Linprog minimises
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(rows, M),
                      bounds=bounds, method="highs")
        if not res.success:
            return None
        g = np.maximum(res.x, 0.0)

        # check all shifts for violations
        conv_fg = _fft_convolve(f, g)
        tol = 1e-9 * (abs(M) + 1.0)
        viol = np.where(conv_fg > M + tol)[0]
        if viol.size == 0:
            return g

        # add newly violated indices to the active set and repeat
        for v in viol:
            active.add(int(v))

    # after max_outer rounds return the last feasible (or near‑feasible) g
    return g

def _lp_refine(initial_a: np.ndarray,
               deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform the monotone LP‑based refinement starting from ``initial_a``.
    Returns the best vector found and its cheap score.
    """
    a = _project_onto_simplex(initial_a, s=1.0)
    best_a = a.copy()
    best_score = _cheap_score(a)

    # mixing parameters
    t_start = 0.5
    t_min = 0.01

    while time.time() < deadline:
        n = a.shape[0]

        # current self‑convolution and bound M
        conv_self = _fft_self_convolve(a)
        M = conv_self.max()

        # active constraints – near‑max shifts
        eps = 1e-9 * (abs(M) + 1.0)
        active = np.flatnonzero(conv_self >= M - eps)
        if active.size == 0:
            active = np.array([np.argmax(conv_self)], dtype=int)

        # LP to obtain a feasible direction g
        g = _lp_feasible_optimize(a, M, active, n)
        if g is None:
            break

        # normalise g (sum‑to‑one)
        g = _project_onto_simplex(g, s=1.0)

        # try to mix with decreasing step size until monotone
        t = t_start
        accepted = False
        while t >= t_min and not accepted and time.time() < deadline:
            a_candidate = (1.0 - t) * a + t * g
            a_candidate = _project_onto_simplex(a_candidate, s=1.0)
            conv_cand = _fft_self_convolve(a_candidate)
            M_cand = conv_cand.max()
            if M_cand <= M + 1e-12:           # non‑increase guarantee
                accepted = True
                a = a_candidate
                # keep the best ever seen (might be slightly better)
                score = 2.0 * n * M_cand
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                # optionally increase step size a little for next round
                t_start = min(0.9, t * 1.1)
            else:
                t *= 0.5

        if not accepted:
            # cannot make progress – stop the refinement
            break

        # early exit if we already beat the target
        if best_score <= 1.503:
            break

    return best_a, best_score

def propose_candidate(seed: int = 1234, budget_s: int = 1000) -> list[float]:
    """

    The routine respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin for final call

    # ------------------------------------------------------------------
    # base length (fallback to a reasonable size if the prior result is absent)
    try:
        base_n = len(height_sequence_1)      # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # lengths to explore – a handful around the base size
    length_factors = (0.75, 1.0, 1.25, 1.5, 2.0)
    lengths = sorted(
        max(3000, int(base_n * f)) for f in length_factors
    )
    # ensure we have at least one length
    if not lengths:
        lengths = [base_n]

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Exploring length n={n}")

        # --------------------------------------------------------------
        # build a pool of analytic / random seeds
        pool = []

        alphas = np.logspace(-0.7, 0.7, 7)   # ≈0.2 … 5
        betas = np.logspace(-0.7, 0.7, 7)
        for a in alphas:
            for b in betas:
                pool.append(_beta_distribution(n, a, b))

        pool.append(_beta_distribution(n, 0.2, 0.8))
        pool.append(_beta_distribution(n, 0.8, 0.2))
        pool.append(_exponential_distribution(n, lam=0.5))
        pool.append(_exponential_distribution(n, lam=1.5))
        pool.append(_initial_vector("prev", n))
        pool.append(_initial_vector("uniform", n))
        for _ in range(3):
            pool.append(_initial_vector("random", n))

        # cheap scoring and keep a handful of promising starters
        scored = [( _cheap_score(a), a ) for a in pool]
        scored.sort(key=lambda x: x[0])
        starters = [a.copy() for _, a in scored[:5]]
        print(f"  top cheap scores (2·n·max(conv)): {[s for s, _ in scored[:5]]}")

        # --------------------------------------------------------------
        for sid, a0 in enumerate(starters):
            if time.time() >= deadline:
                break

            # give each starter a modest slice of the remaining time
            remaining = deadline - time.time()
            sub_deadline = time.time() + min(remaining * 0.25, 150.0)

            a_refined, score_refined = _lp_refine(a0, sub_deadline)

            # final verification with the true black‑box evaluator
            try:
                real_score = evaluate_sequence(a_refined.tolist())
            except Exception:
                real_score = float("inf")

            if real_score < overall_best_score:
                overall_best_score = real_score
                overall_best_seq = a_refined.copy()
                print(
                    f"*** New global best: n={n} starter={sid} "
                    f"score={overall_best_score:.6f}"
                )
                if overall_best_score <= 1.503:
                    print("Target reached – exiting early.")
                    return overall_best_seq.tolist()

    # ------------------------------------------------------------------
    # fallback if nothing succeeded (should never happen)
    if overall_best_seq is None:
        n_fallback = 4000
        overall_best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())
        print("Fallback used.")

    print(
        f"Final best score: {overall_best_score:.6f} "
        f"(length = {len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
