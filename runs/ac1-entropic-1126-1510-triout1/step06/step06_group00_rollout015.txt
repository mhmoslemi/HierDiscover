    # Build A_ub: sp.diags(...).

"""
Propose a step function using a linear‑programming based reduction of the
maximum self‑convolution (the quantity 2·n·max(a*a) for Σa=1),
followed by short projected‑gradient refinements and a final hill‑climbing
phase.  The LP is the classic “LP‑rebalancing” step described in the
statement: for the current f find a g with maximal ℓ1‑norm while keeping
(f∗g)(·)≤M, rescale g to Σg=1 and mix it with f.  Repeating this with a
few mixing steps provably never increases M, and in practice drives the
score below the previous best 1.510112.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a∗a (length 2·n−1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) assuming Σa=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP:
        max Σ g_j
        s.t. (f ∗ g)_k ≤ M   for all k,
             g_j ≥ 0
    The matrix of the convolution constraints is built as a Toeplitz matrix
    using scipy.sparse.diags.
    Returns the optimal g (may be empty if the solver fails).
    """
    n = f.shape[0]
    # Build sparse matrix A (shape (2n‑1, n)) where row k, column j
    # contains f_{k‑j} whenever 0≤k‑j<n, i.e. f_i on diagonal i.
    diags = [np.full(n, f_i, dtype=np.float64) for f_i in f]
    offsets = np.arange(n, dtype=int)
    A = sp.diags(diags, offsets, shape=(2 * n - 1, n), format='csr')
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # maximize Σ g  → minimize -Σ g
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 20000,
                          eta0: float = 0.5, decay: float = 0.9999):
    """Exponentiated‑gradient (mirror descent w.r.t. KL)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # sub‑gradient of max‑convolution (binary active set)
        n = a.shape[0]
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps)

        w = np.zeros_like(b)
        w[active] = 1.0
        conv_rev = np.fft.irfft(
            np.fft.rfft(a[::-1]) * np.fft.rfft(w, n=1 << ((2 * n - 1) - 1).bit_length()),
            n=1 << ((2 * n - 1) - 1).bit_length()
        )[:2 * n - 1]
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]   # simple sub‑gradient

        # KL‑mirrored exponential update
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / n)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score, best_a = cur_score, a.copy()
    return best_a, best_score

    best_a = a.copy()
    best_score = _max_convolution_score(best_a)
    print(f"[{time.time():.1f}] init n={n}, score={best_score:.6f}")

    # ----- LP‑based reduction loop -----------------------------------------
    max_lp_iters = 30
    lp_iter = 0
    while time.time() < deadline and lp_iter < max_lp_iters:
        # current bound
        b = _fft_self_convolve(best_a)
        M = b.max()

        g0 = _lp_step(best_a, M)
        if g0 is None or g0.sum() == 0:
            print(f"[{time.time():.1f}] LP step failed – exiting LP loop.")
            break

        g = g0 / g0.sum()          # renormalise to Σg=1

        # try a decreasing mixing coefficient until improvement
        t = 0.3
        improved = False
        while t > 1e-4:
            cand = (1.0 - t) * best_a + t * g
            cand = _project_onto_simplex(cand, s=1.0)
            cand_score = _max_convolution_score(cand)
            if cand_score < best_score - 1e-12:
                best_a, best_score = cand, cand_score
                improved = True
                print(f"[{time.time():.1f}] LP iter {lp_iter+1} – "
                      f"t={t:.4f} – new best {best_score:.6f}")
                break
            t *= 0.5

        if not improved:
            print(f"[{time.time():.1f}] No improvement with LP iter {lp_iter+1}; stop.")
            break

        # occasional BFGS‑style smooth refinement (few exp‑grad steps)
        if lp_iter % 4 == 0:
            seq, sc = _exp_gradient_descent(best_a, deadline,
                                            max_iters=6000,
                                            eta0=0.6, decay=0.9999)
            if sc < best_score - 1e-12:
                best_a, best_score = seq, sc
                print(f"[{time.time():.1f}] post‑LP exp‑grad improvement {best_score:.6f}")

        lp_iter += 1

    # ----- final hill‑climbing phase ----------------------------------------
    a = best_a.copy()
    cur_score = best_score
    max_transfer = 0.06
    it = 0
    while time.time() < deadline:
        it += 1
        transfer = max_transfer * (0.5 ** (it / 15000.0))

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = rng.random() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_a = a_new.copy()
                best_score = new_score
                print(f"[{time.time():.1f}] hill‑climb iter={it} new best {best_score:.6f}")

        if it % 4000 == 0 and rng.random() < 0.12:
            jitter = rng.normal(scale=1e-4, size=n)
            a = best_a + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    print(f"Final best score: {best_score:.6f} (length={len(best_a)})")
    return best_a.tolist()
