    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    best_a = None
    best_score = float('inf')
    for it in range(1, max_iters+1):
        if time.time() >= deadline: break
        # softmax
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()
        # compute convolution
        conv = _fft_self_convolve_full(a)
        # tau schedule, linear increase
        tau = tau0 + (tau_max - tau0) * (it-1) / max_iters
        # compute softmax weights
        max_conv = conv.max()
        exps = np.exp(tau * (conv - max_conv))
        w = exps / exps.sum()
        # gradient of smooth max w.r.t a
        grad_a = 2.0 * _fft_convolve(a[::-1], w)[n-1:n-1+n]
        # compute raw score for reporting
        cur_max = conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # chain rule to logits
        dot = np.dot(a, grad_a)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x**2)

"""
Propose a step function with a low evaluation score.

Core ideas
----------
* Work directly on the probability simplex (non‑negative numbers that sum to 1).
* Use a smooth approximation of the max‑convolution (log‑sum‑exp) to obtain
  a dense gradient that guides the optimisation more reliably than a pure
  active‑set sub‑gradient.
* Optimise a soft‑max parametrisation with Adam; the temperature of the
  log‑sum‑exp is increased during the run so that the objective gradually
  turns into the true max.
* Warm‑start longer sequences by linear interpolation of the best‑known
  shape from a smaller length.
* After the Adam phase a short projected sub‑gradient descent (as in the
  reference solution) cleans up the remaining plateau.
* Several random initialisations (uniform, Dirichlet with different
  concentrations and very spiky draws) are tried; the best of all is kept.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of max_k (a*a)_k.
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    # active set – tolerance scales with the current maximum
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(conv >= cur_max - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals

def _adam_optimize_smoothed(x0: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 6000,
                            tau0: float = 0.5,
                            tau_max: float = 100.0,
                            lr0: float = 0.12) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the soft‑max parametrisation, using a
    log‑sum‑exp (smooth‑max) surrogate for the convolution supremum.
    Returns the best sequence (a) encountered and its *raw* score
    2·n·max(a*a).
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    best_a = None
    best_score = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        # ----- soft‑max (stable) -----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()          # Σa = 1, a ≥ 0

        # ----- convolution and smooth max -----
        conv = _fft_self_convolve_full(a)        # length 2n‑1
        cur_max = conv.max()
        raw_score = 2.0 * n * cur_max
        if raw_score < best_score:
            best_score = raw_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it}  raw={best_score:.6f}")

        # temperature schedule (linear increase)
        tau = tau0 + (tau_max - tau0) * (it - 1) / max_iters

        # smooth‑max weights (stable softmax)
        max_conv = cur_max
        exps = np.exp(tau * (conv - max_conv))
        w = exps / exps.sum()

        # gradient of the smooth max w.r.t. a
        grad_a = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # ----- chain rule through the soft‑max -----
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to help escape plateaus
        if it % 1000 == 0:
            x += np.random.randn(*x.shape) * 0.01

    # fallback (should never happen)
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_max(best_a)

    return best_a, best_score


# ----------------------------------------------------------------------
def _refine_by_subgradient(a: np.ndarray,
                           deadline: float,
                           start_iter: int = 0) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent for the exact max‑convolution.
    Returns refined (a, raw_score).
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = 2.0 * n * _fft_self_convolve_max(a)
    it = start_iter
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.07 / math.sqrt(it)          # diminishing step
        a = a - step * grad
        a = _project_onto_simplex(a)
        # tiny random perturbations every few hundred iterations
        if it % 800 == 0 and np.random.rand() < 0.06:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a)
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] sub‑grad it={it}  raw={best_score:.6f}")
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Entry point required by the challenge harness.
    Returns a list of non‑negative heights that (hopefully) gives a low
    ``evaluate_sequence`` score.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # small safety margin

    # ------------------------------------------------------------------
    # try to reuse a previously discovered construction if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        s = prev.sum()
        if s > 0.0:
            prev = prev / s
        else:
            prev = None
    except Exception:
        prev = None

    # ------------------------------------------------------------------
    # lengths to explore – the larger the better, but we increase them
    # gradually so that warm‑starts have time to propagate.
    lengths = [4000, 8000, 12000, 20000, 30000]

    best_seq = None
    best_score = float("inf")
    best_n = None

    # baseline evaluation (if a previous sequence is available)
    if prev is not None:
        raw = 2.0 * len(prev) * _fft_self_convolve_max(prev)
        print(f"Baseline (length={len(prev)}) raw score = {raw:.6f}")
        if raw < best_score:
            best_score = raw
            best_seq = prev.copy()
            best_n = len(prev)

    # ------------------------------------------------------------------
    # optimise each length in increasing order
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length {n} ===")
        # ----- build list of start logits -----
        init_logits = []

        # warm‑start from current best (interpolated)
        if best_seq is not None:
            cand = _interpolate_to_length(best_seq, n)
            init_logits.append(np.log(cand + 1e-12))

        # uniform start
        init_logits.append(np.zeros(n, dtype=np.float64))

        # several Dirichlet draws with moderate concentration
        for _ in range(3):
            alpha = np.random.rand() * 4.0 + 0.1   # α ∈ (0.1,4.1)
            d = np.random.gamma(alpha, 1.0, size=n)
            d /= d.sum()
            init_logits.append(np.log(d + 1e-12))

        # a few very spiky Dirichlet draws (tiny concentration)
        for _ in range(2):
            alpha = np.random.rand() * 0.09 + 0.001   # α ∈ (0.001,0.091)
            d = np.random.gamma(alpha, 1.0, size=n)
            d /= d.sum()
            init_logits.append(np.log(d + 1e-12))

        # ----- run optimisation from each start -----
        local_best_seq = None
        local_best_score = float("inf")

        for idx, x0 in enumerate(init_logits):
            if time.time() >= deadline:
                break
            print(f"[{time.time():.1f}] start {idx+1}/{len(init_logits)} for n={n}")

            # 1) Adam with smooth‑max surrogate
            seq, raw_score = _adam_optimize_smoothed(
                x0, n, deadline, max_iters=6000,
                tau0=0.5, tau_max=100.0, lr0=0.12
            )

            # 2) short refinement with the exact sub‑gradient method
            seq, raw_score = _refine_by_subgradient(seq, deadline)

            if raw_score < local_best_score:
                local_best_score = raw_score
                local_best_seq = seq.copy()
                print(f"   *** new local best for n={n}: raw={local_best_score:.6f}")

        if local_best_seq is not None and local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_seq.copy()
            best_n = n
            print(f">>> Updated global best: raw={best_score:.6f} (length={best_n})")

    # ------------------------------------------------------------------
    # final long‑run sub‑gradient refinement on the current global best
    if best_seq is not None:
        a = best_seq.copy()
        a, final_raw = _refine_by_subgradient(a, deadline)
        best_seq = a.copy()
        best_score = final_raw
    else:
        # fallback – uniform distribution of moderate size
        n = 5000
        best_seq = [1.0 / n] * n

    # ------------------------------------------------------------------
    # evaluate through the supplied black‑box and return
    final_list = best_seq.tolist()
    final_score = evaluate_sequence(final_list)   # provided by the harness
    print(f"\nFinal evaluated score: {final_score:.6f} (length={len(final_list)})")
    return final_list
