def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT (full length)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate of max(conv).
    a : probability vector (sum = 1)
    conv : self‑convolution of a
    alpha : temperature multiplier (larger → sharper)
    """
    n = a.shape[0]
    cur_max = conv.max()
    exp_term = np.exp(alpha * (conv - cur_max))
    w = exp_term / exp_term.sum()
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the true max: average over all active indices.
    """
    n = a.shape[0]
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even shifts
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad


# ----------------------------------------------------------------------
def _adam_step(z: np.ndarray,
               g: np.ndarray,
               m: np.ndarray,
               v: np.ndarray,
               t: int,
               lr: float = 0.5,
               beta1: float = 0.9,
               beta2: float = 0.999,
               eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Adam update on the unconstrained variable z."""
    m = beta1 * m + (1.0 - beta1) * g
    v = beta2 * v + (1.0 - beta2) * (g * g)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z, m, v


# ----------------------------------------------------------------------
def _cheap_score(a: np.ndarray) -> float:
    """Score = 2·n·max(self‑convolution); a must sum to 1."""
    n = a.shape[0]
    return 2.0 * n * _fft_self_convolve(a).max()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 1234, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 0. base length (use previous result if available)
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # 1. lengths to explore (a few multiples of base_n)
    length_factors = (0.8, 1.0, 1.2, 1.5, 2.0)
    lengths = sorted(
        max(3000, int(base_n * f))
        for f in length_factors
        if int(base_n * f) >= 3000
    )
    if not lengths:
        lengths = [max(3000, base_n)]

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")

    # --------------------------------------------------------------
    # Candidate generation for every length
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Exploring length n={n}")

        # ---- build the pool -------------------------------------------------
        pool: list[np.ndarray] = []

        # uniform
        pool.append(np.full(n, 1.0 / n, dtype=np.float64))

        # linear ramp (i+1) and its reverse
        ramp = np.arange(1, n + 1, dtype=np.float64)
        ramp = ramp / ramp.sum()
        pool.append(ramp)               # forward ramp
        pool.append(ramp[::-1])         # reverse ramp

        # quadratic ramp
        quad = (np.arange(1, n + 1, dtype=np.float64) ** 2)
        quad = quad / quad.sum()
        pool.append(quad)

        # Beta grid (including very extreme parameters)
        alphas = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
        betas = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
        for a in alphas:
            for b in betas:
                pool.append(_beta_distribution(n, a, b))

        # a few Dirichlet draws with low concentration (moderately sparse)
        for _ in range(5):
            cand = np.random.dirichlet(np.full(n, 0.5))
            pool.append(cand)

        # previous solution, interpolated if needed
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.size != n:
                x_old = np.linspace(0.0, 1.0, prev.size)
                x_new = np.linspace(0.0, 1.0, n)
                prev = np.interp(x_new, x_old, prev)
                prev = _project_onto_simplex(prev, s=1.0)
            else:
                prev = prev.copy()
            pool.append(prev)
        except Exception:
            pass

        # ---- cheap evaluation ------------------------------------------------
        for a in pool:
            a = np.maximum(a, 0.0)
            total = a.sum()
            if total == 0.0:
                continue
            a = a / total
            score = _cheap_score(a)
            if score < overall_best_score:
                overall_best_score = score
                overall_best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] New global best (n={n}) → "
                    f"score={overall_best_score:.6f}"
                )
                if overall_best_score <= 1.503:
                    print("Target reached during cheap scan.")
                    return overall_best_seq.tolist()
        # --------------------------------------------------------------
        # time check after each length
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # If we still have time, refine the current best with Adam on the simplex
    if overall_best_seq is not None and time.time() < deadline - 0.2:
        n = overall_best_seq.shape[0]
        a = np.maximum(overall_best_seq, 1e-15)
        z = np.log(a)                     # unconstrained log‑softmax variables
        m = np.zeros_like(z)
        v = np.zeros_like(z)

        # hyper‑parameters
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        lr0 = 0.6

        temperature = 1.0
        temp_factor = 1.001
        temp_max = 8000.0

        best_local_a = a.copy()
        best_local_score = _cheap_score(a)
        it = 0

        while time.time() < deadline - 0.1:
            it += 1
            a = np.exp(z)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            cur_score = 2.0 * n * conv.max()

            # keep the best seen during refinement
            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local_a = a.copy()
                print(
                    f"[{time.time():.1f}] refinement it={it} score={cur_score:.6f}"
                )
                if cur_score < overall_best_score:
                    overall_best_score = cur_score
                    overall_best_seq = a.copy()
                    if overall_best_score <= 1.503:
                        print("Target reached during refinement.")
                        break

            # gradient (softmax → exact)
            if temperature < temp_max:
                grad_a = _softmax_gradient(a, conv, temperature)
                temperature *= temp_factor
            else:
                grad_a = _exact_gradient(a, conv)

            # convert ∂f/∂a to ∂f/∂z (softmax Jacobian)
            dot = np.dot(a, grad_a)
            grad_z = a * (grad_a - dot)

            # Adam update (iteration index is 1‑based)
            z, m, v = _adam_step(z, grad_z, m, v, it,
                                 lr=lr0, beta1=beta1, beta2=beta2, eps=eps)

            # occasional learning‑rate decay to stabilise late iterations
            if it % 2000 == 0:
                lr0 *= 0.5

        # after refinement store the best found locally
        if best_local_score < overall_best_score:
            overall_best_score = best_local_score
            overall_best_seq = best_local_a.copy()

    # ------------------------------------------------------------------
    # Fallback in the unlikely case that nothing was discovered
    if overall_best_seq is None:
        n_fallback = 4000
        overall_best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        overall_best_score = _cheap_score(overall_best_seq)
        print("Fallback uniform sequence used.")

    print(
        f"Final best cheap score: {overall_best_score:.6f} "
        f"(length = {len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
