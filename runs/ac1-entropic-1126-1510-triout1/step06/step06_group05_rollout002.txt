"""
Propose a step‑function (non‑negative vector) that minimises

    score = 2 * n * max(convolution(a, a)) / (sum(a)^2)

where `evaluate_sequence` will later be called on the returned list.
The algorithm is a streamlined, time‑aware optimiser that

1.  Generates several candidate distributions (arcsine, Beta,
    power‑law, Dirichlet, mixtures) for a few different lengths and
    keeps the best according to the cheap proxy ``2·n·max(conv)``.

2.  Refines the chosen candidate with a projected Adam optimiser on a
    smooth‑max surrogate, followed by a short exact ℓ∞ sub‑gradient
    descent.  All iterates are **symmetrised** (a[i] = a[n‑1‑i]) because the
    optimal solution is known to be symmetric; this halves the effective
    dimensionality and speeds convergence.

3.  Finishes with a light simulated‑annealing that respects the symmetry
    and the simplex constraint.

The whole search respects the provided time budget (default 1000 s) and
returns the best sequence found.
"""

import math
import time
import numpy as np
from numpy.fft import rfft, irfft

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force symmetry a[i] = a[-1-i] by averaging with its reverse."""
    return (v + v[::-1]) * 0.5


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT (length 2·len(a)‑1).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution on {0,…,n‑1} (probability vector)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) on a uniform grid, normalised to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_vals = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    vals[~np.isfinite(vals)] = 0.0
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


def _powerlaw_boundary_distribution(n: int, p: float) -> np.ndarray:
    """Distribution ∝ ((i+0.5)*(n‑i‑0.5))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5) + 1e-12) ** (-p)
    vals /= vals.sum()
    return vals


def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth‑max surrogate
        Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, current_max).
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # stability trick
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 0.5,
                   alpha_factor: float = 1.01,
                   alpha_max: float = 5000.0,
                   lr0: float = 0.25,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """Projected Adam on the smooth‑max surrogate, keeping symmetry."""
    a = _symmetrize(a0.copy())
    a = _project_onto_simplex(a, s=1.0)

    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # enforce symmetry and simplex
        a = _symmetrize(a)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional jitter to escape shallow plateaus
        if t % 400 == 0 and np.random.rand() < 0.1:
            jitter = np.random.randn(n) * 5e-7
            a += jitter
            a = _symmetrize(a)
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


def _max_subgradient_step(a: np.ndarray):
    """
    Exact ℓ∞ sub‑gradient step for the current a (symmetric a).
    Returns (grad, current_peak).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Short exact‑max refinement using sub‑gradient descent (with symmetry)."""
    a = _symmetrize(a0.copy())
    a = _project_onto_simplex(a, s=1.0)

    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.03 / np.sqrt(it)          # a slightly larger initial step
        a = a - step * grad
        a = _symmetrize(a)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        if it % 300 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _symmetrize(a)
            a = _project_onto_simplex(a, s=1.0)

    return best_a


def _simulated_annealing(a0: np.ndarray, deadline: float):
    """Symmetry‑preserving simulated annealing."""
    n = a0.shape[0]
    cur_a = _symmetrize(a0.copy())
    cur_a = _project_onto_simplex(cur_a, s=1.0)
    cur_peak = _fft_self_convolve(cur_a).max()
    best_a = cur_a.copy()
    best_peak = cur_peak

    start = time.time()
    total_time = max(deadline - start, 1e-6)

    # exponential cooling schedule
    start_temp = max(cur_peak, 1e-3) * 0.6
    final_temp = start_temp * 1e-5
    if final_temp <= 0.0:
        final_temp = 1e-9
    cooling_rate = -math.log(final_temp / start_temp) / total_time

    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1

        # pick source and destination (respect symmetry)
        src = np.random.randint(n // 2)  # only left half; right half mirrors
        dst = np.random.randint(n // 2)
        if src == dst:
            continue

        # amount we are allowed to move (cannot exceed either source)
        max_delta = min(cur_a[src], cur_a[n - 1 - src], 0.1)
        if max_delta <= 1e-12:
            continue
        delta = np.random.rand() * max_delta

        a_new = cur_a.copy()
        # remove from both symmetrical sources
        a_new[src] -= delta
        a_new[n - 1 - src] -= delta
        # add to both symmetrical destinations
        a_new[dst] += delta
        a_new[n - 1 - dst] += delta

        new_peak = _fft_self_convolve(a_new).max()
        delta_obj = new_peak - cur_peak

        elapsed = time.time() - start
        temperature = start_temp * math.exp(-cooling_rate * elapsed)

        accept = False
        if delta_obj < 0.0:
            accept = True
        else:
            prob = math.exp(-delta_obj / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_a = a_new
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_a = cur_a.copy()

        # occasional random restart from the current best
        if iter_cnt % 25000 == 0 and np.random.rand() < 0.06:
            cur_a = best_a.copy()
            cur_peak = best_peak

    return best_a, best_peak


def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a non‑negative sequence to a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a low‑score step‑function.  The routine follows the
    multi‑phase scheme described in the module docstring and returns
    a plain Python list ready for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # ------------------------------------------------------------------
    # 1️⃣  =========  Broad initialisation  =========
    # try to reuse a previously discovered construction, if any
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_seq = _project_onto_simplex(base_seq, s=1.0)
        base_n = len(base_seq)
    except Exception:
        base_seq = None
        base_n = 5000   # reasonable default

    # Decide a handful of lengths – we go up to ~150 % of the base size,
    # but never exceed 20000 (still cheap for FFT).
    lengths = [
        max(500, int(base_n * f))
        for f in (0.6, 0.8, 1.0, 1.2, 1.5, 2.0)
        if int(base_n * f) >= 500
    ]
    lengths = sorted(set(lengths))
    # cap maximal length for safety
    lengths = [min(20000, L) for L in lengths]

    overall_best_seq = None
    overall_best_score = float("inf")
    overall_best_n = None

    print("[phase 1] broad search")
    for n in lengths:
        if time.time() >= deadline:
            break

        starters = {}

        # classic families
        starters["arcsine"] = _arcsine_distribution(n)

        # symmetric Beta (close to arcsine) and a few asymmetric variants
        for a in (0.35, 0.40, 0.45, 0.50, 0.55, 0.60, 0.65):
            starters[f"beta_sym_{a:.2f}"] = _beta_distribution(n, a, a)

        for a, b in ((0.3, 0.8), (0.8, 0.3), (0.4, 0.7), (0.7, 0.4)):
            starters[f"beta_{a:.2f}_{b:.2f}"] = _beta_distribution(n, a, b)

        # power‑law family (boundary‑focused)
        for p in (0.45, 0.5, 0.55, 0.6):
            starters[f"power_{p:.2f}"] = _powerlaw_boundary_distribution(n, p)

        # mixtures of arcsine and uniform
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            starters[f"mix_{w:.2f}"] = (1.0 - w) * _arcsine_distribution(n) + w * np.full(n, 1.0 / n)

        # Dirichlet draws (different concentration)
        for conc in (0.2, 0.5, 1.0, 2.0):
            starters[f"dirichlet_{conc:.1f}"] = np.random.dirichlet(np.full(n, conc))

        # if we have a previous good construction, interpolate it
        if base_seq is not None:
            starters["prev_interp"] = _interpolate_to_length(base_seq, n)

        # evaluate cheap proxy for each starter
        for tag, vec in starters.items():
            vec = _project_onto_simplex(vec, s=1.0)
            vec = _symmetrize(vec)                     # enforce symmetry early
            peak = _fft_self_convolve(vec).max()
            fast_score = 2.0 * n * peak
            if fast_score < overall_best_score:
                overall_best_score = fast_score
                overall_best_seq = vec.copy()
                overall_best_n = n
                print(f"[init] n={n:5d} {tag:<15} fast_score={fast_score:.6f}")

    # Fallback – uniform distribution
    if overall_best_seq is None:
        n = max(500, base_n)
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_n = n
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # 2️⃣  =========  Adam + exact max refinement  =========
    time_left = deadline - time.time()
    if time_left > 3.0:
        n = overall_best_n
        print(f"[phase 2] Adam optimisation (n={n}) – {time_left:.1f}s left")
        adam_deadline = time.time() + time_left * 0.55
        a_opt, _ = _adam_optimize(overall_best_seq, adam_deadline,
                                  init_alpha=0.5,
                                  alpha_factor=1.02,
                                  alpha_max=5000.0,
                                  lr0=0.3)
        # exact ℓ∞ refinement
        subgrad_deadline = time.time() + time_left * 0.30
        a_opt = _refine_max(a_opt, subgrad_deadline)
        overall_best_seq = a_opt.copy()
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print(f"[phase 2] post‑refinement fast_score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  =========  Simulated annealing polishing  =========
    time_left = deadline - time.time()
    if time_left > 1.0:
        n = overall_best_n
        print(f"[phase 3] simulated annealing (n={n}) – {time_left:.1f}s left")
        a_final, final_peak = _simulated_annealing(overall_best_seq, deadline)
        overall_best_seq = a_final.copy()
        overall_best_score = 2.0 * n * final_peak
        print(f"[phase 3] SA fast_score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  =========  Final exact evaluation  =========
    try:
        true_score = evaluate_sequence(overall_best_seq.tolist())
        print(f"[final] evaluate_sequence score={true_score:.6f}")
        if true_score < overall_best_score:
            overall_best_score = true_score
    except Exception as exc:
        print("[final] evaluate_sequence raised:", exc)

    print(f"Best score (approx.): {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
