"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)^2)

(the black‑box `evaluate_sequence`).  

Algorithm (≈ 1 000 s budget)

1. **Family sampling** – For several lengths (≈ 0.6…2.0× a known good length)
   we generate many parametric families:
   * arcsine, uniform, random Dirichlet, symmetric Beta,
   * power‑law a_i ∝ ((i+½)(n‑i‑½))^(‑β) for β∈[0.4,1.2],
   * exponential decay,
   * a three‑level block (front‑middle‑back) with random lengths and
     height ratios.
   The fast score `2·n·max(conv)` is evaluated via FFT and the best
   vector is kept.

2. **Two–level block search** – A cheap exhaustive scan of the classic
   (front‑block / back‑block) family (already present in the reference).

3. **Multiple Adam runs** – From the best vector found and a handful of
   diverse seeds (random Dirichlet, best arcsine, best power‑law, …) we
   run Adam on the smooth‑max surrogate

        L_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k)

   with a continuation schedule α→α_max (≈ 2·10⁴).  The best result
   across all runs is retained.

4. **Exact sub‑gradient descent + greedy mass‑transfer** – Starting from
   the Adam result we run a diminishing‑step sub‑gradient descent on the
   true ℓ∞ objective, followed by a directed local search that moves a
   small fraction of mass from indices strongly contributing to the
   current peak to random destinations (only improving moves are kept).

5. **Final evaluation** – The black‑box `evaluate_sequence` is called on
   the best vector; the list is returned.

All steps respect the time budget by checking `time.time()`.  The code
uses only NumPy and the standard library; no closures or lambdas are
used, per the competition rules.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a via real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth‑max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # soft‑max weights (stable)
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.025,
                   alpha_max: float = 20000.0,
                   lr0: float = 0.4,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns (best_vector, best_peak) seen during the run.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule (slow decay)
        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # keep the best exact peak seen so far
        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        # continuation on the smoothness parameter
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny jitter to avoid stagnation
        if t % 500 == 0 and random.random() < 0.12:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak

def _max_subgradient_step(a: np.ndarray):
    """
    Exact ℓ∞ sub‑gradient step.
    Uses a uniform weight over all indices attaining the current maximum.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Sub‑gradient descent on the exact ℓ∞ objective with diminishing step.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.025 / np.sqrt(it)   # diminishing step
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        # occasional jitter
        if it % 700 == 0 and random.random() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a


def _directed_local_search(a0: np.ndarray, deadline: float):
    """
    Greedy mass‑transfer search that always targets the current peak.
    Only improving moves are accepted.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        # pick an index belonging to a maximal convolution term
        candidates = np.where(b >= cur_max - 1e-12)[0]
        if candidates.size == 0:
            break
        max_idx = random.choice(candidates.tolist())

        # indices that contribute to (a*a)[max_idx]
        i_low = max(0, max_idx - (n - 1))
        i_high = min(max_idx, n - 1)
        i_vals = np.arange(i_low, i_high + 1, dtype=int)
        prod = a[i_vals] * a[max_idx - i_vals]

        if prod.sum() == 0.0:
            src = random.randint(0, n - 1)
        else:
            prob = prod / prod.sum()
            src = random.choices(i_vals.tolist(), weights=prob.tolist(), k=1)[0]

        dst = random.randint(0, n - 1)
        if dst == src:
            continue

        # move a modest fraction of source mass (up to 5 %)
        max_delta = min(a[src], 0.05)
        delta = random.random() * max_delta

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        b_new = _fft_self_convolve(a_new)
        new_max = b_new.max()

        if new_max < cur_max - 1e-15:
            a = a_new
            b = b_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()

        # small random jitter to occasionally escape plateaus
        if random.random() < 0.006:
            a += np.random.randn(n) * 1e-7
            a = _project_onto_simplex(a, s=1.0)
            b = _fft_self_convolve(a)
            cur_max = b.max()

    return best_a

# Family generators
def _arcsine_family(n: int) -> np.ndarray:
    """Arcsine density on {0,…,n‑1} normalised to sum 1."""
    vals = 1.0 / np.sqrt((np.arange(n, dtype=np.float64) + 0.5) *
                        (n - np.arange(n, dtype=np.float64) - 0.5))
    vals /= vals.sum()
    return vals


def _beta_family(n: int, a: float) -> np.ndarray:
    """Symmetric Beta(a,a) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_vals = (a - 1.0) * np.log(x) + (a - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    vals[~np.isfinite(vals)] = 0.0
    if vals.sum() == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _power_law_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ ((i+½)(n‑i‑½))^(‑beta) , normalised."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = (i * (n - i)) ** (-beta)
    vals /= vals.sum()
    return vals


def _exponential_family(n: int, lam: float) -> np.ndarray:
    """Simple exponential decay."""
    vals = np.exp(-lam * np.arange(n, dtype=np.float64))
    vals /= vals.sum()
    return vals


def _three_level_family(n: int, deadline: float):
    """
    Random search in the three‑level block family.
    Returns (best_vec, best_score) or (None, inf) if out of time.
    """
    best_vec = None
    best_score = float('inf')
    max_iters = 4000  # hard limit; loop exits early if time runs out
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        # random block lengths (must leave at least 1 element per block)
        L1 = random.randint(1, n - 2)
        L2 = random.randint(1, n - L1 - 1)
        # random ratios >= 1 (log‑uniform)
        r1 = 10 ** random.uniform(0.0, 2.0)   # front/back ratio
        r2 = 10 ** random.uniform(0.0, 2.0)   # middle/back ratio
        # solve for back height r so that sum = 1
        denom = L1 * r1 * r2 + L2 * r2 + (n - L1 - L2)
        r = 1.0 / denom
        q = r2 * r
        p = r1 * q
        seq = np.empty(n, dtype=np.float64)
        seq[:L1] = p
        seq[L1:L1 + L2] = q
        seq[L1 + L2:] = r
        # fast score
        peak = _fft_self_convolve(seq).max()
        score = 2.0 * n * peak
        if score < best_score:
            best_score = score
            best_vec = seq.copy()
    return best_vec, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Main entry point. Returns a Python list of non‑negative numbers.
    The algorithm follows the outline described in the module docstring.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Length handling – reuse a previous construction if available
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        base_seq = _project_onto_simplex(base_seq, s=1.0)
        base_len = len(base_seq)
    except Exception:
        base_seq = None
        base_len = 4000

    # lengths to explore (0.6× … 2.0× base, at least 500)
    factor_grid = [0.6, 0.8, 1.0, 1.2, 1.4, 1.6, 1.8, 2.0]
    lengths = [max(500, int(base_len * f)) for f in factor_grid
               if int(base_len * f) >= 500]
    # cap maximal length to keep FFT cheap
    lengths = [min(L, 8000) for L in lengths]

    overall_best_seq = None
    overall_best_score = float('inf')
    initial_candidates = []   # store a few promising vectors for later Adam

    # ------------------------------------------------------------------
    # Phase 0 – broad family sampling
    print("[phase 0] broad family sampling")
    for n in lengths:
        if time.time() >= deadline:
            break

        families = {}

        # deterministic families
        families["arcsine"] = _arcsine_family(n)
        families["uniform"] = np.full(n, 1.0 / n, dtype=np.float64)
        families["random1"] = np.random.rand(n); families["random1"] /= families["random1"].sum()
        families["random2"] = np.random.rand(n); families["random2"] /= families["random2"].sum()

        # symmetric Beta densities
        for a in (0.25, 0.35, 0.45, 0.55, 0.65, 0.75):
            families[f"beta_{a:.2f}"] = _beta_family(n, a)

        # exponential decay
        for lam in (0.0005, 0.001, 0.002):
            families[f"exp_{lam:.4f}"] = _exponential_family(n, lam)

        # power‑law family
        for beta in np.linspace(0.4, 1.2, 15):
            families[f"pow_{beta:.2f}"] = _power_law_family(n, beta)

        # mixture of arcsine & uniform
        for w in (0.2, 0.4, 0.6):
            families[f"mix_{w:.1f}"] = (1.0 - w) * families["arcsine"] + w * families["uniform"]

        if base_seq is not None:
            # linear interpolation of previous best to current length
            x_old = np.linspace(0.0, 1.0, base_len)
            x_new = np.linspace(0.0, 1.0, n)
            interp = np.interp(x_new, x_old, base_seq)
            interp = np.maximum(0.0, interp)
            interp /= interp.sum()
            families["prev"] = interp

        for tag, vec in families.items():
            vec = _project_onto_simplex(vec, s=1.0)
            peak = _fft_self_convolve(vec).max()
            fast_score = 2.0 * n * peak
            if fast_score < overall_best_score:
                overall_best_score = fast_score
                overall_best_seq = vec.copy()
                print(f"[init] n={n:4d} {tag:<12s} fast_score={fast_score:.6f}")

            # remember a few of the best vectors for Adam seeding
            if fast_score < overall_best_score * 1.02:   # within 2 % of current best
                initial_candidates.append(vec.copy())

        # two‑level block family (already in the reference)
        seq2, score2 = _three_level_family(n, deadline)  # using the same helper for speed
        if seq2 is not None and score2 < overall_best_score:
            overall_best_score = score2
            overall_best_seq = seq2.copy()
            print(f"[three‑level] n={n:4d} fast_score={score2:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – cheap two‑level (front/back) parametric search
    print("[phase 1] two‑level parametric search")
    for n in lengths:
        if time.time() >= deadline:
            break
        # reuse the same routine that scans a modest grid of ratios
        seq, score = _two_level_family(n, deadline)
        if seq is not None and score < overall_best_score:
            overall_best_score = score
            overall_best_seq = seq.copy()
            print(f"[two‑level] n={n:4d} fast_score={score:.6f}")

    # ------------------------------------------------------------------
    # Ensure we have a fallback
    if overall_best_seq is None:
        n = max(500, base_len)
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # Phase 2 – multiple Adam optimisation runs
    time_left = deadline - time.time()
    if time_left > 8.0:
        n = len(overall_best_seq)
        # prepare a diverse set of seeds
        seeds = [overall_best_seq.copy()]
        # add a handful of random Dirichlet vectors
        for _ in range(3):
            rnd = np.random.rand(n)
            seeds.append(rnd / rnd.sum())
        # best arcsine candidate (if present)
        arcs = _arcsine_family(n)
        seeds.append(arcs)
        # best power‑law (we pick the beta that gave the smallest fast score)
        best_pow = None
        best_pow_score = float('inf')
        for beta in np.linspace(0.4, 1.2, 25):
            v = _power_law_family(n, beta)
            sc = 2.0 * n * _fft_self_convolve(v).max()
            if sc < best_pow_score:
                best_pow_score = sc
                best_pow = v
        if best_pow is not None:
            seeds.append(best_pow)

        # unique‑ify seeds (by rounding)
        uniq = []
        seen = set()
        for s_vec in seeds:
            key = tuple(np.round(s_vec, 12))
            if key not in seen:
                seen.add(key)
                uniq.append(s_vec)
        seeds = uniq

        # allocate roughly 70 % of remaining time to Adam, split evenly
        adam_budget = time_left * 0.70
        per_seed = adam_budget / max(1, len(seeds))

        best_adam_vec = overall_best_seq.copy()
        best_adam_peak = _fft_self_convolve(best_adam_vec).max()

        print(f"[phase 2] Adam optimisation – {len(seeds)} seeds, {per_seed:.1f}s each")
        for idx, seed_vec in enumerate(seeds, 1):
            seed_deadline = time.time() + per_seed
            if time.time() >= deadline:
                break
            vec, peak = _adam_optimize(seed_vec, seed_deadline)
            if peak < best_adam_peak - 1e-15:
                best_adam_peak = peak
                best_adam_vec = vec.copy()
                # update fast score for logging
                fast = 2.0 * n * best_adam_peak
                print(f"[adam] seed {idx}/{len(seeds)} improved fast_score={fast:.6f}")

        overall_best_seq = best_adam_vec.copy()
        overall_best_score = 2.0 * n * best_adam_peak

    # ------------------------------------------------------------------
    # Phase 3 – exact sub‑gradient descent + directed local search
    time_left = deadline - time.time()
    if time_left > 6.0:
        n = len(overall_best_seq)
        sub_time = time_left * 0.55
        loc_time = time_left - sub_time
        print(f"[phase 3] sub‑gradient ({sub_time:.1f}s) + local search ({loc_time:.1f}s) – n={n}")

        a_refined = _refine_max(overall_best_seq, time.time() + sub_time)
        a_final = _directed_local_search(a_refined, time.time() + loc_time)

        overall_best_seq = a_final.copy()
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print(f"[phase 3] after refinement fast_score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the true black‑box
    try:
        true_score = evaluate_sequence(overall_best_seq.tolist())
        print(f"[final] evaluate_sequence score={true_score:.6f}")
        if true_score < overall_best_score:
            overall_best_score = true_score
    except Exception as exc:
        print("[final] evaluate_sequence raised:", exc)

    print(f"Best approx. score {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # Return a plain Python list as required
    return overall_best_seq.tolist()

# Helper used in phase 1 (two‑level parametric search) – retained unchanged
def _two_level_family(n: int, deadline: float):
    """
    Search a simple two‑level family:
        a_i = p   for i < L   (front block)
        a_i = q   otherwise   (back block)
    with p = r·q, r≥1.  Returns the best vector found and its fast score.
    """
    best_seq = None
    best_score = float('inf')
    step_L = max(1, n // 100)          # step ≈ 1 % of the length
    # a modest set of ratios to explore (log‑space)
    r_grid = np.logspace(0.0, 2.3, num=25)   # from 1 to ≈200

    for L in range(step_L, n, step_L):
        if time.time() > deadline:
            break
        denom_base = n - L
        for r in r_grid:
            denom = L * r + denom_base
            q = 1.0 / denom
            p = r * q
            seq = np.empty(n, dtype=np.float64)
            seq[:L] = p
            seq[L:] = q
            peak = _fft_self_convolve(seq).max()
            score = 2.0 * n * peak
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
    return best_seq, best_score
