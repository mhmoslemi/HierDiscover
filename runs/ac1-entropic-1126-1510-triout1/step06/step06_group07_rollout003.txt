import time

import numpy as np

def _project_to_simplex(v, z):
    if z == 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _softmax(y):
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / np.sum(e)

def _smooth_obj_and_grad(y, mu, n, S):
    p = _softmax(y)
    a = S * p
    conv = np.convolve(a, a)
    max_c = conv.max()
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv)
        if cnt > 0:
            w[mask] = 1.0 / cnt
        f_val = max_c
    # gradient wrt a
    conv_wr = np.convolve(w, a[::-1])
    g_a = 2.0 * conv_wr[n-1:n-1+n]
    inner = np.dot(g_a, p)
    grad_y = S * p * (g_a - inner)
    return float(f_val), grad_y.astype(np.float64)

class _SmoothMaxObj:
    __slots__ = ("mu","n","S")
    def __init__(self, mu, n, S):
        self.mu = mu
        self.n = n
        self.S = S
    def __call__(self, y):

    conv_wr = np.convolve(w, a[::-1])
    g = 2.0 * conv_wr[n-1:n-1+n]
    return g, max_c

def _pairwise_transfer(a, i, j, delta):
    # delta > 0: move mass from i to j
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    # ensure non-negative (can be slight negative due to rounding)
    a2[i] = max(a2[i], 0.0)
    return a2

"""
Hybrid multi‑restart smooth‑max optimisation, projected sub‑gradient descent,
large‑scale stochastic hill‑climbing and a final simulated‑annealing
finetuning.  The routine explores several problem sizes, keeps the best
non‑negative step function it ever sees and respects the supplied time
budget (default 1000 s).  All helpers are defined at module level – no
closures or lambdas are used.
"""
import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(y: np.ndarray, mu: float,
                         n: int, S: float) -> tuple[float, np.ndarray]:
    """
    Value and gradient of the smooth‑max surrogate:

        f_μ(y) = max(conv) + μ·log Σ exp((conv‑max)/μ)

    Parameters
    ----------
    y : ndarray (n,)
        Unconstrained log‑probabilities (soft‑max argument).
    mu : float (>0).  mu = 0 corresponds to the hard max.
    n, S : problem size and target ℓ¹‑norm.
    """
    p = _softmax(y)               # Σ p_i = 1, p_i ≥ 0
    a = S * p                     # Σ a_i = S

    conv = np.convolve(a, a)      # length 2·n‑1

    max_c = conv.max()
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)        # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        if cnt > 0:
            w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. a  (see derivation in the statement)
    conv_wr = np.convolve(w, a[::-1])
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

    inner = np.dot(g_a, p)               # scalar Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)       # chain rule through soft‑max

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObj:
    """Callable object for scipy.optimize: returns (value, gradient)."""
    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _smooth_obj_and_grad(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
def _hard_subgradient(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient of the true max‑objective  max(conv)  at point a.
    Returns (g, max_conv) where g has the same length as a.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_c = conv.max()
    mask = conv == max_c
    cnt = np.count_nonzero(mask)
    w = np.zeros_like(conv, dtype=float)
    if cnt > 0:
        w[mask] = 1.0 / cnt
    conv_wr = np.convolve(w, a[::-1])
    g = 2.0 * conv_wr[n - 1:n - 1 + n]
    return g, max_c
