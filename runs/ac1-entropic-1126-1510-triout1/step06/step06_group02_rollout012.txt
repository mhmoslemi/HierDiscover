def solve_lp_max_sum(f, M):
    n = len(f)
    # Build Toeplitz matrix A shape (2n-1, n):
    col = np.r_[f, np.zeros(n-1)]
    row = np.r_[f[0], np.zeros(n-1)]
    A = scipy.linalg.toeplitz(col, row)[:2*n-1, :n]
    # Use linprog:
    c = -np.ones(n)  # maximize sum = -c^T x
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError("LP failed")
    g = res.x
    return g
```

"""
Propose a step‑function with a low (≈ 1.5) value of
evaluate_sequence().  The algorithm consists of two stages.

Stage 1 – small scale linear‑programming optimisation
----------------------------------------------------
For a modest size n≈800 we repeatedly solve the LP

    maximise   Σ g_i
    subject to  (f * g)_k ≤ M  for all k          (linear in g)
                g_i ≥ 0,
    where M = max(f*f) and f is the current distribution (Σf=1).

The LP finds a direction g that can increase the total mass while
still respecting the convolution bound M.  After normalising g to a
probability vector we mix f←(1‑t)f + t g (t≈0.2) and repeat.
Since the sum of the vector is kept equal to 1, the objective
2·n·max(f*f) is exactly the value returned by ``evaluate_sequence``
(up to the harmless scaling performed inside that function).

Stage 2 – up‑sampling & projected gradient refinement
----------------------------------------------------
The best small‑scale shape is interpolated to several larger lengths
(≈ 3000–12000).  For each length we run a simple projected sub‑gradient
method on the simplex using the exact sub‑gradient of
max(f*f).  The step size decays as η₀/√t and an Euclidean projection
onto the simplex keeps the variables feasible.  Whenever the cheap
score 2·n·max(f*f) improves we keep the best vector; the loop stops
as soon as the target value 1.503 is reached or the time budget is
exhausted.

Both stages respect the overall 1000 s time budget and print progress
messages.  The final sequence (sum≈1) is returned; ``evaluate_sequence``
will automatically clamp any tiny rounding errors.

The implementation falls back to a pure gradient search if SciPy is
unavailable.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) with respect to a (Σa = 1).
    a : current probability vector (sum = 1)
    conv : self‑convolution of a
    """
    n = a.shape[0]
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size

    # convolution of reversed a with weight vector w
    rev_a = a[::-1]
    # use FFT for speed
    size = 1 << ((rev_a.size + w.size - 1) - 1).bit_length()
    A = np.fft.rfft(rev_a, n=size)
    B = np.fft.rfft(w, n=size)
    conv_rev = np.fft.irfft(A * B, n=size)[: n + w.size - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even shifts (i = j)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad


# ----------------------------------------------------------------------
def _lp_max_sum(f: np.ndarray, M: float):
    """
    Solve the LP:
        maximise   Σ g_i
        subject to (f * g)_k ≤ M   for all k,
                   g_i ≥ 0.
    Returns the optimal g (not normalised).
    """
    n = f.shape[0]
    # Build the Toeplitz matrix A (size (2n-1) × n) such that
    # A @ g = f * g   (linear convolution).
    # First column of the Toeplitz matrix is [f, 0,…,0] (length 2n-1).
    col = np.r_[f, np.zeros(n - 1, dtype=np.float64)]
    row = np.r_[f[0], np.zeros(n - 1, dtype=np.float64)]
    try:
        # SciPy's toeplitz is efficient and returns a dense array.
        from scipy.linalg import toeplitz
        A = toeplitz(col, row)[: 2 * n - 1, :n]
    except Exception as e:
        # Fallback: build dense matrix manually (slow but works for tiny n)
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for k in range(2 * n - 1):
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            A[k, i_min : i_max + 1] = f[k - np.arange(i_min, i_max + 1)]

    # Linear program in standard form: minimise -Σg  (i.e. maximise Σg)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    try:
        from scipy.optimize import linprog
        res = linprog(c,
                      A_ub=A,
                      b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True})
        if not res.success:
            raise RuntimeError(res.message)
        return res.x
    except Exception as exc:
        # If linprog is unavailable or fails, raise to caller.
        raise RuntimeError(f"LP solve failed: {exc}") from exc


# ----------------------------------------------------------------------
def _refine_by_projected_subgrad(a_prob: np.ndarray,
                                n: int,
                                time_limit: float,
                                target: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Refine a probability vector (sum = 1) by projected sub‑gradient descent.
    Returns the best vector seen and its cheap score 2·n·max(conv).
    """
    start = time.time()
    best = a_prob.copy()
    conv = _fft_self_convolve(best)
    best_score = 2.0 * n * conv.max()

    it = 0
    eta0 = 0.7  # initial step size factor
    while time.time() - start < time_limit:
        it += 1
        conv = _fft_self_convolve(a_prob)
        grad = _exact_gradient(a_prob, conv)

        eta = eta0 / math.sqrt(it)
        a_new = a_prob - eta * grad
        a_new = _project_onto_simplex(a_new, s=1.0)

        cur_conv = _fft_self_convolve(a_new)
        cur_score = 2.0 * n * cur_conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best = a_new.copy()
            print(f"[{time.time():.1f}] refine n={n} it={it} score={best_score:.6f}")
            if best_score <= target:
                break

        # Accept the new point if it improves (simple descent); otherwise keep old.
        if cur_score <= 2.0 * n * conv.max():
            a_prob = a_new
        else:
            # occasional jitter to escape shallow plateaus
            if it % 1000 == 0:
                jitter = np.random.randn(n) * 1e-8
                a_prob = _project_onto_simplex(a_prob + jitter, s=1.0)

    return best, best_score
