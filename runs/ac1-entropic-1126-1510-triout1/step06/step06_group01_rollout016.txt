"""
Improved search for a non‑negative step function that minimizes
the evaluate_sequence() objective.

Strategy
--------
1. Scan a handful of analytic families (uniform, arcsine, symmetric
   beta, power‑law tails) for several lengths and keep the most promising
   length / seed.
2. Run many random restarts (Dirichlet draws) for that length.
   Each restart is refined by
   * a multi‑stage smoothed Adam optimisation of the log‑sum‑exp
     approximation of max(a*a);
   * an exact max‑subgradient descent with momentum;
   * a cheap random‑transfer hill‑climber.
3. After each optimisation step the vector is projected back onto the
   probability simplex and symmetrised (a[i] = a[-i‑1]) because the
   optimal shape is known to be symmetric.
4. The best vector seen overall is optionally smoothed with a tiny
   Gaussian kernel and re‑projected; the best of those variants is
   returned.

All heavy work is done with FFT‑based convolutions (O(n log n)) and the
code respects the 1000 s wall‑clock budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_powerlaw(n: int, beta: float) -> np.ndarray:
    """Shape ∝ ((i+0.5)*(n-i-0.5))^{-beta}, symmetric, normalised."""
    i = np.arange(n, dtype=np.float64)
    w = 1.0 / ((i + 0.5) * (n - i - 0.5)) ** beta
    w /= w.sum()
    return w

def _smooth_descent_extended(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the soft‑max approximation of max(a*a).
    Uses a long continuation schedule and symmetrisation.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # (alpha, steps). More steps than the original SOTA.
    schedule = [
        (1.0, 2500),
        (5.0, 2500),
        (20.0, 2500),
        (100.0, 2500),
        (500.0, 2500),
        (2000.0, 2500),
    ]

    for alpha, steps in schedule:
        # learning rate decays roughly like 1/sqrt(alpha) for stability
        lr = base_lr * max(0.1, 1.0 / math.sqrt(alpha))
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score
            t += 1

            b = _fft_self_convolve(a)
            # stable soft‑max
            maxb = b.max()
            exp_terms = np.exp(alpha * (b - maxb))
            w = exp_terms / exp_terms.sum()

            # gradient of μ_α = (1/α) log Σ exp(α b_i)
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # projection + symmetrisation
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a, 1.0)
            a = _symmetrize(a)

            # occasional exact evaluation
            if t % 200 == 0:
                cur_score = _max_convolution_score(a)
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_seq = a.copy()
    return best_seq, best_score

def _exact_subgradient_refine(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑objective, with momentum.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)

    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps)

        w = np.zeros_like(b)
        w[active] = 1.0

        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / math.sqrt(iteration)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)
        a = _symmetrize(a)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] subgrad iter={iteration} new best {best_score:.6f}")

    return best_seq, best_score

def _local_hill_climb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Tiny random‑transfer hill‑climber on the simplex.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)
    cur_score = best_score

    max_transfer = 0.07
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        # exponentially shrinking transfer size
        transfer = max_transfer * (0.5 ** (iteration / 4000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[{time.time():.2f}] hill iter={iteration} improved best {best_score:.6f}")

        # occasional jitter to escape plateaux
        if iteration % 5000 == 0 and np.random.rand() < 0.2:
            pert = np.random.randn(n) * 1e-4
            a = best_seq + pert
            a = _project_onto_simplex(a, 1.0)
            a = _symmetrize(a)
            cur_score = _max_convolution_score(a)

    return best_seq, best_score

def _gaussian_smooth(a: np.ndarray, sigma: float) -> np.ndarray:
    """Mild Gaussian smoothing (wrap‑around ignored, then re‑normalise)."""
    n = a.size
    # kernel centred at 0
    x = np.arange(-n // 2, n // 2 + 1, dtype=np.float64)
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel /= kernel.sum()
    smooth = np.convolve(a, kernel, mode='same')
    smooth = np.maximum(smooth, 0.0)
    smooth /= smooth.sum()
    return smooth


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises evaluate_sequence().
    Returns the best sequence found (as a list of Python floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # 1. pick a promising length by scanning a few cheap analytic seeds
    lengths = [1200, 2000, 3000, 5000, 8000]
    best_len = lengths[0]
    overall_best_seq = None
    overall_best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        # very cheap scan: uniform and arcsine only
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            cand = cand / cand.sum()
            sc = _max_convolution_score(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                best_len = n
                print(f"[{time.time():.2f}] scan len={n} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Random‑restart optimisation on the selected length
    n = best_len
    print(f"[{time.time():.2f}] selected length {n} for intensive search")

    # a few deterministic analytic starts
    for start in _initial_candidates_extended(n):
        if time.time() >= deadline:
            break
        start = start / start.sum()
        sc = _max_convolution_score(start)
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = start.copy()
            print(f"[{time.time():.2f}] analytic init score={sc:.6f}")

    # random Dirichlet restarts
    restarts = 60
    for r in range(restarts):
        if time.time() >= deadline:
            break

        # mixture of sparse (alpha=0.3) and dense (alpha=1.0) Dirichlet draws
        alpha = 0.3 if (r % 2 == 0) else 1.0
        cand = np.random.dirichlet(np.full(n, alpha)).astype(np.float64)

        # smooth Adam phase
        cand, sc = _smooth_descent_extended(cand, deadline)

        # exact sub‑gradient refinement
        cand, sc = _exact_subgradient_refine(cand, deadline)

        # local hill‑climb
        cand, sc = _local_hill_climb(cand, deadline)

        if sc < overall_best_score - 1e-12:
            overall_best_score = sc
            overall_best_seq = cand.copy()
            print(f"[{time.time():.2f}] new overall best {overall_best_score:.6f} (restart {r})")

    # ------------------------------------------------------------------
    # 3. Optional tiny Gaussian smoothing sweep (may shave a few 1e‑4)
    if overall_best_seq is not None:
        for sigma in (0.5, 1.0, 1.5):
            if time.time() >= deadline:
                break
            smooth_seq = _gaussian_smooth(overall_best_seq, sigma)
            sc = _max_convolution_score(smooth_seq)
            if sc < overall_best_score - 1e-12:
                overall_best_score = sc
                overall_best_seq = smooth_seq.copy()
                print(f"[{time.time():.2f}] post‑smooth sigma={sigma} improved to {sc:.6f}")

    # ------------------------------------------------------------------
    # 4. Final sanity check: make sure the vector obeys the constraints
    overall_best_seq = np.maximum(overall_best_seq, 0.0)
    overall_best_seq = overall_best_seq / overall_best_seq.sum()

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
