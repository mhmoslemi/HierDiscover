def _soft_weights(b, tau):
    # compute weights proportional to exp(b/tau)
    # we want sum of weights = 1 (like probability)
    # compute stable: subtract max b to avoid overflow.
    maxb = np.max(b)
    expb = np.exp((b - maxb) / tau)  # careful scaling
    s = expb.sum()
    if s == 0:
        w = np.ones_like(b) / len(b)
    else:
        w = expb / s
    return w

def _optimize_sequence(a0, max_iter, step0, tau0, tau_decay, optimizer_type, time_deadline, rng):

   a = a0.copy()
   best_a = a.copy()
   best_score = raw_score(a)
   # for Adam:
   m = np.zeros_like(a)
   v = np.zeros_like(a)
   beta1 = 0.9
   beta2 = 0.999
   eps = 1e-8
   for it in range(1, max_iter+1):
        if time.time() >= time_deadline:
            break
        b = _self_convolution_full(a)  # shape 2n-1
        cur_max = b.max()
        # compute weights according to optimizer
        if optimizer_type == 'adam_smooth':
            tau = tau0 * (tau_decay ** (it-1))
            w = _soft_weights(b, tau)
        else:
            # active subgradient
            w = np.zeros_like(b)
            tol = max(1e-12, cur_max * 1e-12)
            active = np.where(b >= cur_max - tol)[0]
            if active.size:
                w[active] = 1.0 / active.size

        # gradient:
        a_rev = a[::-1]
        grad = 2.0 * _fft_convolve(a_rev, w)[len(a)-1:len(a)-1+len(a)]

        # step:
        # For Adam:
        if optimizer_type == 'adam_smooth' or optimizer_type == 'adam_active':
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad**2)
            m_hat = m / (1 - beta1**it)
            v_hat = v / (1 - beta2**it)
            lr = step0 / (np.sqrt(v_hat) + eps)
            a = a - lr * m_hat
        else:
            # simple momentum:
            # Not used maybe
            a = a - step0 / math.sqrt(it) * grad

        # projection
        a = project_onto_simplex(a)

        cur_score = raw_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # log improvement

        # occasional jitter maybe
        if it % 1000 == 0 and rng.random() < 0.05:
            a += rng.normal(scale=1e-6, size=a.shape)
            a = project_onto_simplex(a)

   return best_a, best_score

"""
Propose a low‑score step function.

Algorithm
---------
1. **Broad sampling** –  for a few lengths (2000‑16000) we generate a
   modest set of candidate probability vectors:
   * uniform, arcsine (Beta(0.5,0.5))
   * several symmetric Beta(α,α) with α on a small grid
   * Dirichlet draws with random concentration
   * mixtures of two Beta components
   The candidates are projected onto the simplex and evaluated with the
   fast proxy 2·n·max(a*a).  The best candidate (lowest proxy) is kept.

2. **Refinement** – starting from the best proxy candidate we run a
   long–run Adam optimiser on a *smoothed* max‑convolution objective.
   The smoothing is a soft‑max over the convolution values with a
   temperature τ that decays exponentially, turning the problem from a
   smooth one into the exact ∞‑norm as optimisation proceeds.
   After each Adam step the iterate is projected back onto the simplex.
   A tiny random jitter is added occasionally to escape shallow plateaus.

3. The final distribution (still normalised to sum = 1) is handed to
   ``evaluate_sequence`` which rescales it automatically.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution of length n, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf for numerical stability
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logpdf = np.minimum(logpdf, 700.0)               # avoid overflow
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    tot = pdf.sum()
    if tot == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= tot
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a * a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """
    Proxy score assuming Σa = 1 :  2·n·max(a*a).
    It equals the true objective because the evaluator rescales automatically.
    """
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a smoothed max‑convolution objective.
    Returns the best vector seen (projected onto the simplex) and its raw score.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.20                         # base learning rate

    # temperature schedule for soft‑max weighting
    tau0 = 0.5
    tau_min = 1e-4
    tau_decay = 0.9995                # per iteration multiplicative decay

    best_a = a.copy()
    best_val = _raw_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # self‑convolution and soft‑max weights
        b = _self_convolution_full(a)
        cur_max = b.max()
        tau = max(tau_min, tau0 * (tau_decay ** (it - 1)))
        # stable soft‑max
        maxb = cur_max
        w = np.exp((b - maxb) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # gradient of the smoothed max (2 * a_rev * w)
        a_rev = a[::-1]
        grad = 2.0 * _fft_convolve(a_rev, w)[n - 1:n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection onto simplex
        a = _project_onto_simplex(a)

        # occasional tiny jitter (helps escape plateaus)
        if it % 5000 == 0 and rng.random() < 0.12:
            a += rng.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        cur_val = _raw_score(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam it={it:,}  best_score={best_val:.6f}")

    return best_a, best_val

    if remaining > 5.0:
        # reserve ~90 % of the remaining budget for refinement
        refine_deadline = time.time() + 0.9 * remaining
        print(f"Starting refinement (≈{remaining:.1f}s left).")
        refined_seq, refined_score = _adam_optimize(best_seq, refine_deadline, rng)

        if refined_score < best_score:
            best_seq = refined_seq
            best_score = refined_score
            best_len = best_len
            print(f"Refinement improved raw score to {best_score:.6f}")

    # ------------------------------------------------------------------
    final_seq = best_seq
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Final evaluated score: {final_score:.6f} (len={len(final_seq)})")
    return final_seq.tolist()
