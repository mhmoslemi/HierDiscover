"""
Propose a step‑function (non‑negative sequence) that minimizes
``evaluate_sequence``.
The algorithm combines a large pool of analytic candidates,
projected sub‑gradient descent on the simplex and a short
simulated‑annealing fine‑tuning phase.
It works for any length n (≥3000) and respects the given time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float = 1.0) -> np.ndarray:
    """Geometric‑like decreasing shape: a_i ∝ exp(-lam·i/(n‑1))."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * i / max(1, n - 1))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _triangular_distribution(n: int) -> np.ndarray:
    """Symmetric triangular shape, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    tri = np.minimum(i + 1, n - i)
    tri /= tri.sum()
    return tri


# ----------------------------------------------------------------------
def cheap_score(a: np.ndarray) -> float:
    """Cheap surrogate of the true objective (sum of a is assumed to be 1)."""
    n = a.shape[0]
    return 2.0 * n * _fft_self_convolve(a).max()


# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    conv = a * a (full linear convolution).
    """
    n = a.shape[0]
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    # convolution of reversed a with the weight vector w
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even shifts (avoid double‑counting a_i*a_i)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad

def _generate_candidates(n: int) -> list[np.ndarray]:
    """
    Produce a diversified pool of feasible probability vectors of length n.
    Every vector sums to 1 and contains only non‑negative entries.
    """
    candidates = []

    # systematic Beta grid
    alphas = np.logspace(-0.5, 0.5, num=7)   # ~0.316 → 3.16
    betas = np.logspace(-0.5, 0.5, num=7)
    for a in alphas:
        for b in betas:
            candidates.append(_beta_distribution(n, a, b))

    # a few Dirichlet (uniform) samples
    for _ in range(12):
        v = np.random.rand(n)
        v = np.maximum(v, 1e-15)
        v /= v.sum()
        candidates.append(v)

    # exponential shapes
    for lam in (0.3, 0.7, 1.0, 1.5):
        candidates.append(_exponential_distribution(n, lam))

    # special analytic shapes
    candidates.append(_beta_distribution(n, 0.5, 0.5))          # arcsine
    candidates.append(_triangular_distribution(n))          # triangle
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))  # uniform

    # previous best (if it exists) interpolated to current length
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.size != n:

def _optimize_on_simplex(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the simplex to minimise max(a*a).
    Returns the best vector found and its cheap_score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = cheap_score(a)

    # step‑size schedule
    lr0 = 0.6                     # base learning‑rate (absorbs any scaling)
    no_improve = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        grad = _exact_gradient(a, conv)          # ∂max(conv)/∂a

        step = lr0 / np.sqrt(it)                # diminishing step
        a_new = a - step * grad
        a = _project_onto_simplex(a_new, s=1.0)

        new_score = cheap_score(a)
        if new_score + 1e-12 < best_score:
            best_score = new_score
            best_a = a.copy()
            no_improve = 0
            # occasional progress report
            print(f"[{time.time():.1f}] n={n} iter={it} cheap_score={new_score:.6f}")
        else:
            no_improve += 1

        # early stop if stagnated
        if no_improve >= 300:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
def _random_anneal(a_start: np.ndarray,
                   n: int,
                   deadline: float,
                   init_sigma: float = 0.08,
                   min_sigma: float = 0.005) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing style local search around a_start.
    Perturbs the logarithm of the vector and keeps improvements.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = cheap_score(a)

    sigma = init_sigma
    it = 0
    while time.time() < deadline:
        it += 1
        # geometric perturbation (log‑space)
        log_a = np.log(np.maximum(a, 1e-15))
        log_a += np.random.normal(scale=sigma, size=n)
        a2 = np.exp(log_a)
        a2 /= a2.sum()

        score = cheap_score(a2)
        if score < best_score:
            best_score = score
            best_a = a2.copy()
            a = a2
            sigma = max(min_sigma, sigma * 0.98)   # slowly tighten
            print(f"[{time.time():.1f}] SA improvement: cheap_score={best_score:.6f}")
        else:
            # occasional acceptance to escape shallow traps
            if np.random.rand() < 0.003:
                a = a2

        # decay sigma aggressively if we have many iterations without improvement
        if it % 500 == 0:
            sigma = max(min_sigma, sigma * 0.9)

    return best_a, best_score
