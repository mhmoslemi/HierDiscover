"""
Propose a low‑score non‑negative step function.

The search proceeds in three stages:

1. **Candidate generation** – analytic families (beta, arcsine,
   triangular) and a few low‑α Dirichlet draws are sampled on a
   moderately large length (≈ 4000).  The best few seeds (by the cheap
   proxy score 2·n·max(a*a) with ∑a=1) are kept.

2. **Long‑run mirror descent** – each seed is refined with two
   optimiser variants that work on the simplex:
      * exponentiated‑gradient (KL‑mirror descent)
      * Adam‑style adaptive moment descent.
   The optimisation runs until the global time budget is exhausted,
   so we can afford many more iterations than the reference
   implementation (≈ 10⁵ – 10⁶ updates).

3. **Final hill‑climbing** – a simple random‑transfer move is applied
   repeatedly to the current best sequence, accepting any strict
   improvement.  The transfer amount decays slowly, allowing both
   fine‑grained and occasional larger jumps.

All steps respect the 1000 s budget, always return a list of
non‑negative floats and aim for a score below the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Proxy for the true objective when ∑a = 1: 2·n·max(a*a)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution with reversed a
    conv_rev = np.fft.irfft(
        np.fft.rfft(a[::-1]) * np.fft.rfft(w, n=1 << ((a.size + w.size - 1) - 1).bit_length()),
        n=1 << ((a.size + w.size - 1) - 1).bit_length(),
    )
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even‑offset active entries
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 10**7,
                          eta0: float = 0.6, decay: float = 0.9999):
    """Exponentiated‑gradient (KL‑mirror descent)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, cur_max, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / a.size)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 10**7,
                  step0: float = 0.12,
                  beta1: float = 0.9,
                  beta2: float = 0.999):
    """Adam‑style adaptive‑moment optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, cur_max, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _random_dirichlet_alpha(n: int, alpha: float = 1.0) -> np.ndarray:
    """Draw a point on the simplex with Gamma(α,1) components."""
    x = np.random.gamma(alpha, 1.0, size=n)
    s = x.sum()
    return x / s if s > 0 else np.full_like(x, 1.0 / n)


# ----------------------------------------------------------------------
def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Discrete symmetric Beta(p,p) on {0,…,n‑1} (α=β=p)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑space to avoid under‑flow for extreme p
    logw = (p - 1) * (np.log(x) + np.log(1.0 - x))
    w = np.exp(logw - np.max(logw))
    w /= w.sum()
    return w


def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine density ≈ 1/√(x(1−x))."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # 1. Choose a working length.
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        base_n = 3500
    # a little larger than the previous best to give us more freedom
    n = min(5000, max(2000, int(base_n * 1.2)))
    print(f"[{time.time():.1f}] Working length n = {n}")

    # --------------------------------------------------------------
    # 2. Broad sampling of analytic families + heavy‑tailed Dirichlet draws.
    candidates = []   # (array, score)

    # beta family – dense sweep of small exponents (edges heavy)
    for p in np.linspace(0.01, 0.30, 30):
        a = _beta_symmetric_distribution(n, p)
        score = _max_convolution_score(a)
        candidates.append((a, score))

    # a few classic families
    for gen, name in (
        (_arcsine_distribution, "arcsine"),
        (_triangular_distribution, "triangular"),
    ):
        a = gen(n)
        score = _max_convolution_score(a)
        candidates.append((a, score))
        print(f"[{time.time():.1f}] {name:10s} score={score:.6f}")

    # Dirichlet draws with low concentration (α < 1)
    for alpha in (0.3, 0.5, 0.8):
        for _ in range(3):
            a = _random_dirichlet_alpha(n, alpha)
            score = _max_convolution_score(a)
            candidates.append((a, score))

    # include previous best if available (rescaled)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            if prev.shape[0] != n:
                # simple linear interpolation to the new size
                x_old = np.linspace(0, 1, prev.shape[0])
                x_new = np.linspace(0, 1, n)
                prev = np.interp(x_new, x_old, prev)
            prev = prev / prev.sum()
            score = _max_convolution_score(prev)
            candidates.append((prev, score))
            print(f"[{time.time():.1f}] previous best score={score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3. Keep only the most promising seeds.
    candidates.sort(key=lambda x: x[1])
    top_k = 6
    top_candidates = candidates[:top_k]
    print(f"Selected {len(top_candidates)} seeds for intensive refinement "
          f"(best score {top_candidates[0][1]:.6f})")

    best_seq = top_candidates[0][0].copy()
    best_score = top_candidates[0][1]

    # --------------------------------------------------------------
    # 4. Intensive refinement – long‑run mirror descent.
    #   The remaining time is split roughly equally across the seeds.
    remaining_time = deadline - time.time()
    if remaining_time <= 0:
        return best_seq.tolist()
    time_per_seed = remaining_time / len(top_candidates)

    for idx, (seed_seq, seed_score) in enumerate(top_candidates):
        seed_deadline = time.time() + time_per_seed
        print(f"[{time.time():.1f}] Refining seed {idx+1}/{len(top_candidates)} "
              f"(initial score {seed_score:.6f})")

        # 4a. Exponentiated‑gradient phase
        seq_exp, sc_exp = _exp_gradient_descent(seed_seq, seed_deadline,
                                                max_iters=10**7,
                                                eta0=0.6, decay=0.9999)
        if sc_exp < best_score - 1e-14:
            best_seq, best_score = seq_exp.copy(), sc_exp
            print(f"   exp‑grad improvement to {sc_exp:.6f}")

        # 4b. Adam phase (continue from the same seed)
        seq_adam, sc_adam = _adam_descent(seed_seq, seed_deadline,
                                          max_iters=10**7,
                                          step0=0.12)
        if sc_adam < best_score - 1e-14:
            best_seq, best_score = seq_adam.copy(), sc_adam
            print(f"   Adam improvement to {sc_adam:.6f}")

        # keep the best of the two runs as the new seed for the next
        # seed (idea: carry forward any improvement)
        if sc_adam < sc_exp:
            seed_seq = seq_adam
        else:
            seed_seq = seq_exp

    # --------------------------------------------------------------
    # 5. Extra long‑run Adam on the current global best (use whatever time is left).
    if time.time() < deadline:
        print(f"[{time.time():.1f}] Final long‑run Adam on best sequence "
              f"(score {best_score:.6f})")
        seq_final, sc_final = _adam_descent(best_seq, deadline,
                                            max_iters=10**7,
                                            step0=0.12)
        if sc_final < best_score - 1e-14:
            best_seq, best_score = seq_final.copy(), sc_final
            print(f"   final Adam improvement to {sc_final:.6f}")

    # --------------------------------------------------------------
    # 6. Local random‑transfer hill climbing – keep any strict improvement.
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    iteration = 0
    max_transfer = 0.08  # up‑to‑8 % of total mass per move

    while time.time() < deadline:
        iteration += 1
        # geometric decay of transferred mass
        transfer = max_transfer * (0.5 ** (iteration / 20000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_seq = a.copy()
                best_score = new_score
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} "
                      f"new best {best_score:.6f}")

        # occasional tiny jitter to avoid plateaus
        if iteration % 5000 == 0 and np.random.rand() < 0.15:
            jitter = np.random.randn(n) * 1e-4
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
