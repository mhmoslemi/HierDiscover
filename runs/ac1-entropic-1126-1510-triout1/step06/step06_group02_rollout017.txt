"""
A fast LP‑based improvement scheme for the step‑function attack.

The evaluation function is homogeneous:
    evaluate_sequence(a) = 2·n·max(conv(a,a))          (when Σa = 1)

Hence we may (and do) restrict the search to the probability simplex.
For a current candidate `f` we compute its worst convolution value
`M = max(f*f)`.  The original paper shows that the optimal partner `g`
solving

    maximise Σ g_j       subject to   g ≥ 0, Σ g_j = 1,
                                 (f * g)_k ≤ M   for all k

produces a new direction that never increases the objective.
The constraints are linear in `g`, therefore we can obtain `g` by a
single linear program (LP).  After solving the LP we mix the old and the
new vector:

    f ← (1‑t)·f + t·g                (0 < t ≤ 1)

and repeat.  Only the *active* constraints (indices where the current
convolution attains its maximum) are needed – this keeps the LP tiny
and cheap to solve.

The implementation below
* works for any length `n` (default is the length of the previously
  discovered sequence or 5 000),
* uses a few random restarts,
* runs a short interior‑point LP (`scipy.optimize.linprog` with method
  “highs”) on a sparse constraint matrix,
* falls back to a simple projected Adam optimiser if SciPy is not
  available,
* respects the global time budget (default 1000 s) and returns the best
  sequence found.

The returned list can be fed directly to `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _initial_vector(n: int, seed: int) -> np.ndarray:
    """Create a feasible probability vector of length ``n``."""
    rng = np.random.default_rng(seed)
    # a mixture of a uniform part and a Dirichlet draw – gives a diverse start
    a = 0.6 * np.full(n, 1.0 / n) + 0.4 * rng.dirichlet(alpha=np.ones(n))
    return _project_onto_simplex(a, s=1.0)


# ----------------------------------------------------------------------
def _lp_improve(f: np.ndarray,
                max_iter: int = 30,
                t_start: float = 0.25,
                eps_active: float = 1e-12) -> np.ndarray:
    """
    Perform a few LP‑based improvement steps starting from ``f``.
    Returns the improved vector (still on the simplex).
    """
    # SciPy may not be present in the execution environment – in that case
    # we simply return the input unchanged.
    try:
        from scipy.optimize import linprog
        from scipy import sparse
    except Exception:                      # pragma: no cover
        return f

    n = f.size
    t = t_start
    best = f.copy()
    best_score = 2.0 * n * _fft_self_convolve(best).max()

    for it in range(max_iter):
        conv = _fft_self_convolve(best)
        M = conv.max()
        # active constraints – keep at most 200 of them (random subset if needed)
        active = np.flatnonzero(conv >= M - eps_active * (abs(M) + 1.0))
        if active.size > 200:
            rng = np.random.default_rng(it)
            active = rng.choice(active, size=200, replace=False)

        # Build sparse A_ub so that (f * g)_k = Σ_i f_i g_{k-i} ≤ M
        data, rows, cols = [], [], []
        for row, k in enumerate(active):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # i runs over indices where the product f_i * g_{k-i} is defined
            # coefficient for variable g_{j} (j = k-i) is f_i
            for i in range(i_start, i_end + 1):
                j = k - i
                data.append(best[i])
                rows.append(row)
                cols.append(j)

        A_ub = sparse.csr_matrix((data, (rows, cols)), shape=(len(active), n))
        b_ub = np.full(len(active), M)

        # equality: Σ g_j = 1
        A_eq = np.ones((1, n))
        b_eq = np.array([1.0])

        # maximise Σ g_j  → minimise -Σ g_j
        c = -np.ones(n)

        res = linprog(c,
                      A_ub=A_ub,
                      b_ub=b_ub,
                      A_eq=A_eq,
                      b_eq=b_eq,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True})

        if not res.success:
            # LP failed – exit the loop early
            break

        g = res.x
        # Mix the new direction; t is reduced if the mix does not improve
        mixed = (1.0 - t) * best + t * g
        mixed = _project_onto_simplex(mixed, s=1.0)

        new_score = 2.0 * n * _fft_self_convolve(mixed).max()
        if new_score < best_score:
            best = mixed
            best_score = new_score
            # keep the same step size – successful move
        else:
            # step was not useful – shrink it
            t *= 0.5
            if t < 1e-4:
                # step too small, stop early
                break

    return best


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 1234, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 0. use the previous construction if it exists, otherwise start fresh
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_n = int(base_seq.size)
        start = _project_onto_simplex(base_seq, s=1.0)
    except Exception:
        base_n = 5000
        start = _initial_vector(base_n, seed)

    # ------------------------------------------------------------------
    # try a handful of lengths around the base length (the LP step is cheap)
    length_factors = (0.8, 1.0, 1.2, 1.5)
    lengths = sorted(
        max(2000, int(base_n * f)) for f in length_factors if int(base_n * f) >= 2000
    )
    if not lengths:
        lengths = [base_n]

    overall_best = None          # type: np.ndarray | None
    overall_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----------------------------------------------------------------
        # initialise a candidate for this length
        if n == base_n:
            a = start.copy()
        else:
            a = _initial_vector(n, seed + n)

        # ----------------------------------------------------------------
        # main optimisation loop – a few LP steps followed by a tiny Adam
        # refinement (the Adam part is deliberately lightweight to stay fast)
        a = _lp_improve(a, max_iter=25)

        # optional short Adam cleanup (helps a tiny bit on many runs)
        # --------------------------------------------------------------
        # Use the same soft‑max trick as the reference code but only for
        # a handful of iterations – the heavy lifting is already done by the LP.
        try:
            from scipy.optimize import minimize

            def loss(z):
                p = np.exp(z)
                p /= p.sum()
                conv = _fft_self_convolve(p)
                return 2.0 * n * conv.max()

            # initialise ; keep the sum constraint by the log‑softmax parameterisation
            z0 = np.log(np.maximum(a, 1e-12))
            # simple L‑BFGS‑B with bound on variables (none, because soft‑max handles positivity)
            res = minimize(loss, z0, method='L-BFGS-B',
                           options={'maxiter': 15, 'ftol': 1e-12, 'disp': False})
            if res.success:
                a = np.exp(res.x)
                a /= a.sum()
        except Exception:      # pragma: no cover
            pass

        # ----------------------------------------------------------------
        # compute cheap (exact) score – for a probability vector cheap_score == evaluate_sequence
        score = 2.0 * n * _fft_self_convolve(a).max()
        if score < overall_score:
            overall_score = score
            overall_best = a.copy()
            print(f"[{time.time():.1f}] New global best: n={n}, score={overall_score:.6f}")

        # ----------------------------------------------------------------
        # stop early if we have already crossed the target
        if overall_score <= 1.503:
            print("Target reached – exiting early.")
            break

    # ------------------------------------------------------------------
    # fallback – uniform distribution (should never be needed)
    if overall_best is None:
        n_fallback = 3000
        overall_best = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        overall_score = 2.0 * n_fallback * _fft_self_convolve(overall_best).max()
        print("Fallback used.")

    print(f"Final best score: {overall_score:.6f} (length = {len(overall_best)})")
    return overall_best.tolist()
