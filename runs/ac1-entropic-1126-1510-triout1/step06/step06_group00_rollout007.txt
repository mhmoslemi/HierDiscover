"""

Strategy
--------

   limit value 2·sup(g) ≈ 1.5.  As the length grows the discrete version
   approaches that limit, therefore we first try several lengths up to

   * the arcsine shape (the known near‑optimal distribution),

   * a simple exponential fall‑off, the uniform distribution and a few
     convex mixtures of arcsine with uniform.
   All candidates are normalised to ℓ₁‑norm = 1 and scored with the fast
   “max‑convolution” proxy 2·n·max(a*a).

3. **Selection + refinement** – the best few seeds are refined with two
   mirror‑descent schemes that use the exact sub‑gradient of the max‑convolution:
   * exponentiated‑gradient (KL‑mirror descent),
   * Adam‑style adaptive moment method.
   The number of iterations is limited only by the remaining time budget.

4. **Local random‑transfer hill climbing** – a cheap stochastic move
   (transfer a small amount of mass from one index to another) is applied
   repeatedly to the current best sequence; any improvement is kept.

The whole routine respects the supplied 1000 s budget.  The final

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) assuming sum(a)=1 (exact proxy of evaluate_sequence)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.

    Returns:
        grad   – vector of size n,
        cur_max- the current maximal convolution entry,
        active – indices where the maximum is attained (within tolerance).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    # weight vector w (1 on active positions)
    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives gradient before slicing
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (the self‑pair counted only once)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 200000,
                          eta0: float = 0.6,
                          decay: float = 0.9999):
    """Mirror descent with KL‑divergence (exponentiated‑gradient)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)          # stay non‑negative
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 200000,
                  step0: float = 0.12,
                  beta1: float = 0.9,
                  beta2: float = 0.999):
    """Adam‑style adaptive moment optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law – the classic near‑optimal shape."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on {0,…,n‑1} (continuous analogue)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    """A centred exponential tail – useful for extra exploration."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _uniform_distribution(n: int) -> np.ndarray:
    """Plain uniform distribution (baseline)."""
    return np.full(n, 1.0 / n)


def _mix_distributions(a: np.ndarray, b: np.ndarray, lam: float) -> np.ndarray:
    """Convex mix λ·a + (1‑λ)·b, renormalised to ℓ₁‑norm = 1."""
    mix = lam * a + (1.0 - lam) * b
    mix /= mix.sum()
    return mix

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns a list of floats ready for evaluation.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # 1. Lengths to explore – geometric progression up to ~5·10⁴.
    lengths = []
    n = 512
    while n <= 50000:
        lengths.append(n)
        n = n * 2
    # also keep the size of the previous best (if any) as a fallback
    try:
        prev_len = len(height_sequence_1)   # type: ignore[name-defined]
        if prev_len not in lengths:
            lengths.append(prev_len)
    except Exception:
        pass
    lengths = sorted(set(lengths))
    print(f"[{time.time():.1f}] exploring lengths: {lengths}")

    # ------------------------------------------------------------------
    # 2. Broad sampling of analytic families.
    candidates = []      # list of tuples (array, score)

    beta_grid = np.linspace(0.30, 0.70, 21)   # fine grid around the known optimum
    mix_lambdas = [0.6, 0.7, 0.8, 0.9]

    for n in lengths:
        if time.time() >= deadline:
            break

        # arcsine (the champion)
        a = _arcsine_distribution(n)
        sc = _max_convolution_score(a)
        candidates.append((a, sc))
        print(f"[{time.time():.1f}] n={n:5d} arcsine               score={sc:.6f}")

        # uniform (reference)
        u = _uniform_distribution(n)
        sc = _max_convolution_score(u)
        candidates.append((u, sc))

        # exponential tail
        e = _exponential_distribution(n, lam=0.005)
        sc = _max_convolution_score(e)
        candidates.append((e, sc))

        # symmetric beta family
        for p in beta_grid:
            if time.time() >= deadline:
                break
            a = _beta_symmetric_distribution(n, p)
            sc = _max_convolution_score(a)
            candidates.append((a, sc))
        # mixtures of arcsine & uniform
        for lam in mix_lambdas:
            mix = _mix_distributions(_arcsine_distribution(n),
                                     _uniform_distribution(n), lam)
            sc = _max_convolution_score(mix)
            candidates.append((mix, sc))

        # a few random Dirichlet points (they often discover surprising shapes)
        for _ in range(2):
            if time.time() >= deadline:
                break
            rand = np.random.gamma(1.0, 1.0, size=n)
            rand /= rand.sum()
            sc = _max_convolution_score(rand)
            candidates.append((rand, sc))

    # ------------------------------------------------------------------
    # 3. Keep the best few seeds for intensive refinement.
    candidates.sort(key=lambda x: x[1])
    top_k = min(12, len(candidates))
    top_candidates = candidates[:top_k]
    print(f"[{time.time():.1f}] selected {top_k} seeds for refinement "
          f"(best score {top_candidates[0][1]:.6f})")

    best_seq = top_candidates[0][0].copy()
    best_score = top_candidates[0][1]

    # ------------------------------------------------------------------
    # 4. Refinement phase – exponentiated‑gradient + Adam on each seed.
    for idx, (seed_seq, seed_score) in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        n = seed_seq.shape[0]

        # a) Exponentiated‑gradient
        seq, sc = _exp_gradient_descent(seed_seq, deadline,
                                         max_iters=80000,
                                         eta0=0.6,
                                         decay=0.9999)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} exp‑grad improved to {sc:.6f}")

        # b) Adam
        seq, sc = _adam_descent(seed_seq, deadline,
                                 max_iters=80000,
                                 step0=0.12)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} Adam improved to {sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Local random‑transfer hill climbing on the current best.
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    iteration = 0
    max_transfer = 0.06                     # up to 6 % of total mass per trial

    while time.time() < deadline:
        iteration += 1
        # geometrically decay the transferred amount – gives fine tweaks later
        transfer = max_transfer * (0.5 ** (iteration / 12000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} new best {best_score:.6f}")

        # occasional jitter to escape plateaus
        if iteration % 4000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 1e-4
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] final best score {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
