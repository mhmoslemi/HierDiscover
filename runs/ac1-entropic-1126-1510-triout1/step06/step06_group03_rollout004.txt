"""
Propose a step‐function (non‑negative vector) that minimizes
    score = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm improves on the previous baseline by
* employing many Adam restarts with longer runs,
* using a warm‑start from the previous best via linear interpolation,
* adding a cheap stochastic local‑search (pairwise mass transfers) after each
  length optimisation,
* keeping everything on the simplex (sum = 1, a_i ≥ 0).

The whole search respects a 1000 s budget and returns a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of max_k (a*a)_k.
    Returns (grad, cur_max), where grad has the same shape as a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(conv >= cur_max - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals

def _raw_score(a: np.ndarray) -> float:
    """2·n·max(a*a) for a already normalised (sum = 1)."""
    n = a.shape[0]
    return 2.0 * n * _fft_self_convolve_max(a)

def _adam_optimize_from_logits(
    x0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 6000,
    lr0: float = 0.15,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the logit parametrisation (soft‑max → simplex).
    Returns best a (numpy array) and its raw score.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    best_a = None
    best_score = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        # soft‑max (stable)
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # current objective
        cur_score = _raw_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it}  best_raw={best_score:.6f}")

        # sub‑gradient of max
        grad_a, _ = _max_subgradient_step(a)

        # chain rule through soft‑max
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # jitter every 800 iterations
        if it % 800 == 0:
            x += np.random.randn(*x.shape) * 0.01

    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _raw_score(best_a)
    return best_a, best_score


# ----------------------------------------------------------------------
def _optimise_length(
    n: int,
    deadline: float,
    prev_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length n.
    Several initialisations (warm‑start, uniform, random Dirichlet) are tried.
    Returns the best sequence found and its raw score.
    """
    init_logits = []

    # 1) warm‑start from previous best (interpolated)
    if prev_seq is not None:
        cand = _interpolate_to_length(prev_seq, n)
        eps = 1e-12
        init_logits.append(np.log(cand + eps))

    # 2) uniform start
    init_logits.append(np.zeros(n, dtype=np.float64))

    # 3) random Dirichlet draws – number depends on size
    n_random = 5 if n <= 4000 else 2
    for _ in range(n_random):
        alpha = np.random.rand() * 4.0 + 0.1
        sample = np.random.gamma(alpha, 1.0, size=n)
        sample /= sample.sum()
        init_logits.append(np.log(sample + 1e-12))

    best_seq = None
    best_score = float("inf")
    for ix, x0 in enumerate(init_logits):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] length {n}, start {ix + 1}/{len(init_logits)}")
        seq, score = _adam_optimize_from_logits(
            x0, n, deadline, max_iters=6000, lr0=0.15
        )
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    # safety fallback
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _raw_score(best_seq)
    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_search(a: np.ndarray, cur_score: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climbing: move a tiny amount of mass from a random index
    to another while staying on the simplex. Accept only improving moves.
    Runs until the deadline is reached.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_score
    # typical magnitude of one coordinate (≈1/n)
    step = 0.6 / n
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = step * np.random.rand()
        if delta > best_a[j]:
            delta = best_a[j]
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] += delta
        cand[j] -= delta
        # evaluate
        new_score = _raw_score(cand)
        if new_score < best_score:
            best_a = cand
            best_score = new_score
            step *= 1.03   # a tiny boost when we improve
        else:
            step *= 0.995  # slowly shrink when no progress
        # occasional perturbation to shake out of flat regions
        if it % 2000 == 0:
            idx = np.random.choice(n, size=4, replace=False)
            pert = (np.random.rand(4) - 0.5) * step
            pert -= pert.mean()
            cand2 = best_a.copy()
            cand2[idx] += pert
            cand2 = np.maximum(cand2, 0.0)
            s = cand2.sum()
            if s > 0:
                cand2 /= s
                new_score2 = _raw_score(cand2)
                if new_score2 < best_score:
                    best_a = cand2
                    best_score = new_score2
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Main entry point required by the evaluation harness.
    Returns a list of non‑negative heights that (hopefully) yields a low
    ``evaluate_sequence`` score.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Try to reuse a previously discovered construction if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        s = prev.sum()
        if s > 0.0:
            prev = prev / s
        else:
            prev = None
    except Exception:
        prev = None

    # ------------------------------------------------------------------
    # Length schedule – we start small, finish with a fairly large n
    lengths = [2000, 4000, 6000, 8000, 10000]

    best_seq = None
    best_score = float("inf")
    best_n = None

    # evaluate the supplied baseline (if any)
    if prev is not None:
        baseline_score = _raw_score(prev)
        print(f"Baseline (height_sequence_1) raw score = {baseline_score:.6f}")
        if baseline_score < best_score:
            best_score = baseline_score
            best_seq = prev.copy()
            best_n = len(prev)

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length {n} ===")
        seq, score = _optimise_length(n, deadline, prev_seq=best_seq)
        print(f"  length {n} : best raw score = {score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            best_n = n
        # local stochastic refinement – allocate a short slice of remaining time
        remaining = deadline - time.time()
        if remaining > 5.0:  # at least a few seconds left
            ls_deadline = time.time() + min(30.0, remaining / 2.0)
            seq2, score2 = _local_search(best_seq, best_score, ls_deadline)
            print(f"  after local search raw score = {score2:.6f}")
            if score2 < best_score:
                best_score = score2
                best_seq = seq2.copy()
        # warm‑start for the next length
        prev = best_seq

    # ------------------------------------------------------------------
    # Final refined stochastic search on the best length we have so far
    if best_seq is not None and time.time() < deadline:
        print("\n=== Final refinement on best length ===")
        final_seq, final_score = _local_search(best_seq, best_score, deadline)
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
        print(f"Final refined raw score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the sequence (list) – the evaluation harness will call evaluate_sequence
    final_list = best_seq.tolist()
    try:
        final_eval = evaluate_sequence(final_list)  # type: ignore[name-defined]
        print(f"\nFinal evaluated score: {final_eval:.6f} (length = {len(final_list)})")
    except Exception as exc:
        print(f"Failed to evaluate final sequence: {exc}")

    return final_list
