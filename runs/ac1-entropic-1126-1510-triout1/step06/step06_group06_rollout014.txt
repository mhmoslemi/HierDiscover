def _random_two_level_search(n, n_samples, deadline):
    best_seq = None; best_score = inf
    for _ in range(n_samples):
        if time.time() > deadline: break
        L = np.random.randint(1, n)  # front block size
        r = np.exp(np.random.uniform(0, np.log(200)))  # log-uniform
        denom = L * r + (n - L)
        q = 1.0 / denom
        p = r * q
        seq = np.empty(n, dtype=np.float64)
        seq[:L] = p; seq[L:] = q
        # quick evaluate
        b = _fft_self_convolve(seq)
        cur_max = b.max()
        score = 2.0 * n * cur_max
        if score < best_score:
            best_score = score; best_seq = seq.copy()
    return best_seq, best_score

def _anneal_local(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max
    T0 = (cur_max * 0.1)  # starting temperature relative
    T = T0
    it = 0
    while time.time() < deadline:
        it += 1
        # choose a source index weighted by product contributions to current max
        # compute indices participating in the maximum conv terms
        max_idx = np.random.choice(np.where(b >= cur_max - 1e-12)[0])
        i_low = max(0, max_idx - (n - 1))
        i_high = min(max_idx, n - 1)
        i_vals = np.arange(i_low, i_high + 1, dtype=int)
        prod = a[i_vals] * a[max_idx - i_vals]
        if prod.sum() > 0:
            src = np.random.choice(i_vals, p=prod / prod.sum())
        else:
            src = np.random.randint(n)
        dst = np.random.randint(n)
        if dst == src:
            continue
        max_delta = min(a[src], 0.1)  # allow bigger moves
        delta = np.random.rand() * max_delta
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        b_new = _fft_self_convolve(a_new)
        new_max = b_new.max()
        dE = new_max - cur_max
        if dE <= 0 or np.random.rand() < np.exp(-dE / max(T, 1e-12)):
            a = a_new
            b = b_new
            cur_max = new_max

"""
Hybrid search for low‑score step functions.

The routine proceeds in four stages:

0.  Initialise many simple families (arcsine, Beta, exponential,
    uniform, mixtures) for several lengths, keep the best fast‑score
    (2·n·max(conv)) candidate.

1.  Random two‑level (front / back) families are sampled densely
    (log‑uniform ratios, linear break positions) and may replace the
    current best.

2.  Adam optimisation on the smooth‑max surrogate of the ℓ∞
    convolution is performed from several jittered restarts.
    Hyper‑parameters are a little more aggressive than the reference
    implementation.

3.  Exact ℓ∞ sub‑gradient descent is run for a while, followed by a
    simulated‑annealing mass‑transfer local search that accepts
    occasional uphill moves.  The best vector found is returned.

All vectors are projected onto the probability simplex after every
update, guaranteeing non‑negative coefficients that sum to 1.
The algorithm respects the overall time budget (default 1000 s) and
uses only NumPy (no external optimisation packages).
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _arcsine_vec(n: int) -> np.ndarray:
    """U‑shaped arcsine family a_i ∝ 1/√((i+½)(n‑i‑½))."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_vec(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric Beta(α,β) density discretised on the centre of each cell."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use log‑space to avoid overflow at the boundaries
    log_vals = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    # replace nan / inf that arise from the log at the boundaries
    bad = ~np.isfinite(vals)
    if np.any(bad):
        vals[bad] = 0.0
    if vals.sum() == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _exp_vec(n: int, lam: float) -> np.ndarray:
    """Exponential decay a_i ∝ exp(‑lam·i)."""
    v = np.exp(-lam * np.arange(n, dtype=np.float64))
    v /= v.sum()
    return v


# ----------------------------------------------------------------------
def _mix_vectors(v1: np.ndarray, v2: np.ndarray, w: float) -> np.ndarray:
    """Convex mixture (1‑w)·v1 + w·v2, re‑projected onto the simplex."""
    mixed = (1.0 - w) * v1 + w * v2
    return _project_onto_simplex(mixed, s=1.0)


# ----------------------------------------------------------------------
def _random_two_level_search(n: int, n_samples: int, deadline: float):
    """Random search in the two‑level family (front block height p,
    back block height q) with log‑uniform ratios."""
    best_seq = None
    best_score = float("inf")
    for _ in range(n_samples):
        if time.time() > deadline:
            break
        L = np.random.randint(1, n)                # size of the front block
        r = np.exp(np.random.uniform(0.0, math.log(200)))   # p = r·q, 1 … 200
        denom = L * r + (n - L)
        q = 1.0 / denom
        p = r * q
        seq = np.empty(n, dtype=np.float64)
        seq[:L] = p
        seq[L:] = q
        cur_max = _fft_self_convolve(seq).max()
        score = 2.0 * n * cur_max
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    return best_seq, best_score


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """Gradient of the smooth‑max surrogate.
    Returns (grad, current maximum of the convolution)."""
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # stable soft‑max weights
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    # convolution of reversed a with weight vector
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3·n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.02,
                   alpha_max: float = 8000.0,
                   lr0: float = 0.5,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """Adam optimisation of the smooth‑max surrogate."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step-size (slow decay)
        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        # continuation on the smoothness parameter
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny jitter to escape plateaus
        if t % 400 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """Exact ℓ∞ sub‑gradient step (equal weighting of all active indices)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Sub‑gradient descent on the true max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.02 / np.sqrt(it)          # diminishing step
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        # occasional jitter
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


# ----------------------------------------------------------------------
def _anneal_local(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simulated‑annealing mass‑transfer search.
    Moves a random fraction of mass from a source index (chosen
    according to its contribution to the current peak) to a random
    destination.  Uphill moves are occasionally accepted based on a
    temperature schedule.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    # initial temperature – a small fraction of the current peak
    T = max(1e-4, cur_max * 0.02)

    it = 0
    while time.time() < deadline:
        it += 1

        # pick an index that belongs to a maximal convolution term
        max_idx = np.random.choice(np.where(b >= cur_max - 1e-12)[0])
        i_low = max(0, max_idx - (n - 1))
        i_high = min(max_idx, n - 1)
        i_vals = np.arange(i_low, i_high + 1, dtype=int)

        prod = a[i_vals] * a[max_idx - i_vals]
        if prod.sum() > 0:
            src = np.random.choice(i_vals, p=prod / prod.sum())
        else:
            src = np.random.randint(n)

        dst = np.random.randint(n)
        if dst == src:
            continue

        # move a non‑trivial fraction of the source mass
        max_delta = min(a[src], 0.1)          # up to 10 % of src
        delta = np.random.rand() * max_delta

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        b_new = _fft_self_convolve(a_new)
        new_max = b_new.max()

        dE = new_max - cur_max
        if dE <= 0.0 or np.random.rand() < math.exp(-dE / max(T, 1e-12)):
            a = a_new
            b = b_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()

        # geometric cooling
        T *= 0.9995

        # occasional small random jitter to keep exploration alive
        if np.random.rand() < 0.001:
            a += np.random.randn(n) * 1e-7
            a = _project_onto_simplex(a, s=1.0)
            b = _fft_self_convolve(a)
            cur_max = b.max()

    return best_a
