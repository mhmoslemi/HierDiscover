"""
Propose a step‑function that minimizes ``evaluate_sequence``.

Algorithm – multi‑scale stochastic + first‑order optimisation
------------------------------------------------------------
1. Choose a handful of problem sizes (≈ 3000‑16000).  Larger ``n`` gives a
   better asymptotic constant but costs more per iteration.
2. For each size build a *large* pool of analytic candidates:
   * Beta(α,β) densities with α,β ranging from 0.02 to 10 (log‑grid).
   * Exponential shapes with several decay rates.
   * Uniform, a random Dirichlet draw and, if available, the previous
     construction (interpolated to the current length).
3. Keep the best few (≤ 5) candidates per size (cheap score =
   ``2·n·max(conv)`` – exact when the vector sums to 1) and refine them
   with a **projected Adam optimiser on the simplex**:
   * Variables are the unconstrained log‑probabilities ``z = log(a)``.
   * The exact sub‑gradient of the objective (the gradient of the maximum
     convolution entry) is obtained by a fast FFT‑based routine.
   * Adam’s adaptive step‑sizes provide fast, stable convergence without
     hand‑tuned schedules.
4. After the inner optimisation evaluate the best local vector with the
   real ``evaluate_sequence`` – this is the only expensive call.
5. Perform a final *random hill‑climbing* phase on the overall best
   solution (small zero‑mean perturbations, projection onto the simplex)
   to escape possible local minima.
6. Return the best sequence found before the time budget expires.

The whole routine respects the given time limit, prints progress messages
and never uses closures or lambdas – all helpers are top‑level functions.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    a = np.clip(x, eps, 1.0 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1.0 - eps) ** (beta - 1.0)
    vals = a * b
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float = 1.0) -> np.ndarray:
    """Geometric‑like decreasing shape: a_i ∝ exp(-lam·i/(n‑1))."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * i / max(1, n - 1))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the true max, i.e. average over all active indices.
    Returns ∂/∂a of  max(conv) .
    """
    n = a.shape[0]
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    # convolution of reversed a with the weight vector w
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (remove double counting of a_i*a_i for even shifts)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad


# ----------------------------------------------------------------------
def _adam_step(z: np.ndarray,
               g: np.ndarray,
               m: np.ndarray,
               v: np.ndarray,
               t: int,
               lr: float = 0.5,
               beta1: float = 0.9,
               beta2: float = 0.999,
               eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Adam update on the unconstrained variable z."""
    m = beta1 * m + (1.0 - beta1) * g
    v = beta2 * v + (1.0 - beta2) * (g * g)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z, m, v

    return a

            overall_best_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        except Exception:
            n_fallback = 5000
            overall_best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())
        print("Fallback used.")

    print(
        f"Final best score: {overall_best_score:.6f} "
        f"(length = {len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
