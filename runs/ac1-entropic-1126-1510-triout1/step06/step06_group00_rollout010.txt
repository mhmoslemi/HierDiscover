def _smooth_max_gradient_descent(a0, deadline, max_iters=50000, eta0=0.3, tau0=1.0, tau_mult=1.001, tau_max=500):
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    n = len(a)
    tau = tau0
    it=0
    while time.time() < deadline and it < max_iters:
        it+=1
        # compute convolution b
        b = _fft_self_convolve(a)
        cur_max = b.max()
        # compute softmax weights w
        shifted = b - cur_max  # for numeric stability (max of b)
        w = np.exp(tau * shifted)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(b) / b.size
        else:
            w /= w_sum
        # gradient
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2 * conv_rev[n-1:n-1+n]
        # diagonal correction
        diag_idx = 2 * np.arange(n)
        grad -= a * w[diag_idx]
        # update via exponentiated gradient
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        # normalize
        sum_a = a.sum()
        if sum_a == 0:
            a = np.full_like(a, 1.0/n)
        else:
            a /= sum_a
        # evaluate score
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} smooth grad iter {it} score {best_score:.6f} tau {tau:.3f}")
        # update tau
        if tau < tau_max:
            tau *= tau_mult
    return best_a, best_score

"""
Propose a non‑negative step function with a very low evaluation score.

Main ideas
----------
1. **Smooth‑max gradient descent** – instead of the sub‑gradient of the
   hard max, we minimise a differentiable approximation
   `smooth_max(b) = (1/τ)·log Σ exp(τ·b_k)`.  Its gradient is a weighted
   convolution where the weights are the soft‑max of the convolution
   values.  By slowly increasing the temperature τ we follow a continuation
   path from the average‐convolution optimum (uniform distribution) to the
   true sup‑norm optimum.

2. **Exponentiated‑gradient updates** – keep the iterates on the simplex
   (non‑negative and sum‑to‑one) without an expensive projection step.
   After each multiplicative update the vector is renormalised.

3. **Multi‑scale exploration** – several sequence lengths are tried.
   For each length a handful of analytic families (arcsine, beta,
   triangular, exponential) and a few random Dirichlet draws are used as
   starting points.

4. **Local hill‑climbing** – a cheap random‑transfer move is applied to the
   current best sequence until the time budget expires.

The whole routine respects the given 1000 s budget and always returns a
list of non‑negative floats that can be fed directly to `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """
    Score for a probability vector (sum=1):
    2·n·max(a*a).  This equals ``evaluate_sequence`` up to the
    scale‑invariant factor.
    """
    n = a.shape[0]
    return 2.0 * n * _fft_self_convolve(a).max()


# ----------------------------------------------------------------------
def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Gamma‑Dirichlet, α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _smooth_max_gradient_descent(a0: np.ndarray,
                                 deadline: float,
                                 max_iters: int = 20000,
                                 eta0: float = 0.5,
                                 tau0: float = 0.5,
                                 tau_mult: float = 1.001,
                                 tau_max: float = 200.0):
    """
    Continuation method that minimises a smooth approximation of
    ``max(a*a)``.  The temperature τ starts at ``tau0`` and is multiplied
    by ``tau_mult`` after each step, never exceeding ``tau_max``.
    An exponentiated‑gradient update keeps the iterate on the probability
    simplex.
    Returns the best sequence found and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    tau = tau0
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        # convolution of a with itself
        b = _fft_self_convolve(a)
        cur_max = b.max()

        # soft‑max weights (numerically stable)
        shifted = b - cur_max                     # max becomes 0
        w = np.exp(tau * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth max (see analysis)
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # diagonal correction for the i=j terms
        diag_idx = 2 * np.arange(n)                # indices 0,2,4,...,2n-2
        grad -= a * w[diag_idx]

        # exponentiated‑gradient step (clip to avoid overflow)
        eta = eta0 / np.sqrt(it)
        step = -eta * grad
        step = np.clip(step, -50, 50)              # safety
        a = a * np.exp(step)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # score (exact, no temperature)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} iter={it} τ={tau:.3f} "
                  f"score={best_score:.6f}")

        # increase temperature gradually
        if tau < tau_max:
            tau *= tau_mult

    return best_a, best_score
