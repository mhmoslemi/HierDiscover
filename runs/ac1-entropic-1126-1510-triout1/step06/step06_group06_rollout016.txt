"""
Improved step‑function search.

The algorithm builds on the previous four‑stage pipeline but adds
several enhancements:

* a richer family of Beta‑type initialisations (including α<0.5);
* a *symmetric two‑level* parametric family (high blocks at both ends);
* multi‑start Adam optimisation (the best candidate and a random start);
* a larger time budget for the exact sub‑gradient phase;
* a slightly more aggressive step‑size schedule for sub‑gradient descent.

All vectors are kept on the simplex (non‑negative, sum = 1) using a
Euclidean projection.  The routine respects the global time budget
(`budget_s`) and prints intermediate progress.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x≥0, Σx=s} (Euclidean projection)."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth‑max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)

    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # soft‑max weights (stable)
    exp_term = np.exp(alpha * (b - cur_max))
    w_sum = exp_term.sum()
    w = exp_term / w_sum
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 5.0,
                   alpha_factor: float = 1.02,
                   alpha_max: float = 8000.0,
                   lr0: float = 0.3,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.

    Returns the best probability vector encountered and its exact peak.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size (slow decay)
        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # keep the best exact peak seen so far
        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        # continuation on the smoothness parameter
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional jitter to explore
        if t % 400 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """Exact ℓ∞ sub‑gradient step (equal weight on all active max entries)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Sub‑gradient descent on the exact max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        # a slightly larger step than the original code (0.05 instead of 0.02)
        step = 0.05 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        # occasional jitter
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a

def _directed_local_search(a0: np.ndarray, deadline: float):
    """
    Greedy mass‑transfer search focused on the current peak.
    Only improving moves are accepted.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        # pick a convolution index belonging to the current peak
        peak_idxs = np.where(b >= cur_max - 1e-12)[0]
        max_idx = np.random.choice(peak_idxs)

        # indices i that can contribute to (a*a)[max_idx]
        i_low = max(0, max_idx - (n - 1))
        i_high = min(max_idx, n - 1)
        i_vals = np.arange(i_low, i_high + 1, dtype=int)

        prod = a[i_vals] * a[max_idx - i_vals]
        if prod.sum() == 0.0:
            src = np.random.randint(n)
        else:
            prob = prod / prod.sum()
            src = np.random.choice(i_vals, p=prob)

        # destination – try a few candidates and keep the best improvement
        best_candidate = None
        best_new_max = cur_max
        for _ in range(3):
            dst = np.random.randint(n)
            if dst == src:
                continue
            max_delta = min(a[src], 0.07)            # up to 7 % of source mass
            delta = np.random.rand() * max_delta

            a_new = a.copy()
            a_new[src] -= delta
            a_new[dst] += delta
            b_new = _fft_self_convolve(a_new)
            new_max = b_new.max()
            if new_max < best_new_max - 1e-15:
                best_new_max = new_max
                best_candidate = (a_new, b_new)

        if best_candidate is not None:
            a, b = best_candidate
            cur_max = best_new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

        # tiny random jitter
        if np.random.rand() < 0.005:
            a += np.random.randn(n) * 1e-7
            a = _project_onto_simplex(a, s=1.0)
            b = _fft_self_convolve(a)
            cur_max = b.max()

    return best_a

def _two_level_family(n: int, deadline: float):
    """
    Simple (asymmetric) two‑level family:
        a_i = p for i < L   (front block)
        a_i = q otherwise   (back block)
    """
    best_seq = None
    best_score = float('inf')
    step_L = max(1, n // 200)          # ~0.5 % steps
    r_grid = np.logspace(0.0, 2.5, num=30)   # 1 … ≈ 300

    for L in range(step_L, n, step_L):
        if time.time() > deadline:
            break
        denom_base = n - L
        for r in r_grid:
            denom = L * r + denom_base
            q = 1.0 / denom
            p = r * q
            seq = np.empty(n, dtype=np.float64)
            seq[:L] = p
            seq[L:] = q
            peak = _fft_self_convolve(seq).max()
            score = 2.0 * n * peak
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
    return best_seq, best_score

def _two_level_symmetric_family(n: int, deadline: float):
    """
    Symmetric two‑level family:
        a_i = p   for i < L  or i ≥ n‑L   (both edges)
        a_i = q   otherwise
    """
    best_seq = None
    best_score = float('inf')
    step_L = max(1, n // 200)          # ~0.5 % steps
    r_grid = np.logspace(0.0, 2.5, num=30)   # 1 … ≈ 300

    max_L = n // 2
    for L in range(step_L, max_L, step_L):
        if time.time() > deadline:
            break
        denom_base = n - 2 * L
        for r in r_grid:
            denom = 2 * L * r + denom_base
            q = 1.0 / denom
            p = r * q
            seq = np.full(n, q, dtype=np.float64)
            seq[:L] = p
            seq[-L:] = p
            peak = _fft_self_convolve(seq).max()
            score = 2.0 * n * peak
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Main entry point for the competition framework.
    Returns a list of non‑negative heights (sum is 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # --------------------------------------------------------------
    # Load a previously discovered construction if it exists.
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_seq = _project_onto_simplex(base_seq, s=1.0)
        base_n = len(base_seq)
    except Exception:
        base_seq = None
        base_n = 4000

    # --------------------------------------------------------------
    # Lengths to explore – centre around the previous best.
    lengths = [max(500, int(base_n * f))
               for f in (0.6, 0.8, 1.0, 1.2, 1.4, 1.6) if int(base_n * f) >= 500]
    if not lengths:
        lengths = [max(500, base_n)]

    overall_best_seq = None
    overall_best_score = float('inf')

    # --------------------------------------------------------------
    # Phase 0 – Broad sampling of families (more Beta params added)
    print("[phase 0] broad family sampling")
    beta_alphas = np.logspace(np.log10(0.05), np.log10(0.5), num=15)  # include α<0.5
    for n in lengths:
        if time.time() >= deadline:
            break

        starters = {}

        # arcsine (β=0.5) – kept for reference
        starters["arcsine"] = (1.0 / np.sqrt((np.arange(n) + 0.5) *
                                            (n - np.arange(n) - 0.5)))
        starters["arcsine"] /= starters["arcsine"].sum()

        starters["uniform"] = np.full(n, 1.0 / n, dtype=np.float64)

        starters["random1"] = np.random.rand(n)
        starters["random1"] /= starters["random1"].sum()

        starters["random2"] = np.random.rand(n)
        starters["random2"] /= starters["random2"].sum()

        # symmetric Beta families (α = β)
        for a in beta_alphas:
            x = (np.arange(n, dtype=np.float64) + 0.5) / n
            # log‑density to avoid overflow/underflow
            log_vals = (a - 1.0) * (np.log(x) + np.log(1.0 - x))
            log_vals -= np.max(log_vals)          # shift for stability
            vals = np.exp(log_vals)
            vals /= vals.sum()
            starters[f"beta_{a:.3f}"] = vals

        # exponential decay families
        for lam in (0.0003, 0.0006, 0.001):
            vec = np.exp(-lam * np.arange(n))
            vec /= vec.sum()
            starters[f"exp_{lam:.4f}"] = vec

        # mixtures of arcsine and uniform
        for w in (0.2, 0.5, 0.8):
            mix = (1.0 - w) * starters["arcsine"] + w * starters["uniform"]
            starters[f"mix_{w:.1f}"] = mix

        if base_seq is not None:
            # linear interpolation of the previous best to the new length
            old_n = len(base_seq)
            x_old = np.linspace(0.0, 1.0, old_n)
            x_new = np.linspace(0.0, 1.0, n)
            interp = np.interp(x_new, x_old, base_seq)
            interp = np.maximum(0.0, interp)
            interp /= interp.sum()
            starters["prev"] = interp

        # evaluate the fast score of every starter
        for tag, vec in starters.items():
            vec = _project_onto_simplex(vec, s=1.0)
            peak = _fft_self_convolve(vec).max()
            fast_score = 2.0 * n * peak
            if fast_score < overall_best_score:
                overall_best_score = fast_score
                overall_best_seq = vec.copy()
                print(f"[init] n={n:4d} {tag:<12s} fast_score={fast_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Two‑level families (asymmetric + symmetric)
    print("[phase 1] asymmetric two‑level search")
    for n in lengths:
        if time.time() >= deadline:
            break
        seq, score = _two_level_family(n, deadline)
        if seq is not None and score < overall_best_score:
            overall_best_score = score
            overall_best_seq = seq.copy()
            print(f"[asym two‑level] n={n:4d} fast_score={score:.6f}")

    print("[phase 1b] symmetric two‑level search")
    for n in lengths:
        if time.time() >= deadline:
            break
        seq, score = _two_level_symmetric_family(n, deadline)
        if seq is not None and score < overall_best_score:
            overall_best_score = score
            overall_best_seq = seq.copy()
            print(f"[sym two‑level] n={n:4d} fast_score={score:.6f}")

    # fallback if nothing was produced
    if overall_best_seq is None:
        n = max(500, base_n)
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print("[fallback] using uniform distribution")

    # --------------------------------------------------------------
    # Phase 2 – Multi‑start Adam optimisation
    time_left = deadline - time.time()
    if time_left > 8.0:
        n = len(overall_best_seq)
        print(f"[phase 2] Adam optimisation (n={n}) – time left {time_left:.1f}s")
        # allocate half of the remaining budget for Adam, split over 2 starts
        adam_budget = time_left * 0.5
        starts = [overall_best_seq]
        # a random start to diversify
        rnd = np.random.rand(n)
        rnd /= rnd.sum()
        starts.append(rnd)

        best_adam_seq = overall_best_seq
        best_adam_peak = _fft_self_convolve(overall_best_seq).max()

        for start_vec in starts:
            sub_deadline = time.time() + adam_budget / len(starts) - 0.2
            a_opt, _ = _adam_optimize(start_vec, sub_deadline)
            a_opt = _project_onto_simplex(a_opt, s=1.0)
            cur_peak = _fft_self_convolve(a_opt).max()
            if cur_peak < best_adam_peak - 1e-15:
                best_adam_peak = cur_peak
                best_adam_seq = a_opt.copy()
                print(f"[adam] improved peak={cur_peak:.6f}")

        overall_best_seq = best_adam_seq.copy()
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print(f"[phase 2] after Adam fast_score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Exact sub‑gradient descent + aggressive local search
    time_left = deadline - time.time()
    if time_left > 6.0:
        n = len(overall_best_seq)
        print(f"[phase 3] sub‑gradient & directed search (n={n}) – time left {time_left:.1f}s")
        # give ~60 % of remaining time to sub‑gradient descent
        sub_end = time.time() + time_left * 0.60
        a_ref = _refine_max(overall_best_seq, sub_end)

        # remaining time for the greedy local moves
        local_end = deadline
        a_final = _directed_local_search(a_ref, local_end)

        overall_best_seq = a_final.copy()
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print(f"[phase 3] after refinement fast_score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # Final black‑box evaluation
    try:
        true_score = evaluate_sequence(overall_best_seq.tolist())
        print(f"[final] evaluate_sequence score={true_score:.6f}")
        if true_score < overall_best_score:
            overall_best_score = true_score
    except Exception as exc:
        print("[final] evaluate_sequence raised:", exc)

    print(f"Best reported score {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
