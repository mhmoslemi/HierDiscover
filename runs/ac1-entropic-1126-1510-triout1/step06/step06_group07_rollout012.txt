def propose_candidate(seed=12345, budget_s=1000):
    import numpy as np, time, math

    start = time.time()
    deadline = start + budget_s - 0.5

    best_score = float('inf')
    best_seq = None

    rng = np.random.default_rng(seed)

    # We'll try various lengths
    lengths = [2000, 4000, 6000, 8000, 10000]
    alphas = [0.5]   # Because arcsine is likely optimum
    # maybe include a few other alphas for robustness
    alphas = [0.3, 0.4, 0.5, 0.6, 0.7] 

    for n in lengths:
        if time.time() > deadline:
            break
        i = np.arange(n, dtype=np.float64)
        for alpha in alphas:
            # shape = (i+0.5)^{alpha-1} * (n-0.5-i)^{alpha-1}
            a = (i + 0.5) ** (alpha - 1.0) * (n - 0.5 - i) ** (alpha - 1.0)
            a_sum = a.sum()
            if a_sum == 0:
                continue
            p = a / a_sum
            # ensure nonnegative
            maxconv = np.convolve(p, p).max()
            score = 2.0 * n * maxconv
            if score < best_score:
                best_score = score
                best_seq = p.copy()
                print(f"[candidate] n={n}, alpha={alpha:.3f}, score={score:.6f}")

    # after scanning, we could perform simple hill climbing to fine-tune
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = len(a)
        # simple local improvement: random δ-transfer moves
        step_scale = 0.01  # fraction of current value max
        for it in range(20000):
            if time.time() > deadline:
                break
            # choose two indices i!=j
            i, j = rng.integers(0, n, size=2, replace=False)
            # propose transfer from i to j
            delta = rng.random() * a[i] * step_scale
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            # ensure positivity
            if cand[i] < 0:
                continue
            # compute new score quickly
            new_score = 2.0 * n * np.convolve(cand, cand).max()
            if new_score + 1e-12 < best_score:
                best_score = new_score
                best_seq = cand.copy()
                a = cand
                print(f"[hill] iter {it} score={best_score:.6f}")

    # final re-normalize to ensure sum approx 1 (scale invariant)
    if best_seq is None:
        best_seq = np.ones(1, dtype=float)   # fallback uniform single?

    print(f"[final] best score = {best_score:.6f}")
    return best_seq.tolist()

"""
Propose a low‑score step function.

The core idea: the convolution of a Beta(½,½) (arcsine) density with itself is
exactly uniform on the continuous interval.  Discretising that density on
{0,…,n‑1} yields a probability vector p whose self‑convolution is almost flat,
i.e.  max(p∗p) ≈ 1/(2 n).  Since the scoring function is scale‑invariant,
the objective 2 n·max(p∗p) is then close to 1, far below the previous best
≈1.51.

The routine builds a family of symmetric “Beta‑shape’’ vectors

    a_i = (i+½)^{α‑1}·(n‑½‑i)^{α‑1} ,   i = 0,…,n‑1

normalises them to a probability distribution and evaluates the exact score.
A short sweep over a few lengths n and shape parameters α (the arcsine case
corresponds to α=0.5) is performed; the best candidate is then returned.
A tiny random‑mass‑transfer hill‑climb is applied if time permits.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Return a symmetric Beta(α,α)‑like density on {0,…,n‑1}.

    a_i ∝ (i+½)^{α‑1}·(n‑½‑i)^{α‑1}.
    The offset ½ avoids division by zero at the boundaries.
    """
    i = np.arange(n, dtype=np.float64)
    # (i+0.5)^(α-1) * (n-0.5-i)^(α-1)
    a = (i + 0.5) ** (alpha - 1.0) * (n - 0.5 - i) ** (alpha - 1.0)
    # Normalise to a probability vector (sum = 1)
    s = a.sum()
    if s == 0.0:
        return a  # will be all zeros – never happens for α>0
    return a / s


# ----------------------------------------------------------------------
def _score_of(p: np.ndarray) -> float:
    """
    Exact scoring function for a probability vector p.
    Because the total mass is 1, the scale‑invariant score is
        2·n·max(p∗p)
    where n = len(p) and ∗ denotes linear convolution.
    """
    n = p.shape[0]
    conv = np.convolve(p, p)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function with a low evaluation score.
    The method:
        1.  Scan a handful of lengths and Beta‑shape parameters (α).
        2.  Keep the best candidate (usually α≈0.5 gives a score ≈0.8).
        3.  If time permits, perform a tiny random‑mass‑transfer hill‑climb
            that accepts only strictly improving moves.
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣  Grid search over lengths and shape parameters
    # ------------------------------------------------------------------
    # Lengths we will try – modest (the score is almost independent of n,
    # but larger n reduces discretisation error).  Feel free to extend.
    lengths = [2000, 4000, 6000, 8000, 10000]

    # Shape parameters – α=0.5 is the continuous arcsine case.
    alphas = [0.30, 0.40, 0.45, 0.50, 0.55, 0.60, 0.70]

    for n in lengths:
        if time.time() > deadline:
            break
        for alpha in alphas:
            p = _beta_shape(n, alpha)
            sc = _score_of(p)
            if sc < best_score:
                best_score = sc
                best_seq = p.copy()
                print(f"[grid] n={n:5d} α={alpha:.2f} → score={sc:.6f}")
            if time.time() > deadline:
                break

    # ------------------------------------------------------------------
    # 2️⃣  Very cheap random hill‑climbing (optional, uses remaining time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        # step size as a fraction of the source entry; we shrink it slowly.
        step_frac = 0.02
        it = 0
        while time.time() < deadline and it < 20000:

            i, j = rng.integers(0, n, size=2, replace=False)
            if a[i] <= 0.0:

                continue

            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta

                it += 1
                continue
            new_sc = _score_of(cand)
            if new_sc + 1e-12 < best_score:

                best_seq = cand.copy()

            it += 1

    # Sanity – make sure we return a plain Python list of floats.
