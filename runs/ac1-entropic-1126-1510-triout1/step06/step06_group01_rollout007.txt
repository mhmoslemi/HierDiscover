"""
Step‑function optimiser (≈100 s per run).

The code searches for a non‑negative vector `a` (the step heights) that
minimises

    2·n·max(conv(a,a)) / (sum(a))²

The scale of `a` does not matter, so all iterates are kept on the
probability simplex (sum = 1).  The optimiser works in several phases:

1. **Broad initial sweep** – for a handful of lengths (up to 20 000)
   generate analytic families (uniform, arcsine, symmetric beta,
   triangular, exponential, sparse random) and keep the best one.

2. **Smoothed Adam descent** – minimise a smooth approximation of the
   max‑convolution using the log‑sum‑exp (soft‑max) with a continuation
   of the smoothing parameter `α` (1 → 5 → 20 → 100 → 500 → 2000).
   After each Adam step the iterate is projected onto the simplex.

3. **Exact max‑sub‑gradient descent** – switches to the true sub‑gradient
   (active set of maximal convolution entries) with momentum and a
   diminishing stepsize.

4. **Simulated annealing** – random mass transfers between two
   positions are accepted according to a temperature schedule, allowing
   the search to escape shallow local minima.

5. **Local hill climbing** – cheap deterministic moves that only keep
   improvements.

All phases respect the global time budget (default 1000 s).  The best
vector found is returned as a plain Python list of floats.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a*b using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Objective for a simplex‑normalised vector: 2·n·max(a*a)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    # w = x^(p-1) * (1-x)^(p-1); for p<1 this blows up near edges – clip.
    w = (x ** (p - 1)) * ((1 - x) ** (p - 1))
    w = np.clip(w, 0, 1e200)
    s = w.sum()
    if s == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals

    idx = np.random.choice(n, k, replace=False)
    rnd[idx] = np.random.rand(k)
    rnd /= rnd.sum()

def _smooth_descent(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Adam optimisation on a soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # schedule – larger α gives a tighter approximation of the true max
    steps_per_alpha = max(2000, int(0.08 * n))
    schedule = [(1.0, steps_per_alpha), (5.0, steps_per_alpha),
                (20.0, steps_per_alpha), (100.0, steps_per_alpha),
                (500.0, steps_per_alpha), (2000.0, steps_per_alpha)]

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score
            t += 1

            # full self‑convolution (length 2·n‑1)
            b = _fft_self_convolve(a)

            # stable soft‑max weights
            maxb = b.max()
            diff = b - maxb
            diff = np.clip(diff, -np.inf, 0.0)          # keep the max at 0
            exp_terms = np.exp(alpha * diff)
            w = exp_terms / exp_terms.sum()            # size 2·n‑1, sums to 1

            # gradient = 2 * (a_rev * w)  (central n entries)
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam update with Euclidean projection onto the simplex
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a, 1.0)

            # occasional exact evaluation
            if t % 100 == 0:
                cur_score = _max_convolution_score(a)
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_seq = a.copy()
    return best_seq, best_score


# ----------------------------------------------------------------------
def _exact_subgradient_refine(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑convolution (with momentum)."""
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)

    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps)

        w = np.zeros_like(b)
        w[active] = 1.0

        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # momentum / diminishing step
        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / np.sqrt(iteration)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad iter={iteration} new best {best_score:.6f}")

        if best_score <= 1.503:
            break
        if time.time() + 0.5 > deadline:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Random mass‑transfer search with Metropolis acceptance."""
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)
    cur_seq = a.copy()
    cur_score = best_score

    # temperature proportional to current score; will be cooled geometrically
    T = max(1e-4, 0.02 * best_score)
    decay = 0.99995
    max_transfer = 0.07
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        # amount to move – shrink slowly with iterations
        delta = np.random.rand() * min(max_transfer, cur_seq[i])
        if delta <= 0.0:
            continue

        a_new = cur_seq.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            cur_seq = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[{time.time():.1f}] anneal iter={iteration} new best {best_score:.6f}")

        T *= decay
        # gradually tighten the step size
        if iteration % 2000 == 0:
            max_transfer *= 0.98
    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_hill_climb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Simple deterministic hill climbing (only improving moves)."""
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)
    cur_score = best_score

    max_transfer = 0.07
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # exponential decay of the amount transferred
        transfer = max_transfer * (0.5 ** (iteration / 4000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[{time.time():.1f}] hill iter={iteration} improved best {best_score:.6f}")

        # occasional jitter to avoid plateaux
        if iteration % 5000 == 0 and np.random.rand() < 0.2:
            pert = np.random.randn(n) * 1e-4
            a = best_seq + pert
            a = _project_onto_simplex(a, 1.0)
            cur_score = _max_convolution_score(a)
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Main entry point.  Returns a list of non‑negative heights that (hopefully)
    yields a low value of ``evaluate_sequence``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1.  Pick a set of lengths (the larger the better, but we keep it modest)
    lengths = [2000, 4000, 8000, 12000, 16000, 20000]

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 2.  Coarse sweep over analytic families
    for n in lengths:
        if time.time() >= deadline:
            break
        for cand in _initial_candidates(n):
            if time.time() >= deadline:
                break
            cand = cand / cand.sum()
            score = _max_convolution_score(cand)
            if score < overall_best_score:
                overall_best_score = score
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] init n={n} score={score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Include a previously computed solution if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            s = _max_convolution_score(prev)
            if s < overall_best_score:
                overall_best_score = s
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={s:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4.  Fallback (should never trigger)
    if overall_best_seq is None:
        n = 8000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _max_convolution_score(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Refine the current best vector
    a = overall_best_seq.copy()
    a, score = _smooth_descent(a, deadline)

    a, score = _exact_subgradient_refine(a, deadline)

    a, score = _simulated_annealing(a, deadline)

    a, score = _local_hill_climb(a, deadline)

    # ------------------------------------------------------------------
    # 6.  Normalise once more and return
    a = a / a.sum()
    print(f"Final best score (internal estimate): {score:.6f}  length={len(a)}")
    return a.tolist()
