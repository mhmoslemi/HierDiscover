"""
Propose a step‑function (non‑negative sequence) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

where ``conv`` is the linear self‑convolution.
The implementation builds a set of analytic candidates (Beta, exponential,
sine‑shaped, previous best, …) and refines each of them with a projected
Adam optimiser on the simplex.  The optimiser works directly on the
log‑odds (softmax) parametrisation and uses the exact sub‑gradient of the
max‑convolution (no temperature schedule).  A large number of random
re‑initialisations and a generous time budget (≈ 1000 s) give the
algorithm plenty of opportunity to discover a vector with an evaluation
below the previous record (≈ 1.503).

The code respects all constraints: no I/O, only top‑level functions,
no closures or lambdas, and it calls ``evaluate_sequence`` only for the
final candidates (the cheap surrogate uses FFTs and is inexpensive).
"""

from __future__ import annotations
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT (scaled by 1/size)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]      # note: each entry is 1/size times the true value


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT (scaled by 1/size)."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float = 1.0) -> np.ndarray:
    """Geometric‑like decreasing shape: a_i ∝ exp(-lam·i/(n‑1))."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * i / max(1, n - 1))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _sine_distribution(n: int) -> np.ndarray:
    """Positive sine shape (zero at the ends, max in the middle)."""
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)          # strictly positive
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _parabolic_distribution(n: int, power: float = 1.0) -> np.ndarray:
    """Triangular‑ish shape: a_i ∝ (i·(n‑1‑i))**power."""
    i = np.arange(n, dtype=np.float64)
    vals = (i * (n - 1 - i)) ** power
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray, conv_scaled: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the true max of the (unscaled) convolution.
    ``conv_scaled`` is the result of ``_fft_self_convolve`` (i.e. true
    conv multiplied by 1/size).  The returned gradient is correct up to
    the same scale factor, which is irrelevant for optimisation.
    """
    n = a.shape[0]
    # reconstruct true convolution value from the scaled one
    size = 1 << ((2 * n - 1) - 1).bit_length()
    conv = conv_scaled * size
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)

    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (remove double counting of a_i*a_i for even shifts)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad


# ----------------------------------------------------------------------
def _adam_step(z: np.ndarray,
               g: np.ndarray,
               m: np.ndarray,
               v: np.ndarray,
               t: int,
               lr: float = 0.5,
               beta1: float = 0.9,
               beta2: float = 0.999,
               eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Adam update on the unconstrained variable ``z``."""
    m = beta1 * m + (1.0 - beta1) * g
    v = beta2 * v + (1.0 - beta2) * (g * g)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z, m, v


# ----------------------------------------------------------------------
def _initial_vector(kind: str, n: int) -> np.ndarray:
    """Feasible probability vector of length n (sum = 1)."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "arcsine":
        a = _beta_distribution(n, 0.5, 0.5)
    elif kind == "sine":
        a = _sine_distribution(n)
    elif kind == "exp":
        a = _exponential_distribution(n, lam=1.0)
    elif kind == "prev":
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.size != n:
                x_old = np.linspace(0.0, 1.0, prev.size)
                x_new = np.linspace(0.0, 1.0, n)
                a = np.interp(x_new, x_old, prev)
            else:
                a = prev.copy()
        except Exception:
            a = _beta_distribution(n, 0.5, 0.5)
    else:
        a = _beta_distribution(n, 0.5, 0.5)

    a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123456, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and returns
    the best sequence seen so far.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # base length (previous best if available)
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # a few lengths – we go up to roughly twice the base size
    length_factors = (0.75, 1.0, 1.25, 1.5, 2.0)
    lengths = sorted(
        max(3000, int(base_n * f))
        for f in length_factors
        if int(base_n * f) >= 3000 and int(base_n * f) <= 18000
    )
    if not lengths:
        lengths = [max(3000, base_n)]

    # ------------------------------------------------------------------
    # helper to obtain the true max‑convolution (fast, using the scaling factor)
    def true_max_conv(a: np.ndarray) -> float:
        conv_scaled = _fft_self_convolve(a)
        size = 1 << ((2 * a.shape[0] - 1) - 1).bit_length()
        return float(conv_scaled.max() * size)

    # ------------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Exploring length n={n}")

        # ------------------------------------------------------------------
        # build a diverse pool of analytic starters
        pool: list[np.ndarray] = []

        # systematic grid on (α,β) in log‑space
        alphas = np.logspace(-0.7, 0.7, num=7)   # ~0.2 … 5
        betas  = np.logspace(-0.7, 0.7, num=7)
        for a in alphas:
            for b in betas:
                pool.append(_beta_distribution(n, a, b))

        # extra manually‑chosen shapes
        pool.append(_beta_distribution(n, 0.2, 0.8))
        pool.append(_beta_distribution(n, 0.8, 0.2))
        pool.append(_exponential_distribution(n, lam=0.5))
        pool.append(_exponential_distribution(n, lam=1.5))
        pool.append(_sine_distribution(n))
        pool.append(_parabolic_distribution(n, power=0.5))
        pool.append(_initial_vector("prev", n))
        pool.append(_initial_vector("uniform", n))
        pool.append(_initial_vector("arcsine", n))
        pool.append(_initial_vector("sine", n))

        # ------------------------------------------------------------------
        # ranking by the cheap surrogate (2·n·max(conv_scaled) is proportional
        # to the true objective)
        def cheap_score(a: np.ndarray) -> float:
            return 2.0 * n * _fft_self_convolve(a).max()

        scored = [(cheap_score(a), a) for a in pool]
        scored.sort(key=lambda x: x[0])
        starters = [a.copy() for _, a in scored[:8]]   # keep a handful of the best

        print(f"  top cheap scores: {[s for s, _ in scored[:8]]}")

        # ------------------------------------------------------------------
        # optimisation of each starter
        for sid, a0 in enumerate(starters):
            if time.time() >= deadline:
                break

            # allocate a per‑starter budget (roughly a few dozen seconds)
            per_starter_deadline = time.time() + min(120.0,
                                                    deadline - time.time() - 5.0)
            if per_starter_deadline - time.time() < 5.0:
                continue

            # initialise Adam on the log‑odds (softmax) variables
            a = np.maximum(a0, 1e-15)   # avoid log(0)
            z = np.log(a)
            m = np.zeros_like(z)
            v = np.zeros_like(z)

            best_local_a = a.copy()
            best_local_score = true_max_conv(a) * 2.0 * n
            it = 0

            # ------------------------------------------------------------------
            while time.time() < per_starter_deadline:
                it += 1

                # current feasible probability vector
                a = np.exp(z)
                a /= a.sum()

                # cheap (but exact up to a constant factor) objective value
                cur_score = true_max_conv(a) * 2.0 * n
                if cur_score < best_local_score:
                    best_local_score = cur_score
                    best_local_a = a.copy()
                    print(
                        f"[{time.time():.1f}] n={n} starter={sid} it={it} "
                        f"score={cur_score:.6f}"
                    )
                    if best_local_score <= 1.503:
                        break

                # exact sub‑gradient of the max‑convolution
                conv_scaled = _fft_self_convolve(a)
                grad_a = _exact_gradient(a, conv_scaled)

                # gradient w.r.t. the unconstrained logits z
                dot = np.dot(a, grad_a)
                grad_z = a * (grad_a - dot)

                # Adam update
                z, m, v = _adam_step(z, grad_z, m, v, it,
                                     lr=0.5, beta1=0.9, beta2=0.999, eps=1e-8)

                # occasional tiny jitter to escape plateaus
                if it % 4000 == 0 and np.random.rand() < 0.12:
                    z += np.random.randn(n) * 1e-7

                # safety stop when we have already crossed the target
                if best_local_score <= 1.503:
                    break

            # ------------------------------------------------------------------
            # evaluate the best local candidate with the real black‑box
            try:
                final_score = evaluate_sequence(best_local_a.tolist())
            except Exception:
                final_score = best_local_score

            if final_score < best_score:
                best_score = final_score
                best_seq = best_local_a.copy()
                print(
                    f"*** Global improvement: n={n} starter={sid} → "
                    f"score={best_score:.6f}"
                )
                if best_score <= 1.503:
                    print("Target reached – exiting early.")
                    return best_seq.tolist()

    # ------------------------------------------------------------------
    # fallback (should rarely be needed)
    if best_seq is None:
        n_fallback = 4000
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
        print("Fallback used.")

    print(
        f"Final best score: {best_score:.6f} (length = {len(best_seq)})"
    )
    return best_seq.tolist()
