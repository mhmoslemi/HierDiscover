except Exception:  # pragma: no cover
    _CVX_AVAILABLE = False

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_vals = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    vals[~np.isfinite(vals)] = 0.0
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


def _powerlaw_boundary_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5) + 1e-12) ** (-p)
    vals /= vals.sum()
    return vals


def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a non‑negative sequence to a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals

def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """Gradient of Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)."""
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # stability trick
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)        # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.02,
                   alpha_max: float = 3000.0,
                   lr0: float = 0.5,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """Adam optimisation of the smooth‑max surrogate."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional jitter to escape plateaus
        if t % 400 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


def _max_subgradient_step(a: np.ndarray):
    """Sub‑gradient of the exact ℓ∞ objective (max convolution)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-13, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Exact‑max sub‑gradient descent (short, decreasing step)."""
    a = a0.copy()
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.02 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()
        # tiny jitter
        if it % 500 == 0 and np.random.rand() < 0.08:
            a += np.random.randn(a.shape[0]) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


def _deterministic_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Greedy coordinate‑wise moves using the exact sub‑gradient."""
    a = a0.copy()
    n = len(a)
    while time.time() < deadline:
        grad, cur_max = _max_subgradient_step(a)
        i_up = int(np.argmax(grad))   # positive gradient -> moving mass *away* reduces max
        i_dn = int(np.argmin(grad))   # negative gradient -> moving mass *towards* reduces max
        if grad[i_up] - grad[i_dn] < 1e-14:
            break
        # move a small fraction from i_up to i_dn
        delta = min(a[i_up], 0.02)          # at most 2 % of a total mass per move
        if delta <= 0:
            break
        a[i_up] -= delta
        a[i_dn] += delta
        a = _project_onto_simplex(a, s=1.0)
    return a


def _simulated_annealing(a0: np.ndarray, deadline: float):
    """Light simulated‑annealing polish."""
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_peak = _fft_self_convolve(cur_a).max()
    best_a = cur_a.copy()
    best_peak = cur_peak

    start = time.time()
    total_time = max(deadline - start, 1e-6)

    start_temp = max(cur_peak, 1e-3) * 0.5
    final_temp = start_temp * 1e-6
    cooling_rate = -math.log(final_temp / start_temp) / total_time

    it = 0
    while time.time() < deadline:
        it += 1
        src = np.random.randint(n)
        if cur_a[src] <= 1e-12:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue
        # allow relatively large jumps
        max_delta = min(cur_a[src], 0.1)
        delta = np.random.rand() * max_delta

        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_peak = _fft_self_convolve(a_new).max()
        delta_obj = new_peak - cur_peak

        elapsed = time.time() - start
        temperature = start_temp * math.exp(-cooling_rate * elapsed)

        if delta_obj < 0 or np.random.rand() < math.exp(-delta_obj / max(temperature, 1e-12)):
            cur_a = a_new
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_a = cur_a.copy()

        # occasional “restart” to the current best
        if it % 25000 == 0 and np.random.rand() < 0.07:
            cur_a = best_a.copy()
            cur_peak = best_peak

    return best_a, best_peak


def _lp_improve(a: np.ndarray,
                max_iters: int = 5,
                mix_t: float = 0.3,
                inner_deadline: float = None) -> np.ndarray:
    """Linear‑programming polishing (dual‑widening)."""
    if not _CVX_AVAILABLE:
        return a

    import scipy.linalg

    n = a.shape[0]
    for _ in range(max_iters):
        if inner_deadline is not None and time.time() > inner_deadline:
            break
        M = _fft_self_convolve(a).max()
        # Toeplitz matrix representing convolution (dense, but n≤3000)
        col = np.r_[a, np.zeros(n - 1)]
        row = np.r_[a[0], np.zeros(n - 1)]
        A = scipy.linalg.toeplitz(col, row)

        b_var = cp.Variable(n, nonneg=True)
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [A @ b_var <= M])
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception:  # pragma: no cover
            break
        if b_var.value is None:
            break
        b_vec = np.maximum(b_var.value, 0.0)
        s = b_vec.sum()
        if s <= 1e-12:
            break
        g = b_vec / s
        a = (1.0 - mix_t) * a + mix_t * g
        a = _project_onto_simplex(a, s=1.0)
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a step‑function with a low score.
    Returns a plain Python list ready for `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Helper: cheap evaluation (2·n·max(conv))
    def fast_score(vec: np.ndarray) -> float:
        n = len(vec)
        return 2.0 * n * _fft_self_convolve(vec).max()

    # --------------------------------------------------------------
    # Base sequence: try to reuse a previously discovered construction.
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_seq = _project_onto_simplex(base_seq, s=1.0)
        base_n = len(base_seq)
    except Exception:
        base_seq = None
        base_n = 3000

    # --------------------------------------------------------------
    # -------------------------------------------------- Stage 1 : small‑scale optimisation
    # Choose a moderate length that allows LP solving.
    n_small = min(3000, max(800, int(base_n * 0.7)))
    print(f"[stage 1] small‑scale optimisation (n={n_small})")

    # Build a collection of starter vectors (all normalised to sum = 1)
    starters = {}

    starters["arcsine"] = _arcsine_distribution(n_small)

    # Beta families (including very edge‑heavy cases)
    for a in (0.15, 0.25, 0.35, 0.45, 0.55, 0.65, 0.75, 0.85, 1.0):
        tag = f"beta_{a:.2f}"
        starters[tag] = _beta_distribution(n_small, a, a)

    # Power‑law boundary
    for p in (0.35, 0.40, 0.45, 0.50, 0.55, 0.60):
        tag = f"power_{p:.2f}"
        starters[tag] = _powerlaw_boundary_distribution(n_small, p)

    # Dirichlet draws with a range of concentrations
    for conc in (0.2, 0.5, 1.0, 2.0):
        tag = f"dirichlet_{conc:.1f}"
        starters[tag] = np.random.dirichlet(np.full(n_small, conc))

    # Mixtures arcsine / uniform
    for w in (0.2, 0.4, 0.6, 0.8):
        tag = f"mix_{w:.1f}"
        starters[tag] = (1.0 - w) * _arcsine_distribution(n_small) + w * np.full(n_small, 1.0 / n_small)

    if base_seq is not None:
        starters["prev"] = _interpolate_to_length(base_seq, n_small)

    # Pick the best starter according to the cheap proxy
    best_small = None
    best_small_score = float("inf")
    for tag, vec in starters.items():
        vec = _project_onto_simplex(vec, s=1.0)
        sc = fast_score(vec)
        if sc < best_small_score:
            best_small_score = sc
            best_small = vec.copy()
        print(f"[init] {tag:<12s} score={sc:.6f}")

    # --------------------------------------------------------------
    # Dedicated time slice for the small‑scale optimisation
    now = time.time()
    time_small_total = (deadline - now) * 0.45   # roughly 45 % of remaining budget
    deadline_small = now + time_small_total

    # Adam optimisation (smooth‑max surrogate)
    adam_deadline = now + time_small_total * 0.55
    a_opt, _ = _adam_optimize(best_small, adam_deadline)

    # Exact‑max sub‑gradient refinement
    subgrad_deadline = now + time_small_total * 0.30
    a_opt = _refine_max(a_opt, subgrad_deadline)

    # LP polishing (only if we have enough slack)
    lp_deadline = now + time_small_total * 0.05
    if n_small <= 3000 and _CVX_AVAILABLE:
        a_opt = _lp_improve(a_opt,
                            max_iters=8,
                            mix_t=0.25,
                            inner_deadline=lp_deadline)

    # Simulated annealing polishing (short)
    if time.time() < deadline_small:
        a_opt, peak = _simulated_annealing(a_opt, deadline_small)
    else:
        peak = _fft_self_convolve(a_opt).max()

    best_small = a_opt.copy()
    best_small_score = fast_score(best_small)
    print(f"[stage 1] finished – fast_score={best_small_score:.6f}")

    # --------------------------------------------------------------
    # ------------------------------------------------- Stage 2 : upsample & deep refinement
    # Decide final length (larger, but still tractable for FFT)
    N_final = max(4000, int(base_n * 1.2))
    if N_final > 8000:
        N_final = 8000
    print(f"[stage 2] upsample to final length N={N_final}")

    a_big = _interpolate_to_length(best_small, N_final)

    # Allocate most of the remaining budget to this stage
    now = time.time()
    time_big_total = deadline - now - 10.0          # keep a tiny buffer for final evaluation
    if time_big_total < 30.0:
        time_big_total = deadline - now - 0.5
    deadline_big = now + time_big_total

    # Adam (smooth max)
    adam_deadline_big = now + time_big_total * 0.45
    a_big, _ = _adam_optimize(a_big, adam_deadline_big)

    # Exact max sub‑gradient descent
    subgrad_deadline_big = now + time_big_total * 0.30
    a_big = _refine_max(a_big, subgrad_deadline_big)

    # Deterministic coordinate descent – a cheap yet effective clean‑up
    det_deadline = now + time_big_total * 0.10
    a_big = _deterministic_descent(a_big, det_deadline)

    # Longer simulated annealing
    if time.time() < deadline_big:
        a_big, peak_big = _simulated_annealing(a_big, deadline_big)
    else:
        peak_big = _fft_self_convolve(a_big).max()

    final_score = fast_score(a_big)
    print(f"[stage 2] finished – fast_score={final_score:.6f}")

    # --------------------------------------------------------------
    # ------------------------------------------------------- Final evaluation
    # Clip to Python list and let the external evaluator produce the true score.
    try:
        true_score = evaluate_sequence(a_big.tolist())
        print(f"[final] evaluate_sequence returned score={true_score:.6f}")
    except Exception as exc:   # pragma: no cover
        true_score = final_score
        print("[final] evaluate_sequence raised:", exc)

    print(f"Best reported score: {min(true_score, final_score):.6f} (length={len(a_big)})")
    return a_big.tolist()
