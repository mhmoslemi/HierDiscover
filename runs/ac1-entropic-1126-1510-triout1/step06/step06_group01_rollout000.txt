def _smooth_descent(a, deadline):
    n = a.shape[0]
    best = a.copy()
    best_score = _max_conv_score(a)
    # schedule
    alpha_vals = [0.5, 1.0, 5.0, 20.0, 100.0, 500.0]
    steps_each = max(1500, n // 2)  # maybe larger
    # Adam hyper
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    t=0
    for alpha in alpha_vals:
        for _ in range(steps_each):
            if time.time() >= deadline: return best, best_score
            t+=1
            # compute convolution
            b = _fft_self_convolve(a)
            # stable softmax
            maxb = b.max()
            exp_terms = np.exp(alpha * (b - maxb))
            w = exp_terms / exp_terms.sum()
            # gradient via convolution
            grad = 2.0 * _fft_convolve(a[::-1], w)[n-1:n-1+n]
            # adam update
            m = beta1*m + (1-beta1)*grad
            v = beta2*v + (1-beta2)*(grad*grad)
            m_hat = m/(1 - beta1**t)
            v_hat = v/(1 - beta2**t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            # ensure nonneg+ sum=1 via simplex projection
            a = _project_onto_simplex(a, 1.0)
            # optional symmetrization
            # a = _symmetrize(a)
            # evaluation optionally
            if t % 200 == 0:
                cur_score = _max_conv_score(a)
                if cur_score < best_score:
                    best_score, best = cur_score, a.copy()
        # optionally reduce lr
    return best, best_score
```

"""
Propose a non‑negative step‑function that minimises
evaluate_sequence().

Algorithm
---------

1. **Candidate generation** – build a pool of analytic shapes
   (uniform, arcsine, triangular, exponential, symmetric beta) for a few
   lengths and a handful of random Dirichlet draws.  Keep the best
   normalised vector as a seed.

2. **Smooth Adam descent** – minimise a smooth approximation of the
   max‑convolution
        μ_α(a) = (1/α)·log Σ exp(α·(a*a)_k)
   with Adam on the simplex (Euclidean projection after each step).
   The smoothing parameter α is increased gradually; the number of
   iterations per α scales with the length.

3. **Exact max‑subgradient refinement** – switch to the true
   sub‑gradient of max(a*a) (the “active‑set’’ method from the reference
   code) with momentum and a diminishing step size.

4. **Simulated annealing** – a temperature‑controlled random‑transfer
   hill‑climber that accepts worsening moves with probability
   exp(−Δ/T).  This escapes remaining local minima.

5. **Length scaling** – the best vector is interpolated to a few larger
   lengths (up to ~15000) and the whole pipeline (2‑4) is repeated.
   Larger dimensions give more degrees of freedom and may approach the
   theoretical lower bound.

All optimisation stages respect the simplex constraint
∑ a_i = 1, a_i ≥ 0 (scale‑invariance guarantees the same evaluate_sequence
value for any total mass).  The routine respects a hard time budget
(≈1000 s) and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) for a vector already normalised to sum = 1.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _initial_candidates(n: int) -> list[np.ndarray]:
    """Generate a handful of plausible initial vectors for a given length."""
    cand = [
        _distribution_uniform(n),
        _distribution_arcsine(n),
        _distribution_triangular(n),
        _distribution_exponential(n, lam=0.003),
        _distribution_exponential(n, lam=0.01),
    ]
    for p in (0.2, 0.5, 1.0, 2.0):
        cand.append(_distribution_beta_sym(n, p))

    # random sparse start (≈0.5 % non‑zeros)
    rnd = np.zeros(n, dtype=np.float64)
    k = max(2, n // 200)
    idx = np.random.choice(n, k, replace=False)
    rnd[idx] = np.random.rand(k)
    rnd /= rnd.sum()
    cand.append(rnd)

    # a few dense random Dirichlet draws
    for _ in range(3):
        d = np.random.rand(n)
        d /= d.sum()
        cand.append(d)

    return cand

def _smooth_descent(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth max objective.
    Returns (best_vector, best_score) witnessed during the run.
    """
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # smoothing schedule – more iterations for larger n
    steps_per_alpha = max(2000, n // 2)
    schedule = [(0.5, steps_per_alpha), (1.0, steps_per_alpha),
                (5.0, steps_per_alpha), (20.0, steps_per_alpha),
                (100.0, steps_per_alpha), (500.0, steps_per_alpha)]

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best, best_score
            t += 1

            # self‑convolution (full)
            b = _fft_self_convolve(a)

            # soft‑max over b (stable)
            maxb = b.max()
            exp_terms = np.exp(alpha * (b - maxb))
            w = exp_terms / exp_terms.sum()            # weights sum to 1

            # gradient = 2 * conv_rev(a[::-1], w) (first n entries)
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam update + Euclidean projection onto simplex
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a, 1.0)

            # occasional exact evaluation
            if t % 200 == 0:
                cur_score = _objective(a)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best = a.copy()
    return best, best_score

def _exact_subgradient_refine(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Exact max‑subgradient descent with momentum.
    Stops when time runs out or when the objective is already very low.
    """
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)

    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.30
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps)

        w = np.zeros_like(b)
        w[active] = 1.0

        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / np.sqrt(iteration + 1.0)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)

        cur_score = _objective(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time():.1f}] subgrad iter={iteration} best {best_score:.6f}")

        # early exit if we already beat the target bound comfortably
        if best_score <= 1.505:
            break
        if time.time() + 0.5 >= deadline:
            break
    return best, best_score

def _simulated_annealing(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Random‑transfer annealing on the simplex.
    Accepts worsening moves with probability exp(−Δ/T).
    """
    n = a.shape[0]
    cur = a.copy()
    cur_score = _objective(cur)
    best = cur.copy()
    best_score = cur_score

    # annealing schedule parameters
    T_start = 0.02
    T_end = 1e-4
    # exponential decay rate estimated from a generous iteration budget
    decay = 0.9995

    iter_count = 0
    while time.time() < deadline:
        iter_count += 1
        # temperature decays multiplicatively
        T = max(T_start * (decay ** iter_count), T_end)

        # pick two distinct indices
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # propose a transfer; magnitude scales with current temperature
        max_transfer = cur[i] * min(0.5, T * 50.0)   # larger transfers early on
        if max_transfer <= 0:
            continue
        delta = np.random.rand() * max_transfer

        a_new = cur.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = _objective(a_new)

        if new_score < cur_score:
            cur, cur_score = a_new, new_score
            if new_score < best_score:
                best_score, best = new_score, a_new.copy()
                print(f"[{time.time():.1f}] SA iter={iter_count} new best {best_score:.6f}")
        else:
            # Metropolis acceptance
            if np.random.rand() < math.exp(-(new_score - cur_score) / (T + 1e-12)):
                cur, cur_score = a_new, new_score

        # occasional small jitter to avoid stagnation
        if iter_count % 3000 == 0:
            pert = np.random.randn(n) * 1e-4
            cur = _project_onto_simplex(cur + pert, 1.0)
            cur_score = _objective(cur)

        # break if we are already very close to the theoretical lower bound
        if best_score <= 1.503:
            break
    return best, best_score


# ----------------------------------------------------------------------
def _optimize_sequence(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Run the three optimisation stages (smooth Adam → exact sub‑gradient →
    annealing) on a starting vector a.
    """
    # Stage 1: smooth Adam descent
    a, sc = _smooth_descent(a, deadline)
    best_seq, best_score = a.copy(), sc

    # Stage 2: exact sub‑gradient refinement
    a, sc = _exact_subgradient_refine(a, deadline)
    if sc < best_score:
        best_seq, best_score = a.copy(), sc

    # Stage 3: simulated annealing
    a, sc = _simulated_annealing(a, deadline)
    if sc < best_score:
        best_seq, best_score = a.copy(), sc

    return best_seq, best_score
