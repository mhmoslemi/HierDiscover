"""
Search for a non‑negative step function (vector) that minimises
``evaluate_sequence``.  The algorithm explores several lengths,
initialises from a handful of analytic families plus random Dirichlet
samples, runs a smoothed Adam descent on a soft‑max approximation of
the max‑convolution, follows with an exact sub‑gradient refinement,
and finishes with a *gradient‑directed* hill‑climber that moves mass
from positions that increase the active‐set convolution to positions
that decrease it.  Time is split among the different lengths so the
whole search respects the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = _next_pow_two(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = _next_pow_two(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) for a vector already normalised to sum = 1.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals

        (500.0, 2000),
    ]

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score
            t += 1

            b = _fft_self_convolve(a)

            # stable soft‑max over b
            maxb = b.max()
            exp_terms = np.exp(alpha * (b - maxb))
            w = exp_terms / exp_terms.sum()            # soft‑max weights

            # gradient = 2 * conv_rev(a[::-1], w) (first n entries)
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam update + projection onto simplex
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a, 1.0)

            if t % 200 == 0:
                cur_score = _max_convolution_score(a)
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_seq = a.copy()
    return best_seq, best_score


# ----------------------------------------------------------------------
def _exact_subgradient_refine(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Exact max‑subgradient descent with momentum.  Runs until the local
    deadline or the target 1.503 is reached.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)

    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps)

        w = np.zeros_like(b)
        w[active] = 1.0

        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / np.sqrt(iteration)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad iter={iteration} new best {best_score:.6f}")

        if best_score <= 1.503:
            break
        if time.time() + 0.5 > deadline:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
def _gradient_hill_climb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Directed hill‑climbing that always moves a small amount of mass
    from the index with the largest sub‑gradient (i.e. biggest impact on
    the active set) to the index with the smallest sub‑gradient.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)
    cur_score = best_score
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # compute active set and its sub‑gradient
        b = _fft_self_convolve(a)
        maxb = b.max()
        eps = 1e-12 * (abs(maxb) + 1.0)
        active = np.where(b >= maxb - eps)[0]

        w = np.zeros_like(b)
        w[active] = 1.0
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]           # size n

        i_high = np.argmax(grad)
        i_low = np.argmin(grad)

        # if the gradient is flat we are close to a local optimum
        if grad[i_high] - grad[i_low] < 1e-12:
            break

        # step size shrinks with iteration count
        step_factor = 0.08 / np.sqrt(iteration)
        epsilon = min(step_factor, a[i_high])
        if epsilon <= 0:
            break

        # propose transfer
        a_new = a.copy()
        a_new[i_high] -= epsilon
        a_new[i_low] += epsilon
        a_new = _project_onto_simplex(a_new, 1.0)

        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[{time.time():.1f}] grad‑hill iter={iteration} new best {best_score:.6f}")
        else:
            # try a smaller epsilon if the move was not improving
            reduced = False
            for factor in (0.5, 0.25, 0.125):
                eps2 = epsilon * factor
                if eps2 < 1e-12:
                    break
                a_try = a.copy()
                a_try[i_high] -= eps2
                a_try[i_low] += eps2
                a_try = _project_onto_simplex(a_try, 1.0)
                sc = _max_convolution_score(a_try)
                if sc < cur_score - 1e-12:
                    a = a_try
                    cur_score = sc
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_seq = a_try.copy()
                        print(f"[{time.time():.1f}] grad‑hill iter={iteration} (reduced) best {best_score:.6f}")
                    reduced = True
                    break
            if not reduced:
                # no improvement – stop early to avoid endless loops
                break

        # occasional jitter to escape shallow plateaux
        if iteration % 4000 == 0 and np.random.rand() < 0.3:
            jitter = np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a + jitter, 1.0)
            cur_score = _max_convolution_score(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_hill_climb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    The original random‑transfer hill climber kept as a final polish
    step – it is cheap and occasionally finds improvements that the
    directed version misses.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)
    cur_score = best_score

    max_transfer = 0.07          # at most 7 % of total mass per trial
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        transfer = max_transfer * (0.5 ** (iteration / 4000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_onto_simplex(a_new, 1.0)

        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[{time.time():.1f}] random‑hill iter={iteration} improved best {best_score:.6f}")

        if iteration % 5000 == 0 and np.random.rand() < 0.2:
            pert = np.random.randn(n) * 1e-4
            a = _project_onto_simplex(best_seq + pert, 1.0)
            cur_score = _max_convolution_score(a)

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Main entry point.  Returns the best step‑function (list of floats)
    found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # a selection of lengths – we spread the budget uniformly across them
    lengths = [800, 1200, 2000, 3000, 5000, 8000, 12000, 20000]

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  If a previous construction is available, keep it as a baseline.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            overall_best_seq = prev.copy()
            overall_best_score = sc
            print(f"[{time.time():.1f}] loaded previous best {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Iterate over the chosen lengths, allocating equal time shares.
    total_lengths = len(lengths)
    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        remaining = total_lengths - idx
        time_per_len = max(0.0, (deadline - time.time()) / remaining)
        len_deadline = time.time() + time_per_len

        # ----- generate a modest pool of start vectors ----------------
        candidates = _initial_candidates(n)

        # a few extra Dirichlet draws for diversity
        for _ in range(3):
            cand = np.random.rand(n)
            cand /= cand.sum()
            candidates.append(cand)

        # ----- quick screening ---------------------------------------
        best_local = None
        best_local_score = float("inf")
        for cand in candidates:
            if time.time() >= len_deadline:
                break
            score = _max_convolution_score(cand)
            if score < best_local_score:
                best_local_score = score
                best_local = cand.copy()
        if best_local is None:
            continue

        # ----- refinement pipeline ------------------------------------
        a = best_local.copy()

        a, score = _smooth_descent(a, len_deadline)
        a, score = _exact_subgradient_refine(a, len_deadline)
        a, score = _gradient_hill_climb(a, len_deadline)
        a, score = _local_hill_climb(a, len_deadline)

        # ----- record global best ------------------------------------
        if score < overall_best_score:
            overall_best_score = score
            overall_best_seq = a.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST {score:.6f} (len={n})")

        if overall_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 2.  Fallback – should never be needed.
    if overall_best_seq is None:
        n = max(800, lengths[0])
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _max_convolution_score(overall_best_seq)

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
