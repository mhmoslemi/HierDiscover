    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()  # next power of two.
    X = np.fft.rfft(x, size)

"""
Search for a low‑score step function.

The routine works in three stages for a list of problem sizes:
  1️⃣  Multi‑restart smoothed‑max optimisation using L‑BFGS‑B.
      The temperature (μ) is cooled slowly (factor 0.9) and the
      optimiser is given many inner iterations so that the final
      solution of the smooth surrogate is very accurate.
  2️⃣  Projected accelerated sub‑gradient descent on the true
      max‑objective.  A back‑tracking line‑search with a dynamic
      step‑size is used; after a stall we inject a small random
      jitter to escape plateaus.
  3️⃣  Large‑scale stochastic hill‑climbing.  Random swaps,
      δ‑mass transfers and block‑averaging moves are tried
      until the time budget expires.  Every move is accepted only
      if the *exact* evaluate_sequence() value improves.

The search is performed for several lengths (default 2000, 4000,
6000, 8000).  For each length the best sequence found is kept,
and the globally best one is finally returned.

All helper functions are top‑level (no closures) and the routine
obeys the supplied 1000 s time budget.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z == 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (probability vector)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(y: np.ndarray, mu: float, n: int, S: float):
    """
    Smooth‑max surrogate of max(conv) and its gradient w.r.t. y
    (log‑odds of the simplex).
    """
    p = _softmax(y)          # Σp = 1, p ≥ 0
    a = S * p                # Σa = S

    conv = np.convolve(a, a)             # length 2·n‑1
    max_c = conv.max()

    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = mask.sum()
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. a  (see derivation in the description)
    conv_wr = np.convolve(w, a[::-1])
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

    # back‑prop through the soft‑max
    inner = np.dot(g_a, p)          # scalar
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable for scipy.optimize (value, gradient)."""

    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _smooth_obj_and_grad(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
def _hard_subgradient(a: np.ndarray):
    """
    Sub‑gradient of the exact max‑objective max(conv) at a.
    Returns (g, max_conv) where g has the same length as a.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_c = conv.max()
    mask = conv == max_c
    cnt = mask.sum()
    if cnt == 0:
        w = np.zeros_like(conv)
    else:
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
    conv_wr = np.convolve(w, a[::-1])
    g = 2.0 * conv_wr[n - 1:n - 1 + n]
    return g, max_c

                    start = rng.integers(0, n - 2)
                    length = rng.integers(2, min(80, n - start))
                    block = slice(start, start + length)
                    avg = cand[block].mean()
                    cand[block] = avg
                # projection (only needed for move‑type 1, but cheap anyway)
                cand = _project_to_simplex(cand, S)

                sc = evaluate_sequence(cand.tolist())
                if sc < global_best["score"] - 1e-12:
                    global_best["score"] = sc
                    global_best["vec"] = cand.copy()
                    global_best["n"] = n
                    a_hc = cand
                    print(f"[hill] trial {trial:6d} → {sc:.6f}")

        # ----- 4️⃣  Deterministic post‑processing heuristics -------------
        if global_best["vec"] is not None:
            base = global_best["vec"]
            # (a) interleaved big‑small ordering
            vals_desc = np.sort(base)[::-1]
            L = len(vals_desc)
            inter = np.empty_like(vals_desc)
            left = vals_desc[:L // 2][::-1]
            right = vals_desc[L // 2:][::-1]
            inter[0::2] = left
            inter[1::2] = right
            inter = _project_to_simplex(inter, S)
            _evaluate_and_maybe_update(inter, global_best, deadline)

            # (b) pure ascending / descending
            asc = _project_to_simplex(np.sort(base), S)
            _evaluate_and_maybe_update(asc, global_best, deadline)

            desc = _project_to_simplex(vals_desc, S)
            _evaluate_and_maybe_update(desc, global_best, deadline)

        # early stop if we have already beaten the target
        if global_best["score"] < 1.503:
            print("[target] reached required bound – exiting early")
            break

    # ------------------------------------------------------------------
    print(f"\n[final] best score = {global_best['score']:.6f} (n = {global_best['n']})")
    return global_best["vec"].tolist()
