        exp_c = np.exp((c - max_c)/epsilon)  # stable

        # gradient
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2 * conv_rev[n-1:n-1+n]  # no diag correction needed
        # update
        a = a - step0/np.sqrt(t) * grad
        a = _project_onto_simplex(a, s=1.0)

"""
Propose a non‑negative step function that (hopefully) beats the previous
best score ~1.5101.  The algorithm works in three phases:

1. **Broad sampling** – a variety of analytic families (uniform, triangular,
   cosine, Gaussian, exponential, symmetric beta) and many random Dirichlet
   draws are generated for several lengths (roughly 1 k to 8 k).  All candidates
   are normalised to ℓ₁‑norm 1 and scored with the cheap FFT based objective
   ``2·n·max(a*a)``.

2. **Refinement** – the best ~12 seeds are fed to three optimiser kernels:
   * exponentiated‑gradient (mirror descent w.r.t. KL),
   * Adam‑style adaptive moment descent,
   * a smoothed‑max gradient descent (soft‑max of the convolution, which
     provides a smoother search direction).  Each optimiser runs until the
     global time budget is about to expire.

3. **Simulated‑annealing hill‑climb** – the current best sequence is continuously
   perturbed by moving a small random amount of mass between two positions.
   Deteriorating moves are sometimes accepted with a temperature‑dependent
   probability, allowing the search to escape shallow local minima.

The whole routine respects a 1000 s budget (minus a safety margin) and always
returns a list of non‑negative floats that can be fed directly to
``evaluate_sequence``.  All helper functions are defined at module level.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution ``a*a`` (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Score = 2·n·max(a*a) assuming ∑a = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of ``max(a*a)`` on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # correct the double‑counting of the diagonal terms
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 50000,
                          eta0: float = 0.8, decay: float = 0.9999):
    """Exponentiated‑gradient (KL‑mirror descent)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, cur_max, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it) * decay
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 50000,
                  step0: float = 0.12,
                  beta1: float = 0.9,
                  beta2: float = 0.999):
    """Adam‑style adaptive moment optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, cur_max, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray, deadline: float,
                             epsilon: float = 0.02,
                             step0: float = 0.1,
                             max_iters: int = 50000):
    """Gradient descent on a smoothed max (soft‑max) of the convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    n = a.shape[0]
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # smoothed weights
        c = _fft_self_convolve(a)
        mx = c.max()
        # stability: shift by max before exponentiation
        w = np.exp((c - mx) / epsilon)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]   # no diagonal correction needed

        eta = step0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Gamma‑Dirichlet, α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


def _uniform_distribution(n: int) -> np.ndarray:
    """Uniform mass."""
    return np.full(n, 1.0 / n)


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _cosine_distribution(n: int) -> np.ndarray:
    """Raised cosine on [0, n‑1]."""
    i = np.arange(n, dtype=np.float64)
    vals = 0.5 * (1 + np.cos(np.pi * (2 * i + 1 - n) / n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _gaussian_distribution(n: int, sigma_ratio: float = 0.2) -> np.ndarray:
    """Discrete Gaussian centred in the middle, σ = sigma_ratio·n."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = max(1.0, sigma_ratio * n)
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    """Symmetric exponential decay from the centre."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric beta density on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * (np.log(x) + np.log(1.0 - x)))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w
