def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Return the objective 2·n·max(a*a) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals

def _smooth_descent_extended(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max objective.
    Uses a longer schedule (α up to 5000) and many more steps.
    Returns the best vector and the corresponding exact score.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # smoothing schedule – (α, number of steps)
    schedule = [
        (1.0,   2000),
        (5.0,   2000),
        (20.0,  2000),
        (100.0, 2000),
        (500.0, 2000),
        (2000.0,1500),
        (5000.0,1000),
    ]

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score
            t += 1

            # self‑convolution
            b = _fft_self_convolve(a)

            # soft‑max weights for the smoothed max
            maxb = b.max()
            exp_terms = np.exp(alpha * (b - maxb))
            w = exp_terms / exp_terms.sum()          # sums to 1

            # gradient = 2 * conv_rev(a[::-1], w) (first n entries)
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam update + projection onto simplex
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a, 1.0)

            # occasional exact evaluation
            if t % 200 == 0:
                cur_score = _max_convolution_score(a)
                if cur_score < best_score - 1e-13:
                    best_score = cur_score
                    best_seq = a.copy()
    return best_seq, best_score

def _exact_subgradient_refine(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Exact max‑subgradient descent with momentum.
    Continues as long as time permits.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)

    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps)

        w = np.zeros_like(b)
        w[active] = 1.0

        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / math.sqrt(iteration)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] sub‑grad iter={iteration} new best {best_score:.6f}")

        # early stop if we have already reached the target bound
        if best_score <= 1.503:
            break

        # safety break
        if time.time() + 0.3 >= deadline:
            break
    return best_seq, best_score

def _directed_hill_climb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Gradient‑guided hill climbing.
    Moves a small amount of mass from a high‑gradient index to a low‑gradient
    index, accepting only improving moves.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)

    # parameters controlling the size of each transfer
    max_frac = 0.07        # up to 7 % of the current mass at the source
    iteration = 0
    last_print = 0

    while time.time() < deadline:
        iteration += 1

        # compute current active set and the associated sub‑gradient
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps)

        w = np.zeros_like(b)
        w[active] = 1.0
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # choose source (largest grad) and target (smallest grad)
        src = int(np.argmax(grad))
        tgt = int(np.argmin(grad))
        if src == tgt:
            # degenerate case – pick random pair
            src = np.random.randint(n)
            tgt = np.random.randint(n - 1)
            if tgt >= src:
                tgt += 1

        # amount to move – a fraction of the source mass
        delta = min(max_frac * a[src], a[src] * 0.02, 0.02)
        if delta <= 0:
            # nothing to move – exit
            break

        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta

        new_score = _max_convolution_score(a_new)

        if new_score < best_score - 1e-13:
            a = a_new
            best_score = new_score
            best_seq = a_new.copy()
            if iteration - last_print > 500:
                print(f"[{time.time():.1f}] hill iter={iteration} improved best {best_score:.6f}")
                last_print = iteration
        else:
            # occasionally try a random jitter to escape flat plateaus
            if iteration % 3000 == 0 and np.random.rand() < 0.15:
                pert = np.random.randn(n) * 1e-4
                a = best_seq + pert
                a = _project_onto_simplex(a, 1.0)

        # safety check
        if time.time() + 0.2 >= deadline:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
def _optimize_candidate(seed_vec: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Run the three optimisation phases on a single candidate vector.
    Returns the best vector and its exact score.
    """
    a = seed_vec.copy()
    a, sc = _smooth_descent_extended(a, deadline)
    if time.time() >= deadline:
        return a, sc
    a, sc = _exact_subgradient_refine(a, deadline)
    if time.time() >= deadline:
        return a, sc
    a, sc = _directed_hill_climb(a, deadline)
    return a, sc


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  Returns the best sequence found as a list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # safety margin

    # ------------------------------------------------------------------
    # Lengths to explore – a mixture of short and long vectors.
    lengths = [500, 800, 1200, 2000, 3000, 5000, 8000]

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # If a previous solution exists (height_sequence_1), treat it as a seed.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Main loop over lengths
    for n in lengths:
        if time.time() >= deadline:
            break

        # generate candidate pool and keep the best two (by exact score)
        raw_cands = _initial_candidates(n)
        scored = []
        for vec in raw_cands:
            vec = vec / vec.sum()
            sc = _max_convolution_score(vec)
            scored.append((sc, vec))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = vec.copy()
                print(f"[{time.time():.1f}] init n={n} score={sc:.6f}")

        # sort by score, keep top 2 for heavy optimisation
        scored.sort(key=lambda x: x[0])
        top_cands = [vec for _, vec in scored[:2]]

        for seed_vec in top_cands:
            if time.time() >= deadline:
                break
            # run the full optimisation pipeline
            best_vec, best_sc = _optimize_candidate(seed_vec, deadline)
            if best_sc < overall_best_score:
                overall_best_score = best_sc
                overall_best_seq = best_vec.copy()
                print(f"[{time.time():.1f}] length={n} new overall best {best_sc:.6f}")

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded (unlikely) return a uniform vector.
    if overall_best_seq is None:
        n = 800
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _max_convolution_score(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    final_score = _max_convolution_score(overall_best_seq)
    print(f"Final best score: {final_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
