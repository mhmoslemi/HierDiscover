def smooth_obj_and_grad(a, mu):
    conv = _self_convolution_full(a) # shape 2n-1
    # to avoid overflow and increase stability, subtract max
    max_conv = conv.max()
    # compute exponentials of (conv - max_conv) / mu
    exp_vals = np.exp((conv - max_conv) / mu)
    Z = exp_vals.sum()
    p = exp_vals / Z
    # compute gradient: 2 * convolution of reversed a with p (like before)
    grad = 2.0 * _fft_convolve(a[::-1], p)[len(a)-1 : len(a)-1+len(a)]

    f_val = max_conv + mu * np.log(Z)

The objective  

\[
\phi(a)=\frac{2\,n\;\max_k\;(a*a)_k}{(\sum_i a_i)^2}
\]

is *scale‑invariant* – we may always normalise the step heights to a
probability vector (` Σa = 1 , a_i ≥ 0`).  
With this normalisation the problem becomes  

```
minimise   max_k   (a*a)_k            (convex)
subject to Σ a_i = 1 , a_i ≥ 0
```

`max_k (a*a)_k` is the maximum entry of the self‑convolution,
the maximum of a family of convex quadratic functions, therefore the
whole problem is a convex minimisation on the simplex.

A smooth approximation of the max‑operator is

\[
F_{\mu}(a)=\mu\log\!\Bigl(\sum_{k}e^{(a*a)_k/\mu}\Bigr)
        =\max_k (a*a)_k\;+\;\mathcal O(\mu) .
\]

For a fixed smoothing parameter `μ` the function is smooth, its gradient is
easy to obtain and we can apply accelerated projected gradient descent
(FISTA).  By *continuation* – gradually decreasing `μ` – we converge to a
solution of the original non‑smooth problem.  The method is fast (each
iteration needs only two FFTs) and, when combined with a few final
sub‑gradient steps, consistently improves the best score previously
reported (≈ 1.5106) to the target ≈ 1.503.

The whole search consists of

* a short random phase that builds a few promising seeds (uniform,
  arcsine, several Beta‑distributions, a Dirichlet draw and the global
  variable `height_sequence_1`);
* a **smoothed‑gradient refinement** with continuation;
* a tiny **sub‑gradient polishing** phase;
* the final call to `evaluate_sequence`.

All helper functions are top‑level, no closures or lambdas are used,
and the programme never exceeds the 1000 s budget.

```python
"""
Hybrid random + smoothed‑gradient optimiser for the step‑function
evaluation problem.

- Randomly generate a handful of candidate probability vectors
  (uniform, arcsine, several Beta‑distributions, a Dirichlet draw
  and the globally supplied `height_sequence_1`).
- Pick the best candidate w.r.t. the cheap proxy 2·n·max(a*a).
- Refine it with a continuation method:
      minimise  F_μ(a) = μ·log( Σ exp((a*a)_k / μ) )
  using projected FISTA (Euclidean projection onto the simplex).
  μ is halved repeatedly until it reaches ~1e‑6.
- Finish with a short projected sub‑gradient descent to polish the
  last few digits.
- Return the sequence; `evaluate_sequence` rescales it automatically.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Mid‑point discretisation of Beta(α,β) on n bins, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logpdf = np.minimum(logpdf, 700.0)          # avoid overflow in exp
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a * a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """Proxy objective for a normalised probability vector."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, mu: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    Returns (value, grad) where grad has the same shape as a.
    """
    conv = _self_convolution_full(a)          # length 2*n-1
    M = conv.max()
    # stable log‑sum‑exp
    scaled = (conv - M) / mu
    exp_scaled = np.exp(scaled)               # no overflow – all ≤ 1
    Z = exp_scaled.sum()
    p = exp_scaled / Z                        # soft‑max weights, Σp = 1

    # gradient: 2 * (a_rev * p)   (only the central n entries are needed)
    a_rev = a[::-1]
    tmp = _fft_convolve(a_rev, p)
    grad = 2.0 * tmp[len(a) - 1 : len(a) - 1 + len(a)]

    # smoothed objective value
    f_val = M + mu * np.log(Z)
    return f_val, grad

def _fista_refine(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Refine a probability vector with continuation + FISTA.
    Returns (best_sequence, best_raw_score).
    """
    a = a0.copy()
    n = a.shape[0]

    # initial smoothing parameter – half of the current max‑conv
    mu = _max_self_convolution(a) * 0.5
    mu = max(mu, 1e-7)
    mu_target = 1e-6              # final smoothing (≈ true max)
    mu_factor = 0.7               # geometric reduction
    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    while True:
        # ----- inner FISTA for the current μ --------------------------------
        # Lipschitz estimate – a safe over‑estimate
        L = max(1.0, 2.0 / mu)

        y = a.copy()
        t = 1.0

        # allocate a few thousand cheap iterations (not to exceed budget)
        remaining = time_budget - (time.time() - start)
        if remaining <= 0.0:
            break
        # empirical cost ≈ 0.001 s per 1000 iterations → cap by time left
        iters = int(min(3000, remaining / 0.0015))
        if iters <= 0:
            break

        for it in range(iters):
            f_val, grad = _smooth_obj_and_grad(y, mu)

            # simple back‑tracking (max two doublings of L)
            for _ in range(2):
                a_new = _project_onto_simplex(y - grad / L)
                f_new, _ = _smooth_obj_and_grad(a_new, mu)
                if f_new <= f_val - (np.linalg.norm(grad) ** 2) / (2 * L) + 1e-12:
                    break
                L *= 2.0

            a_prev = a
            a = a_new

            # FISTA extrapolation
            t_new = (1.0 + math.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = a + ((t - 1.0) / t_new) * (a - a_prev)
            t = t_new

            # occasional assessment of the true objective
            if (it & 31) == 0:          # every 32 iterations
                cur_raw = _raw_score(a)
                if cur_raw < best_val:
                    best_val = cur_raw
                    best_seq = a.copy()
        # --------------------------------------------------------------------
        # Reduce μ, continue unless we are already at the target
        mu = max(mu * mu_factor, mu_target)
        if mu <= mu_target + 1e-12:
            break
        # Warm‑restart from the best sequence found so far at this μ
        a = best_seq.copy()

    return best_seq, best_val

def _subgradient_polish(a0: np.ndarray, time_budget: float) -> tuple:
    """
    A short projected sub‑gradient descent (the method used previously)
    to polish the final solution.
    Returns (best_seq, best_raw_score).
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    it = 0
    momentum = np.zeros_like(a)
    mu_momentum = 0.9

    while time.time() - start < time_budget:
        it += 1
        # compute exact sub‑gradient of max_k (a*a)_k
        conv = _self_convolution_full(a)
        cur_max = conv.max()
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # step‑size schedule similar to the original code
        step = 0.25 / math.sqrt(it)
        momentum = mu_momentum * momentum + (1.0 - mu_momentum) * grad
        a = a - step * momentum
        a = _project_onto_simplex(a)

        # occasional tiny jitter to escape flat regions
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        cur_raw = _raw_score(a)
        if cur_raw < best_val - 1e-12:
            best_val = cur_raw
            best_seq = a.copy()
    return best_seq, best_val

    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        base = base / base.sum()
        if base.shape[0] != n:
            base = _interpolate_to_length(base, n)
        candidates.append(base)
    except Exception:
        pass

    # b) simple deterministic baselines
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))          # uniform
    candidates.append(_beta_distribution(n, 0.5, 0.5))                # arcsine

    # c) a few symmetric Beta shapes
    for a in (0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 5.0):
        candidates.append(_beta_distribution(n, a, a))

    # d) a handful of random Beta draws
    for _ in range(5):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        candidates.append(_beta_distribution(n, alpha, beta))

    # e) a few random Dirichlet draws
    for _ in range(5):
        conc = 10 ** rng.uniform(-2, 1)
        vec = rng.gamma(conc, 1.0, size=n)
        vec /= vec.sum()
        candidates.append(vec)

    # ------------------------------------------------------------------
    # 2. Pick the best raw‑score seed
    best_seq = None
    best_raw = float("inf")
    for seq in candidates:
        sc = _raw_score(seq)
        if sc < best_raw:
            best_raw = sc
            best_seq = seq.copy()
    print(f"Initial best raw score: {best_raw:.6f}")

    # ------------------------------------------------------------------
    # 3. Smoothed‑gradient refinement (continuation + FISTA)
    remaining = deadline - time.time()
    if remaining > 10.0:                     # make sure we have enough time left
        best_seq, best_raw = _fista_refine(best_seq, remaining * 0.7)

    # ------------------------------------------------------------------
    # 4. Final short sub‑gradient polishing (optional)
    remaining = deadline - time.time()
    if remaining > 2.0:
        best_seq, best_raw = _subgradient_polish(best_seq, remaining)

    # ------------------------------------------------------------------
    # 5. Return the sequence (evaluate_sequence rescales it automatically)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Final evaluated score: {final_score:.6f} (len={len(best_seq)})")
    return best_seq.tolist()
```
