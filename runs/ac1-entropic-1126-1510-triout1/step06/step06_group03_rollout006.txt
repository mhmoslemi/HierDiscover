def _fft_self_convolve_full(a):
   n = len(a)
   size = 1 << ((2*n-1)-1).bit_length()
   A = np.fft.rfft(a, n=size)
   conv = np.fft.irfft(A * A, n=size)
   return conv[:2*n-1]

def _fft_convolve(a,b):
   size = 1 << ((a.size + b.size - 1)-1).bit_length()
   A = np.fft.rfft(a, n=size)
   B = np.fft.rfft(b, n=size)
   conv = np.fft.irfft(A * B, n=size)
   return conv[:a.size + b.size - 1]

def _l2_obj_and_grad(a):
    n = len(a)
    conv = _fft_self_convolve_full(a)
    C = 1.0/(2*n - 1)
    diff = conv - C
    obj = np.sum(diff**2)
    # gradient = 4 * conv_rev(diff)
    a_rev = a[::-1]
    # compute conv_rev(diff) same as earlier: conv between a_rev and diff
    grad_full = _fft_convolve(a_rev, diff)
    grad = 4.0 * grad_full[n-1:n-1+n]   # slice length n
    return obj, grad

def _smooth_max_obj_and_grad(a, beta):
    n = len(a)
    conv = _fft_self_convolve_full(a)
    max_c = conv.max()
    # compute weights via softmax
    max_val = conv.max()
    # stable log-sum-exp
    shifted = conv - max_val
    exp_beta = np.exp(beta * shifted)
    sum_exp = exp_beta.sum()
    # objective approximates max; optionally we can compute s = (max_val + (1/beta)*log(sum_exp))
    obj = max_val + np.log(sum_exp)/beta

    grad_full = _fft_convolve(a_rev, probs)
    grad = 2.0 * grad_full[n-1:n-1+n]
    return obj, grad, max_c

def _adam_optimize(x0, n, deadline, obj_and_grad_fn, max_iters=200000, lr0=0.12, beta1=0.9, beta2=0.999, eps=1e-12, extra_params=None):
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    best_a = None
    best_raw = float('inf')
    extra = extra_params if extra_params else {}
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # softmax
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / np.sum(exp_x)
        # obj and grad
        # obj_and_grad_fn may accept extra parameters (like beta)
        args = (a,)
        if extra:
            # assuming extra contains needed keys e.g., beta
            obj, grad = obj_and_grad_fn(*args, **extra)
        else:
            obj, grad = obj_and_grad_fn(*args)
        # after computing gradient for a, compute gradient for logits if needed
        dot = np.dot(a, grad)
        grad_x = a * (grad - dot)
        m = beta1 * m + (1-beta1) * grad_x
        v = beta2 * v + (1-beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        lr = lr0 * (np.sqrt(1 - beta2**it) / (1 - beta1**it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        # evaluate raw score using current a
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        raw_score = 2.0 * n * cur_max
        if raw_score < best_raw:
            best_raw = raw_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it} raw={best_raw:.6f} obj={obj:.6e}")
        # occasional jitter to escape
        if it % 2000 == 0:
            x += np.random.randn(*x.shape) * 0.01
    # fallback
    if best_a is None:
        best_a = np.full(n, 1.0/n)
        best_raw = 2.0 * n * _fft_self_convolve_full(best_a).max()
    return best_a, best_raw

"""
Improved construction of a non‑negative step function.

The algorithm operates in three phases (all on the probability simplex Σa = 1):

1️⃣  **Flattening (L2) phase** – minimise the ℓ₂‑distance between the
    full self‑convolution a*a and its mean value (1/(2n‑1)).  This makes
    the convolution as flat as possible and quickly reduces the peak.

2️⃣  **Smoothed‑max phase** – minimise a smooth approximation of the peak
    using a log‑sum‑exp (temperature β).  β is increased slowly so the
    optimiser first sees a gentle landscape and finally focuses on the
    true maximum.

3️⃣  **Hard‑max refinement** – a few sub‑gradient steps on the exact
    maximum (identical to the reference solution) polish the result.

The optimisation is performed with Adam on the log‑soft‑max (logits)
parameterisation, guaranteeing non‑negativity and Σa = 1 at every step.
All phases respect the overall time budget (default 1000 s) and the
best sequence found is returned as a plain Python list.
"""

# Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# FFT based convolutions (real, power‑of‑two padding)
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

# Sub‑gradient of the true maximum (used in the final polishing phase)
def _max_subgradient_step(a: np.ndarray):
    """Return (sub‑gradient, current maximum) of maxₖ (a*a)[k]."""
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(conv >= cur_max - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad_full = _fft_convolve(a_rev, w)
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad, cur_max

# L₂ objective (flattening)
def _l2_obj_and_grad(a: np.ndarray):
    """ℓ₂ objective = Σ (cₖ – C)² and its gradient w.r.t. a."""
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)

    return obj, grad

# Smoothed‑max objective (log‑sum‑exp)
def _smooth_max_obj_and_grad(a: np.ndarray, beta: float):
    """
    Smooth approximation of maxₖ(cₖ) via log‑sum‑exp.
    Returns (obj, grad, cur_max) where obj → max as β→∞.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    # stable log‑sum‑exp
    shift = conv - cur_max          # ≤ 0
    exp_beta = np.exp(beta * shift)
    sum_exp = exp_beta.sum()
    obj = cur_max + np.log(sum_exp) / beta

    probs = exp_beta / sum_exp      # softmax weights
    a_rev = a[::-1]
    grad_full = _fft_convolve(a_rev, probs)
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return obj, grad, cur_max

    """

    while it < max_iters and time.time() < deadline:

        # soft‑max (stable)

        # chain rule to obtain gradient w.r.t. logits
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** it)
