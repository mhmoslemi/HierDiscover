def _adam_optimize_softmax_smoothed(x0, n, deadline, max_iters=5000, lr0=0.1, lam0=1.0, lam_factor=1.001):
    x = x0.copy()
    m,v = zeros
    best_a = None, best_score = inf
    lam = lam0
    for it in 1..max_iters:
        if time > deadline: break
        # compute softmax a
        a = softmax(x)
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        cur_score = 2 * n * cur_max
        if cur_score < best_score: update best..
        # compute smoothed weights: p = softmax(lam * conv)
        # Use stable: tmp = lam*conv - max(lam*conv)
        tmp = lam * conv
        tmp_max = tmp.max()
        exp_tmp = np.exp(tmp - tmp_max)
        p = exp_tmp / exp_tmp.sum()
        # grad_a = 2 * conv_rev(p) as defined
        grad_a = 2 * _fft_convolve(a[::-1], p)[n-1:n-1+n]
        # chain rule to logits
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)
        # Adam update...
        m = beta1*m + (1-beta1)*grad_x
        v = beta2*v + (1-beta2)*(grad_x**2)
        m_hat = m/(1-beta1**it)
        v_hat = v/(1-beta2**it)
        lr = lr0 * (np.sqrt(1-beta2**it)/(1-beta1**it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        # optionally adjust lam
        lam *= lam_factor
        if it % 1000 ==0: jitter maybe

-----------------------------------------------------------------
* A probability vector `a` (a_i ≥ 0, Σ a_i = 1) represents the step
  heights after the black‑box normalisation (the ratio in
  ``evaluate_sequence`` is invariant under scaling).

* The exact objective to minimise is
        raw(a) = 2·n·max_k (a*a)_k .
  Instead of using the non‑smooth max we minimise the smooth
  approximation
        f_λ(a) = (1/λ)·log Σ_k exp( λ·(a*a)_k )
  (log‑sum‑exp).  For λ → ∞   f_λ → max.  A continuation schedule
  increases λ during optimisation, providing useful gradient information
  early on.

* The gradient of f_λ is
        ∇_a f_λ = 2·conv_rev(p) ,
  where  p_k = softmax( λ·(a*a)_k )  and
  conv_rev(p) = a_rev  *  p   (restricted to n entries).  The chain‑rule
  through the soft‑max parametrisation (logits `x`) yields a gradient for `x`
  that can be fed to Adam.

* For each chosen length we try several initialisations
  (warm‑start, uniform, random Dirichlet and a deterministic “arcsine”
  shape), run the Adam optimiser with a modest number of iterations,
  keep the best vector, and finally polish it with a projected sub‑gradient
  method (as in the reference solution).

* The routine respects the supplied time budget, prints progress and
  returns a plain Python list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    # points are taken at the mid‑points of each bin
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


# ----------------------------------------------------------------------
def _random_dirichlet_start(n: int, alpha_low: float = 0.1, alpha_high: float = 4.1) -> np.ndarray:
    """Draw a random Dirichlet vector of length n."""
    alpha = np.random.rand() * (alpha_high - alpha_low) + alpha_low
    g = np.random.gamma(alpha, 1.0, size=n)
    if g.sum() == 0:
        g = np.ones_like(g)
    return g / g.sum()


def _arcsine_start(n: int) -> np.ndarray:
    """Deterministic arcsine‑like shape (edges heavier, centre lighter)."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s > 0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals


# ----------------------------------------------------------------------
def _adam_optimize_smoothed(
    x0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 4000,
    lr0: float = 0.12,
    lam_start: float = 0.5,
    lam_factor: float = 1.001,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the logit parametrisation, minimising the
    smoothed log‑sum‑exp of the self‑convolution.  Returns the best
    probability vector found and its raw (non‑smoothed) score.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = None
    best_score = float("inf")
    lam = lam_start

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---------- probability vector ----------
        a = _softmax(x)                     # Σa = 1, a ≥ 0

        # ---------- evaluate raw objective ----------
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} iter={it}  raw={best_score:.6f}  λ={lam:.3f}")

        # ---------- smooth max (log‑sum‑exp) ----------
        # p_k = softmax( λ·conv_k )
        tmp = lam * conv
        tmp_max = tmp.max()
        exp_tmp = np.exp(tmp - tmp_max)
        p = exp_tmp / exp_tmp.sum()          # shape (2n‑1,)

        # ---------- gradient w.r.t. a ----------
        # ∇_a f_λ = 2 * (a_rev * p)   (restricted to n entries)
        grad_a = 2.0 * _fft_convolve(a[::-1], p)[n - 1 : n - 1 + n]

        # ---------- chain rule through soft‑max ----------
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---------- anneal λ ----------
        lam = lam * lam_factor

        # occasional jitter to escape plateaus
        if it % 1000 == 0:
            x += np.random.randn(*x.shape) * 0.01

    # fallback (should never happen)
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_full(best_a).max()

    return best_a, best_score


# ----------------------------------------------------------------------
def _optimise_length(
    n: int,
    deadline: float,
    prev_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length n.
    Several initialisations are tried (warm‑start, uniform, random Dirichlet,
    arcsine).  The best result is returned.
    """
    init_logits = []

    # warm‑start from previous best (interpolated)
    if prev_seq is not None:
        cand = _interpolate_to_length(prev_seq, n)
        eps = 1e-12
        init_logits.append(np.log(cand + eps))

    # uniform start
    init_logits.append(np.zeros(n, dtype=np.float64))

    # a few random Dirichlet draws
    for _ in range(3):
        a = _random_dirichlet_start(n)
        init_logits.append(np.log(a + 1e-12))

    # deterministic arcsine shape
    a_arc = _arcsine_start(n)
    init_logits.append(np.log(a_arc + 1e-12))

    best_local_seq = None
    best_local_score = float("inf")

    for idx, x0 in enumerate(init_logits):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] length {n}, start {idx+1}/{len(init_logits)}")
        seq, score = _adam_optimize_smoothed(
            x0, n, deadline, max_iters=3000,
            lr0=0.12, lam_start=0.5, lam_factor=1.001
        )
        if score < best_local_score:
            best_local_score = score
            best_local_seq = seq.copy()

    # safety fallback
    if best_local_seq is None:
        best_local_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_local_score = 2.0 * n * _fft_self_convolve_full(best_local_seq).max()

    return best_local_seq, best_local_score

    """
    Entry point required by the harness.
    Returns a list of non‑negative heights (a step function) that
    hopefully yields a low ``evaluate_sequence`` score.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin
    deadline = start_time + budget_s - 2.0

    # Try to reuse a previously discovered construction if present
    try:

        prev = np.maximum(prev, 0.0)
        s = prev.sum()
        if s > 0.0:

        else:
            prev = None
    except Exception:
        prev = None

    # ------------------------------------------------------------------
    # Length schedule – larger n gives more freedom but costs more time.
    # We go progressively from a modest size up to a larger one.
    lengths = [2000, 4000, 8000, 12000, 20000]

    best_seq = None
    best_score = float("inf")
    best_n = None

    # evaluate the supplied baseline (if any)
    if prev is not None:
        raw = 2.0 * len(prev) * _fft_self_convolve_full(prev).max()
        print(f"Baseline from height_sequence_1: raw score = {raw:.6f}")
        if raw < best_score:
            best_score = raw
            best_seq = prev.copy()
            best_n = len(prev)

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length {n} ===")
        seq, score = _optimise_length(n, deadline, prev_seq=best_seq)
        print(f"  finished length {n}: best raw score = {score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            best_n = n
        # warm‑start for the next length
        prev = best_seq

    # ------------------------------------------------------------------
    # Final fine‑tuning with a projected sub‑gradient method (as in the
    # reference solution).  This often improves the last few digits.
    if best_seq is not None:
        a = best_seq.copy()
        n = a.shape[0]
        it = 0
        # sub‑gradient loop respects the remaining time budget
        while time.time() < deadline:
            it += 1
            # gradient of the true max (active set, uniform weights)
            # – identical to the helper used earlier for sub‑gradient steps
            conv = _fft_self_convolve_full(a)
            cur_max = conv.max()
            # build uniform weight over active indices
            tol = max(1e-12, cur_max * 1e-12)
            active = np.where(conv >= cur_max - tol)[0]
            w = np.zeros_like(conv)
            if active.size:
                w[active] = 1.0 / active.size
            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

            step = 0.07 / math.sqrt(it)
            a = a - step * grad
            a = _project_onto_simplex(a)

            # occasional tiny random perturbation to shake the iterate
            if it % 800 == 0 and np.random.rand() < 0.06:
                a += np.random.randn(n) * 5e-7
                a = _project_onto_simplex(a)

            cur_score = 2.0 * n * cur_max
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] post‑refinement it={it}  raw={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation through the black‑box

    print(f"\nFinal evaluated score: {final_score:.6f} (length={len(final_list)})")
    return final_list
