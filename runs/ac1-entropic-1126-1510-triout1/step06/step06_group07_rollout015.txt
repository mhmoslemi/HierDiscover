def _targeted_transfer(a, S, rng):
    n = a.shape[0]
    conv = np.convolve(a, a)
    k_max = np.argmax(conv)
    i_start = max(0, k_max - (n - 1))
    i_end = min(k_max, n - 1)
    # contributions high to low
    idx_range = np.arange(i_start, i_end+1)
    # compute contributions
    contribs = a[idx_range] * a[k_max - idx_range]
    i = idx_range[np.argmax(contribs)]   # index with largest product
    # sample j from low indices (those with small a)
    low_idx = np.where(a <= np.quantile(a, 0.2))[0]
    if low_idx.size == 0:
        j = rng.integers(n)
    else:
        j = rng.choice(low_idx)
    max_delta = a[i] * 0.05
    delta = rng.random() * max_delta
    cand = a.copy()
    cand[i] -= delta
    cand[j] += delta
    cand = _project_to_simplex(cand, S)
    return cand
```

"""
Optimise a non‑negative step function to minimise the provided
evaluate_sequence() score.

The algorithm consists of four stages:

1.  Multi‑restart smooth‑max optimisation (L‑BFGS‑B) with a
    geometric cooling schedule for the smoothing parameter μ.
    This quickly drives the solution close to a local optimum.

2.  Hard‑max sub‑gradient descent with an adaptive step‑size
    and occasional random jitter, directly reducing the true max‑convolution.

3.  Aggressive stochastic hill‑climbing.  Moves are:
       * random swaps,
       * targeted δ‑mass transfer from indices that contribute most
         to the current maximal convolution,
       * block‑averaging,
       * small block shuffles,
       * complete random permutation.
    Only strictly improving moves are accepted (strict improvement
    prevents wasteful back‑tracking).

4.  Simple ordering heuristics (descending, ascending, interleaved)
    are finally checked.

All helper routines are defined at module level (no closures or lambdas)
and the routine respects the supplied time budget.  The internal
`fast_evaluate` replicates the behaviour of ``evaluate_sequence`` but
avoids the overhead of the security checks, allowing many thousands of
objective evaluations per second.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (probability vector, Σ=1, all ≥0)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _hard_subgradient(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient of the max‑convolution objective at point a.
    Returns (g, max_conv) where g is a sub‑gradient of size n.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_c = conv.max()
    mask = conv == max_c
    cnt = np.count_nonzero(mask)
    if cnt == 0:
        w = np.zeros_like(conv)
    else:
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
    # gradient wrt a
    conv_wr = np.convolve(w, a[::-1])
    g = 2.0 * conv_wr[n - 1:n - 1 + n]
    return g, max_c


# ----------------------------------------------------------------------
def _compute_smooth_obj_and_grad(y: np.ndarray, mu: float,
                                 n: int, S: float) -> tuple[float, np.ndarray]:
    """
    Smooth‑max surrogate:
        f_μ(y) = max(conv) + μ·log Σ exp((conv‑max)/μ)

    y – free (log‑odds) variables,
    mu – smoothing parameter,
    n – length of the sequence,
    S – target ℓ¹‑norm (√(2n)).
    """
    p = _softmax(y)               # probabilities on the simplex
    a = S * p                     # actual heights, Σa = S
    conv = np.convolve(a, a)      # full auto‑convolution

    max_c = conv.max()
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        if cnt:
            w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient wrt a
    conv_wr = np.convolve(w, a[::-1])
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

    inner = np.dot(g_a, p)                # Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)        # chain rule through soft‑max
    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable used by scipy.optimize; returns (value, gradient)."""
    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_smooth_obj_and_grad(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
def fast_evaluate(a: np.ndarray) -> float:
    """
    Very fast replica of evaluate_sequence (no type checks, no clipping
    beyond the problem limits).  a must be a 1‑D np.ndarray.
    """
    a = np.clip(a, 0.0, 1000.0)
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    conv = np.convolve(a, a)
    max_b = conv.max()
    n = a.shape[0]
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a low‑score non‑negative step function.
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # --------------------------------------------------------------
    # Determine problem size
    try:
        # a reference sequence may be provided in the environment
        ref = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = min(max(len(ref) * 2, 4000), 9000)
        have_ref = True
    except Exception:
        n = 6000
        have_ref = False

    S = math.sqrt(2.0 * n)                # target ℓ¹‑norm

    best_a = np.full(n, S / n, dtype=np.float64)   # uniform start (feasible)
    best_score = fast_evaluate(best_a)

    # ------------------------------------------------------------------
    # Helper to update best solution
    def maybe_update(candidate: np.ndarray) -> None:
        nonlocal best_score, best_a
        sc = fast_evaluate(candidate)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = candidate.copy()
            print(f"[improve] new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Multi‑restart smooth‑max optimisation
    # --------------------------------------------------------------
    init_ys = []

    # 0 – zero vector (soft‑max => uniform)
    init_ys.append(np.zeros(n, dtype=np.float64))

    # 1 – mildly decreasing exponential (log‑odds linear)
    alpha = rng.uniform(0.0005, 0.003) * (n / 2000.0)
    init_ys.append(-alpha * np.arange(n, dtype=np.float64))

    # 2 – opposite exponential
    init_ys.append(-alpha * np.arange(n - 1, -1, -1, dtype=np.float64))

    # 3 – Gaussian random
    init_ys.append(rng.standard_normal(n) * 0.5)

    # 4 – sparse Dirichlet draws
    for _ in range(5):
        dir_alpha = rng.uniform(0.05, 0.8)
        p_dir = rng.dirichlet([dir_alpha] * n)
        init_ys.append(np.log(p_dir + 1e-12))

    # 5 – reference if available
    if have_ref:
        a_ref = _project_to_simplex(ref, S)
        init_ys.append(np.log(a_ref + 1e-12))

    mu_factor = 0.80          # slower cooling → finer final approximation
    mu_min = 1e-12
    maxiter_per_mu = 600
    n_restart = len(init_ys)

    for rid, y0 in enumerate(init_ys):
        if time.time() > deadline:
            break
        print(f"[restart {rid + 1}/{n_restart}]")
        y = y0.copy()
        # initialise μ from the current max‑convolution
        a_cur = S * _softmax(y)
        mu = max(float(np.convolve(a_cur, a_cur).max()), 1.0)

        while mu > mu_min and time.time() < deadline:
            obj = _SmoothMaxObjective(mu, n, S)
            try:
                res = opt.minimize(
                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": maxiter_per_mu, "gtol": 1e-12, "disp": False},
                )
            except Exception as exc:
                print(f"[error] L‑BFGS failed: {exc}")
                break
            y = res.x
            a = S * _softmax(y)
            maybe_update(a)
            mu *= mu_factor

    # ------------------------------------------------------------------
    # 2️⃣  Sub‑gradient descent on the hard objective
    # --------------------------------------------------------------
    if time.time() < deadline and best_score < float("inf"):
        a = best_a.copy()
        step0 = S * 0.5 / n
        it = 0
        max_local_iters = 30000
        while time.time() < deadline and it < max_local_iters:
            g, max_c = _hard_subgradient(a)
            step = step0 / (1.0 + 1e-5 * it)
            improved = False
            # simple back‑tracking line search
            for _ in range(12):
                a_new = a - step * g
                a_new = _project_to_simplex(a_new, S)
                sc = fast_evaluate(a_new)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new
                    improved = True
                    print(f"[subgrad] it {it:05d} step {step:.3e} → {best_score:.6f}")
                    break
                step *= 0.5
            if not improved:
                # random jitter to escape possible plateau
                i, j = rng.choice(n, size=2, replace=False)
                if a[i] > 1e-12:
                    delta = rng.random() * a[i] * 0.03
                    a[i] -= delta
                    a[j] += delta
                    a = _project_to_simplex(a, S)
            it += 1

    # ------------------------------------------------------------------
    # 3️⃣  Aggressive stochastic hill‑climbing
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 1.5 and best_score < float("inf"):
        max_trials = int(min(800000, remaining * 70.0))   # very many cheap trials
        print(f"[hill‑climb] {max_trials} trials")
        a = best_a.copy()
        n_half = n // 2
        for trial in range(max_trials):
            move_type = rng.integers(0, 5)   # 0‑swap,1‑targeted‑δ,2‑avg,3‑shuffle,4‑perm
            cand = a.copy()
            if move_type == 0:          # simple swap
                i, j = rng.choice(n, size=2, replace=False)
                cand[i], cand[j] = cand[j], cand[i]
            elif move_type == 1:        # targeted δ‑transfer
                conv = np.convolve(cand, cand)
                k_max = int(conv.argmax())
                i_start = max(0, k_max - (n - 1))
                i_end = min(k_max, n - 1)
                idxs = np.arange(i_start, i_end + 1, dtype=int)
                # contribution of each i to the maximal entry
                contrib = cand[idxs] * cand[k_max - idxs]
                if contrib.size == 0:
                    continue
                i = idxs[int(contrib.argmax())]
                # pick a low‑value index as destination
                low_candidates = np.where(cand <= np.quantile(cand, 0.2))[0]
                if low_candidates.size == 0:
                    j = rng.integers(n)
                else:
                    j = int(rng.choice(low_candidates))
                max_delta = cand[i] * 0.05
                if max_delta <= 0.0:
                    continue
                delta = rng.random() * max_delta
                cand[i] -= delta
                cand[j] += delta
            elif move_type == 2:        # block averaging
                start = rng.integers(0, n - 1)
                length = rng.integers(2, min(60, n - start))
                block = slice(start, start + length)
                avg = cand[block].mean()
                cand[block] = avg
            elif move_type == 3:        # small block shuffle
                start = rng.integers(0, n - 1)
                length = rng.integers(3, min(30, n - start))
                block = slice(start, start + length)
                sub = cand[block]
                rng.shuffle(sub)               # in‑place shuffling
                cand[block] = sub
            else:                        # complete random permutation
                rng.shuffle(cand)

            # projection (only needed for moves that may break the sum)
            cand = _project_to_simplex(cand, S)

            sc = fast_evaluate(cand)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = cand.copy()
                a = cand
                print(f"[hill] trial {trial:07d} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Simple ordering heuristics
    # --------------------------------------------------------------
    if time.time() < deadline and best_score < float("inf"):
        # descending order
        desc = np.sort(best_a)[::-1]
        maybe_update(_project_to_simplex(desc, S))

        # ascending order
        asc = np.sort(best_a)
        maybe_update(_project_to_simplex(asc, S))

        # interleaved big‑small pattern
        vals = np.sort(best_a)[::-1]
        half = len(vals) // 2
        left = vals[:half][::-1]
        right = vals[half:][::-1]
        inter = np.empty_like(vals)
        inter[0::2] = left
        inter[1::2] = right
        maybe_update(_project_to_simplex(inter, S))

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
