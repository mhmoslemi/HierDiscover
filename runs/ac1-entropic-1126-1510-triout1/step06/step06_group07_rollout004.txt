    A = np.fft.rfft(a, n=m)

def compute_max_conv_and_subgradient(a):

def propose_candidate(seed=12345, budget_s=1000):

    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5

    # Determine n
    try: ref_len = len(height_sequence_1)
    except NameError: ref_len = None
    if ref_len:
        n = min(max(ref_len * 2, 4000), 12000)
    else:
        n = 8000

    S = math.sqrt(2.0 * n)

    best_a = np.zeros(n)
    best_score = float('inf')

    def maybe_update(a):
        nonlocal best_score, best_a
        sc = evaluate_sequence(a.tolist())
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            print(f"[improve] score {best_score:.6f}")

    # Stage 1: L-BFGS on smooth max

    # generate initial y vectors

    # Uniform: zeros
    init_y_list = []
    init_y_list.append(np.zeros(n))

    # Exponential decreasing
    alpha = rng.uniform(0.001, 0.004) * (n/2000.0)
    init_y_list.append(-alpha * np.arange(n))

    # Exponential increasing
    init_y_list.append(-alpha * np.arange(n-1,-1,-1))

    # Random Gaussian
    init_y_list.append(rng.standard_normal(n) * 0.6)

    # Sparse Dirichlet draws
    for _ in range(8):
        d_alpha = rng.uniform(0.02, 0.5)  # more sparsity
        p = rng.dirichlet([d_alpha]*n)
        eps = 1e-12
        init_y_list.append(np.log(p + eps))

    # reference if available
    if ref_len:
        a_ref = np.asarray(height_sequence_1, dtype=float)
        a_ref = project_to_simplex(a_ref, S)
        eps = 1e-12
        init_y_list.append(np.log(a_ref + eps))

    # parameters for cooling
    mu_factor = 0.6
    mu_min = 1e-12
    maxiter_mu = 800

    for rid, y0 in enumerate(init_y_list):
        if time.time() > deadline:
            break
        print(f"[restart {rid+1}/{len(init_y_list)}] start L-BFGS")
        y = y0.copy()

        # compute initial mu as max conv of a
        a_cur = S * softmax(y)
        conv_cur = compute_conv_fft(a_cur)
        mu = max(conv_cur.max(), 1.0)  # at least 1

        while mu > mu_min and time.time() < deadline:
            # define objective wrapper for current mu
            def obj_func(y_vec):
                # compute value and grad
                p = softmax(y_vec)
                a = S * p
                conv = np.convolve(a, a)  # maybe use fft? but okay
                max_c = conv.max()
                # compute smooth weights
                if mu > 0:
                    diff = (conv - max_c) / mu
                    diff = np.clip(diff, -700, 700)
                    exp_diff = np.exp(diff)
                    Z = exp_diff.sum()
                    w = exp_diff / Z
                    f_val = max_c + mu * math.log(Z)
                else:
                    mask = conv == max_c
                    cnt = np.count_nonzero(mask)
                    w = np.zeros_like(conv)
                    if cnt>0:
                        w[mask] = 1.0 / cnt
                    f_val = max_c

                # gradient w.r.t a
                # conv of w and reversed a
                a_rev = a[::-1]
                conv_wr = np.convolve(w, a_rev)
                g_a = 2.0 * conv_wr[len(a)-1:len(a)-1+len(a)]
                # gradient wrt y via softmax Jacobian
                inner = np.dot(g_a, p)
                grad_y = S * p * (g_a - inner)

                return float(f_val), grad_y.astype(np.float64)

            # Use L-BFGS-B
            try:
                res = opt.minimize(obj_func, y, method='L-BFGS-B',
                                   jac=True,
                                   options={'maxiter': maxiter_mu, 'gtol':1e-12, 'disp':False})
            except Exception as e:
                print("[error] L-BFGS:", e)
                break

            y = res.x
            a = S * softmax(y)
            maybe_update(a)

            mu *= mu_factor

    # Stage 2: Accelerated projected subgradient descent
    if time.time() < deadline and best_score < float('inf'):
        a = best_a.copy()
        # momentum variables
        v = np.zeros_like(a)
        # parameters
        it = 0
        max_iters = 20000
        step0 = S * 0.4 / n
        while time.time() < deadline and it < max_iters:
            g, max_c = compute_hard_subgradient(a)  # we need to implement this
            # step size schedule (diminishing)
            step = step0 / (1.0 + 0.0005 * it)
            # Nesterov update: a_new = a - step * g + momentum?
            a_proposed = a - step * g
            a_proj = project_to_simplex(a_proposed, S)
            # evaluate
            sc = evaluate_sequence(a_proj.tolist())
            if sc < best_score - 1e-12:

                best_a = a_proj.copy()
                a = a_proj
                print(f"[subgrad] iter {it} score {best_score:.6f}")
                # optionally reset momentum
                v = np.zeros_like(a)
            else:

                v = 0.9 * v - step * g
                a = project_to_simplex(a + v, S)
            it += 1

    # Stage 3: Simulated annealing hill climbing
    if time.time() < deadline and best_score < float('inf'):
        a = best_a.copy()
        T0 = 0.05 * best_score  # initial temperature as a fraction of score
        t_end = deadline - 0.5
        # linear cooling schedule
        while time.time() < t_end:
            T = T0 * (deadline - time.time()) / (deadline - t_end)
            # propose move
            move_type = rng.integers(0, 4)
            cand = a.copy()
            if move_type == 0:
                # swap random indices
                i, j = rng.choice(n, size=2, replace=False)
                cand[i], cand[j] = cand[j], cand[i]
            elif move_type == 1:
                # transfer delta from i to j
                i = rng.choice(np.where(cand > 1e-12)[0])
                j = rng.choice(n)
                max_delta = cand[i] * 0.1
                delta = rng.random() * max_delta
                cand[i] -= delta
                cand[j] += delta
            elif move_type == 2:
                # block-average but random factor
                start = rng.integers(0, n-1)
                length = rng.integers(2, min(100, n-start))
                avg = cand[start:start+length].mean()
                cand[start:start+length] = avg
            else:
                # Gaussian jitter on a small subset
                idxs = rng.choice(n, size=rng.integers(1,5), replace=False)
                perturb = rng.normal(0, 0.02 * cand[idxs].mean(), size=idxs.shape)
                cand[idxs] += perturb
                # enforce non-negative
                cand[idxs] = np.clip(cand[idxs], 0.0, None)
            # project to simplex for sum constraint
            cand = project_to_simplex(cand, S)
            # evaluate
            sc = evaluate_sequence(cand.tolist())
            delta = sc - best_score
            if delta < -1e-12 or rng.random() < math.exp(-delta / max(T, 1e-12)):
                # accept move
                a = cand
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = cand.copy()
                    print(f"[anneal] score {best_score:.6f}")

    # Stage 4: final sorting heuristics
    if time.time() < deadline:
        # descending
        cand_desc = np.sort(best_a)[::-1]
        cand_desc = project_to_simplex(cand_desc, S)
        maybe_update(cand_desc)
        # interleaved big small
        vals_desc = np.sort(best_a)[::-1]
        half = len(vals_desc)//2
        left = vals_desc[:half][::-1]  # smallest half reversed? Actually we want maybe small at left?
        right = vals_desc[half:][::-1]
        inter = np.empty_like(vals_desc)
        inter[0::2] = left
        inter[1::2] = right
        inter = project_to_simplex(inter, S)
        maybe_update(inter)

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()

def conv_fft(a, b):
    m = next_pow_two(len(a) + len(b) - 1)
    A = np.fft.rfft(a, n=m)
    B = np.fft.rfft(b, n=m)

"""
Propose a step‑function with a low evaluate_sequence() score.

The algorithm consists of four phases:

1.  Multiple random restarts of a smooth‑max surrogate, minimised by
    L‑BFGS‑B.  The surrogate f_μ = max(conv) + μ·log‑sum‑exp((conv‑max)/μ)
    is smoothed with a geometric cooling schedule (μ←0.6·μ) until μ≈0.
    Each restart starts from a different log‑probability vector
    (uniform, exponential, Gaussian, sparse Dirichlet or the reference).

2.  Accelerated projected sub‑gradient descent on the original—
    non‑smoothed—objective.  The sub‑gradient of max(conv) is obtained
    analytically via convolution with an active‑set weight vector w.
    A Nesterov‑style momentum term speeds up convergence.

3.  Simulated‑annealing hill climbing.  Random elementary moves
    (swap, δ‑mass transfer, block‑averaging, jitter) are proposed.
    Moves are accepted if they improve the exact score or, with a
    probability exp(‑Δ/T), if they worsen it; the temperature T decays
    linearly to zero.

4.  Simple deterministic post‑processing (sorted, interleaved) to
    capture any remaining improvement.

All helpers are defined at the top level, the routine respects the
user’s time budget (default 1000 s) and returns the best sequence found.
"""


import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = z} (z ≥ 0).
    """
    if z == 0.0:
        return np.zeros_like(v)

    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _next_pow_two(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full convolution of a and b using FFT (real inputs).
    Returns an array of length len(a)+len(b)-1.
    """
    m = _next_pow_two(len(a) + len(b) - 1)
    A = np.fft.rfft(a, n=m)
    B = np.fft.rfft(b, n=m)
    c = np.fft.irfft(A * B, n=m)
    return c[: len(a) + len(b) - 1]


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(y: np.ndarray, mu: float, n: int, S: float):
    """
    Value and gradient of the smooth‑max surrogate

        f_μ(y) = max(conv) + μ·log Σ exp((conv‑max)/μ)

    Parameters
    ----------
    y : ndarray (n,)
        Unconstrained log‑probabilities (soft‑max will be applied).
    mu : float > 0 (when mu=0 the hard max is used).
    n, S : problem size and target ℓ¹‑norm.

    Returns
    -------
    f_val : float – smooth‑max value
    grad_y : ndarray (n,) – gradient w.r.t. y
    """
    p = _softmax(y)          # Σ p_i = 1, p_i ≥ 0
    a = S * p                # Σ a_i = S

    conv = _conv_fft(a, a)   # full auto‑convolution, length 2·n‑1
    max_c = conv.max()

    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)          # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv)
        if cnt > 0:
            w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. a : g_a[i] = 2 * Σ_k w_k * a_{k-i}
    a_rev = a[::-1]
    conv_wr = _conv_fft(w, a_rev)                   # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1: n - 1 + n]

    # chain rule through the soft‑max
    inner = np.dot(g_a, p)                           # scalar Σ p_j g_a_j
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable used by scipy.optimize for the smoothed max objective."""
    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _smooth_obj_and_grad(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
def _hard_subgradient(a: np.ndarray):
    """
    Sub‑gradient of the true max‑objective  max(conv)  at point a.
    Returns (g, max_c) where g has length n.
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    max_c = conv.max()
    mask = conv == max_c
    cnt = np.count_nonzero(mask)

    w = np.zeros_like(conv)
    if cnt > 0:
        w[mask] = 1.0 / cnt

    a_rev = a[::-1]
    conv_wr = _conv_fft(w, a_rev)                     # length 3·n‑2
    g = 2.0 * conv_wr[n - 1: n - 1 + n]
    return g, max_c
