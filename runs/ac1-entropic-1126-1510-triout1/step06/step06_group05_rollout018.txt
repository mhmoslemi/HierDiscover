analysisThe problem is a search for a nonnegative sequence that minimizes evaluate_sequence, which is defined as:

- Input sequence a_i (list of floats).
- Convert negative to 0, clamp values to [0,1000] but those are irrelevant because we will use normalized values.
- Compute n = len(a)
- Compute b = convolution (full linear) of a with itself, length 2n-1.
- max_b = max(b)
- sum_a = sum(a)
- If sum_a < 0.01: return inf (so we must have sum > 0.01)
- Return 2 * n * max_b / (sum_a**2)

Thus the objective is 2 * n * max(conv(a,a)) / (sum(a)^2). Since we can scale the vector a arbitrarily (nonnegative entries), scaling a by factor c multiplies sum by c and max_b by c^2 (since convolution scales quadratically). Then evaluate_sequence becomes 2 n (c^2 max_b0) / (c^2 sum0^2) = 2 n max_b0 / sum0^2, i.e., scale-invariant. We can normalise sum(a) = 1, or any constant; previously they normalised sum = sqrt(2n). """
Propose a low‑score step‑function.

The search consists of four stages:

1. **Broad seeding** – generate many probability vectors of several
   lengths (arcsine, Beta, Dirichlet, power‑law, mixtures, interpolated
   previous best).  Keep the few with the smallest cheap proxy
   `2·n·max(conv)`.

2. **Refinement** – for each selected seed run a continuation of three
   optimisers:
   * Adam on the smooth‑max surrogate (gradients via FFT).
   * Adaptive sub‑gradient descent on the exact ℓ∞ objective.
   * (optional, for moderate sizes) a linear‑programming “dual‑widening’’
     step that solves `max ∑b` s.t. `f * b ≤ M`.

3. **Polishing** – a light simulated‑annealing phase that moves a small
   amount of mass between two positions while evaluating the exact peak.

4. **Final check** – call the provided `evaluate_sequence` to obtain the
   true score and return the best sequence found before the time budget
   expires.

All operations respect the simplex `Σ a_i = 1, a_i ≥ 0`.  The algorithm
is fully self‑contained, uses only NumPy/ SciPy / optional CVXPy and
fits easily into the 1000 s budget.
"""

import math
import time
import random
import numpy as np
from numpy.fft import rfft, irfft
import scipy.linalg

# optional LP support
try:
    import cvxpy as cp
    _cvxpy_available = True
except Exception:  # pragma: no cover
    _cvxpy_available = False


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution of length `n`."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) on a uniform grid, normalised to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_vals = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    vals[~np.isfinite(vals)] = 0.0
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _powerlaw_boundary_distribution(n: int, p: float) -> np.ndarray:
    """Distribution ∝ ((i+0.5)*(n‑i‑0.5))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5) + 1e-12) ** (-p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth‑max surrogate
    Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k).
    Returns (grad, current_maximum_of_conv).
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.01,
                   alpha_max: float = 2000.0,
                   lr0: float = 0.25,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """Adam optimisation of the smooth‑max surrogate."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny jitter to escape plateaus
        if t % 500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """One sub‑gradient step for the exact ℓ∞ objective."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adaptive_subgrad_refine(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Adaptive sub‑gradient descent on the exact max‑convolution objective.
    Step size decays but is increased when a move succeeds.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    step = 0.02
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)

        a_candidate = a - step * grad
        a_candidate = _project_onto_simplex(a_candidate, s=1.0)
        new_peak = _fft_self_convolve(a_candidate).max()

        if new_peak < cur_max:
            a = a_candidate
            cur_max = new_peak
            step *= 1.1                     # grow step on success
            if new_peak < best_peak:
                best_peak = new_peak
                best_a = a.copy()
        else:
            step *= 0.7                     # shrink step on failure

        # occasional random jitter to avoid stagnation
        if it % 400 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a


# ----------------------------------------------------------------------
def _lp_improve(a: np.ndarray,
                max_iters: int = 5,
                mix_t: float = 0.4,
                inner_deadline: float = None) -> np.ndarray:
    """
    Dual‑widening LP improvement.
    For each iteration maximise Σ b_j subject to (a∗b)_k ≤ M
    where M = max(a∗a).  The resulting normalised vector g is mixed into a.
    """
    if not _cvxpy_available:
        return a

    n = a.shape[0]
    for _ in range(max_iters):
        if inner_deadline is not None and time.time() > inner_deadline:
            break
        M = _fft_self_convolve(a).max()

        # build convolution matrix A (2n‑1)×n  – dense, but n is kept moderate
        col = np.r_[a, np.zeros(n - 1)]
        row = np.r_[a[0], np.zeros(n - 1)]
        A = scipy.linalg.toeplitz(col, row)        # dense array

        b_var = cp.Variable(n, nonneg=True)
        constraints = [A @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception:  # pragma: no cover
            break
        if b_var.value is None:
            break
        b_vec = np.maximum(b_var.value, 0.0)
        s = b_vec.sum()
        if s <= 1e-12:
            break
        g = b_vec / s
        a = (1.0 - mix_t) * a + mix_t * g
        a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, deadline: float):
    """Light simulated‑annealing polish (exact max‑convolution)."""
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_peak = _fft_self_convolve(cur_a).max()
    best_a = cur_a.copy()
    best_peak = cur_peak

    start = time.time()
    total_time = max(deadline - start, 1e-6)

    start_temp = max(cur_peak, 1e-3) * 0.5
    final_temp = start_temp * 1e-5
    if final_temp <= 0.0:
        final_temp = 1e-9
    cooling_rate = -math.log(final_temp / start_temp) / total_time

    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1
        src = np.random.randint(n)
        if cur_a[src] <= 1e-12:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue
        max_delta = min(cur_a[src], 0.05)
        delta = np.random.rand() * max_delta

        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_peak = _fft_self_convolve(a_new).max()
        delta_obj = new_peak - cur_peak

        elapsed = time.time() - start
        temperature = start_temp * math.exp(-cooling_rate * elapsed)

        accept = False
        if delta_obj < 0.0:
            accept = True
        else:
            prob = math.exp(-delta_obj / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_a = a_new
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_a = cur_a.copy()

        if iter_cnt % 20000 == 0 and np.random.rand() < 0.05:
            cur_a = best_a.copy()
            cur_peak = best_peak

    return best_a, best_peak

def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a non‑negative sequence to a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a step‑function with a low `evaluate_sequence` score.
    Returns a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a tiny safety margin

    # ------------------------------------------------------------------
    # Try to obtain a previous construction (may not exist).
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_seq = np.maximum(base_seq, 0.0)
        base_seq = _project_onto_simplex(base_seq, s=1.0)
        base_n = len(base_seq)
    except Exception:
        base_seq = None
        base_n = 2500

    # ------------------------------------------------------------------
    # Lengths to explore (ensure at least 800 entries).
    lengths = [
        max(800, int(base_n * f))
        for f in (0.5, 0.8, 1.0, 1.2, 1.5)
        if int(base_n * f) >= 800
    ]
    if not lengths:
        lengths = [max(800, base_n)]

    print("[stage 1] broad seeding")
    all_candidates = []   # list of (fast_score, vector, length)

    for n in lengths:
        if time.time() >= deadline:
            break

        starters = {}

        # classic families
        starters["arcsine"] = _arcsine_distribution(n)
        starters["uniform"] = np.full(n, 1.0 / n, dtype=np.float64)

        for conc in (0.2, 0.5, 1.0, 2.0):
            starters[f"dirichlet_{conc:.1f}"] = np.random.dirichlet(np.full(n, conc))

        for a in (0.3, 0.4, 0.5, 0.6, 0.7):
            starters[f"beta_eq_{a:.1f}"] = _beta_distribution(n, a, a)

        for a in (0.25, 0.35, 0.45):
            for b in (0.55, 0.65, 0.75):
                starters[f"beta_{a:.2f}_{b:.2f}"] = _beta_distribution(n, a, b)

        for p in (0.4, 0.5, 0.6):
            starters[f"power_{p:.2f}"] = _powerlaw_boundary_distribution(n, p)

        for w in (0.2, 0.5, 0.8):
            starters[f"mix_{w:.1f}"] = (1.0 - w) * _arcsine_distribution(n) + w * np.full(n, 1.0 / n)

        if base_seq is not None:
            starters["prev"] = _interpolate_to_length(base_seq, n)

        # cheap proxy evaluation
        for tag, vec in starters.items():
            vec = _project_onto_simplex(vec, s=1.0)
            peak = _fft_self_convolve(vec).max()
            fast_score = 2.0 * n * peak
            all_candidates.append((fast_score, vec, n))
            print(f"[init] n={n:4d} {tag:<15s} fast_score={fast_score:.6f}")

    # keep only a handful of the best seeds
    all_candidates.sort(key=lambda x: x[0])
    top_candidates = all_candidates[:5]   # five most promising seeds
    print(f"[stage 1] retained {len(top_candidates)} top candidates")

    # ------------------------------------------------------------------
    # Main refinement loop – keep refining while time permits.
    best_seq = top_candidates[0][1].copy()
    best_fast = top_candidates[0][0]

    iteration = 0
    while time.time() < deadline - 5.0:   # reserve a few seconds for final eval
        iteration += 1
        # pick a seed (biased towards better ones)
        weights = np.linspace(1.0, 0.5, len(top_candidates))
        weights /= weights.sum()
        fast, vec, n = random.choices(top_candidates, weights=weights, k=1)[0]

        # allocate a slice of the remaining budget
        remaining = deadline - time.time()
        slice_time = min(remaining * 0.30, 30.0)   # at most 30 s per slice
        slice_deadline = time.time() + slice_time

        # 1) Adam optimisation (smooth‑max)
        a_adam, _ = _adam_optimize(vec, slice_deadline,
                                   init_alpha=0.5, alpha_max=1500.0, lr0=0.2)

        # 2) Adaptive sub‑gradient descent on the exact max
        sub_deadline = time.time() + (slice_deadline - time.time()) * 0.45
        a_sub = _adaptive_subgrad_refine(a_adam, sub_deadline)

        # 3) LP improvement (only for moderate sizes)
        if n <= 1800 and _cvxpy_available:
            lp_deadline = time.time() + (slice_deadline - time.time()) * 0.20
            a_lp = _lp_improve(a_sub, max_iters=7, mix_t=0.4,
                               inner_deadline=lp_deadline)
        else:
            a_lp = a_sub

        # 4) Simulated‑annealing polishing for the leftover slice time
        if time.time() < slice_deadline:
            a_final, final_peak = _simulated_annealing(a_lp, slice_deadline)
        else:
            a_final = a_lp
            final_peak = _fft_self_convolve(a_final).max()

        final_fast = 2.0 * n * final_peak
        if final_fast < best_fast:
            best_fast = final_fast
            best_seq = a_final.copy()
            print(f"[update] iteration {iteration} – new best fast_score={best_fast:.6f} (n={n})")

    # ------------------------------------------------------------------
    # Final exact evaluation.
    try:
        true_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] true evaluate_sequence score={true_score:.6f}")
        if true_score < best_fast:
            best_fast = true_score
    except Exception as exc:
        print("[final] evaluate_sequence raised:", exc)

    print(f"Best score ≈ {best_fast:.6f} with length={len(best_seq)}")
    return best_seq.tolist()
