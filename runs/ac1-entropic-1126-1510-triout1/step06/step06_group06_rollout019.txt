def _two_level_family(n, deadline):
    best_seq = None
    best_score = float('inf')
    step_L = max(1, n // 100) # 1% step
    r_grid = np.logspace(-2, 2.3, num=30) # from 0.01 to ~200

"""
Hybrid search for a low‑score non‑negative step function.

The routine works in several stages under a fixed time budget.

0️⃣  Broad sampling of many analytic families (uniform, arcsine, beta,
    exponential, linear ramps, logistic) for several lengths.  The best
    fast‑score (2·n·max(conv)) candidate is kept.

1️⃣  Simple two‑level block families (front block height p, back block q)
    are searched over block lengths and height ratios (including ratios
    < 1 by swapping the blocks).

2️⃣  Adam optimisation on the smooth‑max surrogate of the ℓ∞‑convolution
    (log‑sum‑exp).  Several restarts with tiny random jitter are performed.

3️⃣  Exact‑max sub‑gradient descent (as in the reference code) followed
    by a specialised “active‑shift” descent that always pushes mass away
    from the currently maximal convolution index.  Finally a greedy
    directed local‑search finishes the refinement.

4️⃣  The best vector is evaluated with the black‑box `evaluate_sequence`
    and returned as a plain Python list.

All vectors are projected onto the simplex Σa_i = 1 after every update,
so the evaluation score is simply 2·n·max(conv).  The algorithm respects
the 1000 s budget (a safety margin of 1 s is kept).  Printing statements
show progress and intermediate scores.
"""

import time
import math

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    exp_term = np.exp(alpha * (b - cur_max))
    w_sum = exp_term.sum()
    w = exp_term / w_sum
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.01,
                   alpha_max: float = 5000.0,
                   lr0: float = 0.3,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best probability vector encountered and its exact peak.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size schedule (slow decay)
        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny jitter
        if t % 400 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    One sub‑gradient step for the exact ℓ∞ objective.
    Uses an equal weight on all indices attaining the current maximum.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Sub‑gradient descent on the exact max‑convolution objective.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.02 / np.sqrt(it)  # diminishing step
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


# ----------------------------------------------------------------------
def _directed_local_search(a0: np.ndarray, deadline: float):
    """
    Greedy mass‑transfer search that always targets the current peak.
    Only improving moves are kept.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        max_idx = np.random.choice(np.where(b >= cur_max - 1e-12)[0])
        i_low = max(0, max_idx - (n - 1))
        i_high = min(max_idx, n - 1)
        i_vals = np.arange(i_low, i_high + 1, dtype=int)
        prod = a[i_vals] * a[max_idx - i_vals]
        if prod.sum() == 0.0:
            src = np.random.randint(n)
        else:
            prob = prod / prod.sum()
            src = np.random.choice(i_vals, p=prob)

        dst = np.random.randint(n)
        if dst == src:
            continue

        max_delta = min(a[src], 0.05)          # up to 5 % of src
        delta = np.random.rand() * max_delta

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        b_new = _fft_self_convolve(a_new)
        new_max = b_new.max()

        if new_max < cur_max - 1e-15:
            a = a_new
            b = b_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()

        if np.random.rand() < 0.005:
            a += np.random.randn(n) * 1e-7
            a = _project_onto_simplex(a, s=1.0)
            b = _fft_self_convolve(a)
            cur_max = b.max()

    return best_a


# ----------------------------------------------------------------------
def _active_shift_descent(a0: np.ndarray, deadline: float):
    """
    Targeted descent that always focuses on the currently
    maximal convolution index (single‑active‑shift gradient).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())                       # most offending shift
        # gradient for this shift: grad_i = 2 * a_{k-i} when i participates
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        idx = np.arange(low, high + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # step size – modest, diminishing
        step = 0.04 / np.sqrt(it)

        # simple backtracking line‑search
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            a_new = a - step * factor * grad
            a_new = _project_onto_simplex(a_new, s=1.0)
            new_peak = _fft_self_convolve(a_new).max()
            if new_peak < best_peak - 1e-15:
                a = a_new
                best_peak = new_peak
                best_a = a_new.copy()
                improved = True
                break
        if not improved:
            # occasional jitter to escape flat regions
            if np.random.rand() < 0.01:
                a += np.random.randn(n) * 1e-7
                a = _project_onto_simplex(a, s=1.0)

    return best_a


# ----------------------------------------------------------------------
def _two_level_family(n: int, deadline: float):
    """
    Search a simple two‑level family:
        a_i = p   for i < L   (front block)
        a_i = q   otherwise   (back block)
    with p = r·q (r may be >1 or <1 – the block order can be swapped).
    Returns the best vector found and its fast score.
    """
    best_seq = None
    best_score = float('inf')
    step_L = max(1, n // 100)          # ≈1 % of length
    # explore ratios from 0.01 to ≈200 (log‑space)
    r_grid = np.logspace(-2, 2.3, num=30)

    for L in range(step_L, n, step_L):
        if time.time() > deadline:
            break
        back_len = n - L
        for r in r_grid:
            if r >= 1.0:
                denom = L * r + back_len
                q = 1.0 / denom
                p = r * q
            else:                                   # swap roles: back block higher
                denom = L + back_len * r
                p = 1.0 / denom
                q = r * p
            seq = np.empty(n, dtype=np.float64)
            seq[:L] = p
            seq[L:] = q
            seq = _project_onto_simplex(seq, s=1.0)
            peak = _fft_self_convolve(seq).max()
            score = 2.0 * n * peak
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
    return best_seq, best_score


# ----------------------------------------------------------------------
def _sample_family(name: str, n: int) -> np.ndarray:
    """Return a normalized vector for a named analytic family."""
    if name == "arcsine":
        v = 1.0 / np.sqrt((np.arange(n) + 0.5) *
                          (n - np.arange(n) - 0.5))
    elif name == "uniform":
        v = np.full(n, 1.0 / n, dtype=np.float64)
    elif name.startswith("beta_"):
        a = float(name.split("_")[1])
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        log_vals = (a - 1.0) * np.log(x) + (a - 1.0) * np.log(1.0 - x)
        v = np.exp(log_vals)
        v[~np.isfinite(v)] = 0.0
        if v.sum() == 0.0:
            v = np.full(n, 1.0 / n)
    elif name.startswith("exp_"):
        lam = float(name.split("_")[1])
        v = np.exp(-lam * np.arange(n))
    elif name == "ramp_up":
        v = np.arange(1, n + 1, dtype=np.float64)
    elif name == "ramp_down":
        v = np.arange(n, 0, -1, dtype=np.float64)
    elif name.startswith("logistic_"):
        # logistic with parameter beta (steepness)
        beta = float(name.split("_")[1])
        x = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
        v = 1.0 / (1.0 + np.exp(beta * x))
    else:
        raise ValueError(f"unknown family {name}")
    v = np.maximum(v, 0.0)
    v /= v.sum()
    return v


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Main entry point – returns a list of non‑negative numbers (the step
    heights) that strives to minimise ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # --------------------------------------------------------------
    # Try to reuse a previously discovered construction if present.
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_seq = _project_onto_simplex(base_seq, s=1.0)
        base_n = len(base_seq)
    except Exception:
        base_seq = None
        base_n = 4000

    # --------------------------------------------------------------
    # Lengths we will explore – a blend around the previous best.
    lengths = [max(500, int(base_n * f))
               for f in (0.6, 0.8, 1.0, 1.2, 1.4, 1.6, 1.8)
               if int(base_n * f) >= 500]
    if not lengths:
        lengths = [max(500, base_n)]

    overall_best_seq = None
    overall_best_score = float('inf')

    # --------------------------------------------------------------
    # Phase 0 – broad family sampling (including new families)
    print("[phase 0] broad family sampling")
    for n in lengths:
        if time.time() >= deadline:
            break

        families = {
            "arcsine": _sample_family("arcsine", n),
            "uniform": _sample_family("uniform", n),
            "beta_0.20": _sample_family("beta_0.20", n),
            "beta_0.30": _sample_family("beta_0.30", n),
            "beta_0.40": _sample_family("beta_0.40", n),
            "beta_0.60": _sample_family("beta_0.60", n),
            "beta_0.80": _sample_family("beta_0.80", n),
            "exp_0.001": _sample_family("exp_0.001", n),
            "exp_0.002": _sample_family("exp_0.002", n),
            "exp_0.005": _sample_family("exp_0.005", n),
            "ramp_up": _sample_family("ramp_up", n),
            "ramp_down": _sample_family("ramp_down", n),
            "logistic_0.02": _sample_family("logistic_0.02", n),
        }

        if base_seq is not None:
            # interpolated version of the previous best to this length
            old_n = len(base_seq)
            x_old = np.linspace(0.0, 1.0, old_n)
            x_new = np.linspace(0.0, 1.0, n)
            interp = np.interp(x_new, x_old, base_seq)
            interp = np.maximum(0.0, interp)
            interp /= interp.sum()
            families["prev"] = interp

        for tag, vec in families.items():
            vec = _project_onto_simplex(vec, s=1.0)
            peak = _fft_self_convolve(vec).max()
            fast_score = 2.0 * n * peak
            if fast_score < overall_best_score:
                overall_best_score = fast_score
                overall_best_seq = vec.copy()
                print(f"[init] n={n:4d} {tag:<12s} fast_score={fast_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – cheap two‑level parametric search
    print("[phase 1] two‑level parametric search")
    for n in lengths:
        if time.time() >= deadline:
            break
        seq, score = _two_level_family(n, deadline)
        if seq is not None and score < overall_best_score:
            overall_best_score = score
            overall_best_seq = seq.copy()
            print(f"[two‑level] n={n:4d} fast_score={score:.6f}")

    # fallback (should never be needed)
    if overall_best_seq is None:
        n = max(500, base_n)
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print("[fallback] using uniform distribution")

    # --------------------------------------------------------------
    # Phase 2 – Adam optimisation (smooth‑max surrogate)
    time_left = deadline - time.time()
    if time_left > 5.0:
        n = len(overall_best_seq)
        print(f"[phase 2] Adam optimisation (n={n}) – time left {time_left:.1f}s")
        phase_end = time.time() + time_left * 0.55
        a_opt, _ = _adam_optimize(overall_best_seq, phase_end)
        overall_best_seq = a_opt.copy()
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print(f"[phase 2] after Adam fast_score={overall_best_score:.6f}")

        # one extra short restart if time permits
        if time.time() + 5.0 < phase_end:
            a_restart = overall_best_seq + np.random.randn(n) * 1e-6
            a_restart = _project_onto_simplex(a_restart, s=1.0)
            a_opt2, _ = _adam_optimize(a_restart, phase_end)
            score2 = 2.0 * n * _fft_self_convolve(a_opt2).max()
            if score2 < overall_best_score:
                overall_best_seq = a_opt2.copy()
                overall_best_score = score2
                print(f"[phase 2] after extra restart fast_score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – exact max descent + dedicated active‑shift refinement
    time_left = deadline - time.time()
    if time_left > 4.0:
        n = len(overall_best_seq)
        print(f"[phase 3] sub‑gradient & local search (n={n}) – time left {time_left:.1f}s")
        sub_end = time.time() + time_left * 0.45
        a_ref = _refine_max(overall_best_seq, sub_end)

        # active‑shift descent (focuses on the current peak)
        a_active = _active_shift_descent(a_ref, deadline)

        # final greedy local search
        a_final = _directed_local_search(a_active, deadline)

        overall_best_seq = a_final.copy()
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print(f"[phase 3] final fast_score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # Final black‑box evaluation
    try:
        true_score = evaluate_sequence(overall_best_seq.tolist())
        print(f"[final] evaluate_sequence score={true_score:.6f}")
        if true_score < overall_best_score:
            overall_best_score = true_score
    except Exception as exc:
        print("[final] evaluate_sequence raised:", exc)

    print(f"Best approx. score {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
