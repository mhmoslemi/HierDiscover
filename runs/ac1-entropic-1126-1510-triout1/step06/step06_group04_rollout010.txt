def init_sequence(rng, n, restart):
    # Base distribution: mixture of up to 3 Beta components
    if restart == 0:
        # uniform
        return np.full(n, 1.0/n)
    elif restart == 1:
        # arcsine / Beta(0.5,0.5)
        return _beta_distribution(n, 0.5, 0.5)
    elif restart == 2:
        # random mixture
        k = rng.integers(2,5) # number components
        weights = rng.dirichlet(np.ones(k))
        vec = np.zeros(n)
        for i in range(k):
            alpha = 10**rng.uniform(-2,1)
            beta = 10**rng.uniform(-2,1)
            vec += weights[i] * _beta_distribution(n, alpha, beta)
        vec = _project_onto_simplex(vec)
        return vec
```

def _smooth_max_gradient(a, tau):
    conv = _self_convolution_full(a)
    max_val = conv.max()
    if tau <= 0 or tau < 1e-12:
        # subgradient
        active = np.where(conv >= max_val - max_val*1e-12)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
    else:
        # compute softmax weights
        # shift for numerical stability
        shifted = conv - max_val
        exp_vals = np.exp(shifted / tau)
        sum_exp = exp_vals.sum()
        if sum_exp == 0:
            w = np.zeros_like(conv)
        else:

    # gradient to increase max (so descent is -grad)
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[len(a)-1 : len(a)-1 + len(a)]

def propose_candidate(seed=0,budget_s=1000):

    start = time.time()
    deadline = start + budget_s - 0.5

    # attempt to load height_sequence_1 as base

    # candidate lengths

    best_seq = None
    best_score = np.inf

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"=== Length {n} ===")

            # mixture of random betas

                a = 10**rng.uniform(-2,1)

def _adam_optimize(a, deadline, rng, max_iters=5000):
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a.copy()
    best_val = _raw_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # compute smoothed gradient
        grad, cur_max = _smooth_max_subgradient(a)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = 0.2 / math.sqrt(it)  # dec

"""
Propose a low‑score step function.

The algorithm works in three stages.

1. **Generate diverse initial candidates** – for each tested length
   (2000, 4000, 8000) we create a handful of hand‑crafted distributions:
   uniform, the arcsine β(½,½), an interpolation of a possible historic
   best, and a few random mixtures of beta components.  All candidates are
   projected onto the probability simplex (Σa=1, a≥0).

2. **Refine with projected Adam on a smooth approximation of the max‑convolution**.
   The true objective is 2·n·max(a∗a) (a is already normalised), i.e.
   minimise the maximal entry of the self‑convolution.  Direct sub‑gradients
   are discontinuous, so we replace the max by the smooth “log‑sum‑exp”
   envelope τ·log(∑exp((a∗a)/τ)).  Its gradient is a weighted sum of the
   individual convolution gradients; the weight vector is a soft‑max over the
   convolution values.  τ is chosen as a fraction of the current maximum and
   is reduced automatically, giving an annealing‑style optimisation.

   Adam (β₁=0.9, β₂=0.999) with a 1/√t step‑size schedule provides fast
   convergence on the simplex – after each update we project back onto the
   simplex.  Small random jitter is added occasionally to escape plateaus.

3. **Optional up‑sampling** – if time permits the best vector is linearly
   interpolated to a longer length (≈12000) and refined again; a finer grid
   can bring the constant factor a little lower.

The best vector found (still normalised to Σa=1) is finally handed to the
provided ``evaluate_sequence`` which rescales it internally.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on n equally spaced mid‑points, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf (stable)
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logpdf = np.minimum(logpdf, 700.0)                     # avoid overflow
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _raw_score(a: np.ndarray) -> float:
    """Proxy score (Σa=1) = 2·n·max(a∗a)."""
    n = a.shape[0]
    return 2.0 * n * _self_convolution_full(a).max()


# ----------------------------------------------------------------------
def _smooth_max_subgradient(a: np.ndarray,
                            tau_factor: float = 0.10,
                            min_tau: float = 1e-12):
    """
    Gradient of the smooth approximation τ·log Σexp((a∗a)/τ).

    Returns (grad, cur_max).  ``grad`` points in the direction of *increasing*
    the smooth max; the descent direction is ``-grad``.
    """
    conv = _self_convolution_full(a)
    cur_max = conv.max()
    tau = max(cur_max * tau_factor, min_tau)

    # soft‑max weights (stable)
    shifted = conv - cur_max                # ≤ 0
    exp_vals = np.exp(shifted / tau)
    sum_exp = exp_vals.sum()
    if sum_exp == 0.0:
        # fall back to true sub‑gradient (active set)
        active = np.where(conv >= cur_max - cur_max * 1e-12)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
    else:
        w = exp_vals / sum_exp

    # gradient of the smooth max (increase direction)
    a_rev = a[::-1]
    full = _fft_convolve(a_rev, w)                     # length = 3n‑2
    n = a.shape[0]
    grad = 2.0 * full[n - 1: n - 1 + n]                # slice to length n
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   max_iters: int = 5000) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation of the smooth‑max objective.

    Returns the best vector (still Σ=1) and its raw score.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_val = _raw_score(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        grad, cur_max = _smooth_max_subgradient(a)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.20 / math.sqrt(it)               # aggressive early, decays
        step = - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a + step
        a = _project_onto_simplex(a)

        # occasional tiny jitter to break symmetry
        if it % 2000 == 0 and rng.random() < 0.07:
            a += rng.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        # monitor progress every 100 iterations
        if it % 100 == 0:
            cur_score = _raw_score(a)
            if cur_score < best_val - 1e-12:
                best_val = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}]   it={it:5d}  max={cur_max:.6e}  score={best_val:.6f}")

    return best_a, best_val
