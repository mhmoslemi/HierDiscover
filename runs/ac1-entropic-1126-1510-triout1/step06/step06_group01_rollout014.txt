def _smooth_descent_alpha(a, alpha, steps, deadline):
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_score(a)

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    for _ in range(steps):
        if time.time() >= deadline:
            break
        t += 1

        b = _fft_self_convolve(a)
        maxb = b.max()
        exp_terms = np.exp(alpha * (b - maxb))
        sum_exp = exp_terms.sum()
        w = exp_terms / sum_exp
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n-1:n-1+n]

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, 1.0)

        if t % 500 == 0:
            cur_score = _max_convolution_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
    return best, best_score

        # optional symmetrize maybe
        # compute score

    return best, best_score

"""
Propose a non‑negative step function that minimizes ``evaluate_sequence``.

Algorithm
---------
1. Generate a diverse set of analytic initial vectors (uniform, arcsine,
   symmetric Beta, exponential, power‑law, random Dirichlet) for several
   lengths. Keep the best one as a seed.

2. Run a long Adam optimisation on a smooth log‑sum‑exp approximation of
   the max‑convolution.  The smoothing parameter α is increased stepwise
   (1 → 5 → 20 → 100 → 500 → 2000).  After every projection onto the simplex
   we enforce symmetry a←(a+a[::-1])/2 – symmetry is known to improve the
   max‑convolution bound.

3. Refine the solution with an exact sub‑gradient method (using the true
   active set of maximal convolution entries) together with momentum and a
   diminishing step size.  This runs for as long as time permits.

4. Final stochastic hill‑climbing (simulated annealing) moves a tiny amount
   of mass between random positions, accepting worsening moves with a
   temperature schedule.  Periodic Gaussian jitter helps escape plateaux.

All helpers are defined at top level, the routine respects the 1000‑second
budget, and the best vector (as a plain Python list) is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """
    Return 2·n·max(a*a) / (sum(a)²).  When sum(a)=1 this is simply 2·n·max.
    """
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid (avoids exact 0/1)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑density to stay stable for very small p
    log_w = (p - 1) * (np.log(x) + np.log(1 - x))
    w = np.exp(log_w - np.max(log_w))   # normalise numerically
    w /= w.sum()
    return w


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_powerlaw(n: int, p: float) -> np.ndarray:
    """U‑shaped power‑law: (i+1)^(-p) + (n-i)^(-p)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p) + (n - i + 1) ** (-p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _generate_initial_candidates(n: int) -> list[np.ndarray]:
    """Return a handful of plausible starts for a given length."""
    cand = []
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_exponential(n, lam=0.002))
    cand.append(_distribution_exponential(n, lam=0.01))
    # symmetric beta with a wide range of p (U‑shaped when p<1)
    for p in (0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        cand.append(_distribution_beta_sym(n, p))
    # power‑law shapes – also U‑shaped
    for p in (0.5, 1.0, 2.0):
        cand.append(_distribution_powerlaw(n, p))
    # a few random Dirichlet draws
    for _ in range(3):
        v = np.random.rand(n)
        v /= v.sum()
        cand.append(v)
    return cand


# ----------------------------------------------------------------------
def _smooth_descent_alpha(a: np.ndarray, alpha: float, steps: int,
                          deadline: float) -> tuple[np.ndarray, float]:
    """Adam optimisation of the log‑sum‑exp smoothed max for a fixed α."""
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_score(a)

    # Adam hyper‑parameters – a conventional choice works well here.
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    for _ in range(steps):
        if time.time() >= deadline:
            break
        t += 1

        # full self‑convolution
        b = _fft_self_convolve(a)

        # stable soft‑max weights over b
        maxb = b.max()
        w = np.exp(alpha * (b - maxb))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(b, 1.0 / b.size)
        else:
            w /= w_sum

        # gradient = 2 * (a reversed) * w   (first n entries)
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # Adam update + projection onto simplex
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_onto_simplex(a, 1.0)

        # enforce symmetry (helps flatten the convolution)
        a = (a + a[::-1]) * 0.5

        if t % 500 == 0:
            cur_score = _max_convolution_score(a)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
    return best, best_score


# ----------------------------------------------------------------------
def _exact_subgradient_refine(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Exact max‑subgradient descent with momentum and diminishing steps."""
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_score(a)

    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps)

        w = np.zeros_like(b)
        w[active] = 1.0

        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # momentum + diminishing step
        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / np.sqrt(iteration)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)
        a = (a + a[::-1]) * 0.5

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time():.1f}] sub‑grad it={iteration} new best {best_score:.6f}")

        # early stop if we are close to the target (optional)
        if best_score < 1.502:
            break
    return best, best_score


# ----------------------------------------------------------------------
def _annealing_hill_climb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climber on the simplex.
    Moves a small amount of probability mass between random indices,
    accepting worsening moves with probability exp(‑Δ/T).
    """
    n = a.shape[0]
    cur = a.copy()
    cur_score = _max_convolution_score(cur)
    best = cur.copy()
    best_score = cur_score

    iteration = 0
    T0 = 0.02               # initial temperature
    decay = 0.9995          # exponential cooling

    while time.time() < deadline:
        iteration += 1
        T = max(T0 * (decay ** iteration), 1e-6)

        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue

        # propose to shift a small amount of mass from i to j
        max_transfer = min(cur[i], 0.15)          # 15 % of mass at i at most
        delta = np.random.rand() * max_transfer
        if delta <= 0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_onto_simplex(cand, 1.0)
        cand = (cand + cand[::-1]) * 0.5          # keep symmetry

        new_score = _max_convolution_score(cand)
        diff = new_score - cur_score

        # Metropolis acceptance
        if diff < 0 or np.random.rand() < math.exp(-diff / T):
            cur = cand
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best = cand.copy()
                print(f"[{time.time():.1f}] anneal iter={iteration} new best {best_score:.6f}")

        # occasional Gaussian jitter to escape plateaux
        if iteration % 5000 == 0:
            pert = np.random.randn(n) * 5e-5
            jitter = best + pert
            jitter = _project_onto_simplex(jitter, 1.0)
            jitter = (jitter + jitter[::-1]) * 0.5
            jitter_score = _max_convolution_score(jitter)
            if jitter_score < best_score - 1e-12:
                best_score = jitter_score
                best = jitter.copy()
                cur = jitter.copy()
                cur_score = jitter_score
                print(f"[{time.time():.1f}] jitter iter={iteration} improved best {best_score:.6f}")

        # stop early if we have already reached the target bound
        if best_score < 1.502:
            break

    return best, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence found (list of Python floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5    # tiny safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Coarse sweep over a few lengths (powers of two are cheap for FFT).
    lengths = [1024, 2048, 4096, 8192, 16384]
    best_vec = None
    best_score = float('inf')
    best_n = None

    for n in lengths:
        if time.time() >= deadline:
            break
        for cand in _generate_initial_candidates(n):
            if time.time() >= deadline:
                break
            cand = cand / cand.sum()
            sc = _max_convolution_score(cand)
            if sc < best_score:
                best_score = sc
                best_vec = cand.copy()
                best_n = n
                print(f"[{time.time():.1f}] init n={n} score={sc:.6f}")

    # 1️⃣b  Try to reuse a previously found construction if present.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_vec = prev.copy()
                best_n = len(prev)
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # Fallback – should never trigger.
    if best_vec is None:
        best_n = 1024
        best_vec = _distribution_uniform(best_n)
        best_score = _max_convolution_score(best_vec)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Long smoothed Adam descent (many more steps and larger α).
    a = best_vec.copy()
    a = a / a.sum()
    # smoothing schedule – more steps than the reference implementation
    schedule = [
        (1.0, 5000),
        (5.0, 5000),
        (20.0, 5000),
        (100.0, 5000),
        (500.0, 5000),
        (2000.0, 5000),
    ]
    for alpha, steps in schedule:
        if time.time() >= deadline:
            break
        a, sc = _smooth_descent_alpha(a, alpha, steps, deadline)
        if sc < best_score:
            best_score = sc
            best_vec = a.copy()
            print(f"[{time.time():.1f}] after smooth α={alpha} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Exact sub‑gradient refinement (more iterations than before).
    if time.time() < deadline:
        a, sc = _exact_subgradient_refine(a, deadline)
        if sc < best_score:
            best_score = sc
            best_vec = a.copy()
            print(f"[{time.time():.1f}] after exact sub‑grad score={sc:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Simulated‑annealing hill climbing.
    if time.time() < deadline:
        a, sc = _annealing_hill_climb(a, deadline)
        if sc < best_score:
            best_score = sc
            best_vec = a.copy()
            print(f"[{time.time():.1f}] after annealing score={sc:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final evaluation with the provided ``evaluate_sequence``.
    final_seq = best_vec.tolist()
    final_score = float(evaluate_sequence(final_seq))
    print(f"Final evaluate_sequence score: {final_score:.6f} (length={len(final_seq)})")
    return final_seq
