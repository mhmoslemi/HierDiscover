"""
Hybrid optimisation for the self‑convolution L∞‑problem.

The objective for a non‑negative vector a (∑a=1) is
    score = 2·n·max_k (a*a)_k .
We minimise the peak of the auto‑convolution, i.e. we try to make the
distribution of the sum of two independent draws from a as flat as possible.

Algorithm outline
-----------------
1. **Broad sampling** – for several lengths n we generate a few analytic
   families (uniform, arcsine, triangular, exponential, symmetric and
   asymmetric beta) plus a handful of random Dirichlet points.  All are
   normalised to ℓ₁‑norm 1 and scored with the cheap FFT based routine.

2. **Smooth‑max descent** – the non‑smooth max is replaced by a
   temperature‑controlled softmax:
        Lβ(a) = log Σ_k exp(β·(a*a)_k) / β .
   For a given β the gradient is a weighted convolution, which is computed
   with real FFTs.  Starting from a small β we run exponentiated‑gradient
   updates, increase β exponentially and keep the best vector seen.
   This yields a much richer descent direction than the plain sub‑gradient.

3. **Adam refinement** – the best few candidates from step 2 are polished
   with a standard Adam optimiser on the exact max‑convolution (the same
   sub‑gradient as in the reference code).

4. **Simulated‑annealing hill climbing** – a final stage performs random
   mass transfers between two coordinates.  Moves that increase the score
   are accepted with a Boltzmann probability; the temperature decays
   geometrically.  This helps escape shallow local minima.

All steps respect a global time budget (≈ 1000 s).  The function returns a
list of non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Score = 2·n·max(a*a) assuming ∑a = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Gamma‑Dirichlet, α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(½,½))."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the grid {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """General (asymmetric) Beta(α,β) on the grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide='ignore', invalid='ignore'):
        w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x))
    w[np.isnan(w)] = 0.0
    s = w.sum()
    if s == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    """Simple symmetric triangular shape."""
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    """Bell‑shaped exponential centred in the middle of the vector."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _smooth_max_descent(a0: np.ndarray,
                       deadline: float,
                       max_iters: int = 20000,
                       lr0: float = 0.5,
                       beta_start: float = 0.5,
                       beta_end: float = 200.0):
    """
    Exponentiated‑gradient descent on the smooth approximation

        Lβ(a) = log( Σ_k exp(β·(a*a)_k) ) / β ,

    with a schedule β ∈ [beta_start, beta_end].

    Returns the best vector seen and its (hard) max‑convolution score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- temperature schedule (smoothness) -----
        beta = beta_start * (beta_end / beta_start) ** (it / max_iters)

        # ----- compute self‑convolution -----
        conv = _fft_self_convolve(a)               # length 2n‑1

        # ----- soft‑max weights -----
        scaled = beta * conv
        # subtract the maximum to avoid overflow
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        ws = w.sum()
        if ws == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= ws

        # ----- weighted gradient (∇ Σ w_k·conv_k) -----
        # gradient = 2 * (a_rev * w)  restricted to the n centre entries
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # ----- exponentiated‑gradient update -----
        eta = lr0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray,
                  deadline: float,
                  max_iters: int = 20000,
                  step0: float = 0.1,
                  beta1: float = 0.9,
                  beta2: float = 0.999):
    """Adam optimisation on the exact max‑convolution (sub‑gradient)."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        # sub‑gradient of the (hard) max
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - eps)
        w = np.zeros_like(conv)
        w[active] = 1.0

        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _annealing_hill_climb(a0: np.ndarray,
                          cur_score: float,
                          deadline: float):
    """
    Simulated‑annealing hill‑climbing on the hard max‑convolution.
    Occasionally accepts worsening moves with probability exp(-(Δ)/T).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_score
    cur = a.copy()
    cur_sc = cur_score

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # geometric temperature decay
        T = max(1e-6, 0.5 ** (iteration / 20000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # move at most 5 % of the mass at i
        delta = np.random.rand() * min(0.05 * cur[i], cur[i])
        if delta <= 0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        new_sc = _max_convolution_score(cand)

        if new_sc < cur_sc:
            cur = cand
            cur_sc = new_sc
            if new_sc < best_score:
                best_score = new_sc
                best_a = cand.copy()
                print(f"[{time.time():.1f}] anneal new best {best_score:.6f}")
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(new_sc - cur_sc) / T):
                cur = cand
                cur_sc = new_sc
    return best_a, best_score

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # small safety margin

    # ------------------------------------------------------------------
    # 1. Choose a collection of lengths to explore.
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 4000

    lengths = {int(base_n * f) for f in (0.5, 0.75, 1.0, 1.25, 1.5, 2.0) if int(base_n * f) >= 500}
    for _ in range(5):
        n = np.random.randint(500, min(20000, base_n * 3) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    global_best_seq = None
    global_best_score = float("inf")

    # ------------------------------------------------------------------
    # 2. Sample candidate families for each length.
    for n in lengths:
        if time.time() >= deadline:
            break
        candidates = []

        # analytic families (both symmetric and a few asymmetric beta shapes)
        families = [
            (_arcsine_distribution, "arcsine"),
            (_triangular_distribution, "triangular"),
            (lambda m: _exponential_distribution(m, lam=0.005), "exponential"),
        ]
        for gen, name in families:
            a = gen(n)
            candidates.append((a, _max_convolution_score(a)))
            print(f"[{time.time():.1f}] n={n:4d} {name:12s} score={candidates[-1][1]:.6f}")

        # symmetric beta with various exponents
        for p in [0.05, 0.10, 0.15, 0.20, 0.30, 0.50, 0.80, 1.00, 1.30, 2.00]:
            a = _beta_symmetric_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))
            print(f"[{time.time():.1f}] n={n:4d} beta_sym(p={p:<4.2f}) score={candidates[-1][1]:.6f}")

        # a few asymmetric beta examples
        for _ in range(2):
            alpha = 10 ** np.random.uniform(-1, 1)   # between 0.1 and 10
            beta = 10 ** np.random.uniform(-1, 1)
            a = _beta_distribution(n, alpha, beta)
            candidates.append((a, _max_convolution_score(a)))
            print(f"[{time.time():.1f}] n={n:4d} beta_asym(a={alpha:.2f},b={beta:.2f}) score={candidates[-1][1]:.6f}")

        # random Dirichlet draws
        for _ in range(3):
            a = _random_dirichlet(n)
            candidates.append((a, _max_convolution_score(a)))
            print(f"[{time.time():.1f}] n={n:4d} random_dirichlet score={candidates[-1][1]:.6f}")

        # include the previously best known sequence (if present)
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
            if prev.ndim == 1:
                prev = prev / prev.sum()
                candidates.append((prev, _max_convolution_score(prev)))
                print(f"[{time.time():.1f}] n={n:4d} previous_best score={candidates[-1][1]:.6f}")
        except Exception:
            pass

        # ------------------------------------------------------------------
        # 3. Select a small number of promising seeds.
        candidates.sort(key=lambda x: x[1])
        top_k = min(8, len(candidates))
        seeds = candidates[:top_k]
        print(f"[{time.time():.1f}] length {n} – keeping {top_k} seeds (best score {seeds[0][1]:.6f})")

        # ------------------------------------------------------------------
        # 4. Smooth‑max descent on each seed.
        for idx, (seed_seq, seed_sc) in enumerate(seeds):
            if time.time() >= deadline:
                break
            seq, sc = _smooth_max_descent(seed_seq, deadline,
                                          max_iters=12000,
                                          lr0=0.7,
                                          beta_start=0.5,
                                          beta_end=300.0)
            if sc < global_best_score - 1e-14:
                global_best_score, global_best_seq = sc, seq.copy()
                print(f"[{time.time():.1f}] n={n} seed{idx+1}/{top_k} smooth‑descent score {sc:.6f}")

        # ------------------------------------------------------------------
        # 5. Optional Adam refinement on the best few results of this length.
        for idx, (seed_seq, _) in enumerate(seeds[:4]):   # a smaller subset
            if time.time() >= deadline:
                break
            seq, sc = _adam_descent(seed_seq, deadline,
                                    max_iters=15000,
                                    step0=0.12)
            if sc < global_best_score - 1e-14:
                global_best_score, global_best_seq = sc, seq.copy()
                print(f"[{time.time():.1f}] n={n} seed{idx+1}/{top_k} Adam score {sc:.6f}")

    # ------------------------------------------------------------------
    # 6. Final simulated‑annealing hill‑climbing on the overall best.
    if global_best_seq is not None:
        a, sc = _annealing_hill_climb(global_best_seq,
                                      global_best_score,
                                      deadline)
        if sc < global_best_score - 1e-14:
            global_best_score, global_best_seq = sc, a.copy()
            print(f"[{time.time():.1f}] final anneal improved to {sc:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best score: {global_best_score:.6f} (length={len(global_best_seq)})")
    return global_best_seq.tolist()
