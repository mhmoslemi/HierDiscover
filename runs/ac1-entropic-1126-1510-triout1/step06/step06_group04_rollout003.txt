def propose_candidate(seed=..., budget_s=1000):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start+budget_s-0.5

    # Phase 1: cheap random search to generate seeds
    candidate_shapes = []
    lengths = [4000, 8000, 12000] maybe limited.

    # For each length generate: uniform, arcsine, beta grid, random betas, Dirichlet samples, mixture etc. (similar to baseline but fewer draws).
    # Keep maybe top K (e.g., 5) sequences per length.
    # We'll store them in a list seeds.

    # Phase 2: refined gradient descent with smoothing
    for each seed_seq in seeds:
        n = len(seed_seq)
        a = seed_seq.copy() # sum? ensure sum=1
        # Initialize Adam m,v zeros
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1=0.9; beta2=0.999; eps=1e-8
        lr = 0.1  (maybe adapt)
        beta = 20.0 # initial smoothing param
        iteration = 0
        while time.time() < deadline:
            iteration+=1
            # compute conv and gradient
            b = _self_convolution_full(a)
            cur_max = b.max()
            # compute smoothed weights
            # using beta variable.
            # select incremental increase schedule: every 2000 it bn increase by factor 1.02 up to maxBeta=5000
            if iteration % 2000 == 0:
                beta = min(beta*1.1, 3000)
            # compute weights
            scaled = beta * b
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            # compute gradient
            a_rev = a[::-1]
            grad = 2.0 * _fft_convolve(a_rev, w)[n-1:n-1+n]
            # Adam update
            m = beta1*m + (1-beta1)*grad
            v = beta2*v + (1-beta2)*(grad*grad)
            m_hat = m/(1-beta1**iteration)
            v_hat = v/(1-beta2**iteration)
            step = lr * m_hat/(np.sqrt(v_hat)+eps)
            a -= step
            # simplex projection
            a = _project_onto_simplex(a)

            # ensure sum not too tiny.
            # compute raw score for tracking
            score = 2.0*n*cur_max
            if score < best_raw:
                best_raw = score
                best_seq = a.copy()
                best_n = n
                print(f"[{time.time():.1f}] new best raw {score:.6f} at n={n}")

            # occasional jitter for global search
            if iteration % 5000 == 0 and rng.random() < 0.1:
                a += rng.normal(scale=1e-6, size=n)
                a = _project_onto_simplex(a)

        # after loop check if time left else break

    # final evaluation of best_seq on evaluate_sequence.
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Final evaluated score: {final_score} (len={best_n})")
    return best_seq.tolist()

"""
Propose a low‑score step function.

The algorithm works in two phases.

1. **Exploratory sampling** – For a few different lengths we generate
   a modest number of candidate distributions from families that are known
   to give good convolutions (uniform, arcsine, symmetric Beta, random
   Beta, Dirichlet, mixtures).  All candidates are projected onto the
   simplex (Σa = 1) and evaluated with the cheap proxy
   ``2·n·max(a*a)`` (the true objective is scale‑invariant).  The best
   few seeds are retained.

2. **Smooth max gradient descent** – Starting from the best seed we
   minimize the *smooth* maximum of the self‑convolution:
        fβ(a) = (1/β)·log Σk exp(β·(a*a)k) .
   For moderate β this function is a smooth approximation of the true
   max, its gradient can be obtained by a single extra convolution.
   We use an Adam‑style adaptive step‑size and project after each step
   onto the simplex.  The smoothing parameter β is increased
   gradually (so the method first follows a smooth landscape, then
   focuses on the true peak).  The loop runs until the time budget is
   exhausted, keeping the best raw score seen.

The final sequence is returned; ``evaluate_sequence`` rescales it
automatically, so the printed score equals the true objective.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution of length n, normalized to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    # log‑pdf (stable)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # clip to avoid overflow in exp
    logpdf = np.minimum(logpdf, 700.0)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf

# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    return _fft_convolve(a, a)[: 2 * a.shape[0] - 1]

def _raw_score(a: np.ndarray) -> float:
    """Proxy score for Σa = 1 : 2·n·max(a*a)."""
    n = a.shape[0]
    return 2.0 * n * _self_convolution_full(a).max()

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """Search for a non‑negative step function with a low evaluate_sequence score."""
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Optional warm‑start from a previously discovered sequence
    base_seq = None
    if "height_sequence_1" in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=np.float64)
            if base_seq.size > 0:
                base_seq = base_seq / base_seq.sum()
        except Exception:
            base_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – generate a handful of seeds
    lengths = [4000, 8000, 12000]
    seed_candidates = []          # list of (seq, raw_score, length)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Exploring length {n}")

        # uniform
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        seed_candidates.append((uniform, _raw_score(uniform), n))

        # arcsine (Beta(0.5,0.5))
        arcsine = _beta_distribution(n, 0.5, 0.5)
        seed_candidates.append((arcsine, _raw_score(arcsine), n))

        # a few symmetric Betas on a coarse grid
        for a in (0.2, 0.5, 1.0, 1.5, 2.0, 3.0, 5.0):
            vec = _beta_distribution(n, a, a)
            seed_candidates.append((vec, _raw_score(vec), n))

        # random Beta draws
        for _ in range(30):
            alpha = 10 ** rng.uniform(-2, 1)   # 0.01 … 10
            beta = 10 ** rng.uniform(-2, 1)
            vec = _beta_distribution(n, alpha, beta)
            seed_candidates.append((vec, _raw_score(vec), n))

        # Dirichlet draws
        for _ in range(30):
            conc = 10 ** rng.uniform(-2, 1)   # 0.01 … 10
            vec = rng.gamma(conc, 1.0, size=n)
            vec /= vec.sum()
            seed_candidates.append((vec, _raw_score(vec), n))

        # mixture of two Betas
        for _ in range(20):
            a1 = 10 ** rng.uniform(-2, 1)
            b1 = 10 ** rng.uniform(-2, 1)
            a2 = 10 ** rng.uniform(-2, 1)
            b2 = 10 ** rng.uniform(-2, 1)
            lam = rng.random()
            p1 = _beta_distribution(n, a1, b1)
            p2 = _beta_distribution(n, a2, b2)
            mix = lam * p1 + (1.0 - lam) * p2
            mix = _project_onto_simplex(mix)
            seed_candidates.append((mix, _raw_score(mix), n))

        # uniform + Beta mixture
        for _ in range(20):
            a = 10 ** rng.uniform(-2, 1)
            b = 10 ** rng.uniform(-2, 1)
            beta_vec = _beta_distribution(n, a, b)
            lam = rng.random()
            mix = lam * beta_vec + (1.0 - lam) * uniform
            mix = _project_onto_simplex(mix)
            seed_candidates.append((mix, _raw_score(mix), n))

        # optionally interpolate the global warm‑start
        if base_seq is not None:
            cand = np.interp(np.linspace(0, 1, n),
                             np.linspace(0, 1, base_seq.shape[0]),
                             base_seq)
            cand = _project_onto_simplex(cand)
            seed_candidates.append((cand, _raw_score(cand), n))

    # keep the 10 best seeds (by raw score)
    seed_candidates.sort(key=lambda t: t[1])
    top_seeds = seed_candidates[:10]

    best_raw = float("inf")
    best_seq = None
    best_len = None
    for seq, scr, n in top_seeds:
        if scr < best_raw:
            best_raw, best_seq, best_len = scr, seq.copy(), n

    if best_seq is None:
        # fallback – uniform of length 4000
        best_len = 4000
        best_seq = np.full(best_len, 1.0 / best_len, dtype=np.float64)
        best_raw = _raw_score(best_seq)

    print(f"Seed chosen: length={best_len}, raw score={best_raw:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smooth‑max Adam optimisation
    a = best_seq.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05                       # base learning rate

    # smoothing parameter β (starts small → smooth landscape)
    smooth_beta = 5.0
    max_smooth_beta = 4000.0
    it = 0
    last_improve_it = 0

    while time.time() < deadline:
        it += 1
        # self‑convolution and current max
        conv = _self_convolution_full(a)        # length 2n‑1
        cur_max = conv.max()
        raw = 2.0 * n * cur_max
        if raw < best_raw - 1e-12:
            best_raw = raw
            best_seq = a.copy()
            best_len = n
            last_improve_it = it
            print(f"[{time.time():.1f}] it={it}  new best raw={best_raw:.6f}")

        # ----- smooth max weights -----
        scaled = smooth_beta * conv
        # subtract max for numerical stability
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()                     # shape (2n‑1,)

        # ----- gradient of smooth max -----
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_onto_simplex(a)

        # ----- schedule for the smoothness parameter -----
        if it % 2000 == 0 and smooth_beta < max_smooth_beta:
            smooth_beta = min(smooth_beta * 1.1, max_smooth_beta)

        # ----- occasional jitter to escape plateaus -----
        if it % 5000 == 0 and rng.random() < 0.07:
            a += rng.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        # safety break if we haven't improved for a long time (optional)
        if it - last_improve_it > 200000 and time.time() > deadline - 5.0:
            # less than 5 s left – stop early
            break

    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Final evaluated score: {final_score:.6f} (len={best_len})")
    return best_seq.tolist()
