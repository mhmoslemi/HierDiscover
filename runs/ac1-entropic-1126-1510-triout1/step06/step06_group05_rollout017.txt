def _exp_gradient_descent(a0, deadline, eta_init=0.1, eta_decay=0.99):
    a = a0.copy()
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it=0
    while time.time() < deadline:
        it+=1
        # compute subgradient with active set (like before)
        grad, cur_max = _max_subgradient_step(a) # returns grad and cur max
        # adapt eta
        eta = eta_init * (eta_decay**it)
        # mirror descent update
        a = a * np.exp(-eta * grad)
        # Normalize
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0/a.size)
        else:
            a /= a_sum
        # Optionally take best when cur_max improved
        if cur_max < best_peak:
            best_peak = cur_max
            best_a = a.copy()
        # occasional random perturbation to escape
        if it % 500 == 0 and np.random.rand() < 0.2:
            # add small Gaussian
            a += np.random.randn(a.size) * 1e-6
            a = _project_onto_simplex(a, 1.0)
    return best_a

"""
Optimisation of the step‑function bound 2·n·max(conv(a,a)) / (∑a)².

The algorithm works in three stages:

1. **Broad generation** – many parametric families (arcsine, Beta,
   power‑law, exponential‑decay, Dirichlet, mixtures) are sampled for
   several lengths (≈500–8000).  The best few candidates according to the
   cheap proxy ``2·n·max(conv)`` are kept.

2. **Intensive refinement** – each top candidate is fed through an
   Adam optimiser on the smooth‑max surrogate (α increased up to 5000),
   then through a mirror‑descent step (exponentiated‑gradient) that works
   directly on the exact ℓ∞ objective, and finally a light simulated‑
   annealing polish.

3. **Final evaluation** – the best sequence found is evaluated with the
   provided ``evaluate_sequence`` function and returned.

All updates respect the simplex (non‑negative, sum = 1) by either the
Euclidean projection or a KL‑projection (multiplicative update).  The
procedure respects the given time budget (default 1000 s) and prints
progress information.
"""

import math
import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_vals = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    vals[~np.isfinite(vals)] = 0.0
    total = vals.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    vals /= total
    return vals


def _beta_sym_distribution(n: int, a: float) -> np.ndarray:
    return _beta_distribution(n, a, a)


def _powerlaw_boundary_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5) + 1e-12) ** (-p)
    vals /= vals.sum()
    return vals


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the edges: λ**i + λ**(n‑1‑i)."""
    i = np.arange(n, dtype=np.float64)
    vals = lam ** i + lam ** (n - 1 - i)
    vals /= vals.sum()
    return vals


def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a non‑negative sequence to a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals

def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """Gradient of Lα(a)= (1/α)·log Σ exp(α·(a*a)_k)."""
    b = _fft_self_convolve(a)
    cur_max = b.max()
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.01,
                   alpha_max: float = 5000.0,
                   lr0: float = 0.30,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """Adam optimisation of the smooth‑max surrogate."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        if t % 500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


def _max_subgradient_step(a: np.ndarray):
    """Exact ℓ∞ sub‑gradient (active‑set) for max(a*a)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


def _exp_gradient_descent(a0: np.ndarray,
                          deadline: float,
                          eta0: float = 0.10,
                          decay: float = 0.9999):
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max objective.
    Uses the sub‑gradient from the active set and performs multiplicative
    updates a_i ← a_i·exp(‑η·g_i), followed by ℓ₁‑normalisation.
    """
    a = a0.copy()
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        eta = eta0 * (decay ** it)
        # multiplicative update
        a = a * np.exp(-eta * grad)
        # avoid all‑zero situation
        total = a.sum()
        if total <= 0.0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a /= total

        if cur_max < best_peak:
            best_peak = cur_max
            best_a = a.copy()

        if it % 2000 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(a.size) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a


def _simulated_annealing(a0: np.ndarray, deadline: float):
    """Light SA that accepts downhill moves always and occasional uphill moves."""
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_peak = _fft_self_convolve(cur_a).max()
    best_a = cur_a.copy()
    best_peak = cur_peak

    start = time.time()
    total_time = max(deadline - start, 1e-6)

    start_temp = max(cur_peak, 1e-3) * 0.5
    final_temp = start_temp * 1e-5
    if final_temp <= 0.0:
        final_temp = 1e-9
    cooling_rate = -math.log(final_temp / start_temp) / total_time

    while time.time() < deadline:
        src = np.random.randint(n)
        if cur_a[src] <= 1e-12:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue
        max_delta = min(cur_a[src], 0.10)          # allow up to 10 % moves
        delta = np.random.rand() * max_delta

        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_peak = _fft_self_convolve(a_new).max()
        delta_obj = new_peak - cur_peak

        elapsed = time.time() - start
        temperature = start_temp * math.exp(-cooling_rate * elapsed)

        accept = False
        if delta_obj < 0.0:
            accept = True
        else:
            prob = math.exp(-delta_obj / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_a = a_new
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_a = cur_a.copy()

        if np.random.rand() < 0.02:
            # occasional restart from best
            cur_a = best_a.copy()
            cur_peak = best_peak

    return best_a, best_peak


def _full_optimize(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Adam → exp‑gradient → SA, respecting the given deadline."""
    # 1) Adam on smooth‑max
    a_adam, _ = _adam_optimize(a0, deadline, alpha_max=5000.0, lr0=0.30)
    if time.time() >= deadline:
        return a_adam

    # 2) Exponentiated‑gradient mirror descent
    a_exp = _exp_gradient_descent(a_adam, deadline)
    if time.time() >= deadline:
        return a_exp

    # 3) Simulated annealing polish
    a_sa, _ = _simulated_annealing(a_exp, deadline)
    return a_sa


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """Search for a low‑score step‑function within the given time budget."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5     # safety margin

    # ------------------------------------------------------------------
    # Base sequence (if the previous run stored one)
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_seq = _project_onto_simplex(base_seq, s=1.0)
        base_n = len(base_seq)
    except Exception:
        base_seq = None
        base_n = 4000

    # ------------------------------------------------------------------
    # Lengths to explore (keep them moderate for speed)
    factors = (0.7, 0.9, 1.0, 1.3, 1.5)
    lengths = sorted({max(500, int(base_n * f)) for f in factors})
    # cap at 8000 to stay memory‑friendly
    lengths = [min(l, 8000) for l in lengths]

    # Keep the best few candidates from the cheap evaluation
    TOP_K = 8
    top_candidates = []   # list of (fast_score, np.ndarray)

    print("[phase 1] broad generation of candidates")
    for n in lengths:
        if time.time() >= deadline:
            break

        # basic distributions
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        arcsine = _arcsine_distribution(n)

        # collection of starters
        starters = {
            "uniform": uniform,
            "arcsine": arcsine,
        }

        # Dirichlet draws with very low concentration (spiky) and some moderate ones
        for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
            starters[f"dirichlet_{conc:.2f}"] = np.random.dirichlet(np.full(n, conc))

        # Symmetric Beta with a wide range (including very edge‑heavy)
        for a in (0.15, 0.20, 0.25, 0.30, 0.35, 0.40, 0.45, 0.50):
            starters[f"beta_{a:.2f}"] = _beta_sym_distribution(n, a)

        # Power‑law boundary
        for p in (0.45, 0.55, 0.65, 0.75, 0.85, 1.0, 1.2):
            starters[f"power_{p:.2f}"] = _powerlaw_boundary_distribution(n, p)

        # Exponential decay from the edges
        for lam in (0.80, 0.85, 0.90, 0.95):
            starters[f"exp_{lam:.2f}"] = _exp_decay_distribution(n, lam)

        # Mixtures (arcsine + uniform)
        for w in (0.2, 0.5, 0.8):
            starters[f"mix_arc_uni_{w:.1f}"] = (1.0 - w) * arcsine + w * uniform

        # Mixtures (arcsine + power‑law)
        for p in (0.6, 0.8, 1.0):
            pw = _powerlaw_boundary_distribution(n, p)
            for w in (0.3, 0.5):
                starters[f"mix_arc_pow_{p:.1f}_{w:.1f}"] = (1.0 - w) * arcsine + w * pw

        # Mixtures (arcsine + exponential)
        for lam in (0.85, 0.90):
            expd = _exp_decay_distribution(n, lam)
            for w in (0.3, 0.6):
                starters[f"mix_arc_exp_{lam:.2f}_{w:.1f}"] = (1.0 - w) * arcsine + w * expd

        # Interpolation of the previous best (if any)
        if base_seq is not None:
            starters["prev"] = _interpolate_to_length(base_seq, n)

        # ------------------------------------------------------------------
        # Evaluate the cheap proxy 2·n·max(conv) and retain the top K.
        for tag, vec in starters.items():
            vec = _project_onto_simplex(vec, s=1.0)
            peak = _fft_self_convolve(vec).max()
            fast_score = 2.0 * n * peak
            # keep top K candidates
            top_candidates.append((fast_score, vec.copy()))
            top_candidates.sort(key=lambda x: x[0])
            if len(top_candidates) > TOP_K:
                top_candidates.pop()  # discard worst
            # occasional progress print
            if fast_score < 2.5:
                print(f"  [init] n={n:4d} {tag:<15s} fast_score={fast_score:.6f}")

    if not top_candidates:
        # fallback: uniform distribution of moderate size
        n = max(500, base_n)
        seq = np.full(n, 1.0 / n, dtype=np.float64)
        top_candidates = [(2.0 * n * _fft_self_convolve(seq).max(), seq)]

    # ------------------------------------------------------------------
    # 2. Intensive refinement of the best few candidates.
    remaining = deadline - time.time()
    if remaining > 5.0:
        # allocate equal time slots to the top candidates (up to 3)
        to_refine = top_candidates[:3]
        per_candidate = remaining / len(to_refine)
        print(f"[phase 2] refining {len(to_refine)} candidates, "
              f"{per_candidate:.1f}s each")
        refined = []
        for idx, (fast_sc, cand) in enumerate(to_refine):
            print(f"   candidate {idx+1}: initial fast_score={fast_sc:.6f}")
            cand_deadline = time.time() + per_candidate
            refined_seq = _full_optimize(cand, cand_deadline)
            refined_peak = _fft_self_convolve(refined_seq).max()
            refined_score = 2.0 * len(refined_seq) * refined_peak
            refined.append((refined_score, refined_seq))
            print(f"   candidate {idx+1}: refined fast_score={refined_score:.6f}")
        # keep the best refined candidate
        refined.sort(key=lambda x: x[0])
        best_seq = refined[0][1]
        best_fast = refined[0][0]
    else:
        # not enough time – just take the current best
        best_seq = top_candidates[0][1]
        best_fast = top_candidates[0][0]

    # ------------------------------------------------------------------
    # 3. Final polishing (simulated annealing) if we still have time.
    time_left = deadline - time.time()
    if time_left > 1.0:
        print(f"[phase 3] final simulated‑annealing polish ({time_left:.1f}s left)")
        best_seq = _simulated_annealing(best_seq, deadline)[0]

    # ------------------------------------------------------------------
    # 4. Exact scoring using the supplied evaluation function.
    try:
        true_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence score={true_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence raised:", exc)
        true_score = float("inf")

    print(f"Best reported score: {true_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
