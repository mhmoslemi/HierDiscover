def _adam_optimize_smoothing(x0, n, alpha, deadline, max_iters=2000, lr0=0.15):
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    best_seq = None
    best_val = float('inf')
    for it in range(1, max_iters+1):
        if time.time()>deadline: break
        # compute a via softmax stable
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x/exp_x.sum()
        # compute full conv
        conv = _fft_self_convolve_full(a)
        # smooth max
        w_raw = np.exp(alpha * conv)
        w = w_raw / w_raw.sum()
        # gradient w.r.t a
        a_rev = a[::-1]
        grad_a = 2.0 * _fft_convolve(a_rev, w)[n-1:n-1+n]
        # raw score
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_val:
            best_val = cur_score
            best_seq = a.copy()
        # chain rule through softmax
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)
        # Adam update
        m = beta1*m + (1-beta1)*grad_x
        v = beta2*v + (1-beta2)*(grad_x*grad_x)
        m_hat = m/(1-beta1**it)
        v_hat = v/(1-beta2**it)
        lr = lr0 * (np.sqrt(1-beta2**it)/(1-beta1**it))
        x = x - lr * m_hat/(np.sqrt(v_hat)+eps)
        # occasional jitter
        if it%500==0:
            x += np.random.randn(*x.shape)*0.01
    return best_seq, best_val

def _adam_optimize_with_smoothing(x0, n, alphas, deadline, iters_per_alpha=2000, lr0=0.12):
    x = x0.copy()
    best_seq = None
    best_score = float('inf')
    for alpha in alphas:
        # warm start for this alpha: keep same x
        # resetting Adam momentum? Could keep same momentum?
        m = np.zeros_like(x)
        v = np.zeros_like(x)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-12
        for it in range(1, iters_per_alpha+1):
            if time.time() >= deadline:
                break
            # softmax
            max_x = np.max(x)
            exp_x = np.exp(x - max_x)
            a = exp_x / exp_x.sum()
            conv = _fft_self_convolve_full(a)
            cur_max = conv.max()
            raw_score = 2.0 * n * cur_max
            if raw_score < best_score:
                best_score = raw_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] n={n} alpha={alpha} it={it} raw_score={best_score:.6f}")
            # compute smoothed gradient
            c = conv
            c_max = c.max()
            # stable softmax of scaled conv
            w = np.exp(alpha * (c - c_max))
            w_sum = w.sum()
            w = w / w_sum
            # gradient wrt a
            a_rev = a[::-1]
            grad_a = 2.0 * _fft_convolve(a_rev, w)[n-1:n-1+n]
            # chain rule through softmax
            dot = np.dot(a, grad_a)
            grad_x = a * (grad_a - dot)
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_x
            v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
            m_hat = m / (1 - beta1**it)
            v_hat = v / (1 - beta2**it)
            lr = lr0 * (np.sqrt(1 - beta2**it) / (1 - beta1**it))
            x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
            # jitter occasionally
            if it % 500 == 0:

"""
Propose a step function using a smoothed max objective and Adam.
The sequence is represented by logits of a soft‑max so the simplex
constraint (non‑negative, unit sum) holds automatically.
We minimise the convex functional

    f(a) = 2·n·max_k (a*a)_k

by first approximating the maximum with a smooth log‑sum‑exp
(temperature α) and performing Adam optimisation on the logits.
The smoothing parameter is increased geometrically,
allowing the optimiser to first see a smooth landscape and then
focus on the true max.  After the smooth phase a few
projected sub‑gradient steps (as in the reference solution) fine‑tune
the result.  The routine respects the time budget and gracefully
falls back to a uniform distribution.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of max_k (a*a)_k.
    Returns (grad, cur_max), where grad has same shape as a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(conv >= cur_max - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


# ----------------------------------------------------------------------
def _adam_optimize_with_smoothing(x0: np.ndarray,
                                  n: int,
                                  alphas: list[float],
                                  deadline: float,
                                  iters_per_alpha: int = 1500,
                                  lr0: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑it representation using a smoothed max.
    Returns the best vector a (sum = 1) found and its raw score
    2·n·max(a*a) evaluated with the true max (not the smoothed one).
    """
    eps = 1e-12
    x = x0.copy()
    best_a = None
    best_score = float("inf")

    for alpha in alphas:
        # reset Adam moments for the new smoothing level
        m = np.zeros_like(x)
        v = np.zeros_like(x)
        beta1, beta2 = 0.9, 0.999
        it = 0
        while it < iters_per_alpha and time.time() < deadline:
            it += 1
            # ---- soft‑max (stable) ----
            max_x = np.max(x)
            exp_x = np.exp(x - max_x)
            a = exp_x / exp_x.sum()          # Σa = 1, a ≥ 0

            # ---- objective values ----
            conv = _fft_self_convolve_full(a)
            cur_max = conv.max()
            raw_score = 2.0 * n * cur_max
            if raw_score < best_score:
                best_score = raw_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] n={n} α={alpha:.1f} it={it}  raw_score={best_score:.6f}")

            # ---- smoothed gradient of the max ----
            c = conv
            c_max = cur_max
            # stable soft‑max of scaled convolution
            w = np.exp(alpha * (c - c_max))
            w /= w.sum()
            a_rev = a[::-1]
            grad_a = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]

            # ---- chain rule through soft‑max ----
            dot = np.dot(a, grad_a)
            grad_x = a * (grad_a - dot)

            # ---- Adam update ----
            m = beta1 * m + (1.0 - beta1) * grad_x
            v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
            x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional jitter to escape plateaus
            if it % 500 == 0:
                x += np.random.randn(*x.shape) * 0.01

        # use the best vector of this stage as the start for the next α
        if best_a is not None:
            x = np.log(best_a + eps)

        if time.time() >= deadline:
            break

    return best_a, best_score
