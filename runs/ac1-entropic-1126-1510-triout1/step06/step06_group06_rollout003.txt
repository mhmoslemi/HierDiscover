"""
Propose a low‑score step‑function.

Algorithm outline
-----------------
1. **Broad family sweep** – For several lengths we generate many
   analytically‑defined candidates (uniform, arcsine, symmetric &
   asymmetric Beta, exponential, two‑level block, random Dirichlet and
   an interpolated copy of the previous best).  The best (lowest) fast
   score 2·n·max(conv) is kept.

2. **Smooth‑max Adam** – Optimize the log‑sum‑exp surrogate of the
   ℓ∞‑convolution using Adam with a continuation on the smoothness
   parameter.  The best vector seen during the run is retained.

3. **Polyak sub‑gradient refinement** – The objective
   f(a)=max_k (a∗a)[k] is convex; we use a Polyak step size
   `gap / ||g||²` where `gap = f(a) – f_best`.  After each update we
   project onto the simplex.

4. **Directed local search** – Greedy mass‑transfer moves that
   target the active peak of the convolution.

5. **Simulated annealing** – Small random transfers are accepted
   according to a decreasing temperature schedule, allowing occasional
   uphill moves to escape shallow minima.

6. **Final evaluation** – The black‑box `evaluate_sequence` is called
   once on the best candidate.

All steps respect the overall time budget (default 1000 s) and keep
the vector non‑negative with unit ℓ¹‑norm (the objective is scale‑free).
"""

import math
import time

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth‑max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    exp_term = np.exp(alpha * (b - cur_max))
    w_sum = exp_term.sum()
    w = exp_term / w_sum
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.01,
                   alpha_max: float = 5000.0,
                   lr0: float = 0.3,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best probability vector encountered and its exact peak.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size schedule (slow decay)
        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # keep the best exact peak seen so far
        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        # continuation on the smoothness parameter
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny jitter
        if t % 400 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    One sub‑gradient step for the exact ℓ∞ objective.
    Uses an equal weight on all indices attaining the current maximum.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


def _polyak_refine_max(a0: np.ndarray, deadline: float, best_peak: float) -> np.ndarray:
    """
    Convex sub‑gradient refinement using a Polyak step size.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = best_peak
    it = 0

    while time.time() < deadline:
        grad, cur_max = _max_subgradient_step(a)

        gap = cur_max - best_val
        if gap <= 0:
            step = 0.0
        else:
            norm_sq = np.dot(grad, grad) + 1e-16
            step = gap / norm_sq

        # limit step size for stability
        max_step = 0.5 / np.sqrt(it + 1)
        if step > max_step:
            step = max_step

        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)

        new_max = _fft_self_convolve(a).max()
        if new_max < best_val - 1e-14:
            best_val = new_max
            best_a = a.copy()

        it += 1
        if it % 400 == 0 and np.random.rand() < 0.05:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

    return best_a

def _directed_local_search(a0: np.ndarray, deadline: float):
    """
    Greedy mass‑transfer search that always targets the current peak.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        # pick one index belonging to a maximal convolution term
        max_idx = np.random.choice(np.where(b >= cur_max - 1e-12)[0])

        # indices that can contribute to (a*a)[max_idx]
        i_low = max(0, max_idx - (n - 1))
        i_high = min(max_idx, n - 1)
        i_vals = np.arange(i_low, i_high + 1, dtype=int)
        prod = a[i_vals] * a[max_idx - i_vals]
        if prod.sum() == 0.0:
            src = np.random.randint(n)
        else:
            prob = prod / prod.sum()
            src = np.random.choice(i_vals, p=prob)

        dst = np.random.randint(n)
        if dst == src:
            continue

        max_delta = min(a[src], 0.05)          # up to 5 % of src
        delta = np.random.rand() * max_delta

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        b_new = _fft_self_convolve(a_new)
        new_max = b_new.max()

        if new_max < cur_max - 1e-15:
            a = a_new
            b = b_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()

        # occasional random jitter

            a = _project_onto_simplex(a, s=1.0)
            b = _fft_self_convolve(a)
            cur_max = b.max()

    return best_a


# ----------------------------------------------------------------------
def _two_level_family(n: int, deadline: float):
    """
    Simple two‑level family:
        a_i = p   for i < L   (front block)
        a_i = q   otherwise   (back block)
    with p = r·q, r≥1.
    Returns the best vector found and its fast score.
    """
    best_seq = None
    best_score = float('inf')
    step_L = max(1, n // 100)          # roughly 1 % steps
    r_grid = np.logspace(0.0, 2.3, num=25)   # 1 … ≈200

    for L in range(step_L, n, step_L):
        if time.time() > deadline:
            break
        denom_base = n - L
        for r in r_grid:
            denom = L * r + denom_base
            q = 1.0 / denom
            p = r * q
            seq = np.empty(n, dtype=np.float64)
            seq[:L] = p
            seq[L:] = q
            peak = _fft_self_convolve(seq).max()
            score = 2.0 * n * peak
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
    return best_seq, best_score

                     (0.5, 3.0), (3.0, 0.5), (0.7, 1.5), (1.5, 0.7)]:
            x = (np.arange(n, dtype=np.float64) + 0.5) / n
            vals = np.exp((a - 1.0) * np.log(x) + (b - 1.0) * np.log(1.0 - x))
            vals[~np.isfinite(vals)] = 0.0
            if vals.sum() == 0.0:
                vals = np.full(n, 1.0 / n)
            else:
                vals /= vals.sum()
            starters[f"beta_{a:.1f}_{b:.1f}"] = vals

        # Exponential decay families
        for lam in (0.0005, 0.001, 0.002, 0.003):
            vec = np.exp(-lam * np.arange(n))
            vec /= vec.sum()
            starters[f"exp_{lam:.4f}"] = vec

        # Random Dirichlet draws (different concentration)
        for _ in range(4):
            conc = np.random.uniform(0.1, 5.0)
            g = np.random.gamma(conc, size=n)
            g /= g.sum()
            starters["rand_dir"] = g

        # Interpolated copy of the previous best, if any
        if base_seq is not None:
            old_n = len(base_seq)
            x_old = np.linspace(0.0, 1.0, old_n)
            x_new = np.linspace(0.0, 1.0, n)
            interp = np.interp(x_new, x_old, base_seq)
            interp = np.maximum(0.0, interp)
            interp /= interp.sum()
            starters["prev"] = interp

        # Evaluate all starters
        for tag, vec in starters.items():
            vec = _project_onto_simplex(vec, s=1.0)
            cur_max = _fft_self_convolve(vec).max()
            fast_score = 2.0 * n * cur_max
            if fast_score < overall_best_score:
                overall_best_score = fast_score
                overall_best_seq = vec.copy()
                best_length = n
                print(f"[init] n={n:5d} tag={tag:<15s} fast_score={fast_score:.6f}")

        # Two‑level parametric family
        seq, score = _two_level_family(n, deadline)
        if seq is not None and score < overall_best_score:
            overall_best_score = score
            overall_best_seq = seq.copy()
            best_length = n
            print(f"[two‑level] n={n:5d} fast_score={score:.6f}")

    # --------------------------------------------------------------
    # Fallback in the extremely unlikely case that nothing was generated
    if overall_best_seq is None:
        n = max(500, base_n)
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        best_length = n
        print("[fallback] using uniform distribution")

    # --------------------------------------------------------------
    # Phase 1 – Adam optimisation of the smooth‑max surrogate
    remaining = deadline - time.time()
    if remaining > 8.0:
        print(f"[phase 1] Adam optimisation on length {best_length} (time left {remaining:.1f}s)")
        phase_end = time.time() + remaining * 0.55
        a_opt, _ = _adam_optimize(overall_best_seq, phase_end)
        overall_best_seq = a_opt.copy()
        overall_best_score = 2.0 * best_length * _fft_self_convolve(overall_best_seq).max()
        print(f"[phase 1] after Adam fast_score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Polyak sub‑gradient refinement
    remaining = deadline - time.time()
    if remaining > 8.0:
        print(f"[phase 2] Polyak sub‑gradient refinement (time left {remaining:.1f}s)")
        sub_end = time.time() + remaining * 0.55
        cur_peak = _fft_self_convolve(overall_best_seq).max()
        refined = _polyak_refine_max(overall_best_seq, sub_end, cur_peak)
        overall_best_seq = refined.copy()
        overall_best_score = 2.0 * best_length * _fft_self_convolve(overall_best_seq).max()
        print(f"[phase 2] after refinement fast_score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Directed greedy local search
    remaining = deadline - time.time()
    if remaining > 8.0:
        print(f"[phase 3] Directed local search (time left {remaining:.1f}s)")
        a_local = _directed_local_search(overall_best_seq, deadline)
        overall_best_seq = a_local.copy()
        overall_best_score = 2.0 * best_length * _fft_self_convolve(overall_best_seq).max()
        print(f"[phase 3] after local search fast_score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Simulated annealing for final fine‑tuning
    remaining = deadline - time.time()
    if remaining > 8.0:
        print(f"[phase 4] Simulated annealing (time left {remaining:.1f}s)")
        a_sa = _simulated_annealing(overall_best_seq, deadline)
        overall_best_seq = a_sa.copy()
        overall_best_score = 2.0 * best_length * _fft_self_convolve(overall_best_seq).max()
        print(f"[phase 4] after SA fast_score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # Final black‑box evaluation
    try:
        true_score = evaluate_sequence(overall_best_seq.tolist())
        print(f"[final] evaluate_sequence score={true_score:.6f}")
        overall_best_score = min(overall_best_score, true_score)
    except Exception as exc:
        print("[final] evaluate_sequence raised:", exc)

    print(f"Best approx. score {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
