"""Algorithm to search for low evaluation score step function.

The search consists of:
  1. Broad enumeration of many parametric families (Beta, power‑law,
     arcsine, mixtures) at several lengths.  The best vectors by the cheap
     proxy 2·n·max(conv) are kept.
  2. For the top candidates a two‑stage continuous refinement is performed:
     - Adam optimisation on a smooth‑max surrogate (high temperature → low,
       then exponential increase of the smooth‑max parameter).
     - Exact ℓ∞ sub‑gradient descent.
  3. Optional light simulated‑annealing polishing.
"""

import math
import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    # Fast case
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _beta_distribution(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0)
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    log_vals = (a - 1.0) * np.log(x) + (b - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    vals[np.isnan(vals)] = 0.0
    total = vals.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    vals /= total
    return vals

def _powerlaw_boundary_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5) + 1e-12) ** (-p)
    vals /= vals.sum()
    return vals

def _powerlaw_interior_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) ** (-p) + (n - i - 0.5) ** (-p))
    vals /= vals.sum()
    return vals

def _exponential_distribution(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(i - center))
    vals /= vals.sum()
    return vals

def _mix_vectors(v1: np.ndarray, v2: np.ndarray, w: float) -> np.ndarray:
    """Linear mixture (1‑w)*v1 + w*v2."""
    return (1.0 - w) * v1 + w * v2

# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """Gradient of the smooth‑max surrogate Lα(a) = (1/α)·log Σ exp(α·(a*a)_k)."""
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # Shift for numerical stability
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max

def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.02,
                   alpha_max: float = 4000.0,
                   lr0: float = 0.3,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-12):
    """Adam optimisation on the smooth‑max surrogate."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny jitter to escape plateaus
        if t % 400 == 0 and np.random.rand() < 0.08:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak

def _max_subgradient_step(a: np.ndarray):
    """Exact ℓ∞ sub‑gradient step."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max

def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Short ℓ∞ sub‑gradient descent after Adam."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.015 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        if it % 300 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)
    return best_a

def _simulated_annealing(a0: np.ndarray, deadline: float):
    """Light simulated‑annealing polishing."""
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_peak = _fft_self_convolve(cur_a).max()
    best_a = cur_a.copy()
    best_peak = cur_peak

    start = time.time()
    remaining = deadline - start
    if remaining <= 0:
        return best_a, best_peak
    start_temp = max(cur_peak, 1e-3) * 0.5
    final_temp = start_temp * 1e-5
    cooling_rate = -math.log(final_temp / start_temp) / remaining

    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1
        src = np.random.randint(n)
        if cur_a[src] <= 1e-12:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue
        # transfer a small random amount
        max_delta = min(cur_a[src], 0.03)
        delta = np.random.rand() * max_delta

        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_peak = _fft_self_convolve(a_new).max()
        delta_obj = new_peak - cur_peak
        elapsed = time.time() - start
        temperature = start_temp * math.exp(-cooling_rate * elapsed)

        accept = False
        if delta_obj < 0:
            accept = True
        else:
            prob = math.exp(-delta_obj / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_a = a_new
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_a = cur_a.copy()

        # occasional reset to best
        if iter_cnt % 15000 == 0 and np.random.rand() < 0.05:
            cur_a = best_a.copy()
            cur_peak = best_peak

    return best_a, best_peak

# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals

1️⃣ Broad generation – many parametric families (Beta, power‑law,
   arcsine, mixtures, exponentials, Dirichlet draws) are sampled at a few
   lengths.  The best vectors by the cheap proxy `2·n·max(conv)` are kept.

2️⃣ Continuous refinement – the top candidates are refined with
   Adam optimisation on a smooth‑max surrogate (α grows up to 4000) and
   a short exact ℓ∞ sub‑gradient descent.  A few restarts are performed.

3️⃣ Light simulated‑annealing – stochastic moves polish the final vector.

All vectors are projected onto the probability simplex (Σ aᵢ = 1,
aᵢ ≥ 0) so that the evaluation score is simply `2·n·max(conv)`.  The
function returns the best sequence found as a plain Python list ready
for `evaluate_sequence`.  The total wall‑clock time is limited by the
`budget_s` argument (≈ 1000 s)."""

import math
import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_distribution(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    log_vals = (a - 1.0) * np.log(x) + (b - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    vals[np.isnan(vals)] = 0.0
    total = vals.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    vals /= total
    return vals


def _powerlaw_boundary_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5) + 1e-12) ** (-p)
    vals /= vals.sum()
    return vals


def _powerlaw_interior_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = (i + 0.5) ** (-p) + (n - i - 0.5) ** (-p)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(i - centre))
    vals /= vals.sum()
    return vals


def _mix_vectors(v1: np.ndarray, v2: np.ndarray, w: float) -> np.ndarray:
    return (1.0 - w) * v1 + w * v2


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """Gradient of Lα(a)= (1/α)·log Σ exp(α·(a*a)_k )."""
    b = _fft_self_convolve(a)
    cur_max = b.max()
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)                # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.02,
                   alpha_max: float = 4000.0,
                   lr0: float = 0.30,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-12):
    """Adam optimisation on the smooth‑max surrogate."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny jitter to escape plateaus
        if t % 400 == 0 and np.random.rand() < 0.08:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


def _max_subgradient_step(a: np.ndarray):
    """Exact ℓ∞ sub‑gradient step for the true max objective."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Short ℓ∞ sub‑gradient descent after Adam."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.015 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        if it % 300 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)
    return best_a


def _simulated_annealing(a0: np.ndarray, deadline: float):
    """Light simulated‑annealing polishing."""
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_peak = _fft_self_convolve(cur_a).max()
    best_a = cur_a.copy()
    best_peak = cur_peak

    start = time.time()
    total = max(deadline - start, 1e-6)

    start_temp = max(cur_peak, 1e-3) * 0.5
    final_temp = start_temp * 1e-5
    if final_temp <= 0.0:
        final_temp = 1e-9
    cooling_rate = -math.log(final_temp / start_temp) / total

    it = 0
    while time.time() < deadline:
        it += 1
        src = np.random.randint(n)
        if cur_a[src] <= 1e-12:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue
        max_delta = min(cur_a[src], 0.03)
        delta = np.random.rand() * max_delta

        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_peak = _fft_self_convolve(a_new).max()
        delta_obj = new_peak - cur_peak
        elapsed = time.time() - start
        temperature = start_temp * math.exp(-cooling_rate * elapsed)

        accept = False
        if delta_obj < 0:
            accept = True
        else:
            prob = math.exp(-delta_obj / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_a = a_new
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_a = cur_a.copy()

        # occasional reset to the best found so far
        if it % 15000 == 0 and np.random.rand() < 0.05:
            cur_a = best_a.copy()
            cur_peak = best_peak

    return best_a, best_peak


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a step‑function with a low score.

    Returns a plain Python list ready for `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Optional warm start from a previously discovered construction.
    base_seq = None
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_seq = _project_onto_simplex(base_seq, s=1.0)
        base_n = len(base_seq)
    except Exception:
        base_seq = None
        base_n = 3500

    # ------------------------------------------------------------------
    # Lengths to explore (ensure at least 500 entries).
    lengths = sorted(
        set(
            max(500, int(base_n * f))
            for f in (0.5, 0.7, 1.0, 1.3, 1.6)
            if int(base_n * f) >= 500
        )
    )
    if not lengths:
        lengths = [max(500, base_n)]

    # Keep the K best candidates (by cheap proxy) for later refinement.
    K = 7
    best_cands = []   # list of tuples (fast_score, vector, tag, n)

    def _push(vec: np.ndarray, tag: str, n: int):
        vec = _project_onto_simplex(vec, s=1.0)
        peak = _fft_self_convolve(vec).max()
        fast = 2.0 * n * peak
        if len(best_cands) < K:
            best_cands.append((fast, vec.copy(), tag, n))
            best_cands.sort(key=lambda x: x[0])
        else:
            if fast < best_cands[-1][0]:
                best_cands[-1] = (fast, vec.copy(), tag, n)
                best_cands.sort(key=lambda x: x[0])

    print("[phase 1] broad exploration of families")
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # deterministic families
        _push(_arcsine_distribution(n), "arcsine", n)
        _push(np.full(n, 1.0 / n, dtype=np.float64), "uniform", n)

        # Dirichlet draws
        for conc in (0.2, 0.5, 1.0, 2.0):
            _push(np.random.dirichlet(np.full(n, conc)), f"dirichlet_{conc}", n)

        # symmetric beta
        for a in (0.35, 0.45, 0.55, 0.65, 0.75):
            _push(_beta_distribution(n, a, a), f"beta_sym_{a:.2f}", n)

        # asymmetric beta (random)
        for _ in range(12):
            a = 10 ** np.random.uniform(-0.5, 0.7)   # ≈0.3 … 5
            b = 10 ** np.random.uniform(-0.5, 0.7)
            _push(_beta_distribution(n, a, b), f"beta_asym_{a:.2f}_{b:.2f}", n)

        # power‑law boundary
        for p in (0.38, 0.42, 0.46, 0.50, 0.54, 0.58):
            _push(_powerlaw_boundary_distribution(n, p), f"pow_bound_{p:.2f}", n)

        # interior power‑law
        for p in (0.6, 0.8, 1.0, 1.2):
            _push(_powerlaw_interior_distribution(n, p), f"pow_int_{p:.2f}", n)

        # exponential centered
        for lam in (0.001, 0.003, 0.008, 0.02):
            _push(_exponential_distribution(n, lam), f"exp_{lam:.4f}", n)

        # mixtures arcsine / uniform
        arcs = _arcsine_distribution(n)
        uni = np.full(n, 1.0 / n, dtype=np.float64)
        for w in (0.2, 0.4, 0.6, 0.8):
            _push(_mix_vectors(arcs, uni, w), f"mix_arc_uni_{w:.1f}", n)

        # previous construction interpolated, if available
        if base_seq is not None:
            _push(_interpolate_to_length(base_seq, n), "prev_interp", n)

    # ------------------------------------------------------------------
    if not best_cands:
        n = max(500, base_n)
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve(best_seq).max()
        print("[fallback] using uniform distribution")
    else:
        best_score, best_seq, best_tag, best_n = best_cands[0]
        print(f"[init] best start: tag={best_tag} n={best_n} fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – continuous refinement on the top few candidates.
    time_left = deadline - time.time()
    if time_left > 8.0 and best_cands:
        num_refine = min(len(best_cands), 3)
        per_refine = time_left * 0.6 / num_refine
        refined = []
        for i in range(num_refine):
            _, vec, tag, n = best_cands[i]
            print(f"[phase 2] refinement #{i+1} ({tag}, n={n}) – {per_refine:.1f}s")
            adam_dead = time.time() + per_refine * 0.55
            a_opt, _ = _adam_optimize(vec, adam_dead)
            sub_dead = time.time() + per_refine * 0.30
            a_opt = _refine_max(a_opt, sub_dead)
            refined.append((2.0 * n * _fft_self_convolve(a_opt).max(), a_opt, n))
        refined.sort(key=lambda x: x[0])
        best_score, best_seq, best_n = refined[0]

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing polishing (if time permits).
    time_left = deadline - time.time()
    if time_left > 1.0:
        print(f"[phase 3] simulated annealing – {time_left:.1f}s left")
        a_fin, fin_peak = _simulated_annealing(best_seq, deadline)
        best_seq = a_fin
        best_score = 2.0 * best_n * fin_peak
        print(f"[phase 3] SA fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final exact evaluation.
    try:
        true_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence score={true_score:.6f}")
        if true_score < best_score:
            best_score = true_score
    except Exception as exc:
        print("[final] evaluate_sequence raised:", exc)

    print(f"Best approx. score={best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
