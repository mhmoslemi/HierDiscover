(sum = 1) because the objective is scale‑invariant.  For a given
length `n` we:

1. Build a modest pool of analytic candidates (Beta, exponential,
   sinusoidal, block, random Dirichlet, previous best if available).
2. Keep the few best according to the cheap objective
   `2·n·max(conv)` (exactly the black‑box value when the sum is 1).
3. Refine each starter with an *exponentiated‑gradient* (mirror‑descent
   on the simplex) on the true sub‑gradient of `max(conv)`.  The
   sub‑gradient is obtained by averaging the gradients of the active
   convolution indices (those attaining the maximum).  A decreasing
   stepsize `η_t = η0/√t` is used, together with occasional jitter to
   escape plateaus.
4. The best vector seen overall (over several lengths) is returned.
   The routine respects the 1000 s budget and prints progress.

The algorithm deliberately avoids the costly soft‑max/Adam machinery
used previously and instead exploits the exact sub‑gradient together
with the natural KL‑mirror step (exponentiated update) which is
well‑suited for simplex optimisation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float = 1.0) -> np.ndarray:
    """Geometric‑like decreasing shape: a_i ∝ exp(-lam·i/(n‑1))."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * i / max(1, n - 1))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _exact_subgradient(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(a * a) (full linear convolution) with respect to a.
    Returns a vector of length n.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_val = conv.max()
    # tolerance to collect the active set
    eps = 1e-12 * (abs(max_val) + 1.0)
    active = np.flatnonzero(conv >= max_val - eps)

    # limit the active set for speed (keep at most 200 indices)
    if active.size > 200:
        # keep the 200 largest values among the active indices
        topk = 200
        act_vals = conv[active]
        idx = np.argpartition(-act_vals, topk - 1)[:topk]
        active = active[idx]

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    # gradient = 2 * (a_rev * w) restricted to the n central entries
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 1234, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step sequence that minimises ``evaluate_sequence``.
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5          # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a few problem sizes around the previous best (if any)
    try:
        base_n = len(height_sequence_1)               # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    factors = (0.7, 0.85, 1.0, 1.15, 1.3, 1.5)
    lengths = sorted({max(3000, int(base_n * f)) for f in factors})
    # impose a hard upper bound to keep each iteration fast
    lengths = [min(15000, n) for n in lengths]

    overall_best_vec = None          # type: np.ndarray | None
    overall_best_score = np.inf

    # ------------------------------------------------------------------
    for n in lengths:
        now = time.time()
        if now >= deadline:
            break
        remaining_global = deadline - now
        # allocate at most 25 % of the remaining time (capped at 120 s)
        slice_budget = min(remaining_global * 0.25, 120.0)
        slice_deadline = now + slice_budget
        print(f"[{now:.1f}] Exploring length n={n}")

        # ------------------------------------------------------------------
        # 1. Build a modest pool of analytic candidates
        pool = []

        # systematic Beta grid (log‑space)
        alphas = np.logspace(-0.7, 0.7, num=7)   # ~0.2 … 5
        betas = np.logspace(-0.7, 0.7, num=7)
        for a in alphas:
            for b in betas:
                pool.append(_beta_distribution(n, a, b))

        # a few hand‑crafted shapes
        pool.append(_beta_distribution(n, 0.2, 0.8))   # decreasing
        pool.append(_beta_distribution(n, 0.8, 0.2))   # increasing
        pool.append(_exponential_distribution(n, lam=0.5))
        pool.append(_exponential_distribution(n, lam=1.5))
        pool.append(_exponential_distribution(n, lam=3.0))

        # sinusoidal shapes
        i = np.arange(n, dtype=np.float64)
        sine = np.sin(np.pi * (i + 0.5) / n) ** 2
        sine /= sine.sum()
        pool.append(sine)

        cosine = (1 - np.cos(2 * np.pi * i / n)) / 2
        cosine /= cosine.sum()
        pool.append(cosine)

        # block shape (first half uniform)
        block = np.zeros(n, dtype=np.float64)
        block[: n // 2] = 1.0
        block /= block.sum()
        pool.append(block)

        # random Dirichlet draws with diverse concentrations
        for conc in (0.05, 0.2, 0.5, 1.0, 5.0, 10.0):
            pool.append(rng.dirichlet([conc] * n))

        # previous construction, interpolated if needed
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
            if prev.size != n:
                x_old = np.linspace(0.0, 1.0, prev.size)
                x_new = np.linspace(0.0, 1.0, n)
                prev = np.interp(x_new, x_old, prev)
            prev = _project_onto_simplex(prev, s=1.0)
            pool.append(prev)
        except Exception:
            pass

        # ------------------------------------------------------------------
        # 2. Cheap score = exact objective for sum = 1
        cheap = lambda a: 2.0 * n * _fft_self_convolve(a).max()
        scored = [(cheap(a), a) for a in pool]
        scored.sort(key=lambda x: x[0])
        starters = [vec.copy() for _, vec in scored[:5]]
        print(f"  top cheap scores (first 5): {[s for s, _ in scored[:5]]}")

        # ------------------------------------------------------------------
        # 3. Refine each starter with exponentiated‑gradient descent
        for sid, a0 in enumerate(starters):
            if time.time() >= slice_deadline:
                break

            a = a0.copy()
            a = np.maximum(a, 1e-15)          # avoid exact zeros for stability
            a /= a.sum()

            # evaluate initial candidate
            cur_score = cheap(a)
            if cur_score < overall_best_score:
                overall_best_score = cur_score
                overall_best_vec = a.copy()
                print(f"[{time.time():.1f}] New global best (init) n={n} sid={sid} score={cur_score:.6f}")

            # optimisation parameters
            max_iters = 20000
            eta0 = 0.30                       # base stepsize
            for it in range(1, max_iters + 1):
                if time.time() >= slice_deadline:
                    break

                # convolution and current maximum
                conv = _fft_self_convolve(a)
                max_val = conv.max()
                cur_score = 2.0 * n * max_val

                # sub‑gradient of the max
                grad = _exact_subgradient(a)

                # diminishing stepsize
                eta = eta0 / np.sqrt(it)

                # exponentiated‑gradient (mirror descent on simplex)
                a = a * np.exp(-eta * grad)
                s = a.sum()
                if s == 0.0 or not np.isfinite(s):
                    # re‑initialise in the unlikely event of numerical collapse
                    a = np.full(n, 1.0 / n, dtype=np.float64)
                else:
                    a /= s

                # occasional random jitter to escape plateaus
                if it % 5000 == 0 and rng.random() < 0.2:
                    jitter = rng.normal(scale=1e-6, size=n)
                    a = a + jitter
                    a = np.maximum(a, 0.0)
                    a /= a.sum()

                # periodic reporting and global‑best update
                if it % 1000 == 0:
                    if cur_score < overall_best_score:
                        overall_best_score = cur_score
                        overall_best_vec = a.copy()
                        print(
                            f"[{time.time():.1f}] Global improvement n={n} sid={sid} it={it} "
                            f"score={overall_best_score:.6f}"
                        )
                        if overall_best_score <= 1.503:
                            print("Target reached – exiting early.")
                            return overall_best_vec.tolist()
        # ------------------------------------------------------------------
        # end of this length, move on
        if time.time() >= deadline:
            break
        if overall_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # fallback in the unlikely event we found nothing
    if overall_best_vec is None:
        n_fallback = 4000
        overall_best_vec = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        overall_best_score = cheap(overall_best_vec)
        print("Fallback uniform distribution used.")

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_vec)})")
    return overall_best_vec.tolist()
