def _simulated_annealing(a, deadline):
    cur_a = a.copy()
    cur_score = _max_convolution_score(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score
    n = len(a)
    iter_count = 0
    T0 = 0.1 * cur_score
    while time.time() < deadline:
        iter_count += 1
        T = T0 * (0.9995 ** iter_count)  # exponential decay
        # propose move
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j: continue
        # move delta mass
        max_delta = min(cur_a[i], 0.05)  # up to 5% of total mass
        delta = rng.random() * max_delta
        a_new = cur_a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # ensure nonnegative and sum unchanged (it does)
        score = _max_convolution_score(a_new)
        delta_score = score - cur_score
        if delta_score < 0 or rng.random() < np.exp(-delta_score / max(T,1e-12)):
            cur_a = a_new
            cur_score = score
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                print(...)
        # occasional restart: random perturbation
        if iter_count % 2000 == 0:
            pert = rng.normal(scale=0.01, size=n)
            cur_a = np.maximum(cur_a + pert, 0)
            cur_a = cur_a / cur_a.sum()
            cur_score = _max_convolution_score(cur_a)
    return best_a, best_score

"""
A fast multi‑restart search for a non‑negative step function (vector) that
minimises the evaluation function

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm is deliberately different from the reference implementation:

* a very cheap global scan over several lengths (up to 16 000) using a few
  analytic “U‑shaped’’ families;
* many random Dirichlet starts (different concentration parameters);
* for each start a short Adam optimisation of the smooth
  log‑sum‑exp approximation of the max‑convolution, followed by
  an exact max‑subgradient phase;
* a stochastic “hill‑climber’’ that transfers a tiny amount of mass
  between two indices;
* a lightweight simulated‑annealing phase that occasionally accepts
  uphill moves; this helps escaping plateaux that Adam/sub‑gradient alone
  cannot leave.
* the best vector ever seen (across all restarts and all lengths) is kept
  and returned.

All vectors are kept on the simplex (sum = 1); the objective is
scale‑invariant, so the final list can be returned as‑is.
The whole search respects the 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of a and b using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) for a vector already normalised to sum = 1.
    This equals the value returned by ``evaluate_sequence`` (scale‑invariant).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_two_exp(n: int, lam: float = 0.02) -> np.ndarray:
    """
    Two symmetric exponentials peaked at the ends (U‑shaped).
    """
    idx = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * idx) + np.exp(-lam * (n - 1 - idx))
    vals /= vals.sum()
    return vals


def _distribution_u_shape(n: int) -> np.ndarray:
    """
    Simple rational U‑shape: 1/(i+1) + 1/(n-i)  (high at the ends, low in centre).
    """
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / (i + 1) + 1.0 / (n - i)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_candidates(n: int) -> list[np.ndarray]:
    """Few analytic starts; also a sparse random start."""
    cand = [
        _distribution_uniform(n),
        _distribution_arcsine(n),
        _distribution_triangular(n),
        _distribution_exponential(n, lam=0.003),
        _distribution_exponential(n, lam=0.02),
        _distribution_two_exp(n, lam=0.02),
        _distribution_u_shape(n),
    ]
    # symmetric beta families (U‑shaped for p<1)
    for p in (0.2, 0.5, 0.8):
        cand.append(_distribution_beta_sym(n, p))

    # a random sparse start (≈1 % non‑zeros)
    rnd = np.zeros(n, dtype=np.float64)
    k = max(2, n // 100)                # ~1 % of entries
    idx = np.random.choice(n, k, replace=False)
    rnd[idx] = np.random.rand(k)
    rnd /= rnd.sum()
    cand.append(rnd)

    return cand


# ----------------------------------------------------------------------
def _smooth_descent(a: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth log‑sum‑exp approximation
    μ_α(a) = (1/α)·log Σ exp(α·(a*a)_k).
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # smoothing schedule – a few thousand steps total per run
    schedule = [(1.0, 800), (5.0, 800), (20.0, 800), (100.0, 800), (500.0, 800)]

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score
            t += 1

            # self‑convolution (full)
            b = _fft_self_convolve(a)

            # soft‑max weights over the convolution entries
            maxb = b.max()
            exp_terms = np.exp(alpha * (b - maxb))
            w = exp_terms / exp_terms.sum()          # shape (2n‑1,); Σ w = 1

            # gradient = 2 * conv_rev(a[::-1], w) (first n entries)
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam update + projection onto simplex
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a, 1.0)

            # occasional exact evaluation
            if t % 200 == 0:
                cur_score = _max_convolution_score(a)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_seq = a.copy()
    return best_seq, best_score


# ----------------------------------------------------------------------
def _exact_subgradient_refine(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Exact max‑subgradient descent with momentum (same idea as the reference).
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)

    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps)

        w = np.zeros_like(b)
        w[active] = 1.0

        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / np.sqrt(iteration + 1.0)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad iter={iteration} new best {best_score:.6f}")

        # early stop if we have already reached the target region
        if best_score <= 1.48:
            break
        # small safety check
        if time.time() + 0.5 > deadline:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_hill_climb(a: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Random transfer hill‑climbing on the simplex.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)
    cur_score = best_score

    max_transfer = 0.08          # up to 8 % of total mass per move
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        transfer = max_transfer * (0.5 ** (iteration / 3500.0))

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = rng.random() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-14:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[{time.time():.1f}] hill iter={iteration} improved best {best_score:.6f}")

        # occasional jitter to shake off plateaux
        if iteration % 4000 == 0 and rng.random() < 0.2:
            pert = rng.normal(scale=1e-4, size=n)
            a = best_seq + pert
            a = _project_onto_simplex(a, 1.0)
            cur_score = _max_convolution_score(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Very lightweight simulated‑annealing on the simplex.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = _max_convolution_score(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    # Initial temperature – a fraction of the current objective.
    T0 = max(0.05 * cur_score, 1e-3)
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        # exponential cooling
        T = T0 * (0.9995 ** iteration)

        # propose a move: transfer a small amount from i to j
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(cur_a[i], 0.07)          # ≤7 % of total mass
        delta = rng.random() * max_delta
        if delta == 0:
            continue

        a_new = cur_a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = _max_convolution_score(a_new)
        dE = new_score - cur_score

        if dE < 0 or rng.random() < math.exp(-dE / max(T, 1e-12)):
            cur_a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[{time.time():.1f}] SA iter={iteration} new best {best_score:.6f}")

        # occasional random re‑perturbation to avoid stagnation
        if iteration % 5000 == 0:
            # blend the current best with a fresh random Dirichlet draw
            alpha = 0.3 + 0.7 * rng.random()
            tmp = rng.dirichlet(np.ones(n) * alpha)
            cur_a = _project_onto_simplex(alpha * best_a + (1 - alpha) * tmp, 1.0)
            cur_score = _max_convolution_score(cur_a)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence (Python list of floats) found within the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # 1.  Scan a handful of lengths with cheap analytic families.
    # ------------------------------------------------------------------
    # Lengths we try – up to about 16 000 (larger start‑up cost but more freedom)
    lengths = [2000, 4000, 8000, 12000, 16000]
    overall_best_seq = None
    overall_best_score = float("inf")
    best_length = None

    for n in lengths:
        if time.time() >= deadline:
            break
        # generate a few analytic candidates for this length
        for cand in _initial_candidates(n):
            if time.time() >= deadline:
                break
            cand = cand / cand.sum()
            score = _max_convolution_score(cand)
            if score < overall_best_score:
                overall_best_score = score
                overall_best_seq = cand.copy()
                best_length = n
                print(f"[{time.time():.1f}] len={n} init score={score:.6f}")

    # ------------------------------------------------------------------
    # 2.  If a previously found construction is present, consider it.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                best_length = prev.size
                print(f"[{time.time():.1f}] prev best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  If nothing survived, fall back to a uniform vector.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = max(lengths[0], 2000)
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _max_convolution_score(overall_best_seq)
        best_length = n
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Main multi‑restart optimisation on the chosen length.
    # ------------------------------------------------------------------
    n = best_length
    # the duration we reserve for the exhaustive phase
    max_restarts = 12
    restart_counter = 0

    while time.time() < deadline and restart_counter < max_restarts:
        restart_counter += 1
        # Choose a start: either a random Dirichlet (different concentration)
        # or reuse the best we have so far with a tiny jitter.
        if rng.random() < 0.3:
            # Dirichlet start – concentration parameter swept between 0.1 and 2
            conc = 10 ** rng.uniform(-1.0, 0.3)   # ~0.1 … 2
            start = rng.dirichlet(np.ones(n) * conc)
        else:
            # jittered copy of current overall best
            start = overall_best_seq + rng.normal(scale=1e-3, size=n)
            start = _project_onto_simplex(start, 1.0)

        # ------------------------------------------------------------------
        # a) smoothed Adam descent
        # ------------------------------------------------------------------
        a1, s1 = _smooth_descent(start, deadline, rng)

        # ------------------------------------------------------------------
        # b) exact max‑subgradient refinement
        # ------------------------------------------------------------------
        a2, s2 = _exact_subgradient_refine(a1, deadline)

        # ------------------------------------------------------------------
        # c) local random‑transfer hill climbing
        # ------------------------------------------------------------------
        a3, s3 = _local_hill_climb(a2, deadline, rng)

        # ------------------------------------------------------------------
        # d) simulated annealing (lightweight)
        # ------------------------------------------------------------------
        a4, s4 = _simulated_annealing(a3, deadline, rng)

        # ------------------------------------------------------------------
        # e) keep the best of this restart
        # ------------------------------------------------------------------
        final_seq, final_score = a4, s4
        if final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = final_seq.copy()
            print(f"[{time.time():.1f}] ***** NEW GLOBAL BEST {overall_best_score:.6f} (len={n})")

        # if we have already crossed the target, we may stop early
        if overall_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 5.  Final optional fine‑tuning of the best vector (time‑permitting).
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        # a brief extra refinement – only subgradient steps because they are cheap
        a_final, s_final = _exact_subgradient_refine(overall_best_seq, deadline)
        if s_final < overall_best_score:
            overall_best_score = s_final
            overall_best_seq = a_final.copy()
            print(f"[{time.time():.1f}] final refinement improved to {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Return the best sequence found (as a Python list of floats)
    # ------------------------------------------------------------------
    print(f"=== DONE === best score {overall_best_score:.6f} (length={overall_best_seq.shape[0]})")
    # The caller will evaluate the vector with ``evaluate_sequence``.
    return overall_best_seq.tolist()
