"""
Propose a low‑score step function.

The search is split in two phases:

1️⃣  **Coarse random exploration** on moderate lengths (2k‑8k).  
    Families: uniform, arcsine, symmetric Beta(α,α) on a grid,
    random Beta draws, Dirichlet draws, mixtures of two/three Betas,
    and mixtures with a uniform component.  All candidates are projected
    onto the simplex (∑a=1) and evaluated with a fast FFT‑based proxy
    `2·n·max(a*a)` (exact for sum‑1 vectors).

2️⃣  **Refinement** – a mirror‑descent (exponentiated‑gradient) on the
    convex objective `max_k (a*a)_k`.  The sub‑gradient of the
    sup‑norm is obtained analytically; the update
    `a ← a·exp(-η·g)` keeps non‑negativity and a final normalisation
    re‑projects onto the simplex.  The step size decays like
    η_t = η₀/√t.  The best sequence is then up‑sampled to larger
    lengths (12k‑20k) and refined again.  The procedure runs until the
    supplied time budget (≈1000 s) expires.

The final sequence (sum≈1) is returned; `evaluate_sequence` rescales it
automatically, so the printed score equals the true objective.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution of length n, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf, stable version
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logpdf = np.minimum(logpdf, 700.0)       # avoid overflow in exp
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a * a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """
    Proxy score for a vector that already satisfies Σa = 1.
    The value equals the true objective because the evaluator is scale‑invariant.
    """
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of   f(a) = max_k (a*a)_k .
    Returns (grad, cur_max).  grad has the same shape as a.
    """
    n = a.shape[0]
    b = _self_convolution_full(a)
    cur_max = b.max()
    # tolerance for the active set
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_mirror_descent(a: np.ndarray, deadline: float,
                           rng: np.random.Generator,
                           eta0: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) refinement on the simplex.
    Returns the best sequence and its raw score found before `deadline`.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _raw_score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        grad, _ = _max_subgradient_step(a)
        eta = eta0 / math.sqrt(it)

        # multiplicative update; clamp exponent to avoid overflow/underflow
        exponent = -eta * grad
        exponent = np.clip(exponent, -50, 50)
        a = a * np.exp(exponent)

        # re‑normalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # occasional tiny jitter to escape plateaus
        if it % 500 == 0 and rng.random() < 0.06:
            a += rng.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        cur_val = _raw_score(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] mirror‑descent len={n} it={it} score={best_val:.6f}")

    return best_a, best_val

            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score, best_len = vec.copy(), sc, n
                print(f"   mix2 λ={lam:.3f} score={best_score:.6f}")

        # Mixture of three Betas
        for _ in range(150):
            if time.time() >= deadline:
                break
            a1, b1 = 10 ** rng.uniform(-2, 1), 10 ** rng.uniform(-2, 1)
            a2, b2 = 10 ** rng.uniform(-2, 1), 10 ** rng.uniform(-2, 1)
            a3, b3 = 10 ** rng.uniform(-2, 1), 10 ** rng.uniform(-2, 1)
            weights = rng.dirichlet([1, 1, 1])
            p1 = _beta_distribution(n, a1, b1)
            p2 = _beta_distribution(n, a2, b2)
            p3 = _beta_distribution(n, a3, b3)
            vec = (weights[0] * p1 + weights[1] * p2 + weights[2] * p3)
            vec = _project_onto_simplex(vec)
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score, best_len = vec.copy(), sc, n
                print(f"   mix3 score={best_score:.6f}")

        # Uniform + Beta mixture
        for _ in range(150):
            if time.time() >= deadline:
                break
            alpha = 10 ** rng.uniform(-2, 1)
            beta = 10 ** rng.uniform(-2, 1)
            beta_vec = _beta_distribution(n, alpha, beta)
            lam = rng.random()
            vec = lam * beta_vec + (1.0 - lam) * uniform
            vec = _project_onto_simplex(vec)
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score, best_len = vec.copy(), sc, n
                print(f"   mixU λ={lam:.3f} score={best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] best score after coarse search = {best_score:.6f} (len={best_len})")

    # ------------------------------------------------------------------
    # Phase 2 – mirror‑descent refinement on the current best length
    if time.time() < deadline:
        a = best_seq.copy()
        a, best_score = _refine_mirror_descent(a, deadline, rng, eta0=0.4)
        best_seq = a.copy()
        best_len = len(a)
        print(f"[{time.time():.1f}] after refinement: score={best_score:.6f} (len={best_len})")

    # ------------------------------------------------------------------
    # Phase 3 – up‑sample to larger lengths (reducing discretisation error)
    larger_lengths = [12000, 16000, 20000]
    for n in larger_lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] up‑sampling to length {n}")
        a = _interpolate_to_length(best_seq, n)
        a = _project_onto_simplex(a)   # safety
        a, scr = _refine_mirror_descent(a, deadline, rng, eta0=0.35)
        if scr + 1e-12 < best_score:
            best_seq, best_score, best_len = a.copy(), scr, n
            print(f"   new best after up‑sample: score={best_score:.6f} (len={best_len})")
        # early stop if we have already passed the target bound
        if best_score <= 1.503:
            print("Target bound reached – exiting early.")
            break

    # ------------------------------------------------------------------
    # Final evaluation with the provided evaluator (exact arithmetic, clipping, etc.)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time():.1f}] final evaluated score: {final_score:.6f} (len={len(best_seq)})")
    return best_seq.tolist()
