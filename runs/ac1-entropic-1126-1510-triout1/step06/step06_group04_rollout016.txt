def _smooth_refine(a, n, max_iters, deadline):
    best_a = a.copy()
    best_score = _raw_score(a)
    beta = 5.0
    step0 = 0.2 # may tune
    momentum = np.zeros_like(a)
    mu = 0.9
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        b = _self_convolution_full(a)  # length 2n-1
        # compute softmax weights with beta scaling
        # for numerical stability subtract max before exp
        max_b = b.max()
        expb = np.exp(beta * (b - max_b))
        w = expb / expb.sum()
        # gradient
        grad = 2.0 * _fft_convolve(a[::-1], w)[n-1 : n-1 + n]
        # step size
        step = step0 / math.sqrt(it)
        # momentum
        momentum = mu * momentum + (1.0 - mu) * grad
        a_new = a - step * momentum
        a_new = _project_onto_simplex(a_new)
        a = a_new
        # maybe increase beta slowly
        if it % 1000 == 0:
            beta = min(beta * 1.1, 5000.0) # avoid overflow
        # optionally compute raw score every few iterations and track best
        if it % 200 == 0:
            cur_score = _raw_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] ref n={n} it={it} score={best_score:.6f}")
    return best_a

"""
Propose a low‑score step function.

The search consists of two phases:
1.  A broad random / grid search over a handful of lengths (2000‑12000)
    using families of Beta, Dirichlet and simple mixtures.  The
    best K≈10 candidates (according to the fast proxy 2·n·max(a*a))
    are kept.
2.  Each retained candidate is refined by a smooth approximation of the
    max‑convolution objective.  For a temperature β the loss
        Lβ(a) = (1/β)·log(∑_k exp(β·(a*a)_k))
    is a smooth, differentiable surrogate for max(a*a).  Its gradient
    can be computed via FFT in O(n log n).  An Adam‑style update with
    projection onto the simplex is applied while β is slowly increased,
    driving the surrogate towards the true max.  The best refined
    sequence over all candidates is finally returned.

The whole routine respects the 1000 s time budget, never returns a
negative‑infinity score and guarantees non‑negative entries (simplex
projection).  ``evaluate_sequence`` is called only once at the end.
"""
import time
import math
import heapq
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) pmf of length n, normalized to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n            # bin mid‑points
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # protect against overflow / under‑flow
    logpdf = np.minimum(logpdf, 700.0)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    tot = pdf.sum()
    if tot == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= tot
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a * a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """
    Proxy score for a normalized to Σa = 1 :  2·n·max(a*a).
    This equals the true evaluator because ``evaluate_sequence`` is
    scale‑invariant.
    """
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _smooth_refine(a0: np.ndarray,
                   max_iters: int,
                   deadline: float,
                   init_beta: float = 5.0,
                   beta_factor: float = 1.02,
                   beta_max: float = 5000.0,
                   lr: float = 0.2) -> np.ndarray:
    """
    Refine ``a0`` by minimizing a smooth approximation of the
    max‑convolution using Adam‑style updates and simplex projection.

    Returns the best vector seen (according to the exact proxy score).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = _raw_score(a)

    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    beta = init_beta
    t = 0

    while t < max_iters and time.time() < deadline:
        t += 1
        # --- compute convolution and softmax weights ---
        b = _self_convolution_full(a)          # length 2n‑1
        max_b = b.max()
        # stability: shift by max before exp
        exp_w = np.exp(beta * (b - max_b))
        w = exp_w / exp_w.sum()                # shape (2n‑1)

        # --- gradient of the smoothed objective ---
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1: n - 1 + n]

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a)

        # --- schedule for β ---
        if t % 500 == 0:
            beta = min(beta * beta_factor, beta_max)

        # --- keep the best exact score seen so far ---
        if t % 200 == 0:
            cur_sc = _raw_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
                print(f"[{time.time():.1f}]  iter={t}  beta={beta:.1f}  score={best_sc:.6f}")

    return best_a
