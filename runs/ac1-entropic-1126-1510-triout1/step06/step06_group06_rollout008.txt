    x = (np.arange(n) + 0.5) / n

"""
Improved step‑function search.

The algorithm consists of four stages:

1️⃣  Broad sampling of many simple families (arcsine, uniform, random,
    symmetric/asymmetric Beta, symmetric exponential and a two‑level
    block family).  The best sequence according to the cheap proxy
    `2·n·max(conv)` is retained.

2️⃣  Adam optimisation of a smooth‑max surrogate of the ℓ∞‑convolution.
    The smoothness parameter is increased continuously up to a large
    value, and the best exact peak encountered is kept.

3️⃣  Projected sub‑gradient descent that always follows the *single*
    active peak (instead of the uniform combination of all active
    peaks).  The step‑size decays as 1/√t.

4️⃣  A greedy mass‑transfer local search that moves a small fraction of
    probability mass from the index that contributes most to the current
    peak towards an index that does not affect that peak (or contributes
    the least).  Only improving moves are accepted.

All vectors are projected onto the probability simplex Σa_i = 1 after
each update, which keeps the evaluation invariant (the black‑box
`evaluate_sequence` only depends on the shape, not on the absolute scale).
The routine respects the supplied time budget (default 1000 s) and prints
progress information.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth‑max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max

def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.02,
                   alpha_max: float = 1e4,
                   lr0: float = 0.35,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best vector encountered and its exact peak.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # keep the best exact peak seen so far
        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        # continuation on the smoothness parameter
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # tiny random jitter every few hundred iterations
        if t % 400 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 4e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak

def _peak_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient step that follows a *single* active peak (the one with
    maximal convolution value).  Returns (grad, cur_max).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()

    # pick a random index among the maximal ones
    peak_idx = np.random.choice(np.where(b >= cur_max - 1e-12)[0])

    # gradient of b[peak_idx] w.r.t a
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = peak_idx - i
    valid = (j >= 0) & (j < n)

    # for i ≠ j the derivative is 2·a[j]; for i == j it is a[i]
    grad[valid] = 2.0 * a[j[valid]]
    if peak_idx % 2 == 0:
        mid = peak_idx // 2
        if mid < n:
            grad[mid] = a[mid]       # correct the double‑count for the diagonal

    return grad, cur_max

def _directed_peak_search(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Greedy mass‑transfer that always attacks the current peak.
    Moves a modest fraction of probability from the index that contributes
    most to the peak towards an index that does *not* affect that peak.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        # pick a peak index (random among the active ones)
        peak_idx = np.random.choice(np.where(b >= cur_max - 1e-12)[0])

        # contribution of each i to that peak:
        i = np.arange(n)
        j = peak_idx - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[valid] = a[i[valid]] * a[j[valid]]

        # source = index with largest contribution (if any)
        if contrib.max() > 0:
            src = np.argmax(contrib)
        else:
            src = np.random.randint(n)

        # destination = index that does *not* affect this peak, i.e. out of range
        out_of_range = np.where(~valid)[0]
        if out_of_range.size > 0:
            dst = np.random.choice(out_of_range)
        else:
            # otherwise choose index with minimal contribution (but not src)
            dst = np.argmin(contrib + (np.arange(n) == src) * 2.0)  # break tie

        if dst == src:
            dst = (src + 1) % n

        # transfer a small fraction of src mass
        max_delta = min(a[src], 0.05)       # at most 5 % of src
        delta = np.random.rand() * max_delta

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        b_new = _fft_self_convolve(a_new)
        new_max = b_new.max()

        if new_max < cur_max - 1e-15:       # accept only improving moves
            a = a_new
            b = b_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()

        if np.random.rand() < 0.004:        # occasional jitter
            a += np.random.randn(n) * 1e-7
            a = _project_onto_simplex(a, s=1.0)
            b = _fft_self_convolve(a)
            cur_max = b.max()

    return best_a

def _two_level_family(n: int, deadline: float):
    """
    Simple two‑level family: a front block of height p and a back block of
    height q with p = r·q, r ≥ 1.
    """
    best_seq = None
    best_score = float('inf')
    step_L = max(1, n // 150)                   # ≈0.7 % of the length
    r_grid = np.logspace(0.0, 3.0, num=30)      # ratios 1 … 1000

    for L in range(step_L, n, step_L):
        if time.time() > deadline:
            break
        denom_base = n - L
        for r in r_grid:
            denom = L * r + denom_base
            q = 1.0 / denom
            p = r * q
            seq = np.empty(n, dtype=np.float64)
            seq[:L] = p
            seq[L:] = q
            peak = _fft_self_convolve(seq).max()
            score = 2.0 * n * peak
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
    return best_seq, best_score

def _beta_symmetric_family(n: int, alist):
    """Return a list of symmetric Beta‑density vectors for given α‑values."""
    families = []
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    for a in alist:
        log_vals = (a - 1.0) * np.log(x) + (a - 1.0) * np.log(1.0 - x)
        vals = np.exp(log_vals)
        vals[~np.isfinite(vals)] = 0.0
        total = vals.sum()
        if total == 0.0:
            vals = np.full(n, 1.0 / n)
        else:
            vals /= total
        families.append(vals)
    return families


def _beta_asymmetric_family(n: int, pairs):
    """Return a list of asymmetric Beta vectors for sampled (α,β) pairs."""
    families = []
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    for a, b in pairs:
        log_vals = (a - 1.0) * np.log(x) + (b - 1.0) * np.log(1.0 - x)
        vals = np.exp(log_vals)
        vals[~np.isfinite(vals)] = 0.0
        total = vals.sum()
        if total == 0.0:
            vals = np.full(n, 1.0 / n)
        else:
            vals /= total
        families.append(vals)
    return families


def _exp_sym_family(n: int, lambdas):
    """Symmetric exponential decay a_i ∝ exp(-λ·i)+exp(-λ·(n‑1‑i))."""
    families = []
    idx = np.arange(n, dtype=np.float64)
    for lam in lambdas:
        v = np.exp(-lam * idx) + np.exp(-lam * (n - 1 - idx))
        v /= v.sum()
        families.append(v)
    return families

def _generate_candidates(lengths, deadline):
    """
    Scan a variety of parametric families for each length and keep the
    best vector (according to the cheap proxy 2·n·max(conv)).
    """
    overall_best_seq = None
    overall_best_score = float('inf')

    # parameters for the families
    sym_beta_alphas = [0.05, 0.07, 0.10, 0.12, 0.15, 0.20,
                       0.25, 0.35, 0.45, 0.55, 0.65, 0.75, 0.85, 1.0]
    # a handful of random asymmetric pairs
    asym_pairs = [(np.random.rand()*1.5+0.05, np.random.rand()*1.5+0.05) for _ in range(15)]

    exp_lams = [0.0001, 0.0002, 0.0005, 0.001, 0.002, 0.003]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[candidate] exploring length {n}")

        starters = {}

        # 1. deterministic families
        starters["arcsine"] = (1.0 / np.sqrt((np.arange(n) + 0.5) *
                                            (n - np.arange(n) - 0.5)))
        starters["arcsine"] /= starters["arcsine"].sum()

        starters["uniform"] = np.full(n, 1.0 / n, dtype=np.float64)

        # 2. random seeds
        for k in range(2):
            rng = np.random.rand(n)
            rng /= rng.sum()
            starters[f"rand{k}"] = rng

        # 3. symmetric Beta families
        for vec, a in zip(_beta_symmetric_family(n, sym_beta_alphas), sym_beta_alphas):
            starters[f"beta_sym_{a:.2f}"] = vec

        # 4. asymmetric Beta families
        for vec, (a, b) in zip(_beta_asymmetric_family(n, asym_pairs), asym_pairs):
            starters[f"beta_asym_{a:.2f}_{b:.2f}"] = vec

        # 5. symmetric exponential families
        for vec, lam in zip(_exp_sym_family(n, exp_lams), exp_lams):
            starters[f"exp_{lam:.4f}"] = vec

        # 6. mixture of arcsine & uniform (a few weights)
        for w in (0.2, 0.4, 0.6):
            mix = (1.0 - w) * starters["arcsine"] + w * starters["uniform"]
            starters[f"mix_{w:.1f}"] = mix

        # 7. two‑level block family (fast‑scored later)
        # (we treat it separately because it is a search loop)

        # evaluate all simple starters
        for tag, vec in starters.items():
            vec = _project_onto_simplex(vec, s=1.0)
            peak = _fft_self_convolve(vec).max()
            fast_score = 2.0 * n * peak
            if fast_score < overall_best_score:
                overall_best_score = fast_score
                overall_best_seq = vec.copy()
                print(f"[init] n={n:4d} {tag:<16s} fast_score={fast_score:.6f}")

        # two‑level family (more expensive, but still cheap)
        seq, score = _two_level_family(n, deadline)
        if seq is not None and score < overall_best_score:
            overall_best_score = score
            overall_best_seq = seq.copy()
            print(f"[two‑level] n={n:4d} fast_score={score:.6f}")

    return overall_best_seq, overall_best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Entry point used by the evaluation framework.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0       # safety margin

    # ------------------------------------------------------------------
    # Try to load the previously found construction, if it exists.
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_seq = _project_onto_simplex(base_seq, s=1.0)
        base_n = len(base_seq)
    except Exception:
        base_seq = None
        base_n = 4000

    # ------------------------------------------------------------------
    # Lengths to explore – a spread around the previously known size.
    factors = (0.6, 0.8, 1.0, 1.2, 1.4, 1.6, 2.0)
    lengths = [max(500, int(base_n * f)) for f in factors if int(base_n * f) >= 500]
    lengths = sorted(set(lengths))

    # ------------------------------------------------------------------
    # Phase 0 – broad family sampling.
    print("[phase 0] broad family sampling")
    overall_best_seq, overall_best_score = _generate_candidates(lengths, deadline)

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation of the smooth‑max surrogate.
    time_left = deadline - time.time()
    if time_left > 6.0 and overall_best_seq is not None:
        n = len(overall_best_seq)
        print(f"[phase 1] Adam optimisation (n={n}) – time left {time_left:.1f}s")
        phase_end = time.time() + time_left * 0.55
        a_opt, _ = _adam_optimize(overall_best_seq, phase_end)
        overall_best_seq = a_opt.copy()
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print(f"[phase 1] after Adam fast_score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact sub‑gradient descent that follows a single peak.
    time_left = deadline - time.time()
    if time_left > 5.0 and overall_best_seq is not None:
        n = len(overall_best_seq)
        print(f"[phase 2] peak‑subgradient descent (n={n}) – time left {time_left:.1f}s")
        sub_end = time.time() + time_left * 0.55
        a_ref = _refine_max_peak(overall_best_seq, sub_end)
        overall_best_seq = a_ref.copy()
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print(f"[phase 2] after sub‑gradient fast_score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – greedy directed peak‑mass‑transfer search.
    time_left = deadline - time.time()
    if time_left > 4.0 and overall_best_seq is not None:
        n = len(overall_best_seq)
        print(f"[phase 3] directed peak search (n={n}) – time left {time_left:.1f}s")
        a_final = _directed_peak_search(overall_best_seq, deadline)
        overall_best_seq = a_final.copy()
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print(f"[phase 3] after local search fast_score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the black‑box objective.
    try:
        true_score = evaluate_sequence(overall_best_seq.tolist())
        print(f"[final] evaluate_sequence score={true_score:.6f}")
        if true_score < overall_best_score:
            overall_best_score = true_score
    except Exception as exc:
        print("[final] evaluate_sequence raised:", exc)

    print(f"Best approx. score {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
