"""
Improved search for a low‑score step function.

The objective 2·n·max(a*a) (with Σa = 1) does not depend on the overall scale,
so we can work on the simplex.  The algorithm consists of three stages:

1. **Broad exploration** – generate a few simple families (uniform,
   arcsine, symmetric Beta on a small grid, random Dirichlet and a sinusoidal
   perturbation of the uniform distribution).  The best raw score is kept.

2. **Additive Adam refinement** – treat the exact max‑convolution as a
   (non‑smooth) objective and run Adam on its sub‑gradient
   (the same sub‑gradient that was used in the previous state‑of‑the‑art).
   Adam’s adaptive step‑size and bias‑correction accelerate convergence
   compared with a plain sub‑gradient method.

3. **Exponentiated‑gradient (mirror‑descent) refinement** – because the
   feasible set is the probability simplex, a multiplicative update
   (a_i ← a_i·exp(‑η·grad_i)) together with a simple re‑normalisation
   often finds a slightly better local optimum.

Both refinements are interleaved with occasional tiny random jitters
to escape flat plateaus.  The whole search stops well before the 1000 s
budget, leaving a safety margin for the final call to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution of length n, normalized to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n           # mid‑points
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logpdf = np.minimum(logpdf, 700.0)                       # prevent overflow
    pdf = np.exp(logpdf)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (a and b are real vectors)."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a."""
    return _self_convolution_full(a).max()


# ----------------------------------------------------------------------
def _raw_score(a: np.ndarray) -> float:
    """Proxy score for Σa = 1 : 2·n·max(a*a)."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray):
    """
    Sub‑gradient of the max‑convolution (the same as in the reference code).
    Returns (grad, current_max).
    """
    n = a.shape[0]
    b = _self_convolution_full(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, gamma: float = 30.0):
    """
    Gradient of the smooth (log‑sum‑exp) approximation of max:
        f(a) = (1/gamma)·log Σ_k exp(gamma·(a*a)_k)
    Returns (grad, smooth_value).
    """
    n = a.shape[0]
    b = _self_convolution_full(a)                       # shape (2n‑1,)
    expb = np.exp(gamma * b - np.max(gamma * b))        # stability
    w = expb / expb.sum()                               # softmax weights, sum = 1
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    smooth_val = (np.log(expb.sum()) + np.max(gamma * b)) / gamma
    return grad, smooth_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step function with a low evaluate_sequence score.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # a safety margin for the final call

    # ------------------------------------------------------------------
    # 1)  Broad exploration – a handful of cheap families, all projected
    #     onto the simplex and evaluated with the fast proxy score.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    best_len = None

    # we try a few different lengths – increasing lengths give a finer
    # discretisation of the continuous optimum.
    lengths = [4000, 8000, 16000, 24000]      # 24000 stays affordable in 1000 s

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Exploring length {n}")

        # uniform baseline
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        sc = _raw_score(uniform)
        if sc < best_score:
            best_score, best_seq, best_len = sc, uniform.copy(), n
            print(f"   uniform score={best_score:.6f}")

        # arcsine (Beta(0.5,0.5))
        arcsine = _beta_distribution(n, 0.5, 0.5)
        sc = _raw_score(arcsine)
        if sc < best_score:
            best_score, best_seq, best_len = sc, arcsine.copy(), n
            print(f"   arcsine score={best_score:.6f}")

        # a few symmetric Beta grid points (these are cheap)
        for a in (0.8, 1.2, 1.6, 2.2):
            vec = _beta_distribution(n, a, a)
            sc = _raw_score(vec)
            if sc < best_score:
                best_score, best_seq, best_len = sc, vec.copy(), n
                print(f"   symBeta a={a:.1f} score={best_score:.6f}")

        # Dirichlet draws (concentration around 1)
        for _ in range(200):
            if time.time() >= deadline:
                break
            conc = 10 ** rng.uniform(-1, 0)               # 0.1 … 1.0
            vec = rng.gamma(conc, 1.0, size=n)
            vec /= vec.sum()
            sc = _raw_score(vec)
            if sc < best_score:
                best_score, best_seq, best_len = sc, vec.copy(), n
                print(f"   dirichlet conc={conc:.3f} score={best_score:.6f}")

        # Simple sinusoidal perturbation of uniform
        amp = rng.uniform(0.0, 0.25)                     # up‑to 25 % wiggle
        phase = rng.uniform(0, 2 * math.pi)
        sine = 1.0 + amp * np.cos(2 * math.pi * np.arange(n) / n + phase)
        vec = uniform * sine
        vec = _project_onto_simplex(vec)
        sc = _raw_score(vec)
        if sc < best_score:
            best_score, best_seq, best_len = sc, vec.copy(), n
            print(f"   sinusoid amp={amp:.3f} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # If nothing was found (should never happen) fall back to a uniform vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 4000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_len = n
        best_score = _raw_score(best_seq)
        print("Fallback to uniform.")

    print(f"Best raw score after exploration: {best_score:.6f} (len={best_len})")

    # ------------------------------------------------------------------
    # 2)  Additive Adam refinement on the exact max‑convolution objective.
    # ------------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = best_score

    # Adam hyper‑parameters
    alpha = 0.12            # base learning‑rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1
        grad, cur_max = _subgradient_max(a)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_onto_simplex(a)

        # occasional tiny jitter to avoid stagnation
        if it % 700 == 0 and rng.random() < 0.08:
            a += rng.normal(scale=3e-7, size=n)
            a = _project_onto_simplex(a)

        cur_val = _raw_score(a)
        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            print(f"[{time.time():.1f}] Adam it={it}  score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 3)  Exponentiated‑gradient (mirror descent) refinement.
    # ------------------------------------------------------------------
    a = best_a.copy()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _subgradient_max(a)

        # learning‑rate schedule – slowly decaying
        eta = 0.04 / math.sqrt(it)

        # multiplicative update, then re‑normalise (implicit projection)
        a = a * np.exp(-eta * grad)
        a = _project_onto_simplex(a)

        if it % 500 == 0 and rng.random() < 0.07:
            a += rng.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        cur_val = _raw_score(a)
        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            print(f"[{time.time():.1f}] Exp‑grad it={it}  score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 4)  (Optional) smooth‑max Adam refinement – a final polishing step.
    # ------------------------------------------------------------------
    a = best_a.copy()
    it = 0
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    gamma = 30.0                       # start with a relatively soft approximation
    while time.time() < deadline:
        it += 1
        grad, smooth_val = _softmax_gradient(a, gamma=gamma)
        # Adam on the smooth objective
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_onto_simplex(a)

        # slowly increase gamma – the smooth loss becomes a tighter
        # approximation of the true max as optimisation proceeds.
        gamma = min(200.0, gamma * 1.001)

        cur_val = _raw_score(a)
        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            print(f"[{time.time():.1f}] smooth‑Adam it={it}  score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 5)  Return the best sequence discovered.
    # ------------------------------------------------------------------
    final_seq = best_a
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Final evaluated score: {final_score:.6f} (len={len(final_seq)})")
    return final_seq.tolist()
