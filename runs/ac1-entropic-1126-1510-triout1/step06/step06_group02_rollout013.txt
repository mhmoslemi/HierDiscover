"""
A fast, deterministic mirror‑descent search for the step‑function that
minimises ``evaluate_sequence``.

The algorithm works on the simplex (non‑negative coefficients that sum to 1)
and exploits the scale‑invariance of the objective.
For a given length *n* it

1. builds a modest pool of analytic candidates (Beta, exponential,
   uniform, interpolation of the previous best, random Dirichlet draws);
2. keeps the few best candidates according to a cheap score
   ``2·n·max(a*a)`` (identical to the real objective when the sum is 1);
3. refines every kept candidate with a large number of mirror‑descent
   (exponentiated‑gradient) iterations using the exact sub‑gradient of the
   ``max`` term; the step size follows  ηₜ = η₀ / √t  and the iterate is
   renormalised to stay on the simplex.
4. The best vector seen over all lengths is returned.
The whole search respects the supplied time budget (≈ 1000 s) and prints
progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1} (sum = 1)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1.0 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1.0 - eps) ** (beta - 1.0)
    vals = a * b
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float = 1.0) -> np.ndarray:
    """Geometric‑like decreasing shape a_i ∝ exp(-lam·i/(n‑1))."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * i / max(1, n - 1))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _rescale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to length ``new_n`` and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        out = seq.copy()
    else:
        x_old = np.linspace(0.0, 1.0, old_n)
        x_new = np.linspace(0.0, 1.0, new_n)
        out = np.interp(x_new, x_old, seq)
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s == 0.0:
        out = np.full(new_n, 1.0 / new_n, dtype=np.float64)
    else:
        out = out / s
    return out


# ----------------------------------------------------------------------
def _max_conv_subgradient(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Returns (max_conv, subgradient) of   max_k (a*a)[k]   at ``a``.
    The subgradient is taken from a single active index (the argmax);
    this is a valid subgradient for the convex max‑function.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_val = conv.max()
    k = int(conv.argmax())                    # shift where the maximum occurs

    # gradient_i = 2 * a_{k-i}  whenever the index k-i lies in [0,n)
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)                     # inclusive
    idx = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - idx
    g[idx] = 2.0 * a[j]
    return max_val, g


# ----------------------------------------------------------------------
def _mirror_descent_optimize(
    a0: np.ndarray,
    deadline: float,
    eta0: float = 0.5,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the simplex.
    Returns the best vector seen and its cheap score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    max_val, _ = _max_conv_subgradient(a)
    best_score = 2.0 * n * max_val

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        max_val, g = _max_conv_subgradient(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # step‑size schedule ηₜ = η₀ / √t
        eta = eta0 / np.sqrt(it)
        # multiplicative update (mirror descent with KL‑Bregman)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
    return best_a, best_score

            a = 10 ** np.random.uniform(-0.5, 0.7)   # roughly 0.3 … 5
            b = 10 ** np.random.uniform(-0.5, 0.7)
            pool.append(_beta_distribution(n, a, b))

        #   d) exponential families
        for lam in (0.5, 1.0, 1.5, 3.0):
            pool.append(_exponential_distribution(n, lam))

        #   e) uniform
        pool.append(np.full(n, 1.0 / n, dtype=np.float64))

        #   f) random Dirichlet draws with a few concentrations
        for conc in (0.3, 0.7, 1.5, 3.0):
            a = np.random.dirichlet(np.full(n, conc), size=1)[0]
            pool.append(a)

        # --------------------------------------------------------------
        # 3️⃣  Keep the few best cheap candidates
        scored = [(cheap_score(vec), vec) for vec in pool]
        scored.sort(key=lambda x: x[0])
        starters = [vec.copy() for _, vec in scored[:5]]
        print(f"   top cheap scores (pre‑refine): {[s for s, _ in scored[:5]]}")

        # --------------------------------------------------------------
        # 4️⃣  Refine every starter with mirror descent
        per_length_deadline = time.time() + min(250.0, (deadline - time.time()) * 0.5)

        for sid, a0 in enumerate(starters):
            if time.time() >= per_length_deadline:
                break

            sub_deadline = time.time() + min(120.0, per_length_deadline - time.time())
            # run a long mirror‑descent optimisation
            best_vec, best_score = _mirror_descent_optimize(
                a0, deadline=sub_deadline, eta0=0.6, max_iters=500_000
            )

            # final evaluation with the true black‑box (identical to cheap score)
            final_score = cheap_score(best_vec)

            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = best_vec.copy()
                print(
                    f"*** Global improvement → n={n} starter={sid} "
                    f"score={overall_best_score:.6f}"
                )
                # early stop if we already beat the target
                if overall_best_score <= 1.503:
                    print("Target achieved – exiting early.")
                    return overall_best_seq.tolist()

        # --------------------------------------------------------------
        # 5️⃣  Global time check
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    if overall_best_seq is None:
        n_fallback = 4000
        overall_best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        overall_best_score = cheap_score(overall_best_seq)
        print("Fallback used.")

    print(
        f"Final best score: {overall_best_score:.6f} "
        f"(length = {len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
