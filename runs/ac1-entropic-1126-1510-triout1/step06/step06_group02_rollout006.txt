"""
search for a non‑negative step function (a list of heights) that
minimises the black‑box ``evaluate_sequence`` objective.

Algorithm
---------
1. Choose a few problem sizes (≈ 3000‑15000).  Larger n gives a better
   asymptotic constant but each evaluation is more expensive.
2. For each length n generate a **large diverse pool** of candidate
   probability vectors (sum = 1) by
   * Dirichlet draws with random concentration parameters,
   * systematic Beta(α,β) densities,
   * power‑law and sinusoidal shapes,
   * exponential decays,
   * the previous best construction (interpolated to the new length)
   * uniform / random vectors.
   The inexpensive “cheap” objective
   ``2·n·max(self‑convolution)`` (exact for sum = 1) is used to rank them.
3. Keep the best K (default 20) candidates and **refine** each with a
   projected Adam optimiser on the simplex:
   * the variables are logits `z` such that `a = softmax(z)`,
   * the sub‑gradient of the max‑convolution is computed exactly,
   * Adam updates are performed on `z` and the best iterate is stored.
4. After refinement evaluate the best local candidate with the real
   ``evaluate_sequence`` call.  The globally best vector seen so far is
   returned when the overall time budget (≈ 1000 s) expires or when a
   target score (≤ 1.503) is reached.
"""

import time
import heapq
import math
from typing import List, Tuple, Optional

import numpy as np

# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolutions -----------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
# Analytic families -----------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


def _exponential_distribution(n: int, lam: float = 1.0) -> np.ndarray:
    """Geometric‑like decreasing shape: a_i ∝ exp(-lam·i/(n‑1))."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * i / max(1, n - 1))
    vals /= vals.sum()
    return vals


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _sinusoid_distribution(
    n: int, freq: float = 1.0, phase: float = 0.0, amp: float = 0.8
) -> np.ndarray:
    """
    Positive sinusoidal shape ``1 + amp·sin(2π·freq·i/n + phase)``,
    normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 + amp * np.sin(2.0 * np.pi * freq * i / n + phase)
    vals = np.clip(vals, 0.0, None)
    if vals.sum() == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Cheap objective -------------------------------------------------------
def _cheap_score(a: np.ndarray) -> float:
    """
    Cheap evaluation for a probability vector (sum≈1):
    2·n·max(conv) – exactly the true objective when sum = 1.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Exact sub‑gradient of max‑convolution ---------------------------------
def _exact_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution w.r.t. a.
    Returns ∂max(conv)/∂a (vector of length n).
    """
    n = a.shape[0]
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    # convolution of reversed a with the weights w
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even shifts (i,i) counted twice
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad


# ----------------------------------------------------------------------
# Adam optimizer step ---------------------------------------------------
def _adam_step(
    z: np.ndarray,
    g: np.ndarray,
    m: np.ndarray,
    v: np.ndarray,
    t: int,
    lr: float = 0.2,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> Tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Adam update on the unconstrained logits z."""
    m = beta1 * m + (1.0 - beta1) * g
    v = beta2 * v + (1.0 - beta2) * (g * g)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z, m, v


# ----------------------------------------------------------------------
# Interpolation of a previous sequence ---------------------------------
def _interpolate_sequence(old: np.ndarray, new_n: int) -> np.ndarray:
    """Resize old vector to length new_n by linear interpolation and renorm."""
    if old.size == new_n:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old.size)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    new = _project_onto_simplex(new, s=1.0)
    return new


# ----------------------------------------------------------------------
# Local optimisation ----------------------------------------------------
def _local_optimize(
    a0: np.ndarray,
    max_iters: int = 2000,
    time_limit: Optional[float] = None,
) -> Tuple[np.ndarray, float]:
    """
    Refine a probability vector with projected Adam on the soft‑max
    parametrisation, using the exact sub‑gradient of the max‑convolution.
    Returns the best vector encountered and its cheap score.
    """
    n = a0.shape[0]
    # logits (avoid log(0))
    eps = 1e-15
    z = np.log(np.maximum(a0, eps))
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    best_a = a0.copy()
    best_score = _cheap_score(best_a)

    for it in range(1, max_iters + 1):
        if time_limit is not None and time.time() > time_limit:
            break

        # current probability vector
        a = np.exp(z)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # exact sub‑gradient of the max‑convolution
        grad_a = _exact_gradient(a, conv)          # ∂max(conv)/∂a
        grad_obj = 2.0 * n * grad_a                # ∂objective/∂a
        dot = np.dot(a, grad_obj)
        grad_z = a * (grad_obj - dot)              # chain rule through softmax

        # Adam step
        z, m, v = _adam_step(z, grad_z, m, v, it, lr=0.2)

        # occasional tiny jitter to escape plateaus
        if it % 1500 == 0 and np.random.rand() < 0.15:
            z += np.random.randn(n) * 1e-7

    return best_a, best_score

# Generation of a diverse candidate pool --------------------------------
def _generate_pool(
    n: int,
    base_seq: Optional[np.ndarray],
    pool_size: int = 3000,
) -> List[np.ndarray]:
    """
    Build a pool of candidate probability vectors for length n.
    The function is deliberately stochastic; the caller may ask for many
    candidates (pool_size) but only a few will be kept later.
    """
    candidates: List[np.ndarray] = []

    # 1) analytic families ------------------------------------------------
    # Beta grid
    alphas = np.logspace(-0.7, 0.7, num=7)   # ≈0.2 … 5.0
    betas = np.logspace(-0.7, 0.7, num=7)
    for a in alphas:
        for b in betas:
            candidates.append(_beta_distribution(n, a, b))

    # Power‑law family
    for p in [0.3, 0.6, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(_power_law_distribution(n, p))

    # Sinusoidal family
    for freq in [1, 2, 3, 4]:
        phase = np.random.rand() * 2 * np.pi
        amp = np.random.uniform(0.2, 0.9)
        candidates.append(_sinusoid_distribution(n, freq=freq, phase=phase, amp=amp))

    # Exponential decays
    for lam in [0.3, 0.7, 1.0, 1.5]:
        candidates.append(_exponential_distribution(n, lam))

    # Uniform and simple random
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))
    candidates.append(np.random.rand(n).astype(np.float64))
    candidates[-1][:] = np.maximum(candidates[-1], 1e-12)
    candidates[-1] /= candidates[-1].sum()

    # Previous construction, if any
    if base_seq is not None:
        candidates.append(_interpolate_sequence(base_seq, n))

    # 2) Random Dirichlet draws -------------------------------------------
    # mixture of concentration parameters
    for _ in range(pool_size - len(candidates)):
        # concentration α from log‑uniform in [1e‑3, 10]
        log_alpha = np.random.uniform(-3.0, 1.0)
        alpha = 10.0 ** log_alpha
        # draw Gamma per coordinate, then normalise
        draws = np.random.gamma(shape=alpha, scale=1.0, size=n)
        if draws.sum() == 0.0:
            draws = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            draws /= draws.sum()
        candidates.append(draws)

    # final safeguard – project everything onto simplex (numerical safety)
    candidates = [_project_onto_simplex(c, s=1.0) for c in candidates]

    return candidates

                    heapq.heapreplace(heap, (-score, cand))

        # Extract the K best candidates (sorted by increasing cheap score)
        best_candidates = [item[1] for item in heap]
        best_candidates.sort(key=_cheap_score)

        print(f"  top cheap scores (length {n}): {[_cheap_score(c) for c in best_candidates[:5]]}")

        # --------------------------------------------------------------
        # 3. Refine each top candidate locally
        per_candidate_time = (slice_deadline - time.time()) / (len(best_candidates) + 1e-9)

        for idx, cand in enumerate(best_candidates):
            # time limit for this local optimisation
            cand_deadline = min(slice_deadline, time.time() + per_candidate_time)
            if time.time() >= cand_deadline:
                break

            # Run Adam‑based refinement
            refined, refined_score = _local_optimize(
                cand, max_iters=2500, time_limit=cand_deadline
            )

            # Evaluate with the true black‑box (cheap_score already matches,
            # but we call the official function to respect clamping etc.)
            try:
                true_score = evaluate_sequence(refined.tolist())
            except Exception:
                true_score = refined_score

            if true_score < best_score:
                best_score = true_score
                best_seq = refined.copy()
                print(
                    f"*** New global best – n={n} starter={idx} "
                    f"score={best_score:.6f}"
                )
                if best_score <= 1.503:
                    print("Target achieved – terminating early.")
                    return best_seq.tolist()

            # Early exit if we already spent the allotted time for this length
            if time.time() >= slice_deadline:
                break

        # ------------------------------------------------------------------
        # global time check
        if time.time() >= deadline:
            break
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback – in the unlikely event that nothing was found
    if best_seq is None:
        n_fallback = 4000
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
        print("Fallback used – uniform sequence.")

    print(
        f"Final best score: {best_score:.6f} (length = {len(best_seq)})"
    )
    return best_seq.tolist()
