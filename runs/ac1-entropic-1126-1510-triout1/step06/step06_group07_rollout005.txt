"""
Propose a step‑function with a low evaluate_sequence() score.

Algorithm outline
-----------------
1.  Choose a problem size n (≈ 9000) and fix the ℓ¹‑norm S = sqrt(2·n).
    All candidates are projected onto the simplex {x≥0, Σx = S}.

2.  **Random seeding** – sample many Dirichlet vectors with a wide
    range of concentration parameters.  Keep the best (lowest max‑convolution).

3.  **Mini genetic optimisation** – maintain a population of ~30 candidates.
    In each generation create children as the (noisy) midpoint of two parents,
    project onto the simplex and keep the best individuals.

4.  **Projected sub‑gradient descent** – starting from the best individual,
    repeatedly move opposite to the sub‑gradient of the true max‑objective
    (the gradient of the active convolution entries) and project back onto the simplex.
    A simple back‑tracking line‑search supplies a suitable step‑size.

5.  **Intensive hill‑climbing** – random swaps, δ‑mass transfers,
    block‑averaging and small Gaussian perturbations are tried until the time budget
    is exhausted; improvements are always accepted.

6.  **Final polishing** – the best vector is combined with the uniform vector,
    sorted, interleaved and reversed; every variant is evaluated and the best
    one is returned.

All heavy evaluations use the fast proxy  max(conv(a,a))  which, because the
candidate always satisfies Σa = S, is exactly the score returned by
``evaluate_sequence`` (up to the clipping of values > 1000, which never occurs
here).  The function respects a user‑supplied time budget (default 1000 s) and
never uses closures or lambdas.

"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Fast evaluation: max(convolution)  (sum(a) = S ⇒ exact score)."""
    # convolution of a with itself (full mode)
    conv = np.convolve(a, a)
    return float(conv.max())


# ----------------------------------------------------------------------
def _hard_subgradient(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient of the hard max‑objective  max(conv)  at point a.
    Returns (g, max_conv) where g is a vector of length n.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_c = conv.max()
    mask = conv == max_c
    cnt = np.count_nonzero(mask)
    if cnt == 0:                     # defensive – should not happen
        w = np.zeros_like(conv)
    else:
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt            # uniform weight over active indices
    conv_wr = np.convolve(w, a[::-1])
    g = 2.0 * conv_wr[n - 1:n - 1 + n]
    return g, max_c

    conc_vals = [0.005, 0.02, 0.05, 0.1, 0.3, 0.7, 1.0, 2.0, 5.0]
