def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (normalised to sum 1)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a non‑negative sequence to a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth‑max surrogate
    L_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k).
    Returns (grad, current_max_conv).
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 10.0,
                   alpha_factor: float = 1.01,
                   alpha_max: float = 2000.0,
                   lr0: float = 0.4,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimiser for the smooth‑max surrogate.
    Returns the best vector seen and its corresponding peak value.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny random jitter to escape plateaux
        if t % 500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """One exact ℓ∞ sub‑gradient step."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Extended exact‑max refinement using sub‑gradient descent."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        # step size proportional to 1/√iteration (a bit larger than previous code)
        step = 0.08 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        # occasional jitter to avoid stagnation
        if it % 400 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, deadline: float):
    """Light simulated‑annealing polish."""
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_peak = _fft_self_convolve(cur_a).max()
    best_a = cur_a.copy()
    best_peak = cur_peak

    start = time.time()
    total_time = max(deadline - start, 1e-6)

    start_temp = max(cur_peak, 1e-3) * 0.5
    final_temp = start_temp * 1e-5
    if final_temp <= 0.0:
        final_temp = 1e-9
    cooling_rate = -math.log(final_temp / start_temp) / total_time

    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1
        src = np.random.randint(n)
        if cur_a[src] <= 1e-12:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue
        max_delta = min(cur_a[src], 0.05)
        delta = np.random.rand() * max_delta

        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_peak = _fft_self_convolve(a_new).max()
        delta_obj = new_peak - cur_peak

        elapsed = time.time() - start
        temperature = start_temp * math.exp(-cooling_rate * elapsed)

        accept = False
        if delta_obj < 0.0:
            accept = True
        else:
            prob = math.exp(-delta_obj / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_a = a_new
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_a = cur_a.copy()

        if iter_cnt % 20000 == 0 and np.random.rand() < 0.05:
            cur_a = best_a.copy()
            cur_peak = best_peak

    return best_a, best_peak


# ----------------------------------------------------------------------
def _lp_improve(a: np.ndarray,
                max_iters: int = 2,
                mix_t: float = 0.5,
                inner_deadline: float = None) -> np.ndarray:
    """
    LP‑based improvement (dual‐widening).  If cvxpy is unavailable
    the function returns ``a`` unchanged.
    """
    if not _cvxpy_available:
        return a

    import scipy.linalg
    n = a.shape[0]
    for _ in range(max_iters):
        if inner_deadline is not None and time.time() > inner_deadline:
            break
        M = _fft_self_convolve(a).max()
        # Build convolution matrix A (2n‑1)×n
        col = np.r_[a, np.zeros(n - 1)]
        row = np.r_[a[0], np.zeros(n - 1)]
        A = scipy.linalg.toeplitz(col, row)        # dense array

        b_var = cp.Variable(n, nonneg=True)
        constraints = [A @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception:  # pragma: no cover
            break
        if b_var.value is None:
            break
        b_vec = np.maximum(b_var.value, 0.0)
        s = b_vec.sum()
        if s <= 1e-12:
            break
        g = b_vec / s
        a = (1.0 - mix_t) * a + mix_t * g
        a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a low‑score step‑function within the supplied time budget.
    Returns a plain Python list ready for `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a little safety margin

    # ------------------------------------------------------------------
    # Choose a working length.  4000 is a good compromise between speed
    # and quality for the available budget.
    n = 4000

    # ------------------------------------------------------------------
    # 1.  Build a set of cheap starter vectors.
    starters = []

    # previous solution if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        prev = _project_onto_simplex(prev, s=1.0)
        if len(prev) != n:
            prev = _interpolate_to_length(prev, n)
        starters.append(prev)
    except Exception:
        pass

    # canonical shapes
    starters.append(_arcsine_distribution(n))
    starters.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Dirichlet draws with a handful of concentration parameters
    for conc in (0.2, 0.5, 1.0, 2.0):
        starters.append(np.random.dirichlet(np.full(n, conc)))

    # mixture arcsine / uniform
    for w in (0.2, 0.4, 0.6, 0.8):
        mixture = (1.0 - w) * _arcsine_distribution(n) + w * np.full(n, 1.0 / n)
        starters.append(mixture)

    # ------------------------------------------------------------------
    # Keep the best simple starter.
    best_seq = None
    best_score = float("inf")
    for vec in starters:
        vec = _project_onto_simplex(vec, s=1.0)
        peak = _fft_self_convolve(vec).max()
        fast_score = 2.0 * n * peak
        if fast_score < best_score:
            best_score = fast_score
            best_seq = vec.copy()
            print(f"[init] fast_score={fast_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Main optimisation loop – many short random restarts.
    # We leave ~30 s for the final simulated‑annealing phase.
    while time.time() < deadline - 30.0:
        time_left = deadline - time.time()
        # allocate a modest slice of the remaining budget to a single restart
        per_restart = min(12.0, time_left * 0.18)   # at most 12 s

        # random Dirichlet start (concentration parameter drawn log‑uniformly)
        conc = 10 ** np.random.uniform(-1.0, 0.7)    # roughly 0.1 … 5
        init = np.random.dirichlet(np.full(n, conc))

        # ---- Adam phase -------------------------------------------------
        adam_deadline = time.time() + per_restart * 0.55
        a_opt, _ = _adam_optimize(init, adam_deadline,
                                   init_alpha=10.0,
                                   lr0=0.4,
                                   alpha_factor=1.01)

        # ---- Exact ℓ∞ refinement ----------------------------------------
        subgrad_deadline = time.time() + per_restart * 0.45
        a_opt = _refine_max(a_opt, subgrad_deadline)

        # ---- Evaluate ----------------------------------------------------
        peak = _fft_self_convolve(a_opt).max()
        fast_score = 2.0 * n * peak
        if fast_score < best_score:
            best_score = fast_score
            best_seq = a_opt.copy()
            print(f"[restart] new best fast_score={fast_score:.6f} "
                  f"time_left={deadline - time.time():.1f}s")

    # ------------------------------------------------------------------
    # 3.  Simulated‑annealing polishing (uses the remaining budget).
    if best_seq is None:   # extremely unlikely, but keep a fallback
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    if time.time() < deadline - 0.5:
        print("[phase 4] simulated annealing polishing")
        best_seq, final_peak = _simulated_annealing(best_seq, deadline)
        best_score = 2.0 * n * final_peak
        print(f"[phase 4] SA fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Optional LP improvement (only if problem size is modest).
    if _cvxpy_available and n <= 3000 and time.time() < deadline - 0.5:
        print("[LP] running optional LP improvement")
        lp_deadline = time.time() + (deadline - time.time()) * 0.6
        best_seq = _lp_improve(best_seq, max_iters=2, mix_t=0.5,
                               inner_deadline=lp_deadline)
        best_score = 2.0 * n * _fft_self_convolve(best_seq).max()
        print(f"[LP] post‑LP fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final exact evaluation (the exact score that will be reported).
    try:
        true_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence score={true_score:.6f}")
        if true_score < best_score:
            best_score = true_score
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    print(f"Best score (approx.): {best_score:.6f} (length={n})")
    return best_seq.tolist()
