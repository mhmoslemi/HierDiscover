def _fft_self_convolve(a: np.ndarray) -> np.ndarray:

    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full auto‑convolution."""
    return _fft_self_convolve(a).max()

def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) (no normalisation assumptions)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)

    w[active] = 1.0

``evaluate_sequence`` objective.

The algorithm proceeds in three phases:

1. **Initialisation** – an analytically motivated arcsine distribution
   (known to give a low value) is created and, if a previous best
   construction ``height_sequence_1`` exists, it is resampled to the same
   length and used as a possible start.

2. **LP‑exchange refinement** – inspired by the classic iterative scheme
   from the problem statement.  For the current sequence `f` we compute
   its autocorrelation bound `M = max(f*f)`.  Then we solve the linear
   program

        maximise   Σ b_i
        subject to  conv(f, b) ≤ M    (component‑wise)
                    b_i ≥ 0

   The solution `b` carries as much ℓ₁‑mass as possible without raising the
   autocorrelation above `M`.  After normalising `b` to the same ℓ₁‑norm
   as `f` we mix it back into `f` with a decaying step size.  Repeating

3. **Hill‑climbing** – a cheap random‑transfer move (move a tiny amount of
   mass from one index to another) is applied while time permits; any

is returned as a plain list of non‑negative floats – it can be fed

# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full auto‑convolution."""
    return _fft_self_convolve(a).max()


def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) (no normalisation assumptions)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    size = 1 << ((a.size + w.size - 1) - 1).bit_length()
    A = np.fft.rfft(a[::-1], n=size)
    W = np.fft.rfft(w, n=size)
    conv_rev = np.fft.irfft(A * W, n=size)
    conv_rev = conv_rev[: a.size + w.size - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


def _solve_lp(f: np.ndarray, M: float):
    """
    Solve the linear program

        maximise   Σ b_i
        subject to  conv(f, b) ≤ M   (component‑wise)
                    b_i ≥ 0

    Returns the optimiser ``b`` as a NumPy array, or ``None`` on failure.
    """
    n = f.shape[0]
    b = cp.Variable(n, nonneg=True)
    constraints = [cp.conv(f, b) <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=5000)
        except Exception:
            return None
    if b.value is None:
        return None
    return np.asarray(b.value, dtype=np.float64)
