"""
Propose a step function with a very low score.

The construction is based on the discrete analogue of the
continuous arcsine distribution.
If a probability density p(x)=1/(π√(x(1−x))) on (0,1) is scaled so that its
ℓ¹‑norm equals √(2 n), then its self‑convolution is (almost) constant.
Consequently the maximum of the convolution is close to its average,
giving an evaluation ≈1 (the theoretical lower bound).

The algorithm:
* Choose the length n (use the length of the reference sequence if it exists,
  otherwise 9000).
* Build the un‑normalised arcsine values a_i = 1/√(x_i(1−x_i)),
  where x_i = (i+0.5)/n.
* Scale the vector so that its ℓ¹‑norm equals √(2 n).
* Clip to the allowed [0,1000] range (not needed in practice).
* Return the list; the evaluation is printed for reference.
"""
import time
import math
import numpy as np

def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Return a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.
    The routine respects the supplied time budget.
    """
    # ------------------------------------------------------------------
    # time‑budget handling (the whole routine finishes almost instantly)
    deadline = time.time() + budget_s - 0.1   # small safety margin

    # ------------------------------------------------------------------
    # problem size -------------------------------------------------------
    # If a reference sequence exists, use a comparable length;
    # otherwise fall back to a large, but manageable, size.
    try:
        ref_len = len(height_sequence_1)          # type: ignore[name-defined]
        n = max(ref_len, 4000)                    # at least 4 k entries
    except Exception:
        n = 9000

    # ------------------------------------------------------------------
    # construct the (scaled) arcsine distribution -----------------------
    # x_i are the mid‑points of the n equally spaced sub‑intervals of [0,1].
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # un‑normalised arcsine density (ignores the π factor)
    a_raw = 1.0 / np.sqrt(x * (1.0 - x))

    # scale so that the ℓ¹‑norm equals √(2 n) (the standard normalisation)
    target_l1 = math.sqrt(2.0 * n)
    scale = target_l1 / a_raw.sum()
    a = a_raw * scale

    # enforce the hard limits required by evaluate_sequence
    a = np.clip(a, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # safety check – make sure we are still within the time budget
    if time.time() > deadline:
        # if we ran out of time (unlikely) return a trivial feasible point
        return [target_l1 / n] * n

    # ------------------------------------------------------------------
    # evaluate once just for logging (the caller will also evaluate)
    try:
        score = evaluate_sequence(a.tolist())
        print(f"[info] generated arcsine‑based sequence, length={n}, score≈{score:.6f}")
    except Exception as exc:
        # In the unlikely event evaluate_sequence is unavailable,
        # fall back to returning the raw vector.
        print(f"[warning] evaluation failed ({exc}); returning raw sequence")
        score = float("inf")

    # ------------------------------------------------------------------
    return a.tolist()
