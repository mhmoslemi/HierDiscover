def _simulated_annealing(a, deadline):
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_score(a)
    cur = a.copy()
    cur_score = best_score
    # temperature schedule
    # start with temp such that probability of accepting small increase of e.g., 1e-4 is moderate.
    init_temp = 0.01 * best_score   # maybe
    final_temp = 1e-6 * best_score
    start_time = time.time()
    max_iter = 2000000
    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        # linear schedule based on elapsed fraction
        frac = (time.time() - start_time) / (deadline - start_time)
        T = init_temp * (1-frac) + final_temp * frac
        # propose move
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        # amount to transfer from i to j
        delta = np.random.rand() * 0.05 * cur[i]  # up to 5% of mass at i
        if delta <= 0:
            continue
        a_new = cur.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)
        dE = new_score - cur_score
        accept = False
        if dE < 0:
            accept = True
        else:
            if T > 0 and np.random.rand() < np.exp(-dE / T):
                accept = True
        if accept:
            cur = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a_new.copy()
                print(f"[{time.time():.1f}] SA iter={it} new best {best_score:.6f}")
    return best, best_score

"""
Fast anti‑concentration search for a non‑negative step function.

The objective to minimise is

    score = 2·n·max( a * a ) / (sum(a)²)

where `*` is the ordinary (linear) convolution and `a` must lie on the
probability simplex (a_i ≥ 0, Σa_i = 1).  The problem is equivalent to
finding a distribution on {0,…,n−1} whose sum‑distribution (the
self‑convolution) has the smallest possible peak.

The algorithm works in several phases:

1. **Broad initial sampling** – for a few lengths we generate many
   analytic families (uniform, arcsine, triangular, exponential) and a
   large collection of random Beta‑ and Dirichlet‑distributed vectors.
   Every candidate is normalised and its exact score is evaluated.
   The best few vectors (one per length) become seeds.

2. **Refinement by smoothed Adam descent** – we minimise a smooth
   log‑sum‑exp surrogate of the max‑convolution.  The gradient is a
   weighted convolution; we use Adam with Euclidean projection onto the
   simplex.  The smoothing parameter α is increased gradually (continuation).

3. **Exact max‑sub‑gradient descent** – after the continuation phase we
   switch to the true sub‑gradient of the max (active set) with momentum
   and a diminishing steps‑size (identical to the previous state‑of‑the‑art).

4. **Simulated‑annealing hill climbing** – random “transfer” moves that
   move a small amount of mass from one coordinate to another are
   accepted either if they improve the score or with a probability that
   depends on a temperature schedule.  This helps escape shallow local
   minima.

5. **Final mirror‑descent pass** – a few iterations of exponentiated‑gradient
   (mirror descent w.r.t. KL) using the exact sub‑gradient give a last polish.

All steps respect the 1000 s wall‑time budget; the code stops a little
early to keep a safety margin.  The best vector ever seen is returned as
a plain Python list of floats.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full (linear) self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors via FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact score for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families -----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_general(n: int, a: float, b: float) -> np.ndarray:
    """Beta(α,β) density on mid‑points, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # work in log‑space to avoid under/overflow
    logw = (a - 1.0) * np.log(x) + (b - 1.0) * np.log(1.0 - x)
    # numerical stabilisation
    logw -= np.max(logw)
    w = np.exp(logw)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_dirichlet(n: int, concentration: float) -> np.ndarray:
    """Draw a Dirichlet sample with a scalar concentration parameter."""
    alpha = np.full(n, concentration, dtype=np.float64)
    sample = np.random.gamma(alpha, 1.0)
    sample_sum = sample.sum()
    if sample_sum == 0.0:
        sample[:] = 1.0 / n
    else:
        sample /= sample_sum
    return sample


# ----------------------------------------------------------------------
def _initial_candidates(n: int) -> list[np.ndarray]:
    """Create a rich set of start vectors for a given length n."""
    cand = []

    # deterministic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))
    cand.append(_distribution_exponential(n, lam=0.003))
    cand.append(_distribution_exponential(n, lam=0.01))

    # random Beta shapes (including asymmetric)
    for _ in range(30):
        a = 10 ** np.random.uniform(-2, 1)   # α ∈ [0.01, 10]
        b = 10 ** np.random.uniform(-2, 1)   # β ∈ [0.01, 10]
        cand.append(_distribution_beta_general(n, a, b))

    # Dirichlet draws with various concentration levels
    for conc in (0.05, 0.2, 0.5, 1.0, 5.0):
        cand.append(_distribution_dirichlet(n, conc))

    # a few sparse random vectors (≈1 % non‑zeros)
    k = max(2, n // 100)
    for _ in range(5):
        vec = np.zeros(n, dtype=np.float64)
        idx = np.random.choice(n, k, replace=False)
        vec[idx] = np.random.rand(k)
        vec_sum = vec.sum()
        if vec_sum == 0.0:
            vec[:] = 1.0 / n
        else:
            vec /= vec_sum
        cand.append(vec)

    # also add the reversed version of every candidate (asymmetry matters)
    rev = [c[::-1] for c in cand]
    cand.extend(rev)

    return cand


# ----------------------------------------------------------------------
def _smooth_descent(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate of max(a*a).
    Returns (best_vector, best_score) for the current length.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # smoothing schedule – larger α gives a tighter approximation of max
    schedule = [
        (1.0, 800),
        (5.0, 800),
        (20.0, 800),
        (100.0, 800),
        (500.0, 800),
        (2000.0, 400),
        (5000.0, 200),
    ]

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score
            t += 1

            # self‑convolution (full)
            b = _fft_self_convolve(a)

            # stable softmax over b
            maxb = b.max()
            exp_terms = np.exp(alpha * (b - maxb))
            w = exp_terms / exp_terms.sum()               # soft‑max weights

            # gradient = 2·(a reversed) * w   (first n entries)
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam update + Euclidean projection onto simplex
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a, 1.0)

            # occasional exact evaluation
            if t % 200 == 0:
                cur_score = _max_convolution_score(a)
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_seq = a.copy()
    return best_seq, best_score


# ----------------------------------------------------------------------
def _exact_subgradient_refine(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Exact max‑subgradient descent (active‑set) with momentum.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)

    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps)

        w = np.zeros_like(b)
        w[active] = 1.0

        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # momentum / diminishing step
        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / np.sqrt(iteration)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] sub‑grad iter={iteration} new best {best_score:.6f}")

        # early stop if we already reach the target bound
        if best_score <= 1.503:
            break
        if time.time() + 0.5 > deadline:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Random transfer hill‑climber with a simple temperature schedule.
    """
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_score(a)
    cur = a.copy()
    cur_score = best_score

    # temperature schedule – start relatively hot and cool to almost zero
    init_T = 0.02 * best_score
    final_T = 1e-6 * best_score
    start_time = time.time()

    it = 0
    while time.time() < deadline:
        it += 1
        # linear cooling based on elapsed fraction
        frac = (time.time() - start_time) / (deadline - start_time)
        T = init_T * (1.0 - frac) + final_T * frac

        # propose a transfer
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        # move at most 5 % of mass currently at i
        delta = np.random.rand() * 0.05 * cur[i]
        if delta <= 0.0:
            continue

        a_new = cur.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = _max_convolution_score(a_new)
        dE = new_score - cur_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if T > 0.0 and np.random.rand() < math.exp(-dE / T):
                accept = True

        if accept:
            cur = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best = a_new.copy()
                print(f"[{time.time():.1f}] SA iter={it} new best {best_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
def _mirror_descent_refine(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Few iterations of KL‑mirror (exponentiated‑gradient) descent using the
    exact max‑subgradient.
    """
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_score(a)

    # work with logits (log‑probabilities)
    eps = 1e-15
    logits = np.log(a + eps)
    lr = 0.7
    it = 0

    while time.time() < deadline:
        it += 1

        # current vector from logits
        max_log = np.max(logits)
        exp_log = np.exp(logits - max_log)
        a = exp_log / exp_log.sum()

        # exact max‑subgradient
        b = _fft_self_convolve(a)
        cur_max = b.max()
        epsg = 1e-12 * (abs(cur_max) + 1.0)
        active = np.where(b >= cur_max - epsg)[0]
        w = np.zeros_like(b)
        w[active] = 1.0
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # mirror‑gradient step on logits
        logits -= lr * grad

        if it % 25 == 0:
            cur_score = _max_convolution_score(a)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] mirror iter={it} new best {best_score:.6f}")

            # optional decay of learning‑rate
            lr *= 0.99

    return best, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found (as a plain Python list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # 1.  Broad sampling over several lengths
    lengths = [800, 1200, 2000, 3000, 5000, 8000]
    global_best_seq = None
    global_best_score = float("inf")

    def consider(vec: np.ndarray) -> None:
        nonlocal global_best_seq, global_best_score
        sc = _max_convolution_score(vec)
        if sc < global_best_score:
            global_best_score = sc
            global_best_seq = vec.copy()
            print(f"[{time.time():.1f}] New global best score={sc:.6f} len={len(vec)}")

    for n in lengths:
        if time.time() >= deadline:
            break
        for cand in _initial_candidates(n):
            if time.time() >= deadline:
                break
            consider(cand)

    # ------------------------------------------------------------------
    # 2.  Include a previous construction if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            consider(prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    if global_best_seq is None:
        n = 1200
        global_best_seq = _distribution_uniform(n)
        global_best_score = _max_convolution_score(global_best_seq)
        print(f"[{time.time():.1f}] Fallback uniform score={global_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Smooth Adam refinement
    a = global_best_seq.copy()
    a, score = _smooth_descent(a, deadline)
    consider(a)

    # ------------------------------------------------------------------
    # 4.  Exact max‑subgradient refinement
    a, score = _exact_subgradient_refine(a, deadline)
    consider(a)

    # ------------------------------------------------------------------
    # 5.  Simulated‑annealing hill climbing
    a, score = _simulated_annealing(a, deadline)
    consider(a)

    # ------------------------------------------------------------------
    # 6.  Final mirror‑descent polish
    a, score = _mirror_descent_refine(a, deadline)
    consider(a)

    # ------------------------------------------------------------------
    print(f"Final best score: {global_best_score:.6f} (length={len(global_best_seq)})")
    return global_best_seq.tolist()
