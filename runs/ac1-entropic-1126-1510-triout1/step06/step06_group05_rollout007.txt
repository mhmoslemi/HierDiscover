def propose_candidate(seed: int=42, budget_s: int=1000):
    np.random.seed(seed)
    start=time.time()
    deadline=start + budget_s -0.5

    # Stage 1: quick search for best shape across lengths and parameters
    lengths=[800,1200,2000,3000,5000]
    p_grid=np.linspace(0.35,0.7,15)
    w_grid=np.linspace(0.0,0.9,10)
    best_fast= float('inf')
    best_vec=None
    for n in lengths:
        if time.time()>=deadline: break
        # uniform baseline
        uniform = np.full(n, 1.0/n)
        fast = _fast_score(uniform)
        if fast < best_fast:
            best_fast, best_vec = fast, uniform.copy()
        # arcsine
        arcs = _arcsine_distribution(n)
        fast=_fast_score(arcs)
        if fast < best_fast:
            best_fast, best_vec = fast, arcs.copy()
        # powerlaw+mix loops
        for p in p_grid:
            base = _powerlaw_boundary_distribution(n,p)
            for w in w_grid:
                vec = (1-w)*base + w*uniform
                vec = _project_onto_simplex(vec)
                fast = _fast_score(vec)
                if fast<best_fast:
                    best_fast,best_vec=fast,vec.copy()
            # also test base alone (w=0) already covered
        # Dirichlet random draws
        for conc in (0.1,0.3,0.5,2.0):
            for _ in range(2):
                vec=np.random.dirichlet(np.full(n, conc))
                fast=_fast_score(vec)
                if fast<best_fast:
                    best_fast,best_vec=fast,vec.copy()

    print("[search] best init fast_score",best_fast, "len",len(best_vec))
    # Stage 2: heavy refinement using Adam+subgrad+SA until deadline
    # allocate time proportion: half to Adam, quarter to subgrad, quarter to SA? But we can adapt
    remaining = deadline - time.time()
    if remaining>5:
        # allocate up to 0.6*remaining to adam

        a_opt, _ = _adam_optimize(best_vec, adam_deadline)

Step‑function optimizer for the “evaluate_sequence” challenge.

The algorithm proceeds in three phases:

1. **Fast parametric sweep** – generate a few hundred candidate probability

   decreasing step‑size.  The whole refinement runs until the allotted time
   expires, typically performing hundreds of thousands of FFT‑based steps.

3. **Light stochastic polish** – a short simulated‑annealing pass (tiny
   temperature schedule) is applied to the refined vector to escape any
   remaining local plateau.

The routine respects the simplex (Σ aᵢ = 1, aᵢ ≥ 0) and works completely
with NumPy/FFT, needing no external data.  The final list is returned ready
for `evaluate_sequence`.
"""

import math
import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine (Beta(½,½)) distribution of length `n`."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _powerlaw_boundary_distribution(n: int, p: float) -> np.ndarray:
    """Distribution ∝ ((i+½)*(n‑i‑½))^{‑p}."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5) + 1e-12) ** (-p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Cheap proxy: 2·n·max(self‑convolution) (sum is 1 after projection)."""
    n = a.shape[0]
    return 2.0 * n * _fft_self_convolve(a).max()


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """Gradient of the smooth‑max surrogate
    Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)."""
    b = _fft_self_convolve(a)
    cur_max = b.max()
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.02,
                   alpha_max: float = 2000.0,
                   lr0: float = 0.30,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """Adam optimiser for the smooth‑max surrogate."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / math.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-16:
            best_peak = cur_max
            best_a = a.copy()

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny jitter to escape plateaus
        if t % 800 == 0 and np.random.rand() < 0.08:
            a += np.random.randn(n) * 6e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """Exact ℓ∞ sub‑gradient step for max‑convolution."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Short exact‑max refinement using sub‑gradient descent."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.025 / math.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-16:
            best_peak = cur_max
            best_a = a.copy()
        if it % 500 == 0 and np.random.rand() < 0.09:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, deadline: float):
    """Very light simulated‑annealing polish (always accept improvements)."""
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_peak = _fft_self_convolve(cur_a).max()
    best_a = cur_a.copy()
    best_peak = cur_peak

    start = time.time()
    total_time = max(deadline - start, 1e-6)

    start_temp = max(cur_peak, 1e-3) * 0.5
    final_temp = start_temp * 1e-5
    if final_temp <= 0.0:
        final_temp = 1e-9
    cooling_rate = -math.log(final_temp / start_temp) / total_time

    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1
        src = np.random.randint(n)
        if cur_a[src] <= 1e-12:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue
        max_delta = min(cur_a[src], 0.07)
        delta = np.random.rand() * max_delta

        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_peak = _fft_self_convolve(a_new).max()
        delta_obj = new_peak - cur_peak

        elapsed = time.time() - start
        temperature = start_temp * math.exp(-cooling_rate * elapsed)

        accept = False
        if delta_obj < 0.0:
            accept = True
        else:
            prob = math.exp(-delta_obj / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_a = a_new
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_a = cur_a.copy()

        if iter_cnt % 25000 == 0 and np.random.rand() < 0.07:
            cur_a = best_a.copy()
            cur_peak = best_peak

    return best_a, best_peak


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Entry point used by the evaluation harness.
    Returns a list of non‑negative heights (probability vector, sum≈1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # tiny safety margin

    # ------------------------------------------------------------------
    # Phase 1 – quick parametric sweep over a handful of lengths.
    lengths = [800, 1200, 2000, 3000, 5000]
    p_grid = np.linspace(0.35, 0.70, 15)        # exponent for power‑law family
    w_grid = np.linspace(0.0, 0.9, 10)          # mixing weight with uniform
    best_vec = None
    best_fast = float("inf")

    print("[phase 1] parametric sweep")
    for n in lengths:
        if time.time() >= deadline:
            break

        # uniform baseline (fast_score = 2.0)
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        fast = _fast_score(uniform)
        if fast < best_fast:
            best_fast, best_vec = fast, uniform.copy()
            print(f"[init] n={n:4d} uniform   fast={fast:.6f}")

        # arcsine (Beta(½,½))
        arcs = _arcsine_distribution(n)
        fast = _fast_score(arcs)
        if fast < best_fast:
            best_fast, best_vec = fast, arcs.copy()
            print(f"[init] n={n:4d} arcsine   fast={fast:.6f}")

        # power‑law + uniform mixtures
        for p in p_grid:
            base = _powerlaw_boundary_distribution(n, p)
            for w in w_grid:
                cand = (1.0 - w) * base + w * uniform
                cand = _project_onto_simplex(cand, s=1.0)
                fast = _fast_score(cand)
                if fast < best_fast:
                    best_fast, best_vec = fast, cand.copy()
                    print(f"[init] n={n:4d} p={p:.3f} w={w:.2f} fast={fast:.6f}")

        # a few Dirichlet draws (different concentration)
        for conc in (0.1, 0.3, 0.5, 2.0):
            for _ in range(2):
                cand = np.random.dirichlet(np.full(n, conc))
                fast = _fast_score(cand)
                if fast < best_fast:
                    best_fast, best_vec = fast, cand.copy()
                    print(f"[init] n={n:4d} dirichlet α={conc:.1f} fast={fast:.6f}")

    if best_vec is None:
        # fallback – uniform of length 1000
        n = 1000
        best_vec = np.full(n, 1.0 / n)
        best_fast = _fast_score(best_vec)
        print("[fallback] using uniform 1000")

    print(f"[phase 1] best initial fast_score={best_fast:.6f} (len={len(best_vec)})")

    # ------------------------------------------------------------------
    # Phase 2 – heavy continuous refinement (Adam + exact sub‑gradient).
    time_left = deadline - time.time()
    if time_left > 5.0:
        # allocate the bulk of the remaining budget to Adam, then sub‑grad.
        adam_deadline = time.time() + time_left * 0.55
        print(f"[phase 2] Adam optimisation – will run until {adam_deadline - start_time:.1f}s")
        a_opt, _ = _adam_optimize(best_vec, adam_deadline)

        # sub‑gradient refinement
        sub_deadline = time.time() + time_left * 0.30
        print(f"[phase 2] Sub‑gradient descent – will run until {sub_deadline - start_time:.1f}s")
        a_opt = _refine_max(a_opt, sub_deadline)

        best_vec = a_opt
        best_fast = _fast_score(best_vec)
        print(f"[phase 2] post‑refinement fast_score={best_fast:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – very light simulated‑annealing polish (if time permits).
    time_left = deadline - time.time()
    if time_left > 1.0:
        print(f"[phase 3] Simulated annealing polish – time left {time_left:.1f}s")
        a_opt, peak = _simulated_annealing(best_vec, deadline)
        best_vec = a_opt
        best_fast = 2.0 * len(best_vec) * peak
        print(f"[phase 3] final fast_score={best_fast:.6f}")

    # ------------------------------------------------------------------
    # Final exact evaluation (the only place where evaluate_sequence is called).
    try:
        true_score = evaluate_sequence(best_vec.tolist())
        print(f"[final] evaluate_sequence score={true_score:.6f}")
        # keep the true score for reporting, but we return the list anyway
    except Exception as exc:  # pragma: no cover
        true_score = None
        print("[final] evaluate_sequence raised:", exc)

    print(f"Best score (approx.): {best_fast:.6f} (length={len(best_vec)})")
    return best_vec.tolist()
