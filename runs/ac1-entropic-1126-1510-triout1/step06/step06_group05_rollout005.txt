"""
Improved multi‑start search for a low‑score step function.

The algorithm works in three phases:

1. **Broad random start generation** – a diverse pool of probability vectors
   (arcsine, power‑law, Dirichlet draws, Beta families, random mixtures) is
   created for a single length *n*.  The cheap proxy `2·n·max(conv)` is used
   to keep the top few candidates.

2. **Quick Adam refinement** – each of the selected candidates receives a
   short Adam optimisation of the smooth‑max surrogate.  The best refined
   vector is kept for the final stage.

3. **Deep optimisation** – a long Adam run followed by exact ℓ∞
   sub‑gradient descent, and a final light simulated‑annealing polish.
   The final answer is returned as a plain Python list ready for
   `evaluate_sequence`.

All operations respect the simplex (Σ a_i = 1, a_i ≥ 0) and stay within the
time budget (default 1000 s).  Optional LP‑based improvement is disabled for
large dimensions to keep memory low.
"""

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex {x ≥ 0, Σ x = s}.
    """
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution ``a*a`` using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(½,½)) of length ``n``."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) on a uniform grid, normalized to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_vals = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    vals[~np.isfinite(vals)] = 0.0
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


def _powerlaw_boundary_distribution(n: int, p: float) -> np.ndarray:
    """Distribution ∝ ((i+½)*(n‑i‑½))^{‑p}."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5) + 1e-12) ** (-p)
    vals /= vals.sum()
    return vals

def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth‑max surrogate
        Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k) .
    Returns ``grad`` (same shape as ``a``) and the current (unsmoothed) maximum.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # subtract cur_max for numerical stability
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()                 # weights ∝ exp(α·b_k)
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)            # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.02,
                   alpha_max: float = 8000.0,
                   lr0: float = 0.25,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best point seen (projected onto the simplex) and its unsmoothed peak.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)          # simple decay
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional jitter to escape flat plateaus
        if t % 400 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak

def _max_subgradient_step(a: np.ndarray):
    """
    One exact ℓ∞ sub‑gradient step.
    Returns ``grad`` (same shape as ``a``) and the current maximal convolution entry.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # tolerance for active set
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Short exact‑max refinement using sub‑gradient descent.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.02 / np.sqrt(it)          # diminishing step size
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        if it % 400 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, deadline: float):
    """
    Light simulated‑annealing polish.
    Returns the best point seen and its peak value.
    """
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_peak = _fft_self_convolve(cur_a).max()
    best_a = cur_a.copy()
    best_peak = cur_peak

    start = time.time()
    total_time = max(deadline - start, 1e-6)

    start_temp = max(cur_peak, 1e-3) * 0.5
    final_temp = start_temp * 1e-5
    if final_temp <= 0.0:
        final_temp = 1e-9
    cooling_rate = -math.log(final_temp / start_temp) / total_time

    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1
        src = np.random.randint(n)
        if cur_a[src] <= 1e-12:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue
        max_delta = min(cur_a[src], 0.05)
        delta = np.random.rand() * max_delta

        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_peak = _fft_self_convolve(a_new).max()
        delta_obj = new_peak - cur_peak

        elapsed = time.time() - start
        temperature = start_temp * math.exp(-cooling_rate * elapsed)

        accept = False
        if delta_obj < 0.0:
            accept = True
        else:
            prob = math.exp(-delta_obj / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_a = a_new
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_a = cur_a.copy()

        if iter_cnt % 20000 == 0 and np.random.rand() < 0.05:
            cur_a = best_a.copy()
            cur_peak = best_peak

    return best_a, best_peak


# ----------------------------------------------------------------------
def _lp_improve(a: np.ndarray,
                max_iters: int = 2,
                mix_t: float = 0.5,
                inner_deadline: float = None) -> np.ndarray:
    """
    Optional LP‑based improvement (only for modest dimensions).
    Maximises ‖b‖₁ under (a∗b) ≤ max(a∗a) and mixes the result back.
    """
    if not _cvxpy_available:
        return a
    n = a.shape[0]
    for _ in range(max_iters):
        if inner_deadline is not None and time.time() > inner_deadline:
            break
        M = _fft_self_convolve(a).max()
        # build convolution matrix A ∈ ℝ^{(2n‑1)×n}
        col = np.r_[a, np.zeros(n - 1)]
        row = np.r_[a[0], np.zeros(n - 1)]
        A = np.linalg.toeplitz(col, row)          # dense but cheap for n≲3000
        b_var = cp.Variable(n, nonneg=True)
        constraints = [A @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception:                         # pragma: no cover
            break
        if b_var.value is None:
            break
        b_vec = np.maximum(b_var.value, 0.0)
        s = b_vec.sum()
        if s <= 1e-12:
            break
        g = b_vec / s
        a = (1.0 - mix_t) * a + mix_t * g
        a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a low‑score step‑function within ``budget_s`` seconds.
    Returns a plain Python list of non‑negative heights ready for
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5            # safety margin

    # --------------------------------------------------------------
    # Choose a working length.  Larger n reduces discretisation error;
    # we keep n modest enough to stay fast.
    n = 5000
    if deadline - time.time() < 200:
        n = 4000                                          # tighter budget → smaller n

    print(f"[setup] chosen length n={n}")

    # --------------------------------------------------------------
    # Phase 1 – generate a diverse pool of start vectors
    starters = {}

    # deterministic families
    starters["arcsine"] = _arcsine_distribution(n)
    starters["uniform"] = np.full(n, 1.0 / n, dtype=np.float64)

    # Dirichlet draws with several concentrations
    for conc in (0.15, 0.3, 0.5, 0.8, 1.0, 2.0):
        starters[f"dirichlet_{conc:.2f}"] = np.random.dirichlet(np.full(n, conc))

    # Symmetric Beta family
    for a in (0.3, 0.4, 0.45, 0.55, 0.6, 0.7):
        starters[f"beta_{a:.2f}"] = _beta_distribution(n, a, a)

    # Power‑law boundary family
    for p in (0.35, 0.40, 0.45, 0.50, 0.55, 0.60):
        starters[f"power_{p:.2f}"] = _powerlaw_boundary_distribution(n, p)

    # Random mixtures of arcsine & Dirichlet
    for i in range(8):
        w = np.random.rand()
        arc = _arcsine_distribution(n)
        rand = np.random.dirichlet(np.full(n, 1.0))
        starters[f"mix_{i}"] = w * arc + (1 - w) * rand

    # --------------------------------------------------------------
    # Fast proxy evaluation: keep the best K candidates
    K = 6
    best_candidates = []
    best_fast = float("inf")
    for tag, vec in starters.items():
        vec = _project_onto_simplex(vec, s=1.0)
        peak = _fft_self_convolve(vec).max()
        fast_score = 2.0 * n * peak
        if fast_score < best_fast:
            best_fast = fast_score
            best_candidates = [(fast_score, vec.copy())]
        elif fast_score < best_fast * 1.02:           # keep a few close ones
            best_candidates.append((fast_score, vec.copy()))
    # keep only the top K
    best_candidates.sort(key=lambda x: x[0])
    best_candidates = best_candidates[:K]
    print(f"[phase 1] fast best ≈ {best_fast:.6f} (kept {len(best_candidates)} candidates)")

    # --------------------------------------------------------------
    # Phase 2 – quick Adam refinements (short time per candidate)
    time_left = deadline - time.time()
    if time_left > 5.0:
        per_cand_deadline = time.time() + time_left * 0.20 / max(len(best_candidates), 1)
        refined = []
        for score, vec in best_candidates:
            if time.time() >= deadline:
                break
            sub_dead = time.time() + (deadline - time.time()) * 0.15
            a_opt, _ = _adam_optimize(vec, sub_dead,
                                      init_alpha=0.5,
                                      alpha_max=4000.0,
                                      lr0=0.20)
            fast = 2.0 * n * _fft_self_convolve(a_opt).max()
            refined.append((fast, a_opt))
        refined.sort(key=lambda x: x[0])
        best_seq = refined[0][1]
        best_fast = refined[0][0]
        print(f"[phase 2] after quick Adam fast_score≈{best_fast:.6f}")
    else:
        best_seq = best_candidates[0][1]

    # --------------------------------------------------------------
    # Phase 3 – deep optimisation on the best candidate
    time_left = deadline - time.time()
    if time_left > 15.0:
        # allocate roughly 55 % for a longer Adam run
        adam_dead = time.time() + time_left * 0.55
        best_a, _ = _adam_optimize(best_seq, adam_dead,
                                    init_alpha=1.0,
                                    alpha_max=8000.0,
                                    lr0=0.25)
        # ℓ∞ sub‑gradient descent (30 % of remaining time)
        sub_dead = time.time() + (deadline - time.time()) * 0.30
        best_a = _refine_max(best_a, sub_dead)
        # optional LP improvement for moderate n
        if n <= 3000 and _cvxpy_available:
            lp_dead = time.time() + (deadline - time.time()) * 0.50
            best_a = _lp_improve(best_a, max_iters=2,
                                 mix_t=0.5,
                                 inner_deadline=lp_dead)
        # final simulated annealing (use whatever time remains)
        if time.time() < deadline:
            best_a, final_peak = _simulated_annealing(best_a, deadline)
        else:
            final_peak = _fft_self_convolve(best_a).max()
        best_seq = best_a
        best_fast = 2.0 * n * final_peak
        print(f"[phase 3] deep optimisation fast_score≈{best_fast:.6f}")
    else:
        # very little time left – evaluate current candidate directly
        best_fast = 2.0 * n * _fft_self_convolve(best_seq).max()

    # --------------------------------------------------------------
    # Final exact evaluation (real objective)
    try:
        true_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence score={true_score:.6f}")
        if true_score < best_fast:
            best_fast = true_score
    except Exception as exc:                               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    print(f"Best (approx) score: {best_fast:.6f}  – length {n}")
    return best_seq.tolist()
