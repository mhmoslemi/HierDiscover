"""
Search for a non‑negative step function that minimizes
evaluate_sequence(sequence).  The objective is

    2·n·max(conv(a,a)) / (sum(a)^2)

and is scale‑invariant, so we keep the vector on the probability simplex
(sum = 1).

The algorithm proceeds in several timed phases:

* Phase 0 – Broad sampling of many analytic families (uniform, arcsine,
  beta, exponential, power‑law, mixtures, and the previous best
  construction) together with a cheap two‑level block search.

* Phase 1 – Random piecewise‑constant families (2–5 blocks) with optional
  smoothing.  The best candidate so far is retained.

* Phase 2 – Adam optimisation of a smooth‑max surrogate of the ℓ∞
  convolution (log‑sum‑exp), with a continuation on the smoothness
  parameter.

* Phase 3 – Exact sub‑gradient descent on the true max‑convolution
  objective, followed by a directed greedy mass‑transfer local search
  that always attacks the current peak.

All vectors are projected onto the simplex after every update.
FFT‑based convolution is used throughout for speed.  The routine respects
the overall time budget (default 1000 s) and returns the best list of
heights found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # Numerically stable soft‑max weights
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.01,
                   alpha_max: float = 5000.0,
                   lr0: float = 0.3,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best probability vector encountered.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule (slow decay)
        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        # continuation on the smoothness parameter
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny jitter
        if t % 400 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    One sub‑gradient step for the exact ℓ∞ convolution objective.
    Uses an equal weight on all indices attaining the current maximum.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Sub‑gradient descent on the exact max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.02 / np.sqrt(it)   # diminishing step
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        # occasional jitter to avoid stagnation
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


# ----------------------------------------------------------------------
def _directed_local_search(a0: np.ndarray, deadline: float):
    """
    Greedy mass‑transfer local search that always attacks the current peak.
    Accepts only strictly improving moves (new max at least 1e‑15 lower).
    """
    a = a0.copy()
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        # pick a shift that is (almost) maximal
        max_idx = np.random.choice(np.where(b >= cur_max - 1e-12)[0])
        # indices that contribute to that shift
        i_low = max(0, max_idx - (n - 1))
        i_high = min(max_idx, n - 1)
        i_vals = np.arange(i_low, i_high + 1, dtype=int)
        prod = a[i_vals] * a[max_idx - i_vals]
        if prod.sum() == 0.0:
            src = np.random.randint(n)
        else:
            prob = prod / prod.sum()
            src = np.random.choice(i_vals, p=prob)

        dst = np.random.randint(n)
        if dst == src:
            continue

        # move a modest (sometimes larger) fraction of source mass
        max_delta = min(a[src], 0.10)          # up to 10 % of source
        delta = np.random.rand() * max_delta

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        b_new = _fft_self_convolve(a_new)
        new_max = b_new.max()

        if new_max < cur_max - 1e-15:         # accept only if improvement
            a = a_new
            b = b_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()

        # occasional random jitter to explore neighbourhood
        if np.random.rand() < 0.005:
            a += np.random.randn(n) * 1e-7
            a = _project_onto_simplex(a, s=1.0)
            b = _fft_self_convolve(a)
            cur_max = b.max()

    return best_a


# ----------------------------------------------------------------------
def _two_level_family(n: int, deadline: float):
    """
    Exhaustive search over a simple two‑level family:
        a_i = p for i < L   (front block)
        a_i = q otherwise   (back block)
    with ratio r = p/q ≥ 1.
    Returns the best vector and its fast score.
    """
    best_seq = None
    best_score = float('inf')
    step_L = max(1, n // 100)          # ≈1 % of the length
    r_grid = np.logspace(0.0, 2.3, num=25)   # from 1 to ≈200

    for L in range(step_L, n, step_L):
        if time.time() > deadline:
            break
        denom_base = n - L
        for r in r_grid:
            denom = L * r + denom_base
            q = 1.0 / denom
            p = r * q
            seq = np.empty(n, dtype=np.float64)
            seq[:L] = p
            seq[L:] = q
            peak = _fft_self_convolve(seq).max()
            score = 2.0 * n * peak
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Main entry point.  Returns a Python list of non‑negative heights that
    (hopefully) minimizes evaluate_sequence().
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # --------------------------------------------------------------
    # Retrieve a previously found construction if present.
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_seq = _project_onto_simplex(base_seq, s=1.0)
        base_n = len(base_seq)
    except Exception:
        base_seq = None
        base_n = 2000

    # --------------------------------------------------------------
    # Lengths to explore – a mix around the previous best.
    lengths = [max(500, int(base_n * f))
               for f in (0.6, 0.8, 1.0, 1.2, 1.4, 1.8, 2.0) if int(base_n * f) >= 500]
    # a few longer candidates (up to ~15000) if time permits
    for mul in (2.5, 3.0, 4.0):
        n = int(base_n * mul)
        if 500 <= n <= 15000:
            lengths.append(n)
    lengths = sorted(set(lengths))

    overall_best_seq = None
    overall_best_score = float('inf')

    # --------------------------------------------------------------
    # Phase 0 – broad analytic families + cheap two‑level search
    print("[phase 0] broad families & two‑level")
    for n in lengths:
        if time.time() >= deadline:
            break

        families = {}

        # arcsine (inverse‑square‑root)
        idx = np.arange(n, dtype=np.float64)
        arcs = 1.0 / np.sqrt((idx + 0.5) * (n - idx - 0.5))
        families["arcsine"] = arcs / arcs.sum()

        # uniform
        families["uniform"] = np.full(n, 1.0 / n)

        # random draws
        for i in range(3):
            r = np.random.rand(n)
            families[f"rand{i}"] = r / r.sum()

        # symmetric beta densities
        for a in (0.25, 0.35, 0.45, 0.55, 0.65, 0.75):
            x = (idx + 0.5) / n
            log_vals = (a - 1.0) * np.log(x) + (a - 1.0) * np.log(1.0 - x)
            vals = np.exp(log_vals)
            vals[~np.isfinite(vals)] = 0.0
            if vals.sum() == 0.0:
                vals = np.full(n, 1.0 / n)
            else:
                vals = vals / vals.sum()
            families[f"beta_{a:.2f}"] = vals

        # exponential decays
        for lam in (0.0005, 0.001, 0.002, 0.005):
            e = np.exp(-lam * idx)
            families[f"exp_{lam:.4f}"] = e / e.sum()

        # power‑law tails
        for p in (0.3, 0.5, 0.8, 1.2):
            pw = 1.0 / ((idx + 1) ** p)
            families[f"pow_{p:.2f}"] = pw / pw.sum()

        # simple log shape
        logv = np.log(idx + 2.0)
        families["log"] = logv / logv.sum()

        # mixture of arcsine and uniform
        for w in (0.2, 0.4, 0.6):
            mix = (1.0 - w) * families["arcsine"] + w * families["uniform"]
            families[f"mix_{w:.1f}"] = mix / mix.sum()

        # interpolate the previous best if available
        if base_seq is not None:
            x_old = np.linspace(0.0, 1.0, base_n)
            x_new = np.linspace(0.0, 1.0, n)
            interp = np.interp(x_new, x_old, base_seq)
            interp = np.maximum(0.0, interp)
            interp /= interp.sum()
            families["prev"] = interp

        # evaluate all families
        for tag, vec in families.items():
            vec = _project_onto_simplex(vec, s=1.0)
            peak = _fft_self_convolve(vec).max()
            fast_score = 2.0 * n * peak
            if fast_score < overall_best_score:
                overall_best_score = fast_score
                overall_best_seq = vec.copy()
                print(f"[init] n={n:4d} {tag:<12s} fast={fast_score:.6f}")

        # cheap two‑level parametric search
        seq2, score2 = _two_level_family(n, deadline)
        if seq2 is not None and score2 < overall_best_score:
            overall_best_score = score2
            overall_best_seq = seq2.copy()
            print(f"[two‑level] n={n:4d} fast={score2:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – random piecewise‑constant families (2–5 blocks)
    time_left = deadline - time.time()
    if time_left > 20.0:
        phase1_deadline = time.time() + time_left * 0.2
        print("[phase 1] random piecewise‑constant search")
        while time.time() < phase1_deadline:
            n = np.random.choice(lengths)
            B = np.random.randint(2, 6)            # number of blocks 2–5
            # choose break points uniformly
            breaks = np.sort(np.random.choice(np.arange(1, n), size=B - 1, replace=False))
            heights = np.exp(np.random.uniform(-2.0, 2.0, size=B))
            seq = np.empty(n, dtype=np.float64)
            start = 0
            for b_pt, h in zip(np.append(breaks, n), heights):
                seq[start:b_pt] = h
                start = b_pt
            seq = seq / seq.sum()
            # occasional tiny smoothing
            if np.random.rand() < 0.3:
                ker = np.exp(-np.linspace(-1.5, 1.5, 7) ** 2)
                ker /= ker.sum()
                seq = np.convolve(seq, ker, mode="same")
                seq = np.maximum(seq, 0.0)
                seq = seq / seq.sum()
            seq = _project_onto_simplex(seq, s=1.0)
            peak = _fft_self_convolve(seq).max()
            fast_score = 2.0 * n * peak
            if fast_score < overall_best_score:
                overall_best_score = fast_score
                overall_best_seq = seq.copy()
                print(f"[piecewise] n={n:4d} B={B} fast={fast_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam optimisation of the smooth‑max surrogate
    time_left = deadline - time.time()
    if time_left > 10.0 and overall_best_seq is not None:
        n = len(overall_best_seq)
        print(f"[phase 2] Adam optimisation (n={n}) – time left {time_left:.1f}s")
        adam_deadline = time.time() + time_left * 0.55
        a_opt, _ = _adam_optimize(overall_best_seq, adam_deadline)
        overall_best_seq = a_opt.copy()
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print(f"[phase 2] after Adam fast={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – exact sub‑gradient descent + directed greedy local search
    time_left = deadline - time.time()
    if time_left > 8.0 and overall_best_seq is not None:
        n = len(overall_best_seq)
        print(f"[phase 3] sub‑gradient & local search (n={n}) – time left {time_left:.1f}s")
        sub_deadline = time.time() + time_left * 0.45
        a_refined = _refine_max(overall_best_seq, sub_deadline)

        # the remaining time is devoted to the directed greedy search
        a_final = _directed_local_search(a_refined, deadline)

        overall_best_seq = a_final.copy()
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print(f"[phase 3] after local search fast={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # Final black‑box evaluation
    try:
        true_score = evaluate_sequence(overall_best_seq.tolist())
        print(f"[final] evaluate_sequence score={true_score:.6f}")
        overall_best_score = min(overall_best_score, true_score)
    except Exception as exc:
        print("[final] evaluate_sequence raised:", exc)

    print(f"Best score {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
