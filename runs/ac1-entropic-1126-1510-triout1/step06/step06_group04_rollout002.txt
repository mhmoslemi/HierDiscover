"""
Propose a low‑score step function.

Algorithm
---------
1. **Random exploration** – generate many candidate probability vectors of
   various lengths (2000‑12000) from a wide range of parametric families:
   uniform, arcsine, symmetric Beta, random Beta, Dirichlet, Gaussian,
   mixtures of two/three Betas or Gaussians, and uniform+Beta mixes.
   Each candidate is projected onto the simplex (Σa_i = 1, a_i ≥ 0) and
   evaluated with the fast proxy 2·n·max(a*a).  The best candidate overall
   is kept as a warm‑start.

2. **Refinement** – a smooth‑max (log‑sum‑exp) approximation of the
   max‑convolution is minimized by projected Adam (adaptive moment) updates.
   For a given temperature β the weighted gradient is

       grad = 2·conv_rev(a, w) ,   w_k ∝ exp(β·(c_k−max(c))) ,

   where c = a*a and conv_rev denotes convolution of the reversed a with w.
   β starts small (≈2) and is increased periodically, turning the smooth
   approximation into the true max.  After every Adam step we project back
   onto the simplex and enforce symmetry (a = (a + a[::-1])/2), which is
   known to be optimal for the convolution problem.  Occasional tiny jitter
   helps escape plateaus.

The routine respects the 1000 s wall‑clock budget, printing progress and
returning the best sequence found (as a plain Python list).  The
`evaluate_sequence` function is called only once on the final candidate.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution of length n, normalized to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf (stable)
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logpdf = np.minimum(logpdf, 700.0)            # avoid overflow
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _gaussian_distribution(n: int, mu: float, sigma: float) -> np.ndarray:
    """Discrete Gaussian (un‑normalised) on the unit interval, projected to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    pdf = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _raw_score(a: np.ndarray) -> float:
    """Proxy score for Σa=1: 2·n·max(a*a)."""
    n = a.shape[0]
    return 2.0 * n * _self_convolution_full(a).max()


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the smooth‑max approximation of max(a*a).
    Returns (grad, current_max_convolution).
    """
    c = _self_convolution_full(a)
    cur_max = c.max()
    # soft‑max weights (numerically stable)
    w = np.exp(beta * (c - cur_max))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum
    # weighted gradient = 2 * conv_rev(a, w)
    grad = 2.0 * _fft_convolve(a[::-1], w)[a.size - 1 : a.size - 1 + a.size]
    return grad, cur_max

def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals
