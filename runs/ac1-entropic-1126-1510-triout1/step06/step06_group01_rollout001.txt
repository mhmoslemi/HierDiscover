"""
Propose a non‑negative step function (probability vector) minimizing
`evaluate_sequence`.  

The algorithm follows three stages:

1. **Coarse sampling** – generate several analytic families (uniform,
   arcsine, triangular, exponential, symmetric beta) and a few random
   Dirichlet starts for a handful of lengths.  Keep the best vector.

2. **Continuation via smoothed max** – minimise a smooth approximation of
   the max‑convolution using Adam on the probability simplex.
   The smoothing parameter α is increased in stages (1 → 5 → 20 →
   100 → 500 → 2000) and each stage is run for many thousands of steps.

3. **Exact max sub‑gradient descent** – after the continuation a true
   sub‑gradient (active‑set) descent with momentum is performed for as
   long as time permits.  Periodic jitter helps escape plateaux.

The best vector found (normalised to sum = 1) is returned as a plain
list of floats.  The whole routine respects a user‑supplied time budget
(default 1000 s) and prints progress.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) for a vector already normalised to sum = 1.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    """Uniform distribution (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_candidates(n: int) -> list[np.ndarray]:
    """Generate a small pool of plausible starts for a given length."""
    cand = []
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))
    cand.append(_distribution_exponential(n, lam=0.003))
    cand.append(_distribution_exponential(n, lam=0.01))
    for p in (0.2, 0.5, 1.0, 2.0):
        cand.append(_distribution_beta_sym(n, p))
    # a few sparse random vectors (≈0.5 % non‑zeros)
    k = max(2, n // 200)
    idx = np.random.choice(n, k, replace=False)
    rand = np.zeros(n, dtype=np.float64)
    rand[idx] = np.random.rand(k)
    if rand.sum() == 0:
        rand[0] = 1.0
    rand /= rand.sum()
    cand.append(rand)
    # a few Dirichlet‑type random vectors
    for _ in range(2):
        d = np.random.gamma(shape=0.3, scale=1.0, size=n)
        d /= d.sum()
        cand.append(d)
    return cand

def _smooth_descent(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max objective (log‑sum‑exp).
    Returns the best vector encountered and its (exact) score.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # smoothing schedule – larger α gives a tighter approximation to max
    schedule = [
        (1.0, 3000),
        (5.0, 3000),
        (20.0, 3000),
        (100.0, 3000),
        (500.0, 3000),
        (2000.0, 3000),
    ]

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score
            t += 1

            # full self‑convolution
            b = _fft_self_convolve(a)

            # soft‑max weights over b
            maxb = b.max()
            exp_terms = np.exp(alpha * (b - maxb))
            sum_exp = exp_terms.sum()
            w = exp_terms / sum_exp          # Σ w = 1

            # gradient of μ_α : 2 * (a reversed) * w
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam update on the simplex
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a, 1.0)

            # occasional exact evaluation
            if t % 150 == 0:
                cur_score = _max_convolution_score(a)
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_seq = a.copy()
    return best_seq, best_score

def _exact_subgradient_refine(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max objective with momentum.
    Continues until time runs out or no improvement is observed for a while.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)

    velocity = np.zeros_like(a)
    beta = 0.85               # momentum factor
    step0 = 1.0               # initial step magnitude
    iteration = 0
    last_improve_iter = 0

    while True:
        if time.time() >= deadline:
            break
        iteration += 1

        # self‑convolution and active set
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.where(b >= cur_max - eps)[0]

        w = np.zeros_like(b)
        w[active] = 1.0

        # sub‑gradient of the max
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # momentum + diminishing steps
        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / np.sqrt(iteration)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            last_improve_iter = iteration
            print(f"[{time.time():.1f}] subgrad iter={iteration} new best {best_score:.6f}")

        # occasional jitter to escape flat plateaux
        if iteration % 4000 == 0:
            a = best_seq + 1e-4 * np.random.randn(n)
            a = _project_onto_simplex(a, 1.0)

        # give up if we haven't improved for a long while
        if iteration - last_improve_iter > 20000:
            # gentle restart around the current best
            a = best_seq + 2e-4 * np.random.randn(n)
            a = _project_onto_simplex(a, 1.0)
            last_improve_iter = iteration

    return best_seq, best_score
