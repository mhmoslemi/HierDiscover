"""
Propose a step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.
The algorithm expands the analytic candidate pool (extreme Beta, exponential,
Dirichlet draws, previous result) and runs an Adam‑based exponentiated‑gradient
optimiser on the simplex for each promising starter.  A soft‑max surrogate is
used for the first phase to obtain smooth gradients; later the exact
sub‑gradient of the max‑convolution is applied.  The routine dynamically
allocates its 1000 s budget, keeps the best vector seen and stops early if the
target 1.503 is reached.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float = 1.0) -> np.ndarray:
    """Geometric‑like decreasing shape: a_i ∝ exp(-lam·i/(n-1))."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * i / max(1, n - 1))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate of max(conv).
    a   : probability vector (sum = 1)
    conv: self‑convolution of a
    alpha: temperature multiplier (larger → sharper)
    """
    n = a.shape[0]
    cur_max = conv.max()
    exp_term = np.exp(alpha * (conv - cur_max))
    w = exp_term / exp_term.sum()
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the true max, i.e. average over all active indices.
    """
    n = a.shape[0]
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even shifts (i == k-i)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad


# ----------------------------------------------------------------------
def _adam_step(z: np.ndarray,
               g: np.ndarray,
               m: np.ndarray,
               v: np.ndarray,
               t: int,
               lr: float = 0.5,
               beta1: float = 0.9,
               beta2: float = 0.999,
               eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Adam update on the unconstrained log‑variables."""
    m = beta1 * m + (1.0 - beta1) * g
    v = beta2 * v + (1.0 - beta2) * (g * g)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z, m, v


# ----------------------------------------------------------------------
def _cheap_score(a: np.ndarray) -> float:
    """Fast score for a probability vector (sum = 1): 2·n·max(conv)."""
    n = a.shape[0]
    return 2.0 * n * _fft_self_convolve(a).max()

    cleaned = []
    for a in candidates:
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            cleaned.append(a / s)
    return cleaned


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 1234, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # 0. Choose a base length (use previous result if available)
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # 1. Lengths to explore – a small set around the base length
    length_factors = (0.8, 1.0, 1.3, 1.6, 2.0)
    lengths = sorted(
        max(3000, int(base_n * f)) for f in length_factors
    )
    # keep at most 3 distinct lengths to stay within the budget
    lengths = lengths[:3]

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    target_score = 1.503

    for n in lengths:
        now = time.time()
        if now >= deadline:
            break
        print(f"[{now:.1f}] ------------------- Exploring n={n}")

        # ----------------------------------------------------------
        # 2. Build a large pool of analytic candidates
        pool = _make_initial_vectors(n)

        # cheap scoring and keep the most promising starters
        scored = [( _cheap_score(a), a) for a in pool]
        scored.sort(key=lambda x: x[0])
        top_k = 6
        starters = [a.copy() for _, a in scored[:top_k]]
        print(f"   top cheap scores: {[s for s, _ in scored[:top_k]]}")

        # ----------------------------------------------------------
        # 3. Refine each starter – Adam on the log‑softmax (exponentiated gradient)
        # Allocate a reasonable time slice per starter
        remaining = max(0.0, deadline - time.time())
        per_start_budget = min(120.0, remaining * 0.5 / max(1, len(starters)))

        for sid, a0 in enumerate(starters):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_start_budget
            if sub_deadline - time.time() < 5.0:
                continue

            # initialise Adam on log‑variables
            a = np.maximum(a0, 1e-15)
            z = np.log(a)
            m = np.zeros_like(z)
            v = np.zeros_like(z)

            # hyper‑parameters
            lr0 = 0.6
            beta1, beta2 = 0.9, 0.999
            eps = 1e-8

            temperature = 1.0
            temp_factor = 1.005
            temp_max = 3000.0

            best_local_a = a.copy()
            best_local_score = _cheap_score(a)

            it = 0
            while time.time() < sub_deadline:
                it += 1

                # probability vector and its self‑convolution
                a = np.exp(z)
                a_sum = a.sum()
                if a_sum == 0:
                    a = np.full_like(a, 1.0 / a.shape[0])
                else:
                    a /= a_sum

                conv = _fft_self_convolve(a)
                cur_score = 2.0 * n * conv.max()

                # keep the best we have seen for this starter
                if cur_score < best_local_score:
                    best_local_score = cur_score
                    best_local_a = a.copy()
                    print(
                        f"[{time.time():.1f}] n={n} starter={sid} it={it} cheap={cur_score:.6f}"
                    )
                    if best_local_score <= target_score:
                        break

                # gradient (soft‑max → exact)
                if temperature < temp_max:
                    grad_a = _softmax_gradient(a, conv, temperature)
                    temperature *= temp_factor
                else:
                    grad_a = _exact_gradient(a, conv)

                # convert to gradient w.r.t. z (mirror‑descent formula)
                dot = np.dot(a, grad_a)
                grad_z = a * (grad_a - dot)

                # Adam update
                z, m, v = _adam_step(z, grad_z, m, v, it,
                                     lr=lr0, beta1=beta1, beta2=beta2, eps=eps)

                # occasional tiny jitter to avoid getting stuck
                if it % 4000 == 0 and np.random.rand() < 0.12:
                    z += np.random.randn(len(z)) * 1e-7

            # ------------------------------------------------------
            # 4. Evaluate the best local candidate with the real black‑box
            try:
                final_score = evaluate_sequence(best_local_a.tolist())
            except Exception:
                final_score = float("inf")
            print(
                f"[{time.time():.1f}] evaluated starter={sid} → score={final_score:.6f}"
            )

            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = best_local_a.copy()
                print(
                    f"*** Global improvement: n={n}, starter={sid}, "
                    f"new best score={overall_best_score:.6f}"
                )
                if overall_best_score <= target_score:
                    print("Target reached – exiting early.")
                    return overall_best_seq.tolist()

        # ----------------------------------------------------------
        # 5. If time permits, launch a few pure random restarts using the
        #    best length discovered so far.
        while time.time() + 8.0 < deadline:
            # pick the length of the best sequence seen so far (fallback to n)
            cur_n = n if overall_best_seq is None else len(overall_best_seq)
            # generate a sparse Dirichlet start
            conc = np.random.choice([0.02, 0.05, 0.1, 0.2])
            a0 = np.random.dirichlet(np.full(cur_n, conc))
            # Run a short optimisation (≈8 s)
            sub_deadline = time.time() + 8.0

            a = np.maximum(a0, 1e-15)
            z = np.log(a)
            m = np.zeros_like(z)
            v = np.zeros_like(z)

            temperature = 1.0
            temp_factor = 1.008
            temp_max = 2500.0

            best_local_a = a.copy()
            best_local_score = _cheap_score(a)

            it = 0
            while time.time() < sub_deadline:
                it += 1
                a = np.exp(z)
                a_sum = a.sum()
                if a_sum == 0:
                    a = np.full_like(a, 1.0 / a.shape[0])
                else:
                    a /= a_sum

                conv = _fft_self_convolve(a)
                cur_score = 2.0 * cur_n * conv.max()
                if cur_score < best_local_score:
                    best_local_score = cur_score
                    best_local_a = a.copy()

                # gradient choice
                if temperature < temp_max:
                    grad_a = _softmax_gradient(a, conv, temperature)
                    temperature *= temp_factor
                else:
                    grad_a = _exact_gradient(a, conv)

                dot = np.dot(a, grad_a)
                grad_z = a * (grad_a - dot)
                z, m, v = _adam_step(z, grad_z, m, v, it,
                                     lr=0.5, beta1=0.9, beta2=0.999, eps=1e-8)

            # evaluate
            try:
                final_score = evaluate_sequence(best_local_a.tolist())
            except Exception:
                final_score = float("inf")
            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = best_local_a.copy()
                print(
                    f"*** New global best from random restart: score={overall_best_score:.6f}"
                )
                if overall_best_score <= target_score:
                    print("Target reached – exiting early.")
                    return overall_best_seq.tolist()

    # ------------------------------------------------------------------
    # 6. Fallback – uniform distribution (should never be needed)
    if overall_best_seq is None:
        fallback_n = max(3000, base_n)
        overall_best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())
        print("Fallback uniform sequence used.")

    print(
        f"Final best score: {overall_best_score:.6f} "
        f"(length = {len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
