assistantfinal**Solution Overview**

We treat the problem as a continuous optimisation on the simplex  

```
minimise   2·n·max_k (a∗a)_k      subject to   a_i ≥ 0 ,  Σ a_i = 1 .
```

The objective is scale‑invariant, therefore the sum is fixed to 1 and the
search space is the probability simplex.  
The code below follows three ideas that were **not** used in the previous
state‑of‑the‑art implementation:

1. **A large pool of parametric seeds** – uniform, arcsine, many β‑shapes,
   exponentials, quadratic and sinusoidal profiles together with a set of
   random sparse starts.  For every candidate we evaluate the exact
   objective (`2·n·max(a∗a)`) with an FFT based convolution.

2. **Multi‑scale refinement** – starting from the best seed we repeatedly
   *upsample* the vector (linear interpolation), re‑normalise it and run a
   long Adam‑based smoothed‑max descent.  Upsampling allows us to explore
   lengths up to ≈ 20 000 without having to run a full optimisation from
   scratch for each size.

3. **Two final polishing phases** – an exact max‑sub‑gradient descent
   (the same scheme as in the reference) and a cheap random‑transfer
   hill‑climber with occasional jitter.

All steps respect the global time budget (`budget_s`).  The implementation
contains only top‑level functions, uses only `numpy`, `math` and the standard
library, and works on a single thread.

```python
"""
Search for a non‑negative step function (probability vector) that minimises
evaluate_sequence.  The algorithm proceeds as follows:

1.  Generate many analytic “seed” vectors for several lengths (uniform,
    arcsine, symmetric beta(p), triangular, quadratic, sinusoidal,
    exponential, and random sparse vectors).  Keep the best seed.

2.  Refine the best seed with a smoothed‑max Adam optimiser (log‑sum‑exp
    approximation).  The optimiser works on the simplex (Euclidean projection
    after each update).

3.  Upsample the refined vector to a larger length (linear interpolation),
    re‑normalise, and repeat step 2.  This multi‑scale scheme lets the search
    explore lengths up to ≈20 000 while keeping the total number of FFTs
    moderate.

4.  Finish with an exact max‑sub‑gradient descent (momentum, diminishing step)
    and a random‑transfer hill‑climber (mass moves between two indices with
    acceptance only on improvement).

The whole procedure stops before the supplied time budget (default 1000 s)
and returns the best vector (as a Python list of floats) found so far.
"""

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σ x = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a∗a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a∗a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# --- Analytic families -------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑space to avoid overflow for extreme p
    logw = (p - 1) * (np.log(x) + np.log(1.0 - x))
    w = np.exp(logw - np.max(logw))
    w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_quadratic(n: int) -> np.ndarray:
    """Concave quadratic shape a_i ∝ (i+½)(n‑i‑½)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = i * (n - i)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    """Positive half‑sine: a_i ∝ sin(π·(i+½)/n)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * i / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, nz_frac: float) -> np.ndarray:
    """Random vector with ≈ nz_frac·n non‑zero entries."""
    k = max(2, int(round(n * nz_frac)))
    idx = np.random.choice(n, k, replace=False)
    vec = np.zeros(n, dtype=np.float64)
    vec[idx] = np.random.rand(k)
    if vec.sum() == 0:
        vec[:] = 1.0 / n
    else:
        vec /= vec.sum()
    return vec

def _smooth_descent(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max objective
    μ_α(a) = (1/α)·log Σ exp(α·(a∗a)_k).
    Returns the best vector (still normalised to sum = 1) and its exact score.
    """
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _max_convolution_score(a)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # smoothing schedule – more steps than the previous code
    schedule = [
        (1.0, 2000),
        (5.0, 2000),
        (20.0, 2000),
        (100.0, 2000),
        (500.0, 2000),
        (2000.0, 1500),
        (5000.0, 1500),
    ]

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_vec, best_score
            t += 1

            # full self‑convolution
            b = _fft_self_convolve(a)

            # stable soft‑max weights
            maxb = b.max()
            exp_terms = np.exp(alpha * (b - maxb))
            w = exp_terms / exp_terms.sum()       # Σ w = 1

            # gradient of the smooth max (2·conv_rev(a_rev, w))
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam update + projection onto simplex
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a, 1.0)

            if t % 200 == 0:
                cur_score = _max_convolution_score(a)
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_vec = a.copy()
    return best_vec, best_score

def _exact_subgradient_refine(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent for the true max objective.
    Momentum and a 1/√t stepsize are used.
    """
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _max_convolution_score(a)

    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0

    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps)

        w = np.zeros_like(b)
        w[active] = 1.0
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[{time.time():.1f}] subgrad iter={it} new best {best_score:.6f}")

        if best_score <= 1.503:          # early stop if target already reached
            break
        if time.time() + 0.5 >= deadline:
            break
    return best_vec, best_score

def _local_hill_climb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Random‑transfer hill climbing on the simplex.
    A small amount of mass is moved from i to j; the move is kept only if it
    improves the exact objective.
    """
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _max_convolution_score(a)
    cur_score = best_score

    max_transfer = 0.07          # start with up to 7 % of total mass
    it = 0

    while time.time() < deadline:
        it += 1
        # geometric decay of the transfer magnitude
        transfer = max_transfer * (0.5 ** (it / 4000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_vec = a_new.copy()
                print(f"[{time.time():.1f}] hill iter={it} improved best {best_score:.6f}")

        # occasional jitter to get out of plateaus
        if it % 5000 == 0 and np.random.rand() < 0.2:
            pert = np.random.randn(n) * 1e-4
            a = best_vec + pert
            a = _project_onto_simplex(a, 1.0)
            cur_score = _max_convolution_score(a)

    return best_vec, best_score


# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length len(a) to new_n (preserves sum)."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    old_x = np.arange(old_n, dtype=np.float64)
    new_x = np.linspace(0, old_n - 1, new_n)
    a_new = np.interp(new_x, old_x, a)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0:
        a_new = np.full(new_n, 1.0 / new_n)
    else:
        a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Main entry point.
    Searches for a step‑function (non‑negative vector) that minimises
    evaluate_sequence and returns the best sequence found as a Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # leave a small safety margin

    # ------------------------------------------------------------------
    # 1. Choose a set of lengths (start small, go up to ~20 k)
    lengths = [800, 1200, 2000, 3000, 5000, 8000, 12000, 16000, 20000]
    max_len = lengths[-1]

    overall_best = None
    overall_score = float("inf")

    # ------------------------------------------------------------------
    # 2. Scan the analytic families for every length
    for n in lengths:
        if time.time() >= deadline:
            break
        for cand in _initial_candidates(n):
            if time.time() >= deadline:
                break
            cand = cand / cand.sum()           # guard against numerical drift
            sc = _max_convolution_score(cand)
            if sc < overall_score:
                overall_score = sc
                overall_best = cand.copy()
                print(f"[{time.time():.1f}] n={n} init score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3. If a previously found vector exists (global name height_sequence_1),
    #    inject it as an extra seed.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            if sc < overall_score:
                overall_score = sc
                overall_best = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. If nothing was found (should never happen) fall back to uniform.
    if overall_best is None:
        n0 = lengths[0]
        overall_best = _distribution_uniform(n0)
        overall_score = _max_convolution_score(overall_best)
        print(f"[{time.time():.1f}] fallback uniform score={overall_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Multi‑scale refinement loop.
    #    Repeatedly upsample (if possible), run a long smooth descent,
    #    and keep the best result.
    while time.time() < deadline - 30.0:   # keep ~30 s for final polishing
        cur_n = overall_best.shape[0]
        if cur_n >= max_len:
            break
        # increase size by roughly 1.5× (capped by max_len)
        new_n = min(max_len, int(cur_n * 1.5))
        up = _upsample(overall_best, new_n)

        # smooth Adam descent on the larger vector
        up, sc = _smooth_descent(up, deadline)
        if sc < overall_score:
            overall_score = sc
            overall_best = up.copy()
            print(f"[{time.time():.1f}] upsampled n={new_n} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 6. Exact sub‑gradient refinement (targeting the true max).
    overall_best, overall_score = _exact_subgradient_refine(overall_best, deadline)

    # ------------------------------------------------------------------
    # 7. Final random‑transfer hill climbing.
    overall_best, overall_score = _local_hill_climb(overall_best, deadline)

    # ------------------------------------------------------------------
    # 8. Normalise one last time (numerical drift) and return.
    overall_best = overall_best / overall_best.sum()
    print(f"Final best score: {overall_score:.6f} (length={len(overall_best)})")
    return overall_best.tolist()
```
