"""
Propose a low‑score step function.

Algorithm
---------
* Choose a problem size (≈ 9 000) and fix the ℓ¹‑norm S = sqrt(2·n).
* Initialise the probability vector p = a / S from many random Dirichlet
  draws (including the reference if available) and work in the log‑odds
  domain y (soft‑max gives p = exp(y)/∑exp(y)).
* Perform a few Adam‑style sub‑gradient steps on the exact objective
  ``max_k (p∗p)_k``.  The gradient is obtained analytically; the whole
  convolution is evaluated by FFT for speed.
* Keep the best candidate seen during optimisation.
* Finish with a cheap stochastic hill‑climber (swap / δ‑transfer /
  block‑averaging moves) that accepts only strict improvements.
* Finally try a few simple deterministic reorderings (ascending,
  descending, interleaved big‑small) and keep any better result.

The routine respects a user supplied time budget (default 1000 s) and
returns the best height sequence found.
"""

import math
import numpy as np
import scipy.optimize as opt

# Global placeholders that will be filled inside propose_candidate
_g_n: int = 0                     # length of the sequence
_g_S: float = 0.0                 # target ℓ¹‑norm (= sqrt(2·n))

def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()

# FFT‑based convolutions – the global size is set once per call.
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full auto‑convolution a*a using FFT (length 2·n‑1)."""
    A = np.fft.rfft(a, _g_fft_size)
    c = np.fft.irfft(A * A, _g_fft_size)
    return c[: 2 * _g_n - 1]


def _conv_wr_fft(w: np.ndarray, a_rev: np.ndarray) -> np.ndarray:
    """Convolution w * (reversed a) using the same FFT size."""
    W = np.fft.rfft(w, _g_fft_size)
    A = np.fft.rfft(a_rev, _g_fft_size)
    c = np.fft.irfft(W * A, _g_fft_size)
    return c[: 2 * _g_n - 1]


def _max_conv_and_grad(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Compute max(conv) and its gradient w.r.t. a for the hard objective.
    Returns (max_value, grad_a) where grad_a has length n.
    """
    # auto‑convolution
    conv = _conv_self_fft(a)
    max_c = conv.max()
    # uniform weight over all active indices
    mask = conv == max_c
    cnt = int(mask.sum())
    if cnt == 0:
        w = np.zeros_like(conv)
    else:
        w = np.zeros_like(conv)
        w[mask] = 1.0 / cnt

    # gradient w.r.t. a
    a_rev = a[::-1]
    conv_wr = _conv_wr_fft(w, a_rev)
    grad_a = 2.0 * conv_wr[_g_n - 1:_g_n - 1 + _g_n]
    return max_c, grad_a


def _hard_subgradient_y(y: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient of the true max‑objective with respect to the
    log‑odds vector y (soft‑max parametrisation).  Returns (grad_y, max_c).
    """
    p = _softmax(y)                     # Σ p_i = 1
    a = _g_S * p                       # Σ a_i = S
    max_c, g_a = _max_conv_and_grad(a)  # g_a = ∂max/∂a
    inner = np.dot(g_a, p)              # scalar Σ_j p_j g_a_j
    grad_y = _g_S * p * (g_a - inner)   # chain rule through soft‑max
    return grad_y, max_c


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step‑function with a low evaluate_sequence score."""
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size ----------------------------------------------------
    try:
        # Use the length of the reference solution (if present) as a baseline.
        ref_len = len(height_sequence_1)   # type: ignore[name-defined]
        n = min(max(ref_len * 2, 4000), 9000)
        have_ref = True
    except Exception:
        n = 6000
        have_ref = False

    global _g_n, _g_S, _g_fft_size
    _g_n = n
    _g_S = math.sqrt(2.0 * n)

    # size for FFT (next power‑of‑two >= 2·n‑1)
    _g_fft_size = 1 << ((2 * n - 1) - 1).bit_length()

    # ------------------------------------------------------------------
    best_a = np.zeros(n, dtype=float)
    best_score = float("inf")

    def _maybe_update(a_vec: np.ndarray) -> None:
        """Evaluate a candidate and keep it if it improves the incumbent."""
        nonlocal best_score, best_a
        sc = evaluate_sequence(a_vec.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a_vec.copy()
            print(f"[improve] new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 0️⃣  Uniform baseline (soft‑max of zero logits)
    y0 = np.zeros(n, dtype=float)
    a0 = _g_S * _softmax(y0)
    _maybe_update(a0)

    # ------------------------------------------------------------------
    # 1️⃣  Adam optimisation from many random starts
    n_restart = 15
    for rid in range(n_restart):
        if time.time() > deadline:
            break
        # random Dirichlet‐type start (concentration sampled uniformly)
        conc = rng.uniform(0.03, 1.2)
        p0 = rng.gamma(shape=conc, scale=1.0, size=n)
        p0 /= p0.sum()
        y = np.log(p0 + 1e-12)

        # occasionally use the reference as a start
        if rid == 1 and have_ref:
            a_ref = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
            a_ref = _project_to_simplex(a_ref, _g_S)
            y = np.log(a_ref / _g_S + 1e-12)

        # Adam hyper‑parameters (a few thousand steps are cheap with FFT)
        max_steps = 2000
        lr = 0.5 / (rid + 1)                     # decreasing learning‑rate
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = np.zeros_like(y)
        v = np.zeros_like(y)

        for t in range(1, max_steps + 1):
            if time.time() > deadline:
                break
            grad_y, _ = _hard_subgradient_y(y)
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact evaluation
            if t % 100 == 0:
                a = _g_S * _softmax(y)
                _maybe_update(a)

        # final evaluation for this restart
        a = _g_S * _softmax(y)
        _maybe_update(a)

    # ------------------------------------------------------------------
    # 2️⃣  Stochastic hill‑climbing on the incumbent
    if time.time() < deadline and best_score < float("inf"):
        a = best_a.copy()
        remaining = deadline - time.time()
        max_trials = int(min(5e4, remaining * 30.0))
        print(f"[hill‑climb] {max_trials} trials")
        for trial in range(max_trials):
            move_type = rng.integers(0, 3)   # 0:swap, 1:δ‑transfer, 2:block‑average
            cand = a.copy()
            if move_type == 0:
                i, j = rng.choice(n, size=2, replace=False)
                cand[i], cand[j] = cand[j], cand[i]
            elif move_type == 1:
                nonzero = np.where(cand > 1e-12)[0]
                if nonzero.size == 0:
                    continue
                i = rng.choice(nonzero)
                j = rng.integers(0, n)
                delta = rng.random() * cand[i] * 0.04
                cand[i] -= delta
                cand[j] += delta
            else:  # block averaging
                start = rng.integers(0, n - 1)
                length = rng.integers(2, min(80, n - start))
                block = slice(start, start + length)
                avg = cand[block].mean()
                cand[block] = avg
            # re‑normalise onto the ℓ¹‑simplex
            cand = _project_to_simplex(cand, _g_S)

            sc = evaluate_sequence(cand.tolist())
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = cand.copy()
                a = cand
                print(f"[hill] trial {trial:5d} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Deterministic re‑orderings (ascending, descending, interleaved)
    if time.time() < deadline and best_score < float("inf"):
        asc = np.sort(best_a)
        _maybe_update(_project_to_simplex(asc, _g_S))

        desc = asc[::-1]
        _maybe_update(_project_to_simplex(desc, _g_S))

        # interleave big–small pattern
        vals = np.sort(best_a)[::-1]               # descending
        inter = np.empty_like(vals)
        half = len(vals) // 2
        inter[0::2] = vals[:half][::-1]           # small half (reversed)
        inter[1::2] = vals[half:][::-1]           # large half (reversed)
        _maybe_update(_project_to_simplex(inter, _g_S))

    # ------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
