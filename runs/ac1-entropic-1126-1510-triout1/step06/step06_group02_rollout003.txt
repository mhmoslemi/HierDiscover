"""
Propose a step‑function that beats the previous best (≈ 1.5105).

Algorithm
---------
1. **Broad exploration** – generate a diverse pool of analytic
   candidates (Beta, exponential, uniform, random Dirichlet) for a few
   problem sizes (≈ 3000‑8000).  Each candidate is cheap‑scored by
   `2·n·max(a*a)` (the exact objective when the vector is normalised to
   sum = 1).  Keep the best few candidates (the top 3).

2. **Deep refinement** – for each kept candidate run a *projected
   sub‑gradient descent* on the simplex with an adaptive step size and
   a simple back‑tracking line‑search.  The exact sub‑gradient of the
   `max` of the self‑convolution is used (averaged over all active
   indices).  Every few thousand iterations a small random jitter is
   added to escape plateaus; after a long stagnation a fresh random
   restart is performed.

3. **Time allocation** – the total budget (≈ 1000 s) is split:
   * exploration  ≈ 10 %,
   * refinement   ≈ 90 %.
   The best sequence seen (according to the cheap score) is finally
   evaluated with the real ``evaluate_sequence`` function and returned.

The method is deliberately simple, uses only NumPy, and respects all
security checks of ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float = 1.0) -> np.ndarray:
    """Geometric‑like decreasing shape: a_i ∝ exp(-lam·i/(n‑1))."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * i / max(1, n - 1))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _cheap_score(a: np.ndarray) -> float:
    """Cheap surrogate of the real objective (sum a = 1)."""
    return 2.0 * a.shape[0] * _fft_self_convolve(a).max()


# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the true max of the self‑convolution.
    Returns ∂max/∂a (no projection).
    """
    n = a.shape[0]
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    # convolution of reversed a with the weight vector
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even shifts (avoid double‑counting squares)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad


# ----------------------------------------------------------------------
def _random_dirichlet(n: int, concentration: float) -> np.ndarray:
    """Draw a Dirichlet vector with a single concentration parameter."""
    y = np.random.gamma(concentration, 1.0, size=n)
    y_sum = y.sum()
    return y / y_sum if y_sum > 0 else np.full(n, 1.0 / n)
