"""
Propose a step‑function via mixture‑Beta initialisation and
projected Adam refinement.

Algorithm
---------
1. Load the previous best sequence (global ``height_sequence_1``) if it
   exists, otherwise start from a Beta(0.5,0.5) distribution.
2. Choose a few lengths around the base size (≈ 0.75×, 1×, 1.5×, 2×).
   Larger lengths give a better asymptotic constant but are slower,
   therefore the set is kept modest.
3. For each length build a *large* pool of analytic candidates:
   * the interpolated previous construction,
   * random mixtures of two Beta densities,
   * a few random Dirichlet draws.
   All candidates are normalised to sum = 1 (probability simplex).
4. Evaluate each candidate cheaply via ``2·n·max(conv)`` (the objective
   when the vector is normalised).  Keep the best few (≤ 2) per length.
5. Refine every kept candidate with Adam on the unconstrained variable
   ``z`` where ``a = softmax(z)``.  The exact sub‑gradient of the true
   ``max(conv)`` is used from the very first iteration, which is cheap
   thanks to FFT‑based convolutions.  The Adam step is projected back to
   the simplex by the soft‑max transformation.
6. After the time‑limited refinement the best local vector is scored
   with the real ``evaluate_sequence`` black‑box.  The globally best
   sequence seen so far is returned (or a uniform fallback if nothing
   succeeded).

The routine respects the supplied time budget (≈ 1000 s) and prints
progress information.  All helper functions are defined at top‑level
as required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the true max of a*a.
    Returns ∂/∂a  max(conv).
    """
    n = a.shape[0]
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for i=j pairs (they appear only once)
    even_mask = (active % 2 == 0)
    diag_idx = active[even_mask] // 2
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad


# ----------------------------------------------------------------------
def _adam_step(z: np.ndarray,
               g: np.ndarray,
               m: np.ndarray,
               v: np.ndarray,
               t: int,
               lr: float = 0.5,
               beta1: float = 0.9,
               beta2: float = 0.999,
               eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Adam update for the unconstrained variable ``z``."""
    m = beta1 * m + (1.0 - beta1) * g
    v = beta2 * v + (1.0 - beta2) * (g * g)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z, m, v


# ----------------------------------------------------------------------
def _cheap_score(a: np.ndarray) -> float:
    """Cheap objective (exact when ``a`` sums to 1)."""
    n = a.shape[0]
    return 2.0 * n * _fft_self_convolve(a).max()


# ----------------------------------------------------------------------
def _mixture_beta(n: int,
                  alpha1: float,
                  beta1: float,
                  alpha2: float,
                  beta2: float,
                  w: float) -> np.ndarray:
    """
    Two‑component Beta mixture, normalised to sum = 1.
    ``w`` is the weight of the first component (0 ≤ w ≤ 1).
    """
    comp1 = _beta_distribution(n, alpha1, beta1)
    comp2 = _beta_distribution(n, alpha2, beta2)
    mix = w * comp1 + (1.0 - w) * comp2
    mix_sum = mix.sum()
    if mix_sum == 0.0:
        mix = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        mix = mix / mix_sum
    return mix


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 1234, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the provided time budget and returns the best sequence
    discovered.  It builds analytic candidates (Mixture‑Beta, Dirichlet, …),
    keeps the most promising ones, and refines them with Adam on the simplex.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 0. retrieve previous construction if present
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if base_seq.size == 0:
            raise ValueError
        base_seq = base_seq / base_seq.sum()
        base_len = base_seq.size
    except Exception:
        # fallback – a generic Beta(0.5,0.5) shape
        base_len = 8000
        base_seq = _beta_distribution(base_len, 0.5, 0.5)

    # ------------------------------------------------------------------
    # 1. lengths to explore (keep them modest)
    length_factors = (0.75, 1.0, 1.5, 2.0)
    lengths = sorted({max(3000, int(base_len * f)) for f in length_factors})
    # ------------------------------------------------------------------
    overall_best_seq = None               # type: np.ndarray | None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] ==== length n={n} ==== ")

        # --------------------------------------------------------------
        # 2. build pool of analytic candidates
        pool = []

        # a) interpolated previous construction
        x_old = np.linspace(0.0, 1.0, base_seq.size)
        x_new = np.linspace(0.0, 1.0, n)
        interp = np.interp(x_new, x_old, base_seq)
        interp = np.maximum(interp, 1e-15)
        interp /= interp.sum()
        pool.append(interp)

        # b) random Mixture‑Beta candidates
        for _ in range(30):
            a1 = 10 ** np.random.uniform(-0.8, 0.8)   # roughly 0.16 … 6.3
            b1 = 10 ** np.random.uniform(-0.8, 0.8)
            a2 = 10 ** np.random.uniform(-0.8, 0.8)
            b2 = 10 ** np.random.uniform(-0.8, 0.8)
            w = np.random.rand()
            mix = _mixture_beta(n, a1, b1, a2, b2, w)
            pool.append(mix)

        # c) a few random Dirichlet draws (Gamma(1) -> uniform Dirichlet)
        for _ in range(5):
            dirich = np.random.gamma(1.0, 1.0, size=n)
            dirich = dirich / dirich.sum()
            pool.append(dirich)

        # --------------------------------------------------------------
        # 3. cheap scores and keep top candidates (≤2)
        scored = [(float(_cheap_score(p)), p) for p in pool]
        scored.sort(key=lambda x: x[0])
        top_vecs = [vec.copy() for _, vec in scored[:2]]
        print(f"  top cheap scores (objective when sum=1): {[s for s, _ in scored[:2]]}")

        # --------------------------------------------------------------
        # 4. refine each top candidate with Adam (exact sub‑gradient)
        for cand_id, a0 in enumerate(top_vecs):
            if time.time() >= deadline:
                break

            # allocate a slice of the remaining budget (≈ 20 % per candidate)
            now = time.time()
            remaining = max(0.0, deadline - now)
            cand_deadline = now + min(remaining * 0.2, 120.0)   # at most 2 min per candidate

            # initialise Adam on log‑odds (soft‑max) variables
            a = np.maximum(a0, 1e-15)
            z = np.log(a)
            m = np.zeros_like(z)
            v = np.zeros_like(z)

            best_local_a = a.copy()
            best_local_score = _cheap_score(a)
            it = 0

            while time.time() < cand_deadline:
                it += 1

                # current simplex point and its self‑convolution
                a = np.exp(z)
                a /= a.sum()
                conv = _fft_self_convolve(a)

                cur_score = 2.0 * n * conv.max()
                if cur_score < best_local_score:
                    best_local_score = cur_score
                    best_local_a = a.copy()
                    # occasional progress print
                    if it % 200 == 0:
                        print(f"[{time.time():.1f}] n={n} cand={cand_id} it={it} score={cur_score:.6f}")

                # exact sub‑gradient of max(conv)
                grad_a = _exact_gradient(a, conv)

                # convert gradient w.r.t. a into gradient w.r.t. z (soft‑max Jacobian)
                dot = np.dot(a, grad_a)
                grad_z = a * (grad_a - dot)

                # Adam step (1‑based iteration index)
                z, m, v = _adam_step(z, grad_z, m, v, it, lr=0.25)

                # keep z bounded to avoid numerical overflow
                if it % 1000 == 0:
                    z = np.clip(z, -30.0, 30.0)

            # ----------------------------------------------------------
            # final evaluation with the true black‑box
            try:
                final_score = evaluate_sequence(best_local_a.tolist())
            except Exception:
                final_score = best_local_score   # fallback to cheap score

            print(f"*** candidate n={n} id={cand_id} final_score={final_score:.6f}")

            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = best_local_a.copy()
                print(
                    f"+++ New global best: n={n}, cand={cand_id}, "
                    f"score={overall_best_score:.6f}"
                )
                if overall_best_score <= 1.503:
                    print("Target achieved – exiting early.")
                    return overall_best_seq.tolist()

    # ------------------------------------------------------------------
    # 5. fallback in case nothing succeeded (should never happen)
    if overall_best_seq is None:
        n_fallback = 4000
        overall_best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())
        print(f"Fallback used, score={overall_best_score:.6f}")

    print(
        f"Final best score: {overall_best_score:.6f} "
        f"(length = {len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
