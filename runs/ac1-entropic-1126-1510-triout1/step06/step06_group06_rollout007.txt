    cur_a = a.copy()
    cur_score = best_score
    it = 0
    while time.time() < deadline:
        it+=1
        # compute gradient parts for max conv: use _max_subgradient_step to get grad and cur_max
        grad, cur_max = _max_subgradient_step(cur_a) # this returns gradient (subgrad) and cur_max

        # Move delta from src to dst
        new_a = cur_a.copy()
        new_a[src] -= delta
        new_a[dst] += delta
        # ensure non-negative automatically

"""
Propose a low‑score step‑function.

The algorithm builds on the previous pipeline but adds several
enhancements:

*   A richer “multi‑level’’ parametric family (three blocks) is
    explored exhaustively on a log‑scale of height ratios.
*   Adam optimisation of a smooth‐max surrogate is run with a more
    aggressive learning‑rate schedule and a higher final smoothness.
*   Sub‑gradient descent uses a back‑tracking line‑search instead of a
    fixed step.
*   A “gradient‑swap’’ phase moves mass from indices that increase the
    current peak to indices that decrease it, based on the exact
    sub‑gradient.
*   A final simulated‑annealing phase performs large random mass‑transfer
    moves that are accepted with a temperature schedule.

All updates keep the vector on the probability simplex
Σa_i = 1, so the evaluation function `2·n·max(conv(a,a))` equals the
true black‑box score.  The routine respects a 1000 s budget, adapts the
time allocated to each phase and prints intermediate progress.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth‑max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # soft‑max weights (numerically stable)
    exp_term = np.exp(alpha * (b - cur_max))
    w_sum = exp_term.sum()
    w = exp_term / w_sum
    # convolution of reversed a with the weight vector
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.02,
                   alpha_max: float = 8000.0,
                   lr0: float = 0.5,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best vector (projected on the simplex) and the best
    exact peak ever seen.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size schedule – slower decay than 1/√t
        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # keep the best exact peak seen so far
        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        # continuation on the smoothness parameter
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional jitter – helps escape flat plateaus
        if t % 400 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Exact ℓ∞ sub‑gradient of the self‑convolution peak.
    Returns (grad, cur_max) where grad points in the direction of
    increase of the peak.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max

def _subgradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Sub‑gradient descent with a simple back‑tracking line search.
    Returns the best vector found before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = 2 * n * _fft_self_convolve(a).max()
    step = 0.2                       # initial trial step
    it = 0

    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)

        # try a few decreasing step sizes until improvement
        improved = False
        for _ in range(5):
            a_new = a - step * grad
            a_new = _project_onto_simplex(a_new, s=1.0)
            new_score = 2 * n * _fft_self_convolve(a_new).max()
            if new_score < best_score - 1e-15:
                a = a_new
                best_a = a_new.copy()
                best_score = new_score
                step *= 1.05            # grow step a bit after success
                improved = True
                break
            else:
                step *= 0.5             # shrink step and retry

        if not improved:
            # occasional random perturbation to avoid stagnation
            if np.random.rand() < 0.02:
                a += np.random.randn(n) * 1e-6
                a = _project_onto_simplex(a, s=1.0)

    return best_a

def _gradient_swap_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Moves mass from indices with large sub‑gradient components to those
    with small components (greedy coordinate swap).  The step size
    (amount transferred) shrinks as 1/√iteration.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = 2 * n * _fft_self_convolve(a).max()
    it = 0

    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)

        src = np.argmax(grad)      # source: contributes most to the peak
        dst = np.argmin(grad)      # destination: contributes least

        if src == dst or grad[src] <= grad[dst] + 1e-15:
            continue

        max_delta = a[src] * 0.5               # move up to 50 % of src
        delta = np.random.rand() * max_delta   # random fraction

        if delta < 1e-12:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        a_new = _project_onto_simplex(a_new, s=1.0)

        new_score = 2 * n * _fft_self_convolve(a_new).max()
        if new_score < best_score - 1e-15:
            a = a_new
            best_a = a_new.copy()
            best_score = new_score

        # occasional jitter
        if it % 2000 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

    return best_a


# ----------------------------------------------------------------------
def _random_moves_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simulated‑annealing style mass transfers that always target the
    currently active convolution peak.  Moves that increase the peak
    are accepted with probability exp(−Δ/T) where the temperature T
    decays linearly to zero.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = 2 * n * _fft_self_convolve(a).max()
    start = time.time()
    init_T = best_score * 0.01 + 1e-12   # start with ~1 % of the current score

    while time.time() < deadline:
        elapsed = time.time() - start
        T = init_T * max(0.0, 1.0 - elapsed / (deadline - start))

        # current peak index
        b = _fft_self_convolve(a)
        k = int(b.argmax())

        # contribution of each index to this peak
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        i_vals = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[i_vals] * a[k - i_vals]
        if contrib.sum() == 0.0:
            src = np.random.randint(n)
        else:
            src = np.random.choice(i_vals, p=contrib / contrib.sum())

        dst = np.random.randint(n)
        if dst == src:
            continue

        max_delta = min(a[src], 0.2)               # up to 20 % of src
        delta = np.random.rand() * max_delta

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        a_new = _project_onto_simplex(a_new, s=1.0)

        new_score = 2 * n * _fft_self_convolve(a_new).max()
        dE = new_score - best_score

        if dE < 0 or np.random.rand() < math.exp(-dE / (T + 1e-12)):
            a = a_new
            if new_score < best_score - 1e-15:
                best_a = a_new.copy()
                best_score = new_score

    return best_a


# ----------------------------------------------------------------------
def _three_level_family_search(n: int, deadline: float):
    """
    Exhaustive search over a simple three‑block family.
    The front block has height p, the middle block q, the back block r,
    with ratios p/q and q/r explored on a log‑scale.
    Returns the best vector and its fast score.
    """
    best_seq = None
    best_score = float('inf')
    # step sizes for the split points (≈5 % of the length)
    step_L = max(1, n // 20)

    # ratios explored in log‑space (from 1 to ≈300)
    r_grid = np.logspace(0.0, 2.5, num=28)   # p/q
    s_grid = np.logspace(0.0, 2.5, num=28)   # q/r

    for L1 in range(step_L, n - step_L, step_L):
        if time.time() > deadline:
            break
        for L2 in range(L1 + step_L, n, step_L):
            for r in r_grid:
                for s in s_grid:
                    # relative heights
                    r_back = 1.0
                    q_mid = s * r_back
                    p_front = r * q_mid
                    denom = L1 * p_front + (L2 - L1) * q_mid + (n - L2) * r_back
                    seq = np.empty(n, dtype=np.float64)
                    seq[:L1] = p_front / denom
                    seq[L1:L2] = q_mid / denom
                    seq[L2:] = r_back / denom
                    peak = _fft_self_convolve(seq).max()
                    score = 2.0 * n * peak
                    if score < best_score:
                        best_score = score
                        best_seq = seq.copy()
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Main entry point.  The routine searches a variety of families,
    runs Adam, sub‑gradient, a gradient‑swap descent and a final
    simulated‑annealing phase while respecting the given time budget.
    The returned list is a non‑negative step‑function (its sum is 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # small safety margin

    # --------------------------------------------------------------
    # 1.  Initialise from a previous construction (if present) or from
    #     a classic arcsine distribution.
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        base_seq = _project_onto_simplex(base_seq, s=1.0)
        base_n = len(base_seq)
    except Exception:
        base_n = 4000
        i = np.arange(base_n, dtype=np.float64) + 0.5
        base_seq = 1.0 / np.sqrt(i * (base_n - i + 0.5))
        base_seq = np.maximum(base_seq, 0.0)
        base_seq /= base_seq.sum()

    overall_best_seq = base_seq.copy()
    overall_best_score = 2 * base_n * _fft_self_convolve(overall_best_seq).max()
    print(f"[init] length={base_n} fast_score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 2.  Broad exploration of simple families (including a 3‑level family)
    lengths = [max(500, int(base_n * f))
               for f in (0.6, 0.8, 1.0, 1.2, 1.4)
               if int(base_n * f) >= 500]
    if not lengths:
        lengths = [max(500, base_n)]

    for n in lengths:
        if time.time() >= deadline:
            break
        # a) classic families
        starters = {}

        starters["arcsine"] = (1.0 / np.sqrt(
            (np.arange(n) + 0.5) * (n - np.arange(n) - 0.5)))
        starters["arcsine"] /= starters["arcsine"].sum()

        starters["uniform"] = np.full(n, 1.0 / n, dtype=np.float64)

        starters["random"] = np.random.rand(n)
        starters["random"] /= starters["random"].sum()

        # Beta(α,α) with several α
        for α in (0.15, 0.25, 0.35, 0.45, 0.55, 0.65, 0.75):
            x = (np.arange(n, dtype=np.float64) + 0.5) / n
            logv = (α - 1.0) * np.log(x) + (α - 1.0) * np.log(1.0 - x)
            v = np.exp(logv)
            v[~np.isfinite(v)] = 0.0
            if v.sum() == 0.0:
                v = np.full(n, 1.0 / n)
            else:
                v /= v.sum()
            starters[f"beta_{α:.2f}"] = v

        # exponential decays
        for lam in (5e-4, 1e-3, 2e-3):
            v = np.exp(-lam * np.arange(n))
            v /= v.sum()
            starters[f"exp_{lam:.4f}"] = v

        # interpolation of the previous best to this length
        x_old = np.linspace(0.0, 1.0, base_n)
        x_new = np.linspace(0.0, 1.0, n)
        interp = np.interp(x_new, x_old, overall_best_seq)
        interp = np.maximum(0.0, interp)
        interp /= interp.sum()
        starters["interp_prev"] = interp

        # evaluate all starters
        for tag, vec in starters.items():
            vec = _project_onto_simplex(vec, s=1.0)
            peak = _fft_self_convolve(vec).max()
            fast_score = 2.0 * n * peak
            if fast_score < overall_best_score:
                overall_best_score = fast_score
                overall_best_seq = vec.copy()
                print(f"[init] n={n:4d} {tag:<12s} fast_score={fast_score:.6f}")

        # b) three‑level parametric family (coarse – cheap)
        tri_seq, tri_score = _three_level_family_search(n, deadline)
        if tri_seq is not None and tri_score < overall_best_score:
            overall_best_score = tri_score
            overall_best_seq = tri_seq.copy()
            print(f"[three‑level] n={n:4d} fast_score={tri_score:.6f}")

    # --------------------------------------------------------------
    # 3.  Adam optimisation of the smooth‑max surrogate
    time_left = deadline - time.time()
    if time_left > 8.0:
        phase_end = time.time() + time_left * 0.35
        a_opt, _ = _adam_optimize(overall_best_seq, phase_end,
                                  init_alpha=1.0,
                                  alpha_factor=1.02,
                                  alpha_max=8000.0,
                                  lr0=0.5)
        cur_score = 2 * len(a_opt) * _fft_self_convolve(a_opt).max()
        if cur_score < overall_best_score:
            overall_best_score = cur_score
            overall_best_seq = a_opt.copy()
        print(f"[adam] after optimisation fast_score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 4.  Exact sub‑gradient descent with back‑tracking line search
    time_left = deadline - time.time()
    if time_left > 6.0:
        sub_end = time.time() + time_left * 0.30
        a_sub = _subgradient_descent(overall_best_seq, sub_end)
        cur_score = 2 * len(a_sub) * _fft_self_convolve(a_sub).max()
        if cur_score < overall_best_score:
            overall_best_score = cur_score
            overall_best_seq = a_sub.copy()
        print(f"[subgrad] after back‑track fast_score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 5.  Gradient‑swap descent – moves mass from high‑gradient to low‑gradient indices
    time_left = deadline - time.time()
    if time_left > 4.0:
        swap_end = time.time() + time_left * 0.30
        a_swap = _gradient_swap_descent(overall_best_seq, swap_end)
        cur_score = 2 * len(a_swap) * _fft_self_convolve(a_swap).max()
        if cur_score < overall_best_score:
            overall_best_score = cur_score
            overall_best_seq = a_swap.copy()
        print(f"[swap] after gradient‑swap fast_score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 6.  Final simulated‑annealing local search (greedy + occasional uphill moves)
    time_left = deadline - time.time()
    if time_left > 2.0:
        a_final = _random_moves_annealing(overall_best_seq, deadline)
        cur_score = 2 * len(a_final) * _fft_self_convolve(a_final).max()
        if cur_score < overall_best_score:
            overall_best_score = cur_score
            overall_best_seq = a_final.copy()
        print(f"[anneal] after SA fast_score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 7.  True black‑box evaluation (the only place we call the provided function)
    try:
        true_score = evaluate_sequence(overall_best_seq.tolist())
        print(f"[final] evaluate_sequence score={true_score:.6f}")
        if true_score < overall_best_score:
            overall_best_score = true_score
    except Exception as exc:            # extremely unlikely, but guard anyway
        print("[final] evaluate_sequence raised:", exc)

    print(f"Best final approx. score {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
