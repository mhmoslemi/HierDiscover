"""
Propose a non‑negative step‑function (a probability vector) that minimises
the objective used by `evaluate_sequence`.  

The algorithm works in three stages:

1. **Coarse scan** – generate a handful of analytic shapes for several
   lengths and keep the best one as a seed.  If a previous solution
   (`height_sequence_1`) exists it is also considered.

2. **Heavy‑weight optimisation** – for the best length we launch many
   random starts (Dirichlet draws with different concentrations) and
   run a long smoothed Adam descent (log‑sum‑exp smooth approximation
   of the max), followed by an exact sub‑gradient descent with momentum
   and finally a short random‑transfer hill‑climber.  All steps respect
   the simplex constraint (non‑negative, sum = 1).

3. **Final polishing** – a last hill‑climb on the overall best vector.

The whole search stops after the allotted time budget (default 1000 s)
and returns the best vector found.
"""

import time
import math
import numpy as np
from scipy import sparse
from scipy.optimize import linprog  # not used now but kept for possible extensions


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _initial_candidates(n: int) -> list[np.ndarray]:
    """A small library of analytic start vectors for a given length."""
    cand = [
        _distribution_uniform(n),
        _distribution_arcsine(n),
        _distribution_triangular(n),
        _distribution_exponential(n, lam=0.003),
        _distribution_exponential(n, lam=0.01),
    ]
    for p in (0.2, 0.5, 1.0, 2.0):
        cand.append(_distribution_beta_sym(n, p))
    # a random sparse start (few non‑zeros)
    rnd = np.zeros(n, dtype=np.float64)
    k = max(2, n // 200)                # about 0.5 % non‑zeros
    idx = np.random.choice(n, k, replace=False)
    rnd[idx] = np.random.rand(k)
    if rnd.sum() > 0:
        rnd /= rnd.sum()
    cand.append(rnd)
    return cand


# ----------------------------------------------------------------------
def _smooth_descent_long(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑sum‑exp smooth approximation of max.
    Runs a fairly long schedule (α up to 5000) and returns the best point.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # schedule: (alpha, steps)
    schedule = [
        (1.0, 2000),
        (5.0, 2000),
        (20.0, 2000),
        (100.0, 2000),
        (500.0, 2000),
        (2000.0, 1500),
        (5000.0, 1000),
    ]

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score
            t += 1

            b = _fft_self_convolve(a)
            maxb = b.max()
            # stable softmax weights
            exp_terms = np.exp(alpha * (b - maxb))
            sum_exp = exp_terms.sum()
            w = exp_terms / sum_exp               # shape (2n‑1,)

            # gradient = 2 * (a reversed convolved with w)
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam update + simplex projection
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a, 1.0)

            if t % 200 == 0:
                cur_score = _max_convolution_score(a)
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_seq = a.copy()
    return best_seq, best_score


def _exact_subgradient_refine_long(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max objective with momentum.
    Runs until time runs out (or a modest iteration limit is reached).
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)

    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.25
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps)

        w = np.zeros_like(b)
        w[active] = 1.0

        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / np.sqrt(iteration)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] subgrad iter={iteration} new best {best_score:.6f}")

        # early termination if progress stalls
        if iteration % 2000 == 0 and time.time() + 0.1 >= deadline:
            break

    return best_seq, best_score


def _local_hill_climb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap random‑transfer hill‑climber.  Only a few thousand trials
    are performed (each trial needs a full convolution, which is why the
    number of iterations is limited).
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)
    cur_score = best_score

    max_transfer = 0.07
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        # exponentially shrinking transfer size
        transfer = max_transfer * (0.5 ** (iteration / 3000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[{time.time():.2f}] hill iter={iteration} improved best {best_score:.6f}")

        if iteration % 5000 == 0 and np.random.rand() < 0.2:
            # tiny jitter to escape tiny plateaux
            pert = np.random.randn(n) * 1e-4
            a = best_seq + pert
            a = _project_onto_simplex(a, 1.0)
            cur_score = _max_convolution_score(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence found (as a Python list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # a small safety margin

    # ------------------------------------------------------------------
    # 1. coarse scan over a few sensible lengths
    lengths = [800, 1200, 2000, 3000, 5000, 8000]
    overall_best_seq = None
    overall_best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        for cand in _initial_candidates(n):
            if time.time() >= deadline:
                break
            cand = cand / cand.sum()
            score = _max_convolution_score(cand)
            if score < overall_best_score:
                overall_best_score = score
                overall_best_seq = cand.copy()
                print(f"[{time.time():.2f}] init n={n} score={score:.6f}")

    # ------------------------------------------------------------------
    # 2. include previously found solution if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.2f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Fallback – should never happen
    if overall_best_seq is None:
        n = max(800, lengths[0])
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _max_convolution_score(overall_best_seq)
        print(f"[{time.time():.2f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. intensive optimisation on the best length found so far
    n = overall_best_seq.shape[0]
    a = overall_best_seq.copy()

    # run a few random restarts (Dirichlet draws with various concentrations)
    conc_list = [0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    restart_budget = (deadline - time.time()) * 0.55   # allocate ~55 % of remaining time
    restart_deadline = time.time() + restart_budget

    while time.time() < restart_deadline:
        # generate a random start
        conc = np.random.choice(conc_list)
        g = np.random.gamma(conc, 1.0, size=n)
        if g.sum() == 0:
            continue
        a0 = g / g.sum()

        # smooth Adam descent
        a_smooth, score = _smooth_descent_long(a0, deadline)
        if score < overall_best_score:
            overall_best_score = score
            overall_best_seq = a_smooth.copy()
            print(f"[{time.time():.2f}] after smooth start score={score:.6f}")

        # exact sub‑gradient refinement
        a_sub, score = _exact_subgradient_refine_long(a_smooth, deadline)
        if score < overall_best_score:
            overall_best_score = score
            overall_best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] after subgrad score={score:.6f}")

        # short hill‑climb on this candidate
        sub_deadline = time.time() + min(4.0, deadline - time.time())
        a_hill, score = _local_hill_climb(a_sub.copy(), sub_deadline)
        if score < overall_best_score:
            overall_best_score = score
            overall_best_seq = a_hill.copy()
            print(f"[{time.time():.2f}] after hill‑climb score={score:.6f}")

    # ------------------------------------------------------------------
    # 4. final hill‑climb on the overall best vector with whatever time is left
    if time.time() < deadline:
        final_deadline = deadline
        a_final, final_score = _local_hill_climb(overall_best_seq.copy(), final_deadline)
        if final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = a_final.copy()
            print(f"[{time.time():.2f}] final hill‑climb best score={final_score:.6f}")

    print(f"=== FINAL BEST:  n={len(overall_best_seq)}  score={overall_best_score:.6f} ===")
    # convert to Python list – evaluate_sequence will apply its own clipping
    return overall_best_seq.tolist()
