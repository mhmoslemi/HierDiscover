The score of a vector `a` (non‑negative, `sum(a)=1`) is  

```
score(a) = 2·n·max( a * a )      (n = len(a))
```

because `evaluate_sequence` first normalises the vector and then computes  
`2·n·max(convolution) / (sum(a)²)`.  
The problem is therefore *“flatten the self‑convolution as much as possible”*.

The algorithm below follows three ideas

1. **Rich set of starts** – for each admissible length we generate many analytic
   families (uniform, arcsine, symmetric β‑distributions, exponential,
   triangular, random Dirichlet).  The best of them is kept as a seed.

2. **Smoothed Adam optimisation in log‑space** –  
   We minimise the smooth surrogate  

```
μ_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
```

   Its gradient is a convolution that can be evaluated with FFTs.
   By working with logits `y` (`a = softmax(y)`) we stay automatically on the
   simplex, so no projection is required.  A continuation on the smoothing
   parameter `α` (1 → 5 → 20 → 100 → 500 → 1000) with many Adam steps drives the
   solution close to a true optimum.

3. **Exact sub‑gradient refinement + cheap hill climbing** –  
   After the smooth phase we switch to the exact sub‑gradient of the
   `max`‑objective (the same routine as the reference code) and finally
   perform a simple random‑transfer hill‑climber that swaps a tiny amount of
   mass between two positions.  All phases respect the 1000 s wall‑clock limit.

The whole search is performed on a few lengths (2000 … 20000).  Longer
vectors give a finer discretisation of the continuous optimal shape, but are
still cheap enough because every iteration uses only FFTs (`O(n log n)`).  
The best vector found (still normalised) is returned as a plain Python list.

```python
"""step‑function optimisation for evaluate_sequence

The algorithm

1.  Generate many analytic starting vectors for several lengths.
2.  Pick the best one and run a long Adam optimiser on the
    smooth log‑sum‑exp surrogate of the max‑convolution.
    The optimiser works in log‑space (soft‑max) → the simplex constraint is
    automatically satisfied.
3.  Refine the result with an exact max‑sub‑gradient method (as in the
    previous SOTA).
4.  Finish with a cheap random‑transfer hill‑climber.
All work is limited by the supplied time budget (default 1000 s).

The returned sequence is a list of non‑negative floats (sum≈1) ready for
evaluate_sequence().
"""

# FFT based convolution utilities

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Exact score 2·n·max(a*a) for a already normalised (sum = 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families (all are normalised to sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on {0,…,n‑1} (density ~ x^{p‑1}(1‑x)^{p‑1})."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Smoothed optimisation (log‑space Adam)
# ----------------------------------------------------------------------
def _smooth_adam_logspace(y0: np.ndarray,
                          schedule: list[tuple[float, int]],
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max‑convolution objective.
    *y0* – logits (a = softmax(y0)).
    *schedule* – list of (alpha, steps) for the continuation.
    Returns the best vector (as a probability vector, sum = 1) and its score.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.15

    best_a = None
    best_score = float("inf")
    t = 0

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_a if best_a is not None else np.exp(y - np.max(y)) / np.exp(y - np.max(y)).sum(), best_score
            t += 1

            # ----- a = softmax(y) -------------------------------------------------
            m_y = np.max(y)
            exp_y = np.exp(y - m_y)                # stable softmax
            a = exp_y / exp_y.sum()

            # ----- self‑convolution ------------------------------------------------
            b = _fft_self_convolve(a)
            maxb = b.max()
            # ----- smooth soft‑max weights ----------------------------------------
            # b – maxb ≤ 0 ⇒ no overflow for large α
            w = np.exp(alpha * (b - maxb))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # ----- gradient wrt a -------------------------------------------------
            conv_rev = _fft_convolve(a[::-1], w)               # length n + len(w) - 1
            grad_a = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

            # ----- gradient wrt logits y (chain rule) ----------------------------
            dot = np.dot(grad_a, a)            # Σ a_i g_i
            grad_y = a * (grad_a - dot)        # ∂L/∂y_i

            # ----- Adam update ----------------------------------------------------
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y * grad_y)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # ----- occasional exact evaluation ------------------------------------
            if t % 500 == 0:
                # recompute a after the update
                m_y = np.max(y)
                exp_y = np.exp(y - m_y)
                cur_a = exp_y / exp_y.sum()
                cur_score = _max_convolution_score(cur_a)
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_a = cur_a.copy()
                    print(f"[{time.time():.1f}] smooth step={t} new best {best_score:.6f}")

    # final evaluation (in case the loop never hit a better point)
    if best_a is None:
        m_y = np.max(y)
        exp_y = np.exp(y - m_y)
        best_a = exp_y / exp_y.sum()
        best_score = _max_convolution_score(best_a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑subgradient refinement (same idea as the reference implementation)
# ----------------------------------------------------------------------
def _exact_subgradient_refine(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0

    while time.time() < deadline:
        it += 1

        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps)

        w = np.zeros_like(b)
        w[active] = 1.0

        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        a = a - eta * velocity
        # project onto the simplex
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {best_score:.6f}")
            if best_score <= 1.503:
                break

        # safety break
        if time.time() + 0.5 > deadline:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Local random‑transfer hill‑climber (very cheap)
# ----------------------------------------------------------------------
def _local_hill_climb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    cur_score = best_score

    max_transfer = 0.07          # at most 7 % of total mass per trial
    it = 0

    while time.time() < deadline:
        it += 1
        # exponentially decaying transfer amount
        transfer = max_transfer * (0.5 ** (it / 4000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[{time.time():.1f}] hill it={it} new best {best_score:.6f}")

        # occasional jitter to escape plateaux
        if it % 5000 == 0 and np.random.rand() < 0.2:
            pert = np.random.randn(n) * 1e-4
            a = best_a + pert
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / a_sum
            cur_score = _max_convolution_score(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The returned list is already normalised (sum≈1) and respects all constraints.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.5   # small safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Scan a handful of lengths with cheap analytic candidates
    # ------------------------------------------------------------------
    lengths = [2000, 5000, 8000, 12000, 16000, 20000]

    overall_best_a = None
    overall_best_score = float("inf")
    overall_best_n = None

    for n in lengths:
        if time.time() >= deadline:
            break
        cand_list = []

        # analytic families
        cand_list.append(_distribution_uniform(n))
        cand_list.append(_distribution_arcsine(n))
        cand_list.append(_distribution_triangular(n))
        cand_list.append(_distribution_exponential(n, lam=0.003))
        cand_list.append(_distribution_exponential(n, lam=0.01))

        # symmetric beta shapes, dense sweep in p
        for p in np.geomspace(0.1, 2.0, num=12):
            cand_list.append(_distribution_beta_sym(n, p))

        # a few random Dirichlet‑like starts
        for _ in range(3):
            exp = np.random.uniform(0.3, 2.0)
            rnd = np.random.rand(n) ** exp
            rnd /= rnd.sum()
            cand_list.append(rnd)

        for cand in cand_list:
            if time.time() >= deadline:
                break
            score = _max_convolution_score(cand)
            if score < overall_best_score:
                overall_best_score = score
                overall_best_a = cand.copy()
                overall_best_n = n
                print(f"[{time.time():.1f}] init n={n} score={score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Fallback (should never be needed)
    # ------------------------------------------------------------------
    if overall_best_a is None:
        n = 2000
        overall_best_a = _distribution_uniform(n)
        overall_best_score = _max_convolution_score(overall_best_a)
        overall_best_n = n
        print(f"[{time.time():.1f}] fallback uniform n={n} score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Main optimisation on the best length found
    # ------------------------------------------------------------------
    a = overall_best_a.copy()
    # initialise logits (log‑space) – small epsilon avoids log(0)
    eps = 1e-14
    y0 = np.log(a + eps)

    # a long continuation schedule (more steps than the reference code)
    schedule = [
        (1.0,   2500),
        (5.0,   2500),
        (20.0,  2500),
        (100.0, 2500),
        (500.0, 2500),
        (1000.0,2500)
    ]

    a, sm_score = _smooth_adam_logspace(y0, schedule, deadline)
    print(f"[{time.time():.1f}] after smooth optimisation score={sm_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Exact max‑subgradient refinement (more iterations)
    # ------------------------------------------------------------------
    a, sub_score = _exact_subgradient_refine(a, deadline)
    print(f"[{time.time():.1f}] after sub‑gradient score={sub_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final cheap hill‑climbing
    # ------------------------------------------------------------------
    a, hill_score = _local_hill_climb(a, deadline)
    print(f"[{time.time():.1f}] final best score={hill_score:.6f}  length={len(a)}")

    # --------------------------------------------------------------
    # 6️⃣  Return the best vector seen (as a plain Python list)
    # --------------------------------------------------------------
    return a.tolist()
```
