"""
Propose a step‑function (list of non‑negative heights) with a low
`evaluate_sequence` score.

Algorithm (smooth‑gradient descent):
* Initialise a probability vector (sum = 1) – by default an arcsine‑type
  distribution, optionally also a random start.
* Repeatedly minimise a smooth approximation of the maximal
  self‑convolution value:
        Lβ(a) = (1/β)·log( Σk exp( β·(a*a)k ) )
  whose gradient can be written in closed form using a convolution.
* The gradient is obtained as
        ∇Lβ = 2·corr( w , a ) ,   w_k = exp(β·b_k)/Σj exp(β·b_j) ,
  where b = a*a (self‑convolution).  Both the convolution and the
  correlation are computed with FFTs (O(n log n)).
* After each gradient step we project back onto the probability simplex
  (non‑negative entries, sum = 1).  Momentum and a decreasing step‑size
  (ηt = η0/√t) are used.  The smoothing parameter β is increased
  gradually (annealing) so the loss converges to the true max‑norm.
* Several lengths and initialisations are tried within the allotted time;
  the best sequence (according to the true black‑box `evaluate_sequence`)
  is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine (Beta(1/2,1/2)) distribution on n points."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _smooth_gradient_step(a: np.ndarray,
                          beta_smooth: float,
                          step_size: float,
                          momentum: np.ndarray,
                          momentum_coeff: float,
                          size_fft: int) -> (np.ndarray, np.ndarray):
    """
    Perform one iteration of the smooth‑gradient descent.

    Returns the updated vector ``a`` and the new momentum term.
    """
    # FFT of current a (padded)
    a_fft = np.fft.rfft(a, n=size_fft)

    # ---- self‑convolution b = a * a (length 2n‑1) ----
    b_fft = a_fft * a_fft
    b_full = np.fft.irfft(b_fft, n=size_fft)
    n = a.shape[0]
    b = b_full[: 2 * n - 1]                     # linear convolution

    # ---- smooth weight vector w ----
    b_max = b.max()
    # subtract the max for numerical stability
    w = np.exp(beta_smooth * (b - b_max))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # ---- gradient = 2 * corr(w, a) ----
    w_fft = np.fft.rfft(w, n=size_fft)
    # correlation via multiplication with conjugate of a_fft
    g_fft = w_fft * np.conj(a_fft)
    g_full = np.fft.irfft(g_fft, n=size_fft)
    # valid part of the correlation (length n)
    g = g_full[n - 1: n - 1 + n]               # corresponds to np.correlate(w, a, 'valid')
    grad = 2.0 * g.real

    # ---- momentum update ----
    velocity = momentum_coeff * momentum + (1.0 - momentum_coeff) * grad

    # ---- gradient step with projection ----
    a_new = a - step_size * velocity
    a_new = _project_onto_simplex(a_new, s=1.0)

    return a_new, velocity, b, b_max
