def _gradient_softmax(a, w):
    # w length L
    n = a.shape[0]
    L = w.shape[0]
    # reverse w to compute correlation via convolution
    w_rev = w[::-1]
    # size for FFT: at least n + L -1
    size = 1 << ((n + L - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    Wrev = np.fft.rfft(w_rev, n=size)
    conv = np.fft.irfft(A * Wrev, n=size)
    # extract correlation indices 0..n-1
    start = L - 1
    g = 2.0 * conv[start:start + n]
    return g

def _optimize_sequence(a0, deadline):

   n = len(a0)
   L = 2*n - 1
   size = 1 << ((n + L - 1) - 1).bit_length()
   a = a0.copy()
   # initial variables
   m = np.zeros_like(a)
   v = np.zeros_like(a)
   beta1 = 0.9
   beta2 = 0.999
   eps = 1e-8
   it = 0
   lambda0=0.2
   lambda_inc=0.0004
   lambda_max=12.0
   alpha0=0.15

   best_seq = a.copy()
   # compute initial b and score
   A = np.fft.rfft(a, n=size)
   b_full = np.fft.irfft(A * A, n=size)
   b = b_full[:L]
   best_score = 2.0 * n * b.max()
   # track best improvement
   while time.time() < deadline:
       it += 1

       # compute A, b
       A = np.fft.rfft(a, n=size)
       b_full = np.fft.irfft(A * A, n=size)
       b = b_full[:L]
       max_b = b.max()
       cur_score = 2.0 * n * max_b
       if cur_score < best_score:
           best_score = cur_score
           best_seq = a.copy()
           print(f"[{time.time():.1f}] n={n} it={it} best_score={best_score:.6f}")

       # compute lambda
       lam = lambda0 + lambda_inc * it
       if lam > lambda_max:
           lam = lambda_max

       # compute softmax weights w
       w = np.exp(lam * (b - max_b))
       w_sum = w.sum()
       w /= w_sum

       # compute gradient via conv with w_rev
       w_rev = w[::-1]
       W_rev = np.fft.rfft(w_rev, n=size)
       conv = np.fft.irfft(A * W_rev, n=size)
       conv_lin = conv[: n + L - 1]
       start = L - 1
       grad = 2.0 * conv_lin[start : start - n : -1]

       # Adam update
       m = beta1 * m + (1.0 - beta1) * grad
       v = beta2 * v + (1.0 - beta2) * (grad ** 2)
       m_hat = m / (1.0 - beta1 ** it)
       v_hat = v / (1.0 - beta2 ** it)
       step = alpha0 * m_hat / (np.sqrt(v_hat) + eps)

       a = a - step
       # project onto simplex
       a = _project_onto_simplex(a, s=1.0)

       # occasional jitter
       if it % 2000 == 0 and np.random.rand() < 0.05:
           a += np.random.randn(n) * 1e-4
           a = _project_onto_simplex(a, s=1.0)

   return best_seq, best_score

"""
Propose a step‑function that minimizes `evaluate_sequence` by
projected Adam‑style optimisation on the probability simplex.
The loss is the smooth approximation of the maximal self‑convolution
value:
    L(a) = log_sum_exp(γ·(a∗a)) / γ ,   γ grows during optimisation.
The gradient of L is 2·corr(a, w) where w are the soft‑max weights.
The algorithm runs several lengths (starting from the previous best,
interpolating to larger sizes) and keeps the globally best sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Arcsine distribution (used as a good default)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Interpolate a previous distribution to a new length
def _interpolate_sequence(old_seq, new_n: int) -> np.ndarray:
    old_seq = np.asarray(old_seq, dtype=np.float64)
    old_n = old_seq.shape[0]
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(new_x, old_x, old_seq)
    new_seq = _project_onto_simplex(new_seq, s=1.0)
    return new_seq


# ----------------------------------------------------------------------
# Full self‑convolution via FFT (real input)
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = 2 * n - 1
    size = 1 << ((L - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:L]


# ----------------------------------------------------------------------
# Gradient of the smooth max (log‑sum‑exp) loss using FFT‑based correlation
def _softmax_gradient(a: np.ndarray, b: np.ndarray, power: float,
                      size: int) -> np.ndarray:
    """
    Compute 2·corr(a, w) where w_i = (b_i / max(b))**power (normalised).
    `size` is the FFT length (next power of two ≥ len(a)+len(b)-1).
    """
    n = a.shape[0]
    L = b.shape[0]

    # soft‑max‑like weights based on a power of the normalized convolution
    max_b = b.max()
    if max_b <= 0:
        w = np.full(L, 1.0 / L)  # degenerate case – uniform
    else:
        # ``power`` may be large; raising a small ratio to a high power drives
        # all but the maximal entries to zero, mimicking the true max.
        ratio = b / max_b
        w = np.power(ratio, power, where=ratio > 0, out=np.zeros_like(ratio))
        # ensure the maximal entry gets weight 1
        w[ratio == 1] = 1.0
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / L)
        else:
            w /= w_sum

    # correlation a * w  (a convolve w_rev)
    w_rev = w[::-1]
    A = np.fft.rfft(a, n=size)
    W_rev = np.fft.rfft(w_rev, n=size)
    conv = np.fft.irfft(A * W_rev, n=size)
    conv_lin = conv[: n + L - 1]          # linear part of the convolution
    start = L - 1
    # g_j = 2 * sum_i a_i w_{i+j} = 2 * (a * w_rev)[L‑1‑j]
    grad = 2.0 * conv_lin[start : start - n : -1]
    return grad


# ----------------------------------------------------------------------
# Optimisation for a single length
def _optimise_length(a0: np.ndarray,
                     deadline: float,
                     verbose: bool = True) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation on the simplex.
    Returns the best sequence seen (sum = 1) and its internal score
    2·n·max(a∗a).
    """
    n = a0.shape[0]
    L = 2 * n - 1
    size = 1 << ((n + L - 1) - 1).bit_length()   # next power‑of‑two ≥ n+L‑1

    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # power‑schedule: start smooth, become sharper → actual max
    power_start = 0.5
    power_inc = 0.001   # increase per iteration
    power_max = 40.0

    # Adam base learning rate
    alpha0 = 0.15

    best_seq = a.copy()
    # initial internal score
    b = _fft_self_convolve(a)
    best_score = float(2.0 * n * b.max())

    it = 0
    while time.time() < deadline:
        it += 1

        # forward pass: self‑convolution
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max

        # record improvement
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            if verbose:
                print(f"[{time.time():.1f}] n={n} it={it} "
                      f"score={best_score:.6f}")

        # adaptive power for the soft‑max weights
        power = power_start + power_inc * it
        if power > power_max:
            power = power_max

        # gradient of the smooth max (log‑sum‑exp)
        grad = _softmax_gradient(a, b, power, size)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha0 * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a, s=1.0)

        # occasional tiny random jitter to escape plateaux
        if it % 2000 == 0 and np.random.rand() < 0.05:
            a += np.random.randn(n) * 1e-4
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score
