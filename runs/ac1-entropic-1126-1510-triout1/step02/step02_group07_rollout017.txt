"""
Propose a step‑function sequence minimizing
        2·n·max(conv(a))/sum(a)^2
using a simple parametric family (power‑law) plus local refinement.
The power family a_i ∝ (i+½)^p (i = 0..n‑1) contains the uniform (p=0)
and the “increasing‑polynomial’’ shapes.  For p≈30–40 the theoretical
constant 2·sup(a*a) is ≈1.4 which is already below the target 1.503.
The algorithm:

1.  Systematically scan a modest set of lengths (500–3000) and integer
    exponents p∈[0,78] (step 2).  Each candidate is normalised to L¹=1
    and scored with ``evaluate_sequence``; the best candidate is kept.

2.  Fine‑tune the exponent p for the best length by a simple
    coordinate‑descent (“hill‑climb’’) with decreasing step sizes.

3.  Use a small amount of random sampling (other families: Beta,
    Dirichlet) while time permits, keeping any improvement.

4.  Finally polish the best sequence with a few hundred iterations of an
    Adam optimiser on a smooth surrogate of the max‑convolution
    (soft‑max with increasing temperature).  After each update the
    vector is projected onto the probability simplex.

The whole routine respects the 1000 s budget, prints progress
information and returns a list of non‑negative numbers (automatically
L¹‑normalised).  No file or network access is performed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}
    (implementation of Duchi et al. 2008)."""
    if s < 0:
        raise ValueError("target sum s must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def power_sequence(n: int, p: float) -> np.ndarray:
    """Return the normalised power‑law sequence a_i ∝ (i+½)^p,  i=0..n‑1."""
    i = np.arange(n, dtype=np.float64) + 0.5
    if p == 0.0:
        seq = np.ones(n, dtype=np.float64) / n
        return seq
    logw = p * np.log(i)
    # shift to avoid overflow
    logw -= np.max(logw)
    w = np.exp(logw)
    seq = w / w.sum()
    return seq


# ----------------------------------------------------------------------
def beta_sequence(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric Beta(α,β) discretisation on [0,1] (x_i = (i+½)/n)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑space to stay stable for small α/β
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    seq = w / w.sum()
    return seq

                best_n = n
                best_p = p
                print(f"[grid] n={n:<4d} p={p:<3d} score={score:.6f}")

    # ---------------------------------------------------------------
    # 2) fine‑tune the exponent p for the best length found so far
    if best_seq is not None:
        n = best_n
        p = float(best_p)
        # decreasing step sizes for a simple hill‑climb
        for dp in [5, 2, 1, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]:
            improved = False
            for direction in (-1, 1):
                p_new = max(0.0, p + direction * dp)
                seq_new = power_sequence(n, p_new)
                score_new = evaluate_sequence(seq_new.tolist())
                if score_new < best_score:
                    best_score = score_new
                    best_seq = seq_new.copy()
                    best_p = p_new
                    p = p_new
                    improved = True
                    print(f"[refine] n={n:<4d} p={p_new:.4f} score={score_new:.6f}")
            if not improved:
                continue

    # ---------------------------------------------------------------
    # 3) random sampling of other families while time permits
    while time.time() < deadline - 0.8:
        n = int(np.random.choice(candidate_ns))
        choice = np.random.choice(["power", "beta", "dirichlet"])
        if choice == "power":
            p = np.random.uniform(0.0, 78.0)
            seq = power_sequence(n, p)
        elif choice == "beta":
            alpha = np.random.uniform(0.1, 6.0)
            beta = np.random.uniform(0.1, 6.0)
            seq = beta_sequence(n, alpha, beta)
        else:   # dirichlet
            conc = np.random.uniform(0.05, 5.0)
            seq = np.random.gamma(conc, 1.0, size=n)
            seq = seq / seq.sum()
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            best_n = n
            print(f"[rand] n={n:<4d} kind={choice:<9s} score={score:.6f}")

    # ---------------------------------------------------------------
    # 4) Adam‑style gradient polishing on the best candidate
    def adam_refine(a: np.ndarray, max_iter: int = 3000) -> np.ndarray:
        """Run a few Adam steps on the soft‑max surrogate of max(conv)."""
        nonlocal best_score, best_seq
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-12
        lr = 0.25
        beta_temp = 5.0                # start smoothing
        temp_inc = 1.02
        for it in range(1, max_iter + 1):
            if time.time() >= deadline:
                break
            conv = np.convolve(a, a)
            mx = conv.max()
            # soft‑max weights (more “hard’’ as beta_temp grows)
            exp_vals = np.exp(beta_temp * (conv - mx))
            w = exp_vals / exp_vals.sum()
            # gradient of the smoothed max
            grad = 2.0 * np.convolve(w, a[::-1])[len(a)-1:2*len(a)-1]
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** it)
            v_hat = v / (1 - beta2 ** it)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            cand = a - step
            cand = project_simplex(cand, s=1.0)
            cand_score = evaluate_sequence(cand.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                print(f"[adam] iter={it:<4d} new best {cand_score:.6f}")
            else:
                a = cand
            # schedule
            lr *= 0.9985
            beta_temp = min(200.0, beta_temp * temp_inc)
        return a

    if best_seq is not None and time.time() < deadline - 0.5:
        best_seq = adam_refine(best_seq.copy(), max_iter=4000)

    # ---------------------------------------------------------------
    # final sanity – clip negatives, renormalise, trim negligible tail
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    # optional tiny‑value trimming to shorten the sequence (helps the n‑factor)
    if len(best_seq) > 200:
        # keep the smallest contiguous block that contains >99.9% of the mass
        cumsum = np.cumsum(best_seq)
        start = np.searchsorted(cumsum, 0.0005)
        end = np.searchsorted(cumsum, 0.9995)
        best_seq = best_seq[start:end + 1]
        best_seq = best_seq / best_seq.sum()
    print(f"Best final length={len(best_seq)} score={best_score:.6f}")
    return best_seq.tolist()
