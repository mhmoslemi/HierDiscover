def _beta_distribution(n, alpha):
    i = np.arange(n, dtype=np.float64)
    # use half-step offset to avoid zero exponent if i=0 or i=n-1
    left = (i + 0.5)
    right = (n - i - 0.5)
    # compute power
    vals = np.power(left, alpha - 1) * np.power(right, alpha - 1)
    # Normalize to sum = 1
    total = vals.sum()
    if total == 0:
        # fallback uniform
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals
```

Note: For alpha < 1, exponent alpha-1 is negative, making values blow near edges but finite due to offset half.

"""
Optimisation of a non‑negative step‑function for the provided
`evaluate_sequence` objective.

Approach
--------
1️⃣  Sweep a parametric family of symmetric Beta‑type distributions
    (discrete analogue of 𝙱𝚎𝚝𝚊(p,p)) on several sequence lengths.
    The parameters (p,q) control the shape: p=q<1 concentrates mass at the
    ends, p=q>1 concentrates mass in the centre.  A small grid and some
    random trials are evaluated using a fast FFT‑based self‑convolution.
    The best candidate (lowest 2·n·max(b)) is retained.

2️⃣  Refine the best candidate with a projected sub‑gradient method
    (the same gradient that the previous solution used) together with
    Nesterov‑style momentum and a diminishing step‑size.
    The iterate is always projected onto the probability simplex
    (non‑negative entries summing to 1).

3️⃣  Occasionally add a tiny random jitter to escape shallow plateaus.

The routine respects the 1000 s wall‑clock budget, prints progress
information and finally returns the best sequence as a plain Python list.
"""

import time

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]                      # descending order
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, p: float, q: float) -> np.ndarray:
    """
    Symmetric (or not) discrete Beta distribution.
    a_i ∝ (i+0.5)^{p‑1}·(n‑i‑0.5)^{q‑1}
    The vector is normalised to sum to 1.
    """
    i = np.arange(n, dtype=np.float64)
    left = i + 0.5
    right = n - i - 0.5

    # guard against overflow/underflow for extreme parameters
    with np.errstate(divide='ignore', invalid='ignore'):
        vals = np.power(left, p - 1.0) * np.power(right, q - 1.0)

    # replace NaNs or infinities by 0 (they arise only when exponent is very
    # negative and the base is 0, which never happens with the +0.5 offset)
    vals = np.nan_to_num(vals, nan=0.0, posinf=0.0, neginf=0.0)

    total = vals.sum()
    if total <= 0.0:
        # fallback to uniform if something went wrong
        return np.full(n, 1.0 / n, dtype=np.float64)

    vals /= total
    # ensure exact simplex feasibility (tiny numerical drift)
    vals = _project_onto_simplex(vals, s=1.0)
    return vals


# ----------------------------------------------------------------------
def _uniform_distribution(n: int) -> np.ndarray:
    """Uniform probability vector of length n."""
    return np.full(n, 1.0 / n, dtype=np.float64)

                        overall_best_seq = mix.copy()
                        overall_best_info = f"n={n} p={p:.3f} q={q:.3f} w={w:.1f}"
                        print(f"🔔 New global best (mix) {overall_best_info} → {mix_score:.6f}")

                if time.time() >= slice_deadline:
                    break
            if time.time() >= slice_deadline:
                break

        # random Beta parameter trials (asymmetric shapes)
        for _ in range(random_trials):
            if time.time() >= slice_deadline:
                break
            p = np.random.uniform(0.2, 2.5)
            q = np.random.uniform(0.2, 2.5)
            a = _beta_distribution(n, p, q)

            b = _fft_self_convolve(a)
            cur_score = 2.0 * n * b.max()
            if cur_score < overall_best_score:
                overall_best_score = cur_score
                overall_best_seq = a.copy()
                overall_best_info = f"n={n} p={p:.3f} q={q:.3f} (rand)"
                print(f"🔔 New global best (rand) {overall_best_info} → {cur_score:.6f}")

        # early exit if we already beat the target
        if overall_best_score <= 1.503:
            break
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # If a historic construction exists, evaluate it as a fallback candidate
    try:
        historic = np.asarray(height_sequence_1, dtype=np.float64)
        historic = _project_onto_simplex(historic, s=1.0)
        hist_score = 2.0 * len(historic) * _fft_self_convolve(historic).max()
        if hist_score < overall_best_score:
            overall_best_score = hist_score
            overall_best_seq = historic.copy()
            overall_best_info = "historic"
            print(f"🔔 New global best (historic) → {hist_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # ----- 2️⃣  LOCAL REFINEMENT (sub‑gradient + momentum) -------------
    if overall_best_seq is None:
        # fallback simple uniform sequence
        n = 2000
        overall_best_seq = _uniform_distribution(n)
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        overall_best_info = "fallback uniform"

    a = overall_best_seq.copy()
    n = a.shape[0]
    best_local_seq = a.copy()
    best_local_score = 2.0 * n * _fft_self_convolve(a).max()

    velocity = np.zeros_like(a)
    beta = 0.9          # momentum parameter
    eta0 = 0.12         # base step size (tuned empirically)

    it = 0
    while time.time() < deadline:
        it += 1

        # 1) current convolution and its maximal entry
        b = _fft_self_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # keep the best seen so far
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_seq = a.copy()
            print(f"[refine] iter={it} score={cur_score:.6f}")

        # 2) sub‑gradient of the active max‑convolution entry
        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]

        # 3) momentum update
        velocity = beta * velocity + (1.0 - beta) * g

        # 4) diminishing step size
        eta = eta0 / np.sqrt(it)

        # 5) gradient step + projection onto simplex
        a = a - eta * velocity
        a = _project_onto_simplex(a, s=1.0)

        # 6) occasional tiny random jitter to escape flat regions
        if it % 3000 == 0:
            jitter = np.random.randn(n) * 1e-4
            a += jitter
            a = _project_onto_simplex(a, s=1.0)

    # ------------------------------------------------------------------
    # Final black‑box evaluation (more reliable than the internal estimate)
    try:
        final_score = evaluate_sequence(best_local_seq.tolist())
        print(f"✅ Final black‑box score = {final_score:.6f}")
        if final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = best_local_seq.copy()
            overall_best_info = "refined"
    except Exception:
        # If the external call fails, keep the internal best
        final_score = best_local_score
        print("⚠️ evaluate_sequence raised an exception; using internal score")

    # ------------------------------------------------------------------
    # Return the best sequence we have found
    print(
        f"🏁 Returning sequence length {len(overall_best_seq)} "
        f"with score {overall_best_score:.6f} (source: {overall_best_info})"
    )
    return overall_best_seq.tolist()
