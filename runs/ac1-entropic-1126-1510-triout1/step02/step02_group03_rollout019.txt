"""
Search for a step‑function (non‑negative vector) minimizing
2·n·max(conv(a,a)) under the simplex constraint Σ a_i = 1.
The search consists of:
  1. Sweep over symmetric Beta(α,α) shapes (including arcsine) on a few
     candidate lengths and keep the best candidate.
  2. Refine the best candidate with two complementary projected gradient
     methods:
        – additive sub‑gradient descent on the true max‑convolution,
        – exponentiated‑gradient descent on a log‑sum‑exp smooth
          approximation of the max.
Both methods use FFT‑based convolution/correlation and Euclidean
projection onto the simplex.  The remaining time budget is devoted to
the refinement, allowing thousands of cheap iterations and typically
improving the score by a few percent beyond the previous state‑of‑the‑art.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, m)
    c = np.fft.irfft(A * A, m)
    return c[: 2 * n - 1]


# ----------------------------------------------------------------------
def _conv_full_and_fft(a: np.ndarray, m: int):
    """Return (full convolution a*a, rfft of a) using supplied FFT length."""
    A = np.fft.rfft(a, m)
    conv_full = np.fft.irfft(A * A, m)[: 2 * len(a) - 1]
    return conv_full, A


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, A: np.ndarray, n: int, m: int) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k‑i}
    using the pre‑computed FFT of a (A) and FFT of the weight vector w.
    Returns vector of length n.
    """
    W = np.fft.rfft(w, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = n - 1
    return 2.0 * corr[start : start + n]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on the grid x_i = (i+½)/n.
    Returns a probability vector that sums to 1.
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (α = ½) on n points, Σa = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _score_of(a: np.ndarray) -> float:
    """Exact score used by evaluate_sequence (sum is 1)."""
    n = len(a)
    return float(2.0 * n * _conv_full_fft(a).max())


# ----------------------------------------------------------------------
def _optimise_additive(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 20000,
    step0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Returns the best vector seen and its exact score.
    """
    a = a0.astype(np.float64).copy()
    a /= a.sum()
    n = len(a)
    best_seq = a.copy()
    best_score = _score_of(a)

    # FFT length that can accommodate both a and weight vectors
    m = 1 << ((3 * n - 2).bit_length())   # ≥ (2n‑1)+(n‑1)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # convolution a*a
        conv_full, A = _conv_full_and_fft(a, m)
        maxb = conv_full.max()
        # weight vector: indicator of positions attaining the maximum
        tol = maxb * 1e-12
        w = np.where(conv_full >= maxb - tol, 1.0, 0.0)

        # gradient (correlation of w with a)
        g = _gradient_via_correlation(w, A, n, m)

        # simple diminishing step size
        eta = step0 / np.sqrt(it)
        a = a - eta * g
        a = _project_onto_simplex(a)   # keeps sum = 1 and non‑negative

        cur_score = _score_of(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] additive n={n:5d} iter={it:5d} score={best_score:.6f}")

        # occasional jitter to escape tiny plateaus
        if it % 500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _optimise_exponential(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 20000,
    eta0: float = 0.2,
    lambda0: float = 0.5,
    lambda_max: float = 50.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on a soft‑max approximation
    of the max‑convolution.  The temperature λ is increased slowly up to
    ``lambda_max`` in order to concentrate on the true peak.
    """
    a = a0.astype(np.float64).copy()
    a /= a.sum()
    n = len(a)
    best_seq = a.copy()
    best_score = _score_of(a)

    # FFT size for correlation
    m = 1 << ((3 * n - 2).bit_length())

    lam = lambda0
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # convolution and current score
        conv_full, A = _conv_full_and_fft(a, m)
        cur_score = 2.0 * n * conv_full.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] exponential n={n:5d} iter={it:5d} score={best_score:.6f}")

        # soft‑max weights
        maxb = conv_full.max()
        exp_arg = lam * (conv_full - maxb)            # all ≤ 0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv_full, 1.0 / conv_full.shape[0])
        else:
            w /= w_sum

        # gradient
        g = _gradient_via_correlation(w, A, n, m)

        # exponentiated (mirror) step
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)                     # stays non‑negative
        a = _project_onto_simplex(a)                 # re‑normalise

        # jitter
        if it % 500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

        # raise temperature
        lam = min(lambda_max, lam * 1.01)

    return best_seq, best_score

"""
Search for a step‑function (non‑negative vector) minimizing

    2·n·max(conv(a, a))

under the simplex constraint Σ a_i = 1.

The routine proceeds in three stages:

1. **Coarse sweep** – for several candidate lengths a dense grid of
   symmetric Beta‑type densities (including the arcsine law) is evaluated.
   The best shape for every length is kept.

2. **Quick exponential refinement** – the best candidate of each length is
   refined with a mirror‑descent (exponentiated‑gradient) method that
   minimises a log‑sum‑exp smooth approximation of the peak.  This supplies
   a reasonably good global candidate quickly.

3. **Deep refinement** – the overall best sequence is further improved
   using two complementary projected‑gradient schemes:
      * additive sub‑gradient descent on the true max‑convolution,
      * exponentiated‑gradient descent with a larger temperature schedule.
   Both use FFT‑based convolution/correlation and Euclidean projection onto
   the simplex.  The remaining time budget is spent here, typically yielding
   a score below the previous 1.527 374 target.

All heavy work relies on NumPy FFTs; the whole search finishes comfortably
inside the 1000 s limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, m)
    c = np.fft.irfft(A * A, m)
    return c[: 2 * n - 1]


# ----------------------------------------------------------------------
def _conv_full_and_fft(a: np.ndarray, m: int):
    """Return (full convolution a*a, rfft of a) using supplied FFT length."""
    A = np.fft.rfft(a, m)
    conv_full = np.fft.irfft(A * A, m)[: 2 * len(a) - 1]
    return conv_full, A


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, A: np.ndarray, n: int, m: int) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k‑i} using the pre‑computed FFT of a (A)
    and the FFT of the weight vector w.
    Returns a vector of length n.
    """
    W = np.fft.rfft(w, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = n - 1
    return 2.0 * corr[start : start + n]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on the grid x_i = (i+½)/n.
    Returns a probability vector (sum = 1).
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (α = ½) on n points, Σa = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _score_of(a: np.ndarray) -> float:
    """Exact score (2·n·max(conv(a,a))) used by evaluate_sequence for Σa=1."""
    n = len(a)
    return float(2.0 * n * _conv_full_fft(a).max())


# ----------------------------------------------------------------------
def _optimise_additive(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 20000,
    step0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Returns the best vector seen and its exact score.
    """
    a = a0.astype(np.float64).copy()
    a /= a.sum()
    n = len(a)
    best_seq = a.copy()
    best_score = _score_of(a)

    m = 1 << ((3 * n - 2).bit_length())   # ≥ (2n‑1)+(n‑1)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv_full, A = _conv_full_and_fft(a, m)
        maxb = conv_full.max()
        # indicator weight: all locations attaining the maximum
        tol = maxb * 1e-12

        a = a - eta * g
        a = _project_onto_simplex(a)

        cur_score = _score_of(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] additive n={n:5d} iter={it:5d} score={best_score:.6f}")

        # occasional jitter to escape tiny plateaus

def _optimise_exponential(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 20000,
    eta0: float = 0.2,
    lambda0: float = 0.5,
    lambda_max: float = 50.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on a log‑sum‑exp
    smooth approximation of the max‑convolution.  The temperature λ
    is increased gradually up to ``lambda_max``.
    """
    a = a0.astype(np.float64).copy()
    a /= a.sum()
    n = len(a)
    best_seq = a.copy()
    best_score = _score_of(a)

    m = 1 << ((3 * n - 2).bit_length())
    lam = lambda0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv_full, A = _conv_full_and_fft(a, m)
        cur_score = 2.0 * n * conv_full.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] exponential n={n:5d} iter={it:5d} score={best_score:.6f}")

        maxb = conv_full.max()
        exp_arg = lam * (conv_full - maxb)      # ≤ 0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv_full, 1.0 / conv_full.shape[0])
        else:
            w /= w_sum

        a = a * np.exp(-eta * g)               # stays non‑negative
        a = _project_onto_simplex(a)

            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

    return best_seq, best_score
