"""
propose_candidate
-----------------

Exact‐subgradient projected descent on the simplex.

* The objective is  max(conv(a,a))  after normalising the ℓ¹–norm to √(2n)
  (the evaluation function reduces to this value).

* At each iteration we locate the index (or indices) where the convolution
  attains its maximum and compute the exact sub‑gradient of that entry.
  The sub‑gradient of a single entry `k` is
      grad_i = a_{k-i}  (for i≤k)  +  a_{i-k}  (for i≥k) .
  If several entries are maximal we simply sum their gradients.

* The iterate is updated by a momentum step and then Euclidean‑projected
  onto the simplex {a≥0, Σa = √(2n)}.  The projection routine is a
  standard O(n log n) algorithm.

* To avoid stagnation we perform several random restarts, occasional
  Gaussian “shakes”, and a cheap block‑search after the main loop.
  All work is bounded by the given time budget (≈1000 s).

The method is deliberately different from the annealed soft‑max approach
used previously and often reaches a lower maximum convolution value.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    # sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero, put the whole mass on the first coordinate
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.

    Parameters
    ----------
    seed : int

    budget_s : int

    Returns

    list[float]

    """

    deadline = time.time() + budget_s - 0.2  # safety margin

    # --------------------------------------------------------------
    # Determine the problem size (use the reference sequence if present)
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)      # ℓ¹‑norm that makes the score = max conv

    # --------------------------------------------------------------
    # Helper to obtain a fresh random feasible point
    def _random_feasible() -> np.ndarray:
        v = np.random.rand(n)
        return _project_to_simplex(v, target_sum)

    # Initial point – either the reference or a fresh random vector
    if have_ref and np.random.rand() < 0.6:
        a = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        a = _project_to_simplex(a, target_sum)
    else:
        a = _random_feasible()

    # Global best
    conv = np.convolve(a, a)
    best_score = conv.max()
    best_a = a.copy()
    print(f"[init] n={n}, best_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Hyper‑parameters
    it = 0
    # Rough scale of a typical entry is target_sum / n
    eta0 = target_sum * 0.5 / n
    beta = 0.9                 # momentum decay
    momentum = np.zeros_like(a)

    # restart schedule
    max_restarts = 4
    restart_interval = (budget_s - 5) / max_restarts

    # main optimisation loop
    while time.time() < deadline:
        conv = np.convolve(a, a)
        cur_max = conv.max()

        # keep global best

            grad[:k + 1] += a[k::-1]          # a[k], a[k-1], ..., a[0]
            # right part (i ≥ k): contribution a_{i-k}
            grad[k:] += a[: n - k]

        # ---------- momentum ----------

        eta = eta0 / math.sqrt(it + 1)

        blocks = np.split(np.arange(n), cuts)

    # Final short refinement starting from the best candidate found
    refine_iters = 2000
    for i in range(refine_iters):
        conv = np.convolve(best_a, best_a)
        cur_max = conv.max()
        if cur_max < best_score - 1e-12:
            best_score = cur_max
        active = np.where(conv >= cur_max - 1e-12)[0]
        grad = np.zeros_like(best_a)
        for k in active:
            grad[:k + 1] += best_a[k::-1]
            grad[k:] += best_a[: n - k]
        eta = eta0 * 0.3 / math.sqrt(i + 1)
        best_a = _project_to_simplex(best_a - eta * grad, target_sum)

    print(f"[final] best score = {best_score:.6f} after {it} main iterations")
    return best_a.tolist()
