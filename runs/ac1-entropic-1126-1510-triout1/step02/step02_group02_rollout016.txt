"""
Propose a low‑score step‑function by combining extensive parametric sampling
with a smooth‑max projected‑Adam refinement.

The algorithm works in three stages:

1. **Broad shape sampling** – several families (inverse‑sqrt, power‑law,
   exponential, cosine‑power, Beta‑PDF, random block, uniform) are sampled
   with many random parameters.  Each candidate is projected onto the simplex
   {a≥0, Σa = 1} and its score (2·n·max(conv)/Σa²) is evaluated.  The best
   few candidates are kept.

2. **Smooth‑max Adam optimisation** – starting from each elite candidate we
   minimise the peak of the self‑convolution using a soft‑max approximation
   to the max.  At iteration t we compute weights w ∝ exp((conv‑max)/T) with a
   slowly decreasing temperature T, obtain the gradient
   ∇ = 2·conv(w, a_rev) (no diagonal correction), and perform an Adam update
   followed by Euclidean projection onto the simplex.  Random jitter is added
   occasionally to escape plateaus.

3. **Final random‑block sweep** – any remaining time is spent generating
   random block‑wise sequences; better ones replace the current best.

The whole search respects the 1000 s budget, always returning the best
sequence found so far.  All numbers are non‑negative; the normalization sum
does not affect the final evaluation score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution using real FFT (output length len(u)+len(v)-1)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full self‑convolution a*a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient_simple(a: np.ndarray, w: np.ndarray,
                             size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    (No diagonal correction; derived analytically as 2·Σ_k w_k·a_{k‑i}).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)          # length = 3n‑2
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative sequence with low evaluation score."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # problem size
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    target_sum = 1.0                         # any positive sum works (invariant)
    size_fft = 1 << ((3 * n - 2).bit_length())   # power of 2 ≥ 3n‑2

    # ------------------------------------------------------------------
    # helpers
    def _score(a: np.ndarray) -> float:
        conv = _self_convolution(a, size_fft)
        return 2.0 * n * conv.max() / (a.sum() ** 2)

    # ------------------------------------------------------------------
    # 1. PARAMETRIC SHAPE SEARCH
    candidates = []

    # uniform
    a = np.ones(n) / n * target_sum
    candidates.append((_score(a), a.copy()))

    # inverse‑sqrt (single‑ended)
    for _ in range(200):
        beta = np.random.rand() * 20.0
        a = 1.0 / np.sqrt(np.arange(n) + beta + 1.0)
        a = _project_to_simplex(a, target_sum)
        candidates.append((_score(a), a.copy()))

    # inverse‑sqrt (double‑ended, symmetric)
    for _ in range(150):
        beta = np.random.rand() * 20.0
        i = np.arange(n)
        a = (1.0 / np.sqrt(i + beta + 1.0) +
             1.0 / np.sqrt((n - 1 - i) + beta + 1.0))
        a = _project_to_simplex(a, target_sum)
        candidates.append((_score(a), a.copy()))

    # power‑law
    for _ in range(150):
        beta = np.random.rand() * 20.0
        p = np.random.rand() * 2.5 + 0.1      # exponent in [0.1, 2.6]
        a = 1.0 / (np.arange(n) + beta + 1.0) ** p
        a = _project_to_simplex(a, target_sum)
        candidates.append((_score(a), a.copy()))

    # exponential decay
    for _ in range(150):
        decay = np.random.rand() * 0.02 + 0.0001   # decay ∈ [1e‑4, 2.1e‑2]
        a = np.exp(-decay * np.arange(n))
        a = _project_to_simplex(a, target_sum)
        candidates.append((_score(a), a.copy()))

    # cosine‑power (large at ends, small centre)
    for _ in range(150):
        p = np.random.rand() * 6.0 + 0.5          # exponent ≥0.5
        a = np.abs(np.cos(np.pi * np.arange(n) / (n - 1))) ** p
        a = _project_to_simplex(a, target_sum)
        candidates.append((_score(a), a.copy()))

    # Beta‑distribution shapes (flexible bell/concave shapes)
    for _ in range(150):
        alpha = np.random.rand() * 5.0 + 0.1
        beta_param = np.random.rand() * 5.0 + 0.1
        x = np.arange(n) / (n - 1)
        a = (x ** alpha) * ((1 - x) ** beta_param) + 1e-9
        a = _project_to_simplex(a, target_sum)
        candidates.append((_score(a), a.copy()))

    # random block candidates (few large constant blocks)
    for _ in range(200):
        nb = np.random.randint(2, 8)                     # 2‑7 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.zeros(n)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        candidates.append((_score(cand), cand.copy()))

    # optionally add the reference sequence (if available)
    if have_ref:
        ref = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        ref = _project_to_simplex(ref, target_sum)
        candidates.append((_score(ref), ref.copy()))

    # keep the best few candidates
    candidates.sort(key=lambda x: x[0])
    best_score, best_a = candidates[0][0], candidates[0][1].copy()
    print(f"[shape search] best initial score = {best_score:.6f}")
    elite = [c[1] for c in candidates[:5]]

    # ------------------------------------------------------------------
    # 2. REFINEMENT WITH SMOOTH‑MAX ADAM
    for run_idx, a0 in enumerate(elite):
        if time.time() >= deadline:
            break
        a = a0.copy()
        # Adam state
        m = np.zeros_like(a)
        v_vec = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        # temperature for soft‑max
        conv = _self_convolution(a, size_fft)
        T = max(1e-4, (conv.max() - conv.min()) * 0.2)
        cooling = 0.99995
        lr0 = 0.3
        it = 0
        max_iters = 500_000

        while time.time() < deadline and it < max_iters:
            it += 1
            conv = _self_convolution(a, size_fft)
            cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

            if cur_score < best_score:
                best_score, best_a = cur_score, a.copy()
                print(f"[run {run_idx}] iter {it} new best = {best_score:.6f}")

            # soft‑max weights
            shifted = conv - conv.max()
            exp_arg = shifted / max(T, 1e-12)
            exp_arg = np.clip(exp_arg, -50, 0)
            w = np.exp(exp_arg)
            w_sum = w.sum()
            if w_sum == 0:
                w.fill(1.0 / w.size)
            else:
                w /= w_sum

            # gradient w.r.t a
            grad = _weighted_gradient_simple(a, w, size_fft)

            # Adam update (minimise → move opposite to gradient)
            m = beta1 * m + (1 - beta1) * grad
            v_vec = beta2 * v_vec + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** it)
            v_hat = v_vec / (1 - beta2 ** it)
            step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

            a = a - step
            a = _project_to_simplex(a, target_sum)

            # temperature cooling
            T *= cooling

            # occasional jitter to escape plateaus
            if it % 5000 == 0:
                a += np.random.randn(n) * target_sum * 5e-4
                a = _project_to_simplex(a, target_sum)

        # final evaluation for this run
        final_sc = _score(a)
        if final_sc < best_score:
            best_score, best_a = final_sc, a.copy()
            print(f"[run {run_idx}] final best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. FINAL RANDOM BLOCK SWEEP (use any remaining time)
    while time.time() < deadline:
        nb = np.random.randint(2, 9)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.zeros(n)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = _score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[random block] new best = {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
