def proj_simplex(v):
    """
    Euclidean projection of vector v onto the probability simplex:
    {x | x_i >= 0, sum x_i = 1}
    Returns projected vector.
    """
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Projected (exponential) gradient descent to flatten self‑convolution.

The evaluation score is   2·n·max_k (a*a)[k]  (since the sum of the
sequence is normalised to 1).  For a uniform vector this score equals 2.
The theoretical optimum is 1 (the average of the convolution), so we
try to make the convolution as flat as possible.

Algorithm
---------
1. Initialise a probability vector `w` (non‑negative, sum = 1).  If the
   global variable ``height_sequence_1`` exists it is used; otherwise a
   uniform vector of length ≈8000 is taken.
2. Repeatedly perform a gradient step that minimises the quadratic loss
   ``||conv(w,w) – 1/(2n‑1)||₂²``:
        * compute the self‑convolution `c = conv(w,w)`;
        * compute the residual `d = c – const`, where the constant is the
          desired uniform value `1/(2n‑1)`;
        * the gradient of the loss w.r.t. `w` is
              g = 2·corr(d, w)    (correlation, mode='valid');
        * update with an exponential (mirror‑descent) step
              w ← w·exp(‑η·g) , then renormalise.
   The learning rate η decays slowly.
3. Keep the best vector seen (according to the true ``evaluate_sequence``
   function) and occasionally accept uphill moves with a tiny simulated‑annealing
   probability to escape shallow local minima.
4. After the gradient phase a short random‑pair polishing stage (mass swaps
   that preserve the simplex) is performed.
5. The best sequence discovered before the time budget expires is returned.

The whole optimisation stops after the allotted time (default 1000 s) and
never performs any file or network I/O.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum x = 1}."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the last index where the condition holds
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random pairwise mass move (preserves sum and non‑negativity)
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Pick two different indices i, j and move a small amount of mass
    (drawn uniformly from [‑σ/2, σ/2]) from one to the other."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    delta = (np.random.rand() - 0.5) * sigma
    if delta > 0:
        delta = min(delta, seq[j])      # take from j
    else:
        delta = -min(-delta, seq[i])    # take from i

    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# Fast score (exactly the value returned by evaluate_sequence for a normalised vector)
# ----------------------------------------------------------------------
def _score_from_vec(vec: np.ndarray) -> float:
    """Return 2·n·max(conv(vec,vec)) – the evaluation score for a normalised vector."""
    n = vec.shape[0]
    conv = np.convolve(vec, vec)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative step heights that (hopefully) minimise
    ``evaluate_sequence`` within the supplied time budget.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a tiny safety margin for the harness
    deadline = start + max(0.0, budget_s - 2.0)

    # ------------------------------------------------------------------
    # 1. Initialise a probability vector (sum = 1)
    # ------------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
        if base.size == 0:
            raise ValueError
        base = np.maximum(base, 0.0)
        if base.sum() == 0.0:
            raise ValueError
        w = base / base.sum()
    except Exception:
        # fallback: a fairly long uniform vector
        n_init = 8000
        w = np.full(n_init, 1.0 / n_init, dtype=float)

    n = w.shape[0]

    # keep the best vector seen so far
    best_vec = w.copy()
    best_score = _score_from_vec(best_vec)

    # ------------------------------------------------------------------
    # 2. Gradient‑descent phase (flatten self‑convolution)
    # ------------------------------------------------------------------
    # target value that the convolution should have everywhere
    target_val = 1.0 / (2 * n - 1)

    # learning‑rate schedule parameters
    eta0 = 0.5          # initial step size
    eta_decay = 5e-5    # decay per iteration
    max_gd_iters = 25000

    it = 0
    current_score = best_score
    while time.time() < deadline and it < max_gd_iters:
        it += 1

        # convolution and gradient
        conv = np.convolve(w, w)                # length 2n‑1
        diff = conv - target_val                # residual (zero‑mean)
        # gradient of the squared error L = ||conv - target||²
        grad = 2.0 * np.correlate(diff, w, mode='valid')   # length n

        # adaptive learning rate
        eta = max(1e-5, eta0 / (1.0 + eta_decay * it))

        # exponential (mirror‑descent) update and re‑normalisation
        w_candidate = w * np.exp(-eta * grad)
        w_candidate_sum = w_candidate.sum()
        if w_candidate_sum == 0.0:
            # numerically unstable – fall back to a tiny random perturbation
            w_candidate = w + 1e-12
            w_candidate_sum = w_candidate.sum()
        w_candidate /= w_candidate_sum

        # evaluate the true score (fast version)
        cand_score = _score_from_vec(w_candidate)

        # keep the globally best vector
        if cand_score < best_score:
            best_score = cand_score
            best_vec = w_candidate.copy()
            print(f"[GD iter {it:05d}] New best score: {best_score:.6f}")

        # simple (annealed) acceptance rule – accept if improving,
        # otherwise accept with a tiny probability that decreases over time.
        if cand_score <= current_score:
            w = w_candidate
            current_score = cand_score
        else:
            # simulated‑annealing style acceptance
            temp = max(0.001, 1.0 - it / max_gd_iters)   # linear cooling
            prob = np.exp(-(cand_score - current_score) / temp)
            if np.random.rand() < prob:
                w = w_candidate
                current_score = cand_score
            # else keep the old w

        # occasional random mass‑swap to inject diversity
        if it % 500 == 0:
            w = _random_pair_move(w, sigma=0.02)
            w /= w.sum()   # re‑normalise (still non‑negative)

    # ------------------------------------------------------------------
    # 3. Short polishing phase (random pair moves)
    # ------------------------------------------------------------------
    polish_iters = 5000
    it = 0
    while time.time() < deadline and it < polish_iters:
        it += 1
        # linear cooling of the perturbation amplitude
        sigma = 0.25 * (1.0 - it / polish_iters) + 0.01
        cand = _random_pair_move(best_vec, sigma)
        cand_score = _score_from_vec(cand)

        if cand_score < best_score:
            best_score = cand_score
            best_vec = cand
            print(f"[Polish iter {it:04d}] New best score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Return the best vector found (as a plain Python list)
    # ------------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)   # safety
    return best_vec.tolist()
