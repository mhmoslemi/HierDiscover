def propose_candidate(seed=42, budget_s=1000):
    import time, numpy as np, math
    # time budget:
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = float('inf')

    # possible lengths
    lengths = [1000, 1500, 2000]

    # attempt arcsine start for each length
    for n in lengths:
        if time.time() >= deadline: break
        seq = np.array(arcsine_sequence(n))
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score, best_seq = score, seq

    # add random restarts
    for trial in range(5):
        if time.time() >= deadline: break
        n = best_seq.shape[0] # use same n
        seq = np.random.rand(n)
        seq /= seq.sum()
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score, best_seq = score, seq

    # optimization loop
    # parameters
    mix_weight = 0.15
    mix_min = 0.01
    mix_decay = 0.95
    step = 0.2
    step_decay = 0.995
    beta = 4.0
    beta_inc = 1.01

    # attempt to import scipy.optimize.linprog
    try:
        from scipy.optimize import linprog
        have_lp = True
    except Exception:
        have_lp = False

    while time.time() < deadline:
        n = best_seq.shape[0]
        # compute conv and metric
        conv = np.convolve(best_seq, best_seq)
        M = conv.max()
        # compute metric using eval for accuracy
        cur_score = evaluate_sequence(best_seq.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = best_seq.copy()

        # decide to do LP if available and enough time left (>2 sec)
        if have_lp and (deadline - time.time() > 2.0):
            # Build A matrix
            # Using Toeplitz: each row k corresponds to conv indices 0..2n-2
            # Instead of building full dense matrix each time, we can use np.zeros then fill
            A = np.zeros((2*n-1, n), dtype=np.float64)
            for i in range(n):
                A[i:i+n, i] = best_seq[i]
            # LP: maximize sum(g) => minimize -1^T g
            c = -np.ones(n, dtype=np.float64)
            # bounds: (0, None)
            bounds = [(0, None) for _ in range(n)]
            # Solve
            res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
            if res.success:
                g_opt = res.x
                sum_g = g_opt.sum()
                if sum_g > 1e-12:
                    # scale to unit sum
                    g_scaled = g_opt / sum_g
                    # mixing
                    new_seq = (1 - mix_weight) * best_seq + mix_weight * g_scaled
                    # project onto simplex
                    new_seq = project_simplex(new_seq, s=1.0)
                    # evaluate
                    new_score = evaluate_sequence(new_seq.tolist())
                    if new_score < best_score:
                        best_score, best_seq = new_score, new_seq
                        print(f"[LP] n={n} improved score={new_score:.6f}")
                    else:
                        # maybe update anyway but with decreased mix_weight
                        best_seq = new_seq
                    # update mixing weight
                    mix_weight = max(mix_min, mix_weight * mix_decay)
                else:
                    # degenerate solution; skip
                    pass
            else:
                # if LP fails, fallback
                have_lp = False
                print("[LP] solve failed, disabling")
            continue  # skip gradient step this iteration

        # gradient step
        # compute w using softmax
        max_val = M
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / exp_vals.sum()
        grad = 2.0 * np.convolve(w, best_seq[::-1])[n-1:2*n-1]
        # gradient descent with momentum maybe
        cand = best_seq - step * grad
        cand = project_simplex(cand, s=1.0)
        cand_score = evaluate_sequence(cand.tolist())

        # decay parameters
        step *= step_decay
        beta = min(200.0, beta * beta_inc)
    # final sanitize and return
    best_seq = np.maximum(best_seq, 0.0)
    best_seq /= best_seq.sum()

"""
Propose a step‑function sequence minimizing
    2·n·max(conv(seq,seq)) / (sum(seq)²)

Algorithm
---------

1. **Initialisation**
   * Try the “arcsine” distribution (C_i = binom(2i,i)/4^i) for a few
     lengths (≈1500‑2500) and keep the best.
   * If a global ``height_sequence_1`` exists it is also examined.
   * A few random Dirichlet starts are tried as fall‑backs.

2. **Alternating optimisation**
   * **LP‑step** – maximise Σg under the linear constraints
       (f * g)[k] ≤ M   (M = max(f*f)) and g ≥ 0.
     The LP is solved with ``scipy.optimize.linprog`` (method='highs').
     The solution g is L1‑normalised and mixed with the current f:
         f ← (1‑t)·f + t·g ,   t starts at 0.2 and decays.
   * **Gradient‑step** – soften the max with a soft‑max weight vector w
     and take a projected gradient descent step on the simplex.
     The step size and the soft‑max temperature β are slowly decreased.

3. **Random perturbations**
   * Occasionally (every ~30 iterations) a small random jitter is added
     to help escape plateaus.

The loop respects a 1000 s budget, constantly records the best score
and returns the corresponding sequence (as a plain Python list)."""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> list[float]:
    """Return the L1‑normalised arcsine coefficients of length n:
       C_0 = 1, C_{i+1}=C_i·(2i+1)/(2(i+1)).
    """
    seq = []
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2.0 * i + 1.0) / (2.0 * (i + 1))
    seq = [x / total for x in seq]
    return seq


# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of vector v onto the simplex {x ≥ 0, Σx = s}.
    Implements the algorithm of Duchi et al. (2008)."""
    if s < 0:
        raise ValueError("Target sum s must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the (2n‑1)×n matrix A such that A @ g = f * g (full convolution)."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column i contributes f[i] on rows i … i+n‑1
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence with minimal evaluation score.
    Returns the best list of coefficients found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_score = float('inf')
    best_n = None

    # --------------------------------------------------------------
    # 1) Initialisation – try several lengths with arcsine
    candidate_lengths = [1500, 1800, 2000, 2200, 2500]
    for n in candidate_lengths:
        if time.time() >= deadline:
            break
        seq = np.asarray(arcsine_sequence(n), dtype=float)
        score = evaluate_sequence(seq.tolist())
        print(f"[init] n={n:<4d}  score={score:.6f}")
        if score < best_score:
            best_score, best_seq, best_n = score, seq, n

    # --------------------------------------------------------------
    # 2) If a historic sequence exists, try it as well
    try:
        global height_sequence_1
        h = np.asarray(height_sequence_1, dtype=float)
        if h.size > 0:
            h = np.maximum(h, 0.0)
            total = h.sum()
            if total > 0:
                h = h / total
                n = h.shape[0]
                score = evaluate_sequence(h.tolist())
                print(f"[global] n={n:<4d}  score={score:.6f}")
                if score < best_score:
                    best_score, best_seq, best_n = score, h, n
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3) Fallback: if nothing succeeded, use a default arcsine of length 2000
    if best_seq is None:
        best_n = 2000
        best_seq = np.asarray(arcsine_sequence(best_n), dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] n={best_n}  score={best_score:.6f}")

    # --------------------------------------------------------------
    # Helper for quick score checks (wrap evaluate_sequence)
    def current_score(a):
        return evaluate_sequence(a.tolist())

    # --------------------------------------------------------------
    # 4) Try to import a linear‑program solver (SciPy)
    try:
        from scipy.optimize import linprog
        have_lp = True
        print("[info] SciPy linprog available – will use LP refinement.")
    except Exception:
        have_lp = False
        print("[info] SciPy linprog NOT available – LP steps will be skipped.")

    # --------------------------------------------------------------
    # Parameters for the alternating scheme
    mix_weight = 0.20          # initial interpolation weight for LP step
    mix_min = 0.01
    mix_decay = 0.95

    grad_step = 0.15           # initial gradient step size
    grad_decay = 0.995
    beta = 4.0                 # soft‑max temperature
    beta_inc = 1.02

    iteration = 0
    last_improve_iter = 0

    # --------------------------------------------------------------
    # Main optimisation loop
    while time.time() < deadline:
        iteration += 1
        n = best_seq.shape[0]

        conv = np.convolve(best_seq, best_seq)
        M = conv.max()
        cur_score = current_score(best_seq)
        if cur_score < best_score:
            best_score, best_seq = cur_score, best_seq.copy()
            best_n = n
            last_improve_iter = iteration
            print(f"[best] iter={iteration:<5d}  new best {cur_score:.6f}")

        # ---- decide whether to run an LP step ----------------------------------------
        time_left = deadline - time.time()
        if have_lp and time_left > 2.5:
            # Build convolution matrix for the current f
            A = build_conv_matrix(best_seq)
            # Linear program: maximise Σ g  <=> minimise -Σ g
            c = -np.ones(n, dtype=np.float64)
            b_ub = np.full(2 * n - 1, M, dtype=np.float64)
            bounds = [(0.0, None) for _ in range(n)]

                print(f"[LP] solve exception: {e}")
                have_lp = False
                continue

            if not res.success or res.x is None:

                have_lp = False

            sum_g = g_opt.sum()
            if sum_g < 1e-12:

                continue

            g_scaled = g_opt / sum_g
            new_seq = (1.0 - mix_weight) * best_seq + mix_weight * g_scaled

            new_score = current_score(new_seq)
            if new_score < best_score:

                best_seq = new_seq

            continue

        grad = 2.0 * np.convolve(w, best_seq[::-1])[n - 1:2 * n - 1]

        cand = project_simplex(cand, s=1.0)

        cand_score = current_score(cand)
        if cand_score < best_score:

            best_seq = cand
            last_improve_iter = iteration
            print(f"[grad] iter={iteration:<5d} new best {cand_score:.6f}")

        grad_step *= grad_decay
        beta = min(200.0, beta * beta_inc)

        # ---- occasional random jitter to escape flat regions -------------------------
        if iteration % 30 == 0 and time.time() < deadline:

            cand2 = np.maximum(cand2, 0.0)
            cand2 = cand2 / cand2.sum()
            cand2_score = current_score(cand2)
            if cand2_score < best_score:
                best_score = cand2_score
                best_seq = cand2
                last_improve_iter = iteration
                print(f"[jitter] iter={iteration:<5d} improved to {cand2_score:.6f}")

        # early stop if no improvement for a long time (optional)
        if iteration - last_improve_iter > 300 and grad_step < 1e-6:

            break

    # --------------------------------------------------------------
    # Final sanitisation
    best_seq = np.maximum(best_seq, 0.0)
    if best_seq.sum() == 0:
        # fallback to uniform if something went wrong
        best_seq = np.full(best_seq.shape[0], 1.0 / best_seq.shape[0])
    else:
        best_seq = best_seq / best_seq.sum()

    print(f"Best final sequence length = {len(best_seq)}  score = {best_score:.6f}")
    return best_seq.tolist()
