def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    # Compute full convolution using FFT.
    n = len(x) + len(y) - 1
    size = 1 << ((n) - 1).bit_length() # next power of two
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    Z = X * Y
    conv = np.fft.irfft(Z, n=size)
    return conv[:n]

"""
Propose a step‑function (list of non‑negative heights) with a low evaluation
score.  The method combines three ideas:

1. **Warm start** – use the historic best sequence (if available) or an
   analytically motivated arcsine distribution.
2. **Smooth max optimisation** – replace the non‑smooth max(b) by a
   temperature‑controlled soft‑max, which yields a dense gradient
   (weighted sum of all convolution shifts).  The temperature is annealed
   during the run.
3. **Simplex parametrisation via softmax + Adam** – the coefficients are
   represented as a softmax of unconstrained “logits’’ `u`.  This guarantees
   non‑negativity and unit sum without an explicit projection.  An Adam
   optimiser (first‑ and second‑moment adaptation) provides a robust step‑size.

Each iteration:
    * Build `a = softmax(u)`.
    * Compute the self‑convolution `b = a * a` (FFT).
    * Form soft‑max weights `w` from `b` with temperature `τ`.
    * Compute the gradient of the smoothed objective
          ∇_a  = 2 * Σ_k w_k a_{k−i}
      via one extra FFT convolution of `a` with the reversed weight vector.
    * Transform the gradient to the logit space and update `u` with Adam.
    * Keep the best sequence (according to the cheap internal objective
      `2·n·max(b)`) and periodically evaluate it with the supplied
      `evaluate_sequence`.

The routine runs for at most `budget_s` seconds, tries a few sequence
lengths and initialisations and finally returns the best list it found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

    elif kind == "arcsine":
        a = _arcsine_distribution(n)
    elif kind == "prev":
        try:
            a = np.asarray(height_sequence_1, dtype=np.float64)

        except Exception:

        else:
            a = a / a.sum()
    else:
        a = _arcsine_distribution(n)

    # ensure exact feasibility
    a = _project_onto_simplex(a, s=1.0)
    return a

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Return a * a using FFT (real input). Length = 2·len(a)‑1."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:n]

            for f in (0.5, 0.75, 1.0, 1.25, 1.5)
            if int(base_n * f) >= 500
        }

        lengths = [8000]

    overall_best_score = float("inf")

        for init_kind in ("prev", "arcsine", "uniform", "random"):
            # initial probability vector

            # logits for the soft‑max parametrisation
            eps = 1e-12

            v = np.zeros_like(u)

            beta2 = 0.999
            eps_adam = 1e-8
            lr0 = 0.05                       # base learning rate

            best_local_seq = None

                max_u = np.max(u)
                a = np.exp(u - max_u)

                u = u - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

                # ----- occasional random jitter to escape plateaux -------
                if it % 1000 == 0 and np.random.rand() < 0.05:
                    u += np.random.randn(n) * 1e-3

                # (no explicit projection needed – soft‑max guarantees feasibility)

            # ------------------------------------------------------------------
            # Final evaluation with the black‑box routine
            try:
                final_score = evaluate_sequence(best_local_seq.tolist())
            except Exception:
                final_score = best_local_score

            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = best_local_seq.copy()
                print(
                    f"*** New global best -> len={n} init={init_kind} "
                    f"score={overall_best_score:.6f}"
                )

            # early exit if we have already beaten the target
            if overall_best_score <= 1.503:
                break

        if time.time() >= deadline or overall_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback – should never be needed but kept for safety
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())

    print(
        f"Final best score: {overall_best_score:.6f} "
        f"(length = {len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
