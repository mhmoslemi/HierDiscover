"""
Gradient‑descent with soft‑max smoothing on the simplex.

The objective that `evaluate_sequence` returns (up to the constant factor
`2*n`) is the L‑∞ norm of the linear convolution of a non‑negative vector
`a` that is normalised to sum = 1.  We therefore try to minimise
``max_k (a * a)[k]`` on the probability simplex.

*  The current max‑convolution is computed each iteration.
*  A soft‑max weighting `w` (parameter β) turns the non‑smooth max into a
   smooth surrogate.  For β≈0 the weight is almost uniform; for large β it
   concentrates on the largest convolution entries.
*  The gradient of the weighted sum is
      ∂/∂a_i  Σ_k w_k (a*a)[k] = 2 Σ_k w_k a_{k‑i},
   which can be obtained by a single linear convolution
   `conv(w, a[::-1])`.
*  The gradient is projected onto the affine space `∑a_i = 0` (by subtracting
   its mean) and an additive step is taken.
*  After the step we clip negatives to zero and renormalise – a cheap
   projection onto the simplex.
*  A small random jitter is injected every few hundred iterations and a
   deterministic “targeted move’’ (as used in the reference implementation)
   is applied periodically to escape plateaux.
*  If no improvement is seen for a long stretch the algorithm restarts from a
   fresh random point (simple stagnation handling).

The loop runs until the time budget (≈ 1000 s) is exhausted and returns the
best sequence discovered.  The fast `_score` function (identical to the real
evaluation apart from the trivial `∑a_i = 1` normalisation) is used for the
inner optimisation; the final list is handed back to the harness where the
full `evaluate_sequence` wrapper is applied.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _clip_and_normalise(v: np.ndarray) -> np.ndarray:
    """Clip negative entries to zero and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        v = np.full_like(v, 1.0 / v.size)
    else:
        v = v / s
    return v


# ----------------------------------------------------------------------
# Fast (exact) score – identical to evaluate_sequence for sum = 1 vectors
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """2 * n * max(conv(seq,seq)) for a probability vector."""
    n = seq.size
    conv = np.convolve(seq, seq)          # linear convolution, length 2n‑1
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Soft‑max weights for the convolution entries
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Return a probability vector over lags, concentrated on large conv values."""
    max_c = conv.max()
    # subtraction prevents overflow, all entries ≤ 1
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        # should not happen, but safe fallback
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    return w


# ----------------------------------------------------------------------
# Gradient of Σ_k w_k (a*a)[k] w.r.t. a (full length n)
# ----------------------------------------------------------------------
def _gradient(seq: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Compute 2 * Σ_k w_k a_{k-i} for all i."""
    # convolution of w with the reversed sequence
    rev = seq[::-1]
    full = np.convolve(w, rev)                 # length = (2n‑1) + n‑1 = 3n‑2
    n = seq.size
    # take the middle slice that corresponds to i = 0 … n‑1
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Deterministic targeted move – moves a tiny amount of mass away from the
# pair that contributes most to the maximal convolution entry.
# ----------------------------------------------------------------------
def _targeted_move(seq: np.ndarray, frac: float = 0.03) -> np.ndarray:
    """Shift a small fraction of mass away from the dominant pair."""
    n = seq.size
    conv = np.convolve(seq, seq)
    k = int(conv.argmax())                     # lag with maximal value

    i_arr = np.arange(n)
    j = k - i_arr
    mask = (j >= 0) & (j < n)
    contrib = np.zeros_like(seq)
    contrib[mask] = seq[i_arr[mask]] * seq[j[mask]]

    if not np.any(mask):
        # very unlikely; fall back to a random jitter
        cand = seq + frac * (np.random.rand(n) - 0.5)
        return _clip_and_normalise(cand)

    i_max = int(np.argmax(contrib))
    j_max = k - i_max

    # move mass from the larger of the two indices
    src = i_max if seq[i_max] >= seq[j_max] else j_max
    dst = np.random.randint(n)
    while dst == src:
        dst = np.random.randint(n)

    amount = frac * seq[src] * np.random.rand()
    cand = seq.copy()
    cand[src] -= amount
    cand[dst] += amount
    return _clip_and_normalise(cand)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Gradient‑descent optimiser for the step‑function problem.

    Returns a list of non‑negative heights that (approximately) minimises
    `evaluate_sequence`.  The algorithm works on the probability simplex,
    uses a soft‑max smoothed surrogate for the L‑∞ convolution norm, and
    periodically performs targeted deterministic moves together with small
    random jitters to avoid stagnation.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for the harness
    deadline = start + max(0.0, budget_s - 0.5)

    # ------------------------------------------------------------------
    # Initialise the problem dimension and starting vector
    # ------------------------------------------------------------------
    try:
        # use the best known construction if it exists
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        seq = _clip_and_normalise(base)          # put on simplex
    except Exception:
        # fallback – a reasonably large dimension (larger dimensions give
        # more freedom to reduce the max‑convolution)
        n = 10000
        seq = np.full(n, 1.0 / n, dtype=float)

    # keep track of the best solution seen so far
    best_seq = seq.copy()
    best_score = _score(best_seq)

    print(f"[Init] n = {n}, initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters
    # ------------------------------------------------------------------
    init_eta = 0.25                 # initial additive step size
    beta = 0.5                      # start of the soft‑max temperature
    beta_max = 40.0
    beta_mul = 1.001                # geometric increase per iteration
    stagnation_limit = 3000        # iterations without improvement → restart
    iter_no_improve = 0
    iteration = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ---- compute current convolution and soft‑max weights -------------
        conv = np.convolve(seq, seq)          # length 2n‑1
        max_conv = conv.max()
        beta = min(beta_max, beta * beta_mul)
        w = _softmax_weights(conv, beta)

        # ---- gradient of the weighted sum (sub‑gradient of the max) -----
        grad = _gradient(seq, w)

        # zero‑mean the gradient so that the additive step does not change the
        # total mass before the projection step
        grad -= grad.mean()

        # annealed step size (slowly decreasing)
        eta = init_eta / np.sqrt(iteration)

        # ---- additive update + projection onto the simplex ----------------
        cand = seq - eta * grad
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            cand = np.full_like(cand, 1.0 / n)
        else:
            cand = cand / s

        # ---- small random jitter every few hundred iterations ------------
        if iteration % 250 == 0:
            jitter = 0.015 * (np.random.rand(n) - 0.5)
            cand += jitter
            cand = np.maximum(cand, 0.0)
            cand = cand / cand.sum()

        # ---- evaluate ----------------------------------------------------
        cand_score = _score(cand)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            iter_no_improve = 0
            print(f"[Iter {iteration:5d}] New best = {best_score:.6f}")

        # ---- occasional targeted deterministic move ----------------------
        if iteration % 60 == 0:
            tm = _targeted_move(best_seq, frac=0.02)
            tm_score = _score(tm)
            if tm_score < best_score:
                best_score = tm_score
                best_seq = tm.copy()
                print(f"[Iter {iteration:5d}] Targeted move improved to {best_score:.6f}")

        # ---- handle stagnation – simple random restart -------------------
        iter_no_improve += 1
        if iter_no_improve > stagnation_limit:
            seq = np.random.rand(n)
            seq = _clip_and_normalise(seq)
            iter_no_improve = 0
            beta = 0.5
            print(f"[Iter {iteration:5d}] Restart due to stagnation")
        else:
            seq = cand

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    print(f"[Done] Best score after {iteration} iterations = {best_score:.6f}")
    return best_seq.tolist()
