        M = float(conv.max())
        # LP step if enough time
        if (deadline - time.time() > 2.0):
            A = np.zeros((2*n-1, n), dtype=float)
            for i in range(n):
                A[i:i+n, i] = best_seq[i]
            c = -np.ones(n)
            bounds = [(0, None)]*n
            b_ub = M*np.ones(2*n-1)
            try:

                    g = res.x
                    sum_g = g.sum()
                    if sum_g > 1e-12:
                        g_scaled = g / sum_g
                        # mix
                        new_f = (1.0 - mix_weight) * best_seq + mix_weight * g_scaled
                        new_f = project_simplex(new_f, s=1.0)

                    f ∗ g ≤ M                (element‑wise)

   using ``scipy.optimize.linprog`` (HiGHS).  The optimum `g` tries to add as
   much ℓ₁‑mass as possible without increasing the maximal convolution value.
   After normalising `g` to unit ℓ₁‑norm we convex‑combine it with `f` using a
   mixing weight `t` that slowly decays.  The new vector is projected back onto
   the probability simplex.

3. **Gradient polishing** – when the LP would take too long (or fails) we
   perform a smooth‑max gradient step.  The soft‑max weights
   `w_k = exp(β·(conv_k−max))/∑exp` give a differentiable approximation of the
   max‑operator and the sub‑gradient of the smoothed objective is
   `grad = 2·(w ∗ a[::-1])`.  After a step we again project onto the simplex.

4. **Book‑keeping** – after every update we compute the true score
   ``evaluate_sequence`` and keep the best sequence seen so far.  The loop runs
   until the 1000 s budget expires.

The LP step is the main source of improvement over the previous implementation
(which lacked a linear‑programming solver).  By repeatedly “filling the slack”
in the convolution constraints we can push the maximal self‑convolution below
the value attained by the arcsine construction, moving the evaluation score
closer to the theoretical optimum.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> list[float]:
    """
    Normalised (L1‑norm = 1) arcsine coefficients.
    C_0 = 1, C_{i+1}=C_i·(2i+1)/(2(i+1)).
    """
    seq = []
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2.0 * i + 1.0) / (2.0 * (i + 1))
    seq = [x / total for x in seq]          # L1‑normalise to 1
    return seq


# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x ≥ 0, Σ x = s}.
    Implements the algorithm of Duchi et al. (2008)."""
    if s < 0:
        raise ValueError("target sum s must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the matrix A with entries A[k,i] = a[i]  iff 0 ≤ k‑i < n.
    For a given vector g, (A @ g)[k] == (a * g)[k].
    """
    n = a.shape[0]
    A = np.zeros((2 * n - 1, n), dtype=float)
    for i in range(n):
        A[i:i + n, i] = a[i]
    return A


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` within the allotted time budget.
    Returns the best list found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = float('inf')
    best_n = None

    # ---------------------------------------------------------------
    # 1) Initialise with a few lengths and two simple shapes
    candidate_lengths = [1500, 1800, 2000, 2200, 2500]
    for n in candidate_lengths:
        if time.time() >= deadline:
            break
        # arcsine
        seq = np.asarray(arcsine_sequence(n), dtype=float)
        sc = evaluate_sequence(seq.tolist())
        print(f"[init‑arc] n={n:<4d}  score={sc:.6f}")
        if sc < best_score:
            best_seq, best_score, best_n = seq.copy(), sc, n

        # uniform
        seq_u = np.full(n, 1.0 / n, dtype=float)
        sc_u = evaluate_sequence(seq_u.tolist())
        print(f"[init‑uni] n={n:<4d}  score={sc_u:.6f}")
        if sc_u < best_score:
            best_seq, best_score, best_n = seq_u.copy(), sc_u, n

    # fallback (should never happen)
    if best_seq is None:
        best_n = 2000
        best_seq = np.asarray(arcsine_sequence(best_n), dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] n={best_n}  score={best_score:.6f}")

    # ---------------------------------------------------------------
    # Helper: recompute true score (list → float)
    def true_score(a: np.ndarray) -> float:
        return evaluate_sequence(a.tolist())

    # ---------------------------------------------------------------
    # Parameters
    mix_weight = 0.2          # initial convex‑combination weight
    mix_min = 0.02
    mix_decay = 0.97

    grad_step = 0.12          # step size for gradient polishing
    grad_decay = 0.995
    beta = 5.0                # soft‑max temperature
    beta_inc = 1.02

    # ---------------------------------------------------------------
    # Main optimisation loop
    while time.time() < deadline:
        n = best_seq.shape[0]

        # -----------------------------------------------------------
        # current convolution and its peak
        conv = np.convolve(best_seq, best_seq)
        M = float(conv.max())

        # -----------------------------------------------------------
        # 2) Linear‑programming refinement (if we have enough time)
        if deadline - time.time() > 1.5:
            try:
                A = build_conv_matrix(best_seq)                # (2n‑1)×n matrix
                c = -np.ones(n)                               # maximise Σ g  → minimise –Σ g
                b_ub = M * np.ones(2 * n - 1)
                bounds = [(0, None)] * n

                res = scipy.optimize.linprog(c,
                                             A_ub=A,
                                             b_ub=b_ub,
                                             bounds=bounds,
                                             method='highs',
                                             options={'presolve': True})
                if res.success and res.x is not None:
                    g = np.asarray(res.x, dtype=float)
                    sum_g = g.sum()
                    if sum_g > 1e-12:
                        g_norm = g / sum_g                     # unit ℓ₁‑norm
                        new_f = (1.0 - mix_weight) * best_seq + mix_weight * g_norm
                        new_f = project_simplex(new_f, s=1.0)

                        # record improvement
                        sc = true_score(new_f)
                        if sc < best_score:
                            best_score, best_seq = sc, new_f.copy()
                            print(f"[LP] n={n:<4d} improved to {sc:.6f}")

                        # update mixing weight and continue to next iteration
                        mix_weight = max(mix_min, mix_weight * mix_decay)
                        continue
            except Exception as e:
                print(f"[LP] exception: {e}")

        # -----------------------------------------------------------
        # 3) Gradient‑descent polishing (soft‑max surrogate)
        max_val = conv.max()
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / exp_vals.sum()
        # gradient of the smoothed max: 2·(w ∗ a[::-1]) restricted to the overlap part
        grad = 2.0 * np.convolve(w, best_seq[::-1])[n - 1:2 * n - 1]

        cand = best_seq - grad_step * grad
        cand = project_simplex(cand, s=1.0)

        cand_score = true_score(cand)
        if cand_score < best_score:
            best_score, best_seq = cand_score, cand.copy()
            print(f"[grad] n={n:<4d} new best {cand_score:.6f}")

        # decay step‑size / increase temperature
        grad_step *= grad_decay
        beta = min(200.0, beta * beta_inc)

    # ---------------------------------------------------------------
    # Final sanity: enforce non‑negativity and normalise
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        # degenerate – fall back to a uniform distribution of the same length
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"Best sequence length = {len(best_seq)}   final score = {best_score:.6f}")
    return best_seq.tolist()
