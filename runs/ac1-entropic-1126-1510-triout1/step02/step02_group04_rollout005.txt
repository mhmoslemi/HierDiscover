def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time, numpy as np, math
    np.random.seed(seed)
    start = time.time()
    deadline = start + max(0, budget_s - 0.5)

    # Determine length and initial a
    try:
        base = np.array(height_sequence_1, dtype=float).copy()
        base = np.maximum(base, 0.0)
        if base.sum() == 0:
            raise Exception
        a = base / base.sum()
        n = a.size
    except Exception:
        n = 8000
        a = np.full(n, 1.0/n, dtype=float)

    # Helper functions
    def full_conv(x):
        # uses FFT
        n= x.shape[0]
        size = 1 << ((2*n - 1).bit_length())
        X = np.fft.rfft(x, size)
        conv = np.fft.irfft(X*X, size)[:2*n-1]
        return conv

    def grad(active, a):
        n = a.shape[0]
        # active length = 2n-1
        size = 1 << ((len(active) + n - 1).bit_length())
        # reversed a
        a_rev = a[::-1]
        F_active = np.fft.rfft(active, size)
        F_a_rev = np.fft.rfft(a_rev, size)
        conv = np.fft.irfft(F_active * F_a_rev, size)[:len(active) + n -1]
        # extract slice at n-1 .. n-1+n-1
        start = n - 1
        g = 2.0 * conv[start:start+n]
        return g

    best_a = a.copy()
    best_max = full_conv(a).max()
    best_score = 2 * n * best_max
    iter_count = 0
    stagnation = 0
    init_eta = 0.7

    while time.time() < deadline:
        iter_count +=1
        conv = full_conv(a)
        cur_max = conv.max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
            best_score = 2 * n * best_max
            stagnation = 0
            if iter_count % 100 == 0:
                print(f"[Iter {iter_count:6d}] New best max={best_max:.6e} score={best_score:.6f}")
        else:
            stagnation +=1

        # compute active mask
        # tolerance relative: 1e-12*cur_max maybe small
        tol = max(cur_max * 1e-12, 1e-14)
        active = (conv >= cur_max - tol).astype(float)

        # gradient
        grad_vec = grad(active, a)
        # step size
        eta = init_eta / math.sqrt(iter_count)

        # exponentiated update
        # to avoid overflow, subtract max
        exp_factor = -eta * (grad_vec - grad_vec.max())
        a = a * np.exp(exp_factor)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0/n, dtype=float)
        else:
            a = a / a_sum

        # occasional random jitter

            jitter = 0.01 * (np.random.rand(n) - 0.5)

            print(f"[Iter {iter_count}] Restart due to stagnation")

    # final best
    print(f"[Done] iter={iter_count}, best score={best_score:.6f}")

"""
Simple exponentiated‑gradient optimiser for the step‑function problem.

The algorithm works on the probability simplex (non‑negative entries,
sum = 1).  In each iteration it:
  * evaluates the full auto‑convolution,
  * builds a binary mask of the (almost) maximal convolution lags,
  * computes the sub‑gradient of the max‑lag functional,
  * performs an exponentiated‑gradient (multiplicative) update that
    moves probability mass away from indices with large gradient,
  * re‑normalises to stay on the simplex.
Occasionally a tiny random jitter is added and a simple stagnation
restart (draw a fresh Dirichlet point) is performed.
The best solution seen during the run is remembered and returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: full linear convolution via FFT (real input → real output)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return np.convolve(x, x) using a real FFT (size is a power of two)."""
    n = x.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, size)
    conv = np.fft.irfft(X * X, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: gradient of the max‑convolution functional.
#   grad_i = 2 * sum_{k active} a_{k-i}
# This is realised as a convolution of the active‑mask with the reversed a.
# ----------------------------------------------------------------------
def _gradient(active: np.ndarray, a: np.ndarray) -> np.ndarray:
    """Compute the sub‑gradient (multiplied by 2) of max_k (a*a)_k."""
    n = a.shape[0]
    # length of the convolution we need: len(active)+n-1 = 3n-2
    size = 1 << ((active.shape[0] + n - 1).bit_length())
    F_a = np.fft.rfft(a[::-1], size)               # a reversed
    F_act = np.fft.rfft(active, size)
    conv = np.fft.irfft(F_act * F_a, size)[: active.shape[0] + n - 1]
    start = n - 1                                   # centre of the result
    grad = 2.0 * conv[start : start + n]
    return grad


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Exponentiated‑gradient optimiser for the step‑function problem.
    Returns a list of non‑negative heights that (hopefully) improves on
    the current best score.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + max(0.0, budget_s - 0.5)

    # ------------------------------------------------------------------
    # Initialise a feasible point on the simplex.
    # ------------------------------------------------------------------
    try:
        # use the previously discovered construction if it exists
        a = np.array(height_sequence_1, dtype=float).copy()
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total == 0.0:
            raise ValueError
        a = a / total
        n = a.size
    except Exception:
        # fallback – a long uniform vector
        n = 8000
        a = np.full(n, 1.0 / n, dtype=float)

    # best solution seen so far
    conv = _full_convolution(a)
    best_max = conv.max()
    best_a = a.copy()
    best_score = float(2.0 * n * best_max)

    # ------------------------------------------------------------------
    # Optimisation loop
    # ------------------------------------------------------------------
    iteration = 0
    no_improve = 0                     # generations without improvement
    init_eta = 0.7                      # initial step size for exp‑gradient

    while time.time() < deadline:
        iteration += 1

        # ----- evaluate current state ----------------------------------
        conv = _full_convolution(a)
        cur_max = conv.max()

        # ----- keep the best -------------------------------------------
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
            best_score = float(2.0 * n * best_max)
            no_improve = 0
            if iteration % 100 == 0:
                print(
                    f"[Iter {iteration:6d}] New best max={best_max:.6e} "
                    f"score={best_score:.6f}"
                )
        else:
            no_improve += 1

        # ----- build active mask (indices that are (almost) maximal) ---
        # tolerance is tiny – enough to capture ties caused by floating error
        tol = max(cur_max * 1e-12, 1e-14)
        active = (conv >= cur_max - tol).astype(float)

        # ----- compute sub‑gradient ------------------------------------
        grad_vec = _gradient(active, a)

        # ----- exponentiated‑gradient step ------------------------------
        # step size decays like 1/sqrt(t) (standard for EG)
        eta = init_eta / math.sqrt(iteration)
        # exponent with shift to avoid overflow (global factor cancels after renorm)
        shift = grad_vec.max()
        exp_factor = -eta * (grad_vec - shift)
        a = a * np.exp(exp_factor)
        s = a.sum()
        if s == 0.0:
            # numerical safety net – fall back to uniform
            a = np.full(n, 1.0 / n, dtype=float)
        else:
            a = a / s

        # ----- occasional random jitter to escape plateaus ------------
        if iteration % 200 == 0:
            jitter = 0.01 * (np.random.rand(n) - 0.5)
            a = np.maximum(a + jitter, 0.0)
            s = a.sum()
            if s == 0.0:
                a = np.full(n, 1.0 / n, dtype=float)
            else:
                a = a / s

        # ----- simple stagnation restart --------------------------------
        if no_improve > 5000:
            a = np.random.rand(n)
            a = a / a.sum()
            no_improve = 0
            print(f"[Iter {iteration:6d}] Restart due to stagnation")

    # ------------------------------------------------------------------
    # Return the best sequence discovered
    # ------------------------------------------------------------------
    print(
        f"[Done] iterations={iteration}, best score={best_score:.6f} "
        f"(max conv={best_max:.6e})"
    )
    return best_a.tolist()
