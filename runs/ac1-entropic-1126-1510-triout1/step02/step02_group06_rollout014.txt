def convolve_self(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:2*n - 1]

def gradient_from_softmax(a, p):
    # a: np.ndarray (n,)
    # p: np.ndarray (2n-1,)
    n = a.shape[0]
    L = 2*n - 1
    # reverse p
    p_rev = p[::-1]
    size = 1 << ((L + n - 1).bit_length())
    P_rev = np.fft.rfft(p_rev, n=size)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(P_rev * A, n=size)
    start = L - 1
    grad = 2.0 * conv[start:start + n]
    return grad
```

def optimize_sequence_ls(init_seq, deadline, n_iters=20000):
    a = np.asarray(init_seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_metric = metric(a)
    lr = 0.2
    alpha = 1.0
    for it in range(n_iters):
        if time.time() >= deadline:
            break
        conv = convolve_self(a)
        max_c = conv.max()
        metric_val = 2.0 * n * max_c
        if metric_val < best_metric:
            best_metric = metric_val
            best_a = a.copy()
        # smoothing
        exp_vals = np.exp(alpha * (conv - max_c))
        p = exp_vals / exp_vals.sum()
        grad = gradient_from_softmax(a, p)
        # update via exponentiated gradient
        a = a * np.exp(-lr * grad)
        a_sum = a.sum()
        if a_sum <= 0 or np.isnan(a_sum):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum
        # adapt parameters
        if it % 500 == 0 and it > 0:
            # reduce lr slowly
            lr = max(lr * 0.9, 1e-4)
            alpha = min(alpha * 1.05, 30.0)
    return best_a.tolist(), best_metric

def propose_candidate(seed=0, budget_s=1000):
    import time
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2

    lengths = [1500, 2500, 3500, 4500]  # adjustable within budget
    alpha_vals = [0.5, 0.7, 1.0, 1.3, 1.6]  # for beta distribution
    best_seq = None
    best_score = float('inf')
    total_lengths = len(lengths)

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break
        per_len_deadline = deadline - (deadline - time.time()) * (total_lengths - idx - 1) / (total_lengths - idx) # not simple, simpler allocate equal portion
        # Simpler: allocate equal portion of remaining time
        remaining = deadline - time.time()
        per_len_budget = remaining / (total_lengths - idx)
        print(f"Optimising length {n}, allocated {per_len_budget:.2f}s")
        # Try different initializations
        init_sequences = []
        init_sequences.append(arcsine_sequence(n))
        init_sequences.append([1.0/n]*n)  # uniform
        for a in alpha_vals:
            init_sequences.append(beta_sequence(n, a))

        for _ in range(2):
            init_sequences.append(np.random.dirichlet(np.ones(n)).tolist())
        # evaluate each initial using quick metric

        per_init_budget = per_len_budget / len(init_sequences)
        for init_idx, init_seq in enumerate(init_sequences):
            if time.time() >= deadline:
                break
            # ensure sum=1
            init_seq = np.array(init_seq, dtype=np.float64)

                best_seq = seq

    # final sanity clipping
    if best_seq is None:
        best_seq = arcsine_sequence(2000)
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]

def gradient_softmax(a, p):

"""
Propose a low‑value step‑function sequence.

The objective of ``evaluate_sequence`` (up to a scale factor) is

    2·n·max_k (a * a)_k    with   sum_i a_i = 1 , a_i ≥ 0

i.e. minimise the peak of the self‑convolution of a probability vector.
We start from several well‑known initial shapes (arcsine, uniform,
Beta(α,α) with a few α values and random Dirichlet draws) and
run a *smooth* exponentiated‑gradient descent that minimises a
soft‑max surrogate of the peak.  The surrogate is

    L_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k)

which converges to the true max when α grows.  Its gradient is a
weighted sum of the usual sub‑gradient and can be computed with a
single FFT‑based convolution.  The update
    a_i ← a_i·exp(−η·∂L/∂a_i)
keeps the vector non‑negative; we renormalise to the simplex after
each step.  The smoothing parameter α and the learning rate η are
adapted during the run.  Several restarts and several sequence lengths
are tried within the 1000 s budget; the best sequence found is returned.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> list[float]:
    """First n coefficients of (1‑z)^‑1/2, normalised to sum 1."""
    seq = []
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    seq = [x / total for x in seq]
    return seq


def beta_sequence(n: int, alpha: float) -> list[float]:
    """
    Beta(α,α) density on the midpoint grid,
    normalised to sum 1 (α>0, α=0.5 gives the arcsine law).
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    total = a.sum()
    if total <= 0 or np.isnan(total):
        a = np.full(n, 1.0 / n)
    else:
        a = a / total
    return a.tolist()


# ----------------------------------------------------------------------
def _metric_of_seq(seq: np.ndarray) -> float:
    """2·n·max(convolution) for a sequence that already sums to 1."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    return 2.0 * n * conv.max()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _gradient_from_softmax(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate.
    a  – current probability vector (length n)
    p  – softmax weights of length 2n‑1, Σ p = 1
    Returns vector of length n with entries 2·Σ_i p_{i+j} a_i .
    """
    n = a.shape[0]
    L = 2 * n - 1
    # reversed softmax weights
    p_rev = p[::-1]
    size = 1 << ((L + n - 1).bit_length())
    P = np.fft.rfft(p_rev, n=size)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(P * A, n=size)
    start = L - 1
    grad = 2.0 * conv[start : start + n]
    return grad


# ----------------------------------------------------------------------
def _optimize_with_smoothing(
    init_seq: list[float],
    deadline: float,
    max_iter: int = 200_000,
) -> tuple[list[float], float]:
    """
    Exponentiated‑gradient descent on the soft‑max surrogate.
    Returns the best sequence (sum = 1) seen before `deadline`
    together with its true metric (2·n·max(conv)).
    """
    a = np.asarray(init_seq, dtype=np.float64)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _metric_of_seq(a)

    # hyper‑parameters
    eta = 0.20          # learning‑rate for the multiplicative update
    alpha = 1.0         # soft‑max smoothing parameter
    no_improve = 0

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self_fft(a)
        max_c = conv.max()
        cur_score = 2.0 * n * max_c

        # keep best ever
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            if it % 500 == 0:
                print(f"[iter {it:6d}] new best metric {best_score:.6f} (n={n})")
        else:
            no_improve += 1

        #--- smooth max -------------------------------------------------
        # soft‑max weights (stable by subtracting the max)
        exp_vals = np.exp(alpha * (conv - max_c))
        p = exp_vals / exp_vals.sum()

        # gradient of the surrogate
        grad = _gradient_from_softmax(a, p)

        # multiplicative (mirror‑descent) step
        a = a * np.exp(-eta * grad)

        # projection back onto the probability simplex
        s = a.sum()
        if s <= 0 or np.isnan(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        #--- schedule ----------------------------------------------------
        if it % 1_000 == 0:
            # increase smoothing, slowly decay the step size
            alpha = min(alpha * 1.05, 30.0)
            eta = max(eta * 0.95, 1e-4)

        # occasional random jitter to escape flat plateaux
        if no_improve % 500 == 0 and no_improve > 0:
            i, j = np.random.choice(n, 2, replace=False)
            delta = (np.random.rand() - 0.5) * 0.002   # ±0.001
            a[i] = max(0.0, a[i] + delta)
            a[j] = max(0.0, a[j] - delta)
            a = a / a.sum()

        # early termination if we are stuck
        if no_improve > 5_000:
            break

    return best_a.tolist(), best_score

    """

    """

    # a tiny safety margin – the caller will deem a timeout as failure
