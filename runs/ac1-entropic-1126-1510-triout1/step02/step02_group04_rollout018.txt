def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.2

    # choose n
    try:
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        a = base / np.sum(base)  # ensure sum = 1
    except Exception:
        n = 12000
        # edge-heavy init
        idx = np.arange(n)
        a = 1.0/(idx+1) + 1.0/(n-idx)
        a = a / a.sum()
    best = a.copy()
    best_score = _score(best)

    # parameters
    eta0 = 0.5
    top_k = 5
    move_frac = 0.03
    iter_cnt=0
    last_best_improve_iter=0

    while time.time() < deadline:
        iter_cnt += 1
        # conv
        conv = np.convolve(a, a)
        # pick top peaks
        # ideally many peaks. Use argpartition
        k_peaks = np.argpartition(-conv, range(top_k))[:top_k]
        # compute subgradient
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in k_peaks:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]
        # step size schedule
        eta = eta0 / np.sqrt(iter_cnt)
        # multiplicative update
        # using mirror descent on simplex: a_i ← a_i * exp(-eta * g_i)
        a = a * np.exp(-eta * g)
        a = a / a.sum()

        # targeted redistribution at each iteration for up to top_k
        for k in k_peaks:
            # find i maximizing a_i * a_{k-i}
            i_arr = np.arange(n)
            j = k - i_arr
            mask = (j >= 0) & (j < n)
            # compute products
            prod = np.empty_like(a)
            prod[:] = 0.0
            prod[mask] = a[i_arr[mask]] * a[j[mask]]
            if prod.max() == 0.0:
                continue
            i_max = int(np.argmax(prod))
            j_max = k - i_max
            # choose source index as larger mass
            src = i_max if a[i_max] >= a[j_max] else j_max
            dst = np.random.randint(n)
            while dst == src:
                dst = np.random.randint(n)
            # transfer fraction of source
            amount = move_frac * a[src] * np.random.rand()
            a[src] -= amount
            a[dst] += amount

        # after modifications renorm
        a = a / a.sum()

        # optionally inject random perturbations if no improvement for many iterations
        if iter_cnt - last_best_improve_iter > 2000:
            # random mixing: new = alpha*best + (1-alpha)*random
            alpha = 0.7
            random_vec = np.random.rand(n)
            random_vec = random_vec / random_vec.sum()
            a = alpha * best + (1-alpha) * random_vec
            a = a / a.sum()
            last_best_improve_iter = iter_cnt  # reset counter

        # evaluate score
        score = _score(a)
        if score < best_score:
            best_score = score
            best = a.copy()
            last_best_improve_iter = iter_cnt
            if best_score < 1.51:
                print(f"[Iter {iter_cnt}] New best {best_score:.6f}")

        # early exit maybe find <1.503 target
        if best_score <= 1.503:
            print(f"[Success] Target reached {best_score:.6f} at iter {iter_cnt}")
            break

    print(f"[Done] Best score = {best_score:.6f} after {iter_cnt} iterations")
    return best.tolist()

"""
Mirror‑descent optimiser for the step‑function problem.

The objective 2·n·max(conv(a,a))/sum(a)^2 is invariant to scaling,
so the algorithm works on the probability simplex (non‑negative,
sum = 1).  At each iteration the lag(s) with the largest convolution
value are identified; the sub‑gradient of the max‑convolution (the sum
of the gradients of a few top peaks) is computed and a mirror‑descent
(step with the exponential map) move is performed, followed by a
small “targeted mass transfer” that explicitly reduces the dominant
pair contributions.  The step size decays as 1/√t, occasional random
perturbations are added to escape stagnation and the best solution
found so far is tracked.  The method starts from the previously
discovered construction (``height_sequence_1``) when available and
falls back to a uniform vector of a moderately large dimension
(~15000) otherwise.

The implementation is pure NumPy, runs in well under the 1000 s
budget, and returns the best sequence as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: project a vector onto the probability simplex (non‑negative,
# sum = 1).
# ----------------------------------------------------------------------
def _clip_and_normalise(v: np.ndarray) -> np.ndarray:
    """Clip negatives to zero and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        v = np.full_like(v, 1.0 / v.size)
    else:
        v = v / s
    return v


# ----------------------------------------------------------------------
# Fast scoring – identical to evaluate_sequence() when the input already
# satisfies the security checks (non‑negative, sum > 0.01).
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    n = seq.size
    conv = np.convolve(seq, seq)           # linear convolution
    max_conv = conv.max()
    # denominator (sum(seq))**2 == 1 because we stay on the simplex
    return float(2.0 * n * max_conv)


# ----------------------------------------------------------------------
# Gradient of the (soft) maximum of several convolution lags.
# For each selected lag k we add the sub‑gradient 2·a_{k‑i}.
# ----------------------------------------------------------------------
def _gradient_for_peaks(seq: np.ndarray, peaks: np.ndarray) -> np.ndarray:
    n = seq.size
    g = np.zeros_like(seq)
    idx = np.arange(n)
    for k in peaks:
        j = k - idx
        mask = (j >= 0) & (j < n)
        # contribution of a_i to conv[k] is 2*a_{k-i}
        g[mask] += 2.0 * seq[j[mask]]
    return g


# ----------------------------------------------------------------------
# Mirror‑descent step on the simplex (exponential map).
# ----------------------------------------------------------------------
def _mirror_descent_step(seq: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """Perform a multiplicative update a_i <- a_i * exp(-eta * grad_i)."""
    # avoid overflow/underflow by clipping the exponent
    exp_factor = np.exp(-eta * grad)
    new_seq = seq * exp_factor
    # guarantee non‑negativity and renormalise
    return _clip_and_normalise(new_seq)


# ----------------------------------------------------------------------
# Targeted move: for each peak lag, move a small fraction of mass from
# the larger of the two indices contributing the largest product
# a_i*a_{k-i} to a random destination.
# ----------------------------------------------------------------------
def _multi_targeted_move(seq: np.ndarray, peaks: np.ndarray, frac: float) -> np.ndarray:
    n = seq.size
    for k in peaks:
        i_arr = np.arange(n)
        j = k - i_arr
        mask = (j >= 0) & (j < n)
        if not mask.any():
            continue
        prod = np.empty_like(seq)
        prod[:] = 0.0
        prod[mask] = seq[i_arr[mask]] * seq[j[mask]]
        if prod.max() == 0.0:
            continue
        i_max = int(np.argmax(prod))
        j_max = k - i_max
        # source = the index that holds the larger amount of mass
        src = i_max if seq[i_max] >= seq[j_max] else j_max
        # destination: any other index
        dst = np.random.randint(n)
        while dst == src:
            dst = np.random.randint(n)
        # transfer up to `frac` of the source mass
        amount = frac * seq[src] * np.random.rand()
        seq[src] -= amount
        seq[dst] += amount
    return seq

            peak_idxs = np.argpartition(-conv, range(peak_k))[:peak_k]
        else:
            peak_idxs = np.arange(conv.size)

        # ----- mirror‑descent step (sub‑gradient of the max) -----
        grad = _gradient_for_peaks(a, peak_idxs)
        eta = eta0 / np.sqrt(iter_cnt)           # diminishing step size
        a = _mirror_descent_step(a, grad, eta)

        # ----- targeted mass redistribution (helps the active set) -----
        a = _multi_targeted_move(a, peak_idxs, frac_move)

        # ----- occasional random jitter to avoid exact stagnation -----
        if iter_cnt % 200 == 0:
            jitter = (np.random.rand(n) - 0.5) * 1e-3
            a = _clip_and_normalise(a + jitter)

        # ----- evaluate current candidate -----
        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:5d}] New best score = {best_score:.6f}")

        # ----- simple stagnation handling -----
        if iter_cnt - last_improve > stagnation_limit:
            # restart from the current best with a small random perturbation
            noise = (np.random.rand(n) - 0.5) * 5e-3
            a = _clip_and_normalise(best + noise)
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:5d}] Restart due to stagnation")

        # ----- early exit if target reached -----
        if best_score <= 1.503:
            print(f"[Success] Target bound reached: {best_score:.6f} at iteration {iter_cnt}")
            break

    total_iters = iter_cnt
    print(f"[Done] Best score after {total_iters} iterations = {best_score:.6f}")
    # Return as plain Python list (the harness will call evaluate_sequence on it)
    return best.tolist()
