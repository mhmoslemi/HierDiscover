"""
LP‑guided Evolution on the Simplex.

We keep a probability vector `f` (sum = 1, non‑negative) and iteratively
reduce the infinity‑norm of its auto‑convolution `c = np.convolve(f, f)`.
Given the current peak `M = max(c)`, the linear programme

    maximise   sum(g_i)
    subject to   (f * g)[k] ≤ M   for all k
                 g_i ≥ 0

has a feasible region that contains directions which move mass from
indices with large `f` to indices that are “quiet’’ for the convolution.
The optimal `g` is found with `scipy.optimize.linprog`; it is then
normalised to sum = 1 and mixed into the current population with a
small step size `t`.  Periodically a few projected‑gradient steps on the
largest convolution peaks are performed to fine‑tune the solution.
If no progress is seen for a while the algorithm restarts from a random
simplex point.  The whole search runs until the 1000 s budget expires
and returns the best sequence discovered.

The score on the simplex equals the value returned by `evaluate_sequence`,
so we can compute it cheaply as `2 * n * max(conv)`.
"""

import time
import numpy as np
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Simplex projection (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Clip negatives and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        v[:] = 1.0 / v.size
    else:
        v /= s
    return v


# ----------------------------------------------------------------------
# Fast scoring on the simplex (equivalent to evaluate_sequence)
# ----------------------------------------------------------------------
def _score_on_simplex(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for a vector that already sums to 1."""
    n = seq.size
    conv = np.convolve(seq, seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Gradient step that reduces the `top_k` largest convolution entries.
# ----------------------------------------------------------------------
def _multi_lag_gradient_step(seq: np.ndarray, eta: float, top_k: int = 3) -> np.ndarray:
    """One projected gradient step aimed at the biggest convolution peaks."""
    n = seq.size
    conv = np.convolve(seq, seq)
    # indices of the k largest entries
    top_idxs = np.argpartition(-conv, range(top_k))[:top_k]

    g = np.zeros_like(seq)
    idx = np.arange(n)
    for k in top_idxs:
        j = k - idx
        mask = (j >= 0) & (j < n)
        g[mask] += 2.0 * seq[j[mask]]

    new_seq = seq - eta * g
    return _project_to_simplex(new_seq)


# ----------------------------------------------------------------------
# Linear programme: maximise the total mass of g while keeping f*g ≤ M
# ----------------------------------------------------------------------
def _solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve
        maximise   sum(g)
        s.t.       (f * g)[k] ≤ M  for all k
                   g ≥ 0
    Returns the optimal g (unnormalised) or None if the solver fails.
    """
    n = f.size
    rows = 2 * n - 1

    # Build the Toeplitz‑like constraint matrix A where
    #   (A @ g)[k] = (f * g)[k]
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        # column j is f shifted down by j positions
        A[j:j + n, j] = f

    c = -np.ones(n, dtype=np.float64)               # maximise sum(g)  → minimise -sum(g)
    b = np.full(rows, M, dtype=np.float64)          # upper bound M for every lag

    # Use the HiGHS solver (fast and reliable for moderate sizes)
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs',
                  options={'presolve': True, 'time_limit': 30})

    if not res.success or res.x is None:
        return None
    return res.x

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    LP‑guided optimiser for the step‑function problem.
    Returns a list of non‑negative heights that (approximately) minimise
    `evaluate_sequence`.  The function respects the 1000 s time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 0.5)   # tiny safety margin

    # --------------------------------------------------------------
    # Initialise dimension and starting point
    # --------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        n = base.size
        f = _project_to_simplex(base)                     # put on simplex
    except Exception:
        # a moderate size gives a good trade‑off between speed and
        # expressive power; we keep it a power of two for nicer indexing.
        n = 2048
        f = np.full(n, 1.0 / n, dtype=float)

    best = f.copy()
    best_score = _score_on_simplex(best)
    print(f"[Init] n = {n}, initial score = {best_score:.6f}")

    # algorithmic parameters
    mix_step = 0.25               # initial convex‑combination weight
    min_step = 0.02
    max_step = 0.35

    # Return the best sequence discovered (as plain Python list)
    # ------------------------------------------------------------------
    print(f"[Done] Best score after {generation} generations = {best_score:.6f}")
    # The evaluator is homogeneous; any positive scaling is fine.
    # We keep the simplex normalisation (sum = 1).
