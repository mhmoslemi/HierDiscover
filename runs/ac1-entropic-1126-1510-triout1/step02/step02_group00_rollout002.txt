"""
Propose a step‑function (list of non‑negative heights) that minimizes
evaluate_sequence by directly smoothing the max‑convolution objective.
The algorithm:

* Initialise a distribution on the simplex (usually the discrete arcsine law)
  – it already yields a nearly flat self‑convolution.
* Optimise the smoothed objective
        f(a) = log‑sum‑exp( λ·(a∗a) )
  using the Adam optimiser.  For λ → ∞ the objective approaches the
  true maximum, but a moderate λ gives a smooth landscape that can be
  descended efficiently.
* The gradient of f can be expressed by a single convolution:
        ∇f_j = (2·λ / Z) * (w ∗ a_rev)_{j+n‑1}
  where w = exp(λ·(a∗a)), Z = Σ w and a_rev is a reversed copy of a.
  All convolutions are computed with an FFT in O(n log n) time.
* After every Adam step the vector is projected back onto the probability
  simplex (non‑negative, sum = 1).  A tiny random jitter is occasionally
  added to escape shallow plateaux.
* Several lengths and initialisations are tried within the 1000 s budget.
  The best sequence found (according to the black‑box evaluate_sequence)
  is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law on {0,…,n‑1}, already normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _initial_sequence(n: int, kind: str) -> np.ndarray:
    """Create a feasible start vector of length n."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif kind == "arcsine":
        a = _arcsine_distribution(n)
    elif kind == "prev":

                raise ValueError
        except Exception:
            a = _arcsine_distribution(n)
    else:
        a = _arcsine_distribution(n)
    return _project_onto_simplex(a, s=1.0)


# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Real FFT based convolution, returning length len(x)+len(y)-1."""
    n = len(x)
    m = len(y)
    size = 1 << ((n + m - 1) - 1).bit_length()   # next power of two
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    Z = X * Y
    conv = np.fft.irfft(Z, n=size)
    return conv[: n + m - 1]
