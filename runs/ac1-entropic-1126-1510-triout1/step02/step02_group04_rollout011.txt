"""
Gradient‑flattening optimizer for the step‑function problem.

The algorithm starts from the best known construction (if available) or
a uniform simplex point.  At each iteration it:

1. Computes the linear convolution a * a by FFT (O(n log n)).
2. Identifies the K largest convolution lags.
3. Builds a sub‑gradient that points in the direction of increasing the
   largest lags (∂(a*a)_ℓ/∂a_i ≈ 2·a_{ℓ‑i} for valid indices).
4. Takes a small projected step opposite to this gradient, clips to the
   non‑negative simplex and evaluates the new score.
5. Accepts the move if it improves the objective, otherwise it may be
   accepted with a tiny temperature‑dependent probability (simulated
   annealing) to escape shallow local minima.
6. Every few iterations a targeted “mass‑transfer’’ move is tried –
   it moves a fraction of mass from the pair that contributes most to the
   current maximal convolution value to a random low‑mass index.

The step size decays slowly, while a small random perturbation is applied
occasionally to maintain diversity.  The loop runs until the time budget
expires (≈ 1000 s) and the best sequence found is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: project onto the probability simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _clip_and_normalise(v: np.ndarray) -> np.ndarray:
    """Clip negatives to zero and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        v = np.full_like(v, 1.0 / v.size)
    else:
        v /= s
    return v


# ----------------------------------------------------------------------
# FFT‑based linear convolution (real inputs)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using zero‑padded FFT of length `fft_len`."""
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    # Numerical noise may produce tiny negative values
    conv_full[conv_full < 0.0] = 0.0
    return conv_full


# ----------------------------------------------------------------------
# Compute the evaluation score (2 n · max(convolution)).
# The sum of the vector is assumed to be 1, therefore the denominator is 1.
# ----------------------------------------------------------------------
def _score_from_conv(conv: np.ndarray, n: int) -> float:
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Targeted move – tries to lower the current maximal convolution entry.
# ----------------------------------------------------------------------
def _targeted_move(a: np.ndarray, conv: np.ndarray, frac: float = 0.02) -> np.ndarray:
    n = a.size
    lag = int(conv.argmax())

    # contribution = a[i] * a[lag‑i] for all feasible i
    i_arr = np.arange(n)
    j = lag - i_arr
    mask = (j >= 0) & (j < n)
    contrib = np.zeros_like(a)
    contrib[mask] = a[i_arr[mask]] * a[j[mask]]

    if not np.any(mask):
        # fallback – small random jitter
        cand = a + frac * (np.random.rand(n) - 0.5)
        return _clip_and_normalise(cand)

    i_max = int(np.argmax(contrib))
    j_max = lag - i_max
    src = i_max if a[i_max] >= a[j_max] else j_max

    # destination – pick a low‑mass index (random, but avoid src)
    dst = np.random.randint(n)
    while dst == src:
        dst = np.random.randint(n)

    # move a small fraction of the source mass
    amount = frac * a[src] * np.random.rand()
    cand = a.copy()
    cand[src] -= amount
    cand[dst] += amount
    return _clip_and_normalise(cand)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Gradient‑flattening optimiser for the step‑function problem.
    Returns a list of non‑negative heights that (approximately) minimises
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a small safety margin for the harness
    deadline = start + max(0.0, budget_s - 0.5)

    # --------------------------------------------------------------
    # Initialise the sequence
    # --------------------------------------------------------------
    try:
        # Use the previously discovered construction if present
        base = np.array(height_sequence_1, dtype=float)
    except Exception:
        # Fallback – a uniform simplex point of moderate length
        n = 8000
        base = np.full(n, 1.0 / n, dtype=float)

    n = base.size
    a = _clip_and_normalise(base)            # current iterate (sum = 1)

    # Pre‑compute FFT length (next power of two ≥ 2 n‑1)
    fft_len = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # Initial evaluation
    # --------------------------------------------------------------
    conv_a = _conv_fft(a, fft_len)
    best = a.copy()
    best_score = _score_from_conv(conv_a, n)
    best_conv = conv_a.copy()
    print(f"[Init] n = {n}, initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------------
    top_k = 30                 # number of largest lags used for the gradient
    base_step = 0.15           # target maximal change per iteration
    step_decay = 2e-5          # linear decay of the step size
    rand_move_prob = 0.012     # probability of a pure random perturbation
    iter_since_improve = 0
    max_stall = 1500           # restart if no improvement for many iterations

    iteration = 0

    # --------------------------------------------------------------
    # Optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ------------------------------------------------------------------
        # Evaluate current iterate
        # ------------------------------------------------------------------
        conv_a = _conv_fft(a, fft_len)
        cur_score = _score_from_conv(conv_a, n)

        # Update global best
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            best_conv = conv_a.copy()
            iter_since_improve = 0
            print(f"[Iter {iteration:5d}] New best score = {best_score:.6f}")

        else:
            iter_since_improve += 1

        # ------------------------------------------------------------------
        # Gradient step based on the K largest convolution lags
        # ------------------------------------------------------------------
        top_idxs = np.argpartition(-conv_a, top_k)[:top_k]

        g = np.zeros_like(a)
        idx = np.arange(n)
        for lag in top_idxs:
            j = lag - idx
            mask = (j >= 0) & (j < n)
            g[idx[mask]] += 2.0 * a[j[mask]]          # sub‑gradient approx.

        # Determine a step size that limits the maximal change in any entry.
        # The target maximum change shrinks slowly with the iteration count.
        target_max_change = max(base_step - step_decay * iteration, 1e-4)
        max_grad = np.max(np.abs(g)) + 1e-16
        step = target_max_change / max_grad

        cand = a - step * g
        cand = _clip_and_normalise(cand)

        cand_score = _score_from_conv(_conv_fft(cand, fft_len), n)

        if cand_score < cur_score:
            a = cand
            continue  # successful descent, skip other moves this iteration

        # ------------------------------------------------------------------
        # Occasional targeted move (exploits the most active lag)
        # ------------------------------------------------------------------
        if iteration % 9 == 0:
            cand2 = _targeted_move(a, conv_a, frac=0.015)
            cand2_score = _score_from_conv(_conv_fft(cand2, fft_len), n)
            if cand2_score < cur_score:
                a = cand2
                continue

        # ------------------------------------------------------------------
        # Small random perturbation (exploration)
        # ------------------------------------------------------------------
        if np.random.rand() < rand_move_prob:
            perturb = 0.02 * (np.random.rand(n) - 0.5)
            cand3 = a + perturb
            cand3 = _clip_and_normalise(cand3)
            cand3_score = _score_from_conv(_conv_fft(cand3, fft_len), n)
            if cand3_score < cur_score:
                a = cand3
                continue

        # ------------------------------------------------------------------
        # Simulated‑annealing style fallback: accept worsening move with tiny prob.
        # ------------------------------------------------------------------
        # Temperature decays linearly to zero near the end of the budget.
        remaining = max(0.0, deadline - time.time())
        total = max(1e-12, deadline - start)
        temperature = remaining / total
        if temperature > 0.0:
            prob = np.exp(-(cand_score - cur_score) / (temperature * 0.05 + 1e-12))
            if np.random.rand() < prob:
                a = cand
                continue

        # ------------------------------------------------------------------
        # Stagnation restart (inject diversity)
        # ------------------------------------------------------------------
        if iter_since_improve > max_stall:
            a = _clip_and_normalise(np.random.rand(n))
            iter_since_improve = 0
            print(f"[Iter {iteration:5d}] Restart due to stagnation")
            continue

    # --------------------------------------------------------------
    # Return the best sequence discovered
    # --------------------------------------------------------------
    print(f"[Done] Best score after {iteration} iterations = {best_score:.6f}")
    return best.tolist()
