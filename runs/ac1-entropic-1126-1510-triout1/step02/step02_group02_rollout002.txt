"""
Hybrid shape‑search + Adam‑based smoothed max‑convolution optimizer.

1. **Shape search** – sample a large collection of simple parametric families
   (inverse‑sqrt, power‑law, double‑ended, exponential, cosine, mixtures and
   random block patterns).  Every candidate is projected onto the simplex
   {a_i ≥ 0, Σ a_i = √(2 n)} and its exact objective  max(a*a) is computed via
   FFT.  The best few candidates are retained.

2. **Refinement** – starting from the best candidate we minimise a smooth
   approximation of the max‑convolution using a soft‑max of the convolution
   entries.  The gradient of the weighted sum of convolution values is
   obtained analytically (cross‑correlation) and the parameters are updated
   with the Adam optimiser.  After each step the vector is projected back onto
   the simplex.  The temperature of the soft‑max is slowly annealed, turning
   the smooth objective into the true max.

3. **Final random block sweep** – after the time budget expires we perform a
   short random‑block search to possibly escape a local minimum.

All operations respect the security checks performed by `evaluate_sequence`,
and the function returns the best sequence found before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σ x = z}."""
    n = v.shape[0]
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries would be zero, put the whole mass on the first coordinate
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using FFT (real input)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Return the full self‑convolution a*a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """Gradient of Σ_k w_k·(a*a)_k w.r.t. a (weights w are fixed)."""
    # cross‑correlation of w with a can be obtained as convolution of w with rev(a)
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)          # length = 3n‑2
    start = a.shape[0] - 1                         # slice aligning with indices 0…n‑1
    grad = 2.0 * conv[start:start + a.shape[0]]
    return grad


# ----------------------------------------------------------------------
def _score(a: np.ndarray, size_fft: int, n: int) -> float:
    """Exact objective value max(a*a) (equivalent to evaluate_sequence)."""
    conv = _self_convolution(a, size_fft)
    return float(conv.max())

            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)
            # reset Adam moments after a stochastic jump
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[{it:07d}] random perturbation")

        a = a_new
        # geometric cooling of temperature
        T *= 0.9995
        # keep temperature bounded away from zero
        if T < 1e-8:
            T = 1e-8

    # ------------------------------------------------------------------
    # 3. FINAL RANDOM BLOCK SEARCH (short budget)
    remaining = deadline - time.time()
    if remaining > 5.0:
        extra_iters = 200
        print("[final] random block sweep")
        for _ in range(extra_iters):
            nb = np.random.randint(2, 7)
            cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = np.random.rand(nb) + 0.1
            cand = np.zeros(n)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, target_sum)
            sc = _score(cand, size_fft, n)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
                print(f"[block] improved to {best_score:.6f}")

    print(f"[done] best score = {best_score:.6f}")
    return best_a.tolist()
