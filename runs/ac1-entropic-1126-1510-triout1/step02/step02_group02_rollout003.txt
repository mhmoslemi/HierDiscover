"""
Propose a step‑function sequence with a low evaluation score.

The algorithm consists of three stages:

1. **Broad parametric shape search** – many families (inverse‑sqrt,
   power‑law, exponential, logistic, cosine, mixtures, random blocks)
   are sampled, each candidate is projected onto the simplex
   Σa = √(2 n) and its true score (max convolution) is evaluated.
   The best candidate is retained.

2. **Adam‑based smooth max‑gradient descent** – starting from the best
   shape, we minimise the maximal convolution value M = max(a⋆a)
   using a soft‑max approximation of the max (temperature‑controlled)
   and Adam updates, with Euclidean projection onto the simplex after
   each step.  The temperature is slowly annealed, turning the smooth
   objective into the true max.  Random perturbations are injected
   periodically to escape plateaus.

3. **Final random‑block refinement** – a quick extra search over
   random block‑wise sequences to catch any missed improvement.

All operations respect the security checks of `evaluate_sequence`
(the optimiser works with non‑negative numbers and a fixed sum, so
the final score is invariant under scaling).  The function returns the
best sequence found before the 1000 s budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using FFT (output length len(u)+len(v)-1)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full convolution a * a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    w must sum to 1 and be non‑negative.
    """
    n = a.shape[0]
    # reverse a for convolution
    a_rev = a[::-1]
    # w * a_rev convolution
    conv = _conv_fft(w, a_rev, size_fft)      # length = 2n-1 + n-1 = 3n-2
    start = n - 1
    grad = 2.0 * conv[start:start + n]        # 2·Σ_k w_k a_{k-i}
    # subtract diagonal over‑count (i = k-i) which should be counted once
    diag_w = w[0:2 * n - 1:2]                # elements w_{2i}, length n
    grad -= a * diag_w
    return grad


# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, temperature: float) -> np.ndarray:
    """Soft‑max (log‑sum‑exp) weights for a convolution vector."""
    # shift for numerical stability
    shifted = conv - conv.max()
    # prevent division by zero
    T = max(temperature, 1e-12)
    exp_arg = shifted / T
    exp_arg = np.clip(exp_arg, -60, 0)      # avoid overflow
    w = np.exp(exp_arg)
    total = w.sum()
    if total == 0.0:
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= total
    return w


# ----------------------------------------------------------------------
def _score_from_conv(conv: np.ndarray) -> float:
    """Return the value of the evaluation function (with sum already normalised)."""
    # after normalisation Σa = √(2 n) ⇒ denominator = 2 n, score == max(conv)
    return float(conv.max())


# ----------------------------------------------------------------------
def _generate_candidate(shape_func, params, n: int, target_sum: float) -> np.ndarray:
    """Create a raw candidate by calling shape_func(params, n) and project."""
    raw = shape_func(params, n)
    proj = _project_to_simplex(raw, target_sum)
    return proj

# ----- shape families ---------------------------------------------------
def _shape_uniform(_: dict, n: int) -> np.ndarray:
    return np.ones(n)


def _shape_inv_sqrt(params: dict, n: int) -> np.ndarray:
    beta = params["beta"]
    i = np.arange(n)
    return 1.0 / np.sqrt(i + beta + 1.0)


def _shape_double_inv_sqrt(params: dict, n: int) -> np.ndarray:
    beta = params["beta"]
    i = np.arange(n)
    left = 1.0 / np.sqrt(i + beta + 1.0)
    right = 1.0 / np.sqrt((n - 1 - i) + beta + 1.0)
    return left + right


def _shape_power_law(params: dict, n: int) -> np.ndarray:
    beta = params["beta"]
    p = params["p"]
    i = np.arange(n)
    return 1.0 / (i + beta + 1.0) ** p


def _shape_exponential(params: dict, n: int) -> np.ndarray:
    decay = params["decay"]
    i = np.arange(n)
    return np.exp(-decay * i)


def _shape_cosine(params: dict, n: int) -> np.ndarray:
    p = params["p"]
    i = np.arange(n)
    return np.abs(np.cos(np.pi * i / (n - 1))) ** p


def _shape_logistic(params: dict, n: int) -> np.ndarray:
    alpha = params["alpha"]
    shift = params["shift"]
    i = np.arange(n, dtype=np.float64)
    return 1.0 / (1.0 + np.exp(alpha * (i - shift)))


def _shape_mixture_uniform_inv(params: dict, n: int) -> np.ndarray:
    beta = params["beta"]
    lam = params["lam"]
    inv = 1.0 / np.sqrt(np.arange(n) + beta + 1.0)
    return lam * np.ones(n) + (1.0 - lam) * inv


def _shape_two_power(params: dict, n: int) -> np.ndarray:
    """a_i = (i+beta)^(-p) * (1 - i/(n‑1))^q  (both ends decay)."""
    beta = params["beta"]
    p = params["p"]
    q = params["q"]
    i = np.arange(n)
    left = (i + beta + 1.0) ** (-p)
    right = (1.0 - i / (n - 1.0)) ** q
    return left * right


def _shape_random_blocks(params: dict, n: int) -> np.ndarray:
    nb = params["nb"]
    cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
    blocks = np.split(np.arange(n), cuts)
    vals = np.random.rand(nb) + 0.1
    cand = np.zeros(n)
    for blk, v in zip(blocks, vals):
        cand[blk] = v
    return cand


def _shape_random_spikes(params: dict, n: int) -> np.ndarray:
    m = params["m"]
    positions = np.random.choice(n, m, replace=False)
    heights = np.random.rand(m) + 0.1
    cand = np.zeros(n)
    cand[positions] = heights
    return cand

        print(f"[shape] after {shape_func.__name__} best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. REFINEMENT VIA ADAM‑BASED SMOOTH MAX GRADIENT DESCENT
    a = best_a.copy()
    conv = _self_convolution(a, size_fft)
    # initialise temperature as a fraction of the spread
    T = max(1e-4, (conv.max() - conv.min()) * 0.5)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # base learning rate – proportional to the typical magnitude of a
    lr0 = 0.6 * target_sum / n
    lr = lr0

    it = 0
    no_improve = 0
    max_iters = 5_000_000   # upper bound; will be cut by the time budget

    while time.time() < deadline and it < max_iters:
        conv = _self_convolution(a, size_fft)
        cur_score = float(conv.max())

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[{it:07d}] new best = {best_score:.6f}")
        else:
            no_improve += 1

        # ------- soft‑max weights (smooth approximation of max) -------
        w = _softmax_weights(conv, T)

        # ------- gradient of weighted sum of convolutions -------------
        grad = _weighted_gradient(a, w, size_fft)

        # ------- Adam update -----------------------------------------
        it1 = it + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it1)
        v_hat = v / (1.0 - beta2 ** it1)

        step = - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a + step

        # project back onto simplex (non‑negative & Σ = target_sum)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a_new, target_sum / n)
        else:
            a_new = a_new * (target_sum / s)

        # occasional random perturbation to escape plateaus
        if (it % 4000 == 0 and it > 0) or no_improve > 4000:
            noise = np.random.randn(n) * target_sum * 5e-4
            a_new = np.maximum(a_new + noise, 0.0)
            a_new = a_new * (target_sum / a_new.sum())
            no_improve = 0
            print(f"[{it:07d}] random perturbation")

        a = a_new
        # cool temperature very slowly
        T *= 0.99995
        # slowly decay learning rate
        lr = lr0 / (1.0 + 1e-6 * it)

        it += 1

    # --------------------------------------------------------------
    # 3. FINAL RANDOM BLOCK SEARCH (quick local improvement)
    for _ in range(300):
        nb = np.random.randint(2, 7)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.zeros(n)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[block] improved to {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f} after {it} iterations")
    return best_a.tolist()
