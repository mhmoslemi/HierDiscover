   cssv = np.cumsum(u) - 1

   cond = u - cssv / ind > 0

   w = np.maximum(v - theta, 0)
   return w

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Optimise the step‑function heights using a smoothed max‑convolution
    gradient descent (soft‑max of the autocorrelation) followed by a
    short random‑pair local‑search polish.  The algorithm works on the
    probability simplex (sum of heights = 1) so the objective reduces
    to 2·n·max(conv) and the update is performed with an exponentiated‑gradient
    step on the soft‑max surrogate of the max‑convolution.
    """
    import time, math, numpy as np
    # set seed
    np.random.seed(seed)
    start = time.time()
    deadline = start + max(0.0, budget_s - 2.0)   # safety margin
    # ------------------------------------------------------------------
    # Helper: projection (simple normalisation)
    # ------------------------------------------------------------------
    def normalise(v):
        s = v.sum()
        if s <= 0:
            # fall back to uniform
            n = v.shape[0]
            return np.full(n, 1.0 / n)
        return v / s

    # ------------------------------------------------------------------
    # Helper: fast power‑of‑two
    # ------------------------------------------------------------------
    def next_pow2(x):
        return 1 << (x - 1).bit_length()

    # ------------------------------------------------------------------
    # Initialise a starting sequence.
    # ------------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        if base.size == 0:
            raise ValueError
        base = np.maximum(base, 0.0)
    except Exception:
        # fallback: modest uniform of length 4000
        base = np.full(4000, 1.0 / 4000.0, dtype=float)
    # Normalise to sum = 1 (the evaluation is scale‑invariant)
    base = normalise(base)

    best_seq = base.copy()
    # compute fast‑score (2·n·max) and keep official score for safety
    n_best = best_seq.shape[0]
    b0 = np.convolve(best_seq, best_seq)   # small length, ok
    best_score = float(2 * n_best * b0.max())
    # verify through the official evaluator (may clamp, etc.)
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        pass
    print(f"[Init] length={n_best}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation: smoothed max‑convolution descent.
    # ------------------------------------------------------------------
    a = base.copy()
    n = a.shape[0]

    # Pre‑compute FFT lengths (will stay constant for a fixed n)
    Lc = next_pow2(2 * n - 1)       # for self‑convolution
    Lg = next_pow2(3 * n - 2)       # for gradient (a_rev * w)

    # Hyper‑parameters (tuned empirically)
    beta = 0.5            # temperature for the soft‑max
    beta_inc = 0.0008     # increase per iteration
    eta0 = 1.0            # initial step‑size for the exponentiated update
    max_iters = 200000    # will be limited by the time budget
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- (1) self‑convolution (fast via FFT) -----
        Af = np.fft.rfft(a, Lc)
        conv_full = np.fft.irfft(Af * Af, Lc)
        b = conv_full[:2 * n - 1]
        max_b = b.max()

        # ----- (2) soft‑max weights -----
        # shift so the maximum becomes zero → the largest weight is exactly 1
        shift = b - max_b
        w = np.exp(beta * shift)        # 0 ≤ w_i ≤ 1
        Z = w.sum() + 1e-30

        # ----- (3) gradient of the soft‑max surrogate -----
        # correlation of a_rev with w gives Σ_i a_i w_{i+j}
        a_rev = a[::-1]
        Af_rev = np.fft.rfft(a_rev, Lg)
        Wf = np.fft.rfft(w, Lg)
        corr = np.fft.irfft(Af_rev * Wf, Lg)
        grad = 2.0 * corr[n - 1 : n - 1 + n] / Z

        # ----- (4) exponentiated‑gradient update (mirrored on the simplex) -----
        eta = eta0 / math.sqrt(it)          # decreasing step‑size
        # protect against overflow/underflow in the exponent
        update_log = -eta * grad
        update_log = np.clip(update_log, -50, 50)
        a *= np.exp(update_log)
        # renormalise to keep the vector on the probability simplex
        a = normalise(a)

        # ----- (5) keep the best candidate seen so far -----
        score = 2.0 * n * max_b
        if score < best_score:

            best_seq = a.copy()
            print(f"[Iter {it:05d}] new best score (fast) = {best_score:.6f}")

        beta = min(20.0, beta + beta_inc)

            # random pair move with a modest sigma
            sigma = 0.05

                delta = -min(-delta, a[i])

    # Ensure non‑negativity and reasonable length before returning.
    best_seq = np.maximum(best_seq, 0.0)

"""
Optimise a non‑negative step‑function by minimising the
max‑autocorrelation (the quantity evaluated by ``evaluate_sequence``).

The method works on the probability simplex (∑a_i = 1, a_i ≥ 0).  For a
vector a the objective reduces to 2·n·max(conv(a,a)).  We use a
*soft‑max* surrogate of the max:
    J(a) = (1/β)·log(∑_k exp(β·b_k)),   b = a*a .
Its gradient is a weighted sum of the gradients of the individual
convolution entries and can be computed by a single FFT‑based correlation.
An exponentiated‑gradient (mirror‑descent) step keeps a on the simplex.
The temperature β is slowly increased so that the surrogate becomes
sharper, while the step size η decays as 1/√t.  After the main phase a
short random‑pair local‑search polish (simulated‑annealing style) is
performed.  The best sequence found is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level – no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def normalise(v: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector to have ℓ₁‑norm = 1."""
    s = v.sum()
    if s <= 0:
        # fallback to a uniform distribution
        n = v.shape[0]
        return np.full(n, 1.0 / n, dtype=float)
    return v / s


def convolve_autocorr(a: np.ndarray) -> np.ndarray:
    """
    Return the autocorrelation a*a (full convolution) using FFT.
    Length of the result is 2·len(a)‑1.
    """
    n = a.shape[0]
    Lc = next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, Lc)
    conv_full = np.fft.irfft(Af * Af, Lc)
    return conv_full[: 2 * n - 1]


def random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """
    Pick two distinct indices i, j and move a small amount of mass
    (drawn from a symmetric uniform interval of width ``sigma``) from
    one to the other while preserving the total sum.
    """
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    delta = (np.random.rand() - 0.5) * sigma
    if delta > 0:
        delta = min(delta, seq[j])          # take from j
    else:
        delta = -min(-delta, seq[i])        # take from i

    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + max(0.0, budget_s - 2.0)

    # --------------------------------------------------------------
    # 1️⃣  Initialise a starting sequence.
    # --------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
        if base.size == 0:
            raise ValueError
        base = np.maximum(base, 0.0)
    except Exception:
        # fallback: a modest‑size uniform vector (≈4000 entries)
        base = np.full(4000, 1.0 / 4000.0, dtype=float)

    base = normalise(base)
    best_seq = base.copy()

    # Fast score = 2·n·max(b) (identical to the official score for sum=1)
    n_best = best_seq.shape[0]
    max_b0 = convolve_autocorr(best_seq).max()
    best_score = 2.0 * n_best * max_b0
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        pass
    print(f"[Init] length={n_best}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Main optimisation – softened max‑autocorrelation descent.
    # --------------------------------------------------------------
    a = base.copy()
    n = a.shape[0]

    # FFT sizes that stay constant for the whole run (chosen once).
    Lc = next_pow2(2 * n - 1)          # for self‑convolution
    Lg = next_pow2(3 * n - 2)          # for a_rev * w correlation

    # Hyper‑parameters (empirically tuned)
    beta = 0.5                         # temperature for the soft‑max
    beta_inc = 0.0008                  # increase per iteration (≈20 % after 20 k iters)
    eta0 = 1.0                         # initial step‑size for the exponentiated update
    max_iters = 200_000                # hard cap – actual number limited by time

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ---- (a) compute autocorrelation (FFT) ----
        Af = np.fft.rfft(a, Lc)
        conv_full = np.fft.irfft(Af * Af, Lc)
        b = conv_full[: 2 * n - 1]
        max_b = b.max()

        # ---- (b) soft‑max weights ----
        # shift so that the maximal entry gets weight = 1
        shift = b - max_b
        w = np.exp(beta * shift)                # 0 ≤ w_i ≤ 1
        Z = w.sum() + 1e-30                     # avoid division by zero

        # ---- (c) gradient of the soft‑max surrogate ----
        a_rev = a[::-1]
        Af_rev = np.fft.rfft(a_rev, Lg)
        Wf = np.fft.rfft(w, Lg)
        corr = np.fft.irfft(Af_rev * Wf, Lg)    # length ≥ 3·n‑2
        grad = 2.0 * corr[n - 1 : n - 1 + n] / Z

        # ---- (d) exponentiated‑gradient (mirror‑descent) step ----
        eta = eta0 / math.sqrt(it)              # decreasing step size
        update_log = -eta * grad
        update_log = np.clip(update_log, -50, 50)  # protect exp()
        a *= np.exp(update_log)
        a = normalise(a)                         # stay on the simplex

        # ---- (e) keep the best candidate ----
        fast_score = 2.0 * n * max_b
        if fast_score < best_score:

            best_seq = a.copy()
            print(f"[Iter {it:05d}] new best (fast) = {best_score:.6f}")

        # ---- (f) schedule updates ----
        beta = min(20.0, beta + beta_inc)

        # ---- (g) occasional tiny random perturbation to escape plateaus ----
        if it % 500 == 0:
            sigma = 0.05
            a = random_pair_move(a, sigma)

    # --------------------------------------------------------------
    # 3️⃣  Polishing – short simulated‑annealing / random‑pair search.
    # --------------------------------------------------------------
    polish_iters = 2000
    it = 0
    while time.time() < deadline and it < polish_iters:
        it += 1
        # linear cooling for the proposal amplitude
        temp = 1.0 - it / polish_iters
        sigma = 0.2 * temp + 0.01

        cand = random_pair_move(best_seq, sigma)    # sum stays unchanged

        # cheap fast evaluation (same as before)
        max_b = convolve_autocorr(cand).max()
        cand_score = 2.0 * n * max_b
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            print(f"[Polish {it:04d}] improved (fast) = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final verification with the official evaluator.
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[Final] official score = {final_score:.6f}")
    except Exception:
        final_score = best_score
        print("[Final] official evaluator failed – using fast score")

    # Ensure the returned sequence respects the required bounds.
    best_seq = np.maximum(best_seq, 0.0)
    return best_seq.tolist()
