"""
Propose a step‑function sequence with a low value of
    max_k (a * a)[k]   (the evaluation score).

The algorithm combines three ideas:

1.  Projected Adam optimisation.
    – While a “soft‑max” surrogate of the max‑convolution is used
      (high temperature) the gradient is cheap (single convolution).
    – After the temperature is cooled the exact sub‑gradient of the
      (hard) max is employed.
    – After every update the point is Euclidean‑projected onto the
      simplex {a ≥ 0, Σa = √(2 n)}.

2.  Several random restarts (different initialisations, periodic
    Gaussian shakes) to avoid being trapped in a poor basin.

3.  A cheap block‑wise random search that often discovers patterns
    missed by the gradient phase.

The code respects the 1000 s budget, prints intermediate progress, and
returns the best feasible sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _grad_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of the smoothed‑max (softmax) surrogate."""
    # convolution w * a_rev
    conv_wr = np.convolve(w, a[::-1])
    n = a.shape[0]
    # central n entries correspond to Σ_k w_k a_{k-i}
    sum_term = conv_wr[n - 1:n - 1 + n]
    grad = 2.0 * sum_term
    # diagonal correction for the i=j terms (k = 2i)
    double_idx = 2 * np.arange(n)
    mask = double_idx < conv_wr.size
    grad[mask] -= w[double_idx[mask]] * a[mask]
    return grad


# ----------------------------------------------------------------------
def _grad_hard_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Exact sub‑gradient of max_k (a*a)[k] (all indices attaining the max)."""
    n = a.shape[0]
    max_val = conv.max()
    # tolerance to capture all active indices
    idxs = np.where(conv >= max_val - 1e-12)[0]
    grad = np.zeros_like(a)
    i_arr = np.arange(n)

    for k in idxs:
        j = k - i_arr
        mask = (j >= 0) & (j < n)
        grad[mask] += a[j[mask]]
        # extra contribution when i=j (i = k/2, integer)
        if (k % 2) == 0:
            i_eq = k // 2
            if i_eq < n:
                grad[i_eq] += a[i_eq]
    return grad

            a = _project_to_simplex(a, target_sum)
            m_vec.fill(0.0)
            v_vec.fill(0.0)
            cur_max, cur_conv = _score_and_conv(a)
            mu = max(cur_max * mu_factor, 1.0)
            no_improve = 0
            next_restart = time.time() + restart_interval

        it += 1
        # ------------------------------------------------------------------
        # Safety break‑out near deadline.
        if time.time() > deadline - 5.0:
            break

    # ------------------------------------------------------------------
    # ----- cheap block‑search (post‑processing) -----------------------
    # Try a few hundred random piece‑wise constant patterns.
    block_trials = 300
    for _ in range(block_trials):
        nb = np.random.randint(2, 7)                     # 2‑6 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb)
        cand = np.zeros(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = np.convolve(cand, cand).max()
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[block] improved block score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # ----- final tiny local‑move refinement (if time permits) -------
    def _local_refine(a_vec: np.ndarray, cur_sc: float, max_trials: int = 5000):
        n_ = a_vec.shape[0]
        best_local = a_vec.copy()
        best_local_sc = cur_sc
        for _ in range(max_trials):
            i = np.random.randint(n_)
            if best_local[i] <= 1e-12:
                continue
            j = np.random.randint(n_)
            if i == j:
                continue
            # transfer a modest fraction of mass from i to j
            delta = best_local[i] * np.random.uniform(0.02, 0.08)
            delta = min(delta, best_local[i])
            cand = best_local.copy()
            cand[i] -= delta
            cand[j] += delta
            # sum unchanged, non‑negativity respected
            sc = np.convolve(cand, cand).max()
            if sc < best_local_sc:
                best_local_sc = sc
                best_local = cand
        return best_local, best_local_sc

    if time.time() < deadline - 2.0 and best_a is not None:
        refined_a, refined_sc = _local_refine(best_a, best_score, max_trials=8000)
        if refined_sc < best_score:
            best_score = refined_sc
            best_a = refined_a
            print(f"[refine] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    if best_a is None:
        # fallback: uniform distribution
        best_a = np.full(n, target_sum / n, dtype=float)

    print(f"[final] best score = {best_score:.6f} after {it} iterations")
    return best_a.tolist()
