def _subgradient_descent(a, n, deadline, max_iters=2000, eta0=0.2):
    best_seq = a.copy()
    best_score = _score(a)
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        b = _conv_full_fft(a)
        max_idx = np.argmax(b)
        # compute gradient g = 2*shift(a, max_idx)
        g = np.zeros_like(a)
        low = max(0, max_idx - (n-1))
        high = min(max_idx, n-1)
        i_range = np.arange(low, high+1)
        g[i_range] = a[max_idx - i_range]
        g *= 2.0
        # step size
        eta = eta0 / np.sqrt(it)
        a_tmp = a - eta * g
        # project onto simplex
        a_tmp = _project_onto_simplex(a_tmp, s=1.0)
        a = a_tmp
        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad n={n} it={it} score={best_score:.6f}")
        # occasional jitter
        if it % 500 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0, None)
            a /= a.sum()
    return best_seq, best_score

"""
Propose a low‑score step‑function sequence.

Algorithm
---------
1.  Load a historic sequence (if present) and keep it as a baseline.
2.  For several problem sizes `n` (4000‑10000) generate a diverse pool of
    initial candidates:
        * discrete arcsine law,
        * symmetric Beta(α,α) for many α (including random draws),
        * uniform,
        * a few‑spike random vector,
        * a Dirichlet draw with low concentration.
3.  Refine each candidate with two fast first‑order methods:
    a)  Exponentiated‑gradient descent on a smooth “soft‑max”
        approximation of the peak of the self‑convolution.
        The temperature λ is increased linearly to a modest λmax
        (≈30) while the step size η decays as 1/√t.
    b)  Sub‑gradient descent on the exact max‑convolution peak,
        projected onto the simplex after each step.
    The best sequence seen for a given length is kept.
4.  After all lengths have been processed, perform a short stochastic
    hill‑climbing phase on the overall best sequence: repeatedly move a
    tiny amount of mass from one index to another and keep the change if
    the (unsmoothed) score improves.
5.  Return the best sequence (as a plain list).  All vectors are kept
    non‑negative and normalised to sum 1, which leaves the objective
    invariant under the evaluator’s scaling rules.

The whole search respects the global time budget (default 1000 s).  The
implementation uses only NumPy and a handful of helper routines that
operate in O(n log n) via FFTs, allowing thousands of iterations within
the time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the grid x_i = (i+½)/n (normalised)."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (α=½) on n points, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT (length 2·n‑1)."""
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, m)
    C = np.fft.irfft(A * A, m)
    return C[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k−i} (correlation) using FFT.
    w – weight vector of length 2·n‑1, a – current sequence (length n),
    m – FFT length (power of two, at least len(w)+len(a)-1).
    Returns g[0…n‑1].
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start:start + len(a)]


# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a normalised to sum 1."""
    n = len(a)
    return float(2.0 * n * _conv_full_fft(a).max())

def _initial_candidates(n: int) -> list[np.ndarray]:
    """Generate a small, diverse set of initial probability vectors for size n."""
    cand = []

    # 1) arcsine
    cand.append(_arcsine_distribution(n))

    # 2) uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 3) symmetric betas with a grid of α
    for alpha in (0.25, 0.35, 0.45, 0.55, 0.65, 0.75, 0.9, 1.2):
        cand.append(_beta_distribution(n, alpha))

    # 4) random betas
    for _ in range(3):
        alpha = np.random.uniform(0.1, 2.0)
        cand.append(_beta_distribution(n, alpha))

    # 5) a sparse “spike” vector (few non‑zero entries)
    k = max(2, n // 500)               # a few spikes, roughly 0.2 % of positions
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = np.random.rand(k) + 0.1   # avoid exact zeros on selected spots
    cand.append(a / a.sum())

    # 6) Dirichlet draw with low concentration (produces many tiny entries)
    a = np.random.dirichlet(np.full(n, 0.3))
    cand.append(a)

    return cand


# ----------------------------------------------------------------------
def _optimise_exponential(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 6000,
    eta0: float = 0.15,
    lambda0: float = 0.3,
    lambda_max: float = 30.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror descent (exponentiated‑gradient) on a soft‑max approximation of
    max(conv(a,a)).  Returns the best vector (sum 1) seen and its true
    unsmoothed score.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()                       # safety normalisation
    best_seq = a.copy()
    best_score = _score(a)

    # FFT size for correlation (w length 2n‑1, a length n)
    m = 1 << ((3 * n - 2).bit_length())

    # Linear λ schedule
    lambda_inc = (lambda_max - lambda0) / max_iters
    lamb = lambda0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        b = _conv_full_fft(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} it={it:5d} score={best_score:.6f}")

        # soft‑max weights
        maxb = b.max()
        exp_arg = lamb * (b - maxb)          # always ≤0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.shape[0])
        else:
            w /= w_sum

        # gradient via correlation
        g = _gradient_via_correlation(w, a, m)

        # exponentiated‑gradient step
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)

        # projection onto the simplex (renormalise)
        a_sum = a.sum()
        if a_sum > 0.0:
            a = a / a_sum
        else:
            a = np.full(n, 1.0 / n)

        # occasional jitter to escape tiny plateaus
        if it % 800 == 0 and np.random.rand() < 0.15:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

        # increase λ
        lamb = min(lambda_max, lamb + lambda_inc)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_descent(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 2000,
    eta0: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution peak.
    Uses the sub‑gradient g_i = 2·a_{k*−i} where k* is an argmax of conv.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = _score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        b = _conv_full_fft(a)
        k_star = int(np.argmax(b))

        # build sub‑gradient g_i = 2·a_{k*−i}
        g = np.zeros_like(a)
        low = max(0, k_star - (n - 1))
        high = min(k_star, n - 1)
        i_range = np.arange(low, high + 1, dtype=int)
        g[i_range] = a[k_star - i_range]
        g *= 2.0

        eta = eta0 / np.sqrt(it)
        a = _project_onto_simplex(a - eta * g, s=1.0)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} sub it={it:5d} score={best_score:.6f}")

        # jitter
        if it % 500 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _hill_climb(
    a: np.ndarray,
    deadline: float,
    max_trials: int = 15000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search: move a tiny mass from one index to
    another and keep the change if the (unsmoothed) objective improves.
    """
    n = len(a)
    best_seq = a.copy()
    best_score = _score(best_seq)

    for t in range(max_trials):
        if time.time() >= deadline:
            break

        # pick donor and receiver
        i = np.random.randint(n)
        if best_seq[i] < 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # amount to transfer (up to 5 % of donor)
        delta = np.random.rand() * min(best_seq[i] * 0.05, 0.01)

        trial = best_seq.copy()
        trial[i] -= delta
        trial[j] += delta
        # projection not needed – sum unchanged and non‑negative by construction

        cur = _score(trial)
        if cur < best_score:
            best_score = cur
            best_seq = trial
            print(f"[{time.time():.1f}] hill‑climb trial={t} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a very low
    `evaluate_sequence` score.  Returns the best sequence found as a plain
    Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣  historic sequence, if available
    try:
        hist = np.array(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        hist = np.clip(hist, 0.0, None)
        if hist.sum() > 0.0:
            hist = hist / hist.sum()
            hist_score = _score(hist)
            overall_best_seq = hist.copy()
            overall_best_score = hist_score
            print(f"Historic sequence: length={len(hist)} score={hist_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2️⃣  problem sizes to explore
    candidate_lengths = [4000, 6000, 8000, 10000]
    # keep only lengths that still leave a reasonable amount of time
    candidate_lengths = [n for n in candidate_lengths if n >= 100]

    for n in candidate_lengths:
        if time.time() >= deadline:
            break

        print(f"--- Exploring length n={n} ---")
        # pool of starting points
        init_pool = _initial_candidates(n)

        # keep best for this n
        best_n_seq = None
        best_n_score = float("inf")

        # ------------------------------------------------------------------
        # a) smooth exponential mirror descent
        for init in init_pool:
            if time.time() >= deadline:
                break
            seq, sc = _optimise_exponential(
                n=n,
                init_seq=init,
                deadline=deadline,
                max_iters=6000,
                eta0=0.15,
                lambda0=0.3,
                lambda_max=30.0,
            )
            if sc < best_n_score:
                best_n_score = sc
                best_n_seq = seq.copy()
                print(f"  >>> New best for n={n}: score={best_n_score:.6f}")

        # ------------------------------------------------------------------
        # b) sub‑gradient refinement (if time permits)
        if best_n_seq is not None and time.time() < deadline:
            sub_deadline = deadline - 0.5   # keep a tiny margin
            seq, sc = _subgradient_descent(
                n=n,
                init_seq=best_n_seq,
                deadline=sub_deadline,
                max_iters=2000,
                eta0=0.25,
            )
            if sc < best_n_score:
                best_n_score = sc
                best_n_seq = seq.copy()
                print(f"  *** Sub‑grad refined n={n} score={best_n_score:.6f}")

        # ------------------------------------------------------------------
        # c) update global best
        if best_n_seq is not None and best_n_score < overall_best_score:
            overall_best_score = best_n_score
            overall_best_seq = best_n_seq.copy()
            print(f"*** Global best updated: length={n} score={overall_best_score:.6f}")

        # early exit if we already beat the target
        if overall_best_score <= 1.503:
            print("Target reached – exiting early.")
            break

    # ------------------------------------------------------------------
    # 3️⃣  final local hill‑climbing refinement
    if overall_best_seq is not None and time.time() < deadline:
        hill_deadline = deadline
        seq, sc = _hill_climb(overall_best_seq, hill_deadline, max_trials=15000)
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = seq.copy()
            print(f"*** After hill‑climb: score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  fallback (should never trigger)
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = _score(overall_best_seq)
        print("Fallback uniform sequence used.")

    print(f"Final best: length={len(overall_best_seq)} score={overall_best_score:.6f}")
    # Return as plain Python list for the evaluation harness
    return overall_best_seq.tolist()
