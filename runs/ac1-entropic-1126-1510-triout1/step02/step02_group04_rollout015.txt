def _gradient_step(seq, top_k, eta):

    n = len(seq)

    conv = np.convolve(seq, seq)

    top_idxs = np.argpartition(-conv, range(top_k))[:top_k]

    grad = np.zeros_like(seq)

    idx = np.arange(n)

    for k in top_idxs:

        j = k - idx

        mask = (j >= 0) & (j < n)

        grad[mask] += 2 * seq[j[mask]]

    # step (descend)
    new_seq = seq - eta * grad

    # Ensure nonnegativity, renorm

    new_seq = _clip_and_normalise(new_seq)

    return new_seq

"""
Propose a step‑function with a low evaluate_sequence() value.

The algorithm works in two stages:

1. **Exploratory random sampling** (≈ 30 % of the time budget).
   Many Dirichlet‑distributed vectors are generated, optionally smoothed
   or power‑transformed, and the one with the smallest convolution peak
   (hence smallest score) is kept.

2. **Local refinement** of the best sample.
   A deterministic projected sub‑gradient descent is performed on the
   objective  f(p)=max_k (p*p)[k] while staying on the probability simplex.
   For the current maximal lag we also apply a targeted mass‑transfer
   move that reduces the largest product a_i·a_{k‑i}.  The step size
   decays as 1/√iteration.  If progress stalls a short random restart is
   triggered.

All vectors are kept non‑negative and renormalised to sum = 1, so the
evaluate_sequence() wrapper returns a finite value.  The routine returns
the best sequence found before the time budget (≈ 1000 s) expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _clip_and_normalise(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise so that sum(v)=1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        v = np.full_like(v, 1.0 / v.size)
    else:
        v = v / s
    return v


def _score(seq: np.ndarray) -> float:
    """Fast score = 2·n·max(conv(seq,seq)).  All sequences are assumed to sum to 1."""
    n = seq.size
    conv = np.convolve(seq, seq)
    return float(2.0 * n * conv.max())


def _gradient_step(seq: np.ndarray, top_k: int, eta: float) -> np.ndarray:
    """
    One projected sub‑gradient step.
    Gradient is the sum of the sub‑gradients of the `top_k` largest convolution entries.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    # indices of the k largest entries
    top_idxs = np.argpartition(-conv, range(top_k))[:top_k]

    grad = np.zeros_like(seq)
    idx = np.arange(n)
    for k in top_idxs:
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad[mask] += 2.0 * seq[j[mask]]

    new_seq = seq - eta * grad
    return _clip_and_normalise(new_seq)


def _targeted_transfer(seq: np.ndarray, conv: np.ndarray, frac: float = 0.02) -> np.ndarray:
    """
    Reduce the current maximal convolution entry by moving a small fraction of mass
    from the index that contributes most to that entry to the index with the
    smallest current weight.
    """
    n = seq.size
    k_max = int(conv.argmax())

    # contribution a_i * a_{k_max-i}
    idx = np.arange(n)
    j = k_max - idx
    mask = (j >= 0) & (j < n)
    contrib = np.zeros_like(seq)
    contrib[mask] = seq[idx[mask]] * seq[j[mask]]

    if not np.any(mask):
        # degenerate – fallback to a tiny random perturbation
        cand = seq + frac * (np.random.rand(n) - 0.5)
        return _clip_and_normalise(cand)

    src = int(np.argmax(contrib))
    # destination: smallest weight (avoid src)
    dst_candidates = np.argsort(seq)[:5]  # a few of the smallest entries
    dst = int(dst_candidates[0]) if dst_candidates[0] != src else int(dst_candidates[1])

    amount = frac * seq[src] * np.random.rand()
    cand = seq.copy()
    cand[src] -= amount
    cand[dst] += amount
    return _clip_and_normalise(cand)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative sequence minimising `evaluate_sequence`.
    Returns a plain Python list of heights.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + max(0.0, budget_s - 0.5)

    # --------------------------------------------------------------
    # Determine the dimensionality (use the known construction if present)
    # --------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        base = _clip_and_normalise(base)          # normalize to the simplex
    except Exception:
        n = 8000                                   # a reasonable default
        base = np.full(n, 1.0 / n, dtype=float)

    # --------------------------------------------------------------
    # Phase 1 – random sampling
    # --------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    # allocate roughly 35 % of the time budget to sampling
    sample_deadline = start_time + (budget_s * 0.35)

    while time.time() < sample_deadline:
        # Dirichlet(1) – exponential draws give a uniform point on the simplex
        v = np.random.exponential(1.0, size=n)
        v = _clip_and_normalise(v)

        # occasional smoothing / power‑transform to diversify the search space
        if np.random.rand() < 0.5:                 # smoothing with prob 0.5
            kernel = np.array([0.25, 0.5, 0.25])
            v = np.convolve(v, kernel, mode='same')
            v = _clip_and_normalise(v)

        if np.random.rand() < 0.3:                 # apply a power‑law transformation
            power = np.random.choice([0.5, 0.8, 1.0, 1.2])
            v = v ** power
            v = _clip_and_normalise(v)

        cur_score = _score(v)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = v.copy()
            print(f"[Sample] New best score {best_score:.6f} (t={time.time()-start_time:.1f}s)")

    # fallback – should never happen, but keep the algorithm robust
    if best_seq is None:
        best_seq = base.copy()
        best_score = _score(best_seq)

    # --------------------------------------------------------------
    # Phase 2 – deterministic local refinement
    # --------------------------------------------------------------
    seq = best_seq.copy()
    cur_score = best_score
    top_k = 5            # number of active convolution lags used for the gradient
    eta0 = 0.15          # initial step size (will decay)
    iteration = 0
    last_improve_iter = 0
    stagnation_limit = 5000   # generations without improvement trigger a restart

    while time.time() < deadline:
        iteration += 1

        conv = np.convolve(seq, seq)
        max_conv = conv.max()
        cur_score = float(2.0 * n * max_conv)

        # record best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = seq.copy()
            last_improve_iter = iteration
            print(f"[Iter {iteration:5d}] New best score = {best_score:.6f}")

        # ----- 1) gradient descent on the active set -----
        eta = eta0 / np.sqrt(iteration)
        seq_candidate = _gradient_step(seq, top_k=top_k, eta=eta)
        cand_score = _score(seq_candidate)

        # ----- 2) targeted transfer move (focus on the maximal lag) -----
        seq_target = _targeted_transfer(seq, conv, frac=0.02)
        targ_score = _score(seq_target)

        # pick the best among the three candidates (old, grad, target)
        if cand_score < cur_score and cand_score <= targ_score:
            seq = seq_candidate
            cur_score = cand_score
        elif targ_score < cur_score:
            seq = seq_target
            cur_score = targ_score
        # else keep current sequence

        # ----- occasional random jitter to escape plateaus -----
        if iteration % 250 == 0:
            jitter = 0.005 * np.random.randn(n)
            seq = _clip_and_normalise(seq + jitter)

        # ----- stagnation handling -----
        if iteration - last_improve_iter > stagnation_limit:
            # re‑initialise from the best found so far with a small perturbation
            jitter = 0.02 * np.random.randn(n)
            seq = _clip_and_normalise(best_seq + jitter)
            last_improve_iter = iteration
            print(f"[Iter {iteration:5d}] Restart after stagnation")

    print(f"[Done] Best score after {iteration} iterations = {best_score:.6f}")
    return best_seq.tolist()
