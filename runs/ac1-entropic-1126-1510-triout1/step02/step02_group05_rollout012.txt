def project_to_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1  # zero-index
    theta = cssv[cond][-1] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Gradient‑descent optimisation of a step‑function for the evaluation metric.

The metric 2·n·max(conv(f,f))/sum(f)^2 is invariant under scaling, therefore we
fix the ℓ1‑norm of the candidate to 1 and minimise   max(conv(f,f)).
We use a smooth approximation of the max (log‑sum‑exp) with a temperature
parameter ρ that is increased during the run.  The gradient of the smooth
objective is

    ∇f L = 2· (w * rev(f))   with   w_k ∝ exp(ρ·c_k)

where c = conv(f,f) and rev(f) is f reversed.  All convolutions are performed
with FFT (O(N log N)).  After each gradient step the vector is projected onto
the probability simplex (non‑negative, sum = 1).  The best sequence seen is
kept, and a short random‑pair “polish” phase finishes the search.

The routine respects the given time budget (default 1000 s) and returns the
best list of heights.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper: projection onto the probability simplex (Σx_i = 1, x_i ≥ 0)
# --------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the largest rho where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are ≤ 0 → return uniform distribution
        return np.full(n, 1.0 / n, dtype=float)
    rho = ind[cond][-1] - 1               # zero‑based index
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Helper: full (linear) convolution using FFT (real inputs)
# --------------------------------------------------------------
def _full_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the linear convolution x*y (length len(x)+len(y)-1)."""
    n = x.shape[0] + y.shape[0] - 1
    # next power of two for efficient FFT
    L = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, L)
    Y = np.fft.rfft(y, L)
    Z = X * Y
    conv = np.fft.irfft(Z, L)[:n]
    return conv


# --------------------------------------------------------------
# Helper: cheap random pairwise mass exchange (keeps ℓ1‑norm = 1)
# --------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Pick two indices i≠j and move a small amount of mass (|δ| ≤ σ) from one to the other."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    # draw a symmetric uniform proposal
    delta = (np.random.rand() - 0.5) * sigma

    # clamp to keep non‑negativity
    if delta > 0:
        delta = min(delta, seq[j])      # take from j
    else:
        delta = -min(-delta, seq[i])    # take from i

    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq

    max‑convolution, followed by a short random‑pair polishing phase.
    """

    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 2.0)   # tiny safety margin

    # ------------------------------------------------------------------
    # Initialise a starting sequence.
    # ------------------------------------------------------------------
    try:

        n = 2500

    try:

    except Exception:

    print(f"[Init] length={a.size}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # PARAMETERS FOR THE GRADIENT‑DESENT LOOP
    # ------------------------------------------------------------------
    rho = 5.0            # starting temperature for the soft‑max
