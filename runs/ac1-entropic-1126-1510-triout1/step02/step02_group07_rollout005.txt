"""
Propose a step‑function sequence that minimises the metric

    2·n·max(conv) / (∑a)²

where conv = a * a (self‑convolution).  The search proceeds in three
phases:

1. **Broad parameter sweep** – try several sequence lengths and a few
   families of probability densities (Beta(α,α), geometric, arcsine and
   versions with zero “gaps’’ at the ends).  The best candidate of this
   sweep is taken as the starting point.

2. **Projected sub‑gradient descent** – the exact sub‑gradient of the
   max‑convolution is computed (the gradient of max(conv) w.r.t. a).
   A small step in the negative direction is taken and the result is
   projected back onto the probability simplex.  The step size decays
   slowly.

3. **Random pairwise “hill‑climbing’’** – occasionally a random transfer
   of a tiny amount of mass between two coordinates is tried; if it
   improves the score the move is kept.  The algorithm alternates the two
   move types until the time budget (≈1000 s) expires.

Both phases keep the sequence non‑negative and normalised (∑a = 1), so
the evaluation metric is scale‑invariant.  The best sequence found
before the deadline is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> np.ndarray:
    """Arcsine (Beta(½,½)) coefficients, normalised to sum = 1."""
    seq = np.empty(n, dtype=float)
    cur = 1.0
    for i in range(n):
        seq[i] = cur
        if i + 1 < n:
            cur = cur * (2.0 * i + 1.0) / (2.0 * (i + 1))
    seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
def beta_sequence(n: int, alpha: float, beta: float | None = None) -> np.ndarray:
    """Discrete Beta(α,β) density on mid‑points of n equal sub‑intervals,
    normalised to sum = 1.  If β is omitted it is set equal to α (symmetric case)."""
    if beta is None:
        beta = alpha
    # mid‑points to avoid the singularities at 0 and 1
    x = (np.arange(n, dtype=float) + 0.5) / n
    logB = math.lgamma(alpha) + math.lgamma(beta) - math.lgamma(alpha + beta)
    log_pdf = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x) - logB
    pdf = np.exp(log_pdf)
    pdf /= pdf.sum()
    return pdf


# ----------------------------------------------------------------------
def geometric_sequence(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised to sum = 1 (0 < r < 1)."""
    idx = np.arange(n, dtype=float)
    weights = r ** idx
    weights /= weights.sum()
    return weights


# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x ≥ 0, Σ x = s}.  Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Target sum s must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def compute_score(seq: np.ndarray) -> float:
    """Exact copy of evaluate_sequence (without the defensive wrappers) for speed."""
    seq = np.asarray(seq, dtype=float)
    if np.any(np.isnan(seq)) or np.any(np.isinf(seq)):
        return float('inf')
    n = seq.size
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)

    of coefficients found.
    """

    best_seq: np.ndarray | None = None
    best_score = float('inf')
    best_n = None

    # --------------------------------------------------------------
    # 0) try the previously discovered global sequence if available
    try:
        global height_sequence_1
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.size > 0:
            seq = np.maximum(seq, 0.0)
            if seq.sum() > 0:
                seq = seq / seq.sum()
                sc = compute_score(seq)
                print(f"[global] len={seq.size:<5d} score={sc:.6f}")
                if sc < best_score:
                    best_seq = seq.copy()
                    best_score = sc
                    best_n = seq.size
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1) Broad sweep over lengths and families
    lengths = [2000, 2500, 3000, 3500, 4000, 4500, 5000]
    alphas = [0.35, 0.40, 0.45, 0.50, 0.55, 0.60, 0.65, 0.70]
    geom_rs = [0.93, 0.95, 0.97, 0.99]

    for n in lengths:
        if time.time() >= deadline:
            break

        # --- arcsine -------------------------------------------------
        seq = arcsine_sequence(n)
        sc = compute_score(seq)
        if sc < best_score:
            best_seq, best_score, best_n = seq, sc, n
            print(f"[init] arcsine  n={n:<5d} score={sc:.6f}")

        # --- symmetrised arcsine ------------------------------------
        sym = (seq + seq[::-1]) * 0.5
        sym /= sym.sum()
        sc = compute_score(sym)
        if sc < best_score:
            best_seq, best_score, best_n = sym, sc, n
            print(f"[init] sym_arc n={n:<5d} score={sc:.6f}")

        # --- Beta(α,α) -----------------------------------------------
        for a in alphas:
            if time.time() >= deadline:
                break
            seq = beta_sequence(n, a)
            sc = compute_score(seq)
            if sc < best_score:
                best_seq, best_score, best_n = seq, sc, n
                print(f"[grid] beta  n={n:<5d} α={a:.2f} score={sc:.6f}")

        # --- Geometric -----------------------------------------------
        for r in geom_rs:
            if time.time() >= deadline:
                break
            seq = geometric_sequence(n, r)
            sc = compute_score(seq)
            if sc < best_score:
                best_seq, best_score, best_n = seq, sc, n
                print(f"[grid] geom  n={n:<5d} r={r:.2f} score={sc:.6f}")

        # --- Gap‑augmented arcsine -----------------------------------
        # Zero a small number of entries at both ends and place an arcsine on the interior.
        for gap_frac in (0.0, 0.01, 0.02, 0.05):
            gap = int(round(gap_frac * n))
            if 2 * gap >= n:
                continue
            interior_len = n - 2 * gap
            interior = arcsine_sequence(interior_len)
            full = np.concatenate([np.zeros(gap), interior, np.zeros(gap)])
            # interior already sums to 1, zeros keep the sum 1.
            sc = compute_score(full)
            if sc < best_score:
                best_seq, best_score, best_n = full, sc, n
                print(f"[gap] n={n:<5d} gap={gap} score={sc:.6f}")

    # --------------------------------------------------------------
    # Fallback – if nothing was successful, use a moderate arcsine.
    if best_seq is None:
        best_n = 3000
        best_seq = arcsine_sequence(best_n)
        best_score = compute_score(best_seq)
        print(f"[fallback] n={best_n} score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Refinement loop: sub‑gradient descent + random pairwise moves
    step_size = 0.10            # initial step magnitude
    step_decay = 0.9995
    grad_prob = 0.45            # probability to perform a sub‑gradient step
    no_improve = 0
    max_no_improve = 3000

    while time.time() < deadline:
        n = best_seq.shape[0]

        # Current convolution and set of maximisers
        conv = np.convolve(best_seq, best_seq)
        M = conv.max()
        active = conv >= (M - 1e-12)          # Boolean mask of active constraints

        # Exact sub‑gradient of max(conv) (sum of gradients of all active entries)
        w = active.astype(float)
        grad = 2.0 * np.convolve(w, best_seq[::-1])[n - 1:2 * n - 1]

        # Decide which move to try this iteration
        if np.random.rand() < grad_prob:
            # ----- sub‑gradient descent step -----
            cand = best_seq - step_size * grad
        else:
            # ----- random pairwise transfer -----
            i, j = np.random.choice(n, 2, replace=False)
            delta = step_size * best_seq[i]
            if delta <= 0.0:
                continue
            cand = best_seq.copy()
            cand[i] -= delta
            cand[j] += delta

        # Project back onto the simplex and enforce non‑negativity
        cand = project_simplex(cand, s=1.0)

        # Occasionally try symmetrising – it never hurts the max‑conv if the distribution is already symmetric
        sym = (cand + cand[::-1]) * 0.5
        sym /= sym.sum()
        if compute_score(sym) < compute_score(cand):
            cand = sym

        cand_score = compute_score(cand)

        if cand_score + 1e-12 < best_score:
            best_seq = cand
            best_score = cand_score
            best_n = n
            no_improve = 0
            print(f"[improve] score={best_score:.6f} n={n}")
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                # If we are stuck, increase the step size a little to escape
                step_size = min(0.5, step_size * 1.5)
                no_improve = 0

        # Decay step size (but never let it become too tiny)
        step_size = max(1e-7, step_size * step_decay)

    # --------------------------------------------------------------
    # Final sanity – ensure feasibility
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total == 0:
        # extremely unlikely – fall back to uniform
        best_seq = np.ones_like(best_seq) / best_seq.size
    else:
        best_seq /= total

    print(f"Best final length = {len(best_seq)}  score = {best_score:.6f}")
    return best_seq.tolist()
