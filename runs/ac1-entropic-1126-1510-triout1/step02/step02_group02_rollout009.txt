assistantfinal**Solution Overview**

The goal is to find a non‑negative sequence `a` (step‑function heights) with a fixed ℓ¹ norm  
`∑a = √(2·n)` that minimises  

```
score(a) = 2·n·max(a*a) / (∑a)²                (a*a = full linear convolution)
```

The problem is convex in the *max‑convolution* but the convolution itself is quadratic.
A practical approach is:

1. **Generate many simple shapes** (uniform, inverse‑sqrt, power‑law, exponential, cosine,
   random block structures, Dirichlet samples).  
   All candidates are projected onto the simplex `{a≥0, Σa = √(2n)}` and the best
   one is kept.

2. **Refine the best candidate by smooth projected gradient descent**.  
   * Smoothing – a soft‑max with temperature `T` approximates the max‑operator.  
   * Gradient – for a weighted sum `∑ w_k (a*a)_k` the exact gradient is  

   ```
   ∂/∂a_i  Σ w_k (a*a)_k = 2· Σ_k w_k a_{k-i}
   ```

   which can be evaluated by a single FFT‑based convolution (no diagonal correction
   needed).  
   * Update – an Adam‑style adaptive step is used, followed by Euclidean projection
   onto the simplex.  
   * Temperature is exponentially cooled; occasional Gaussian noise is added to escape
   plateaus.

3. **If time permits** a final short random‑block search is performed.

All heavy work (convolutions, gradients) is performed with FFTs of size
`≥ 3·n‑2`, giving an `O(n log n)` cost per iteration.  
The algorithm respects the security checks of `evaluate_sequence` because
only non‑negative numbers are ever generated and the ℓ¹ norm is fixed.

The implementation below follows the described strategy and returns the best
sequence found before the time budget expires.

* a fast parametric shape sweep,
* Adam‑style projected gradient descent on a soft‑max approximation of the

* occasional random perturbations,
* a final short random‑block search.

All updates keep the vector on the simplex {a ≥ 0, Σa = √(2·n)} so the
objective is scale‑invariant and the security checks of evaluate_sequence()
are never triggered.

def project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:

    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution via FFT (full length = len(u)+len(v)-1)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Return the full self‑convolution a*a."""
    return conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a (weights w are fixed).
    Exact formula:  ∂/∂a_i = 2· Σ_k w_k a_{k-i}.
    Implemented by one FFT convolution of w with the reversed a.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    conv = conv_fft(w, a_rev, size_fft)          # length = 3n‑2
    start = n - 1
    grad = 2.0 * conv[start:start + n]           # 2· Σ_k w_k a_{k-i}
    return grad


# ----------------------------------------------------------------------
def simple_score(a: np.ndarray, size_fft: int) -> float:
    """Fast evaluation of the objective (without security checks)."""
    n = a.shape[0]
    conv = self_convolution(a, size_fft)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a low‑score step‑function within the given time budget."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)      # type: ignore[name-defined]
    except Exception:
        n = 3000
    target_sum = math.sqrt(2.0 * n)      # the normalisation used in the paper
    size_fft = 1 << ((3 * n - 2).bit_length())   # power‑of‑two ≥ 3n‑2

    # --------------------------------------------------------------
    # 1) QUICK SHAPE SEARCH
    # --------------------------------------------------------------
    best_a = None
    best_score = float("inf")

    def try_candidate(raw: np.ndarray, label: str = "") -> None:
        nonlocal best_a, best_score
        cand = project_to_simplex(raw.astype(float), target_sum)
        sc = simple_score(cand, size_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[shape] new best ({label}) = {sc:.6f}")

    # uniform
    try_candidate(np.ones(n), "uniform")

    # inverse‑sqrt families
    for _ in range(70):
        beta = np.random.rand() * 10.0
        cand = 1.0 / np.sqrt(np.arange(n) + beta + 1.0)
        try_candidate(cand, "inv‑sqrt")

    # double‑ended inverse‑sqrt
    for _ in range(70):
        beta = np.random.rand() * 10.0
        i = np.arange(n)
        cand = (1.0 / np.sqrt(i + beta + 1.0) +
                1.0 / np.sqrt((n - 1 - i) + beta + 1.0))
        try_candidate(cand, "double‑inv‑sqrt")

    # power‑law
    for _ in range(70):
        beta = np.random.rand() * 10.0
        pexp = np.random.rand() * 1.7 + 0.3
        cand = 1.0 / (np.arange(n) + beta + 1.0) ** pexp
        try_candidate(cand, "power‑law")

    # exponential decay
    for _ in range(70):
        decay = np.random.rand() * 0.01 + 0.0005
        cand = np.exp(-decay * np.arange(n))
        try_candidate(cand, "exp")

    # cosine‑based (large at the ends, small in the middle)
    for _ in range(70):
        p = np.random.rand() * 5.0 + 0.5
        cand = np.abs(np.cos(np.pi * np.arange(n) / (n - 1))) ** p
        try_candidate(cand, "cosine")

    # mixture of uniform and inverse‑sqrt
    for _ in range(70):
        beta = np.random.rand() * 10.0
        lam = np.random.rand()
        cand = lam * np.ones(n) + (1.0 - lam) * (1.0 /
                np.sqrt(np.arange(n) + beta + 1.0))
        try_candidate(cand, "mix‑uniform‑inv")

    # random block structures (2‑6 blocks)
    for _ in range(150):
        nb = np.random.randint(2, 7)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.zeros(n)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        try_candidate(cand, "block")

    # random Dirichlet / exponential samples
    for _ in range(120):
        cand = np.random.exponential(scale=1.0, size=n)
        try_candidate(cand, "dirichlet")

    print(f"[shape] best score after sweep = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) REFINEMENT VIA SMOOTHED PROJECTED GRADIENT (Adam)
    # --------------------------------------------------------------
    def refine(start_vec: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
        """Run Adam‑style projected gradient descent for at most time_limit seconds."""
        a = start_vec.copy()
        best_local = a.copy()
        best_local_score = simple_score(a, size_fft)

        # temperature – start with a fraction of the current spread
        conv = self_convolution(a, size_fft)
        T = max(1e-3, (conv.max() - conv.min()) * 0.5)

        # Adam parameters
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        it = 0
        no_improve = 0
        lr0 = target_sum * 0.5 / n                # base learning rate
        lr = lr0

        while time.time() < time_limit and it < 2_000_000:
            # evaluate current objective
            conv = self_convolution(a, size_fft)
            cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local = a.copy()
                no_improve = 0
                print(f"[refine {it:07d}] new best = {cur_score:.6f}")
            else:
                no_improve += 1

            # -------- soft‑max weights ----------
            if T < 1e-8:
                # hard max (subgradient) when temperature vanished
                mask = (conv >= conv.max() - 1e-12)
                w = mask.astype(float)
                w /= w.sum()
            else:
                shifted = conv - conv.max()
                exp_arg = shifted / T
                exp_arg = np.clip(exp_arg, -50, 0)     # stability
                w = np.exp(exp_arg)
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

            # -------- gradient ----------
            grad = weighted_gradient(a, w, size_fft)

            # -------- Adam update ----------
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = a - step
            a_new = project_to_simplex(a_new, target_sum)

            # occasional perturbation to get out of plateaus
            if (it % 5000 == 0) or (no_improve > 5000):
                noise = np.random.randn(n) * target_sum * 1e-3
                a_new = a_new + noise
                a_new = project_to_simplex(a_new, target_sum)
                no_improve = 0
                print(f"[refine {it:07d}] perturbation")

            a = a_new
            # cool temperature and learning rate slowly
            T *= 0.9999
            lr *= 0.9999

        return best_local, best_local_score

    # Allocate time: 80 % for refinement, rest for final block search
    remaining = deadline - time.time()
    refine_time = remaining * 0.80
    final_block_time = remaining - refine_time

    # First refinement from the best shape discovered above
    best_vec, best_vec_score = refine(best_a, start_time + refine_time)

    # If we still have a few seconds left, try a few extra random starts
    extra_start_deadline = start_time + refine_time
    extra_runs = 0
    while time.time() < extra_start_deadline and extra_runs < 3:
        # random Dirichlet start
        rnd = np.random.exponential(scale=1.0, size=n)
        rnd = project_to_simplex(rnd, target_sum)
        vec, sc = refine(rnd, extra_start_deadline)
        if sc < best_vec_score:
            best_vec_score, best_vec = sc, vec.copy()
        extra_runs += 1

    # --------------------------------------------------------------
    # 3) FINAL SHORT RANDOM BLOCK SEARCH (if time permits)
    # --------------------------------------------------------------
    if final_block_time > 1.0:
        end_block_deadline = start_time + budget_s - 0.5
        while time.time() < end_block_deadline:
            nb = np.random.randint(2, 7)
            cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = np.random.rand(nb) + 0.1
            cand = np.zeros(n)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = project_to_simplex(cand, target_sum)
            sc = simple_score(cand, size_fft)
            if sc < best_vec_score:
                best_vec_score, best_vec = sc, cand.copy()
                print(f"[final block] improved to {sc:.6f}")

    print(f"[final] best score = {best_vec_score:.6f}")
    # Return as plain Python list (evaluate_sequence expects a list of floats)
    return best_vec.tolist()
