"""
Propose a step‑function sequence with a low self‑convolution peak.

Algorithm
---------

1. **Initialisation** – choose a moderate length (≈2000) and start from the
   “arcsine” distribution (the coefficients C_i = binom(2i,i)/4^i, L1‑normalised).

2. **Alternating LP / gradient refinement** – repeat the following while time
   remains.

   *Compute the current max of the self‑convolution*  M = max(a∗a).

   *Linear‑programming step* – with the current sequence a fixed, solve the LP

        maximise    Σ g_i
        subject to  g_i ≥ 0
                    a ∗ g ≤ M                (2n‑1 linear inequalities)

   This is linear in the unknown g, therefore we can use ``scipy.optimize.linprog``.
   The solution g is L1‑normalised and mixed with a:

        a ← (1‑t)·a + t·g ,   t decreasing from 0.5 → 0.01

   Several candidate mixing weights are tried; the one giving the smallest
   self‑convolution peak is kept.

   time is left) a smooth approximation of the max (soft‑max) is used and a
   projected gradient step on the probability simplex is taken.

3. **Book‑keeping** – after each update the true score
   ``evaluate_sequence`` is computed; the best sequence seen so far is stored

The routine works even when ``scipy`` is unavailable (it then falls back to

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> list[float]:
    """Return a length‑n arcsine distribution normalised to L1 = 1."""
    seq = []
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2.0 * i + 1.0) / (2.0 * (i + 1))
    seq = [x / total for x in seq]
    return seq


# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σ x = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("target sum s must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (shape (2n‑1, n)) such that (A @ g)[k] = (f ∗ g)[k]
    for all k.  Column i of A contains f[i] on rows i … i+n‑1.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for i, fi in enumerate(f):
        A[i:i + n, i] = fi
    return A


# ----------------------------------------------------------------------
def solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max Σ g_i  subject to g ≥ 0,  f ∗ g ≤ M.
    Returns a normalised (L1 = 1) vector g, or None if the LP fails.
    """
    try:
        from scipy.optimize import linprog
    except Exception:
        return None

    n = f.shape[0]
    A = build_conv_matrix(f)                     # (2n‑1) × n
    b = M * np.ones(2 * n - 1, dtype=float)

    # maximise Σ g_i  → minimise -Σ g_i
    c = -np.ones(n, dtype=float)

    bounds = [(0, None)] * n
    # Use the HiGHS interior‑point solver (fast and robust)
    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 30},
    )

    if not res.success or res.x is None:
        return None

    g = np.asarray(res.x, dtype=float)
    sum_g = g.sum()
    if sum_g <= 0:
        return None
    return g / sum_g   # normalise to L1 = 1
