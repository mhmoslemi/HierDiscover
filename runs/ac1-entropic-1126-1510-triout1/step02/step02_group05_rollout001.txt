def simplex_projection(v):
    """Project v onto the probability simplex {x >=0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]  # last true index
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function that (hopefully) beats the previous best score.

The algorithm has three stages

1. **Initialisation** – use the globally supplied `height_sequence_1`
   (if it exists) otherwise fall back to a uniform distribution.
   The vector is normalised to unit ℓ¹‑norm (the objective of
   ``evaluate_sequence`` is scale‑invariant).

2. **Exponential‑gradient descent on the worst autocorrelation offset** –
   at each iteration we locate the index `k_max` where the linear
   convolution `a*a` attains its maximum `M`.  The sub‑gradient of this
   maximum w.r.t. the vector `a` is `g_i = 2·a_{k_max‑i}` (zero outside
   the support).  A mirror‑descent step on the probability simplex is
   performed by the multiplicative update
        a_i ← a_i·exp(‑η·g_i)
   followed by renormalisation.  The step size η decreases like 1/√t.
   This update directly reduces the product `a_i·a_{k_max‑i}` that is
   responsible for the current worst value, and therefore tends to lower
   the overall objective `2·n·max(a*a)`.

3. **Short simulated‑annealing polish** – after the deterministic phase
   we run a few thousand random pair‑wise “mass‑swap’’ moves (the same
   move used in the reference implementation).  The temperature (the
   Gaussian‑like width `σ`) is linearly cooled; a move is kept only if it
   improves the best score seen so far.

The best sequence encountered during the whole run is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper for random pairwise mass‑swap (keeps ℓ¹‑norm exactly 1)
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Pick two distinct positions i, j and transfer a small amount of
    mass (drawn from a symmetric uniform interval of width `sigma`) from
    one to the other while preserving non‑negativity."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    # propose a zero‑mean transfer
    delta = (np.random.rand() - 0.5) * sigma

    if delta > 0:
        # move mass from j to i
        delta = min(delta, seq[j])
        new_seq = seq.copy()
        new_seq[i] += delta
        new_seq[j] -= delta
    else:
        # move mass from i to j
        delta = -min(-delta, seq[i])
        new_seq = seq.copy()
        new_seq[i] += delta   # delta is negative
        new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# Exponential‑gradient (mirror descent) step
# ----------------------------------------------------------------------
def _exp_gradient_step(seq: np.ndarray, k_max: int, eta: float) -> np.ndarray:
    """
    Perform one mirror‑descent step on the simplex that reduces the
    contribution to the autocorrelation value at offset `k_max`.

    The sub‑gradient of the active autocorrelation entry is
    g_i = 2·seq[k_max‑i] (zero if the index is out of range).  The update
    is multiplicative: seq_i ← seq_i·exp(‑η·g_i) and the vector is
    re‑normalised to sum to one.
    """
    n = seq.shape[0]
    i = np.arange(n)
    j = k_max - i
    valid = (j >= 0) & (j < n)

    # compute sub‑gradient (zero where invalid)
    grad = np.zeros_like(seq)
    grad[valid] = 2.0 * seq[j[valid]]

    # multiplicative mirror‑descent update
    new_seq = seq * np.exp(-eta * grad)

    # avoid numerical collapse to exact zeros
    eps = 1e-30
    new_seq = np.maximum(new_seq, eps)

    # renormalise to the probability simplex
    new_seq /= new_seq.sum()
    return new_seq
