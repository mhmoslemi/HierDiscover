def propose_candidate(seed=0, budget_s=1000):
    import time, math, numpy as np
    from scipy.optimize import linprog

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2

    # Determine n
    try:
        n = len(height_sequence_1)
        have_ref = True
    except Exception:
        n = 2500 # fallback moderate length
        have_ref = False

    target_sum = math.sqrt(2.0 * n)

    # Initialize a
    if have_ref and np.random.rand() < 0.5:
        a = np.array(height_sequence_1, dtype=float)
    else:
        a = np.random.rand(n)
    # Project onto simplex
    a = _project_to_simplex(a, target_sum)

    # Starting best
    best_a = a.copy()
    best_score = np.max(np.convolve(best_a, best_a))
    print(f"[init] n={n} target_sum={target_sum:.3f} start score={best_score:.6f}")

    # hyperparameters
    t = 0.2    # step size for convex combination
    min_t = 0.02
    decay_t = 0.99  # decay per iteration
    iter = 0
    while time.time() < deadline:
        conv = np.convolve(a, a)
        M = conv.max()
        # solve LP for g0
        # Build constraint matrix
        rows = 2*n - 1
        A = np.zeros((rows, n), dtype=float)
        for j in range(n):
            A[j:j+n, j] = a
        # Right-hand side vector
        b_ub = np.full(rows, M, dtype=float)

        # Use linprog
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n,
                      method='highs', options={'presolve': True, 'time_limit': max(1.0, (deadline - time.time()))})
        if not res.success:
            # fallback: keep a unchanged
            g = a.copy()
        else:
            g0 = res.x
            sum_g0 = g0.sum()
            if sum_g0 > 0:
                g = g0 * target_sum / sum_g0
            else:
                g = a.copy()

        # combine
        a = (1 - t) * a + t * g
        # a is already nonnegative and sum preserved (both sums = target_sum)
        # compute score
        new_score = np.max(np.convolve(a, a))
        if new_score < best_score - 1e-9:
            best_score = new_score
            best_a = a.copy()
            print(f"[{iter}] new best = {best_score:.6f}")
        # adapt step size
        iter += 1
        t = max(min_t, t * decay_t)

        # optional restart
        if np.random.rand() < 0.01:
            a = np.random.rand(n)
            a = _project_to_simplex(a, target_sum)

    # final local block search (same as previous) to fine-tune
    block_trials = 200
    for _ in range(block_trials):
        nb = np.random.randint(2, 7)
        cuts = np.sort(np.random.choice(n-1, nb-1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb)
        cand = np.zeros(n)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = np.max(np.convolve(cand, cand))
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[block] improved block score = {best_score:.6f}")
        if time.time() >= deadline:
            break

    print(f"[final] best score={best_score:.6f} after {iter} iterations")
    return best_a.tolist()

    n = a.shape[0]
    m = np.zeros_like(a)
    beta = 0.9
    mu = np.max(_convolve(a))
    mu_decay = 0.9995
    mu_min = 1e-6
    best_a = a.copy()
    best_score = np.max(_convolve(a))
    it = 0
    top_k = max(5, int(0.001 * n))
    while time.time() < deadline:
        conv = _convolve(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            print(f"[phase1-{it}] new best = {best_score:.6f}")
        # gradient:
        if mu > 0.01 * cur_max:
            diff = conv - cur_max
            w = np.exp(diff / mu)
            w /= w.sum()
            # convolution of w with a reversed
            conv_wr = np.convolve(w, a[::-1])
            sum_term = conv_wr[n-1:n-1+n]
            grad = 2.0 * sum_term
            # diagonal correction
            diag_idx = 2 * np.arange(n)
            mask = diag_idx < len(conv)
            grad[mask] -= w[diag_idx[mask]] * a[mask]
        else:
            # top-k subgradient
            idxs = np.argpartition(-conv, top_k - 1)[:top_k]
            grad = np.zeros_like(a)
            i = np.arange(n)
            for k in idxs:
                j = k - i
                mask = (j >= 0) & (j < n)
                grad[mask] += 2.0 * a[j[mask]]
                # diagonal correction automatically accounted
        # momentum
        m = beta * m + (1.0 - beta) * grad
        # step size
        eta0 = target_sum * 0.4 / n
        eta = eta0 / (1.0 + 1e-4 * it)
        a_new = a - eta * m

            a_new = _project_to_simplex(a_new, target_sum)
        a = a_new

"""
propose_candidate – hybrid optimizer for the max‑convolution problem.

The evaluation function is 2 n·max(a∗a)/(∑a)².  After projecting the
sequence onto the simplex {a≥0, Σa = √(2 n)} the denominator equals 2 n,
so the score is exactly the maximum of the autocorrelation (a∗a).

The algorithm consists of three stages:

1. **Annealed projected gradient** – a fast Adam‑style descent whose
   gradient is the sub‑gradient of a soft‑max (high temperature) or of the
   *k* largest convolution entries (low temperature).  The iterate is
   projected onto the simplex after every step.

2. **Linear‑programming refinement** – for the current iterate `a` we
   solve

        maximise   Σ b_j
        subject to   (a∗b)_k ≤ M  for all k,
                     b_j ≥ 0, b_j ≤ 1000

   where `M = max(a∗a)`.  The LP pushes mass into coordinates that do not
   increase the current worst autocorrelation.  The solution `b` is scaled
   to the required ℓ¹‑norm and mixed with `a`.

3. **Local post‑processing** – cheap block‑wise random constructions and
   tiny pairwise mass transfers are tested; any improvement is kept.

All steps respect the non‑negativity and the ℓ¹‑norm constraint, and the
total wall‑clock time never exceeds the given budget (default 1000 s).
"""

import time
import math
import numpy as np
from scipy.optimize import linprog


def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = z}."""
    if z <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _convolve(a: np.ndarray) -> np.ndarray:
    """Fast convolution a∗a using real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    # size = next power of two ≥ 2·n‑1 for speed
    size = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _gradient_topk(a: np.ndarray, conv: np.ndarray, top_k: int = 5) -> np.ndarray:
    """Sub‑gradient of the max‑convolution using the `top_k` largest entries."""
    n = a.shape[0]
    idxs = np.argpartition(-conv, top_k - 1)[:top_k]
    grad = np.zeros_like(a)
    i = np.arange(n)
    for k in idxs:
        j = k - i
        mask = (j >= 0) & (j < n)
        grad[mask] += 2.0 * a[j[mask]]
    return grad


def _phase1(a: np.ndarray, target_sum: float, deadline: float):
    """Annealed projected (Adam‑style) gradient descent."""
    n = a.shape[0]
    m = np.zeros_like(a)          # first moment (Adam)
    v = np.zeros_like(a)          # second moment (Adam)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0
    best_a = a.copy()
    best_score = np.max(_convolve(a))
    mu = best_score               # temperature start (soft‑max width)
    mu_decay = 0.9995
    mu_min = 1e-6
    top_k = max(5, int(0.002 * n))

    while time.time() < deadline:
        conv = _convolve(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            print(f"[phase1 {it:06d}] new best = {best_score:.6f}")

        # ---- gradient -------------------------------------------------
        if mu > 0.01 * cur_max:                     # high temperature -> smooth‑max
            diff = conv - cur_max                   # ≤ 0
            w = np.exp(diff / mu)
            w /= w.sum()
            # gradient via a single convolution (w * a[::-1])
            grad_full = np.convolve(w, a[::-1])
            grad = 2.0 * grad_full[n - 1 : n - 1 + n]
            # diagonal correction (subtract w_{2i}·a_i)
            diag_idx = 2 * np.arange(n)
            mask = diag_idx < conv.size
            grad[mask] -= w[diag_idx[mask]] * a[mask]
        else:                                        # low temperature -> top‑k sub‑gradient
            grad = _gradient_topk(a, conv, top_k=top_k)

        # ---- Adam update -----------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = target_sum * 0.5 / n                # crude base step size
        step = step / (np.sqrt(v_hat) + eps)       # Adam‑style scaling

        a_new = a - step * m_hat
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional random shake to escape plateaus
        if it % 4000 == 0 and it > 0:
            a_new += np.random.randn(n) * target_sum * 5e-4
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        mu = max(mu * mu_decay, mu_min)
        it += 1

    return a, best_score, best_a


def _phase2(a: np.ndarray, target_sum: float, deadline: float):
    """Linear‑programming refinement (few LP solves, heavy but powerful)."""
    n = a.shape[0]
    best_a = a.copy()
    best_score = np.max(_convolve(a))
    t = 0.3                      # mixing factor with the LP solution
    min_t = 0.02

    it = 0
    while time.time() < deadline:
        conv = _convolve(a)
        M = conv.max()

        # Build the dense constraint matrix A such that (a∗b)_k = A[k,:]·b
        rows = 2 * n - 1
        A = np.empty((rows, n), dtype=float)
        for j in range(n):
            A[j : j + n, j] = a

        b_ub = np.full(rows, M, dtype=float)
        c = -np.ones(n, dtype=float)                     # maximise Σb  == minimise –Σb
        bounds = [(0.0, 1000.0) for _ in range(n)]        # respects the external caps

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method="highs", options={"presolve": True})
        if not res.success:
            print(f"[LP {it}] linprog failed: {res.message}")
            break

        g0 = res.x
        s = g0.sum()
        if s > 0:
            g = g0 * target_sum / s      # rescale to the required ℓ¹‑norm
        else:
            g = a.copy()

        # convex combination (keeps feasibility)
        a = (1.0 - t) * a + t * g
        a = _project_to_simplex(a, target_sum)   # numerical clean‑up

        new_score = np.max(_convolve(a))
        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[LP {it}] new best = {best_score:.6f}")

        t = max(min_t, t * 0.95)   # slowly reduce the mixing factor
        it += 1

    return a, best_score, best_a


def _final_search(a: np.ndarray, target_sum: float, deadline: float):
    """Cheap block search + tiny pairwise transfers."""
    n = a.shape[0]
    best_a = a.copy()
    best_score = np.max(_convolve(best_a))

    # ----- block‑wise random constructions -----
    block_trials = 200
    for _ in range(block_trials):
        if time.time() >= deadline:
            break
        nb = np.random.randint(2, 7)            # 2‑6 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb)
        cand = np.zeros(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = np.max(_convolve(cand))
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[block] improved = {best_score:.6f}")

    # ----- tiny random pairwise mass moves -----
    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = min(best_a[i], target_sum * 0.001)   # modest move
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = np.max(_convolve(cand))
        if sc < best_score:
            best_score = sc
            best_a = cand
            print(f"[pair] improved = {best_score:.6f}")

    return best_a, best_score
