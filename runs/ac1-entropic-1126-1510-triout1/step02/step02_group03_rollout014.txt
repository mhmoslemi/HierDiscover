def _optimize_l2(a, n, deadline, max_iters=4000, eta0=0.2):
    a = a.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = float('inf')
    L = 2*n - 1
    uniform = np.full(L, 1.0 / L, dtype=np.float64)
    # Precompute FFT length
    m = 1 << ((3 * n - 2).bit_length())
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # conv
        A = np.fft.rfft(a, m)
        conv = np.fft.irfft(A * A, m)[:L]
        # compute score
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n} L2 iter={it} score={best_score:.6f}")
        # error
        e = conv - uniform
        # correlation gradient
        E = np.fft.rfft(e, m)
        corr = np.fft.irfft(np.conj(E) * A, m)
        start = n - 1
        grad = 2.0 * corr[start:start+n]
        # step size
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        # projection onto simplex
        a = _project_onto_simplex(a)
    return best_seq, best_score

    uₖ = 1/(2 n‑1).  The gradient of the loss   ½‖a*a‑u‖²
    is  2·corr(e,a)  with  e = a*a‑u, which can be evaluated by a
    single FFT‑based correlation.  After every step we Euclidean‑project
    back onto the simplex {x≥0, Σx = 1}.

3.  **Fine‑tune the peak** – a simple sub‑gradient method for the
    true objective  maxₖ (a*a)ₖ .  At the current worst shift k* the
    gradient is  gᵢ = 2·a_{k*‑i} (zero outside the support); again we
    step and project.

Both phases use a decreasing step size ηₜ = η₀/√t and occasional
random jitter to escape tiny plateaus.  The loop runs as long as the
global 1000 s budget allows, stopping early if a score ≤ 1.503 is found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on the grid x_i = (i+½)/n.
    Returns a probability vector (sum = 1).
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (α=½) on n points, Σa = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a power‑of‑two FFT.
    Returned length = 2·len(a)‑1.
    """
    n = len(a)
    L = 2 * n - 1
    m = 1 << ((L).bit_length())          # next power of two ≥ L
    A = np.fft.rfft(a, m)
    c = np.fft.irfft(A * A, m)
    return c[:L]


# ----------------------------------------------------------------------
def _l2_optimize(a: np.ndarray,
                 n: int,
                 deadline: float,
                 max_iters: int = 3000,
                 eta0: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Minimise ½‖a*a - uniform‖² by projected gradient descent.
    Returns the best vector seen and its (unsmoothed) score.
    """
    a = a.astype(np.float64).copy()
    a /= a.sum()
    L = 2 * n - 1
    uniform = np.full(L, 1.0 / L, dtype=np.float64)

    # FFT size sufficient for a (len n) and error (len L)
    m = 1 << ((3 * n - 2).bit_length())   # ≥ (2n‑1)+(n‑1)

    best_seq = a.copy()
    best_score = float("inf")
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # convolution a*a
        A = np.fft.rfft(a, m)
        conv = np.fft.irfft(A * A, m)[:L]

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} L2‑iter={it:4d} score={best_score:.6f}")

        # error w.r.t. flat target
        e = conv - uniform

        # gradient = 2·corr(e, a)
        E = np.fft.rfft(e, m)
        corr = np.fft.irfft(np.conj(E) * A, m)
        start = n - 1
        grad = 2.0 * corr[start:start + n]

        # step size, projection
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a)

        # occasional jitter
        if it % 500 == 0 and rng.random() < 0.2:
            a += rng.normal(scale=1e-4, size=n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _max_subgradient_optimize(a: np.ndarray,
                              n: int,
                              deadline: float,
                              max_iters: int = 3000,
                              eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent for the true objective maxₖ (a*a)ₖ.
    Returns the best sequence seen and its score.
    """
    a = a.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = float("inf")
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_full_fft(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} sub‑grad‑iter={it:4d} score={best_score:.6f}")

        k = conv.argmax()                     # shift giving the worst peak

        # gradient of the peak (hard max) : gᵢ = 2·a_{k‑i} where defined
        g = np.zeros_like(a)
        if k < n:                             # left side of the convolution
            g[: k + 1] = 2.0 * a[k::-1]
        else:                                 # right side
            offset = k - (n - 1)
            # slice a[n‑1:offset‑1:-1] is a[n‑1],…,a[offset]
            g[offset:] = 2.0 * a[n - 1 : offset - 1 : -1]

        # step size & projection
        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_onto_simplex(a)

        # jitter to avoid tiny plateaus
        if it % 500 == 0 and rng.random() < 0.2:
            a += rng.normal(scale=1e-4, size=n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a very low
    `evaluate_sequence` score.
    Several lengths are tried; for each we start from a few analytic
    candidates, run an L²‑flattening phase and then a hard‑max
    sub‑gradient phase.  The best sequence found before the time budget
    expires is returned as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  Historic solution, if it exists.
    try:
        hist = np.array(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        hist = np.clip(hist, 0.0, None)
        if hist.sum() > 0.0:
            hist = hist / hist.sum()
            sc = 2.0 * len(hist) * _conv_full_fft(hist).max()
            overall_best_seq = hist.copy()
            overall_best_score = sc
            print(f"Historic sequence: length={len(hist)} score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Lengths to explore – larger n gives more resolution.
    lengths = [2000, 4000, 8000, 12000, 20000, 30000]

    for n in lengths:
        if time.time() >= deadline:
            break

        # allocate a slice of the remaining budget for this length
        remaining = deadline - time.time()
        slice_deadline = time.time() + max(5.0, remaining * 0.5)

        # --------------------------------------------------------------
        # a) generate a modest pool of starter vectors
        starters = []

        # symmetric Beta laws with a range of shape parameters
        alphas = list(rng.uniform(0.15, 1.5, size=8))
        alphas += [0.3, 0.5, 0.7, 1.0, 1.5]
        for a in alphas:
            starters.append(_beta_distribution(n, float(a)))

        # arcsine law, uniform, and a few Dirichlet draws
        starters.append(_arcsine_distribution(n))
        starters.append(np.full(n, 1.0 / n, dtype=np.float64))
        for _ in range(4):
            starters.append(rng.dirichlet(np.ones(n)))

        # --------------------------------------------------------------
        # b) keep the best few initial points (by the raw max‑convolution)
        scores = []
        for cand in starters:
            conv = _conv_full_fft(cand)
            scores.append(2.0 * n * conv.max())
        top_idx = np.argsort(scores)[:2]          # keep the two most promising
        top_candidates = [starters[i] for i in top_idx]

        # --------------------------------------------------------------
        # c) optimise each candidate
        for cand in top_candidates:
            if time.time() >= deadline:
                break

            # ---- L²‑flattening -------------------------------------------------
            a, best_sc = _l2_optimize(
                a=cand,
                n=n,
                deadline=slice_deadline,
                max_iters=3000,
                eta0=0.25,
            )
            # ---- Hard‑max sub‑gradient phase ----------------------------------
            a, best_sc = _max_subgradient_optimize(
                a=a,
                n=n,
                deadline=slice_deadline,
                max_iters=3000,
                eta0=0.6,
            )

            if best_sc < overall_best_score:
                overall_best_score = best_sc
                overall_best_seq = a.copy()
                print(
                    f"*** New global best → length={n:5d}  score={overall_best_score:.6f}"
                )
                if overall_best_score <= 1.503:
                    # target reached – stop early
                    break

        # early global stop if target already met
        if overall_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 2️⃣  Fallback – ensure something is returned
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = 2.0 * n * _conv_full_fft(overall_best_seq).max()

    print(
        f"Final best (approx) score: {overall_best_score:.6f}  length={len(overall_best_seq)}"
    )
    return overall_best_seq.tolist()
