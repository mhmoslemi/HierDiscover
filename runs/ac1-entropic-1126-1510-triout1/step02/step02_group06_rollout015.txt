def lp_optimize(init_seq, t=0.2, max_iter=10, deadline=None):
    import time
    from scipy.optimize import linprog
    a = np.array(init_seq, dtype=np.float64)
    n = len(a)
    best_a = a.copy()
    best_metric = compute_metric(a)
    iteration = 0
    while time.time() < deadline and iteration < max_iter:
        iteration += 1
        # compute convolution
        conv = np.convolve(a, a)
        M = conv.max()
        # Build A
        rows = 2*n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        for i in range(n):
            A[i:i+n, i] = a
        # Linear program: maximize sum(g) -> minimize -sum(g)
        c = -np.ones(n, dtype=np.float64)
        b = np.full(rows, M, dtype=np.float64)
        res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
        if not res.success:
            break
        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break
        g = g0 / sum_g0  # normalize to sum 1
        # Mix
        a = (1 - t) * a + t * g
        metric = compute_metric(a)
        if metric < best_metric:
            best_metric = metric
            best_a = a.copy()
    return best_a.tolist(), best_metric

        A = build_A_sparse(a)
        # LP
        c = -np.ones(n, dtype=np.float64)
        b = np.full(2*n - 1, M, dtype=np.float64)
        res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
        if not res.success:

        sum_g0 = g0.sum()

        # Mix
        a = (1 - t) * a + t * g
        metric = compute_metric(a)
        if metric < best_metric:
            best_metric = metric
            best_a = a.copy()
    return best_a.tolist(), best_metric

def hill_climb(seq, deadline):
    import time
    n = len(seq)
    a = np.array(seq, dtype=np.float64)
    conv = np.convolve(a, a)
    cur_metric = 2 * n * conv.max()
    best_a = a.copy()
    best_metric = cur_metric
    step_factor = 0.2  # fraction of source mass to move each iteration
    min_step = 1e-6
    tolerance = 1e-12
    it = 0
    no_improve = 0
    while time.time() < deadline:
        it += 1
        # compute weight for each index based on max indices
        max_val = conv.max()
        K_max = np.where(conv == max_val)[0]
        weight = np.zeros(n, dtype=int)
        for k in K_max:
            lo = max(0, k - (n-1))
            hi = min(k, n-1)
            weight[lo:hi+1] += 1
        # select source i with max weight, target j with min weight
        i = np.flatnonzero(weight == weight.max())
        i = np.random.choice(i)
        j = np.flatnonzero(weight == weight.min())
        j = np.random.choice(j)
        if i == j:
            # pick random different index
            j = (i + np.random.randint(1, n)) % n
        # Determine transfer amount
        max_transfer = step_factor * a[i]
        if max_transfer < min_step:
            step_factor *= 0.5
            continue
        delta = np.random.rand() * max_transfer
        # copy arrays for trial
        a_new = a.copy()
        conv_new = conv.copy()
        delta_i = -delta
        delta_j = delta
        conv_new[i:i+n] += 2 * delta_i * a
        conv_new[j:j+n] += 2 * delta_j * a
        conv_new[2*i] += delta_i * delta_i
        conv_new[2*j] += delta_j * delta_j
        conv_new[i+j] += 2 * delta_i * delta_j
        a_new[i] += delta_i
        a_new[j] += delta_j
        new_metric = 2 * n * conv_new.max()
        if new_metric < cur_metric - tolerance:
            a = a_new
            conv = conv_new
            cur_metric = new_metric
            no_improve = 0
            if new_metric < best_metric:
                best_metric = new_metric
                best_a = a.copy()
        else:
            no_improve += 1
            # occasionally jitter (random small perturbation)
            if no_improve % 5000 == 0:
                # random jitter: pick two indices and move a small mass
                i1, i2 = np.random.choice(n, 2, replace=False)
                epsilon = 1e-4
                if a[i1] > epsilon:
                    a2 = a.copy()
                    conv2 = conv.copy()
                    a2[i1] -= epsilon
                    a2[i2] += epsilon
                    conv2[i1:i1+n] += 2 * (-epsilon) * a
                    conv2[i2:i2+n] += 2 * epsilon * a
                    conv2[2*i1] += epsilon*epsilon
                    conv2[2*i2] += epsilon*epsilon
                    conv2[i1+i2] += 2 * (-epsilon) * epsilon
                    metric2 = 2 * n * conv2.max()
                    if metric2 < cur_metric:
                        a = a2
                        conv = conv2
                        cur_metric = metric2
                        if metric2 < best_metric:
                            best_metric = metric2
                            best_a = a.copy()
        # gradually reduce step_factor
        if it % 1000 == 0:
            step_factor = max(step_factor * 0.95, min_step)
    return best_a.tolist(), best_metric

"""
Propose a non‑negative step‑function sequence with a low evaluation score.

Algorithm
---------

1. **Start** from a few simple distributions (arcsine, uniform, random Dirichlet)
   normalised to L1‑norm 1 for a chosen length *n*.

2. **Linear‑programming mix** – repeat a few times:
   * Compute M = max(f * f) (the peak of the self‑convolution).
   * Build the Toeplitz‐matrix A that maps a candidate g to the convolution
     (f * g) (A is very sparse: each column i contains the whole vector f
     shifted by i positions).
   * Solve the LP  

        maximise   1ᵀ g
        s.t.      A g ≤ M·1 ,   g ≥ 0

     The optimum g pushes many constraints to the bound M.
   * Normalise g to L1‑norm 1 and replace the current distribution by a small
     convex combination `f ← (1‑t) f + t g` (t≈0.2).  This step provably never
     increases the peak M.

   After a handful of such mixes the peak shrinks substantially.

3. **Fine‑grained hill‑climbing** – starting from the best LP‑mixed sequence,
   repeatedly move a tiny amount of mass from an index that contributes most
   to the peak to an index that contributes least.  The convolution is updated
   via a cheap O(n) incremental formula, so millions of tiny moves can be
   tried within the time budget.

4. Keep the best sequence seen over all tried lengths and return it.

The combination of a global LP step and a local hill‑climbing refinement
often beats plain mirror‑descent and drives the evaluation score well below
the previous 1.5709 barrier.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> list[float]:
    """First n coefficients of (1‑z)^‑1/2, normalised to sum 1."""
    seq = []
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    seq = [x / total for x in seq]
    return seq


# ----------------------------------------------------------------------
def _metric(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for a sequence with L1‑norm 1."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def _build_A_sparse(f: np.ndarray):
    """
    Build the (2n‑1) × n Toeplitz matrix A that maps g → f*g.
    Sparse representation is used because the matrix is very large but
    contains only n² non‑zero entries.
    """
    from scipy.sparse import csr_matrix

    n = f.shape[0]
    rows = 2 * n - 1
    # each column i contributes the whole vector f shifted by i positions
    data = np.tile(f, n)                     # length n*n
    col_idx = np.repeat(np.arange(n), n)     # length n*n
    row_idx = np.concatenate(
        [np.arange(i, i + n, dtype=np.int64) for i in range(n)]
    )
    return csr_matrix((data, (row_idx, col_idx)), shape=(rows, n))


# ----------------------------------------------------------------------
def _lp_mix(
    init_seq: list[float],
    t: float = 0.2,
    max_iter: int = 8,
    deadline: float = 0.0,
) -> tuple[list[float], float]:
    """
    Perform a few LP‑based mixture steps starting from ``init_seq``.
    Returns the best sequence found and its metric.
    """
    from scipy.optimize import linprog

    a = np.array(init_seq, dtype=np.float64)   # L1‑norm = 1
    n = a.shape[0]
    best_a = a.copy()
    best_score = _metric(a)

    iter_cnt = 0
    while time.time() < deadline and iter_cnt < max_iter:
        iter_cnt += 1

        # current peak of the self‑convolution
        conv = np.convolve(a, a)
        M = conv.max()

        # matrix for the linear constraints (f * g) ≤ M
        A = _build_A_sparse(a)

        # LP: maximise 1ᵀ g  <=> minimise –1ᵀ g
        c = -np.ones(n, dtype=np.float64)
        b = np.full(2 * n - 1, M, dtype=np.float64)

        # solve
        res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method="highs")
        if not res.success:
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            break

        # normalise g to have L1‑norm 1 and mix
        g = g0 / sum_g0
        a = (1.0 - t) * a + t * g

        cur_score = _metric(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
def _hill_climb(
    init_seq: list[float],
    deadline: float,
) -> tuple[list[float], float]:
    """
    Tiny‑mass hill‑climbing on the peak of the self‑convolution.
    O(n) incremental convolution updates are used, allowing many trials.
    """
    n = len(init_seq)
    a = np.array(init_seq, dtype=np.float64)   # L1‑norm = 1
    conv = np.convolve(a, a)
    cur_metric = 2.0 * n * conv.max()
    best_a = a.copy()
    best_metric = cur_metric

    # step size as a fraction of the source mass
    step_frac = 0.2
    min_step = 1e-6
    tolerance = 1e-12
    it = 0
    no_improve = 0

    while time.time() < deadline:
        it += 1

        # indices where the peak is attained
        max_val = conv.max()
        K_max = np.where(conv == max_val)[0]

        # how often each a_i participates in a maximal convolution entry
        weight = np.zeros(n, dtype=int)
        for k in K_max:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            weight[lo : hi + 1] += 1

        # pick a source (high weight) and a target (low weight)
        src_candidates = np.flatnonzero(weight == weight.max())
        tgt_candidates = np.flatnonzero(weight == weight.min())
        i = np.random.choice(src_candidates)
        j = np.random.choice(tgt_candidates)
        if i == j:                         # ensure i ≠ j
            j = (i + np.random.randint(1, n)) % n

        # amount to move from i to j
        max_transfer = step_frac * a[i]
        if max_transfer < min_step:
            step_frac = max(step_frac * 0.5, min_step)
            continue
        delta = np.random.rand() * max_transfer   # 0 ≤ delta ≤ max_transfer

        # trial update (copy‑on‑write)
        a_new = a.copy()
        conv_new = conv.copy()
        di = -delta
        dj = delta

        # incremental convolution update
        conv_new[i : i + n] += 2.0 * di * a          # contribution of the changed a_i
        conv_new[j : j + n] += 2.0 * dj * a          # contribution of the changed a_j
        conv_new[2 * i] += di * di
        conv_new[2 * j] += dj * dj
        conv_new[i + j] += 2.0 * di * dj

        a_new[i] += di
        a_new[j] += dj

        new_metric = 2.0 * n * conv_new.max()

        if new_metric < cur_metric - tolerance:
            # accept
            a = a_new
            conv = conv_new
            cur_metric = new_metric
            no_improve = 0
            if new_metric < best_metric:
                best_metric = new_metric
                best_a = a.copy()
        else:
            no_improve += 1
            # occasional random jitter to escape plateaux
            if no_improve % 5000 == 0:
                i1, i2 = np.random.choice(n, 2, replace=False)
                eps = 1e-4
                if a[i1] > eps:
                    a_tmp = a.copy()
                    conv_tmp = conv.copy()
                    a_tmp[i1] -= eps
                    a_tmp[i2] += eps
                    conv_tmp[i1 : i1 + n] += 2.0 * (-eps) * a
                    conv_tmp[i2 : i2 + n] += 2.0 * eps * a
                    conv_tmp[2 * i1] += eps * eps
                    conv_tmp[2 * i2] += eps * eps
                    conv_tmp[i1 + i2] += 2.0 * (-eps) * eps
                    tmp_metric = 2.0 * n * conv_tmp.max()
                    if tmp_metric < cur_metric:
                        a = a_tmp
                        conv = conv_tmp
                        cur_metric = tmp_metric
                        if tmp_metric < best_metric:
                            best_metric = tmp_metric
                            best_a = a.copy()

        # slowly decay the step fraction
        if it % 1000 == 0:
            step_frac = max(step_frac * 0.95, min_step)

    return best_a.tolist(), best_metric

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a good step‑function within the allotted budget.
    Several lengths are tried; for each length a short LP‑mix phase is
    followed by a hill‑climbing refinement.  The best sequence found is
    returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # safety margin

    # lengths that fit comfortably into memory and run fast enough
    lengths = [1500, 2000, 2500, 3000]

    best_seq = None
    best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # three simple starting points
        start_points = [
            arcsine_sequence(n),
            [1.0 / n] * n,                                   # uniform
            (np.random.rand(n) / np.random.rand(n).sum()).tolist(),  # random Dirichlet
        ]

        for idx, init_seq in enumerate(start_points):
            if time.time() >= deadline:
                break

            # split the remaining time roughly equally among the remaining runs
            remaining = deadline - time.time()
            runs_left = len(start_points) - idx
            per_run_deadline = time.time() + max(5.0, remaining / runs_left)

            print(f"\n--- Run {idx + 1}/{len(start_points)} for length {n} ---")

            # 1) LP‑mix phase (very cheap, a few iterations)
            seq_lp, score_lp = _lp_mix(
                init_seq,
                t=0.2,
                max_iter=8,

            # 2) hill‑climbing refinement
            seq_ref, score_ref = _hill_climb(seq_lp, per_run_deadline)
            print(f"Hill‑climb score: {score_ref:.6f}")

            if score_ref < best_score:
                best_score = score_ref
                best_seq = seq_ref
                print(f"*** New overall best (len={n}) score = {best_score:.6f}")

    if best_seq is None:
        # fallback – should never happen
        best_seq = arcsine_sequence(2000)

    # final safety clipping as required by evaluate_sequence
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    print(f"\n--- FINAL RESULT ---\nBest length = {len(best_seq)} with metric = {best_score:.6f}")
    return best_seq
