"""
Propose a step‑function (list of non‑negative heights) that minimises
`evaluate_sequence`.  The algorithm is a fast projected Adam optimiser
that minimises a *smoothed* version of the max self‑convolution.  The
smoothing (log‑sum‑exp) gives a well‑behaved gradient instead of using
only the single active index, which allows the optimiser to escape
local plateaux that limited the previous sub‑gradient method.

Key ideas
---------
* Work on the probability simplex (sum of entries = 1).  The objective
  `evaluate_sequence` is scaling‑invariant, so fixing the sum does not
  affect the score.
* At each iteration compute the self‑convolution `b = a * a` with an FFT
  (O(n log n)).
* Replace `max(b)` by a smoothed max:
      w_k = exp(α (b_k – max(b))) / Σ_j exp(α (b_j – max(b)) )
  and use `∇ = 2·conv(w, a_rev)[(n‑1)−i]` as gradient.

  `η_t = η0 / sqrt(t)` and projection onto the simplex yields fast
  convergence.

  arcsine, uniform, random).  The best sequence seen overall is returned.
* The internal estimate `2·n·max(b)` equals the true

# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, sum(x)=s}
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Discrete arcsine distribution (probability vector)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

# FFT‑based convolution (real inputs)
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the linear convolution x*y using real FFT."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:n]


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Return a*a (self‑convolution) using FFT. Length = 2·len(a)‑1."""
    return _fft_convolve(a, a)

        ϕ_α(a) = log‑sum‑exp(α·(a*a)) / α,
    i.e. a smooth approximation of max(a*a).
    The gradient formula reduces to a convolution of the weight vector
    with the reversed a.
    """
    n = a.shape[0]

    # self‑convolution
    b = _fft_self_convolve(a)                     # length 2n‑1
    max_b = b.max()

    # smoothed weights (log‑sum‑exp trick)
    # subtract max_b for numerical stability
    w = np.exp(alpha * (b - max_b))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum                                 # normalised weights

    # gradient = 2 * conv(w, a_rev)[(n‑1)‑i]
    a_rev = a[::-1]
    conv = _fft_convolve(w, a_rev)               # length = 3n‑2
    idx = (n - 1) - np.arange(n)                 # positions for i = 0..n‑1
    g = 2.0 * conv[idx]
    return g

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score step‑function within the given time budget.
    Returns the best list of non‑negative heights found.
    """
    np.random.seed(seed)
    # Reserve a couple of seconds for finalisation / printing
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # Choose a set of lengths to explore (>= 500)
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        base_n = 8000
    # We explore a small neighbourhood around the historic size
    lengths = [
        max(500, int(base_n * f))
        for f in (0.75, 1.0, 1.25, 1.5)
        if int(base_n * f) >= 500
    ]
    if not lengths:
        lengths = [8000]

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    for n in lengths:
        # allocate roughly a third of the remaining time to the current length
        slice_deadline = time.time() + max(0.1, (deadline - time.time()) * 0.33)

        for init_kind in ("prev", "arcsine", "uniform", "random"):
            a = _initial_sequence(n, init_kind)

            # Adam state
            m = np.zeros_like(a)          # first moment
            v = np.zeros_like(a)          # second moment
            beta1, beta2 = 0.9, 0.999
            eps = 1e-8
            eta0 = 0.8                     # base learning rate
            t = 0
            alpha = 1.0                   # start with gentle smoothing

            # local best for this run (optional, not strictly needed)
            best_local_score = 2.0 * n * _fft_self_convolve(a).max()
            best_local_seq = a.copy()

            while time.time() < slice_deadline:
                t += 1

                # ----------- adaptive smoothing schedule -------------
                # slowly increase α to focus on the true max
                if t % 2000 == 0:
                    alpha = min(30.0, alpha * 1.3)

                # ----------- gradient of smoothed objective -----------
                g = _grad_smoothed_max(a, alpha)

                # ----------- Adam update ------------------------------
                m = beta1 * m + (1.0 - beta1) * g
                v = beta2 * v + (1.0 - beta2) * (g * g)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)

                # diminishing step size
                eta = eta0 / np.sqrt(t)

                a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

                # ----------- projection onto simplex ------------------
                a = _project_onto_simplex(a, s=1.0)

                # occasional jitter to avoid stagnation
                if (t % 500 == 0) and (np.random.rand() < 0.08):
                    a += np.random.randn(n) * 5e-4
                    a = _project_onto_simplex(a, s=1.0)

                # ----------- evaluate internal score ------------------
                b = _fft_self_convolve(a)
                cur_score = 2.0 * n * b.max()

                if cur_score < best_local_score:
                    best_local_score = cur_score
                    best_local_seq = a.copy()
                    # Also update the global best if it improves
                    if cur_score < overall_best_score:
                        overall_best_score = cur_score
                        overall_best_seq = best_local_seq.copy()
                        print(
                            f"[{time.time():.1f}] len={n} init={init_kind} "
                            f"iter={t} score={overall_best_score:.6f}"
                        )
                # early termination if the target is already met
                if overall_best_score <= 1.503:
                    break
            # end while (time slice)

            if overall_best_score <= 1.503:
                break
        # end init loop
        if time.time() >= deadline:
            break
        if overall_best_score <= 1.503:
            break
    # end length loop

    # ------------------------------------------------------------------
    # Fallback in the unlikely case that nothing was produced
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = float(2 * n * _fft_self_convolve(overall_best_seq).max())

    # Final verification with the provided black‑box (optional but cheap)
    try:
        final_score = evaluate_sequence(overall_best_seq.tolist())
        overall_best_score = final_score
    except Exception:
        pass

    print(
        f"Final best score: {overall_best_score:.6f} "
        f"(length = {len(overall_best_seq)})"
    )
    # Return a plain Python list as required by the runner
    return overall_best_seq.tolist()
