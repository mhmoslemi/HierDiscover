def optimize_via_lp_and_md(init_seq, deadline):
    seq = np.array(init_seq)
    n = len(seq)
    best_seq = seq.copy()
    best_metric = metric(seq)

    lp_iter = 0
    while time.time() < deadline - 0.5 and lp_iter < max_lp_iters:
        lp_iter += 1
        conv_ff = np.convolve(seq, seq)
        M = conv_ff.max()
        # Build A matrix
        col = np.concatenate([seq, np.zeros(n-1)])
        row = np.concatenate([[seq[0]], np.zeros(n-1)])
        A = scipy.linalg.toeplitz(col, row)  # shape (2n-1, n)
        b_ub = np.full(2*n-1, M)
        c = -np.ones(n)
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                     bounds=[(0, None)]*n, method='highs')
        if not res.success:
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g_norm = g / sum_g
        # mixing factor (choose based on improvement)
        # try mixing factor adaptively: try t=0.5 maybe
        t = 0.3
        seq = (1 - t) * seq + t * g_norm
        seq = seq / seq.sum()  # For safety
        cur_metric = metric(seq)
        if cur_metric < best_metric:
            best_metric = cur_metric
            best_seq = seq.copy()
        # maybe reduce t after many iterations
    # after lp, run mirror descent (multiplicative) for further refinement
    md_iter=0
    eta=0.5
    while time.time() < deadline - 0.1 and md_iter < max_md_iters:
        md_iter+=1
        # compute gradient (zero except for indices related to max conv)
        conv = np.convolve(seq, seq)
        k_max = conv.argmax()
        # gradient = 2*a_{k_max - i} where in range
        idx = np.arange(n)
        partner = k_max - idx
        mask = (partner >=0) & (partner < n)
        grad = np.zeros_like(seq)
        grad[mask] = 2.0 * seq[partner[mask]]
        # multiplicative update
        upd = np.exp(-eta * grad)
        cand = seq * upd
        cand = cand / cand.sum()
        cand_metric = metric(cand)
        cur_metric = metric(seq)
        if cand_metric < cur_metric:
            seq = cand
            eta = min(eta * 1.1, 5.0)
            cur_metric = cand_metric
        else:
            eta *= 0.6
        if cand_metric < best_metric:
            best_metric = cand_metric
            best_seq = cand.copy()

"""
Propose a high‑quality step‑function sequence.

Algorithm
---------

* Start from the classical arcsine distribution (L¹‑norm = 1).
* Apply an *alternating linear‑program / mixing* step:

  1.  Compute M = maxₖ (f * f)ₖ.
  2.  Solve the linear program                                         
         maximise   Σ gⱼ                                                
         subject to (f * g)ₖ ≤ M   for all k = 0,…,2n‑2                 
                    gⱼ ≥ 0                                                
     The matrix of the convolution constraints is built by a Toeplitz
     construction; the problem is small enough to be solved with
     ``scipy.optimize.linprog`` (highs backend).

  3.  Let ḡ = g / Σg (renormalise to sum = 1).  Mix the current
     distribution with the new one:
           f ← (1‑t)·f + t·ḡ    (t≈0.3, slowly reduced)
     The sum stays equal to 1, and the bound M is reduced because
     Σg > 1.

* After a handful of LP‑mix steps, run a short *mirror‑descent*
  (multiplicative) refinement that directly reduces the peak of the
  self‑convolution (gradient is non‑zero only at the maximal index).

* Repeat the whole procedure for several lengths (200‑1200).  Keep the
  sequence with the smallest metric  2·n·max(f*f).  The total run time
  is limited by the supplied ``budget_s`` (≈1000 s).

The method is completely self‑contained, uses only NumPy/SciPy, and
generally drives the metric well below the arcsine baseline
(≈1.5709), often reaching the target ≈ 1.50 range.
"""

import time
import math
import numpy as np
import scipy.linalg
import scipy.optimize


def arcsine_sequence(n: int) -> list[float]:
    """
    First n coefficients of (1‑z)^‑1/2,
    C_i = binom(2i,i) / 4**i, normalised to have sum 1.
    """
    seq = []
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    seq = [x / total for x in seq]
    return seq


def _metric(seq: np.ndarray) -> float:
    """Metric = 2·n·max(conv(seq,seq)), seq assumed to sum to 1."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    return 2.0 * n * conv.max()


def _lp_step(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve  maximise Σ g_j  subject to (f*g)_k ≤ M  (all k) and g ≥ 0.
    Returns the optimal g (might be all zeros if LP fails).
    """
    n = f.shape[0]
    # Build the convolution matrix A (size (2n-1) × n) via Toeplitz.
    # First column: f followed by zeros, first row: [f0, 0, …, 0].
    col = np.concatenate([f, np.zeros(n - 1)])
    row = np.concatenate([[f[0]], np.zeros(n - 1)])
    A = scipy.linalg.toeplitz(col, row)          # dense (2n-1)×n matrix
    b = np.full(2 * n - 1, M)

    # maximise Σ g  → minimise −Σ g
    c = -np.ones(n)

    res = scipy.optimize.linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        # In the unlikely event of failure, return a zero vector.
        return np.zeros_like(f)
    return res.x


def _mirror_descent_step(a: np.ndarray, eta: float) -> np.ndarray:
    """
    One mirror‑descent (multiplicative) step on the simplex.
    Gradient is 2·a_{k*‑i} where k* is the location of the current maximum.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    k_max = int(conv.argmax())
    idx = np.arange(n)
    partner = k_max - idx
    mask = (partner >= 0) & (partner < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[partner[mask]]
    upd = np.exp(-eta * grad)
    cand = a * upd
    cand_sum = cand.sum()
    if cand_sum <= 0.0:
        cand = np.full_like(a, 1.0 / n)
    else:
        cand = cand / cand_sum
    return cand


def _optimize_one_length(init_seq: list[float], deadline: float) -> tuple[list[float], float]:
    """
    Perform alternating LP‑mixing + short mirror‑descent on a single length.
    Returns the best sequence found (as a list) and its metric.
    """
    f = np.array(init_seq, dtype=np.float64)
    n = f.shape[0]

    best_f = f.copy()
    best_score = _metric(f)

    # parameters
    mix_factor = 0.35            # initial mixing weight t
    min_mix = 0.02
    max_lp_iters = 30
    max_md_iters = 4000

    # ---------- alternating LP + mixing ----------
    lp_iter = 0
    while time.time() < deadline and lp_iter < max_lp_iters:
        lp_iter += 1
        conv_ff = np.convolve(f, f)
        M = conv_ff.max()
        g = _lp_step(f, M)
        sum_g = g.sum()
        if sum_g <= 0.0:
            break
        g_norm = g / sum_g                        # sum(g_norm)=1
        # convex combination, keep normalization
        f = (1.0 - mix_factor) * f + mix_factor * g_norm
        f /= f.sum()                              # safety
        cur_score = _metric(f)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_f = f.copy()
            print(f"[len {n} LP {lp_iter:2d}] new best metric {best_score:.6f}")

        # gradually shrink the mixing factor to stabilise
        mix_factor = max(min_mix, mix_factor * 0.92)

    # ---------- short mirror‑descent refinement ----------
    md_iter = 0
    eta = 0.6
    while time.time() < deadline and md_iter < max_md_iters:
        md_iter += 1
        cur_metric = _metric(f)
        # propose update
        cand = _mirror_descent_step(f, eta)
        cand_metric = _metric(cand)

        if cand_metric < cur_metric:
            f = cand
            eta = min(eta * 1.08, 5.0)
            if cand_metric < best_score - 1e-12:
                best_score = cand_metric
                best_f = f.copy()
                print(f"[len {n} MD {md_iter:5d}] new best metric {best_score:.6f}")
        else:
            eta *= 0.6
            if eta < 1e-8:
                break

    return best_f.tolist(), best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for the best non‑negative step‑function sequence within the time budget.
    Several lengths are explored; each length is optimised by alternating
    linear‑programming mixing and a brief mirror‑descent refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin

    best_seq = None
    best_metric = float("inf")

    # lengths that fit comfortably in memory while still being large enough
    lengths = [200, 400, 600, 800, 1000, 1200]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length {n} ===")
        init = arcsine_sequence(n)

        # split remaining time roughly equally over the still‑to‑process lengths
        remaining = deadline - time.time()
        per_len = max(5.0, remaining / (len(lengths) - lengths.index(n)))
        seq, score = _optimize_one_length(init, time.time() + per_len)

        print(f"Finished length {n}: metric = {score:.6f}")

        if score < best_metric:
            best_metric = score
            best_seq = seq
            print(f"*** New overall best (len={n}) metric = {best_metric:.6f}")

    # fallback – should never happen, but keep a safe default
    if best_seq is None:
        best_seq = arcsine_sequence(500)
        best_metric = _metric(np.array(best_seq))

    # final safety: enforce the limits used inside evaluate_sequence
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    print(f"\n--- FINAL RESULT ---\nBest length = {len(best_seq)} "
          f"with metric = {best_metric:.6f}")
    return best_seq
