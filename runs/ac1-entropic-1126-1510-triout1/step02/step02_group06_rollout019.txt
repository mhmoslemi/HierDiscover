def anneal_sequence(a_init, deadline):
    a = a_init.copy()
    n = a.shape[0]
    cur_metric = compute_metric(a)
    best_a = a.copy()
    best_metric = cur_metric

    # parameters
    T = 0.01  # initial temperature
    Tmin = 1e-8
    alpha = 0.9995  # cooling factor per iteration (adjust)
    max_step = 0.1  # maximal mass to transfer per move (initial)
    step_factor = 0.999  # reduce step gradually

    it = 0
    while time.time() < deadline:
        it += 1
        # random indices
        i, j = np.random.choice(n, 2, replace=False)
        # propose delta
        delta = (np.random.rand() * 2 - 1) * max_step
        # clamp to keep nonnegative
        lo = -a[i]  # cannot decrease below zero
        hi = a[j]   # cannot increase above available mass at j
        # Actually we move mass from j to i: delta positive means add delta to i and subtract from j.
        delta = np.clip(delta, lo, hi)
        if abs(delta) < 1e-12:
            continue

        a_new = a.copy()
        a_new[i] += delta
        a_new[j] -= delta

        # compute new metric
        new_metric = compute_metric(a_new)

        # acceptance criteria: if improvement accept, else accept w.p. exp(-deltaMetric/T)
        accept = False
        if new_metric < cur_metric:
            accept = True
        else:
            # probability
            prob = np.exp(-(new_metric - cur_metric) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_new
            cur_metric = new_metric
            # update best
            if cur_metric < best_metric:
                best_metric = cur_metric
                best_a = a.copy()
                # maybe print progress
                if it % 1000 == 0:
                    print(f"[iter {it}] New best metric {best_metric:.6f}")

        # cooling schedule
        T = max(T * alpha, Tmin)
        # step schedule
        max_step *= step_factor
        if max_step < 1e-6:
            max_step = 1e-6

        # if cur_metric close to best (no improvement for many iterations) maybe break
        # Not needed.

    return best_a.tolist(), best_metric

    # arcsin sqrt x
    y = np.arcsin(np.sqrt(x))
    a = (2.0 / np.pi) * (y[1:] - y[:-1])

def smooth_descent(a_init, deadline, beta_start=0.5, beta_end=100.0, eta_start=0.5, max_iter=10**6):
    a = a_init.copy()
    n = len(a)
    m1 = 1 << ((2*n-1).bit_length())
    m2 = 1 << ((3*n-2).bit_length())
    best_a = a.copy()
    cur_metric = compute_metric(a)
    best_metric = cur_metric
    eta = eta_start
    beta = beta_start
    # schedule: increase beta linearly
    while time.time() < deadline and iteration < max_iter:
        # conv of a*a using FFT
        A = np.fft.rfft(a, n=m1)
        conv = np.fft.irfft(A * A, n=m1)[:2*n-1]
        max_conv = conv.max()
        metric = 2.0 * n * max_conv
        if metric < best_metric - 1e-12:
            best_metric = metric
            best_a = a.copy()
            if iteration % 1000 == 0:
                print(f"[{iteration}] New best metric {best_metric:.6f}")

        # smooth weights
        w = np.exp(beta * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient via correlation
        a_rev = a[::-1]
        W_fft = np.fft.rfft(w_norm, n=m2)
        A_rev_fft = np.fft.rfft(a_rev, n=m2)
        corr = np.fft.irfft(W_fft * A_rev_fft, n=m2)
        grad = 2.0 * corr[n-1: n-1+n]  # length n

        # multiplicative mirror descent step
        # clip gradient exponent
        upd = np.exp(-eta * grad)
        cand = a * upd
        sum_cand = cand.sum()
        if sum_cand > 0:
            cand = cand / sum_cand
        else:
            cand = np.full_like(a, 1.0 / n)

        # compute metric of cand
        A_cand = np.fft.rfft(cand, n=m1)
        conv_cand = np.fft.irfft(A_cand * A_cand, n=m1)[:2*n-1]
        metric_cand = 2.0 * n * conv_cand.max()

        # acceptance
        if metric_cand < metric:
            a = cand
            metric = metric_cand
            # possibly increase eta
            eta = min(eta * 1.05, 10.0)
        else:
            eta = eta * 0.7
            # occasional random jitter when stuck
            if iteration % 5000 == 0:
                # random small move
                i, j = np.random.choice(n, 2, replace=False)
                delta = (np.random.rand() - 0.5) * 1e-3
                delta = np.clip(delta, -a[i], a[j])
                a[i] += delta
                a[j] -= delta
                a = np.maximum(a, 0)
                a = a / a.sum()

        # increase beta gradually up to limit
        beta = min(beta_end, beta * 1.02)

        iteration += 1

        # break if eta too small
        if eta < 1e-10:
            break

    return best_a.tolist(), best_metric

High‑precision search for a low‑evaluation step‑function.

The algorithm proceeds in three stages:

1.  Initialise a probability vector from the continuous arcsine law.
    The exact integral of 1/(π√(x(1‑x))) over each bin yields a more
    accurate discrete representation than the classic binomial formula.

2.  Perform a smoothed mirror‑descent on the objective
        f(a) = max_k (a*a)[k]               (self‑convolution peak)
    The max is approximated by a soft‑maximum

    Starting with a small β we obtain a global, smooth direction;
    β is increased gradually so that the method finally behaves like
    the standard “max‑gradient’’ descent.  The update is multiplicative
    (mirror descent on the simplex) and the step‑size η is adapted
    automatically.  Random jitter is injected when progress stalls.

3  The whole routine is run for several lengths (powers of two are used
    for FFT efficiency).  The best sequence (smallest evaluation score)
    found within the time budget is returned.

All computations stay in the probability simplex (sum = 1), therefore
the scaling invariance of the original `evaluate_sequence` is respected.
The final list is clipped to the limits enforced by the evaluator.

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small utility: next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def arcsine_integral_seq(n: int) -> np.ndarray:
    """
    Discrete approximation of the arcsine density
        f(x) = 1/(π·√(x·(1‑x)))   on [0,1]
    by integrating the density over each bin of width 1/n.
    The resulting probabilities sum to 1.
    """
    # points x_i = i/n, i = 0..n
    x = np.arange(n + 1, dtype=np.float64) / n
    y = np.arcsin(np.sqrt(x))                     # integral of 1/√(x(1‑x))
    a = (2.0 / np.pi) * (y[1:] - y[:-1])         # difference gives bin mass
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total <= 0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
def _metric_of(seq: np.ndarray) -> float:
    """
    Evaluation metric for a probability vector (sum ≈ 1):
        2·n·max(conv(seq,seq))
    The convolution is performed by FFT for speed.
    """
    n = seq.shape[0]
    m = _next_pow2(2 * n - 1)
    A = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(A * A, n=m)[: 2 * n - 1]
    max_conv = conv.max()
    return float(2.0 * n * max_conv)


# ----------------------------------------------------------------------
def _smooth_descent(
    init_seq: np.ndarray,
    deadline: float,
    beta_start: float = 0.5,
    beta_end: float = 80.0,
    eta_start: float = 0.5,
    max_iter: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Smoothed mirror‑descent on the simplex.
    The soft‑max temperature β grows from ``beta_start`` to ``beta_end``.
    Returns the best vector seen and its metric.
    """
    a = init_seq.copy()
    n = a.shape[0]
    m1 = _next_pow2(2 * n - 1)            # length for a*a convolution
    m2 = _next_pow2(3 * n - 2)            # length for correlation (w * a_rev)

    best_a = a.copy()
    best_score = _metric_of(a)

    eta = eta_start
    beta = beta_start
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        # ----- current metric -------------------------------------------------
        cur_score = _metric_of(a)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            if it % 500 == 0:
                print(f"[iter {it:6d}] new best metric {best_score:.9f}")
        else:
            no_improve += 1

        # ----- convolution a*a ------------------------------------------------
        A = np.fft.rfft(a, n=m1)
        conv = np.fft.irfft(A * A, n=m1)[: 2 * n - 1]
        max_conv = conv.max()

        # ----- smooth weights (soft max) ------------------------------------
        # w_k ∝ exp(β·(conv_k−max_conv)), max weight = 1
        w = np.exp(beta * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # ----- gradient = 2·corr(w_norm, a) ---------------------------------
        a_rev = a[::-1]
        W_fft = np.fft.rfft(w_norm, n=m2)
        A_rev_fft = np.fft.rfft(a_rev, n=m2)
        corr_full = np.fft.irfft(W_fft * A_rev_fft, n=m2)
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]   # length n

        # ----- mirror‑descent step (multiplicative) -------------------------
        # protect exponent overflow/underflow
        upd = np.exp(-eta * np.clip(grad, -50, 50))
        cand = a * upd
        s = cand.sum()
        if s > 0:
            cand /= s
        else:
            cand = np.full_like(a, 1.0 / n)

        cand_score = _metric_of(cand)

        # ----- accept / reject ------------------------------------------------
        if cand_score < cur_score:
            a = cand
            eta = min(eta * 1.08, 5.0)   # slight increase after success
        else:
            eta *= 0.65
            # occasional random jitter to escape flat regions
            if no_improve % 2000 == 0:
                i, j = np.random.choice(n, 2, replace=False)
                delta = (np.random.rand() - 0.5) * 1e-3   # tiny mass shift
                # keep feasibility
                delta = np.clip(delta, -a[i], a[j])
                a[i] += delta
                a[j] -= delta
                a = np.maximum(a, 0.0)
                a /= a.sum()

        # ----- anneal β (focus more on the true maximum) --------------------
        beta = min(beta_end, beta * 1.015)

        # safety break
        if eta < 1e-10:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Search for the best non‑negative step‑function sequence within the
    allotted time.  Several lengths are examined; each length is
    optimised starting from a high‑quality arcsine approximation using
    smoothed mirror‑descent.  The sequence with the smallest evaluation
    score is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin

    # lengths chosen as powers of two (FFT‑friendly)
    lengths = [1024, 2048, 4096, 5000]       # 5000 is not a power of two but still fine
    best_seq = None
    best_score = float("inf")

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # --- initialise from the exact arcsine integral --------------------
        a0 = arcsine_integral_seq(n)

        # allocate time roughly equally among remaining lengths
        remaining = deadline - time.time()
        parts_left = len(lengths) - idx
        per_len_budget = max(5.0, remaining / parts_left)

        # run the smoothed descent
        seq, score = _smooth_descent(
            a0,
            deadline=time.time() + per_len_budget,
            beta_start=0.5,
            beta_end=80.0,
            eta_start=0.5,
        )
        print(f"Finished length {n}: metric = {score:.9f}")

        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"*** New overall best (len={n}) score = {best_score:.9f}")

    # ----------------------------------------------------------------------
    if best_seq is None:
        # fallback – should never happen, but keep a safe default
        best_seq = arcsine_integral_seq(2000)

    # Clip to the limits enforced by evaluate_sequence (non‑negative, ≤1000)
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    print(
        f"\n--- FINAL RESULT ---\nBest length = {len(best_seq)} "
        f"with metric = {best_score:.9f}"
    )
    return best_seq
