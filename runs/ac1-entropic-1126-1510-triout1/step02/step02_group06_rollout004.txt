"""
Propose a low‑value step‑function sequence.

The evaluation metric is

    score = 2 * n * max(a * a) / (sum(a) ** 2)

where ``a * a`` is the discrete self‑convolution.  Because the score is
scale‑invariant we can restrict ourselves to probability vectors
(``sum(a) == 1``) and minimise the peak of the convolution.

The algorithm works as follows:

1.  Try a handful of sensible initial shapes (the classical monotone
    arcsine, a symmetric arcsine, and a few Beta‑type “U‑shaped”
    distributions).  All are normalised to have L1‑norm 1.

2.  For each initial vector run a projected‑gradient descent on a
    *smooth* approximation of the max‑function:
    ```
        fβ(a) = (1/β) * log Σ_k exp(β * (a*a)_k)
    ```
    When ``β`` is small the objective is almost the mean of the
    convolution, while large ``β`` approaches the true max.
    β is increased slowly during optimisation (annealing).

3.  The gradient of ``fβ`` is
    ```
        ∂fβ/∂a_i = 2 * Σ_k w_k a_{k-i},
        w_k = exp(β * (a*a)_k) / Σ_j exp(β * (a*a)_j)
    ```
    which can be evaluated efficiently by a single correlation
    (`np.correlate`).  An Adam‑style update is applied and the result
    is projected back onto the simplex (non‑negative entries summing to
    one).

4.  The best sequence found for any length is kept and finally returned.

The whole search fits comfortably into the 1000 s budget; it uses only
NumPy and a few hundred thousand cheap iterations, yet the annealed
smooth‑max plus Adam+simplex projection typically pushes the metric
below the old value (π/2 ≈ 1.5708) and often reaches the target
≈ 1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho_arr = np.nonzero(u > cssv / np.arange(1, n + 1))[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> np.ndarray:
    """Classic monotone arcsine coefficients C_i = binom(2i,i)/4^i (L1=1)."""
    seq = np.empty(n, dtype=np.float64)
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq[i] = cur
        total += cur
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    seq /= total
    return seq


# ----------------------------------------------------------------------
def symmetric_arcsine_sequence(n: int) -> np.ndarray:
    """Symmetric arcsine a_i ∝ 1 / sqrt( (i+0.5)*(n-0.5-i) )."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-14
    seq = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
def beta_u_shape_sequence(n: int, alpha: float) -> np.ndarray:
    """
    U‑shaped beta distribution a_i ∝ ((i+1)*(n-i))**alpha.
    Negative alpha concentrates mass near the ends.
    """
    i = np.arange(n, dtype=np.float64)
    seq = ((i + 1) * (n - i)) ** alpha
    seq = np.where(seq > 0, seq, 0.0)
    if seq.sum() == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
def smooth_max_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute soft‑max weights w_k = exp(beta * conv_k) / Σ exp(beta * conv).
    The subtraction of the max value keeps the exponentials stable.
    """
    shift = conv.max()
    exp_w = np.exp(beta * (conv - shift))
    total = exp_w.sum()
    if total == 0.0:
        # fallback – uniform weights
        return np.full_like(conv, 1.0 / conv.size)
    return exp_w / total


# ----------------------------------------------------------------------
def gradient_smooth_max(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth‑max objective.
    Returns a vector of length n (same as a).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)                     # length 2n‑1
    w = smooth_max_weights(conv, beta)           # same length
    # correlation a ⋆ w gives Σ_j a_j w_{i+j}
    corr = np.correlate(a, w, mode='full')
    offset = len(w) - 1
    grad = 2.0 * corr[offset: offset + n]
    return grad, conv.max()


# ----------------------------------------------------------------------
def adam_simplex_optimize(
    a0: np.ndarray,
    deadline: float,
    beta_max: float = 40.0,
    max_iter: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max objective with simplex projection.
    Returns the best sequence seen and its (non‑smooth) score
    2 * n * max(conv).
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # current best
    conv = np.convolve(a, a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()

    # beta schedule (annealing)
    beta = 1.0
    beta_increment = (beta_max / beta) ** (1.0 / max_iter)

    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        # gradient of smooth max
        grad, cur_max = gradient_smooth_max(a, beta)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule – slowly decreasing
        lr = 0.05 / (np.sqrt(t))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = project_onto_simplex(a_candidate)

        # evaluate the true (non‑smooth) metric
        conv_candidate = np.convolve(a_candidate, a_candidate)
        score_candidate = 2.0 * n * conv_candidate.max()

        # accept if improvement (otherwise keep current a)
        if score_candidate < best_score:
            best_score = score_candidate
            best_a = a_candidate.copy()
        # always move – Adam already incorporates momentum; we keep the
        # candidate for the next iteration.
        a = a_candidate

        # increase beta (approach true max)
        if beta < beta_max:
            beta = min(beta * beta_increment, beta_max)

        # occasional logging
        if t % 2000 == 0:
            print(
                f"[n={n:4d} iter={t:6d}] "
                f"beta={beta:.2f} score={best_score:.6f}"
            )
    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a step‑function sequence with a low evaluation score.
    The function respects the global 1000 s time limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1    # small safety margin

    # Lengths to try – a trade‑off between resolution and speed.
    lengths = [1400, 1800, 2200, 2600, 3000]

    # Stores the overall best result.
    best_seq = None
    best_score = float("inf")

    # Helper to generate a small pool of starting points for a given n.
    def initial_pool(n: int) -> list[np.ndarray]:
        pool = []
        pool.append(arcsine_sequence(n))
        pool.append(symmetric_arcsine_sequence(n))
        for alpha in (-0.6, -0.4, -0.2):
            pool.append(beta_u_shape_sequence(n, alpha))
        # a few random Dirichlet vectors (very sparse)
        for _ in range(2):
            pool.append(np.random.dirichlet(np.full(n, 0.1)))
        return pool

    # Main loop over lengths.
    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break
        print(f"\n--- Optimising length {n} ---")
        remaining = deadline - time.time()
        # allocate roughly equal time for each remaining length
        per_len_deadline = time.time() + max(5.0, remaining / (len(lengths) - idx))

        # generate several initial candidates
        candidates = initial_pool(n)

        # optimise each candidate (share the same per‑length budget)
        per_cand_deadline = per_len_deadline - time.time()
        per_cand_budget = per_cand_deadline / len(candidates) if candidates else 0.0

        for cand_idx, cand in enumerate(candidates):
            if time.time() >= per_len_deadline:
                break
            cand_deadline = time.time() + per_cand_budget
            print(f"  candidate {cand_idx+1}/{len(candidates)} (n={n})")
            seq, score = adam_simplex_optimize(cand, cand_deadline)
            print(f"    candidate best score = {score:.6f}")
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"    *** New overall best (len={n}) score = {best_score:.6f}")

    # Fallback in the unlikely case we never produced a candidate.
    if best_seq is None:
        best_seq = arcsine_sequence(2000)

    # Apply the clipping that evaluate_sequence enforces (harmless but tidy).
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    print(
        f"\n=== FINAL RESULT ===\n"
        f"Length = {len(best_seq)}  score = {best_score:.6f}"
    )
    return best_seq
