"""
propose_candidate
=================

Search for a non‑negative step‑function (a probability vector) whose
self‑convolution peak  max(a*a)  is as small as possible.
The evaluation metric is

    score = 2 * n * max(a*a)          (the sum of a is kept equal to 1)

The algorithm combines three ideas:

1. **Initialization** – start from the arcsine distribution (the classic
   “optimal” shape) or from the previously found global `height_sequence_1`,
   whichever gives the lower score.  Length is capped at ``MAX_N`` (≈2500)
   to keep the linear programs tractable.

2. **Alternating linear‑programming step** – with a fixed sequence `f`,
   the convolution constraint  (f * g)[k] ≤ M  is linear in `g`.  We solve

        maximise    1ᵀ g
        subject to  g ≥ 0,
                    f * g ≤ M,

   using ``scipy.optimize.linprog`` (HiGHS solver).  The optimum `g`
   tries to fill all slack while keeping the same maximal convolution `M`.
   A convex combination ``f ← (1‑t)·f + t·ĝ`` (where ``ĝ`` is `g` normalised
   to sum = 1) strictly reduces the peak.  The mixing weight `t` decays
   geometrically.

3. **Projected soft‑max gradient descent** – after the LP phase (or when
   the LP gives no improvement) we minimise a smooth surrogate of
   ``max(conv)`` :

        F(f) = (1/β)·log Σₖ exp(β·convₖ) ,

   whose gradient is  ``∇F = 2·conv(w, f_rev)`` with soft‑max weights
   ``wₖ = exp(β·(convₖ−max(conv))) / Σ exp``.
   An adaptive back‑tracking line‑search ensures a decrease of ``F``.
   After every step we project back onto the probability simplex.

The two phases are interleaved until the time budget expires.  The best
sequence encountered (lowest score) is returned.  The implementation
uses only NumPy, SciPy and the provided ``evaluate_sequence`` for a final
sanity check.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> np.ndarray:
    """
    Return an n‑point discretisation of the arcsine density on [0,1],
    normalised so that the sum of the entries equals 1.
    Recurrence C_{i+1}=C_i·(2i+1)/(2(i+1)) is used for stability.
    """
    seq = np.empty(n, dtype=float)
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq[i] = cur
        total += cur
        if i + 1 < n:
            cur = cur * (2.0 * i + 1.0) / (2.0 * (i + 1))
    seq /= total
    return seq


# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σ x = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("target sum s must be non‑negative")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A such that (A @ g) = conv(f, g)  (full linear convolution
    of length 2·n−1).  A has shape (2·n‑1, n) and is dense.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    # each column i creates a shifted copy of f[i]
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


# ----------------------------------------------------------------------
def lp_step(f: np.ndarray, M: float, mix_weight: float):
    """
    Perform one LP‑based refinement step:
        maximise    1ᵀ g
        subject to  g ≥ 0,
                    conv(f, g) ≤ M .
    Returns the updated sequence (projected onto the simplex) and a flag
    indicating whether the LP succeeded and produced an improvement.
    """
    try:
        from scipy.optimize import linprog
    except Exception:
        return f, False

    n = f.shape[0]

    # Build the convolution matrix for the current f
    A = build_conv_matrix(f)

    # Linear program data
    c = -np.ones(n)                     # maximise sum(g)  → minimise -sum(g)
    b_ub = M * np.ones(2 * n - 1)       # RHS of the inequality constraints
    bounds = [(0, None)] * n

    # Solve with HiGHS (fast, robust)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})

    if not res.success or res.x is None:
        return f, False

    g = np.asarray(res.x, dtype=float)
    sum_g = g.sum()
    if sum_g <= 0:
        return f, False

    # Normalise g to sum = 1 (the metric is scale‑invariant)
    g_norm = g / sum_g

    # Convex combination with the current sequence
    new_f = (1.0 - mix_weight) * f + mix_weight * g_norm

    # Project back onto the probability simplex
    new_f = project_simplex(new_f, s=1.0)

    return new_f, True


# ----------------------------------------------------------------------
def gradient_step(f: np.ndarray, step: float, beta: float):
    """
    Perform a soft‑max gradient descent step on the smooth surrogate of
    max(conv).  Returns the new sequence (projected onto the simplex) and
    the achieved decrease flag.
    """
    n = f.shape[0]

    # self‑convolution
    conv = np.convolve(f, f)
    max_val = conv.max()

    # soft‑max weights (stable)
    exp_vals = np.exp(beta * (conv - max_val))
    w = exp_vals / exp_vals.sum()

    # gradient = 2 * conv(w, f[::-1])[n‑1:2·n‑1]
    grad_full = np.convolve(w, f[::-1])
    grad = 2.0 * grad_full[n - 1:2 * n - 1]

    # Back‑tracking line search (Armijo)
    f_new = None
    improved = False
    cur_score = 2.0 * n * max_val   # because sum(f)≈1
    for _ in range(12):   # at most 12 reductions of the step
        cand = f - step * grad
        cand = project_simplex(cand, s=1.0)

        # compute surrogate (soft‑max) value for the candidate
        cand_conv = np.convolve(cand, cand)
        cand_max = cand_conv.max()
        cand_score = 2.0 * n * cand_max

        if cand_score < cur_score - 1e-12:
            f_new = cand
            improved = True
            break
        step *= 0.5

    if not improved:
        # fallback: keep the original vector (no change)
        f_new = f.copy()

    return f_new, improved, step
