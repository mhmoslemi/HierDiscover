def conv_max(a):
    n = a.shape[0]
    # Determine fft size
    n_fft = 1 << ((2*n-1).bit_length())
    # Use real FFT (rfft) for speed.
    A = np.fft.rfft(a, n_fft)
    conv = np.fft.irfft(A * A, n_fft)[:2*n-1]
    return conv.max()
```

def softmax_grad(a, conv, mu):
    cur_max = conv.max()
    diff = conv - cur_max   # <=0
    w = np.exp(diff / mu)
    w_sum = w.sum()
    w /= w_sum

    sum_term = conv_wr[len(a)-1 : len(a)-1 + len(a)]  # from n-1 index to n-1 + n-1 inclusive
    grad = 2.0 * sum_term
    # No subtraction
    return grad

    # sum_term is indices n-1 to n-1 + n - 1 inclusive (len n)

    return grad

"""
Propose a low‑score step‑function.

The algorithm proceeds in three stages:

1. **Broad exploration** – generate many symmetric block‑wise candidates,
   and a few parametric families (uniform, exponential, power‑law).  All
   vectors are projected onto the simplex {a≥0, Σa = √(2 n)}.  The best
   few are kept.

2. **Projected Adam optimisation** – starting from the current best
   candidate we run an Adam‑style projected gradient descent.
   The gradient is taken either from a smooth soft‑max surrogate of the
   max‑convolution (high temperature) or from a top‑k sub‑gradient
   (low temperature).  The temperature μ is cooled geometrically.
   After each step the iterate is projected back onto the simplex and
   symmetrised (a = (a+a[::-1])/2) to keep the search space small.

3. **Local random swaps** – finally a cheap hill‑climbing phase moves a
   small amount of mass between two random positions, accepting the move
   only if it improves the max‑convolution.  This often removes tiny
   remaining peaks.

The whole procedure respects the 1000 s wall‑clock budget, prints a few
status messages and returns the best sequence found as a plain Python
list.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force a vector to be symmetric (a_i = a_{n-1-i})."""
    return (v + v[::-1]) * 0.5


# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, n_fft: int) -> np.ndarray:
    """Full self‑convolution a*a using FFT (length 2·len(a)-1)."""
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * a.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray, n_fft: int) -> float:
    """Maximum entry of a*a."""
    return float(_full_convolution_fft(a, n_fft).max())


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      mu: float,
                      n_fft_grad: int) -> np.ndarray:
    """
    Gradient of the smooth soft‑max surrogate μ·log Σ exp(conv_k/μ).

    Returns ∇ Σ w_k·conv_k where w_k ∝ exp(conv_k/μ) and Σ w_k = 1.
    """
    cur_max = conv.max()
    # shifted soft‑max for numerical stability
    w = np.exp((conv - cur_max) / mu)
    w /= w.sum()

    # convolution w * a_rev (a reversed)
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=n_fft_grad)
    AR = np.fft.rfft(a_rev, n=n_fft_grad)
    conv_wr = np.fft.irfft(W * AR, n=n_fft_grad)[: len(w) + a.shape[0] - 1]

    n = a.shape[0]
    sum_term = conv_wr[n - 1 : n - 1 + n]          # length n slice
    grad = 2.0 * sum_term
    return grad


# ----------------------------------------------------------------------
def _gradient_topk(a: np.ndarray, conv: np.ndarray, top_k: int = 10) -> np.ndarray:
    """
    Sub‑gradient based on the `top_k` largest convolution entries.
    Identical to the helper used in the reference implementation.
    """
    n = a.shape[0]
    idxs = np.argpartition(-conv, top_k - 1)[:top_k]
    grad = np.zeros_like(a)
    i = np.arange(n)
    for k in idxs:
        j = k - i
        mask = (j >= 0) & (j < n)
        grad[mask] += 2.0 * a[j[mask]]
        diag = mask & (i == j)
        grad[diag] -= a[i[diag]]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns the best sequence found (as a plain list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin

    # --------------------------------------------------------------
    # Length of the problem (fallback to 3000 if the reference is absent)
    try:
        n = len(height_sequence_1)         # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)       # exact ℓ¹‑norm used in the score

    # FFT sizes (constant throughout the run)
    n_fft_conv = 1 << ((2 * n - 1).bit_length())      # for a*a
    n_fft_grad = 1 << ((3 * n - 2).bit_length())      # for w * a_rev

    # --------------------------------------------------------------
    # ---------- 0. INITIAL CANDIDATE POOL -------------------------
    candidates = []

    # reference solution (if available)
    if have_ref:
        a_ref = np.asarray(height_sequence_1, dtype=float)
        a_ref = _project_to_simplex(a_ref, target_sum)
        a_ref = _symmetrize(a_ref)
        candidates.append(a_ref)

    # uniform vector
    candidates.append(np.full(n, target_sum / n))

    # exponential decays (symmetric)
    for rate in [0.0005, 0.001, 0.002, 0.004]:
        x = np.arange(n)
        a = np.exp(-rate * x)
        a = _symmetrize(a)
        a = _project_to_simplex(a, target_sum)
        candidates.append(a)

    # power‑law shapes
    for p in np.linspace(-2.0, 2.0, 17):
        d = np.minimum(np.arange(n), n - 1 - np.arange(n)) + 1
        a = d ** p
        a = _symmetrize(a)
        a = _project_to_simplex(a, target_sum)
        candidates.append(a)

    # random blockwise candidates (fast cheap exploration)
    block_trials = 600
    for _ in range(block_trials):
        nb = np.random.randint(2, 9)                     # 2‑8 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb)
        cand = np.zeros(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _symmetrize(cand)
        cand = _project_to_simplex(cand, target_sum)
        candidates.append(cand)

    # Evaluate all candidates (fast using FFT)
    best_a = None
    best_score = np.inf
    for a in candidates:
        sc = _conv_max(a, n_fft_conv)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    print(f"[init] n={n}, evaluated {len(candidates)} candidates, best score {best_score:.6f}")

    # --------------------------------------------------------------
    # ---------- 1. PROJECTED ADAM OPTIMISATION --------------------
    a = best_a.copy()
    m = np.zeros_like(a)          # first moment (Adam)
    v = np.zeros_like(a)          # second moment (Adam)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0

    # Adam base learning rate – comparable to the step size used before
    eta0 = target_sum * 0.5 / n

    # Soft‑max temperature schedule
    mu = best_score * 5.0          # start fairly “hot”
    mu_decay = 0.9995
    mu_min = 1e-6

    # Some algorithmic switches
    top_k_hard = 10                # used when mu is small

    # Main optimisation loop -------------------------------------------------
    while time.time() < deadline:
        # full self‑convolution (FFT)
        conv = _full_convolution_fft(a, n_fft_conv)
        cur_max = conv.max()

        # record global best
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            print(f"[{it:08d}] new best = {best_score:.6f}")

        # choose gradient source
        if mu > 0.02 * cur_max:                     # warm phase -> smooth max
            grad = _softmax_gradient(a, conv, mu, n_fft_grad)
        else:                                        # cold phase -> top‑k sub‑grad
            grad = _gradient_topk(a, conv, top_k=top_k_hard)

        # Adam update (bias‑corrected)
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_to_simplex(a_new, target_sum)
        a_new = _symmetrize(a_new)                   # keep symmetry

        # occasional tiny random shake to escape flat regions
        if it % 4000 == 0:
            a_new += np.random.randn(n) * target_sum * 5e-5
            a_new = _project_to_simplex(a_new, target_sum)
            a_new = _symmetrize(a_new)

        a = a_new

        # temperature cooling
        mu = max(mu * mu_decay, mu_min)

        # light‑weight time guard (check every few thousand iters)
        if it % 5000 == 0 and time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # ---------- 2. LOCAL RANDOM SWAPS (hill climbing) -------------
    a = best_a.copy()                # start from the best found so far
    cur_score = best_score
    swap_trials = 0

    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # move a small random amount δ from i to j (preserve sum)
        max_move = min(a[i], target_sum * 0.01)    # limit to 1 % of the total mass
        if max_move <= 0:
            continue
        delta = np.random.rand() * max_move

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _symmetrize(a_new)                 # keep symmetry

        new_score = _conv_max(a_new, n_fft_conv)
        swap_trials += 1
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            best_a = a.copy()
            best_score = new_score
            print(f"[swap {swap_trials}] improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # ---------- 3. FINISH -----------------------------------------
    print(f"[final] best score = {best_score:.6f} after {it} Adam iterations")
    return best_a.tolist()
