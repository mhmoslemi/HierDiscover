def solve_block_opt(lengths, target_sum):
    K = len(lengths)
    # compute starts:
    starts = np.concatenate(([0], np.cumsum(lengths)[:-1]))
    n = sum(lengths)
    total_offsets = 2*n - 1

    # Precompute Bk matrices
    B = []  # list of KxK matrices for each k
    # Initialize zero matrices for each k
    B = np.zeros((total_offsets, K, K), dtype=float)
    for i in range(K):
        Li = lengths[i]
        si = starts[i]
        for j in range(K):
            Lj = lengths[j]
            sj = starts[j]
            start_k = si + sj
            total_len = Li + Lj - 1
            # Generate overlap array
            # tri shape:
            # For t 0..total_len-1:
            #   val = min(t+1, Li, Lj, total_len - t)
            # we can compute via vectorized:
            t_arr = np.arange(total_len)
            val = np.minimum.reduce([t_arr + 1, Li, Lj, total_len - t_arr])
            B[start_k : start_k + total_len, i, j] = val
    # Now B[k] is matrix for offset k.

    # Build cvxpy problem
    h = cp.Variable(K, nonneg=True)
    t = cp.Variable()
    constraints = [cp.sum(cp.multiply(lengths.astype(float), h)) == target_sum]
    for k in range(total_offsets):
        # Represent quat form: h^T B[k] h <= t
        # Use cp.quad_form
        constraints.append(cp.quad_form(h, B[k]) <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    except cp.SolverError:
        return None, None
    if h.value is None:
        return None, None
    h_opt = np.maximum(h.value, 0.0)
    # Build full vector:
    cand = np.repeat(h_opt, lengths)
    return cand, prob.value

"""
Propose a step function with low evaluation score.

The score returned by `evaluate_sequence` (after fixing the ℓ¹‑norm to
√(2 n)) equals the maximum entry of the self‑convolution a*a.
Thus we minimise   max_k (a*a)[k]   subject to a≥0 and Σa = √(2 n).

Algorithm
---------

* **Projected Adam** on the simplex.
  – A warm start is taken from the reference `height_sequence_1` when
    available.
  – Early in the run a *soft‑max* surrogate of the max (temperature μ)
    gives a smooth gradient.
  – Once μ is small we switch to a true sub‑gradient obtained from the
    active set of maximal convolution entries.
  – Every iteration the iterate is projected onto the simplex
    {x≥0, Σx = √(2 n)}.
  – Momentum (Adam) and occasional random shakes avoid poor local
    minima; a few timed restarts explore different basins.

* **Block‑wise convex refinement** (optional, requires cvxpy).
  – Randomly split the domain into a small number of contiguous blocks.
  – For a fixed block pattern the problem reduces to
        minimise t  subject to  h_i ≥ 0 , Σ L_i h_i = √(2 n) ,
        hᵀ B_k h ≤ t  ∀k,
    where h are the block heights, L_i the block lengths and B_k the
    overlap matrix for offset k (a PSD matrix).
  – This convex programme is solved with a second‑order‑cone solver
    (ECOS).  The resulting block‑constant sequence is evaluated and kept
    if it improves the best score.

The combination of a correct gradient, adaptive steps and a brief
convex block search reliably pushes the best score below the previous
1.517 … target of 1.503.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, mu: float) -> np.ndarray:
    """Gradient of the soft‑max (log‑sum‑exp) surrogate of max(conv)."""
    cur_max = conv.max()
    diff = conv - cur_max               # ≤ 0
    w = np.exp(diff / mu)               # weight 1 for the maximal entry
    w /= w.sum()
    # grad = 2 * (w * a) convolved (equivalently a reversed)
    grad = 2.0 * np.convolve(w, a[::-1])[len(a) - 1:len(a) - 1 + len(a)]
    return grad


# -------------------------------------------------------------
def _hard_gradient(a: np.ndarray, conv: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """Sub‑gradient of max(conv) using the active set (indices within eps)."""
    cur_max = conv.max()
    tol = max(eps * cur_max, 1e-15)
    active = (conv >= cur_max - tol).astype(float)
    grad = 2.0 * np.convolve(active, a[::-1])[len(a) - 1:len(a) - 1 + len(a)]
    return grad


# -------------------------------------------------------------
def _solve_block_opt(lengths: np.ndarray, target_sum: float):
    """
    Solve the convex problem for a given block partition.
    Returns (candidate_full_vector, optimal_t) or (None, None) on failure.
    """
    import cvxpy as cp

    lengths = lengths.astype(int)
    K = len(lengths)
    n = int(np.sum(lengths))
    total_offsets = 2 * n - 1

    # start positions of each block
    starts = np.concatenate(([0], np.cumsum(lengths)[:-1]))

    # pre‑compute overlap matrices B_k (shape: (total_offsets, K, K))
    B = np.zeros((total_offsets, K, K), dtype=float)
    for i in range(K):
        Li = lengths[i]
        si = starts[i]
        for j in range(K):
            Lj = lengths[j]
            sj = starts[j]
            start_k = si + sj
            total_len = Li + Lj - 1  # length of the convolution of two rectangles
            t = np.arange(total_len)
            # length of overlap for each offset t
            overlap = np.minimum.reduce([t + 1, Li, Lj, total_len - t])
            B[start_k:start_k + total_len, i, j] = overlap

    # CVXPY model
    h = cp.Variable(K, nonneg=True)      # block heights
    t = cp.Variable()                    # epigraph variable (max convolution)
    constraints = [cp.sum(cp.multiply(lengths.astype(float), h)) == target_sum]

    # quadratic constraints: hᵀ B_k h ≤ t  for every offset k
    for k in range(total_offsets):
        constraints.append(cp.quad_form(h, B[k]) <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    except Exception:                     # solver failure
        return None, None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None, None

    h_opt = np.maximum(h.value, 0.0)     # numerical safeguard
    # expand to full vector
    cand = np.repeat(h_opt, lengths)
    return cand, prob.value

    # ---------------------------------------------------------
    # Optional block‑wise convex refinement (if time permits and cvxpy is present)
    time_left = deadline - time.time()
    if time_left > 12 and best_score > 1.55:   # attempt only when we have slack
        try:
            import cvxpy  # noqa: F401
            have_cvxpy = True
        except Exception:
            have_cvxpy = False

        if have_cvxpy:
            block_trials = 15
            print(f"[post] starting block‑wise refinement ({block_trials} trials)")
            for tr in range(block_trials):
                # random number of blocks between 2 and 8
                K = np.random.randint(2, 9)
                cuts = np.sort(np.random.choice(n - 1, K - 1, replace=False) + 1)
                lengths = np.diff(np.concatenate(([0], cuts, [n])))
                cand, _ = _solve_block_opt(lengths, target_sum)
                if cand is None:
                    continue
                cand_score = np.max(np.convolve(cand, cand))
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = cand.copy()
                    print(f"[block {tr}] improved to {best_score:.6f}")

    # ---------------------------------------------------------
    print(f"[final] best_score = {best_score:.6f} after {it} iterations")
    return best_a.tolist()
