import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

def solve_partner(f):
    n = len(f)
    target_sum = np.sqrt(2.0 * n)
    # variable x = [g (n), M (1)]
    # Constraints:
    # sum(g) = target_sum -> A_eq: [1..1, 0] equality.
    # For each k in [0,2n-2]:
    # sum_i f_i * g_{k - i} - M <= 0
    # Build sparse matrix for A_ub.
    rows = []
    cols = []
    data = []
    # We'll fill rows for each k.
    for k in range(2*n-1):
        # range of i such that 0 <= i < n and 0 <= k-i < n
        i_start = max(0, k - (n-1))
        i_end = min(k, n-1)  # inclusive
        # i runs from i_start to i_end
        # for each i, g_{k-i} factor = f_i
        # Let j = k - i (g index)
        # We'll store row k for column j
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])   # coefficient f_i
        # coefficient for M is -1
        rows.append(k)
        cols.append(n)   # index of M variable
        data.append(-1.0)
    # Build csr matrix
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n+1))
    b_ub = np.zeros(2*n-1)

    # Equality constraint sum(g) == target_sum
    A_eq = sp.csr_matrix(np.concatenate([np.ones((1, n)), np.zeros((1,1))], axis=1))
    b_eq = np.array([target_sum])

    # bounds: g_i >= 0; M free? Actually M >= 0, lower bound 0.
    bounds = [(0, None)] * n + [(0, None)]

    # solve
    res = linprog(c=np.concatenate([np.zeros(n), np.array([1.0])]), A_ub=A_ub, b_ub=b_ub,
                  A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                  method='highs')
    if not res.success:
        print("LP failed:", res.message)
        return f  # fallback
    g = res.x[:n]
    # ensure sum close to target
    g = np.maximum(g, 0)
    # Project onto simplex to correct numerical issues
    g = _project_to_simplex(g, target_sum)
    return g

    # main loop
    while time.time() < deadline:
        # choose n

        target_sum = math.sqrt(2.0 * n)

        # choose initialization method
        init_type = np.random.choice(['random', 'linear', 'quadratic', 'exponential', 'block', 'ref'])
        if init_type == 'ref' and have_ref and n == len(height_sequence_1):

        elif init_type == 'random':
            a = np.random.rand(n)
        elif init_type == 'linear':

            idx = np.arange(n)
            dist = np.abs(idx - (n-1)/2)
            a = dist.astype(float)
        elif init_type == 'quadratic':
            idx = np.arange(n)
            dist = np.abs(idx - (n-1)/2) / ((n-1)/2)
            a = dist**2
        elif init_type == 'exponential':
            idx = np.arange(n)
            dist = np.abs(idx - (n-1)/2) / ((n-1)/2)
            a = np.exp(dist * 3)  # exponential growth towards edges
        elif init_type == 'block':
            blocks = np.random.randint(2, 8)  # random small number of blocks
            cuts = np.sort(np.random.choice(n-1, blocks-1, replace=False) + 1)
            block_ids = np.split(np.arange(n), cuts)
            vals = np.random.rand(blocks)
            a = np.zeros(n)
            for blk, v in zip(block_ids, vals):
                a[blk] = v
        else:
            a = np.random.rand(n)

        a = _project_to_simplex(a, target_sum)

        max_iters = 3000

"""
propose_candidate
-----------------

Search for a non‑negative step function (list of heights) that minimizes the
score returned by ``evaluate_sequence``.  The score equals the maximum entry
of the self‑convolution when the ℓ¹‑norm of the sequence is normalised to
sqrt(2 n).  The algorithm mixes three complementary ideas:

*   **Simplex‑projected sub‑gradient descent** that directly minimises the
    maximum convolution entry.  The gradient for the active shift k is
    2·a_{k‑i} (zero outside the support).  After each step the vector is
    Euclidean‑projected onto the simplex {a≥0, Σa = sqrt(2 n)}.

*   **Linear‑programming refinement**.  For a fixed sequence f, the problem

        minimise   M
        s.t.       f * g ≤ M   (component‑wise)
                   Σ g = sqrt(2 n) , g ≥ 0

    is a linear programme in the variables (g,M).  Solving it yields a
    partner g that gives the smallest possible maximum convolution with f.
    Replacing f by this partner (or a convex combination) often reduces
    the self‑convolution dramatically.

*   **Diverse random restarts** using several initial families
    (uniform, U‑shaped, exponential, block‑wise).  The best sequence found
    before the time budget expires is returned.

The whole search stops after the prescribed wall‑clock budget (default
1000 s) and always returns a feasible sequence; otherwise the harness will
assign ``inf`` points.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """ Euclidean projection of v onto the simplex {x≥0, Σx = z}. """
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray) -> float:
    """Return max_k (a*a)[k] (linear convolution)."""
    return float(np.max(np.convolve(a, a)))


# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] w.r.t. a.
    Uses the (first) index attaining the maximum.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    k = int(np.argmax(conv))
    # gradient_i = 2 * a_{k-i} (zero if out of range)
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    idx_i = np.arange(i_start, i_end + 1)
    idx_j = k - idx_i
    grad[idx_i] = 2.0 * a[idx_j]
    return grad


# ----------------------------------------------------------------------
def _solve_partner(f: np.ndarray) -> np.ndarray:
    """
    Solve the linear programme

        minimise   M
        s.t.       (f * g)_k ≤ M   for k = 0 … 2n‑2
                   Σ g = sqrt(2 n)
                   g ≥ 0 , M ≥ 0

    and return the optimal partner g.
    If a suitable LP solver is unavailable, the input f is returned unchanged.
    """
    try:
        from scipy.optimize import linprog
        import scipy.sparse as sp
    except Exception:                                    # pragma: no cover
        return f

    n = f.shape[0]
    target_sum = math.sqrt(2.0 * n)

    # ----- Build sparse constraint matrix ---------------------------------
    # Variables are [g_0 … g_{n‑1} , M]  (n+1 variables)
    rows = []
    cols = []
    data = []

    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # contribution of each g_j with coefficient f_i, where i = k-j
        for i in range(i_start, i_end + 1):
            j = k - i                     # index of g
            rows.append(k)
            cols.append(j)
            data.append(f[i])
        # coefficient of M is -1 in every inequality
        rows.append(k)
        cols.append(n)                    # column of M
        data.append(-1.0)

    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)

    # Equality: Σ g = target_sum
    A_eq = sp.csr_matrix(np.concatenate([np.ones((1, n)), np.zeros((1, 1))], axis=1))
    b_eq = np.array([target_sum])

    # Objective: minimise M
    c = np.concatenate([np.zeros(n), np.array([1.0])])

    bounds = [(0, None)] * n + [(0, None)]

    # ----- Solve -----------------------------------------------------------
    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'quiet': True})

    if not res.success:               # pragma: no cover
        # In the unlikely case the LP fails, fall back to the current point.
        return f

    g = res.x[:n]
    # Numerical noise may violate the simplex constraint by a tiny amount.
    g = np.maximum(g, 0.0)
    g = _project_to_simplex(g, target_sum)
    return g


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a list of non‑negative heights (a step function) with a low
    evaluation score.  The function respects the time budget (default 1000 s)
    and may invoke ``evaluate_sequence`` only indirectly via the objective
    (max convolution) it computes.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # Helper to evaluate the current max‑convolution (the true score)
    def _score(a):
        return _max_convolution(a)

    # ------------------------------------------------------------------
    # Different initial families that are known to be diverse.
    def _initial_sequence(n, kind):
        target_sum = math.sqrt(2.0 * n)

        if kind == "random":
            a = np.random.rand(n)
        elif kind == "linear_u":                # U‑shaped: distance from centre
            idx = np.arange(n)
            dist = np.abs(idx - (n - 1) / 2)
            a = dist.astype(float) + 1e-12       # avoid all‑zero vector
        elif kind == "quadratic_u":
            idx = np.arange(n)
            dist = np.abs(idx - (n - 1) / 2) / ((n - 1) / 2)
            a = (dist ** 2) + 1e-12
        elif kind == "exponential_u":
            idx = np.arange(n)
            dist = np.abs(idx - (n - 1) / 2) / ((n - 1) / 2)
            a = np.exp(3.0 * dist) + 1e-12
        elif kind == "block":
            blocks = np.random.randint(2, 7)                # 2‑6 blocks
            cuts = np.sort(np.random.choice(n - 1, blocks - 1,
                                            replace=False) + 1)
            blk_ids = np.split(np.arange(n), cuts)
            values = np.random.rand(blocks)
            a = np.zeros(n, dtype=float)
            for ids, v in zip(blk_ids, values):
                a[ids] = v
        else:                                         # fallback to uniform
            a = np.full(n, 1.0)

        a = _project_to_simplex(a, target_sum)
        return a

    # ------------------------------------------------------------------
    # Main search loop – runs as many independent trials as time permits.
    init_kinds = ["random", "linear_u", "quadratic_u", "exponential_u", "block"]
    length_options = [1800, 2500, 3130]   # a few sensible sizes

    while time.time() < deadline:
        # --------------------------------------------------------------
        # Choose a problem size and an initialisation method
        n = int(np.random.choice(length_options))
        kind = np.random.choice(init_kinds)
        a = _initial_sequence(n, kind)

        target_sum = math.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Sub‑gradient descent on the max‑convolution.
        max_iters = 1500
        eta0 = target_sum * 0.5 / n          # initial step size ≈ average entry
        for it in range(max_iters):
            if time.time() >= deadline:
                break

            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] new best {best_score:.6f} (n={n}, kind={kind})")

            grad = _subgradient_maxconv(a)
            # Simple diminishing step size – sqrt decay works well for sub‑gradients
            eta = eta0 / math.sqrt(it + 1)
            a = a - eta * grad
            a = _project_to_simplex(a, target_sum)

            # Small random perturbation occasionally to escape plateaus
            if it % 400 == 0 and it > 0:
                a += np.random.randn(n) * target_sum * 5e-5
                a = _project_to_simplex(a, target_sum)

        # --------------------------------------------------------------
        # LP‑based refinement: replace the current point by the optimal partner.
        # Perform a few fixed‑point iterations (g ← partner(g)).
        for lp_it in range(3):
            if time.time() >= deadline:
                break
            g = _solve_partner(a)
            new_score = _score(g)
            if new_score < best_score:
                best_score = new_score
                best_seq = g.copy()
                print(f"[LP {lp_it}] improved to {best_score:.6f} (n={n})")
            # Stop early if the LP does not improve the objective.
            if new_score >= _score(a) - 1e-8:
                break
            a = g

    # ------------------------------------------------------------------
    # Fallback – should never be needed, but guarantees a valid return value.
    if best_seq is None:
        n = 2000
        a = np.full(n, math.sqrt(2.0 * n) / n)
        best_seq = a

    print(f"[final] best score = {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
