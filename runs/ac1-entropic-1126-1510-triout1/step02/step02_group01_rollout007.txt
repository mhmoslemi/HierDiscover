def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1

    # Determine n
    try:
        n = len(height_sequence_1)
        start_seq = np.asarray(height_sequence_1, dtype=float)
    except NameError:
        n = 3000
        start_seq = np.random.rand(n)

    target_sum = math.sqrt(2.0 * n)

    # initial a: start_seq projected or random Dirichlet
    a = _project_to_simplex(start_seq, target_sum)

    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    print(f"[init] n={n}, score={best_score}")

    # parameters
    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = target_sum * 0.1 / n  # initial step size

    # SA temperature
    T0 = 0.001  # maybe half difference typical
    T = T0
    alpha = 0.9999  # cooling per iteration

    it = 0
    while time.time() < deadline:
        # compute convolution and max index
        conv = np.convolve(a, a)
        max_val = conv.max()
        max_idx = conv.argmax()
        # compute gradient for subgradient (hard max)
        # For each i, compute grad = 2 * a[max_idx - i] if within range else 0
        grad = np.zeros_like(a)
        # compute indexes where max_idx - i within [0, n-1]
        idxs = np.arange(n)
        partner = max_idx - idxs
        mask = (partner >= 0) & (partner < n)
        grad[mask] = 2.0 * a[partner[mask]]
        # For diagonal (i == partner) is included correctly

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_to_simplex(a_new, target_sum)

        # Evaluate
        cur_score = evaluate_sequence(a.tolist())
        new_score = evaluate_sequence(a_new.tolist())

        # SA decision: if new_score < cur_score accept
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
        else:
            # accept with probability exp(-(new_score - cur_score)/T)
            if np.random.rand() < math.exp(-(new_score - cur_score) / (T + 1e-12)):
                a = a_new
                cur_score = new_score
            # else keep old a

        # Keep best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{it}] new best {best_score:.6f}")

        # update temperature
        T = max(T * alpha, 1e-8)

        # occasional heavy random restart: if stuck
        if it % 20000 == 0 and it > 0:
            # Random perturbation
            a_rand = np.random.rand(n)
            a_rand = _project_to_simplex(a_rand, target_sum)
            # Blend with current best
            a = 0.5 * a + 0.5 * a_rand
            a = _project_to_simplex(a, target_sum)
            # reset Adam
            m.fill(0.0)
            v.fill(0.0)

        # occasional local transfer SA:
        if it % 500 == 0:
            # propose mass transfer
            i, j = np.random.choice(n, 2, replace=False)
            max_transfer = min(a[i], target_sum * 0.01) # up to 1% of total sum
            if max_transfer > 0:
                delta = np.random.rand() * max_transfer
                a_tmp = a.copy()
                a_tmp[i] -= delta
                a_tmp[j] += delta
                # keep nonneg and sum automatically holds
                # compute score
                tmp_score = evaluate_sequence(a_tmp.tolist())
                if tmp_score < cur_score or np.random.rand() < math.exp(-(tmp_score - cur_score) / (T + 1e-12)):
                    a = a_tmp
                    cur_score = tmp_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()
                        print(f"[{it}] new best via transfer {best_score:.6f}")

        it += 1

    # final block local search similar to previous
    block_trials = 2000
    for _ in range(block_trials):
        nb = np.random.randint(2, 8)
        cuts = np.sort(np.random.choice(n-1, nb-1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb)
        cand = np.zeros(n)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[block] improved to {best_score:.6f}")
    print(f"[final] best {best_score:.6f}")
    return best_a.tolist()

def project_to_weighted_simplex(v, w, S):
    # Solve for y = w * x, projecting v*w onto standard simplex of mass S
    # y = w * v (elementwise). Then project y onto simplex with sum S.
    y = v * w
    y_proj = _project_to_simplex(y, S)
    # x = y_proj / w
    x = np.zeros_like(v)
    mask = w != 0
    x[mask] = y_proj[mask] / w[mask]
    return x

"""
propose_candidate
-----------------

Hierarchical block optimisation for the step‑function problem.

The algorithm starts with a very coarse representation of the
step‑function (a small number of constant blocks).  It performs an
Adam‑style projected gradient descent on the block heights, where the
gradient is the sub‑gradient of the current convolution maximum.
When progress stalls, every block that is longer than one element is
split into two sub‑blocks (preserving the total L¹‑mass) and the
optimisation continues at the higher resolution.  This multi‑scale
scheme explores the global shape early on and refines locally later,
which helps escaping the plateaux that often trap pure
gradient‑descent methods.

All updates respect the simplex constraint Σa = √(2 n) by a
weighted‑simplex Euclidean projection.  Random “shakes’’ and a final
cheap block‑search are added to increase robustness.

The implementation runs for the full allotted wall‑clock budget
(~1000 s) and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _project_to_weighted_simplex(v: np.ndarray, w: np.ndarray, S: float) -> np.ndarray:
    """
    Project v onto {x ≥ 0, Σ_i w_i x_i = S},
    where w_i > 0 are integer block lengths.
    """
    # work in the space y_i = w_i * x_i, which must satisfy Σ y_i = S
    y = v * w
    y_proj = _project_to_simplex(y, S)
    # avoid division by zero (w_i are always >0)
    x = np.empty_like(v)
    x = y_proj / w
    return x
