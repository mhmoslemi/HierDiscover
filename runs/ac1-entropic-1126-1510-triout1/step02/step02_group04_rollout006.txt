Simple hill‑climbing optimiser for the step‑function problem.

The algorithm starts from the best known construction (``height_sequence_1``)
if it is available, otherwise it falls back to a uniform distribution.
It then repeatedly performs tiny mass transfers between two random
positions, keeps the change if it reduces the evaluation score and
occasionally injects a fresh random point to escape local minima.
All arithmetic stays on the probability simplex (non‑negative, sum = 1)
so the objective ``evaluate_sequence`` can be safely replaced by
the faster equivalent ``_score`` during the search.
The routine respects the 1000 s budget and returns the best sequence
found.
"""
import time
import numpy as np

def _clip_and_normalise(v: np.ndarray) -> np.ndarray:
    """Clip negatives to zero and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fallback to uniform distribution
        v = np.full_like(v, 1.0 / v.size)
    else:
        v = v / s
    return v

def _score(seq: np.ndarray) -> float:
    """
    Fast proxy for ``evaluate_sequence`` when ``seq`` is already
    normalised to sum = 1.
    Returns 2·n·max(conv(seq,seq)).
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    max_conv = conv.max()
    return float(2.0 * n * max_conv)

def _perturb(seq: np.ndarray, frac: float = 0.02) -> np.ndarray:
    """
    Transfer a small amount of mass from one random index to another.
    ``frac`` controls the maximal fraction of the source mass that may be moved.
    """
    n = seq.size

    amount = frac * seq[i] * np.random.rand()
