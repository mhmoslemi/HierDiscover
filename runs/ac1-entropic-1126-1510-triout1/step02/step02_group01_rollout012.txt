"""
Propose a step–function (non‑negative vector) that minimizes the
evaluation score 2·n·max(conv) / (sum(a)^2).  The algorithm is a
projected Adam optimiser on the simplex (Σa = √(2 n)) with an
annealed soft‑max surrogate for the max‑convolution.  It combines
several restarts, occasional random shakes and a cheap block‑search
at the end.  Compared to the reference code it uses a full‑active‑set
gradient (instead of a fixed top‑k), Adam’s adaptive steps, and a
more aggressive restart schedule, which usually pushes the max‑convolution
below the previous 1.5172 barrier.
"""

import time
import math
import numpy as np


def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_weighted(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k (a*a)[k] w.r.t. a.
    `w` must have length 2·n‑1 (the full convolution length).
    """
    n = a.shape[0]
    # convolution of the weight vector with reversed a
    conv_wr = np.convolve(w, a[::-1])
    # extract the part that corresponds to each component of a
    sum_term = conv_wr[n - 1 : n - 1 + n]
    grad = 2.0 * sum_term

    # diagonal correction (terms where i = j)
    double_idx = 2 * np.arange(n)
    mask = double_idx < w.shape[0]
    grad[mask] -= w[double_idx[mask]] * a[mask]
    return grad


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a low‑score step function within the time budget."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # problem size (use the reference size when it exists)
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)   # Σa = √(2 n) → score = max(conv)

    # ------------------------------------------------------------------
    # initialise a feasible vector
    if have_ref and np.random.rand() < 0.6:
        a = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
    else:
        a = np.random.rand(n)

    a = _project_to_simplex(a, target_sum)

    # best ever found
    conv = np.convolve(a, a)
    best_score = conv.max()
    best_a = a.copy()
    print(f"[init] n={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Adam state
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = beta1
    beta2_pow = beta2

    # learning‑rate schedule (≈ average entry)
    lr0 = target_sum * 0.5 / n

    # soft‑max temperature
    mu = best_score
    mu_decay = 0.9997
    mu_min = 1e-7

    # restart handling
    max_restarts = 8
    restart_interval = (budget_s - 5) / max_restarts
    next_restart = time.time() + restart_interval
    restarts_done = 0

    it = 0
    while time.time() < deadline:
        # --------------------------------------------------------------
        # convolution & current max
        conv = np.convolve(a, a)
        cur_max = conv.max()

        # keep global best
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            print(f"[{it:08d}] new best = {best_score:.6f}")

        # --------------------------------------------------------------
        # build weight vector w
        if mu > 0.01 * cur_max:                     # high‑temperature (soft‑max)
            diff = conv - cur_max                   # ≤ 0, max entry = 0
            w = np.exp(diff / mu)
            w_sum = w.sum()
            if w_sum > 0:
                w /= w_sum
        else:                                       # low‑temperature (hard max)
            tol = max(cur_max * 1e-9, 1e-12)
            active = np.where(conv >= cur_max - tol)[0]
            w = np.zeros_like(conv)
            w[active] = 1.0          # unnormalised – scaling is absorbed by the step size

        # --------------------------------------------------------------

        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = np.convolve(cand, cand).max()
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[block] improved block score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # final tiny local refinements: move a tiny amount of mass between
    # random indices and keep any improvement.
    local_iters = 2000
    a_loc = best_a.copy()
    conv_loc = np.convolve(a_loc, a_loc)
    cur_best = best_score
    for _ in range(local_iters):

        if a_loc[i] <= 1e-12:
            continue
        delta = a_loc[i] * np.random.rand() * 0.5      # move up to 50 % of a_i
        a_new = a_loc.copy()

        a_new[j] += delta

            a_loc = a_new
            best_score = sc
            best_a = a_new.copy()
            print(f"[local] improved to {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f} after {it} iterations")
    return best_a.tolist()
