"""
Propose a step‑function sequence with a low evaluation score.

The search consists of three stages:

1. **Broad parametric shape sampling** – Generate many simple
   families (inverse‑power, exponential, cosine, mixtures, random block
   structures) and keep the best candidate.  Each candidate is projected
   onto the simplex {a≥0, Σa = √(2 n)}.

2. **Refinement by simulated‑annealing local moves** – Starting from the
   best shape, repeatedly transfer a small amount of mass from one index
   to another while keeping the total sum constant.  Moves are accepted
   with a Metropolis criterion using a temperature that cools
   geometrically.  The current best, as well as the global best, are
   updated throughout.

3. **Final projection** – The best sequence found is projected one last
   time onto the simplex to guarantee the required sum.

All operations respect the security checks of `evaluate_sequence`; the
algorithm works entirely with non‑negative numbers and a fixed total
mass, so the returned sequence can be fed directly to `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_convolution(seq: np.ndarray, nfft: int) -> float:
    """Return max entry of the linear convolution seq * seq."""
    # Linear convolution via FFT (full length = 2*len(seq)-1)
    f = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(f * f, n=nfft)
    # The first (2*len(seq)-1) entries contain the linear convolution.
    max_val = conv[: 2 * len(seq) - 1].max()
    return float(max_val)


# ----------------------------------------------------------------------
def _make_inverse_power(beta: float, p: float, n: int) -> np.ndarray:
    """Symmetric inverse‑power shape a_i = (i+β+1)^(-p) + (n‑1‑i+β+1)^(-p)."""
    idx = np.arange(n, dtype=float)
    left = 1.0 / (idx + beta + 1.0) ** p
    right = 1.0 / ((n - 1 - idx) + beta + 1.0) ** p
    return left + right


# ----------------------------------------------------------------------
def _make_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential decay a_i = exp(-α·i) + exp(-α·(n‑1‑i))."""
    idx = np.arange(n, dtype=float)
    left = np.exp(-alpha * idx)
    right = np.exp(-alpha * (n - 1 - idx))
    return left + right


# ----------------------------------------------------------------------
def _make_cosine(p: float, n: int) -> np.ndarray:
    """Cosine‑based shape a_i = |cos(π·i/(n‑1))|^p."""
    idx = np.arange(n, dtype=float)
    base = np.abs(np.cos(np.pi * idx / (n - 1))) ** p
    return base


# ----------------------------------------------------------------------
def _make_mixture(beta: float, lam: float, n: int) -> np.ndarray:
    """Mixture of uniform and inverse‑sqrt."""
    inv = 1.0 / np.sqrt(np.arange(n, dtype=float) + beta + 1.0)
    return lam * np.ones(n) + (1.0 - lam) * inv


# ----------------------------------------------------------------------
def _make_random_blocks(nb: int, n: int) -> np.ndarray:
    """Create a random block‑wise constant sequence with nb blocks."""
    cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
    blocks = np.split(np.arange(n), cuts)
    vals = np.random.rand(nb) + 0.1
    cand = np.empty(n, dtype=float)
    for blk, v in zip(blocks, vals):
        cand[blk] = v
    return cand

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
