def compute_score(seq):
    s = np.sum(seq)
    if s < 0.01:
        return float('inf')
    n = len(seq)
    # next power of two for efficient FFT:
    m = 1 << (2*n-1).bit_length()
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)
    # conv length is m; actual convolution length 2*n-1 is first 2*n-1 entries
    conv = conv[:2*n-1]
    max_b = conv.max()
    return float(2 * n * max_b / (s*s))

"""
Improved search for a low‑autocorrelation step function.

The evaluation function is scale‑invariant:
    score = 2·n·max(conv(f,f)) / (sum(f)²)

Hence we can normalise any candidate to sum = 1 and minimise
    max(conv(f,f)).
For a fixed multiset of values the sum of products with the reversed
vector (the central convolution entry) is minimised by pairing the
largest values with the smallest ones (rearrangement inequality).
The algorithm therefore

1. draws many random value‑multisets (Dirichlet / Gamma samples) with
   various concentration parameters,
2. arranges each multiset in the “anti‑sorted’’ order (largest,
   smallest, second largest, second smallest, …),
3. evaluates the score (FFT‑based convolution, O(n log n)),
4. keeps the best candidate and finally
5. runs a short simulated‑annealing polish that moves a tiny amount of
   mass between two positions while preserving the total sum.

The whole search respects the 1000 s budget, uses only NumPy and the
provided ``evaluate_sequence`` for the final answer, and prints progress.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast evaluation of the objective (identical to evaluate_sequence
# but without the security checks).  The function works on a NumPy array
# and assumes all entries are finite non‑negative numbers.
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Return the evaluation score for a non‑negative array ``seq``."""
    s = seq.sum()
    if s < 0.01:                     # same guard used in evaluate_sequence
        return float("inf")
    n = seq.size
    # Length needed for exact linear convolution
    m = 1 << ((2 * n - 1).bit_length())
    # FFT‑based linear convolution (real input -> rfft)
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Helper: construct an anti‑sorted arrangement of a value multiset.
# Largest value goes to the leftmost free slot, second largest to the
# rightmost free slot, and so on.
# ----------------------------------------------------------------------
def _anti_sorted(vals: np.ndarray) -> np.ndarray:
    """Return an anti‑sorted sequence built from the values in ``vals``."""
    n = vals.size
    # sort descending
    sorted_vals = np.sort(vals)[::-1]
    seq = np.empty(n, dtype=float)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if (i & 1) == 0:            # even index → left side
            seq[left] = v
            left += 1
        else:                       # odd index → right side
            seq[right] = v
            right -= 1
    return seq

    try:

        n = 8192                          # a power of two makes FFT cheap

    # --------------------------------------------------------------
    best_seq = None

        temperature = (0.05 * (1.0 - t_frac) + 0.0005 * t_frac)

        # ---- propose a random pair move (preserves total mass) ----
        i, j = np.random.choice(n, size=2, replace=False)
        sigma = 0.02 * (1.0 - t_frac) + 0.001    # step size shrinks
        delta = (np.random.rand() - 0.5) * sigma

        # Clip to keep both entries non‑negative
        if delta > 0.0:
            delta = min(delta, cur_seq[j])
            cand = cur_seq.copy()
            cand[i] += delta
            cand[j] -= delta
        else:
            delta = -min(-delta, cur_seq[i])
            cand = cur_seq.copy()
            cand[i] += delta
            cand[j] -= delta

        cand_score = _fast_score(cand)

        # ---- Metropolis acceptance ----
        if cand_score < cur_score:
            cur_seq, cur_score = cand, cand_score
            if cand_score < best_score:
                best_score, best_seq = cand_score, cand.copy()
                print(f"[anneal {it:05d}] New best score: {best_score:.6f}")
        else:
            # probability exp(-(Δ)/T) – note Δ>0 here
            if temperature > 0.0 and np.random.rand() < math.exp(-(cand_score - cur_score) / temperature):
                cur_seq, cur_score = cand, cand_score

    # --------------------------------------------------------------
    # Final safety: ensure non‑negativity and return a plain list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    return best_seq.tolist()
