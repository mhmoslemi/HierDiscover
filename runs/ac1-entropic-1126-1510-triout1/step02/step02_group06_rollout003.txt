def conv_fft(a):

    n = len(a)
    size = 2*n-1
    # choose fft length as power of 2 >= size

"""
Propose a step‑function sequence by a two‑phase optimisation:

1.  Initialise the probability vector with the classic arcsine coefficients
    (and a few random Dirichlet seeds).  The vector is kept on the
    probability simplex (non‑negative, sum = 1).

2.  Perform many mirror‑descent steps on a *smoothed* version of the
    objective  max_k (a∗a)_k .  The smoothing is a soft‑max
    (log‑sum‑exp) with a temperature β that is slowly increased.
    The mirror update is multiplicative (a_i ← a_i·exp(‑η·∂)),
    which automatically preserves non‑negativity; after the update the
    vector is re‑normalised to sum 1.

3.  Every few thousand iterations an LP is solved (if SciPy is available):
    maximise Σ_i g_i subject to a∗g ≤ current peak M.
    The solution g typically has a larger ℓ₁‑norm while respecting the
    same maximum convolution value.  A convex combination of the current
    vector and the normalised LP solution is taken, which provably never
    increases the evaluation score.

4.  The process is repeated for several sequence lengths
    (≈1500–3000).  The best sequence found (with the smallest value of
    ``evaluate_sequence``) is returned.

The whole routine stops before the allotted time limit (default 1000 s)
and never writes to disk.  It uses only NumPy, SciPy (optional) and the
provided ``evaluate_sequence`` function.
"""

import time
import math
import numpy as np

# Optional – SciPy is used only for the linear‑programming refinement.
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:                     # pragma: no cover
    _SCIPY_AVAILABLE = False


def arcsine_sequence(n: int) -> list[float]:
    """First n coefficients of (1‑z)^‑1/2, normalised to sum 1."""
    seq = []
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    seq = [x / total for x in seq]
    return seq


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full (linear) convolution a∗a using NumPy (fast enough for n≤3000)."""
    return np.convolve(a, a)


def _metric(a: np.ndarray) -> float:
    """Exact evaluation metric for a probability vector (sum≈1)."""
    n = a.shape[0]
    conv = _conv_full(a)
    return 2.0 * n * conv.max()


def _solve_lp_max_sum(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ g_i   subject to   a∗g ≤ M,   g≥0.
    Returns the optimal g (not normalised) or None on failure.
    """
    if not _SCIPY_AVAILABLE:                     # pragma: no cover
        return None
    n = a.shape[0]
    rows = 2 * n - 1

    # Build the Toeplitz‑like constraint matrix A where
    #   (A g)_k = Σ_i a_i g_{k-i}  (= (a∗g)_k).
    A = np.empty((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = a   # shift a into column i

    b = np.full(rows, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # minimise –Σ g_i  → maximise Σ g_i
    bounds = [(0, None)] * n

    # ``highs`` is fast for medium‑size dense LPs.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None


def _smooth_mirror_descent(a0: np.ndarray,
                           deadline: float,
                           max_iter: int = 200_000,
                           eta0: float = 0.5,
                           beta0: float = 1.0,
                           beta_max: float = 2000.0,
                           lp_interval: int = 5000) -> tuple[list[float], float]:
    """
    Mirror‑descent on the smoothed max‐convolution objective.
    Occasionally mixes in an LP solution to accelerate progress.
    Returns the best sequence (as a Python list) and its metric.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _metric(a)

    eta = eta0
    beta = beta0
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_full(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # keep the best ever seen
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            if it % 200 == 0:
                print(f"[len={n} iter={it:6d}] new best metric {best_score:.6f}")

        # ---------- smoothed max ----------
        # soft‑max (log‑sum‑exp) weights
        shift = cur_max                     # numeric stabiliser
        exp_vals = np.exp(beta * (conv - shift))
        w = exp_vals / exp_vals.sum()       # w has length 2n‑1, sums to 1

        # gradient of the smoothed objective (mirror‑descent on the simplex)
        #   ∂/∂a_i f = 2· Σ_k w_k a_{k‑i}
        grad = 2.0 * np.correlate(w, a, mode='valid')   # length n

        # multiplicative (exponential) mirror‑descent update
        a = a * np.exp(-eta * grad)

        # re‑normalise to the simplex
        s = a.sum()
        if s <= 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ----- anneal β (make the approx tighter) -----
        if beta < beta_max:
            beta = min(beta_max, beta * 1.001)

        # ----- occasional LP refinement -----
        if _SCIPY_AVAILABLE and (it % lp_interval == 0):
            M = conv.max()
            g = _solve_lp_max_sum(a, M)
            if g is not None:
                sum_g = g.sum()
                if sum_g > 1.0 + 1e-9:
                    # normalise the LP solution and mix it in.
                    g_norm = g / sum_g
                    # mixing weight – we never exceed the contribution that would
                    # increase the ℓ₁‑norm beyond the current one.
                    t = min(0.2, (sum_g - 1.0) / sum_g)
                    a = (1.0 - t) * a + t * g_norm
                    a /= a.sum()           # keep exact normalisation

        # ----- adapt step size if we are stuck -----
        if it % 1000 == 0:
            # simple heuristic: if the metric hasn't improved recently, shrink η
            if cur_score > best_score * 1.001:
                eta = max(eta * 0.7, 1e-4)
            else:
                eta = min(eta * 1.05, 5.0)

        # safety guard – avoid endless tiny steps
        if eta < 1e-9:
            break

    return best_a.tolist(), best_score


def _optimise_one_length(n: int, time_budget: float) -> tuple[list[float], float]:
    """
    Optimise a single length n using several random starts.
    Returns the best sequence found and its metric.
    """
    deadline = time.time() + time_budget
    best_seq = None
    best_score = float("inf")

    # start points: classical arcsine + a few Dirichlet draws
    starts = [np.array(arcsine_sequence(n), dtype=np.float64)]
    rng = np.random.default_rng(12345)
    for _ in range(3):
        starts.append(rng.dirichlet(alpha=np.full(n, 0.5)))
    # optionally use a globally supplied sequence if present
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if hs.size == n:
            starts.append(hs / hs.sum())

    for idx, a0 in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"\n--- Length {n} – start #{idx + 1} (budget {(deadline - time.time()):.1f}s) ---")
        seq, score = _smooth_mirror_descent(a0, deadline)
        print(f"Finished start #{idx + 1}: metric = {score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"*** New best for length {n}: {best_score:.6f}")

    return best_seq, best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for the lowest possible evaluation score within the allocated time.
    Several sequence lengths are explored; each length receives a share of
    the remaining budget and is refined by a smoothed mirror‑descent
    plus occasional LP mixing.
    The best sequence (as a list of floats) is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # Lengths that comfortably fit in memory and run fast.
    # The exact set can be tuned; we keep a moderate spread.
    lengths = [1500, 1800, 2100, 2400, 2700, 3000]

    best_overall_seq = None
    best_overall_score = float("inf")

    for idx, n in enumerate(lengths):
        now = time.time()
        if now >= deadline:
            break

        # Allocate roughly equal time for the remaining lengths.
        remaining_lengths = len(lengths) - idx
        remaining_time = deadline - now
        per_len_budget = max(5.0, remaining_time / remaining_lengths)

        print(f"\n=== Optimising length {n} (budget ≈ {per_len_budget:.1f}s) ===")
        seq, score = _optimise_one_length(n, per_len_budget)

        if seq is None:
            continue

        print(f"Best metric for length {n}: {score:.6f}")
        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = seq
            print(f"*** New overall best (len={n}) score = {best_overall_score:.6f}")

    # Fallback – should never be needed.
    if best_overall_seq is None:
        best_overall_seq = arcsine_sequence(2000)
        best_overall_score = _metric(np.array(best_overall_seq))

    # Final clipping to respect the bounds imposed by ``evaluate_sequence``.
    best_overall_seq = [float(max(0.0, min(1000.0, x))) for x in best_overall_seq]

    print(f"\n--- FINAL RESULT ---\nBest length = {len(best_overall_seq)} "
          f"with metric = {best_overall_score:.6f}")
    return best_overall_seq
