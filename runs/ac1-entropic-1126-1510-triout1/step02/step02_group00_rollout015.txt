def compute_gradient(a, w):
    n = a.shape[0]
    w_rev = w[::-1]
    size = 1 << ((n + len(w_rev) - 1).bit_length())
    A = np.fft.rfft(a, n=size)

    conv = np.fft.irfft(A * Wr, n=size)
    offset = len(w_rev) - 1
    grad = 2.0 * conv[offset: offset + n]
    return grad

* Work on the probability simplex (sum = 1) – the objective
  2·n·max(a∗a) is invariant under scaling.
* At each iteration compute the self‑convolution `b = a∗a` via a single
  FFT (real‐to‑real) and the “soft‑max” weights
      w_k ∝ exp((b_k–max(b))/σ) .
  When σ → 0 these weights converge to a one‑hot vector on the maximal
  entry, i.e. we recover the sub‑gradient used in the reference code.
* The gradient of the smoothed objective is
      g = 2·(a ⋆ w) ,   (cross‑correlation)

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine density on {0,…,n‑1}, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

        a /= a.sum()

        a = _arcsine_distribution(n)
    a = _project_onto_simplex(a, s=1.0)
    return a


def _next_pow_two(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray, size: int) -> np.ndarray:
    """Return a∗a (real convolution) using FFT of given size."""
    A = np.fft.rfft(a, n=size)
    b = np.fft.irfft(A * A, n=size)
    return b[: 2 * a.shape[0] - 1]


def _grad_via_softmax(a: np.ndarray, b: np.ndarray, sigma: float,
                      size: int) -> np.ndarray:
    """
    Compute g = 2·(a ⋆ w) where w is the soft‑max of b with temperature sigma.
    `b` is the full self‑convolution (length 2·n‑1) of `a`.
    """
    n = a.shape[0]
    max_b = b.max()
    if sigma <= 0.0:
        # pure sub‑gradient (hard max)
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        # stable soft‑max
        exp_arg = (b - max_b) / sigma
        # avoid overflow in exp
        exp_arg = np.clip(exp_arg, -700, 0)   # values ≥0 are already 0 after subtraction
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
        else:
            w /= w_sum

    # cross‑correlation a ⋆ w  =  (a * reverse(w))
    w_rev = w[::-1]
    A = np.fft.rfft(a, n=size)
    Wr = np.fft.rfft(w_rev, n=size)
    conv = np.fft.irfft(A * Wr, n=size)
    offset = len(w_rev) - 1          # = 2·n‑2
    grad = 2.0 * conv[offset: offset + n]
    return grad
