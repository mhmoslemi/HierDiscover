    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1

    """Project onto the probability simplex (nonnegative, sum=1)."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # Sort in descending order
    u = np.sort(v)[::-1]

        theta = 0

"""
Propose a step‑function (non‑negative list) that minimizes the
``evaluate_sequence`` score.

The algorithm works in three stages:

1. **Warm‑start** – use the globally supplied ``height_sequence_1`` if
   it exists, otherwise fall back to a uniform distribution.  The
   vector is normalised to unit L1‑norm (the objective is scale invariant).

2. **LP refinement (Frank–Wolfe style)** – repeated linear‑programming
   steps increase the total mass of a candidate while keeping the
   current autocorrelation bound.  The best mixing weight *t* is found
   by a cheap line search on the convex combination of the three
   convolutions (f*f, f*g, g*g).

3. **Projected gradient descent on a smoothed ∞‑norm** – a soft‑max
   approximation of the maximal autocorrelation is minimised by
   gradient descent.  Each step is followed by a projection onto the
   probability simplex (non‑negative, sum = 1).  Several step‑sizes are
   tried and the best update is kept.  The smoothing parameter α is
   slowly increased so that the method focuses on the true maximum.

A short random‑pair polishing phase finishes the search.  The best
sequence found before the time budget expires is returned as a plain
list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional import of cvxpy – used for the LP refinement.
# ----------------------------------------------------------------------
try:
    import cvxpy as cp   # type: ignore
    _HAS_CVX = True
except Exception:        # pragma: no cover
    _HAS_CVX = False


# ----------------------------------------------------------------------
# Helper: random pairwise mass‑swap (keeps sum exactly 1).
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Move a small amount of mass between two random positions."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)
    delta = (np.random.rand() - 0.5) * sigma   # uniform in [-sigma/2, sigma/2]

    # keep both entries non‑negative
    if delta > 0:
        delta = min(delta, seq[j])            # take from j
    else:
        delta = -min(-delta, seq[i])          # take from i

    out = seq.copy()
    out[i] += delta
    out[j] -= delta
    return out


# ----------------------------------------------------------------------
# Linear‑programming step used in the Frank–Wolfe refinement.
# ----------------------------------------------------------------------
def _lp_max_mass(f: np.ndarray, M: float):
    """Solve   maximise Σ g   s.t.   (f * g) ≤ M ,   g ≥ 0.
    Returns the optimal g (numpy array) or ``None`` on failure."""
    if not _HAS_CVX:
        return None

    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)

    # cvxpy's built‑in convolution (linear in g)
    conv_fg = cp.conv(cp.Constant(f), g)        # length 2n‑1
    constraints = [conv_fg <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                    max_iters=500, verbose=False)
    except Exception:                           # pragma: no cover
        return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None

    g_val = g.value
    if g_val is None:
        return None
    return np.asarray(g_val, dtype=float)


# ----------------------------------------------------------------------
# Fast (FFT based) linear convolution.
# ----------------------------------------------------------------------
def _convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a power‑of‑two FFT."""
    n = len(a) + len(b) - 1
    L = 1 << ((n - 1).bit_length())            # next power of two
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    fc = fa * fb
    conv = np.fft.irfft(fc, n=L)[:n]
    return conv


# ----------------------------------------------------------------------
# Linear cross‑correlation via FFT (needed for the gradient).
# ----------------------------------------------------------------------
def _cross_corr(p: np.ndarray, f: np.ndarray) -> np.ndarray:
    """
    Compute the correlation (p ⋆ f)[i] = Σ_j p[j] * f[i+j]  (zero‑padded)
    for i = 0 … len(f)-1.
    """
    rev_f = f[::-1]
    conv = _convolve_fft(p, rev_f)          # length = 3*len(f)-2
    start = len(f) - 1
    return conv[start:start + len(f)]


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Σ x_i = 1, x_i ≥ 0).
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights (a step function) that tries
    to minimise ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + max(0.0, budget_s - 2.0)   # small safety margin

    # --------------------------------------------------------------
    # 1. Initialise a starting sequence (unit L1‑norm).
    # --------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        if base.size == 0:
            raise ValueError
        base = np.maximum(base, 0.0)
        base /= max(base.sum(), 1e-12)
    except Exception:                                    # pragma: no cover
        # Fallback: a uniform vector of a convenient length (power of two).
        n = 8192
        base = np.full(n, 1.0 / n, dtype=float)

    n = base.shape[0]
    best_seq = base.copy()
    conv_best = _convolve_fft(best_seq, best_seq)
    best_score = 2.0 * n * conv_best.max()

    # --------------------------------------------------------------
    # 2. LP refinement (Frank–Wolfe style) – limited number of iters.
    # --------------------------------------------------------------
    f = base.copy()
    current_score = best_score
    max_lp_iters = 40
    lp_iter = 0

    while time.time() < deadline and lp_iter < max_lp_iters:
        lp_iter += 1
        conv_ff = _convolve_fft(f, f)
        M = conv_ff.max()

        g_raw = _lp_max_mass(f, M)
        if g_raw is None:
            break          # LP not available or failed

        sum_g = g_raw.sum()
        if sum_g <= 1.0 + 1e-12:
            break          # stationary point

        g_norm = g_raw / sum_g               # Σ g_norm = 1

        # pre‑compute the needed convolutions
        conv_fg = _convolve_fft(f, g_norm)
        conv_gg = _convolve_fft(g_norm, g_norm)

        # ----------------------------------------------------------
        # Find the best mixing coefficient t ∈ [0,1] by a cheap grid.
        # ----------------------------------------------------------
        t_grid = np.linspace(0.0, 1.0, 21)    # step 0.05
        best_t, best_val = 0.0, np.inf

        for t in t_grid:
            conv_t = ((1 - t) ** 2) * conv_ff + (2 * t * (1 - t)) * conv_fg + (t ** 2) * conv_gg
            val = conv_t.max()
            if val < best_val:
                best_val = val
                best_t = t

        # finer search around the provisional optimum
        t_fine = np.linspace(max(0.0, best_t - 0.05), min(1.0, best_t + 0.05), 11)
        for t in t_fine:
            conv_t = ((1 - t) ** 2) * conv_ff + (2 * t * (1 - t)) * conv_fg + (t ** 2) * conv_gg
            val = conv_t.max()
            if val < best_val:
                best_val = val
                best_t = t

        # construct the candidate
        f_candidate = (1.0 - best_t) * f + best_t * g_norm
        cand_score = 2.0 * n * best_val

        # keep track of the absolute best
        if cand_score < best_score:
            best_score = cand_score
            best_seq = f_candidate.copy()
            print(f"[LP iter {lp_iter:02d}] New best score: {best_score:.6f}")

        # accept only if we improve the current iterates
        if cand_score < current_score:
            f = f_candidate
            current_score = cand_score
        else:
            # no improvement – stop early
            break

    # --------------------------------------------------------------
    # 3. Projected gradient descent on a smoothed max‑autocorrelation.
    # --------------------------------------------------------------
    f = best_seq.copy()
    gd_iter = 0
    while time.time() < deadline and gd_iter < 30000:
        gd_iter += 1

        # current convolution & objective
        conv_f = _convolve_fft(f, f)
        max_c = conv_f.max()
        score = 2.0 * n * max_c
        if score < best_score:
            best_score = score
            best_seq = f.copy()
            print(f"[GD iter {gd_iter:04d}] New best score: {best_score:.6f}")

        # smoothing parameter α – gradually increase
        alpha = 20.0 + (gd_iter / 5000.0) * 180.0   # ends near 200
        # soft‑max weights for the convolution entries
        exp_shift = np.exp(alpha * (conv_f - max_c))
        p = exp_shift / exp_shift.sum()

        # gradient of the smoothed max‑norm
        grad = 2.0 * _cross_corr(p, f)

        # try a few step sizes and keep the best update
        eta0 = 0.5 / (1.0 + gd_iter * 5e-4)   # slowly decreasing
        candidate_etas = [eta0, eta0 * 0.5, eta0 * 2.0]
        best_local_f = f
        best_local_score = score

        for eta in candidate_etas:
            f_tmp = f - eta * grad
            f_tmp = _project_simplex(f_tmp)
            conv_tmp = _convolve_fft(f_tmp, f_tmp)
            s_tmp = 2.0 * n * conv_tmp.max()
            if s_tmp < best_local_score:
                best_local_score = s_tmp
                best_local_f = f_tmp

        f = best_local_f

    # --------------------------------------------------------------
    # 4. Final cheap polishing with random pairwise moves.
    # --------------------------------------------------------------
    polish_iters = 4000
    it = 0
    while time.time() < deadline and it < polish_iters:
        it += 1
        sigma = 0.3 * (1.0 - it / polish_iters) + 0.02
        candidate = _random_pair_move(best_seq, sigma)

        conv_c = _convolve_fft(candidate, candidate)
        cand_score = 2.0 * n * conv_c.max()
        if cand_score < best_score:
            best_score = cand_score
            best_seq = candidate
            print(f"[Polish iter {it:04d}] New best score: {best_score:.6f}")

    # --------------------------------------------------------------
    # Return the best sequence found.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    return best_seq.tolist()
