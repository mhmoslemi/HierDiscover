```
def optimize_sequence(a0, time_deadline, max_iters=...):
    a = a0.copy()
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1
    iter = 0
    best_local_a = a.copy()
    best_local_score = evaluate_sequence(a.tolist())
    beta = 1.0

    while time.time() < time_deadline and iter < max_iters:
        # compute convolution
        conv = np.convolve(a, a)
        max_val = conv.max()
        # compute w = softmax(beta*(conv - max_val))
        # use stable exp
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / exp_vals.sum()
        # gradient (softmax) = 2 * conv(w, a_rev)
        grad = 2.0 * np.convolve(w, a[::-1])[n-1:2*n-1]
        # Adam update on a (before projection)
        iter += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**iter)
        v_hat = v / (1 - beta2**iter)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # enforce non-negativity:
        a = np.maximum(a, 0.0)
        # renormalize to sum 1
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s
        # evaluate
        curr_score = evaluate_sequence(a.tolist())
        if curr_score < best_local_score:
            best_local_score = curr_score
            best_local_a = a.copy()
        # update beta; increase gradually
        beta = min(200.0, beta * 1.02)
        # optionally adjust lr: maybe decay slowly
        lr *= 0.9995
    return best_local_a, best_local_score
```

```
def local_optimize(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    beta = 2.0  # start with relatively low
    eta = 1.0   # step size for exponentiated gradient

    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    # optionally a small safety check for sum>0.01 (ensured)
    iter_no_improve = 0

    while time.time() < deadline:
        conv = np.convolve(a, a)
        max_val = conv.max()
        # compute weights
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / exp_vals.sum()
        # gradient
        grad = 2.0 * np.convolve(w, a[::-1])[n - 1:2 * n - 1]
        # exponentiated gradient step
        a = a * np.exp(-eta * grad)
        # renorm
        total = a.sum()
        if total == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        # evaluate seldom: every 5 iterations
        iter_no_improve += 1
        if iter_no_improve % 5 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                iter_no_improve = 0
                print(f"[local opt] improved to {sc:.6f} (n={n})")
        # schedule updates
        beta = min(500.0, beta * 1.02)
        eta = max(1e-6, eta * 0.9995)

        # optionally break if max_iter > some big number
    return best_a, best_score
```

"""
Propose a step‑function sequence with low self‑convolution peak using
exponentiated‑gradient descent on the simplex.

The algorithm:
* Choose a length n (≈4000) and generate several seed sequences
  (arcsine, uniform, 1/√i, exponential decay, random Dirichlet).
* For each seed run a fast exponentiated‑gradient loop.
  – The gradient of a smooth approximation (soft‑max) of the maximal
    convolution value is computed analytically.
  – An exponentiated‑gradient (multiplicative‑weights) update keeps the
    vector non‑negative and automatically respects the ∑a = 1 simplex
    constraint.
  – β (soft‑max temperature) is slowly increased so the optimisation
    initially sees a smooth landscape and finally concentrates on the
    true maximum.
  – η (step size) is slowly decayed; occasional tiny random perturbations
    are added to escape shallow local minima.
* The best sequence found across all seeds (and a final refinement of the
  current best) is returned.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> np.ndarray:
    """Return a length‑n non‑negative sequence proportional to the
    arcsine (binomial) coefficients, L1‑normalised to sum = 1.
    """
    seq = np.empty(n, dtype=float)
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq[i] = cur
        total += cur
        if i + 1 < n:
            cur = cur * (2.0 * i + 1.0) / (2.0 * (i + 1))
    seq /= total
    return seq
