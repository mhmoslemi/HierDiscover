  a = a_init.copy()
  m = np.zeros_like(a)
  v = np.zeros_like(a)
  beta1 = 0.9
  beta2 = 0.999
  eps = 1e-8
  beta = beta_start
  best_local_score = evaluate_sequence(a.tolist())
  best_local_seq = a.copy()
  for t in range(1, max_iters+1):
      # compute conv and gradient
      conv = np.fft.ifft(np.fft.fft(a, conv_len) * np.fft.fft(a, conv_len)).real
      conv = conv[:2*len(a)-1]  # maybe linear conv length
      max_val = conv.max()
      # compute w:
      exp_terms = np.exp(beta * (conv - max_val))
      w = exp_terms / exp_terms.sum()
      # compute gradient w.r.t a
      rev_a = a[::-1]
      grad_fft = np.fft.ifft(np.fft.fft(w, conv_len) * np.fft.fft(rev_a, conv_len)).real

      a = a - step
      # project onto simplex
      a = project_simplex(a, s=1.0)
      # increase beta
      beta = min(200.0, beta * beta_factor)
      # optionally decay step size
      step_size *= step_decay
      # evaluate score occasionally, e.g., every 10 iters or each

Implementation of LP using SciPy:

def lp_refinement(a, M, t_mix, deadline):

    n = len(a)
    # Build A as sparse matrix:
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    data = []
    for i in range(n):
        start = i
        end = i + n
        # rows indices start..end-1
        # each row k where k = start + j (j from 0..n-1)
        # col i = coefficient a[i]
        # So for each k from start to start+n-1, we add entry.
        for k in range(start, end):
            rows.append(k)
            cols.append(i)
            data.append(a[i])
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # RHS
    b_ub = np.full(2*n-1, M)
    # bounds (0, None)
    bounds = [(0, None)] * n
    c = -np.ones(n)  # maximize sum => minimize -sum

    # Solve with linprog
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    g_opt = res.x  # size n
    sum_g = g_opt.sum()
    if sum_g <= 0:
        return None
    g_scaled = g_opt / sum_g
    # convex combination
    new_a = (1 - t_mix) * a + t_mix * g_scaled
    # project onto simplex
    new_a = project_simplex(new_a, s=1.0)
    return new_a

        start_time = time.time()
    a = a_init.copy()
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step = 0.3
    step_decay = 0.9995
    beta = 1.0
    beta_factor = 1.02
    beta_max = 200.0
    best_seq = a.copy()
    best_score = evaluate_sequence(a.tolist())
    # Record iteration when improved
    improve_counter = 0

    for t in range(1, max_iters+1):
        if time.time() > deadline - 0.2:
            break
        # Compute self-convolution
        conv = self_convolution(a)
        max_val = conv.max()
        # softmax
        exps = np.exp(beta * (conv - max_val))
        w = exps / exps.sum()
        # gradient
        grad = gradient_wrt_a(a, w)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step_vec = step * m_hat / (np.sqrt(v_hat) + eps)
        a -= step_vec
        # project onto simplex
        a = project_simplex(a, 1.0)
        # Update beta and step
        beta = min(beta_max, beta * beta_factor)
        step *= step_decay
        # occasionally evaluate real score
        if t % 10 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score - 1e-9:
                best_score = sc
                best_seq = a.copy()
                improve_counter = 0
                print(f"[adam] iter={t} n={n} score={sc:.6f} beta={beta:.2f}")
            else:
                improve_counter += 1
                # early stop if no improvement for many iterations
                if improve_counter >= 200:
                    break
    return best_seq, best_score

    import scipy.optimize as opt
    import scipy.sparse as sp
    n = len(a)
    # build matrix A
    rows = []
    cols = []
    data = []
    for i in range(n):
        start = i
        # end exclusive = i + n
        for k in range(start, start + n):
            rows.append(k)
            cols.append(i)
            data.append(a[i])
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)
    c = -np.ones(n)  # maximize sum(g)
    # bounds: g_i >= 0
    bounds = [(0, None)] * n

    if not res.success:
        return None
    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return None
    g_scaled = g / sum_g
    new_a = (1 - mix_weight) * a + mix_weight * g_scaled
    new_a = project_simplex(new_a, 1.0)
    return new_a

"""
Simple Adam‐based optimizer with optional linear‑programming refinement.
Starts from several initialisations (arcsine, random Dirichlet) for medium
length sequences (≈1000‑2000).  The objective
    2·n·max(a*a)/(∑a)²
is approximated by a smooth soft‑max of the self‑convolution, and we
optimise on the simplex using Adam.  If SciPy’s ``linprog`` is available
we additionally solve the linear programme
    maximise 1ᵀ g    s.t. g≥0,  a∗g ≤ M
to “fill the slack’’ after a few Adam steps.  The best sequence found
before the time budget expires is returned.
"""

import time
import numpy as np

def arcsine_sequence(n: int) -> list[float]:
    """Return length‑n non‑negative coefficients proportional to
    binom(2i,i)/4**i and L¹‑normalised to sum = 1."""
    seq = []
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2.0 * i + 1.0) / (2.0 * (i + 1))
    seq = np.array(seq, dtype=float)
    seq /= seq.sum()
    return seq.tolist()


def project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Small utility – smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def self_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (non‑circular) self‑convolution a∗a using FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of soft‑max( a∗a ) with respect to a.
    w is the soft‑max weight vector (same length as the convolution)."""
    n = a.shape[0]
    rev_a = a[::-1]
    size = _next_pow2(len(w) + n - 1)
    fw = np.fft.rfft(w, n=size)
    fr = np.fft.rfft(rev_a, n=size)
    conv = np.fft.irfft(fw * fr, n=size)
    start = n - 1
    grad = 2.0 * conv[start : start + n]
    return grad


def adam_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 2000,
    beta_start: float = 1.0,
    beta_factor: float = 1.02,
    beta_max: float = 200.0,
    step0: float = 0.3,
    step_decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the simplex (sum = 1). Returns the best sequence
    observed (as a NumPy array) together with its true evaluation score."""
    a = a0.copy()
    n = a.shape[0]
    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    beta = beta_start
    step = step0

    best_seq = a.copy()
    best_score = evaluate_sequence(a.tolist())
    no_imp = 0

    for t in range(1, max_iters + 1):
        if time.time() > deadline - 0.2:
            break

        conv = self_convolution(a)
        max_val = conv.max()
        # soft‑max over the convolution entries
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / exp_vals.sum()

        # gradient w.r.t. a
        grad = gradient_wrt_a(a, w)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the probability simplex
        a = project_simplex(a, s=1.0)

        # schedule parameters
        beta = min(beta_max, beta * beta_factor)
        step *= step_decay

        # occasional true evaluation (every 10 steps)
        if t % 10 == 0:
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                no_imp = 0
                print(f"[adam] iter={t:4d} n={n:<5d} score={cur_score:.6f} β={beta:.2f}")
            else:
                no_imp += 1
                if no_imp >= 300:
                    # stagnated – stop early
                    break

    return best_seq, best_score


def lp_refinement(a: np.ndarray, M: float, mix: float, deadline: float):
    """Linear‑programming refinement using scipy.optimize.linprog.
    Returns a new sequence or ``None`` if the LP fails."""
    try:
        from scipy.optimize import linprog
        from scipy.sparse import csr_matrix
    except Exception:
        return None

    n = a.shape[0]
    # build Toeplitz‑type constraint matrix A (shape (2n‑1, n))
    rows = []
    cols = []
    data = []
    for i in range(n):
        start = i
        for k in range(start, start + n):
            rows.append(k)
            cols.append(i)
            data.append(a[i])
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    b_ub = np.full(2 * n - 1, M, dtype=float)
    c = -np.ones(n, dtype=float)  # maximise sum(g) → minimise –sum(g)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method="highs")
    if not res.success or res.x is None:
        return None

    g = res.x
    s = g.sum()
    if s <= 0:
        return None
    g_scaled = g / s
    new_a = (1.0 - mix) * a + mix * g_scaled
    new_a = project_simplex(new_a, s=1.0)
    return new_a


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Search for a non‑negative step‑function sequence with a low evaluation
    score.  The routine respects the supplied time budget and returns the
    best sequence (as a plain Python list) found so far."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # a tiny safety margin

    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # optional global starting point
    try:
        global height_sequence_1
        if isinstance(height_sequence_1, (list, tuple, np.ndarray)):
            tmp = np.asarray(height_sequence_1, dtype=float)
            if tmp.size > 0:
                tmp = np.maximum(tmp, 0.0)
                if tmp.sum() > 0:
                    tmp /= tmp.sum()
                    sc = evaluate_sequence(tmp.tolist())
                    print(f"[global] n={tmp.shape[0]} score={sc:.6f}")
                    if sc < best_score:
                        best_score = sc
                        best_seq = tmp.copy()
                        best_n = tmp.shape[0]
    except Exception:
        pass

    # ------------------------------------------------------------------
    # candidate lengths – we keep them moderate to stay inside the
    # LP memory budget while still allowing a good asymptotic value.
    lengths = [800, 1200, 1600, 2000]

    for n in lengths:
        if time.time() >= deadline:
            break

        # ---- start from the classic arcsine shape
        init = np.asarray(arcsine_sequence(n), dtype=float)
        sc = evaluate_sequence(init.tolist())
        print(f"[init] n={n:<5d} arcsine score={sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_seq = init.copy()
            best_n = n

        # ---- a few random Dirichlet restarts
        for restart in range(3):
            if time.time() >= deadline:
                break
            rand = np.random.rand(n)
            rand /= rand.sum()
            # blend a bit with the arcsine shape to stay in a promising region
            alpha = np.random.rand()
            seq0 = alpha * init + (1 - alpha) * rand
            seq0 /= seq0.sum()
            cur_sc = evaluate_sequence(seq0.tolist())
            print(f"[init] n={n:<5d} restart={restart} blended score={cur_sc:.6f}")

            # ---- Adam optimisation
            seq_opt, seq_score = adam_optimize(
                seq0,
                deadline=deadline,
                max_iters=4000,
                beta_start=1.0,
                beta_factor=1.02,
                beta_max=200.0,
                step0=0.3,
                step_decay=0.9995,
            )
            if seq_score < best_score:
                best_score = seq_score
                best_seq = seq_opt.copy()
                best_n = n
                print(f"[adam] n={n} restart={restart} -> new best {seq_score:.6f}")

            # ---- occasional LP refinement (only if time permits)
            if time.time() < deadline - 2.0:
                conv = self_convolution(best_seq)
                M = conv.max()
                # we use a decreasing mix weight, starting from 0.2
                mix = 0.2
                refined = lp_refinement(best_seq, M, mix, deadline)
                if refined is not None:
                    sc_ref = evaluate_sequence(refined.tolist())
                    if sc_ref < best_score:
                        best_score = sc_ref
                        best_seq = refined.copy()
                        best_n = n
                        print(f"[LP] n={n} refinement improved to {sc_ref:.6f}")

    # ------------------------------------------------------------------
    # final fine‑tuning on the overall best sequence (remaining time)
    if best_seq is not None and time.time() < deadline - 1.0:
        seq_opt, seq_score = adam_optimize(
            best_seq,
            deadline=deadline,
            max_iters=8000,
            beta_start=5.0,
            beta_factor=1.02,
            beta_max=300.0,
            step0=0.15,
            step_decay=0.9998,
        )
        if seq_score < best_score:
            best_score = seq_score
            best_seq = seq_opt.copy()
            print(f"[final] further improvement to {seq_score:.6f}")

    # ------------------------------------------------------------------
    # final sanity check – enforce non‑negativity and unit sum
    best_seq = np.maximum(best_seq, 0.0)
    if best_seq.sum() == 0:
        # fallback – uniform distribution (should never happen)
        best_seq = np.ones(best_seq.shape[0], dtype=float) / best_seq.shape[0]
    else:
        best_seq = best_seq / best_seq.sum()

    print(f"Best sequence length = {len(best_seq)}  score = {best_score:.6f}")
    return best_seq.tolist()

"""
Simple Adam‑based optimizer with optional linear‑programming refinement.

The objective to minimise is

    2·n·max(conv(a,a)) / (∑ a)²   (∑ a is forced to 1)

We approximate the non‑smooth max by a soft‑max with temperature β and
optimise the resulting smooth function on the probability simplex using
the Adam optimiser.  For a few iterations we also solve the linear program

    maximise   1ᵀ g
    subject to   g ≥ 0 ,   a ∗ g ≤ M   (element‑wise)

with SciPy’s ``linprog`` (high‑s) – this “fills the slack’’ without raising
the current maximal convolution M.  Several initialisations (arcsine,
random Dirichlet, mixed blends) are tried for a handful of moderate
lengths (≈800‑2000).  The best sequence seen before the time budget runs
out is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> list[float]:
    """Return a length‑n non‑negative sequence proportional to
    binom(2i,i)/4**i and normalised to have L¹‑norm = 1."""
    seq = []
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2.0 * i + 1.0) / (2.0 * (i + 1))
    seq = np.array(seq, dtype=float)
    seq /= seq.sum()
    return seq.tolist()


# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper – smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def self_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (non‑circular) self‑convolution a∗a via FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of  (1/β)·log‑sum‑exp(β·(a∗a))   w.r.t. a.
    w is the soft‑max weight vector (same length as a∗a)."""
    n = a.shape[0]
    rev_a = a[::-1]
    size = _next_pow2(len(w) + n - 1)
    fw = np.fft.rfft(w, n=size)
    fr = np.fft.rfft(rev_a, n=size)
    conv = np.fft.irfft(fw * fr, n=size)
    start = n - 1
    grad = 2.0 * conv[start : start + n]
    return grad


# ----------------------------------------------------------------------
def adam_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 2000,
    beta_start: float = 1.0,
    beta_factor: float = 1.02,
    beta_max: float = 200.0,
    step0: float = 0.3,
    step_decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the probability simplex.
    Returns the best sequence encountered and its true evaluation score.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    beta = beta_start
    step = step0

    best_seq = a.copy()
    best_score = evaluate_sequence(a.tolist())
    stagnation = 0

    for t in range(1, max_iters + 1):
        if time.time() > deadline - 0.2:
            break

        conv = self_convolution(a)
        max_val = conv.max()

        # soft‑max over convolution entries
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / exp_vals.sum()

        # gradient wrt a
        grad = gradient_wrt_a(a, w)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # bring back onto the simplex
        a = project_simplex(a, s=1.0)

        # schedule changes
        beta = min(beta_max, beta * beta_factor)
        step *= step_decay

        # occasional exact evaluation
        if t % 10 == 0:
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                stagnation = 0
                print(f"[adam] iter={t:4d} n={n:<5d} score={cur_score:.6f} β={beta:.2f}")
            else:
                stagnation += 1
                if stagnation >= 300:
                    break

    return best_seq, best_score


# ----------------------------------------------------------------------
def lp_refinement(a: np.ndarray, M: float, mix: float, deadline: float):
    """Linear‑programming refinement (SciPy linprog).  Returns a new sequence
    or ``None`` if the LP fails or time runs out."""
    try:
        from scipy.optimize import linprog
        from scipy.sparse import csr_matrix
    except Exception:
        return None

    n = a.shape[0]

    # Build the Toeplitz‑type matrix for the convolution constraints
    rows = []
    cols = []
    data = []
    for i in range(n):
        start = i
        for k in range(start, start + n):
            rows.append(k)
            cols.append(i)
            data.append(a[i])
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    b_ub = np.full(2 * n - 1, M, dtype=float)
    c = -np.ones(n, dtype=float)          # maximise Σ g  ⇔ minimise –Σ g
    bounds = [(0, None)] * n               # g ≥ 0

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success or res.x is None:
        return None

    g = res.x
    g_sum = g.sum()
    if g_sum <= 0:
        return None
    g_scaled = g / g_sum
    new_a = (1.0 - mix) * a + mix * g_scaled
    new_a = project_simplex(new_a, s=1.0)
    return new_a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence with a low evaluation
    score.  The routine respects the supplied time budget and returns the
    best sequence (as a plain Python list) found so far.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Optional global start point (may be defined by the surrounding runner)
    try:
        global height_sequence_1
        if isinstance(height_sequence_1, (list, tuple, np.ndarray)):
            tmp = np.asarray(height_sequence_1, dtype=float)
            if tmp.size > 0:
                tmp = np.maximum(tmp, 0.0)
                if tmp.sum() > 0:
                    tmp /= tmp.sum()
                    sc = evaluate_sequence(tmp.tolist())
                    print(f"[global] n={tmp.shape[0]} score={sc:.6f}")
                    if sc < best_score:
                        best_score = sc
                        best_seq = tmp.copy()
                        best_n = tmp.shape[0]
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Candidate lengths – moderate sizes keep the LP tractable.
    lengths = [800, 1200, 1600, 2000]

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- 1) classic arcsine start
        init = np.asarray(arcsine_sequence(n), dtype=float)
        sc = evaluate_sequence(init.tolist())
        print(f"[init] n={n:<5d} arcsine score={sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_seq = init.copy()
            best_n = n

        # ----- 2) a few random Dirichlet restarts
        for restart in range(3):
            if time.time() >= deadline:
                break

            rand = np.random.rand(n)
            rand /= rand.sum()
            # blend with arcsine to stay near a promising region
            alpha = np.random.rand()
            seq0 = alpha * init + (1 - alpha) * rand
            seq0 /= seq0.sum()
            sc0 = evaluate_sequence(seq0.tolist())
            print(f"[init] n={n:<5d} restart={restart} blended score={sc0:.6f}")

            # ----- Adam optimisation
            seq_opt, seq_score = adam_optimize(
                seq0,
                deadline=deadline,
                max_iters=4000,
                beta_start=1.0,
                beta_factor=1.02,
                beta_max=200.0,
                step0=0.3,
                step_decay=0.9995,
            )
            if seq_score < best_score:
                best_score = seq_score
                best_seq = seq_opt.copy()
                best_n = n
                print(f"[adam] n={n} restart={restart} -> new best {seq_score:.6f}")

            # ----- occasional LP refinement (if time permits)
            if time.time() < deadline - 2.0:
                M = self_convolution(best_seq).max()
                refined = lp_refinement(best_seq, M, mix=0.2, deadline=deadline)
                if refined is not None:
                    sc_ref = evaluate_sequence(refined.tolist())
                    if sc_ref < best_score:
                        best_score = sc_ref
                        best_seq = refined.copy()
                        best_n = n
                        print(f"[LP] n={n} refinement improved to {sc_ref:.6f}")

    # ------------------------------------------------------------------
    # Final fine‑tuning on the overall best sequence with remaining time.
    if best_seq is not None and time.time() < deadline - 1.0:
        seq_opt, seq_score = adam_optimize(
            best_seq,
            deadline=deadline,
            max_iters=8000,
            beta_start=5.0,
            beta_factor=1.02,
            beta_max=300.0,
            step0=0.15,
            step_decay=0.9998,
        )
        if seq_score < best_score:
            best_score = seq_score
            best_seq = seq_opt.copy()
            print(f"[final] further improvement to {seq_score:.6f}")

    # ------------------------------------------------------------------
    # Sanity – enforce non‑negativity and unit L¹‑norm.
    best_seq = np.maximum(best_seq, 0.0)
    if best_seq.sum() == 0:
        best_seq = np.ones_like(best_seq) / best_seq.shape[0]
    else:
        best_seq = best_seq / best_seq.sum()

    print(f"Best sequence length = {len(best_seq)}  score = {best_score:.6f}")
    return best_seq.tolist()
