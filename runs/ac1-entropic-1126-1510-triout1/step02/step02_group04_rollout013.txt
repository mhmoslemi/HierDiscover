def downsample(arr, target_len):
    n = len(arr)
    if n == target_len:
        return arr.copy()
    if n < target_len:
        # pad or replicate
        pad = np.full(target_len - n, arr.mean())
        return np.concatenate([arr, pad])
    else:
        factor = n / target_len
        # average over blocks:
        out = np.empty(target_len)
        for i in range(target_len):
            start = int(round(i * factor))
            end = int(round((i+1) * factor))
            if end <= start:
                end = start+1
            out[i] = arr[start:end].mean()
        return out

"""
LP‑based iterative improvement for the step‑function problem.

The algorithm works with a fixed length `n` (default 2000) and keeps
the coefficients on the probability simplex (non‑negative, sum = 1).

1. Initialise `f` from the best known global construction (`height_sequence_1`)
   or from a random point.
2. Repeatedly:
   * Compute the maximal autocorrelation `M = max(f*f)`.
   * Build the linear constraint matrix `A` such that `A @ g` equals the
     convolution `f*g`.
   * Solve the linear programme
         maximise 1ᵀ g
         s.t.   A @ g ≤ M ,  g ≥ 0
     (SciPy's high‑performance HiGHS solver is used).
   * Scale the optimum `g` to have the same L¹‑norm as `f` (here 1).
   * Mix `f` and `g` with a small step size `t` (line‑search) and keep the
     best mixture that lowers the objective `2·n·max(f*f)`.
   * Occasionally apply a few projected‑gradient steps that directly
     decrease the largest convolution entries.
   * If no progress is made, perform a small random perturbation to escape
     a possible stall.

The loop stops when the time budget (≈ 1000 s) is exhausted or when the
objective drops below the target `1.503`.  The best sequence found is
returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _clip_and_normalise(v: np.ndarray) -> np.ndarray:
    """Clip negative entries to zero and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        v = np.full_like(v, 1.0 / v.size)
    else:
        v = v / s
    return v


def _downsample(arr: np.ndarray, target_len: int) -> np.ndarray:
    """Simple block‑wise down‑sampling of a 1‑D array."""
    n = arr.size
    if n == target_len:
        return arr.copy()
    if n < target_len:
        # Pad with the mean value if the source is shorter.
        pad = np.full(target_len - n, arr.mean())
        return np.concatenate([arr, pad])
    # n > target_len : average over roughly equal sized blocks.
    factor = n / target_len
    out = np.empty(target_len, dtype=np.float64)
    for i in range(target_len):
        start = int(round(i * factor))
        end = int(round((i + 1) * factor))
        if end <= start:
            end = start + 1
        out[i] = arr[start:end].mean()
    return out


def _score_on_simplex(seq: np.ndarray) -> float:
    """Objective when the sequence is already normalised (sum = 1)."""
    n = seq.size
    conv = np.convolve(seq, seq)
    return float(2.0 * n * conv.max())


def _build_constraint_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A such that (A @ g)[k] = (f * g)[k] for all k.
    Shape of A is (2·n‑1, n).
    """
    n = f.shape[0]
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


def _multi_lag_gradient_step(seq: np.ndarray, eta: float, top_k: int = 3) -> np.ndarray:
    """
    One projected gradient step that reduces the `top_k` largest
    convolution entries.  `eta` is a step size.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    # indices of the k largest entries (ties broken arbitrarily)
    top_idxs = np.argpartition(-conv, range(top_k))[:top_k]

    g = np.zeros_like(seq)
    idx = np.arange(n)
    for k in top_idxs:
        j = k - idx
        mask = (j >= 0) & (j < n)
        g[mask] += 2.0 * seq[j[mask]]

    new_seq = seq - eta * g
    return _clip_and_normalise(new_seq)


def _lp_improve(f: np.ndarray, max_time: float) -> tuple[np.ndarray, float]:
    """
    Perform one LP‑based improvement step for a fixed vector `f`
    (already normalised).  Returns the best new vector and its score.
    """
    n = f.size
    # current maximal autocorrelation
    conv_ff = np.convolve(f, f)
    M = conv_ff.max()

    # Build the linear constraint matrix A (dense, size (2n‑1)×n)
    A = _build_constraint_matrix(f)

    # Linear programme: maximise 1ᵀg  subject to A g ≤ M, g ≥ 0
    c = -np.ones(n, dtype=np.float64)        # maximise sum(g)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Solve using HiGHS (fast interior‑point/simplex implementation)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True, 'time_limit': max_time})

    if not res.success or res.x is None:
        # LP failed – fall back to the original vector
        return f, _score_on_simplex(f)

    g = res.x
    s_g = g.sum()
    if s_g <= 0.0:
        return f, _score_on_simplex(f)

    # Scale g back to the simplex (sum = 1)
    g = g / s_g

    # Try a few mixing coefficients and keep the best improvement.
    best_seq = f
    best_score = _score_on_simplex(f)
    improved = False
    for t in (0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01):
        mixed = (1.0 - t) * f + t * g
        mixed = _clip_and_normalise(mixed)
        score = _score_on_simplex(mixed)
        if score < best_score - 1e-12:
            best_seq, best_score = mixed, score
            improved = True
            break

    if not improved:
        # No mixing helped – keep the original vector.
        best_seq, best_score = f, _score_on_simplex(f)

    return best_seq, best_score

    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns a list of heights (float) that achieves a low upper bound.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 0.5)   # small safety margin

    # --------------------------------------------------------------
    # Choose a reasonable dimension (large enough to give flexibility,
    # small enough to keep the LP tractable).
    # --------------------------------------------------------------
    TARGET_N = 2000

    # Try to initialise from the previously discovered construction.
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        base = _downsample(base, TARGET_N)
    except Exception:
        base = np.random.rand(TARGET_N)

    f = _clip_and_normalise(base)          # start on the simplex
    best_f = f.copy()
    best_score = _score_on_simplex(best_f)
    print(f"[Init] n = {TARGET_N}, initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Main optimisation loop (LP steps + occasional gradient refinements)
    # --------------------------------------------------------------
    iteration = 0
    no_improve_iters = 0
    while time.time() < deadline:
        iteration += 1
        time_left = deadline - time.time()
        if time_left <= 0.0:
            break

        # ----- LP improvement -------------------------------------------------
        new_f, new_score = _lp_improve(best_f, max_time=time_left)
        if new_score < best_score - 1e-12:
            best_f, best_score = new_f, new_score
            print(f"[Iter {iteration:5d}] LP improved score = {best_score:.6f}")
            no_improve_iters = 0
        else:
            no_improve_iters += 1

        # ----- Gradient refinement (few steps) ---------------------------------
        # This cheap local step often squeezes out a little extra gain.
        for step in range(5):
            tmp = _multi_lag_gradient_step(best_f, eta=0.5 / (step + 1.0), top_k=3)
            tmp_score = _score_on_simplex(tmp)
            if tmp_score < best_score - 1e-12:
                best_f, best_score = tmp, tmp_score
                print(f"[Iter {iteration:5d}] Grad‑step {step+1} improved to {best_score:.6f}")

        # ----- Stagnation handling --------------------------------------------
        if no_improve_iters >= 8:
            # Random jitter to escape potential local minima.
            jitter = 0.02 * (np.random.rand(TARGET_N) - 0.5)
            cand = best_f + jitter
            cand = _clip_and_normalise(cand)
            cand_score = _score_on_simplex(cand)
            if cand_score < best_score:
                best_f, best_score = cand, cand_score
                print(f"[Iter {iteration:5d}] Random jitter improved to {best_score:.6f}")
            no_improve_iters = 0

        # ----- Early exit if we already beat the target -----------------------
        if best_score < 1.503:
            print(f"[Done] Target reached after {iteration} iterations, score = {best_score:.6f}")
            break

    print(f"[Done] Best score after {iteration} iterations = {best_score:.6f}")
    # Return the best sequence as a plain Python list.
    return best_f.tolist()
