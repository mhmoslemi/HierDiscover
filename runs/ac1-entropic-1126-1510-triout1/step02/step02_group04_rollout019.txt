def project_onto_simplex(v):
    """Project onto probability simplex."""
    # Works for 1D array.
    # Using sorting method
    n = v.shape[0]
    u = np.sort(v)[::-1]   # decreasing
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].sum() / rho
    w = np.maximum(v - theta, 0)
    return w

def targeted_move(seq, frac=0.01):
    n = len(seq)
    conv = conv_fft(seq)
    k = int(conv.argmax())
    # compute contributions for each pair:
    # contributions = seq[i] * seq[k-i] if indices valid
    i_arr = np.arange(n)
    j_arr = k - i_arr
    mask = (j_arr >= 0) & (j_arr < n)
    contrib = np.zeros(n)
    contrib[mask] = seq[i_arr[mask]] * seq[j_arr[mask]]
    # choose index with max contrib
    i_max = int(np.argmax(contrib))
    j_max = k - i_max
    src = i_max if seq[i_max] >= seq[j_max] else j_max
    # pick random destination not equal src
    dst = np.random.randint(n)
    while dst == src:
        dst = np.random.randint(n)
    amount = frac * seq[src] * np.random.rand()
    new_seq = seq.copy()
    new_seq[src] -= amount
    new_seq[dst] += amount
    return project_onto_simplex(new_seq)

    """Exact Euclidean projection onto the probability simplex."""
    # Implementation per Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    # Find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all <=0, return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = cssv[cond].sum() / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Hybrid projected sub‑gradient optimiser for the step‑function problem.

The algorithm works on the probability simplex (non‑negative entries,
sum = 1).  Starting from the supplied construction (height_sequence_1) or
a random Dirichlet point it repeatedly

* evaluates the convolution by means of an FFT,
* builds a sub‑gradient from the *top_k* largest convolution entries,
* takes a projected gradient step,
* occasionally applies a small targeted mass‑transfer move that directly
  attacks the current maximal lag,
* restarts from a fresh random point when no improvement has been seen
  for a while.

Both the gradient and the targeted move are projected back onto the
simplex by an exact Euclidean projection (Duchi et al.).  The whole
search runs until the time budget is exhausted and returns the best
sequence found (or the original construction if it happens to be better).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum(x) = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: uniform point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].sum() / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast convolution using a power‑of‑two FFT (linear convolution)
# ----------------------------------------------------------------------
def _conv(seq: np.ndarray) -> np.ndarray:
    """Linear convolution of `seq` with itself via FFT."""
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fft_seq = np.fft.rfft(seq, size)
    conv = np.fft.irfft(fft_seq * fft_seq, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Scoring (the exact evaluation without the safety wrapper)
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv) for a sequence already on the simplex."""
    n = seq.size
    max_conv = _conv(seq).max()
    return float(2.0 * n * max_conv)


# ----------------------------------------------------------------------
# Sub‑gradient built from the `top_k` largest convolution entries.
# ----------------------------------------------------------------------
def _subgradient(seq: np.ndarray, conv: np.ndarray, top_k: int = 3) -> np.ndarray:
    """Sub‑gradient of the (soft) max‑convolution w.r.t. `seq`."""
    n = seq.size
    idx = np.arange(n)
    # indices of the k largest convolution values (ties broken arbitrarily)
    top_idxs = np.argpartition(-conv, range(top_k))[:top_k]
    g = np.zeros_like(seq)
    for k in top_idxs:
        delta = k - idx
        mask = (delta >= 0) & (delta < n)
        g[mask] += 2.0 * seq[delta[mask]]
    return g


# ----------------------------------------------------------------------
# Targeted move: steal a little mass from the larger member of the pair
# that contributes most to the current maximal convolution entry.
# ----------------------------------------------------------------------
def _targeted_move(seq: np.ndarray, frac: float = 0.02) -> np.ndarray:
    n = seq.size
    conv = _conv(seq)
    k = int(conv.argmax())

    i_arr = np.arange(n)
    j_arr = k - i_arr
    mask = (j_arr >= 0) & (j_arr < n)

    if not np.any(mask):
        # fallback – random jitter
        cand = seq + frac * (np.random.rand(n) - 0.5)
        return _project_to_simplex(cand)

    contrib = np.zeros_like(seq)
    contrib[mask] = seq[i_arr[mask]] * seq[j_arr[mask]]

    i_max = int(np.argmax(contrib))
    j_max = k - i_max
    src = i_max if seq[i_max] >= seq[j_max] else j_max

    dst = np.random.randint(n)
    while dst == src:
        dst = np.random.randint(n)

    amount = frac * seq[src] * np.random.rand()
    cand = seq.copy()
    cand[src] -= amount
    cand[dst] += amount
    return _project_to_simplex(cand)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Optimise a non‑negative step‑function under the evaluation
    ``2·n·max(conv)`` (the denominator after normalising the L1‑norm).
    Returns a list of heights.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 0.5)   # small safety margin

    # --------------------------------------------------------------
    # Initialise a starting point (prefer the known construction)
    # --------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        seq = _project_to_simplex(base)
    except Exception:
        # fallback – random Dirichlet point with moderate length
        n = 7200                     # close to the size of the best known solution
        seq = np.random.rand(n)
        seq = _project_to_simplex(seq)

    # --------------------------------------------------------------
    # Records for the best solution seen so far
    # --------------------------------------------------------------
    best_seq = seq.copy()
    best_score = _score(best_seq)
    print(f"[Init] n = {n}, initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------------
    top_k = 5                # how many peaks feed the sub‑gradient
    eta0 = 0.06              # initial step size
    restart_interval = 2000 # generations without improvement -> restart
    iter_cnt = 0
    last_improve = 0
    avg_success = 0.0

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        iter_cnt += 1

        # ----- evaluate current sequence -----
        conv = _conv(seq)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:5d}] New best score = {best_score:.6f}")

        # ----- sub‑gradient step -----
        # step size decays slowly
        eta = eta0 / (1.0 + 0.0005 * iter_cnt)
        grad = _subgradient(seq, conv, top_k=top_k)
        cand = seq - eta * grad
        cand = _project_to_simplex(cand)

        # ----- occasional targeted perturbation (every 30 iters) -----
        if iter_cnt % 30 == 0:
            tmove = _targeted_move(best_seq, frac=0.015)
            tscore = _score(tmove)
            if tscore < best_score:
                best_score = tscore
                best_seq = tmove.copy()
                last_improve = iter_cnt
                print(f"[Iter {iter_cnt:5d}] Targeted move improved best = {best_score:.6f}")
            # also let the current candidate be the targeted move
            cand = tmove

        # ----- small random jitter (helps escaping plateaux) -----
        if iter_cnt % 10 == 0:
            jitter = (np.random.rand(n) - 0.5) * 0.008
            cand = cand + jitter
            cand = _project_to_simplex(cand)

        # ----- accept the candidate (always, we are doing a descent) -----
        seq = cand

        # ----- stagnation restart -----
        if iter_cnt - last_improve > restart_interval:
            seq = np.random.rand(n)
            seq = _project_to_simplex(seq)
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:5d}] Restarted due to stagnation")

    # --------------------------------------------------------------
    # After the time budget pick the best among our search and the
    # original construction (the latter might still be the winner).
    # --------------------------------------------------------------
    try:
        base_seq = np.array(height_sequence_1, dtype=float)
        base_seq = _project_to_simplex(base_seq)
        base_score = _score(base_seq)
        if base_score < best_score:
            best_seq, best_score = base_seq, base_score
    except Exception:
        pass

    print(f"[Done] Best score = {best_score:.6f} after {iter_cnt} iterations")
    return best_seq.tolist()
