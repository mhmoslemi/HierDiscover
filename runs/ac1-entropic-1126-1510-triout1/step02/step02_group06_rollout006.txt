def project_simplex(v):
    # returns v projected onto simplex sum=1, v>=0
    # using algorithm from Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import time
import numpy as np

def arcsine_sequence(n: int) -> list[float]:
    """Return the first n coefficients of (1‑z)^‑1/2 (the arcsine law),
    normalized to have L1‑norm 1."""
    seq = []
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    seq = np.array(seq, dtype=float)
    seq /= seq.sum()
    return seq.tolist()

def uniform_sequence(n: int) -> list[float]:
    """Uniform probability vector of length n."""
    return (np.full(n, 1.0 / n)).tolist()

def random_dirichlet_sequence(n: int, rng: np.random.Generator, alpha: float = 1.0) -> list[float]:
    """Draw a random probability vector from a Dirichlet(alpha) distribution."""
    vec = rng.dirichlet(np.full(n, alpha))
    return vec.tolist()

def smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Given a convolution vector `conv` (sum of products, not necessarily normalized),
    compute a soft‑max weight vector w_k proportional to exp(beta * conv_k)."""
    # subtract max for numerical stability
    max_c = conv.max()
    exps = np.exp(beta * (conv - max_c))
    w = exps / exps.sum()
    return w

def gradient_smooth(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of the smooth max S_beta(a) = (1/beta) log sum_k exp(beta * (a*a)_k)
    with respect to a (assumes sum(a)=1).  This equals 2 * correlation(w, a)."""
    # np.correlate implements exactly sum_k w_{i+k} a_k when mode='valid'
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad

def metric_from_seq(a: np.ndarray) -> float:
    """Return the evaluation metric for a probability vector a (sum to 1)."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max()

def optimize_sequence(init_seq: np.ndarray, deadline: float) -> tuple[list[float], float]:
    """
    Refine a probability vector using a smoothed‑max mirror‑descent.
    The routine runs until ``deadline`` (a unix timestamp) and returns the
    best sequence seen along with its metric.
    """
    a = init_seq.copy()
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_metric = metric_from_seq(a)

    # hyper‑parameters
    beta = 2.0          # start temperature (smoothness)
    beta_inc = 0.5
    beta_max = 80.0
    eta = 0.5           # step size for multiplicative update
    eta_inc = 1.05
    eta_dec = 0.6
    it = 0
    max_iter = 1_000_000  # safety bound

    while time.time() < deadline and it < max_iter:
        it += 1

        # convolution and metric of current a
        conv = np.convolve(a, a)
        max_c = conv.max()
        cur_metric = 2.0 * n * max_c

        # keep best
        if cur_metric < best_metric - 1e-12:
            best_metric = cur_metric
            best_a = a.copy()
            # a small boost for the learning rate when we succeed
            eta = min(eta * eta_inc, 5.0)

        w = smooth_weights(conv, beta)
        grad = gradient_smooth(a, w)

        grad_clipped = np.clip(grad, -100.0, 100.0)
        a_candidate = a * np.exp(-eta * grad_clipped)
        # renormalise to the simplex

        if a_candidate_sum == 0.0:

        else:

        conv_cand = np.convolve(a_candidate, a_candidate)
        cand_metric = 2.0 * n * conv_cand.max()

        if cand_metric < cur_metric:
            a = a_candidate
            eta = min(eta * eta_inc, 5.0)
        else:

                    a[i] -= eps
                    a[j] += eps
                    a = a / a.sum()

        # increase beta slowly (focus more on the true max)
        beta = min(beta + beta_inc, beta_max)

        # tiny safeguard: if eta collapses, we give up early
        if eta < 1e-12:
            break

"""
High‑level search for a step‑function that minimizes ``evaluate_sequence``.

The algorithm mixes several ideas:
  1.  A collection of deterministic and random starting points
      (arcsine, uniform, Dirichlet draws, smoothed arcsine, a previous
      solution if available).
  2.  A *smoothed‑max* mirror‑descent: instead of using the exact
      maximum of the self‑convolution we optimise the soft‑max
      ``S_β(a)= (1/β) log Σ_k exp(β·(a*a)_k)``.
      Its gradient can be written in closed form as
          ∇S_β(a) = 2·corr(w, a) ,
      where ``w`` are the soft‑max weights and ``corr`` is a
      cross‑correlation.  The update is multiplicative
          a_i ← a_i·exp(‑η·∇S_β(a)_i)
      followed by normalisation (the iterate stays on the probability
      simplex).  The temperature ``β`` is increased slowly so the
      optimisation gradually focuses on the true maximum.
  3.  Adaptive step‑size (increase on success, decrease on failure) and
      occasional random jitter to escape plateaux.
  4.  A simple time‑budget manager that distributes the total 1000 s
      among several sequence lengths and several restarts.

The code returns the best sequence found before the deadline.
"""

import time
import numpy as np


def arcsine_sequence(n: int) -> list[float]:
    """First ``n`` coefficients of (1‑z)^‑1/2, normalised to L1‑norm = 1."""
    seq = []
    cur = 1.0
    tot = 0.0
    for i in range(n):
        seq.append(cur)
        tot += cur
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    seq = np.array(seq, dtype=float)
    seq /= seq.sum()
    return seq.tolist()


def uniform_sequence(n: int) -> list[float]:
    """Uniform probability vector of length ``n``."""
    return (np.full(n, 1.0 / n, dtype=float)).tolist()


def random_dirichlet_sequence(n: int, rng: np.random.Generator,
                              alpha: float = 1.0) -> list[float]:
    """Random probability vector drawn from Dirichlet(alpha)."""
    vec = rng.dirichlet(np.full(n, alpha))
    return vec.tolist()


def smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights w_k ∝ exp(β·conv_k).  The subtraction of the max
    improves numerical stability."""
    max_c = conv.max()
    exps = np.exp(beta * (conv - max_c))
    w = exps / exps.sum()
    return w


def gradient_smooth(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of the smooth max w.r.t. ``a``.
    Result = 2·corr(w, a)  (cross‑correlation, mode='valid')."""
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad


def metric_from_seq(a: np.ndarray) -> float:
    """Exact evaluation metric for a probability vector ``a`` (∑a=1):
    2·n·max((a*a))."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max()


def optimize_sequence(init_seq: np.ndarray, deadline: float) -> tuple[list[float], float]:
    """
    Refine a probability vector with smoothed‑max mirror‑descent.
    Runs until ``deadline`` (unix timestamp) and returns the best
    sequence seen together with its metric.
    """
    a = np.maximum(init_seq, 0.0)
    if a.sum() == 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_metric = metric_from_seq(a)

    # hyper‑parameters
    beta = 2.0          # start temperature
    beta_inc = 0.5
    beta_max = 80.0
    eta = 0.5           # multiplicative step size
    eta_inc = 1.05
    eta_dec = 0.6
    it = 0
    max_iter = 1_000_000

    while time.time() < deadline and it < max_iter:
        it += 1

        # current metric
        conv = np.convolve(a, a)
        cur_metric = 2.0 * n * conv.max()

        # keep the best ever seen
        if cur_metric < best_metric - 1e-12:
            best_metric = cur_metric
            best_a = a.copy()
            eta = min(eta * eta_inc, 5.0)   # reward improvement

        # ---- smooth‑max gradient step ----
        w = smooth_weights(conv, beta)
        grad = gradient_smooth(a, w)

        # clip gradient to avoid overflow in the exponential
        grad_clipped = np.clip(grad, -100.0, 100.0)

        # multiplicative update on the simplex
        cand = a * np.exp(-eta * grad_clipped)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            cand = np.full_like(a, 1.0 / n)
        else:
            cand = cand / cand_sum

        # evaluate candidate
        cand_metric = 2.0 * n * np.convolve(cand, cand).max()

        if cand_metric < cur_metric:
            a = cand
            eta = min(eta * eta_inc, 5.0)
        else:
            # reject, shrink step size
            eta = max(eta * eta_dec, 1e-8)
            # occasional jitter
            if it % 200 == 0:
                i, j = np.random.randint(0, n, size=2)
                eps = 1e-4
                if a[i] > eps:
                    a[i] -= eps
                    a[j] += eps
                    a = a / a.sum()

        # raise temperature slowly – we move from a smooth proxy
        # towards the true max as optimisation proceeds
        beta = min(beta + beta_inc, beta_max)

        # early exit if the step size has collapsed
        if eta < 1e-12:
            break

    return best_a.tolist(), best_metric
