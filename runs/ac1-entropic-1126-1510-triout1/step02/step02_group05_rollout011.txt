"""
Gradient‑guided mass redistribution to flatten self‑convolution.

The candidate starts from the known high‑quality construction (or a Dirichlet
sample), then repeatedly:
  * compute the full self‑convolution using an FFT (O(n log n));
  * find the index(es) where the convolution attains its maximum;
  * evaluate the sub‑gradient of the max‑convolution w.r.t. each coefficient;
    this gradient tells which positions contribute most to the peak;
  * move a small amount of mass from the coefficient with the largest
    gradient to the coefficient with the smallest gradient (steepest‑descent
    step).  The step size decays linearly with the elapsed time.
The loop runs until the time budget expires; occasional random
pair‑wise swaps are injected to avoid stagnation.
Finally a short random‑pair polishing phase is performed.
The best sequence seen is returned.
"""

import time
import numpy as np

def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Transfer a small amount of mass between two random indices."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)
    delta = (np.random.rand() - 0.5) * sigma
    if delta > 0:
        delta = min(delta, seq[j])
    else:
        delta = -min(-delta, seq[i])
    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Fast full convolution a * a using real FFT (returns length 2*len-1)."""
    n = a.shape[0]
    size = int(2**np.ceil(np.log2(2 * n - 1)))
    A = np.fft.rfft(a, size)
    c = np.fft.irfft(A * A, size)[: 2 * n - 1]
    return c

def _gradient_from_active(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max‑convolution at the active indices."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active_idx:
