def grad_smooth(a, lam):
    n = len(a)
    L = 1 << ((2*n-1).bit_length())
    A = np.fft.fft(a, L)
    conv = np.fft.ifft(A * A).real[:2*n-1]
    # compute w
    exps = np.exp(lam * conv - lam * conv.max())
    w = exps / exps.sum()
    # correlation using FFT:
    W = np.fft.fft(w, L)
    grad_full = np.fft.ifft(W * np.conj(A)).real[:n]  # correlation result
    grad = 4 * n * grad_full
    return grad, conv, w

def project_to_simplex(v):
    # ensures v >=0 and sum=1
    # Using algorithm from Duchi et al. 2008
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all <=0
        return np.full_like(v, 1.0/n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def adam_optimize(a0, deadline, lam_start=5.0, lam_max=200.0, lam_factor=1.02, lr=0.01, iter_max=10**7):
    a = np.array(a0, dtype=np.float64)
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    best = a.copy()
    best_metric = metric_of(a)
    t = 0
    lam = lam_start
    # Compute L for FFT
    L = 1 << ((3*n-2).bit_length())
    while time.time() < deadline and t < iter_max:
        t += 1
        # compute FFT of a
        A_fft = np.fft.fft(a, L)
        conv = np.fft.ifft(A_fft * A_fft).real[:2*n-1]
        # compute w
        # Use scaling for exponent to avoid overflow
        max_val = conv.max()
        exps = np.exp(lam * (conv - max_val))
        w = exps / exps.sum()
        # compute gradient via correlation
        W_fft = np.fft.fft(w, L)
        corr = np.fft.ifft(W_fft * np.conj(A_fft)).real[:n]  # correlation result
        grad = 4.0 * n * corr
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a_new = a - step
        # projection to simplex
        a_new = project_to_simplex(a_new)
        a = a_new
        # evaluation metric
        cur_metric = metric_of(a)  # conv max
        if cur_metric < best_metric:
            best_metric = cur_metric
            best = a.copy()
        # increase lambda gradually
        if t % 500 == 0:
            lam = min(lam * lam_factor, lam_max)
    return best, best_metric

The evaluation function is  

\[
\text{score}(a)=\frac{2n\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\]

where `a*a` is the self‑convolution.  
All entries are forced to be non‑negative, therefore we can keep the
vector on the probability simplex (`∑a_i = 1`).  
The only difficult part is the *max* in the numerator.  
We replace the max by a smooth log‑sum‑exp approximation  

\[
\max_k x_k\;\approx\;\frac{1}{\lambda}\log\!\Bigl(\sum_k e^{\lambda x_k}\Bigr)
\]

and minimise the smoothed objective with a **projected Adam** optimiser.
The smoothing parameter `λ` is increased slowly (annealing) so that the
optimizer first sees a gentle landscape and later focuses on the true
maximum.

All heavy work (self‑convolution, the weights for the gradient and the
gradient itself) is performed with FFT – a linear‑time operation for the
chosen sizes (≤ 4000).  The algorithm is fast enough to perform many
thousands of iterations within the 1000 s budget.

The overall search strategy

* try a handful of lengths (`≈1500…3500`);
* for each length run a few Adam restarts (different random jitter);
* keep the best sequence;
* finish with a tiny hill‑climbing/annealing pass to polish the result.

The code below follows exactly this plan and returns a list of
non‑negative heights ready for `evaluate_sequence`.  It is completely
self‑contained and uses only the allowed standard libraries.

```python
"""
High‑quality step‑function search.

The algorithm works on the probability simplex (∑a_i = 1, a_i ≥ 0).
It minimises a smooth approximation of the peak of the self‑convolution
by projected Adam.  The smoothing parameter λ starts small (soft max) and
is multiplied by a factor after a few thousand iterations – this
anneals the problem to the true max‑norm.  FFT is used for the
convolution and for the gradient (cross‑correlation).  Several lengths
and random restarts are tried; the best sequence found before the time
budget expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> list[float]:
    """
    Classical arcsine (1‑z)^‑1/2 coefficients, normalised to L1‑norm 1.
    """
    seq = []
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    seq = np.array(seq, dtype=np.float64) / total
    return seq.tolist()


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.  Implements the algorithm of
    Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection works on 1‑D arrays")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _metric_of(a: np.ndarray) -> float:
    """
    Original evaluation metric for a sequence with L1‑norm 1:
    2·n·max( a * a ).
    The convolution is performed with FFT for speed.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    A = np.fft.fft(a, L)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _adam_optimize(
    init_a: np.ndarray,
    deadline: float,
    lam_start: float = 5.0,
    lam_max: float = 200.0,
    lam_factor: float = 1.02,
    lr: float = 0.03,
    max_iter: int = 10_000_000,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the smoothed objective.
    Returns the best sequence (still on the simplex) and its exact metric.
    """
    a = init_a.copy()
    n = a.shape[0]

    # length needed for linear correlation (conv length 2n‑1, correlation 3n‑2)
    Lcorr = _next_pow2(3 * n - 2)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # initial metric
    best_a = a.copy()
    best_score = _metric_of(a)

    lam = lam_start
    it = 0
    t1 = time.time()
    while time.time() < deadline and it < max_iter:
        it += 1

        # ------------------------------------------------------------
        # 1) self‑convolution (FFT)
        A = np.fft.fft(a, Lcorr)
        conv = np.fft.ifft(A * A).real[: 2 * n - 1]
        max_conv = conv.max()

        # 2) soft‑max weights
        #   w_i ∝ exp(λ * conv_i).  Subtract max to avoid overflow.
        max_val = conv.max()
        w_raw = np.exp(lam * (conv - max_val))
        w = w_raw / w_raw.sum()

        # 3) gradient = 4 n · corr(w, a)
        W = np.fft.fft(w, Lcorr)
        # cross‑correlation (linear, not circular) via FFT:
        corr = np.fft.ifft(W * np.conj(A)).real[:n]
        grad = 4.0 * n * corr

        # ------------------------------------------------------------
        # Adam update on the (unconstrained) vector, then projection
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias correction
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = _project_to_simplex(a_new)

        # store metric of the *new* point (need a fresh convolution)
        A_new = np.fft.fft(a_new, Lcorr)
        conv_new = np.fft.ifft(A_new * A_new).real[: 2 * n - 1]
        cur_score = 2.0 * n * conv_new.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # ------------------------------------------------------------
        # λ‑annealing (slowly tighten the approximation of the max)
        if it % 2000 == 0:
            lam = min(lam * lam_factor, lam_max)

        # occasional progress report (does not affect timing)
        if it % 5000 == 0:
            print(
                f"[len {n}] iter {it:7d}  score {best_score:.6f}  λ {lam:.1f}"
            )

    return best_a, best_score


# ----------------------------------------------------------------------
def _local_refine(
    a: np.ndarray,
    deadline: float,
    jitter: float = 0.002,
    inner_iter: int = 5_000,
) -> tuple[np.ndarray, float]:
    """
    Tiny hill‑climbing pass: move a tiny amount of mass between two random
    coordinates and keep the change if it improves the exact metric.
    Used only when a small amount of time is left.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _metric_of(best_a)

    while time.time() < deadline:
        for _ in range(inner_iter):
            i, j = np.random.choice(n, 2, replace=False)
            delta = (np.random.rand() - 0.5) * 2.0 * jitter
            if delta > 0 and best_a[i] < delta:
                continue
            if delta < 0 and best_a[j] < -delta:
                continue

            cand = best_a.copy()
            cand[i] -= delta
            cand[j] += delta
            # keep non‑negative
            if cand[i] < 0 or cand[j] < 0:
                continue
            cand = _project_to_simplex(cand)

            cur = _metric_of(cand)
            if cur < best_score:
                best_score = cur
                best_a = cand
                break
        else:
            # no improvement in this batch – exit early
            break
    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a step‑function with the smallest possible evaluation
    score within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # tiny safety margin

    # lengths that comfortably fit into memory and give enough resolution
    lengths = [1500, 2000, 2500, 3000, 3400]

    # keep the overall best
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) try the known global sequence if it exists (often already very good)
    if "height_sequence_1" in globals():
        seq0 = np.array(globals()["height_sequence_1"], dtype=np.float64)
        seq0 = np.clip(seq0, 0.0, None)
        s = seq0.sum()
        if s > 0:
            seq0 = seq0 / s
            # give it a modest amount of time for refinement
            remaining = deadline - time.time()
            if remaining > 5.0:
                print("[global] refining previously known sequence")
                seq0, score0 = _adam_optimize(
                    seq0,
                    time.time() + min(20.0, remaining),
                )
                print(f"[global] refined metric {score0:.6f}")
                if score0 < best_score:
                    best_score = score0
                    best_seq = seq0

    # ------------------------------------------------------------------
    # 2) systematic optimisation over the chosen lengths
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # allocate a slice of the remaining budget (roughly equal per length)
        remaining = deadline - time.time()
        per_len = max(5.0, remaining / (len(lengths) - lengths.index(n)))

        # we will perform a few independent restarts for this length
        n_restarts = 3
        per_restart = per_len / n_restarts

        best_local_seq = None
        best_local_score = float("inf")

        for r in range(n_restarts):
            if time.time() >= deadline:
                break

            # start from arcsine plus a small random jitter
            base = np.array(arcsine_sequence(n), dtype=np.float64)
            jitter = np.random.randn(n) * 0.001
            start = base + jitter
            start = np.clip(start, 0.0, None)
            start /= start.sum()

            seq_opt, score_opt = _adam_optimize(
                start,
                time.time() + per_restart,
                lam_start=5.0,
                lam_max=250.0,
                lam_factor=1.02,
                lr=0.025,
            )
            print(f"[len {n} restart {r+1}] metric {score_opt:.6f}")

            if score_opt < best_local_score:
                best_local_score = score_opt
                best_local_seq = seq_opt

        # optional polishing with a tiny local search
        if best_local_seq is not None and time.time() < deadline:
            seq_refined, score_refined = _local_refine(
                best_local_seq,
                deadline,
                jitter=0.001,
                inner_iter=3_000,
            )
            if score_refined < best_local_score:
                best_local_score = score_refined
                best_local_seq = seq_refined

        print(f"Finished length {n}: best metric = {best_local_score:.6f}")

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_seq
            print(f"*** New overall best (len={n}) score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) final safety – make sure the numbers respect the limits of
    #    evaluate_sequence() and re‑normalise to L1‑norm 1.
    if best_seq is None:
        # Should never happen: fall back to a safe default.
        best_seq = np.array(arcsine_sequence(2000), dtype=np.float64)
        best_score = _metric_of(best_seq)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(
        f"\n--- FINAL RESULT ---\n"
        f"Best length = {len(best_seq)}  metric = {best_score:.6f}"
    )
    return best_seq.tolist()
```
