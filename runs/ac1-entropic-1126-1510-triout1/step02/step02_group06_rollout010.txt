def project_simplex(v):
    """Project vector v onto probability simplex (non-negative sum to 1)."""
    # Standard algorithm
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _optimize_softmax(init_seq:list,float, deadline, max_iter=200000):
    a = np.array(init_seq, dtype=float)
    a = a / a.sum()
    n = a.shape[0]
    L = 1 << ((3*n - 1).bit_length())
    best_a = a.copy()
    # compute initial score
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    best_score = 2 * n * conv.max()
    # adam variables
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    t=0
    alpha = 5.0  # start smoothness
    alpha_max=200.0

    no_improve = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        # compute convolution in frequency domain
        fa = np.fft.rfft(a, n=L)
        # conv = a*a
        conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
        # compute softmax weights
        maxc = conv.max()
        # may subtract maxc to avoid overflow
        exp_vals = np.exp(alpha * (conv - maxc))
        # avoid overflow/underflow extremes
        sum_exp = exp_vals.sum()
        if sum_exp == 0 or np.isnan(sum_exp):
            # fallback to uniform weights
            w = np.full_like(conv, 1.0/conv.size)
        else:
            w = exp_vals / sum_exp

        # compute gradient via correlation
        fw = np.fft.rfft(w, n=L)
        corr_full = np.fft.irfft(np.conj(fa) * fw, n=L)
        offset = len(w) - 1  # 2n-2
        grad = 2.0 * corr_full[offset:offset + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step_size = 0.1 / (np.sqrt(t) + 1)  # decaying step
        a_new = a - step_size * m_hat / (np.sqrt(v_hat) + eps)

        # project onto simplex
        a_new = project_simplex(a_new)

        # compute new metric
        fa_new = np.fft.rfft(a_new, n=L)
        conv_new = np.fft.irfft(fa_new * fa_new, n=L)[:2*n-1]
        new_score = 2 * n * conv_new.max()

        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            no_improve = 0
            # maybe reduce step size less aggressively
        else:
            no_improve += 1

        # acceptance step: always take a_new (since we project) but maybe keep old if too far?
        a = a_new

        # adjust alpha gradually (increase to focus near max)
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * 1.01)

        # if stagnation for many iterations, break or jitter
        if no_improve >= 5000:
            # random jitter
            jitter = np.random.normal(scale=0.001, size=n)
            a = project_simplex(best_a + jitter)
            no_improve = 0
            # maybe reset Adam moments
            m.fill(0)
            v.fill(0)
        # maybe early break if time left insufficient
    return best_a.tolist(), best_score

def softmax_optimize(init_seq, deadline):
    a = np.array(init_seq, dtype=np.float64)
    a = a / a.sum()
    n = a.shape[0]
    L = 1 << ((3*n - 1).bit_length())
    best_a = a.copy()
    # compute initial score
    best_score = compute_metric(a)
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    # smoothing parameter
    alpha = 5.0
    alpha_max = 200.0
    # counters
    no_improve = 0
    while time.time() < deadline:
        t += 1
        # Compute FFT of a (fa) only once
        fa = np.fft.rfft(a, n=L)
        # conv
        conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
        maxc = conv.max()
        # compute weights for gradient using smooth max
        # subtract maxc to stabilize
        exp_vals = np.exp(alpha * (conv - maxc))
        sum_exp = exp_vals.sum()
        if sum_exp == 0 or np.isinf(sum_exp):
            # fallback uniform
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = exp_vals / sum_exp

        # Gradient via correlation
        fw = np.fft.rfft(w, n=L)
        corr_full = np.fft.irfft(np.conj(fa) * fw, n=L)
        offset = len(w) - 1  # = 2n-2
        grad = 2.0 * corr_full[offset:offset + n]  # shape n

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = 0.5 / (np.sqrt(t) + 1)  # decaying step
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a_new = project_simplex(a_new)

        # evaluate new metric
        fa_new = np.fft.rfft(a_new, n=L)
        conv_new = np.fft.irfft(fa_new * fa_new, n=L)[:2*n-1]
        new_score = 2.0 * n * conv_new.max()

        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # increase smoothing gradually

            m.fill(0)
            v.fill(0)
            no_improve = 0

        # break if step becomes too small
        if step < 1e-8:
            break

    return best_a.tolist(), best_score

"""
High‑quality step‑function search (soft‑max gradient descent).

The evaluation metric is
    2·n·max(conv(a,a)) / (sum(a)²)

When the coefficients are normalised to sum = 1 the metric reduces to
    2·n·max(conv(a,a)) .
The algorithm works directly with this simplified objective.

We use a *soft‑max* approximation of the maximum:
    w_k ∝ exp(α·conv_k) .
For a given α, the gradient of the smooth objective is
    ∇ = 2·corr(a, w) ,
where corr denotes the (non‑reversed) cross‑correlation.
All convolutions / correlations are performed with FFTs, giving an
O(n log n) per‑iteration cost.

An Adam optimiser (first‑ and second‑moment adaptation) updates the
vector; after each update the iterate is projected onto the probability
simplex (non‑negative entries summing to 1).  The smoothing parameter
α starts small (≈5) and is increased slowly so the method first makes
large, smooth moves and later focuses on the true peak.

Multiple initialisations (arcsine, half‑sine, exponential, random Dirichlet)
are tried for several lengths; the best sequence found before the total
budget (default 1000 s) expires is returned.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive → uniform distribution
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # re‑normalise to avoid tiny numerical drift
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        w = w / w_sum
    return w


# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> list[float]:
    """First n coefficients of (1‑z)^‑½, normalised to L1‑norm = 1."""
    seq = []
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    seq = [x / total for x in seq]
    return seq


# ----------------------------------------------------------------------
def half_sine_sequence(n: int) -> list[float]:
    """a_i ∝ sin(π·(i+½)/n), i.e. a half‑sine wave on [0,n)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    seq = np.sin(np.pi * i / n)
    seq = np.maximum(seq, 0.0)
    seq = seq / seq.sum()
    return seq.tolist()


# ----------------------------------------------------------------------
def exp_sequence(n: int, lam: float = 0.001) -> list[float]:
    """Exponential decay a_i ∝ exp(‑lam·i), normalised."""
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * i)
    seq = seq / seq.sum()
    return seq.tolist()


# ----------------------------------------------------------------------
def _metric_fast(a: np.ndarray) -> float:
    """Fast evaluation of 2·n·max(conv(a,a)) using FFT (a is already normalised)."""
    n = a.shape[0]
    L = 1 << ((3 * n - 1).bit_length())        # next power of two ≥ 3n‑2
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _softmax_optimize(
    init_seq: list[float],
    deadline: float,
    max_iter: int = 200_000,
) -> tuple[list[float], float]:
    """
    Soft‑max gradient descent with Adam on the objective
    ``2·n·max(conv(a,a))``.
    Returns the best sequence (as a list) and its metric.
    """
    a = np.array(init_seq, dtype=np.float64)
    a = a / a.sum()
    n = a.shape[0]
    L = 1 << ((3 * n - 1).bit_length())    # FFT length (power of two)

    best_a = a.copy()
    best_score = _metric_fast(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    # soft‑max temperature (larger → sharper)
    alpha = 5.0
    alpha_max = 200.0

    # stagnation detector
    no_improve = 0

    while time.time() < deadline and t < max_iter:
        t += 1

        # ----- convolution and soft‑max weights -----
        fa = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        maxc = conv.max()

        # soft‑max over the convolution entries
        exp_vals = np.exp(alpha * (conv - maxc))      # shift to avoid overflow
        sum_exp = exp_vals.sum()
        if sum_exp == 0.0 or not np.isfinite(sum_exp):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = exp_vals / sum_exp

        # ----- gradient (2·correlation(a, w)) -----
        fw = np.fft.rfft(w, n=L)
        corr_full = np.fft.irfft(np.conj(fa) * fw, n=L)
        offset = len(w) - 1                     # = 2n‑2
        grad = 2.0 * corr_full[offset : offset + n]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.05                               # fixed base learning rate
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex
        a_new = project_simplex(a_new)

        # compute new metric
        fa_new = np.fft.rfft(a_new, n=L)
        conv_new = np.fft.irfft(fa_new * fa_new, n=L)[: 2 * n - 1]
        new_score = 2.0 * n * conv_new.max()

        # keep the best ever seen
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # slowly increase the soft‑max sharpness
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * 1.02)

        # occasional jitter to escape plateaux
        if no_improve >= 3000:
            jitter = np.random.normal(scale=0.001, size=n)
            a = project_simplex(best_a + jitter)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        # safety break
        if lr * np.max(np.abs(grad)) < 1e-12:
            break

    return best_a.tolist(), best_score

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # tiny safety margin

    # Lengths to explore – they fit easily into memory and give a decent
    # trade‑off between expressiveness and speed.
    lengths = [1800, 2600, 3400, 5000]
    best_seq = None
    best_score = float("inf")

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # share the remaining budget evenly among the still‑to‑process lengths
        remaining = deadline - time.time()
        per_len_budget = max(5.0, remaining / (len(lengths) - idx))

        # a few diverse starting points
        starts = [
            arcsine_sequence(n),
            half_sine_sequence(n),

    if best_seq is None:                         # very unlikely

        best_score = _metric_fast(np.array(best_seq))

    # final safety: enforce the bounds required by evaluate_sequence
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    print(

        f"with metric = {best_score:.6f}"
    )
    return best_seq
