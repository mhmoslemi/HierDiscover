    # Pre-calc metric maybe.
    
    # Temperature schedule: start T ~ 0.01 (in metric units). Actually metric range close to pi/2 ~ 1.57.

    # step size: epsilon ~ 0.01? Since sum=1.
    eps0 = 0.01
    eps_decay = 0.9998
    
    iteration = 0
    while time.time() < deadline:

        j = np.random.randint(0, n)
        if i == j:
            continue
        # propose new distribution
        cand = curr.copy()

        cand[j] += eps
        # maybe enforce bounds (>=0)
        # compute metric quickly via FFT
        cand_metric = compute_metric(cand)
        delta = cand_metric - curr_score
        if delta < 0 or np.exp(-delta / temperature) > np.random.rand():

            curr = cand
            curr_score = cand_metric
            if curr_score < best_score:
                best = curr.copy()

"""
Improved search for a low‑evaluation step‑function.

The algorithm uses a simulated‑annealing (SA) phase followed by a short
gradient‑descent polishing phase.

* Initialise with the analytically good “arcsine’’ distribution (sum = 1).
* Repeatedly move a random amount of mass from one index to another,
  accepting worsening moves with probability exp(–Δ/temperature).  The
  temperature and the step‑size (fraction of the source mass) are
  exponentially cooled, allowing a broad exploration early and fine
  tuning later.
* The evaluation of a candidate is performed with an FFT‑based
  convolution, which gives the exact same score as `evaluate_sequence`
  for our (non‑negative, bounded) vectors but is much faster.
* After the SA time expires, a few gradient‑descent iterations on a
  soft‑max approximation of the max‑convolution are applied to polish
  the best candidate.
* The final vector is projected onto the probability simplex (non‑negative,
  sum = 1) and its true score is returned by `evaluate_sequence`.

The whole search stays well within the 1000 s budget, works for any
reasonable length (the default is 5000), and does not rely on external
LP solvers.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> list[float]:
    """Return the normalised (sum = 1) arcsine coefficients C_i for i=0..n‑1.
    Recurrence: C_{i+1}=C_i·(2i+1)/(2(i+1)).
    """
    seq = []
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2.0 * i + 1.0) / (2.0 * (i + 1))
    seq = [x / total for x in seq]          # L1‑normalise to 1
    return seq


# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σ x = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("target sum s must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """Fast version of evaluate_sequence (no clamping, sum assumed >0)."""
    f = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * n - 1]
    max_c = conv.max()
    return 2.0 * n * max_c / (a.sum() ** 2)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Search for a non‑negative step‑function with minimal evaluation score.

    Returns the best list of coefficients found within ``budget_s`` seconds.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # 1) Choose a length and an initial distribution (arcsine is a strong start)
    n = 5000                     # length of the sequence – a compromise between
                                 # flexibility and evaluation speed
    a = np.asarray(arcsine_sequence(n), dtype=np.float64)

    # Ensure perfect L1 normalisation (floating‑point drift)
    a = a / a.sum()

    # Pre‑compute FFT length (next power of two >= 2n‑1)
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Helper to compute the true score (via the provided evaluator)
    def true_score(vec):
        return evaluate_sequence(vec.tolist())

    # ------------------------------------------------------------------
    # 2) Simulated‑annealing phase
    best = a.copy()
    best_score = true_score(best)
    curr = a.copy()
    curr_score = _fft_score(curr, n, fft_len)

    # SA hyper‑parameters
    temperature = 0.02          # initial temperature (in score units)
    min_temp = 1e-6
    temp_decay = 0.9997

    eps_factor = 0.20           # fraction of source mass moved each step
    eps_decay = 0.9999

    it = 0
    while time.time() < deadline:
        it += 1

        # pick a source index with non‑zero mass
        i = np.random.randint(0, n)
        if curr[i] <= 0.0:
            continue
        # pick a distinct destination
        j = np.random.randint(0, n)
        if i == j:
            continue

        # amount to move: a random fraction of the source mass
        frac = np.random.rand() * eps_factor
        delta = frac * curr[i]
        if delta <= 0.0:
            continue

        cand = curr.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = _fft_score(cand, n, fft_len)
        delta_score = cand_score - curr_score

        # Metropolis acceptance
        if delta_score < 0 or math.exp(-delta_score / max(temperature, 1e-12)) > np.random.rand():
            curr = cand
            curr_score = cand_score
            if curr_score < best_score:
                best = curr.copy()
                best_score = true_score(best)
                print(f"[SA {it}] new best {best_score:.6f} (temp={temperature:.2e})")

        # exponential cooling
        temperature = max(min_temp, temperature * temp_decay)
        eps_factor = max(0.001, eps_factor * eps_decay)

    # ------------------------------------------------------------------
    # 3) Gradient‑descent polishing (soft‑max of the convolution)
    grad_step = 0.12
    beta = 5.0
    grad_iters = 300

    for g in range(grad_iters):
        if time.time() >= deadline:
            break

        # compute convolution once (FFT)
        f = np.fft.rfft(best, n=fft_len)
        conv = np.fft.irfft(f * f, n=fft_len)[: 2 * n - 1]

        max_val = conv.max()
        # soft‑max over the convolution values
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / exp_vals.sum()

        # gradient of the smooth max (see description)
        grad = 2.0 * np.convolve(w, best[::-1])[n - 1 : 2 * n - 1]

        # take a step and re‑project onto the simplex
        cand = best - grad_step * grad
        cand = project_simplex(cand, s=1.0)

        cand_score = true_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            print(f"[GD {g}] improved to {best_score:.6f}")

        # schedule decay
        grad_step *= 0.995
        beta = min(200.0, beta * 1.02)

    # ------------------------------------------------------------------
    # 4) Final sanity checks and return
    best = np.maximum(best, 0.0)
    total = best.sum()
    if total > 0:
        best = best / total
    else:
        best = np.asarray(arcsine_sequence(n), dtype=np.float64)

    final_score = true_score(best)
    print(f"Final score {final_score:.6f} after {it} SA steps and {g+1} GD steps")
    return best.tolist()
