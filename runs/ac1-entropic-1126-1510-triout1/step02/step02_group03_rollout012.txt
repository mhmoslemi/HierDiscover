The score returned by `evaluate_sequence` for a non‑negative vector `a` with  
`Σa = 1` is  

```
score(a) = 2·n·max_k (a * a)[k]                (1)
```

where `n = len(a)` and `*` is the linear convolution.  
Consequently the optimisation problem is

* minimise the maximal entry of the self‑convolution,
* under the simplex constraint `a ≥ 0 , Σa = 1`.

The problem is non‑convex, but a smooth approximation of the `max`
function works very well in practice:

```
F_λ(a) = (1/λ) · log Σ_k exp( λ·(a*a)[k] )      (2)
```

`F_λ → max` when `λ → ∞`.  
The gradient of (2) can be written as a *correlation* (a single FFT)

```
∇F_λ(a)_i = 2 · Σ_k w_k · a_{k‑i}               (3)
w_k = softmax_k ( λ·(a*a)[k] )
```

Using (3) we can run a fast mirror‑descent (exponentiated‑gradient)
procedure on the simplex.  
Two practical improvements over the reference implementation are

1. **Much larger λ** – we raise λ up to `200`.  This forces the surrogate
   to concentrate on the true peak and gives a far tighter approximation.
2. **Adam‑style step‑size control** – the learning‑rate is adapted
   automatically, which makes the optimisation far more robust.
3. **Better initialisation** – a wide sweep over the symmetric Beta
   family (`α ∈ [10⁻³, 5]`), a few random Dirichlet draws and an
   up‑sampled solution from a smaller dimension are tried.
4. **More iterations per length** – the time budget is split between a
   few problem sizes, but each size receives a generous number of
   iterations (up to several ten‑thousands).  The whole search finishes
   well inside the 1000 s limit.

All heavy work (convolution, correlation) is performed with NumPy FFTs,

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the grid x_i = (i+½)/n, normalised."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, m)
    c = np.fft.irfft(A * A, m)
    return c[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k‑i} via FFT.
    w – weight vector (length 2·n‑1)
    a – current sequence (length n)
    fft_len – power‑of‑two size (≥ len(w)+len(a)-1)
    Returns the slice of length n.
    """
    W = np.fft.rfft(w, fft_len)
    A = np.fft.rfft(a, fft_len)
    corr = np.fft.irfft(np.conj(W) * A, fft_len)
    start = len(a) - 1
    return 2.0 * corr[start:start + len(a)]


# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False) + 0.5 / old_n
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False) + 0.5 / new_n
    new_vals = np.interp(new_x, old_x, seq)
    # re‑normalise to be a probability vector
    new_vals /= new_vals.sum()
    return new_vals

def _optimise_adam(
    n: int,
    deadline: float,
    init_seq: np.ndarray,
    max_iters: int = 20000,
    eta0: float = 0.2,
    lambda0: float = 0.5,
    lambda_max: float = 200.0,
    beta1: float = 0.9,
    beta2: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Refine `init_seq` (already normalised to sum=1) with an Adam‑style
    exponentiated‑gradient mirror descent on the smooth max surrogate.
    Returns the best sequence seen and the exact (unsmoothed) score.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()

    best_seq = a.copy()
    best_score = float(2.0 * n * _conv_full_fft(a).max())

    # FFT length that can accommodate both a (len=n) and w (len=2n-1)
    fft_len = 1 << ((3 * n - 2).bit_length())   # >= (2n-1)+(n-1)

    # Adam moments on the *log* of a (so positivity is automatic)
    m1 = np.zeros_like(a)
    m2 = np.zeros_like(a)
    eps = 1e-12
    lam = lambda0
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # current exact score
        b = _conv_full_fft(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} iter={it:6d} score={best_score:.6f}")

        # smooth max – soft‑max weights
        maxb = b.max()
        # (b-maxb) ≤ 0, safe for exp
        w = np.exp(lam * (b - maxb))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.size)
        else:
            w /= w_sum

        # gradient (correlation) of the smooth max
        g = _gradient_via_correlation(w, a, fft_len)

        # Adam on log‑a
        m1 = beta1 * m1 + (1.0 - beta1) * g
        m2 = beta2 * m2 + (1.0 - beta2) * (g * g)
        m1_hat = m1 / (1.0 - beta1 ** it)
        m2_hat = m2 / (1.0 - beta2 ** it)

        # learning rate schedule – slowly decaying
        eta = eta0 / np.sqrt(it)

        # mirror‑descent step on log‑a
        log_a = np.log(a + eps) - eta * (m1_hat / (np.sqrt(m2_hat) + eps))
        a = np.exp(log_a)

        # project back onto simplex (ensures Σa = 1, a ≥ 0)
        a = _project_onto_simplex(a, s=1.0)

        # occasional small random jitter to escape plateaus
        if it % 1000 == 0 and np.random.rand() < 0.15:
            a += np.random.randn(n) * 5e-5
            a = np.clip(a, 0.0, None)
            a = _project_onto_simplex(a, s=1.0)

        # increase λ – exponential schedule, capped at lambda_max
        lam = min(lambda_max, lam * 1.03)

    return best_seq, best_score

        if overall_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 4️⃣  Fallback – guarantee a return value
    if overall_best_seq is None:
        n_fallback = 2000
        overall_best_seq = np.full(n_fallback, 1.0 / n_fallback)
        overall_best_score = float(2.0 * n_fallback *
                                   _conv_full_fft(overall_best_seq).max())

    print(f"Finished – best approximate score: {overall_best_score:.6f} "
          f"length={len(overall_best_seq)}")
    # convert to plain Python list for the external runner
    return overall_best_seq.tolist()
