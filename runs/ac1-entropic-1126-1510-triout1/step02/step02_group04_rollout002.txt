"""
Gradient‑based optimiser for the step‑function bound.

The algorithm works on the probability simplex (non‑negative, sum = 1)
and repeatedly:
  * evaluates the self‑convolution,
  * builds a soft‑max weighting of the convolution entries (the weight
    concentrates on the largest peaks as a smoothing of the max‑operator),
  * computes the corresponding gradient (a correlation of the weights
    with the current sequence),
  * takes a projected gradient step,
  * occasionally performs a small targeted move that shifts mass away from
    the pair of indices responsible for the current maximal convolution
    entry, and
  * slowly anneals the soft‑max temperature (β) to focus on the true max.

The best sequence seen during the run is returned.  The code respects the
time budget (≈ 1000 s) and only uses NumPy and the provided
`evaluate_sequence` function.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Clip negatives to zero and renormalise to sum = 1 (simple simplex)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fallback to uniform point
        v = np.full_like(v, 1.0 / v.size)
    else:
        v = v / s
    return v


# ----------------------------------------------------------------------
# Fast linear convolution via FFT (full mode)
# ----------------------------------------------------------------------
def _linear_convolution(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of two real vectors using FFT."""
    n = x.size + y.size - 1
    # next power of two for efficient FFT
    N = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    conv = np.fft.irfft(X * Y, N)
    return conv[:n]


# ----------------------------------------------------------------------
# Convolution of a sequence with itself (used many times)
# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full length 2·len(a)−1)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Score identical to evaluate_sequence for sum‑=‑1 vectors:
#   2·n·max(conv)  (since denominator = (sum a)^2 = 1)
# ----------------------------------------------------------------------
def _score_simplex(a: np.ndarray) -> float:
    n = a.size
    conv = _self_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Gradient of the soft‑max weighted sum of convolution entries.
# For a weight vector w (same length as conv) the gradient is
#   2 * correlation(w, a)    (the correlation is a convolution of w with a[::-1])
# ----------------------------------------------------------------------
def _gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    n = a.size
    rev = a[::-1]
    corr = _linear_convolution(w, rev)               # length = len(w)+n-1
    # the needed slice corresponds to lags 0 … n−1
    grad = 2.0 * corr[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Targeted move – shift a tiny fraction of mass away from the pair that
# contributes most to the currently maximal convolution entry.
# ----------------------------------------------------------------------
def _targeted_move(a: np.ndarray, frac: float = 0.02) -> np.ndarray:
    n = a.size
    conv = _self_convolution(a)
    k = int(conv.argmax())                     # lag with maximal entry

    # contributions a[i] * a[k‑i] for all feasible i
    i_arr = np.arange(n, dtype=int)
    j_arr = k - i_arr
    mask = (j_arr >= 0) & (j_arr < n)

    if not np.any(mask):
        # fallback: a small random perturbation
        cand = a + frac * (np.random.rand(n) - 0.5)
        return _project_to_simplex(cand)

    contrib = np.zeros_like(a)
    contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

    i_max = int(np.argmax(contrib))
    j_max = k - i_max

    # move a fraction of the larger of the two entries to a random location
    src = i_max if a[i_max] >= a[j_max] else j_max
    dst = random.randrange(n)
    while dst == src:
        dst = random.randrange(n)

    amount = frac * a[src] * random.random()
    cand = a.copy()
    cand[src] -= amount
    cand[dst] += amount
    return _project_to_simplex(cand)

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Gradient / simulated‑annealing optimiser for the step‑function bound.
    Returns a list of non‑negative heights that minimise `evaluate_sequence`.
    """
    # ------------------------------------------------------------------
    # Initialise random generators
    # ------------------------------------------------------------------
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + max(0.0, budget_s - 0.5)

    # ------------------------------------------------------------------
    # Load a sensible starting point (previous best if available)
    # ------------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        n = base.size
    except Exception:
        n = 8000
        base = np.full(n, 1.0 / n, dtype=float)

    # Normalise onto the simplex
    base = _project_to_simplex(base)

    # ------------------------------------------------------------------
    # Optimisation state
    # ------------------------------------------------------------------
    seq = base.copy()                 # current candidate
    best_seq = seq.copy()
    best_score = _score_simplex(best_seq)

    print(f"[Init] n = {n}, initial score = {best_score:.6f}")

    # Hyper‑parameters
    beta = 1.0                         # soft‑max temperature
    beta_max = 30.0
    beta_factor = 1.02                # slowly anneal towards a sharper max
    base_lr = 0.6                     # initial step‑size
    lr_decay = 5e-4
    iteration = 0
    last_improve = 0
    temp_start = 0.02                 # annealing temperature for Metropolis moves
    temp_end = 0.0001
    max_no_improve = 4000             # stagnation threshold for a random restart

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ----- evaluate current candidate -----
        conv = _self_convolution(seq)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # ----- keep the best seen so far -----
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = seq.copy()
            last_improve = iteration
            print(f"[Iter {iteration:5d}] New best score = {best_score:.6f}")

        # ----- compute soft‑max weights for the convolution entries -----
        # Normalise the convolution to its maximum before exponentiation
        # to keep the exponent well‑scaled.
        w = np.exp(beta * (conv / (cur_max + 1e-16) - 1.0))
        w_sum = w.sum()
        if w_sum == 0.0:                     # should not happen, but be safe
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient step -----
        grad = _gradient(seq, w)
        eta = base_lr / (1.0 + lr_decay * iteration)   # diminishing step
        cand = seq - eta * grad
        cand = _project_to_simplex(cand)

        # ----- accept / reject (simple Metropolis on the true max‑conv) -----
        cand_conv = _self_convolution(cand)
        cand_score = 2.0 * n * cand_conv.max()

        if cand_score < cur_score:
            seq = cand
        else:
            # probabilistic acceptance to escape shallow local minima
            elapsed = time.time() - start_time
            t_frac = elapsed / budget_s
            temperature = temp_start * (1.0 - t_frac) + temp_end * t_frac
            accept_prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            if np.random.rand() < accept_prob:
                seq = cand

        # ----- occasional targeted move (every ~10 iterations) -----
        if iteration % 10 == 0:
            seq_tm = _targeted_move(seq, frac=0.02)
            tm_conv = _self_convolution(seq_tm)
            tm_score = 2.0 * n * tm_conv.max()
            if tm_score < cur_score:
                seq = seq_tm
                # we deliberately do not update “cur_score” here – it will be
                # recomputed at the start of the next loop.

        # ----- anneal the soft‑max temperature -----
        if beta < beta_max:
            beta = min(beta_max, beta * beta_factor)

        # ----- stagnation handling -----
        if iteration - last_improve > max_no_improve:
            # inject fresh randomness while keeping a memory of the best solution
            mixed = seq * 0.7 + 0.3 * np.random.rand(n)
            seq = _project_to_simplex(mixed)
            beta = 1.0
            last_improve = iteration
            print(f"[Iter {iteration:5d}] Restart due to stagnation")

        # ----- early exit if we already beat the target -----
        if best_score < 1.503:
            print(f"[Iter {iteration:5d}] Target achieved (score={best_score:.6f})")
            break

    # ------------------------------------------------------------------
    # Final reporting
    # ------------------------------------------------------------------
    print(f"[Done] Best score after {iteration} iterations = {best_score:.6f}")

    # A final safety check using the official evaluator (optional)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[Check] evaluate_sequence reports {final_val:.6f}")
    except Exception:
        pass

    return best_seq.tolist()
