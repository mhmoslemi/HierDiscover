       else:

       if accept:

           a[source] -= epsilon
           a[dest] += epsilon
           # avoid negative due to numerical errors

               best_metric = true_score  # we store true maybe

           # update step and temp

"""
Local‑search (guided by the current maximal convolution point) for a
non‑negative step function that minimises

    2·n·max(conv(a,a)) / (sum(a))²

The search starts from an arcsine distribution (the former state‑of‑the‑art)
or from the globally stored sequence `height_sequence_1`.  At each iteration
the index `k_max` where the self–convolution attains its maximum is located.
A gradient‑like vector `g_i = a_{k_max-i}` tells how the peak changes when a
mass is moved from `i` to another index `j`.  The algorithm repeatedly moves
a small amount of probability mass from a source with large `g_i` to a sink
with small `g_j`.  Moves are accepted greedily, with a tiny simulated‑annealing
probability to escape plateaus.  The convolution is updated in O(n) by a rank‑1
update (no full re‑convolution each step).  Every few thousand steps the full
convolution is recomputed to remove numerical drift.  The best sequence found
is returned; the final result is normalised to have unit L₁‑norm (the metric
is scale‑invariant).

The method runs within the supplied time budget (≈1000 s) and uses only
NumPy and the provided `evaluate_sequence` function.
"""

import time
import math
import numpy as np


def arcsine_sequence(n: int) -> list[float]:
    """
    Normalised (sum = 1) discrete arcsine coefficients:
        C_0 = 1,
        C_{i+1} = C_i * (2*i+1) / (2*(i+1)).
    """
    seq = []
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2.0 * i + 1.0) / (2.0 * (i + 1))
    seq = np.array(seq, dtype=float)
    seq /= seq.sum()
    return seq.tolist()


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns the best list of coefficients found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # spare a bit for finalisation

    # ------------------------------------------------------------------
    # 1) initialise a – try global sequence, otherwise arcsine
    # ------------------------------------------------------------------
    try:
        global height_sequence_1
        seq0 = np.asarray(height_sequence_1, dtype=float)
        if seq0.size == 0:
            raise ValueError
        seq0 = np.maximum(seq0, 0.0)
        total0 = seq0.sum()
        if total0 <= 1e-12:
            raise ValueError
        seq0 /= total0
        n = seq0.shape[0]
        # limit size to keep updates cheap
        if n > 4000:
            n = 4000
            seq0 = seq0[:n]
            seq0 /= seq0.sum()
        a = seq0.copy()
        print(f"[init] using global sequence, n={n}")
    except Exception:
        # fallback to arcsine of moderate length
        n = 3200
        a = np.array(arcsine_sequence(n), dtype=float)
        print(f"[init] using arcsine sequence, n={n}")

    # ------------------------------------------------------------------
    # Helper to compute the metric (exact, since sum(a)=1)
    # ------------------------------------------------------------------
    def metric_from_conv(conv_arr: np.ndarray) -> float:
        """2 * n * max(conv)  (valid when sum(a)=1)."""
        return 2.0 * n * float(conv_arr.max())

    # initial full convolution and metric
    conv = np.convolve(a, a)
    cur_metric = metric_from_conv(conv)
    best_seq = a.copy()
    best_metric = cur_metric
    true_best = evaluate_sequence(best_seq.tolist())
    print(f"[start] metric≈{cur_metric:.6f}, true score={true_best:.6f}")

    # ------------------------------------------------------------------
    # 2) parameters for the guided local search
    # ------------------------------------------------------------------
    step_factor = 0.20          # fraction of source weight moved each step
    step_decay = 0.9997         # slowly shrink step size
    min_step = 1e-7

    temperature = 1e-4          # annealing temperature for occasional uphill moves
    temp_decay = 0.9999

    recompute_interval = 5000   # full recompute to avoid drift
    last_improve = 0            # iteration index of last improvement

    iteration = 0
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # locate the current peak of the convolution
        # --------------------------------------------------------------
        k_max = int(conv.argmax())

        # gradient‑like vector g_i = a_{k_max-i}  (zero outside the support)
        indices = np.arange(n)
        idx = k_max - indices
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros(n, dtype=float)
        grad[mask] = a[idx[mask]]

        # --------------------------------------------------------------
        # choose a source (large gradient) and a destination (small gradient)
        # --------------------------------------------------------------
        positive_mask = a > 0
        if not np.any(positive_mask):
            break  # nothing to move

        # source: among positions with mass, pick one with maximal gradient
        source_candidates = np.where(positive_mask)[0]
        src_grad_vals = grad[source_candidates]
        src = source_candidates[int(np.argmax(src_grad_vals))]

        # destination: index with minimal gradient (can be zero‑mass)
        dst = int(np.argmin(grad))
        if dst == src:
            # pick the second smallest if it coincides
            sorted_idx = np.argsort(grad)
            for cand in sorted_idx:
                if cand != src:
                    dst = int(cand)
                    break

        # --------------------------------------------------------------
        # magnitude of the transfer
        # --------------------------------------------------------------
        max_eps = step_factor * a[src]
        if max_eps <= min_step:
            # dwindling steps – stop early
            break
        epsilon = np.random.random() * max_eps
        if epsilon <= 0:
            continue

        # --------------------------------------------------------------
        # fast rank‑1 update of the convolution (O(n))
        # --------------------------------------------------------------
        new_conv = conv.copy()
        # decrease contribution from the source
        new_conv[src:src + n] -= 2.0 * epsilon * a
        # increase contribution from the destination
        new_conv[dst:dst + n] += 2.0 * epsilon * a

        eps_sq = epsilon * epsilon
        new_conv[2 * src] += eps_sq
        new_conv[2 * dst] += eps_sq
        new_conv[src + dst] -= 2.0 * eps_sq

        new_metric = metric_from_conv(new_conv)

        # --------------------------------------------------------------
        # acceptance test (mostly greedy, with occasional uphill moves)
        # --------------------------------------------------------------
        accept = False
        if new_metric < cur_metric:
            accept = True
        else:
            # simulated‑annealing style uphill acceptance
            if temperature > 0:
                prob = math.exp(-(new_metric - cur_metric) / temperature)
                if np.random.random() < prob:
                    accept = True

        if accept:
            # update the state
            a[src] -= epsilon
            a[dst] += epsilon

            # numerical clean‑up (avoid tiny negatives)
            if a[src] < 0:
                a[src] = 0.0
            conv = new_conv
            cur_metric = new_metric

            # keep the globally best sequence (true evaluation)
            if cur_metric < best_metric:
                # Re‑evaluate with the official function for safety
                true_score = evaluate_sequence(a.tolist())
                if true_score < best_metric:
                    best_metric = true_score
                    best_seq = a.copy()
                    last_improve = iteration
                    print(f"[iter {iteration}] new best ≈{cur_metric:.6f} (true {true_score:.6f})")
                else:
                    # keep the previous best if the official score did not improve
                    best_metric = true_score

        # --------------------------------------------------------------
        # decay step size & temperature
        # --------------------------------------------------------------
        step_factor = max(min_step, step_factor * step_decay)
        temperature = max(0.0, temperature * temp_decay)

        # --------------------------------------------------------------
        # periodic full recompute to erase accumulated numerical error
        # --------------------------------------------------------------
        if iteration % recompute_interval == 0:
            conv = np.convolve(a, a)
            cur_metric = metric_from_conv(conv)
            # optional tiny global perturbation if stuck for long
            if iteration - last_improve > recompute_interval * 5:
                # add a little random noise, re‑normalise
                noise = np.random.randn(n) * 0.001
                a = a + noise
                a = np.maximum(a, 0.0)
                if a.sum() > 0:
                    a /= a.sum()
                conv = np.convolve(a, a)
                cur_metric = metric_from_conv(conv)
                print(f"[shake] iteration {iteration}, jitter applied")

    # ------------------------------------------------------------------
    # final sanitisation
    # ------------------------------------------------------------------
    final_seq = np.maximum(best_seq, 0.0)
    total = final_seq.sum()
    if total > 0:
        final_seq /= total
    else:
        final_seq = np.array(arcsine_sequence(n), dtype=float)

    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Finished – best true score = {final_score:.6f}  length = {len(final_seq)}")
    return final_seq.tolist()
