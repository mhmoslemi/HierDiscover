def proj_simplex(v):
    # Projects onto {x >= 0, sum x = 1}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _project_onto_simplex(v):
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function (non‑negative heights) that minimises
``evaluate_sequence``.

The algorithm mixes three ideas:

1. **Random restarts** – many short runs start from a random Dirichlet
   distribution of a chosen length (2 k … 8 k).  This explores many
   different mass patterns.

2. **LP refinement** – if CVXPY is available we solve the linear program
   used in the reference implementation:

        maximise   Σ g_j
        subject to (f * g)_k ≤ M   (M = max(f*f))
                     g_j ≥ 0

   The solution pushes extra total mass into ``g`` while keeping the
   autocorrelation bound M.  After scaling ``g`` back to a probability
   vector we take a convex combination with the current ``f``.  Repeating
   this step (up to 60 iterations per restart) steadily reduces the
   evaluation score.

3. **Local polishing** – a cheap “random‑pair” move (preserving the
   L1‑norm) is applied many times (≈ 5000 steps) with a cooling schedule.
   Any improvement is kept, guaranteeing that the best sequence seen
   across all restarts is returned.

If CVXPY is not present the method falls back to a simple projected
sub‑gradient descent that directly reduces the maximal convolution
value.  Both branches keep the vector normalised to unit L1‑norm; the
scale invariance of ``evaluate_sequence`` then guarantees that the
objective value equals ``2·n·max(conv)``.

The whole search respects the 1000 s budget and prints progress
information.
"""

# ----------------------------------------------------------------------
# Helper: project a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the simplex {x ≥ 0, Σx = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerical safety – fall back to uniform distribution.
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: a cheap random pairwise mass‑swap (keeps sum exactly 1).
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Pick two distinct indices i, j and move a small amount of mass
    (drawn from a symmetric uniform interval of width ``sigma``) from one
    to the other while preserving non‑negativity."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    # Symmetric proposal.
    delta = (np.random.rand() - 0.5) * sigma

    if delta > 0:
        delta = min(delta, seq[j])          # take from j
    else:
        delta = -min(-delta, seq[i])        # take from i

    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# Linear‑programming step (if cvxpy is available):
# given a current f and its max autocorrelation M,
# find a new g that maximises total mass while respecting (f * g) ≤ M.
# ----------------------------------------------------------------------
def _lp_max_mass(f: np.ndarray, M: float):
    """Solve   maximise Σ g   subject to   (f * g) ≤ M , g ≥ 0.
    Returns the optimal g as a NumPy array or ``None`` on failure."""
    if not _HAS_CVX:
        return None

    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)

    # Convolution of the constant vector f with the variable g.
    conv_fg = cp.conv(cp.Constant(f), g)          # length 2n‑1
    constraints = [conv_fg <= M]                  # component‑wise

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                    max_iters=500, verbose=False)
    except Exception:  # pragma: no cover
        return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None

    g_val = g.value
    if g_val is None:
        return None
    return np.asarray(g_val, dtype=float)


# ----------------------------------------------------------------------
# Sub‑gradient descent (fallback when cvxpy is absent).
# ----------------------------------------------------------------------
def _subgradient_descent(f: np.ndarray, max_iters: int,
                         time_deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a simple projected sub‑gradient descent on the quantity
    ``max(conv)``.  The vector ``f`` is assumed to sum to one.
    Returns the best vector found and its evaluation score.
    """
    best_vec = f.copy()
    try:
        best_score = evaluate_sequence(best_vec.tolist())
    except Exception:
        best_score = float("inf")

    it = 0
    while time.time() < time_deadline and it < max_iters:
        it += 1
        conv = np.convolve(f, f)
        k_max = int(np.argmax(conv))
        M = conv[k_max]

        # Gradient of the max‑convolution term.
        grad = np.zeros_like(f)
        k = k_max
        n = f.shape[0]
        if k < n:
            # contribution from f[i] * f[i+k]
            grad[: n - k] += f[k:]
            grad[k:] += f[: n - k]
        else:
            # For k >= n the overlap is small; we simply ignore it
            # (this case seldom happens for the optimum).
            pass

        # Step size decreases like 1/√t.
        step = 0.5 / np.sqrt(it)
        f_new = f - step * grad
        f = _project_onto_simplex(f_new)

        # Periodically evaluate.
        if it % 20 == 0:
            try:
                sc = evaluate_sequence(f.tolist())
            except Exception:
                sc = float("inf")
            if sc < best_score:
                best_score = sc
                best_vec = f.copy()
                print(f"[SubGrad iter {it:05d}] New best score: {best_score:.6f}")

    return best_vec, best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):

    Return a list of non‑negative heights (a step function) that strives
    to minimise ``evaluate_sequence`` within the allotted time.

      * applies LP refinement (if CVXPY is available) or sub‑gradient descent,
      * finishes with a short simulated‑annealing style polishing,

    """

    best_overall_score = float("inf")

        alpha = np.random.uniform(0.2, 2.0)

                if time.time() >= deadline:
                    break
                # 1. compute M = max(f*f)
                conv_ff = np.convolve(f_current, f_current)

                # 2. Solve the LP that pushes as much total mass as possible
                #    without raising the bound M.

                if sum_g <= 1.0 + 1e-12:

                best_score = cand_score

    # Safety and final return

    best_overall_seq = np.maximum(best_overall_seq, 0.0)
    return best_overall_seq.tolist()
