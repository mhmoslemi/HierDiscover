def refine(a_init):
    a = a_init.copy()
    best_local = a.copy()
    best_local_score = compute_score(a)
    # set variables
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = target_sum * 0.5 / n
    # compute initial conv for temperature
    conv = _self_convolution(a, size_fft)
    T = max((conv.max() - conv.min()) * 0.5, 1e-4)

    it = 0
    patience = 0
    max_patience = 2000
    while time.time() < deadline and it < max_iter:
        conv = _self_convolution(a, size_fft)
        cur_score = 2.0*n*conv.max()/(a.sum()**2)
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local = a.copy()
            patience = 0
            # optional print every improvement
            print(f"[refine] it {it:,} new best {best_local_score:.6f}")
        else:
            patience +=1
        # compute softmax weights
        # avoid overflow: use shift = max
        shifted = conv - conv.max()  # negative values
        exp_arg = shifted / T
        exp_arg = np.clip(exp_arg, -50, 0)   # ensure nonpositive
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient
        grad = _weighted_gradient(a, w, size_fft)
        # Adam step
        it1 = it + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1**it1)
        v_hat = v / (1.0 - beta2**it1)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = _project_to_simplex(a_new, target_sum)

        # random perturbation if stuck
        if patience >= max_patience:
            noise = np.random.randn(n) * target_sum * 1e-3
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)
            patience = 0
            print(f"[refine] it {it:,} random perturbation")
        a = a_new
        # cool temperature gradually
        T = max(T * 0.9995, 1e-8)
        it += 1
        # optional early break if T is tiny
    return best_local, best_local_score

"""
Propose a step‑function sequence with a low self‑convolution peak.
The algorithm consists of three stages:

1. **Broad parametric search** – generate many simple families
   (inverse‑sqrt, power‑law, exponential, sinusoidal, double‑ended,
   random block, uniform) and keep the few best candidates
   (according to the exact evaluation score).

2. **Refinement** – from each of the top candidates run a smooth
   max‑convolution minimisation using a soft‑max approximation.
   The true gradient of the smoothed objective is computed
   analytically with FFTs (the previous version had an
   incorrect diagonal correction).  A Nesterov‑type momentum step
   with a decreasing step size and temperature is employed,
   followed by Euclidean projection onto the simplex
   {a ≥ 0, Σa = √(2 n)}.

3. **Final random‑block sweep** – a quick pass of random block
   candidates to exploit any remaining plateaus.

All operations respect the security checks performed later by
`evaluate_sequence`.  The function returns the best sequence found
before the 1000 s budget expires.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using FFT (full length)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    # keep only the non‑zero part (length = len(u)+len(v)-1)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full convolution a * a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of   Σ_k w_k·(a*a)_k   w.r.t. a  (weights w sum to 1).
    The exact expression is   2·(w ↔ a_rev)  restricted to n entries.
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)          # length = 3·n‑2
    start = len(w) - len(a)                       # = n‑1
    grad = 2.0 * conv[start:start + len(a)]
    return grad


# ----------------------------------------------------------------------
def _score(a: np.ndarray, n: int) -> float:
    """Exact evaluation score (no security checks)."""
    conv = _self_convolution(a, _size_fft_global)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ----------------------------------------------------------------------
def _refine_candidate(a0: np.ndarray,
                      deadline: float,
                      n: int,
                      size_fft: int,
                      target_sum: float,
                      max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Smooth max‑convolution minimisation starting from a0.
    Returns the best sequence encountered and its score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a, n)

    # momentum variables
    vel = np.zeros_like(a)
    momentum = 0.9
    # base step size (scaled to the problem size)
    eta0 = target_sum * 0.5 / n
    decay = 1e-5

    # initialise temperature from the spread of the initial convolution
    conv = _self_convolution(a, size_fft)
    T = max(1e-4, (conv.max() - conv.min()) * 0.5)

    no_improve = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        # ----- evaluate current point -----
        conv = _self_convolution(a, size_fft)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[refine] it {it:,} new best = {best_score:.6f}")
        else:
            no_improve += 1

        # ----- soft‑max weights (smooth approximation of max) -----
        shifted = conv - conv.max()                # now ≤ 0
        exp_arg = shifted / max(T, 1e-8)
        exp_arg = np.clip(exp_arg, -50, 0)         # prevent under/overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        grad = _weighted_gradient(a, w, size_fft)

        # ----- Nesterov‑type momentum update -----
        eta = eta0 / (1.0 + decay * it)
        vel = momentum * vel - eta * grad
        a_new = a + vel

        # ----- simplex projection (keeps Σa = target_sum) -----
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional random kick to escape plateaus
        if (it % 5000 == 0 and it > 0) or no_improve > 5000:
            noise = np.random.randn(n) * target_sum * 1e-3
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)
            no_improve = 0
            print(f"[refine] it {it:,} random perturbation")

        a = a_new
        T *= 0.9999                # slowly cool
        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    the evaluation score.  The function respects a 1000 s time budget and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Problem size
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)          # the normalisation used in the paper
    global _size_fft_global
    _size_fft_global = 1 << ((3 * n - 2).bit_length())   # >= 3n‑2

    # ------------------------------------------------------------------
    # Helper for quick scoring (no security checks)
    def quick_score(a: np.ndarray) -> float:
        conv = _self_convolution(a, _size_fft_global)
        return 2.0 * n * conv.max() / (a.sum() ** 2)

    # ------------------------------------------------------------------
    # Keep the K best raw candidates from the exploratory phase
    K = 10
    best_candidates = []   # list of (score, array)

    def consider(score: float, a: np.ndarray) -> None:
        """Maintain the top‑K list (ascending order)."""
        if len(best_candidates) < K:
            best_candidates.append((score, a.copy()))
            best_candidates.sort(key=lambda x: x[0])
        else:
            if score < best_candidates[-1][0]:
                best_candidates[-1] = (score, a.copy())
                best_candidates.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # ---------- 1. PARAMETRIC SHAPE SEARCH -------------------------
    # Uniform
    a = np.ones(n)
    a = _project_to_simplex(a, target_sum)
    sc = quick_score(a)
    consider(sc, a)
    print(f"[init] uniform score = {sc:.6f}")

    # a few families – each with a modest random sweep
    idx = np.arange(n)

    # Inverse‑sqrt (single‑ended)
    for _ in range(80):
        beta = np.random.rand() * 10.0
        a = 1.0 / np.sqrt(idx + beta + 1.0)
        a = _project_to_simplex(a, target_sum)
        consider(quick_score(a), a)

    # Double‑ended inverse‑sqrt
    for _ in range(80):
        beta = np.random.rand() * 10.0
        a = (1.0 / np.sqrt(idx + beta + 1.0) +
             1.0 / np.sqrt((n - 1 - idx) + beta + 1.0))
        a = _project_to_simplex(a, target_sum)
        consider(quick_score(a), a)

    # Power‑law
    for _ in range(80):
        beta = np.random.rand() * 10.0
        pexp = np.random.rand() * 1.7 + 0.3          # exponent ∈ [0.3, 2.0]
        a = 1.0 / (idx + beta + 1.0) ** pexp
        a = _project_to_simplex(a, target_sum)
        consider(quick_score(a), a)

    # Exponential decay
    for _ in range(80):
        decay = np.random.rand() * 0.01 + 0.0005    # ∈ [5e‑4, 1.05e‑2]
        a = np.exp(-decay * idx)
        a = _project_to_simplex(a, target_sum)
        consider(quick_score(a), a)

    # Cosine‑based (large at ends, small in the middle)
    for _ in range(80):
        p = np.random.rand() * 5.0 + 0.5
        a = np.abs(np.cos(np.pi * idx / (n - 1))) ** p
        a = _project_to_simplex(a, target_sum)
        consider(quick_score(a), a)

    # Sine‑based (zero at ends, peak in the middle)
    for _ in range(80):
        p = np.random.rand() * 5.0 + 0.5
        a = np.sin(np.pi * idx / (n - 1)) ** p
        a = _project_to_simplex(a, target_sum)
        consider(quick_score(a), a)

    # Linear ramp (ascending)
    for _ in range(40):
        p = np.random.rand() * 3.0 + 0.5
        a = (idx + 1) ** p
        a = _project_to_simplex(a, target_sum)
        consider(quick_score(a), a)

    # Quadratic / polynomial
    for _ in range(40):
        p = np.random.rand() * 2.0 + 1.0
        a = (idx + 1) ** p
        a = _project_to_simplex(a, target_sum)
        consider(quick_score(a), a)

    # Random block candidates (few large blocks)
    for _ in range(200):
        nb = np.random.randint(2, 7)                     # 2‑6 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        consider(quick_score(cand), cand)

    # Report best from the exploratory phase
    best_candidates.sort(key=lambda x: x[0])
    print(f"[shape search] best raw score = {best_candidates[0][0]:.6f}")

    # ------------------------------------------------------------------
    # ---------- 2. REFINEMENT (smooth max‑conv descent) ------------
    global_best_a = best_candidates[0][1].copy()
    global_best_score = best_candidates[0][0]

    # iterate over the K best raw candidates (most promising ones first)
    for rank, (raw_score, a0) in enumerate(best_candidates):
        if time.time() >= deadline:
            break
        print(f"[refine start] candidate #{rank+1} (raw score {raw_score:.6f})")
        # allow a generous portion of the remaining budget to each candidate,
        # but stop early if time is running out.
        remaining = deadline - time.time()
        max_iters = int(min(400_000, remaining * 1500))  # rough budget → 1500 it/s
        refined_a, refined_score = _refine_candidate(
            a0, deadline, n, _size_fft_global, target_sum, max_iters
        )
        if refined_score < global_best_score:
            global_best_score = refined_score
            global_best_a = refined_a.copy()
            print(f"[global] new best after refinement = {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # ---------- 3. FINAL RANDOM BLOCK SWEEP -----------------------
    while time.time() < deadline:
        nb = np.random.randint(2, 7)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = quick_score(cand)
        if sc < global_best_score:
            global_best_score = sc
            global_best_a = cand.copy()
            print(f"[final block] improved to {global_best_score:.6f}")
        # a modest limit to avoid endless loop if time is short
        if time.time() + 0.02 > deadline:
            break

    print(f"[final] best score = {global_best_score:.6f}")
    return global_best_a.tolist()
