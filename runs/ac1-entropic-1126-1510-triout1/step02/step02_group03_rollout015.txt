```python
"""Search for low‑score step‑function sequence.

The algorithm proceeds in two phases:

1. **Coarse shape sweep** – for several candidate lengths we generate
   symmetric Beta(α,α) densities (including a simple triangular shape)
   and keep the best (lowest ``2·n·max(conv)``) candidate.

2. **Fine optimisation** – starting from the best candidate we run a
   long mirror‑descent optimisation on the simplex.  The objective is
   approximated by the smooth log‑sum‑exp of the convolution entries,
   with the temperature parameter λ increased gradually up to a large
   value (≈200).  The update is an exponentiated‑gradient step
   (multiplicative mirror descent) followed by a Euclidean projection onto
   the simplex.  Random jitter is occasionally added to escape plateaus.
   The whole run stops when the time budget expires and the best sequence
   found is returned as a plain list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on the grid x_i = (i+½)/n.
    Returns a probability vector (sum = 1).
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals_sum = vals.sum()
    if vals_sum == 0.0:
        # fallback to uniform
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / vals_sum
    return vals


# ----------------------------------------------------------------------
def _triangular_distribution(n: int) -> np.ndarray:
    """Simple symmetric triangular shape (linear rise then fall)."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1, dtype=np.float64)
        a = np.concatenate([left, left[::-1]])
    else:
        left = np.arange(1, half + 1, dtype=np.float64)
        mid = np.full(1, half + 1, dtype=np.float64)
        a = np.concatenate([left, mid, left[::-1]])
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a power‑of‑two FFT.
    Returned length = 2·len(a)‑1.
    """
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, m)
    B = A * A
    c = np.fft.irfft(B, m)
    return c[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k‑i} (correlation) using FFT.
    w – weight vector of length 2·n‑1, a – current sequence (length n),
    m – FFT length (power of two, at least len(w)+len(a)-1).
    Returns the slice g[0…n‑1].
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start : start + len(a)]


# ----------------------------------------------------------------------
def _smooth_max_weights(b: np.ndarray, lamb: float) -> np.ndarray:
    """Return the soft‑max weights w_k = exp(lamb·(b_k‑max(b))) / Σ."""
    maxb = b.max()
    # Safe exponentials
    exp_arg = lamb * (b - maxb)          # ≤ 0
    # Clip to avoid underflow in extreme cases
    exp_arg = np.clip(exp_arg, -700, 0)
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(b, 1.0 / b.shape[0])
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
def _mirror_descent_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    eta0: float = 0.8,
    lambda0: float = 0.5,
    lambda_max: float = 200.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) optimisation of the
    smoothed max‑convolution objective.
    Returns the best sequence seen and its (unsmoothed) score.
    """
    a = a0.astype(np.float64).copy()
    a /= a.sum()
    n = len(a)
    best_seq = a.copy()
    best_score = float(2.0 * n * _conv_full_fft(a).max())

    # FFT size for correlation: must accommodate lengths 2n‑1 + n‑1
    m = 1 << ((3 * n - 2).bit_length())   # sufficient power of two

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current score (unsmoothed)
        b = _conv_full_fft(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} it={it:6d} score={best_score:.6f}")

        # temperature schedule – linear raise from lambda0 to lambda_max
        lamb = lambda0 + (lambda_max - lambda0) * it / max_iters
        w = _smooth_max_weights(b, lamb)

        # gradient via correlation
        g = _gradient_via_correlation(w, a, m)

        # step size (decreasing)
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * g)

        # project back onto the simplex (simple Euclidean projection)
        a = _project_onto_simplex(a, s=1.0)

        # occasional jitter to escape plateaus
        if it % 1000 == 0 and np.random.rand() < 0.25:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score


# ----------------------------------------------------------------------

            hist_score = 2.0 * len(hist) * _conv_full_fft(hist).max()

    except Exception:

        alphas = np.concatenate([alphas_grid, alphas_random])

        if tri_score < best_local_score:

    print(

"""Search for a low‑score step‑function sequence.

The routine works in two phases:

1️⃣  A short sweep over a few lengths and over symmetric Beta(α,α)
    densities (plus a simple triangular shape).  The best shape according
    to the raw objective 2·n·max(conv) is kept.

2️⃣  Starting from that best candidate we run a long mirror‑descent
    optimisation on the simplex.  The objective is a smooth
    log‑sum‑exp approximation of the peak of the self‑convolution; the
    temperature λ is increased linearly up to a large value (≈200) so
    that the algorithm eventually minimises the true peak.  The update
    is an exponentiated‑gradient step (multiplicative mirror descent)
    followed by an Euclidean projection onto the simplex.  Random jitter
    is occasionally added to escape plateaus.  The best sequence found
    before the time limit expires is returned as a plain list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s} (Euclidean)."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on points x_i = (i+½)/n, normalised to sum 1.
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    s = vals.sum()
    if s == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / s
    return vals


# ----------------------------------------------------------------------
def _triangular_distribution(n: int) -> np.ndarray:
    """Simple symmetric triangular shape on {0,…,n‑1}."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1, dtype=np.float64)
        a = np.concatenate([left, left[::-1]])
    else:
        left = np.arange(1, half + 1, dtype=np.float64)
        mid = np.full(1, half + 1, dtype=np.float64)
        a = np.concatenate([left, mid, left[::-1]])
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a via FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, m)
    B = A * A
    c = np.fft.irfft(B, m)
    return c[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g_i = 2·∑_k w_k·a_{k‑i} using FFT.
    w – length 2·n‑1 weight vector,
    a – current sequence (length n),
    m – FFT length (power of two, at least len(w)+len(a)-1).
    Returns vector of length n.
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start : start + len(a)]


# ----------------------------------------------------------------------
def _smooth_max_weights(b: np.ndarray, lamb: float) -> np.ndarray:
    """Soft‑max weights w_k = exp(lamb·(b_k‑max(b))) / Σ."""
    maxb = b.max()
    exp_arg = lamb * (b - maxb)          # ≤ 0
    exp_arg = np.clip(exp_arg, -700, 0)  # avoid under‑/overflow
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(b, 1.0 / b.shape[0])
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
def _mirror_descent_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    eta0: float = 0.8,
    lambda0: float = 0.5,
    lambda_max: float = 200.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) optimisation of the smoothed
    max‑convolution objective.
    Returns the best sequence seen and its (unsmoothed) score.
    """
    a = a0.astype(np.float64).copy()
    a /= a.sum()
    n = len(a)
    best_seq = a.copy()
    best_score = float(2.0 * n * _conv_full_fft(a).max())

    # FFT size sufficient for correlation (len=2n‑1) with a (len=n)
    m = 1 << ((3 * n - 2).bit_length())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current (unsmoothed) score
        b = _conv_full_fft(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} it={it:6d} score={best_score:.6f}")

        # temperature schedule – linear ramp to lambda_max
        lamb = lambda0 + (lambda_max - lambda0) * it / max_iters
        w = _smooth_max_weights(b, lamb)

        # gradient via correlation
        g = _gradient_via_correlation(w, a, m)

        # step size (decreasing like 1/√t)
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update (multiplicative mirror step)
        a = a * np.exp(-eta * g)

        # project back onto the simplex
        a = _project_onto_simplex(a, s=1.0)

        # occasional jitter to avoid stagnation
        if it % 1000 == 0 and np.random.rand() < 0.25:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score
