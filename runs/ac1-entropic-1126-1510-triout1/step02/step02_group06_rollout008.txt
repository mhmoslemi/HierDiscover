    end_time = start_time + budget_s - 0.1

    np.random.seed(seed)

    lengths = [2000, 2500, 3000, 3500] # for variety.

    best_seq = None
    best_score = float('inf')

        cur_score = 2.0 * n * conv.max()

        best_local_seq = a.copy()
        best_local_score = cur_score

        # parameters

        # we will schedule step and T decreasing linearly.

        iteration = 0

            elapsed = time.time() - start_time
            fraction = elapsed / budget_s

"""
Propose a high‑quality step‑function sequence.

Algorithm
---------

1. Choose a few candidate lengths (2000–4000).  
2. For each length initialise the distribution from several
   different starting points (arcsine, Dirichlet with various
   concentration parameters).  Parameterise the distribution by
   unconstrained logits `x` (soft‑max) so the L1‑norm is always 1
   and all entries stay non‑negative.
3. Optimise the non‑smooth objective  

        Φ(a) = 2·n·max_k (a∗a)_k         (sum(a)=1)

   using a quasi‑Newton method (L‑BFGS‑B).  The gradient of the
   max‑convolution is obtained analytically:
   if the maximum occurs at index `k*` then  
        ∂Φ/∂a_i = 2·n·2·a_{k*‑i}  (when 0≤k*‑i<n, else 0).

   The chain rule converts this to a gradient w.r.t. the logits.
4. After the L‑BFGS run perform a short mirror‑descent (multiplicative)
   fine‑tuning using the exact sub‑gradient of the true max.
5. Keep the best sequence (smallest evaluation score) over all runs
   and return it.

The code runs entirely in memory, uses only NumPy and SciPy,
and respects the 1000 s time budget.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize


def arcsine_sequence(n: int) -> list[float]:
    """First `n` coefficients of (1‑z)^‑1/2, normalised to sum 1."""
    seq = []
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    seq = [x / total for x in seq]
    return seq


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(x)
    e = np.exp(x - m)
    s = e.sum()
    if s == 0.0:
        # fallback to uniform
        return np.full_like(x, 1.0 / x.shape[0])
    return e / s


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return a∗a (real part) using an FFT of length `fft_len`."""
    A = np.fft.rfft(a, fft_len)
    c = np.fft.irfft(A * A, fft_len)
    # the true convolution length is 2·len(a)‑1
    return c[: 2 * a.shape[0] - 1]


def _objective_and_grad(x: np.ndarray, n: int, fft_len: int):
    """
    Compute Φ(a) and its gradient wrt logits `x`.

    a = softmax(x)  (∑a = 1, a_i ≥ 0)
    Φ(a) = 2·n·max_k (a∗a)_k
    """
    a = _softmax(x)

    # self‑convolution and its maximum
    conv = _conv_self(a, fft_len)
    k_max = int(conv.argmax())
    max_val = conv[k_max]

    # objective (the scaling 2·n cancels in the gradient later)
    phi = 2.0 * n * max_val

    # gradient of max_k (a∗a)_k w.r.t. a (sub‑gradient)
    grad_a = np.zeros_like(a)
    partners = k_max - np.arange(n)          # partner index for each i
    mask = (partners >= 0) & (partners < n)
    grad_a[mask] = 2.0 * a[partners[mask]]

    # chain rule: dΦ/dx = a * (grad_a - <a,grad_a>)
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)

    return phi, grad_x


def _lbfgs_optimize(start_a: np.ndarray, n: int, max_iter: int = 200) -> np.ndarray:
    """
    Run L‑BFGS‑B on the log‑it variables starting from `start_a`
    (which is already a probability vector).  Returns the improved
    probability vector.
    """
    # initialise logits from the probability vector (log‑it)
    eps = 1e-12
    x0 = np.log(np.clip(start_a, eps, None))

    # FFT length – the next power of two ≥ 2·n‑1
    fft_len = 1 << ((2 * n - 1).bit_length())

    # SciPy minimize (L‑BFGS‑B) with analytic gradient
    res = minimize(
        fun=_objective_and_grad,
        x0=x0,
        args=(n, fft_len),
        method="L-BFGS-B",
        jac=True,
        options={"maxiter": max_iter, "disp": False},
    )
    # convert final logits back to a probability vector
    a_opt = _softmax(res.x)
    return a_opt


def _mirror_descent_step(a: np.ndarray, eta: float) -> np.ndarray:
    """
    One mirror‑descent step (multiplicative) on the simplex,
    using the true sub‑gradient of the max‑convolution.
    """
    n = a.shape[0]
    conv = _conv_self(a, 1 << ((2 * n - 1).bit_length()))
    k_max = int(conv.argmax())
    partners = k_max - np.arange(n)
    mask = (partners >= 0) & (partners < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[partners[mask]]

    # multiplicative update, protect against overflow
    upd = np.exp(-eta * grad)
    cand = a * upd
    cand_sum = cand.sum()
    if cand_sum == 0.0:
        cand = np.full_like(a, 1.0 / n)
    else:
        cand /= cand_sum
    return cand


def _refine_by_mirror(a: np.ndarray,
                      n: int,
                      budget_s: float,
                      start_eta: float = 0.5) -> np.ndarray:
    """
    Short mirror‑descent polishing phase (few thousand cheap steps).
    Stops when `budget_s` seconds have elapsed.
    """
    end = time.time() + budget_s
    eta = start_eta
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a, 1 << ((2 * n - 1).bit_length())).max()

    while time.time() < end:
        cand = _mirror_descent_step(a, eta)
        cur_score = 2.0 * n * _conv_self(cand,
                                         1 << ((2 * n - 1).bit_length())
                                         ).max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand.copy()
        # accept if improvement, otherwise shrink step‑size
        if cur_score <= 2.0 * n * _conv_self(a,
                                            1 << ((2 * n - 1).bit_length())
                                            ).max():
            a = cand
            eta = min(eta * 1.1, 5.0)
        else:
            eta *= 0.6
        if eta < 1e-8:
            break
    return best_a


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a list of non‑negative heights (the sequence is already
    normalised; the evaluation routine rescales it anyway).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1  # safety margin

    # lengths to investigate – a compromise between quality and speed
    lengths = [1800, 2400, 3000, 3600]

    # random Dirichlet concentration parameters we will try
    conc_params = [0.3, 0.5, 1.0, 2.0]

    best_seq = None
    best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # pre‑compute FFT length once per n
        fft_len = 1 << ((2 * n - 1).bit_length())

        # ----- 1. generate initial candidates ---------------------------------
        inits = []

        # a) classic arcsine
        inits.append(np.array(arcsine_sequence(n), dtype=np.float64))

        # b) several Dirichlet draws with different concentration
        for alpha in conc_params:
            draws = np.random.gamma(alpha, 1.0, size=n)
            draws /= draws.sum()
            inits.append(draws)

        # ----- 2. run L‑BFGS from each start --------------------------------
        for idx, a0 in enumerate(inits):
            if time.time() >= deadline:
                break

            # give each start an equal slice of the remaining budget
            rem = deadline - time.time()
            per_start = max(5.0, rem / (len(inits) - idx))

            # Run L‑BFGS (fast; a few hundred iterations)
            a_opt = _lbfgs_optimize(a0, n, max_iter=200)

            # Short mirror‑descent polishing (use up to half of the slice)
            polish_time = per_start * 0.5
            a_opt = _refine_by_mirror(a_opt, n, polish_time, start_eta=0.5)

            # compute the true evaluation score (calls the provided function)
            seq = a_opt.tolist()
            # clipping to the limits enforced by evaluate_sequence
            seq = [float(max(0.0, min(1000.0, x))) for x in seq]
            try:
                score = evaluate_sequence(seq)
            except Exception:      # safety – the evaluator is assumed to exist
                score = float('inf')

            print(f"  start {idx+1}/{len(inits)} – score = {score:.6f}")

            if score < best_score:
                best_score = score
                best_seq = seq
                print(f"*** New overall best (len={n}) score = {best_score:.6f}")

        # End of length looping
        print(f"Finished length {n}: best score so far = {best_score:.6f}")

    if best_seq is None:
        # Fallback – never should happen
        best_seq = arcsine_sequence(2000)

    # final safety clipping (already done above, but enforce once more)
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    print(f"\n--- FINAL RESULT ---\nBest length = {len(best_seq)} "
          f"with metric = {best_score:.6f}")
    return best_seq
