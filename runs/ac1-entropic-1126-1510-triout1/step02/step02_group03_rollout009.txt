    a = init_seq.copy()
    a = a / a.sum()
    best_seq = a.copy()
    best_score = evaluate_score(a)
    # FFT size
    m = 1 << ((3 * n - 2).bit_length())
    # Adam params
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m_t = np.zeros(n)
    v_t = np.zeros(n)
    t_iter = 0
    lambda_val = 0.5
    lambda_max = 12.0
    # step size schedule
    base_lr = 0.1
    while time.time() < deadline:
        t_iter += 1
        # Compute convolution
        b = conv_full_fft(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n} it={t_iter} score={best_score:.6f}")
        # compute weights for smoothed max
        maxb = b.max()
        # if lambda large maybe numerical issues; compute exponent safely
        exp_arg = lambda_val * (b - maxb)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(b, 1.0 / b.size)
        else:
            w = w / w_sum
        # gradient
        g = gradient_via_correlation(w, a, m)
        # Clip gradient
        grad_clip = 10.0
        g = np.clip(g, -grad_clip, grad_clip)
        # Adam update
        m_t = beta1 * m_t + (1 - beta1) * g
        v_t = beta2 * v_t + (1 - beta2) * (g ** 2)
        m_hat = m_t / (1 - beta1 ** t_iter)
        v_hat = v_t / (1 - beta2 ** t_iter)
        lr = base_lr / (np.sqrt(t_iter) + 1)  # decreasing lr
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = a + jitter
            a = np.clip(a, 0, None)
            a = project_onto_simplex(a)
    return best_seq, best_score

"""
Propose a low‑score step‑function.

The algorithm proceeds in three phases:

1. **Initial sweep** –  For a handful of candidate lengths we
   generate cheap candidates:
   * uniform,
   * the discrete arcsine distribution,
   * several symmetric Beta(α,α) densities (α∈[0.2,1.8]),
   * a few random Dirichlet draws.
   Each candidate is symmetrised, normalised to Σa=1 and evaluated
   with the exact convolution.  The best candidate over all lengths
   is kept.

2. **Smooth optimisation** –  Starting from the best candidate we
   minimise a smooth approximation of the peak of the self‑convolution.
   For a current sequence `a` we define

        Fλ(a) = (1/λ)·log Σ_k exp( λ·(a*a)_k ),

   whose gradient can be written as
   `∇Fλ(a) = 2·corr(w, a)` with `w` the soft‑max weights.
   We perform a projected Adam optimisation (additive updates,
   Euclidean projection onto the simplex) while slowly increasing λ.
   Gradient clipping and occasional jitter keep the iterates stable.
   The best sequence seen during the optimisation is returned.

3. **Fallback** –  If nothing is found we fall back to a uniform
   distribution of moderate size.

All heavy work (full convolution, correlation) uses power‑of‑two FFTs.
The routine respects the 1000 s time budget and prints progress
information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (α=½) on n points, Σa = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on the grid x_i = (i+½)/n.
    Returns a probability vector (sum = 1).
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1.0 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, m)
    C = np.fft.irfft(A * A, m)
    return C[: 2 * n - 1]


# ----------------------------------------------------------------------
def gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k‑i} (correlation) using FFT.
    w – weight vector of length 2·n‑1, a – current sequence (length n),
    m – FFT length (power of two, at least len(w)+len(a)-1).
    Returns the slice g[0…n‑1].
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = a.shape[0] - 1
    return 2.0 * corr[start:start + a.shape[0]]


# ----------------------------------------------------------------------
def evaluate_score_seq(a: np.ndarray) -> float:
    """Exact score 2·n·max(conv(a,a)) for a normalised (Σa=1) sequence."""
    n = a.shape[0]
    conv = conv_full_fft(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def refine_sequence_smooth(init_seq: np.ndarray,
                           deadline: float) -> tuple[np.ndarray, float]:
    """
    Refine `init_seq` (already normalised to Σ=1) by minimising a smooth
    approximation of max_conv using projected Adam with gradient clipping.
    Returns the best sequence seen and its exact score.
    """
    a = np.clip(init_seq, 0.0, None)
    a /= a.sum()
    n = a.shape[0]

    # pre‑compute an FFT length large enough for correlation (w length = 2n‑1)
    m = 1 << ((3 * n - 2).bit_length())   # >= (2n‑1)+(n‑1)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m_t = np.zeros(n, dtype=np.float64)
    v_t = np.zeros(n, dtype=np.float64)

    # smooth‑max schedule
    lam = 0.5
    lam_max = 10.0
    lam_factor = 1.002

    # learning‑rate schedule
    lr0 = 0.12

    best_seq = a.copy()
    best_score = evaluate_score_seq(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # current convolution and score
        conv = conv_full_fft(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] n={n:5d} iter={it:5d} score={best_score:.6f}"
            )

        # smooth‑max (soft‑max) weights
        maxc = conv.max()
        exp_arg = lam * (conv - maxc)          # ≤ 0, safe numerically
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.shape[0])
        else:
            w /= w_sum

        # gradient of the smoothed objective
        g = gradient_via_correlation(w, a, m)

        # gradient clipping (helps stability for large lam)
        g = np.clip(g, -5.0, 5.0)

        # Adam update (additive)
        m_t = beta1 * m_t + (1.0 - beta1) * g
        v_t = beta2 * v_t + (1.0 - beta2) * (g * g)
        m_hat = m_t / (1.0 - beta1 ** it)
        v_hat = v_t / (1.0 - beta2 ** it)

        lr = lr0 / np.sqrt(it)               # slowly decaying step size
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection back onto the probability simplex
        a = project_onto_simplex(a)

        # increase smoothing parameter gradually
        lam = min(lam_max, lam * lam_factor)

        # occasional jitter to escape tiny plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a = project_onto_simplex(a)

    return best_seq, best_score
