"""
Simulated‑annealing optimizer for the self‑convolution max‑norm.

The function starts from the best known construction (the global variable
``height_sequence_1`` if it exists) and tries to improve it by repeatedly
moving a small amount of mass from one position to another while keeping the
ℓ₁‑norm fixed (the objective is scale‑invariant).  An annealing schedule
accepts occasional worsening moves, allowing the search to escape local minima.
All candidates are projected onto the simplex {a≥0, Σa = √(2 n)} after each
change.  The convolution is evaluated with an FFT (O(n log n)) and the score
is simply the maximal convolution coefficient (because Σa is kept constant).

The algorithm runs until the time budget expires (default 1000 s) and returns
the best sequence found.
"""

import time
import math
import numpy as np


def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – put all mass on the first coordinate
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using a power‑of‑2 FFT."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2 n‑1)."""
    return _conv_fft(a, a, size_fft)


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """Return a step‑function (list of non‑negative heights) with a low score."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # problem size
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)          # the normalisation used in the paper
    fft_len = 1 << ((2 * n - 1).bit_length())   # power of two >= 2 n‑1

    # --------------------------------------------------------------
    # initialise a – use the best known construction if available
    if have_ref:
        a = np.asarray(height_sequence_1, dtype=float)
        a = _project_to_simplex(a, target_sum)
    else:
        # simple exponential decay as a fallback
        decay = 0.001
        a = np.exp(-decay * np.arange(n))
        a = _project_to_simplex(a, target_sum)

    # current score (max of the convolution; because Σa is fixed this equals the
    # evaluate_sequence value)
    cur_conv = _self_convolution(a, fft_len)
    cur_score = float(cur_conv.max())
    best_score = cur_score
    best_a = a.copy()

    print(f"[init] n={n}, initial score={cur_score:.6f}")

    # --------------------------------------------------------------
    # simulated annealing parameters
    temperature = cur_score * 0.10          # start with 10 % of the score
    move_fraction = 0.25                    # at most 25 % of source mass moved
    it = 0
    stagnation = 0                          # how many iterations without improvement

    while time.time() < deadline and it < 5_000_000:
        # pick two distinct indices, ensure the source has some mass
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # amount to transfer from i to j
        delta = a[i] * (np.random.rand() * move_fraction)

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # keep the ℓ₁‑norm exact (tiny drift may appear due to floating point)
        # projecting onto the simplex also guarantees non‑negativity
        a_new = _project_to_simplex(a_new, target_sum)

        # evaluate
        conv = _self_convolution(a_new, fft_len)
        new_score = float(conv.max())

        # Metropolis acceptance
        if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / max(temperature, 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                stagnation = 0
                print(f"[{it:07d}] new best = {best_score:.6f}")
            else:
                stagnation += 1
        else:
            stagnation += 1

        # temperature schedule (exponential cooling)
        if it % 1000 == 0 and it > 0:
            temperature *= 0.9995

        # occasional large perturbation if stuck for a long time
        if stagnation > 200_000:
            # inject a fresh random block pattern
            nb = np.random.randint(2, 6)
            cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = np.random.rand(nb) + 0.1
            cand = np.zeros(n)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            a = _project_to_simplex(cand, target_sum)
            cur_score = float(_self_convolution(a, fft_len).max())
            stagnation = 0
            print(f"[{it:07d}] random block perturbation (score={cur_score:.6f})")

        it += 1

    print(f"[final] best score = {best_score:.6f} after {it} iterations")
    # Return the best sequence (already respects all guards of evaluate_sequence)
    return best_a.tolist()
