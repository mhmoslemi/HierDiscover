"""
Propose a low‑score step function.

The algorithm is a hybrid of three simple ideas
    1.  Start from the known best construction (``height_sequence_1``) if it
        exists, otherwise from a random point on the simplex
        { a ≥ 0 , Σa = √(2 n) }.
    2.  Perform a projected (soft‑max) gradient descent with
        momentum – the same basic scheme as the reference code,
        but with more aggressive hyper‑parameters
        (larger top‑k, slower cooling, more restarts) in order to
        push the max‑convolution a little further down.
    3.  After the long gradient phase, run a cheap but exhaustive
        block‑search (piecewise‑constant patterns) that can still
        discover tiny improvements.

The whole routine respects the 1000 s wall‑clock budget
and returns the best list of heights found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _gradient_topk(a: np.ndarray, conv: np.ndarray, top_k: int = 10) -> np.ndarray:
    """Sub‑gradient of max conv based on the ``top_k`` largest entries."""
    n = a.shape[0]
    idxs = np.argpartition(-conv, top_k - 1)[:top_k]
    grad = np.zeros_like(a)
    i = np.arange(n)
    for k in idxs:
        j = k - i
        mask = (j >= 0) & (j < n)
        grad[mask] += 2.0 * a[j[mask]]
        diag = mask & (i == j)
        grad[diag] -= a[i[diag]]
    return grad

    for _ in range(block_trials):
        nb = np.random.randint(2, 12)                       # 2‑11 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb)
        cand = np.zeros(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = _max_conv(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[block] improved block score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final tiny hill‑climb (swapping mass between two positions).
    # Run while there is a little time left.
    while time.time() < deadline:
        # pick two distinct indices, one with positive mass, one arbitrary.
        nz = np.flatnonzero(best_a > 1e-12)
        if nz.size == 0:
            break
        i = np.random.choice(nz)
        j = np.random.randint(0, n)
        if i == j:
            continue
        delta = best_a[i] * np.random.uniform(0.1, 0.5)  # move a fraction
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        # project back only to keep feasibility (sum unchanged, non‑negative)
        cand = _project_to_simplex(cand, target_sum)
        sc = _max_conv(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand
            print(f"[swap] new best = {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
