Now about simplex projection: Use algorithm from "Efficient projection onto the simplex." Implementation:

def project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

    # Use Adam
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    while time.time() < deadline:
        t += 1
        # compute conv
        conv = np.convolve(seq, seq)
        # smoothing
        max_conv = conv.max()
        # compute softmax weights
        # shift for numerical stability
        shift = max_conv
        exp_alpha = np.exp(alpha * (conv - shift))
        sum_exp = exp_alpha.sum()
        w = exp_alpha / sum_exp  # shape 2n-1
        # compute gradient g = 2 * sum_k w_k * a_{k-i}

        grad = 2.0 * conv_rev[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        alpha = min(alpha * 1.01, 150.0)   # cap

        if t % 100 == 0:
            stepsize *= 0.9

    # final best

"""
Gradient‑descent optimiser for the step‑function problem.

The objective is 2·n·max(conv(a,a)) / (sum(a)**2).  After fixing the
scale (sum(a)=1) the problem reduces to minimising the maximal value of
the self‑convolution.  We approximate the max by a smooth log‑sum‑exp
and perform projected Adam updates on the probability simplex.
A simple simplex projection (the algorithm of Duchi et al.) guarantees
non‑negative entries that sum to one.  The routine runs until the time
budget (≈ 1000 s) expires and returns the best sequence found.

The implementation uses FFTs for the convolution and for the gradient
(computed as the cross‑correlation of the current weights with the
soft‑max vector), which makes each iteration O(n log n) and allows many
thousands of iterations in the allotted time.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Projects an arbitrary vector onto the probability simplex
    {x | x_i >= 0, sum_i x_i = 1}.
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries are zero or negative – return the uniform point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast linear convolution using a power‑of‑two FFT (real inputs)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Returns the linear convolution a * b using an FFT of size
    next power of two.  Result length = len(a) + len(b) - 1.
    """
    L = a.size + b.size - 1
    N = 1 << ((L - 1).bit_length())          # next power of two ≥ L
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:L]
    return conv


# ----------------------------------------------------------------------
# Gradient of the smooth‑max (log‑sum‑exp) of the convolution
# ----------------------------------------------------------------------
def _smooth_max_gradient(seq: np.ndarray, alpha: float) -> np.ndarray:
    """
    Computes ∇_a  ( 1/α log Σ_k exp(α·conv_k) )  for `seq`.
    Returns a vector of length len(seq).
    """
    n = seq.size
    # convolution a*a (length 2n-1)
    conv = _conv_fft(seq, seq)

    # soft‑max weights (stable by subtracting the max)
    shift = conv.max()
    exp_alpha = np.exp(alpha * (conv - shift))
    w = exp_alpha / exp_alpha.sum()                 # shape (2n-1,)

    # gradient = 2 * Σ_k w_k * a_{k-i}
    # This is a cross‑correlation between `seq` and the weight vector `w`.
    # By reversing `w` we can obtain it with a single convolution.
    w_rev = w[::-1]
    conv_rev = _conv_fft(seq, w_rev)                # length 3n-2
    idx = (2 * n - 2) - np.arange(n)                # indices giving Σ_k w_k a_{k-i}
    grad = 2.0 * conv_rev[idx]
    return grad, conv.max()                         # also return current max for monitoring


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Finds a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine runs a projected Adam optimiser with
    a smooth‑max surrogate, starting either from the supplied global
    `height_sequence_1` (if present) or from a random simplex point.
    The best sequence encountered before the time budget expires is
    returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # Choose a convenient dimension (trade‑off between speed and expressivity)
    # ------------------------------------------------------------------
    n = 2500                                          # can be changed if desired

    # ------------------------------------------------------------------
    # Initialise the population – try to reuse a previously discovered
    # construction if it exists.
    # ------------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        # Resize / truncate / pad to the target length `n`
        if base.size > n:
            # pick roughly evenly spaced entries
            idx = np.linspace(0, base.size - 1, n).astype(int)
            base = base[idx]
        elif base.size < n:
            base = np.pad(base, (0, n - base.size), mode="constant")
        seq = _project_to_simplex(base)
    except Exception:
        # random start on the simplex
        seq = _project_to_simplex(np.random.rand(n))

    # ------------------------------------------------------------------
    # Adam state
    # ------------------------------------------------------------------
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    step_size = 0.5

    # smoothing parameter for the log‑sum‑exp approximation
    alpha = 0.5
    alpha_max = 80.0
    alpha_factor = 1.02

    # best solution seen so far (store as Python list for final return)
    best_seq = seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[Init] n = {n}, initial score = {best_score:.6f}")

    iteration = 0
    eval_interval = 30          # call the heavy `evaluate_sequence` only every few steps

    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # Compute smoothed gradient
        # --------------------------------------------------------------
        grad, cur_max = _smooth_max_gradient(seq, alpha)

        # --------------------------------------------------------------
        # Adam update (gradient descent on the smooth surrogate)
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        seq = seq - step_size * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # Projection back onto the probability simplex
        # --------------------------------------------------------------
        seq = _project_to_simplex(seq)

        # --------------------------------------------------------------
        # Occasionally inject a tiny random perturbation (helps escape plateaus)
        # --------------------------------------------------------------
        if iteration % 500 == 0:
            noise = (np.random.rand(n) - 0.5) * 1e-2
            seq = _project_to_simplex(seq + noise)

        # --------------------------------------------------------------
        # Periodically evaluate the true objective (expensive!)
        # --------------------------------------------------------------
        if iteration % eval_interval == 0:
            score = float(evaluate_sequence(seq.tolist()))
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[Iter {iteration:5d}] New best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Anneal the smoothing parameter – gradually turn the surrogate
        # into the true max.
        # --------------------------------------------------------------
        if alpha < alpha_max:
            alpha = min(alpha * alpha_factor, alpha_max)

        # --------------------------------------------------------------
        # Slowly decay the step size to aid convergence.
        # --------------------------------------------------------------
        if iteration % 200 == 0:
            step_size = max(step_size * 0.9, 1e-5)

    print(f"[Done] Finished after {iteration} iterations – best score = {best_score:.6f}")
    return best_seq.tolist()
