"""
Propose a step‑function sequence that minimizes the metric

    2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm combines three ideas:

1.  **Start from a good analytic shape** – the discrete “arcsine”
    sequence (C_i = binom(2i,i)/4^i) is known to give a low value
    (≈π/2).  For comparison we also try a uniform vector.

2.  **LP refinement** – for a current f, solve the linear program

        maximise    Σ g_j
        subject to  g ≥ 0,
                    f ∗ g ≤ M    (element‑wise),

   where M = max(f∗f).  The constraints are linear in the unknown g,
   so a dense matrix representation of the convolution is built and
   SciPy’s HiGHS LP solver is used.  The optimum g fills the slack
   without increasing M.  After solving we renormalise g to L¹‑norm = 1
   and mix it with f:  f ← (1‑t)·f + t·g, projecting back onto the
   probability simplex.  The mixing weight t starts at 0.3 and decays.

3.  **Gradient‑descent polishing** – when the LP step is not possible
   (e.g. SciPy missing or time‑critical) we minimise the variance of the
   convolution (which pushes it toward a flat profile) using a projected
   gradient descent with an Adam‑like step‑size schedule.

The search runs for the allotted time (≈1000 s) and remembers the best
sequence seen (lowest value reported by ``evaluate_sequence``).  The
returned list satisfies all security checks of the evaluator.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> np.ndarray:
    """Discrete arcsine coefficients, L¹‑normalised to 1."""
    seq = np.empty(n, dtype=float)
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq[i] = cur
        total += cur
        if i + 1 < n:
            cur = cur * (2.0 * i + 1.0) / (2.0 * (i + 1))
    seq /= total
    return seq


# ----------------------------------------------------------------------
def uniform_sequence(n: int) -> np.ndarray:
    """Flat distribution (all entries equal)."""
    return np.full(n, 1.0 / n, dtype=float)


# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
    (Duchi et al., 2008).
    """
    if s < 0:
        raise ValueError("target sum s must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def convolution(seq: np.ndarray) -> np.ndarray:
    """
    Compute the (linear) self‑convolution of a real vector.
    Returned array has length 2·len(seq)‑1 and is real‑valued.
    """
    n = seq.shape[0]
    m = 2 * n - 1
    # next power of two is not needed; np.fft works fine with arbitrary length
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)
    return conv


# ----------------------------------------------------------------------
def lp_refine(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    One LP refinement step.
    Solves maximise Σ g   s.t.   g ≥ 0,   f∗g ≤ M.
    Returns the new mixed vector (1‑t)·f + t·g_scaled,
    or None if the LP could not be solved.
    """
    n = f.shape[0]

    # Build dense convolution matrix A where (A @ g)[k] = (f∗g)[k]
    A = np.zeros((2 * n - 1, n), dtype=float)
    for i in range(n):
        A[i:i + n, i] = f[i]

    # Linear programme (SciPy – minimise –cᵀx)
    from scipy.optimize import linprog
    c = -np.ones(n)                 # maximise sum(g)  → minimise -sum(g)
    bounds = [(0, None)] * n
    ub = M * np.ones(2 * n - 1)

    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": 30},
        )
    except Exception:
        return None

    if not res.success or res.x is None:
        return None

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return None
    g_scaled = g / sum_g
    return g_scaled


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """Search for a sequence with a low evaluation score within the time budget."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = float('inf')
    best_n = None

    # ------------------------------------------------------------------
    # 1) Choose a starting length and initialise with a few candidates
    lengths = [1500, 2000, 2500, 3000, 3500, 4000]
    for n in lengths:
        if time.time() >= deadline:
            break
        seqs = {
            "arcsine": arcsine_sequence(n),
            "uniform": uniform_sequence(n),
        }
        for name, seq in seqs.items():
            score = evaluate_sequence(seq.tolist())
            print(f"[init] n={n:<4d} {name:<8s} score={score:.6f}")
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_n = n

    # fallback sanity
    if best_seq is None:
        best_n = 2000
        best_seq = arcsine_sequence(best_n)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] n={best_n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Helper to evaluate quickly (list → float)
    def current_score(a: np.ndarray) -> float:
        return evaluate_sequence(a.tolist())

    # ------------------------------------------------------------------
    # Parameters for the alternating LP scheme
    mix_weight = 0.30        # start mixing amount
    mix_min = 0.02
    mix_decay = 0.96

    # Parameters for gradient‑descent polishing (variance of convolution)
    adam_alpha = 0.05
    adam_beta1 = 0.9
    adam_beta2 = 0.999
    eps = 1e-12
    m = np.zeros_like(best_seq)
    v = np.zeros_like(best_seq)
    iter_adam = 0

    # Main optimisation loop
    while time.time() < deadline:
        n = best_seq.shape[0]

        # ------------------------------------------------------------------
        # current convolution and bound M
        conv = convolution(best_seq)
        M = float(conv.max())

        # ------------------------------------------------------------------
        # try an LP refinement (preferring it because it has a proven
        # monotone decrease of the objective)
        try:
            from scipy.optimize import linprog  # noqa: F401 – just to check availability
            have_scipy = True
        except Exception:
            have_scipy = False

        if have_scipy and (deadline - time.time() > 2.0):
            g_scaled = lp_refine(best_seq, M)
            if g_scaled is not None:
                # mixing step
                new_f = (1.0 - mix_weight) * best_seq + mix_weight * g_scaled
                new_f = project_simplex(new_f, s=1.0)
                # evaluate
                sc = current_score(new_f)
                if sc < best_score:
                    best_score = sc
                    best_seq = new_f
                    print(f"[LP] n={n:<4d} improvement → {sc:.6f}")
                # decay mixing weight
                mix_weight = max(mix_min, mix_weight * mix_decay)
                continue   # go to next iteration, skip gradient step

        # ------------------------------------------------------------------
        # Gradient‑descent polishing – minimise variance of convolution
        # Target value μ = average of convolution = 1/(2n‑1)
        mu = 1.0 / (2 * n - 1)
        err = conv - mu                      # vector length 2n‑1
        # gradient of Σ err²  w.r.t. a  (see analysis notes)
        grad = 2.0 * np.convolve(err, best_seq[::-1])[n - 1:2 * n - 1]

        # Adam‑style update
        iter_adam += 1
        m = adam_beta1 * m + (1 - adam_beta1) * grad
        v = adam_beta2 * v + (1 - adam_beta2) * (grad ** 2)
        m_hat = m / (1 - adam_beta1 ** iter_adam)
        v_hat = v / (1 - adam_beta2 ** iter_adam)
        step = adam_alpha * m_hat / (np.sqrt(v_hat) + eps)

        cand = best_seq - step
        cand = project_simplex(cand, s=1.0)
        cand_score = current_score(cand)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            print(f"[grad] n={n:<4d} new best → {cand_score:.6f}")

        # occasional random restart if stagnating
        if iter_adam % 300 == 0:
            # small random perturbation of the best sequence
            noise = np.random.rand(n)
            noise = noise / noise.sum()
            perturbed = 0.95 * best_seq + 0.05 * noise
            perturbed = project_simplex(perturbed, s=1.0)
            pert_score = current_score(perturbed)
            if pert_score < best_score:
                best_score = pert_score
                best_seq = perturbed
                print(f"[rand] n={n:<4d} random improvement → {pert_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – enforce positivity and unit sum
    best_seq = np.maximum(best_seq, 0.0)
    best_seq /= best_seq.sum()

    print(f"Best sequence length = {len(best_seq)}  score = {best_score:.6f}")
    return best_seq.tolist()
