"""
Propose a candidate step function using Adam‑based projected gradient descent
with a temperature‑controlled soft‑max surrogate for the convolution maximum.
The method runs on several lengths (the known reference length and a larger
one), performs periodic restarts and a final random block search.  The
projection onto the simplex enforces the ℓ₁‑norm constraint Σa = √(2 n) so
that the evaluation score equals the maximum auto‑convolution value.
"""

import time
import math
import numpy as np

def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries will be zero except possibly one
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _softmax_weights(conv: np.ndarray, mu: float) -> np.ndarray:
    """Return weights proportional to exp((conv - max(conv))/mu)."""
    cur_max = conv.max()
    diff = conv - cur_max        # ≤ 0
    # when mu is extremely small, exp may underflow to 0 → keep at least the max entry
    w = np.exp(diff / mu)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        w_sum = 1.0
    w /= w_sum
    return w

def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Search for a low‑score step function.

    The algorithm runs an Adam‑style projected gradient descent on the
    simplex Σa = √(2 n) using a temperature‑controlled smooth approximation
    of the ℓ∞‑norm of the auto‑convolution.  After the main loop a cheap
    random block search is performed.  Several problem sizes are tried;
    the best sequence found across all sizes is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # lengths to explore – the reference length and a larger one.
    lengths = [3130, 5000]
    best_score = float("inf")
    best_seq = None

    for n in lengths:
        # stop if we ran out of time
        if time.time() >= deadline:
            break

        # target ℓ1‑norm that makes the evaluation equal to max(conv)
        target_sum = math.sqrt(2.0 * n)

        # initialise a – either from the reference (if length matches) or random
        try:
            # height_sequence_1 may have the reference length
            if n == len(height_sequence_1):      # type: ignore[name-defined]
                a = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
            else:
                raise NameError
        except Exception:
            a = np.random.rand(n)

        a = _project_to_simplex(a, target_sum)

        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8

        # learning‑rate – roughly the magnitude of a typical entry
        eta0 = 0.6 * target_sum / n

        # temperature for soft‑max surrogate
        cur_conv = np.convolve(a, a)
        mu = cur_conv.max()          # start at current max
        mu_decay = 0.9993
        mu_min = 1e-9

        # restart schedule
        max_restarts = 2
        restart_interval = (budget_s - 5) / max_restarts
        next_restart = time.time() + restart_interval
        restarts_done = 0

        it = 0
        no_improve = 0

        while time.time() < deadline:
            conv = np.convolve(a, a)
            cur_max = conv.max()

            # record global best
            if cur_max < best_score:
                best_score = float(cur_max)
                best_seq = a.copy()
                no_improve = 0
                print(f"[n={n}] it={it:07d} new best={best_score:.6f}")
            else:

                # compute sum_k w_k a_{k-i} efficiently via one convolution
                # use reversed a to align indices
                conv_wr = np.convolve(w, a[::-1])

                    mask = (j >= 0) & (j < n)
                    grad[mask] += 2.0 * a[j[mask]]

            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            a = a - step

            a = _project_to_simplex(a, target_sum)

            # occasional random shake to escape plateaus

                a += np.random.randn(n) * target_sum * 5e-5
                a = _project_to_simplex(a, target_sum)

            # temperature cooling
            mu = max(mu * mu_decay, mu_min)

            # restart if needed
            if time.time() > next_restart and restarts_done < max_restarts:
                restarts_done += 1
                print(f"[n={n}] restart #{restarts_done} at it={it}")
                a = np.random.rand(n)
                a = _project_to_simplex(a, target_sum)
                m.fill(0.0)
                v.fill(0.0)
                mu = np.max(np.convolve(a, a))
                next_restart = time.time() + restart_interval

            # early exit if we haven't improved for a long stretch
            if no_improve > 200000:
                # treat as stagnation – perform a soft restart
                print(f"[n={n}] stagnation at it={it}, performing soft restart")
                a = np.random.rand(n)
                a = _project_to_simplex(a, target_sum)
                m.fill(0.0)
                v.fill(0.0)
                mu = np.max(np.convolve(a, a))
                no_improve = 0

        # ----------- cheap random block search for this n -------------
        block_trials = 400
        for _ in range(block_trials):
            if time.time() >= deadline:
                break
            nb = np.random.randint(2, 15)               # 2‑14 blocks
            cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = np.random.rand(nb)
            cand = np.zeros(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, target_sum)
            cur_max = np.max(np.convolve(cand, cand))
            if cur_max < best_score:
                best_score = float(cur_max)
                best_seq = cand.copy()
                print(f"[n={n} block] improved to {best_score:.6f}")

    # fallback – should never happen, but keep a valid sequence
    if best_seq is None:
        n = 2000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)

    # Convert to plain Python list of floats
    result = [float(x) for x in best_seq]
    print(f"[final] best score = {best_score:.6f} using length {len(result)}")
    return result

Propose a candidate step function using Adam‑based projected gradient descent.

projection onto the simplex Σ a_i = √(2 n), and does periodic random restarts.
When the time budget is nearly exhausted a cheap random block search is
performed.  The best sequence found across all lengths is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    # Find the largest rho satisfying the condition
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, mu: float) -> np.ndarray:
    """Weights proportional to exp((conv - max(conv))/mu), normalised."""
    cur_max = conv.max()
    diff = conv - cur_max          # ≤ 0
    w = np.exp(diff / mu)
    w_sum = w.sum()
    if w_sum == 0.0:               # numerical safety when mu is tiny
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        w_sum = 1.0
    w /= w_sum
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Search for a low‑score step function.

    The procedure explores two lengths (the reference length 3130 and a
    larger one 5000), runs Adam‑style projected gradient descent with a
    temperature‑controlled smooth approximation of the ℓ∞‑norm of the
    auto‑convolution, performs a few random restarts and finishes with a
    cheap block‑search.  The best sequence found is returned as a list of
    Python floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # Problem sizes to try – keep the list short to respect the time budget.
    lengths = [3130, 5000]

    best_score = float("inf")
    best_seq = None

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)      # Σa = √(2 n) ⇒ score = max(conv)

        # Initialise a: use the reference if the length matches, otherwise random.
        try:
            if n == len(height_sequence_1):   # type: ignore[name-defined]
                a = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
            else:
                raise NameError
        except Exception:
            a = np.random.rand(n)

        a = _project_to_simplex(a, target_sum)

        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8

        # Base step size (≈ average entry magnitude)
        eta0 = 0.6 * target_sum / n

        # Temperature for the smooth‑max surrogate
        cur_conv = np.convolve(a, a)
        mu = cur_conv.max()
        mu_decay = 0.9993
        mu_min = 1e-9

        # Restart handling
        max_restarts = 2
        restart_interval = (budget_s - 5) / max_restarts
        next_restart = time.time() + restart_interval
        restarts_done = 0

        it = 0
        no_improve = 0

        # ------------------- main optimisation loop --------------------
        while time.time() < deadline:
            conv = np.convolve(a, a)
            cur_max = conv.max()

            # Record global best
            if cur_max < best_score:
                best_score = float(cur_max)
                best_seq = a.copy()
                no_improve = 0
                print(f"[n={n}] it={it:07d} new best={best_score:.6f}")
            else:
                no_improve += 1

            # ----------- gradient computation -------------------------
            if mu > max(0.01 * cur_max, 1e-8):
                # Soft‑max (smooth‑max) gradient
                w = _softmax_weights(conv, mu)

                # Compute Σ_k w_k a_{k-i} for every i via one convolution.
                conv_wr = np.convolve(w, a[::-1])          # length 3n‑2
                sum_term = conv_wr[n - 1 : n - 1 + n]      # indices i = 0 … n‑1
                grad = 2.0 * sum_term

                # Diagonal correction for k = 2i (the a_i·a_i terms)
                double_idx = 2 * np.arange(n)
                mask = double_idx < conv.size
                grad[mask] -= w[double_idx[mask]] * a[mask]
            else:
                # Hard‑max sub‑gradient using a few top entries
                top_k = 7
                idxs = np.argpartition(-conv, top_k - 1)[:top_k]
                grad = np.zeros_like(a)
                i = np.arange(n)
                for k in idxs:
                    j = k - i
                    mask = (j >= 0) & (j < n)
                    grad[mask] += 2.0 * a[j[mask]]
                    diag = mask & (i == j)
                    grad[diag] -= a[i[diag]]

            # ---------------------- Adam step -------------------------
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = eta0 * m_hat / (np.sqrt(v_hat) + eps)

            a = a - step
            a = _project_to_simplex(a, target_sum)

            # Occasional random perturbation to escape plateaus
            if it % 5000 == 0:
                a += np.random.randn(n) * target_sum * 5e-5
                a = _project_to_simplex(a, target_sum)

            # Cool the temperature
            mu = max(mu * mu_decay, mu_min)

            # ----------- restart if the schedule says so -------------
            if time.time() > next_restart and restarts_done < max_restarts:
                restarts_done += 1
                print(f"[n={n}] restart #{restarts_done} at it={it}")
                a = np.random.rand(n)
                a = _project_to_simplex(a, target_sum)
                m.fill(0.0)
                v.fill(0.0)
                mu = np.max(np.convolve(a, a))
                next_restart = time.time() + restart_interval

            # Simple stagnation detection – soft restart
            if no_improve > 200_000:
                print(f"[n={n}] stagnation at it={it}, soft restart")
                a = np.random.rand(n)
                a = _project_to_simplex(a, target_sum)
                m.fill(0.0)
                v.fill(0.0)
                mu = np.max(np.convolve(a, a))
                no_improve = 0

        # --------------------- cheap block search --------------------
        block_trials = 400
        for _ in range(block_trials):
            if time.time() >= deadline:
                break
            nb = np.random.randint(2, 15)               # 2‑14 blocks
            cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = np.random.rand(nb)
            cand = np.zeros(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, target_sum)
            cur_max = np.max(np.convolve(cand, cand))
            if cur_max < best_score:
                best_score = float(cur_max)
                best_seq = cand.copy()
                print(f"[n={n} block] improved to {best_score:.6f}")

    # ------------------- fallback (should never trigger) -----------
    if best_seq is None:
        n = 2000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)

    result = [float(x) for x in best_seq]
    print(f"[final] best score = {best_score:.6f} using length {len(result)}")
    return result
