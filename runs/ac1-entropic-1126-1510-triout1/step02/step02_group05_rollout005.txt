def project_simplex(v, s):
    # s target sum
    # ensure non-negative
    # using Duchi et al.
    if v.sum() == s and np.all(v >= 0):
        return v
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def simplex_projection(v, s):
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find rho
    rho_arr = u - cssv / (np.arange(1, n+1))

    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Gradient‑Descent on the peak of the self‑convolution.

The evaluation is   score = 2·n·max(a⋆a) / (sum(a))² .
Because it is scale‑invariant we fix   sum(a) = n   (average = 1) and
minimise the peak of the convolution.  Starting from the best known
construction (``height_sequence_1`` when available) we repeatedly

1. compute the convolution by FFT,
2. take the few largest convolution entries,
3. form the gradient of their contributions (each term contributes
   2·a[k‑j] to the gradient at position j),
4. move a short step opposite to that gradient,
5. project back onto the probability simplex Σa = n, a≥0,
6. accept the move if the peak decreased, otherwise shrink the step.

When the gradient phase stalls we perform a long simulated‑annealing
polish consisting of random pairwise mass swaps that keep the total sum
exactly n.  The best sequence seen is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: simplex projection (Duchi et al.)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # find the largest rho satisfying u[rho] > cssv[rho] / rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1  # zero‑based index
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # numerical safety
    w[w < 0] = 0.0
    return w

# ----------------------------------------------------------------------
# Helper: convolution (self‑convolution) using FFT – returns full vector
# ----------------------------------------------------------------------
def _self_convolution(seq: np.ndarray) -> np.ndarray:
    """Return np.convolve(seq, seq) using an FFT of length power‑of‑2."""
    n = seq.shape[0]
    # length for zero‑padding: next power of two ≥ 2n‑1
    m = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, m)
    conv_full = np.fft.irfft(fft_seq * fft_seq, m)[: 2 * n - 1]
    return conv_full

# ----------------------------------------------------------------------
# Helper: compute peak and its index
# ----------------------------------------------------------------------
def _max_conv(seq: np.ndarray) -> tuple[float, int, np.ndarray]:
    """Return (peak value, index of peak, full convolution)."""
    conv = _self_convolution(seq)
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv

# ----------------------------------------------------------------------
# Helper: gradient of the (soft) maximum of the convolution.
# We sum the contributions of the top K convolution entries.
# ----------------------------------------------------------------------
def _gradient_of_peak(seq: np.ndarray, conv: np.ndarray, top_k: int = 5) -> np.ndarray:
    n = seq.shape[0]
    g = np.zeros_like(seq)
    # indices of the K largest convolution entries
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(conv, -top_k)[-top_k:]
    for idx in idxs:
        # overlapping region for this shift
        start_j = max(0, idx - (n - 1))
        end_j = min(n - 1, idx)
        js = np.arange(start_j, end_j + 1)
        # contribution 2 * a[idx - j]
        g[js] += 2.0 * seq[idx - js]
    return g

# ----------------------------------------------------------------------
# Helper: cheap random pairwise mass move that preserves Σa = n.
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Move a small amount of mass (drawn from Uniform[-σ,σ]) between
    two distinct positions while keeping the sum unchanged."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)
    delta = (np.random.rand() - 0.5) * sigma  # centred uniform
    if delta > 0:
        delta = min(delta, seq[j])   # take from j
    else:
        delta = -min(-delta, seq[i])  # take from i
    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # give ourselves a small safety margin
    deadline = start_time + max(0.0, budget_s - 2.0)

    # --------------------------------------------------------------
    # Initialise a starting sequence.
    # --------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
        if base.ndim != 1 or base.size == 0:
            raise ValueError
        n = base.shape[0]
    except Exception:
        # fallback: a modest uniform vector
        n = 8000
        base = np.full(n, 1.0, dtype=float)   # sum = n

    # enforce non‑negativity and normalise Σa = n (average = 1)
    base = np.maximum(base, 0.0)
    total = base.sum()
    if total <= 0:
        base = np.full(n, 1.0, dtype=float)
        total = n
    base = base * (n / total)   # now Σbase = n

    # --------------------------------------------------------------
    # Variables for tracking the best solution
    # --------------------------------------------------------------
    a = base.copy()
    best_seq = a.copy()
    max_val, max_idx, conv = _max_conv(a)
    best_score = 2.0 * max_val / n   # because Σa = n
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Gradient‑descent phase on the peak of the self‑convolution
    # --------------------------------------------------------------
    lr = 0.6               # initial step length
    min_lr = 1e-6
    max_gd_iters = 20000   # generous upper bound
    top_k = 5              # number of large convolution entries used for gradient
    gd_iter = 0
    no_improve_counter = 0

    while time.time() < deadline and gd_iter < max_gd_iters:
        gd_iter += 1

        # gradient of the current peak (sum of top‑K contributions)
        g = _gradient_of_peak(a, conv, top_k=top_k)
        # direction that reduces the peak
        direction = -g

        # line‑search: try diminishing step sizes
        step = lr
        accepted = False
        while step > min_lr:
            cand = a + step * direction
            # ensure non‑negativity before projection
            cand = np.maximum(cand, 0.0)
            # project back onto the simplex Σx = n
            cand = _project_onto_simplex(cand, n)

            # recompute convolution for the candidate
            cand_max, cand_idx, cand_conv = _max_conv(cand)

            # improvement?
            if cand_max < max_val - 1e-12:
                # accept the move
                a = cand
                conv = cand_conv
                max_val = cand_max
                max_idx = cand_idx
                accepted = True
                # adapt learning rate upward a little
                lr = min(lr * 1.05, 1.0)
                break
            else:
                step *= 0.5   # shrink step and retry

        if not accepted:
            # no improvement – shrink step size more aggressively
            lr = max(lr * 0.5, min_lr)
            no_improve_counter += 1
        else:
            no_improve_counter = 0

        # keep track of the globally best sequence
        cur_score = 2.0 * max_val / n
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[gd iter {gd_iter:04d}] new best score: {best_score:.6f}")

        # early termination if we have stalled for many iterations
        if no_improve_counter >= 2000:
            break

    # --------------------------------------------------------------
    # Polishing phase – simulated‑annealing style random pair moves
    # --------------------------------------------------------------
    polish_iters = 50000
    it = 0
    while time.time() < deadline and it < polish_iters:
        it += 1
        # linear cooling of the move magnitude
        t_frac = it / polish_iters
        sigma = 0.3 * (1.0 - t_frac) + 0.01 * t_frac   # from 0.3 down to 0.01

        cand = _random_pair_move(best_seq, sigma)
        # quick score from the cached peak (recompute only the peak)
        cand_max, _, _ = _max_conv(cand)
        cand_score = 2.0 * cand_max / n

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            print(f"[polish {it:05d}] new best score: {best_score:.6f}")

    # --------------------------------------------------------------
    # Final safety clamp and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)   # enforce non‑negative
    # The evaluation function also caps values at 1000 – we stay far below that
    return best_seq.tolist()
