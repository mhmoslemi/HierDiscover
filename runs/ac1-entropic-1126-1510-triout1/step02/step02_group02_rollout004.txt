def _clip_and_rescale(a, target_sum):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 0:
        a = a * (target_sum / s)
    else:
        # all zeros; set uniform
        a = np.full_like(a, target_sum / a.size)
    return a

   (inverse‑sqrt, power‑law, exponential, cosine, mixtures and random

   {a ≥ 0, Σa = √(2 n)} and forced to be symmetric (a[i] = a[n‑1‑i]).

    _consider(np.ones(n))                       # uniform baseline

    for _ in range(120):                        # inverse‑sqrt
        beta = np.random.rand() * 10.0
        a = 1.0 / np.sqrt(np.arange(n) + beta + 1.0)
        _consider(a)

    for _ in range(120):                        # double‑ended inverse‑sqrt
        beta = np.random.rand() * 10.0
        i = np.arange(n)
        a = (1.0 / np.sqrt(i + beta + 1.0) +
             1.0 / np.sqrt((n - 1 - i) + beta + 1.0))
        _consider(a)

    for _ in range(80):                         # power‑law
        beta = np.random.rand() * 10.0
        pexp = np.random.rand() * 1.7 + 0.3
        a = 1.0 / (np.arange(n) + beta + 1.0) ** pexp
        _consider(a)

    for _ in range(80):                         # exponential decay
        decay = np.random.rand() * 0.01 + 0.0005
        a = np.exp(-decay * np.arange(n))
        _consider(a)

    for _ in range(80):                         # cosine‑based
        p = np.random.rand() * 5.0 + 0.5
        a = np.abs(np.cos(np.pi * np.arange(n) / (n - 1))) ** p
        _consider(a)

    for _ in range(80):                         # mixture uniform + inverse‑sqrt
        beta = np.random.rand() * 10.0
        lam = np.random.rand()
        a = lam * np.ones(n) + (1.0 - lam) * (1.0 /
                np.sqrt(np.arange(n) + beta + 1.0))
        _consider(a)

    for _ in range(300):                        # random block candidates
        nb = np.random.randint(2, 7)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.zeros(n)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        _consider(cand)

    if have_ref:
        _consider(np.array(height_sequence_1, dtype=float))

    print(f"[shape search] best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Adam optimisation with soft‑max smoothing
    a = best_a.copy()
    conv = _self_convolution(a, size_fft)
    T = max(1e-4, (conv.max() - conv.min()) * 0.5)   # initial temperature

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = target_sum * 0.8 / n                       # base step size
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    no_improve = 0

    while time.time() < deadline and it < 4_000_000:
        conv = _self_convolution(a, size_fft)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[{it:07d}] new best = {best_score:.6f}")
        else:
            no_improve += 1

        # soft‑max weights (smooth approximation of max)
        max_c = conv.max()
        shifted = conv - max_c
        if T < 1e-10:
            T = 1e-10
        exp_arg = shifted / T
        exp_arg = np.clip(exp_arg, -50, 0)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
            w_sum = 1.0

        # gradient of the smoothed max (normalised soft‑max)
        grad = _weighted_gradient(a, w, size_fft) / w_sum

        # enforce symmetry on the gradient to keep a symmetric
        grad = (grad + grad[::-1]) * 0.5

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _clip_and_rescale(a, target_sum)
        a = (a + a[::-1]) * 0.5                     # re‑impose symmetry

        # occasional perturbation to avoid stagnation
        if (it % 5000 == 0) or (no_improve > 5000):
            noise = np.random.randn(n) * target_sum * 1e-3
            a = a + noise
            a = _clip_and_rescale(a, target_sum)
            a = (a + a[::-1]) * 0.5
            no_improve = 0
            print(f"[{it:07d}] random perturbation")

        # anneal temperature very slowly
        T *= 0.9999

    print(f"[adam] best score after optimisation = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Final light random‑block refinement
    for _ in range(400):
        nb = np.random.randint(2, 7)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.zeros(n)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _clip_and_rescale(cand, target_sum)
        cand = (cand + cand[::-1]) * 0.5
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[block] improved to {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
