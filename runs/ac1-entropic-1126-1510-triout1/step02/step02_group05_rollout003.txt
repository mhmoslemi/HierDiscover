def project_simplex(v):
    # ensure sum to 1, all entries >=0
    # Duchi algorithm:
    u = np.sort(v)[::-1]
    sv = np.cumsum(u)
    rho = np.where(u > (sv - 1) / np.arange(1, len(v)+1))[0][-1]
    theta = (sv[rho] - 1) / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

def move_reduce_peak(seq, k_max):
    n = len(seq)
    # compute contributions to max shift
    partner = k_max - np.arange(n)
    mask = (partner >= 0) & (partner < n)
    contributions = np.zeros_like(seq)
    contributions[mask] = seq[mask] * seq[partner[mask]]
    # find index with largest contribution
    i = int(contributions.argmax())
    j = int(k_max - i) if 0 <= k_max - i < n else None

"""
Propose a step‑function sequence that minimises the objective
``2 n · max(f*f) / (∑f)²``.
The algorithm starts from the best known construction (global
``height_sequence_1`` when available) and then performs a long
simulated‑annealing / sub‑gradient search:

*   Each iteration evaluates the (linear‑time) autocorrelation via FFT.
*   The shift k* with the largest autocorrelation is located.
*   A *targeted* move reduces the contribution of the pair(s) that
    create this peak by moving a small fraction of their mass to random
    positions.
*   With probability 0.3 a simple random‑pair mass‑swap is tried instead
    (exploration).
*   Moves are accepted if they improve the fast score
    ``score = 2·n·max(conv)`` (the sum is normalised to 1).  Otherwise
    they are accepted according to a decreasing temperature (simulated
    annealing).
*   The best sequence seen so far is continually recorded.  When time
    runs out a short polishing phase (only random pair swaps) refines
    the result.

All heavy work uses FFT‑based convolution, which is O(n log n) and fast
for the typical length (~7 k).  The routine never calls the slow
``evaluate_sequence`` inside the main loop – it recomputes the objective
directly, guaranteeing that the 1000 s budget is respected.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast linear‑time convolution using FFT (real input)
# ----------------------------------------------------------------------
def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """Linear convolution of *seq* with itself via FFT (real‑valued)."""
    n = seq.shape[0]
    m = 2 * n - 1
    # next power of two for efficient FFT
    size = 1 << ((m - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[:m]
    return conv


# ----------------------------------------------------------------------
# Helper: compute current score = 2·n·max(conv)  (sum of seq is 1)
# ----------------------------------------------------------------------
def _score_from_conv(conv: np.ndarray, n: int, total: float) -> float:
    """Return the evaluation score given the convolution and length."""
    max_b = conv.max()
    # ``evaluate_sequence`` multiplies by 2·n and divides by (sum)².
    # The sum may drift slightly after numerical steps; we keep it in the formula.
    return 2.0 * n * max_b / (total * total)


# ----------------------------------------------------------------------
# Helper: random pairwise mass move (preserves ℓ₁‑norm)
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Move a small amount of mass between two randomly chosen indices."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    # symmetric uniform proposal
    delta = (np.random.rand() - 0.5) * sigma

    if delta > 0:
        delta = min(delta, seq[j])          # take from j -> i
    else:
        delta = -min(-delta, seq[i])        # take from i -> j

    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# Helper: targeted move that reduces the contribution of the largest peak
# ----------------------------------------------------------------------
def _targeted_move(seq: np.ndarray, k_max: int) -> np.ndarray:
    """
    Identify the index pair that contributes most to the convolution at
    ``k_max`` and transfer a fraction of its mass to two random other
    positions.  The sequence is assumed to have a positive sum (normally 1)
    and the result is renormalised to the original sum.
    """
    n = seq.shape[0]
    idx = np.arange(n)
    partner = k_max - idx
    mask = (partner >= 0) & (partner < n)

    # contribution of each *i* to the peak (i supplies a_i * a_{k_max-i})
    contrib = np.zeros_like(seq)
    contrib[mask] = seq[idx[mask]] * seq[partner[mask]]

    # pick the index with the largest contribution
    i = int(contrib.argmax())
    j = k_max - i
    if not (0 <= j < n):
        j = None

    # fraction of mass that will be moved; adaptively small to stay stable
    frac = 0.08
    if j is not None:
        delta = frac * min(seq[i], seq[j])
    else:
        delta = frac * seq[i]

    if delta <= 0:
        return seq  # nothing to move

    # choose random target indices distinct from the source(s)
    candidates = np.arange(n)
    forbid = np.full(n, True, dtype=bool)
    forbid[i] = False
    if j is not None:
        forbid[j] = False
    free = candidates[forbid]

    if free.size == 0:
        return seq

    t1 = np.random.choice(free)
    new_seq = seq.copy()
    new_seq[i] -= delta
    new_seq[t1] += delta

    if j is not None:
        # pick a second target different from t1 and the sources
        forbid[t1] = False
        free2 = candidates[forbid]
        if free2.size == 0:
            # fallback – give the mass back to t1
            new_seq[t1] += delta
        else:
            t2 = np.random.choice(free2)
            new_seq[j] -= delta
            new_seq[t2] += delta

    # renormalise to keep the total mass unchanged (helps numerical stability)
    total = new_seq.sum()
    if total > 0:
        new_seq /= total
    return new_seq

    # Return the best sequence found (as a plain Python list).
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    return best_seq.tolist()
