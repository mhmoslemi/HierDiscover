---------

# ----------------------------------------------------------------------
# Projection onto the probability simplex {x ≥ 0, sum(x)=s}
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    # Duchi et al. (2008) – sort descending and find rho
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Discrete arcsine distribution (probability vector)
def _arcsine_distribution(n: int) -> np.ndarray:
    """Return a length‑n arcsine‑like distribution (sums to 1)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Fast self‑convolution using real FFT (full length 2·n‑1)
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Return a * a (full convolution) using an FFT of size power‑of‑2."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]

                best_local_seq = a.copy()
                print(f"[{time.time():.1f}] restart={restart} iter={t} "
                      f"score={best_local_score:.6f}")

            # ---- occasional tiny random jitter (helps escape plateaux) ----
            if t % 500 == 0 and np.random.rand() < jitter_prob:
                a += np.random.randn(n) * jitter_scale
                a = _project_onto_simplex(a, s=1.0)

        # ----------------------------------------------------------------
        # Final precise evaluation of the best vector from this run
        try:
            final_score = evaluate_sequence(best_local_seq.tolist())
        except Exception:
            # fallback to the cheap internal estimate if the black‑box fails
            final_score = best_local_score

        if final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = best_local_seq.copy()
            print(f"*** New global best after restart {restart}: "
                  f"score={overall_best_score:.6f}")

        # Early exit if we have already beaten the target threshold
        if overall_best_score <= 1.503:
            break

        # If time is almost exhausted, stop spawning new restarts
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Fallback – should never happen, but guarantees a return value.
    if overall_best_seq is None:
        overall_best_seq = np.full(1000, 1.0 / 1000, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())

    print(f"Final best score: {overall_best_score:.6f} "
          f"(length = {len(overall_best_seq)})")
    return overall_best_seq.tolist()
