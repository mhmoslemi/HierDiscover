Gradient‑guided local search for the step‑function evaluation problem.

The algorithm proceeds as follows:
1. Initialise a feasible sequence (use the known best construction if
   available, otherwise a uniform vector) and normalise it to L1‑norm 1.
2. Optionally run a few LP refinement steps (the same LP used in the
   reference solution) to obtain a good starting point.
3. Perform a long inexpensive local search:
   * The current convolution f*f is maintained and updated incrementally
     after each move (O(n) per move, no full recomputation).
   * In each iteration the index kₘₐₓ where the convolution attains its
     maximum is identified.  The gradient of this maximal term w.r.t.
     the coefficients is g_i = f_i + f_{kₘₐₓ‑i} (if the shifted index
     exists).  Moving mass from indices with large g_i⋅f_i to indices

   * The convolution and the score are updated via a cheap O(n) formula.
   * Improvements are kept; occasional random pair moves and periodic
     full recomputation of the convolution avoid drift.
4. The best sequence seen before the time budget expires is returned.

The incremental update makes it possible to run tens of thousands of
iterations within the 1000‑second budget, which typically yields a

# Helper: cheap random pairwise move (used for occasional exploration).

def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)
    delta = (np.random.rand() - 0.5) * sigma
    if delta > 0:
        delta = min(delta, seq[j])
    else:
        delta = -min(-delta, seq[i])
    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# Linear‑programming step: find a vector g with maximal L1 mass under the
# same autocorrelation bound M of the current f.  Returns g or None.
# ----------------------------------------------------------------------
def _lp_max_mass(f: np.ndarray, M: float):
    if not _HAS_CVX:
        return None
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)
    conv_fg = cp.conv(cp.Constant(f), g)        # length 2n‑1
    constraints = [conv_fg <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                    max_iters=500, verbose=False)
    except Exception:                           # pragma: no cover
        return None
    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    if g.value is None:
        return None
    return np.asarray(g.value, dtype=float)


# ----------------------------------------------------------------------
# Incremental convolution update when moving mass δ from i→j.
# ----------------------------------------------------------------------
def _apply_move(seq: np.ndarray, conv: np.ndarray, i: int, j: int,
                delta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns (seq', conv') after transferring 'delta' from position i to j.
    The update follows:
        conv' = conv + 2·δ·(f_{·‑j} – f_{·‑i}) + δ²·(e_{2i}+e_{2j}–2e_{i+j})
    where e_k denotes a unit impulse at index k.
    """
    n = seq.shape[0]
    new_seq = seq.copy()
    new_seq[i] -= delta
    new_seq[j] += delta

    new_conv = conv.copy()
    # slice updates for the linear part 2·δ·(f_{k‑j} – f_{k‑i})
    new_conv[i:i + n] -= 2.0 * delta * seq
    new_conv[j:j + n] += 2.0 * delta * seq
    # quadratic part δ²
    new_conv[2 * i] += delta * delta
    new_conv[2 * j] += delta * delta
    new_conv[i + j] -= 2.0 * delta * delta
    return new_seq, new_conv


# ----------------------------------------------------------------------
# Compute the gradient of the maximal autocorrelation term.
# ----------------------------------------------------------------------
def _gradient_at_max(seq: np.ndarray, kmax: int) -> np.ndarray:
    """Returns g_i = f_i + f_{kmax‑i} (zero if the shifted index is out of range)."""
    n = seq.shape[0]
    grad = seq.copy()
    i_start = max(0, kmax - (n - 1))
    i_end = min(kmax, n - 1)
    if i_start <= i_end:
        idx = np.arange(i_start, i_end + 1)
        grad[idx] += seq[kmax - idx]
    return grad

    except Exception:                                     # pragma: no cover
        # modest uniform start; length around 2500 gives a decent trade‑off.
        base = np.full(2500, 1.0 / 2500.0, dtype=float)

    base = np.maximum(base, 0.0)
    base_sum = base.sum()
    if base_sum == 0:
        base = np.full_like(base, 1.0 / base.size)
        base_sum = 1.0
    f = base / base_sum                     # normalise to L1‑norm 1
    n = f.shape[0]

    # --------------------------------------------------------------
    # 2. (Optional) Run a few LP refinement steps to get a better start.
    # --------------------------------------------------------------
    if _HAS_CVX:
        # a small number of LP iterations (fast, cheap) – adapt mixing.
        mix_coef = 0.5
        for lp_iter in range(10):
            if time.time() >= deadline:
                break
            conv_ff = np.convolve(f, f)
            M = conv_ff.max()
            g_raw = _lp_max_mass(f, M)
            if g_raw is None:
                break
            sum_g = g_raw.sum()
            if sum_g <= 1.0 + 1e-12:          # stationary point reached
                break
            g_norm = g_raw / sum_g
            f_candidate = (1.0 - mix_coef) * f + mix_coef * g_norm
            # evaluate (cheap formula – no full convolution needed)
            conv_cand = np.convolve(f_candidate, f_candidate)
            cand_score = 2.0 * n * conv_cand.max() / (f_candidate.sum() ** 2)
            cur_score = 2.0 * n * M / (f.sum() ** 2)
            if cand_score < cur_score:
                f = f_candidate
                mix_coef = min(0.9, mix_coef * 1.05)
            else:
                mix_coef = max(0.1, mix_coef * 0.5)

    # --------------------------------------------------------------
    # 3. Prepare data structures for the incremental search.
    # --------------------------------------------------------------
    conv = np.convolve(f, f)
    M = conv.max()
    best_seq = f.copy()
    best_score = 2.0 * n * M / (f.sum() ** 2)
    best_it = 0

    # configuration of the search
    max_iters = 200_000                     # hard limit; actual stop = deadline
    it = 0
    stagnate_counter = 0
    # parameters for the step size schedule
    base_step = 0.2
    min_step = 0.005
    # probability of using a pure random pair move (for exploration)
    random_move_prob = 0.08
    # full recompute interval (prevent drift)
    recompute_interval = 5000

    while time.time() < deadline and it < max_iters:
        it += 1

        # every few thousand iterations recompute the convolution from scratch
        if it % recompute_interval == 0:
            conv = np.convolve(f, f)
            M = conv.max()
            # note: we keep the current f unchanged – only conv refreshed

        # --------------------------------------------------------------
        # Determine the dominant index (where max autocorrelation occurs)
        # --------------------------------------------------------------
        M = conv.max()
        max_idxs = np.flatnonzero(conv == M)
        kmax = np.random.choice(max_idxs)           # break ties randomly

        # --------------------------------------------------------------
        # Gradient‑guided move (majority of the time)
        # --------------------------------------------------------------
        if np.random.rand() > random_move_prob:
            grad = _gradient_at_max(f, kmax)

            # source weights: favour positions that both carry mass and have large gradient
            src_weights = f * grad
            src_total = src_weights.sum()
            if src_total == 0:
                src_weights = f + 1e-15
                src_total = src_weights.sum()
            src = np.random.choice(n, p=src_weights / src_total)

            # destination weights: prefer low‑gradient locations
            eps = 1e-12
            dest_weights = 1.0 / (grad + eps)
            dest_weights[src] = 0.0                     # avoid self‑move
            dest_total = dest_weights.sum()
            if dest_total == 0:
                # fall back to a random destination
                dest = (src + 1) % n
            else:
                dest = np.random.choice(n, p=dest_weights / dest_total)

            # step size – decays linearly with iteration count
            step_frac = max(base_step * (1.0 - it / max_iters) + min_step, min_step)
            delta = step_frac * f[src]
            # protect against numerical zero
            if delta < 1e-15:
                continue

            # ----------------------------------------------------------
            # Incremental update
            # ----------------------------------------------------------
            new_f, new_conv = _apply_move(f, conv, src, dest, delta)

            # safeguard against negative entries caused by rounding
            new_f = np.maximum(new_f, 0.0)
            # (the L1‑sum is preserved up to rounding)
            new_M = new_conv.max()
            new_score = 2.0 * n * new_M / (new_f.sum() ** 2)

        else:
            # ----------------------------------------------------------
            # Pure random pair move (exploration)
            # ----------------------------------------------------------
            sigma = 0.2 * (1.0 - it / max_iters) + 0.02
            cand = _random_pair_move(f, sigma)
            new_conv = np.convolve(cand, cand)
            new_M = new_conv.max()
            new_score = 2.0 * n * new_M / (cand.sum() ** 2)
            new_f = cand

        # --------------------------------------------------------------
        # Accept or reject the move.
        # --------------------------------------------------------------
        if new_score < best_score:
            best_score = new_score
            best_seq = new_f.copy()
            best_it = it
            print(f"[Iter {it:06d}] New best score: {best_score:.6f}")

        if new_score < (2.0 * n * M / (f.sum() ** 2)):
            f = new_f
            conv = new_conv
            M = new_M
            stagnate_counter = 0
        else:
            stagnate_counter += 1

        # --------------------------------------------------------------
        # Occasionally restart from the best solution with a small perturbation
        # if we have been stagnant for a while.
        # --------------------------------------------------------------
        if stagnate_counter >= 10_000:
            # perturb a small random subset of entries
            pert = best_seq.copy()
            # add tiny Gaussian noise
            noise = np.random.normal(scale=0.01, size=n)
            pert += noise
            pert = np.maximum(pert, 0.0)
            pert /= max(pert.sum(), 1e-12)      # renormalise
            f = pert
            conv = np.convolve(f, f)
            M = conv.max()
            stagnate_counter = 0
            print(f"[Iter {it:06d}] Restarted with small perturbation.")

    # --------------------------------------------------------------
    # Return the best sequence found.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    # final safety normalisation (evaluate_sequence will re‑scale anyway)
    best_seq /= max(best_seq.sum(), 1e-12)
    return best_seq.tolist()
