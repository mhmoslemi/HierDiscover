"""
Propose a step‑function sequence whose self‑convolution peak is as small as
possible.  

The algorithm consists of three stages:

1. **Initial candidates** – evaluate several arcsine‑type sequences of
   different lengths, a few simple permutations of them and the previously
   discovered global sequence (if present).  The best of those initial
   candidates is kept.

2. **Exploratory search** – a simulated‑annealing loop perturbs the current
   sequence with Gaussian noise, projects the result onto the probability
   simplex and accepts the move according to a decreasing temperature schedule.
   Periodic deterministic mixes with a uniform distribution and with the
   reversed sequence are also tried.

3. **Polishing** – a soft‑max based gradient descent (the same idea that was
   used in the reference code) finishes the optimisation.

The whole search respects the global time budget (≈1000 s).  All numbers are
non‑negative and the final sequence is L1‑normalised, as required by the
`evaluate_sequence` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> list[float]:
    """
    Return the arcsine coefficients C_i = binom(2i,i)/4^i for i=0..n‑1,
    normalised so that their sum equals 1.
    """
    seq = []
    cur = 1.0
    for i in range(n):
        seq.append(cur)
        if i + 1 < n:
            cur = cur * (2.0 * i + 1.0) / (2.0 * (i + 1))
    total = sum(seq)
    seq = [x / total for x in seq]
    return seq


# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x ≥ 0, Σ x = s}.  Implements the O(n log n) algorithm of
    Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("target sum s must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    cur_score = best_score
    temperature = 1.0
    sigma = 0.02                     # std‑dev of the Gaussian perturbation
    decay_temp = 0.9995
    decay_sigma = 0.9999
    it = 0

    while time.time() < deadline:
        it += 1
        # random Gaussian perturbation + projection onto the simplex
        cand = project_simplex(cur + sigma * np.random.randn(n))
        cand_score = eval_seq(cand)

        # keep the globally best solution
        if cand_score < best_score:
            best_score, best_seq = cand_score, cand.copy()
            print(f"[SA] it={it}  new best {cand_score:.6f}  "
                  f"temp={temperature:.2e}  sigma={sigma:.5f}")

        # Metropolis acceptance for the *current* state
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
        else:
            if temperature > 1e-12:
                prob = np.exp(-(cand_score - cur_score) / temperature)
                if np.random.rand() < prob:
                    cur, cur_score = cand, cand_score

        # annealing schedules
        temperature *= decay_temp
        sigma *= decay_sigma

    # ------------------------------------------------------------------
    # 5)  Gradient‑descent polishing (soft‑max of the convolution)
    # ------------------------------------------------------------------
    beta = 5.0            # soft‑max temperature
    grad_step = 0.12
    beta_inc = 1.02

    while time.time() < deadline:
        conv = np.convolve(best_seq, best_seq)
        max_val = conv.max()
        # soft‑max weights that concentrate on the indices attaining the max
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / exp_vals.sum()
        # gradient of the smoothed max (see the reference code)
        grad = 2.0 * np.convolve(w, best_seq[::-1])[n - 1:2 * n - 1]
        cand = project_simplex(best_seq - grad_step * grad)
        cand_score = eval_seq(cand)
        if cand_score < best_score:
            best_score, best_seq = cand_score, cand.copy()
            print(f"[grad] improved to {cand_score:.6f}")
        grad_step *= 0.995
        beta = min(200.0, beta * beta_inc)

    # ------------------------------------------------------------------
    # Final sanity checks and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = best_seq / best_seq.sum()
    print(f"Best sequence length = {len(best_seq)}  score = {best_score:.6f}")
    return best_seq.tolist()
