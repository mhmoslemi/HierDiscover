def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2

    # Determine n: use existing height_sequence_1 length else set to 8000
    try:
        base = np.array(height_sequence_1, dtype=float)
        n = len(base)
        # Normalize to sum 1
        base = np.maximum(base, 0)
        base_sum = base.sum()
        if base_sum == 0:
            base = np.full(n, 1.0/n)
        else:
            base = base / base_sum
    except Exception:
        n = 8000
        base = np.full(n, 1.0/n)
    # Ensure within bounds
    base = np.clip(base, 0, 1000)

    best_seq = base.copy()
    best_score = score(best_seq)

    #====================================================================
    # Stage 1: LP refinement with line search
    #====================================================================
    f = base.copy()
    current_score = best_score
    # compute M
    M = convolution_max(f)
    # parameters
    max_lp_iters = 80
    mix_candidates = [0.1,0.2,0.3,0.5,0.7,0.9]
    for lp_iter in range(max_lp_iters):
        if time.time() > deadline:
            break
        M = convolution_max(f)
        g_raw = _lp_max_mass(f, M)
        if g_raw is None:
            break
        sum_g = g_raw.sum()
        if sum_g <= 1 + 1e-12:
            # reached stationary
            break
        g_norm = g_raw / sum_g
        # line search over mixing coefficient
        best_local_seq = None
        best_local_score = float('inf')
        best_t = None
        for t in mix_candidates:
            cand = (1 - t) * f + t * g_norm
            cand = np.clip(cand, 0, 1000)
            cand = project_to_simplex(cand)   # ensure sum=1
            cand_score = score(cand)
            if cand_score < best_local_score:
                best_local_score = cand_score
                best_local_seq = cand
                best_t = t
        # accept best among candidates if improvement
        if best_local_score < current_score - 1e-12:
            f = best_local_seq
            current_score = best_local_score
            if current_score < best_score:
                best_score = current_score
                best_seq = f.copy()
                print(f"[LP {lp_iter+1}] Improved to {best_score:.6f}")
        else:
            # no improvement; break early
            break

    #====================================================================
    # Stage 2: Simulated annealing polishing
    #====================================================================
    # tuning
    total_iters = 20000
    it = 0
    # initial temperature based on current score
    T0 = best_score * 0.05  # ~5% of score maybe
    while time.time() < deadline and it < total_iters:
        it += 1
        frac = it / total_iters
        # temperature schedule, linear cooling
        T = T0 * (1 - frac)
        sigma = 0.3 * (1 - frac) + 0.02 * frac  # magnitude of move
        # propose move
        if np.random.rand() < 0.8:

        # clip

        # project to simplex (total sum 1)
        candidate = project_to_simplex(candidate)
        cand_score = score(candidate)
        delta = cand_score - best_score
        if delta < 0 or np.random.rand() < np.exp(-delta / T):

            best_seq = candidate

            # maybe print updates occasionally

    #====================================================================

    best_seq = np.maximum(best_seq, 0.0)
    # ensure sum non-zero
    if best_seq.sum() < 1e-12:

"""
Propose a non‑negative step‑function (list of heights) that aims to minimise

    2 * n * max(convolve(seq, seq)) / (sum(seq) ** 2)

within the allotted time.  The algorithm consists of three stages:

1. **Initialisation** – use the previously found best sequence if available,
   otherwise fall back to a uniform distribution and normalise to sum = 1.

2. **LP refinement** – repeatedly solve a linear program that, for the current
   sequence `f` and its maximal convolution value `M`, finds a new sequence
   `g` with the largest possible ℓ₁‑mass while respecting the same bound
   `f * g ≤ M`.  A short line‑search over several mixing coefficients
   `t∈[0,1]` produces a convex combination `(1‑t)·f + t·g` that improves the
   objective.  The process stops when no progress is possible or the time budget
   is exhausted.

3. **Simulated‑annealing polishing** – a long stochastic search that repeatedly
   perturbs the current best vector by moving a small amount of mass between
   entries (pair‑wise or symmetric pair‑wise moves) and accepts the move either
   if it improves the score or with a temperature‑dependent probability.
   The temperature is linearly cooled; the move size shrinks as the search
   proceeds.  The best sequence seen is returned.

All arithmetic is performed with NumPy; convolution maxima are obtained via
FFT for speed.  The final sequence is clipped to the allowed range and
normalised to sum‑one, guaranteeing compatibility with `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional CVXPY import – used for the LP refinement.
# ----------------------------------------------------------------------
try:
    import cvxpy as cp  # type: ignore
    _HAS_CVX = True
except Exception:  # pragma: no cover
    _HAS_CVX = False

# ----------------------------------------------------------------------
# Helper: fast convolution maximum (FFT based) – O(n log n).
# ----------------------------------------------------------------------
def _max_convolution(seq: np.ndarray) -> float:
    """Return max_{k} (seq * seq)[k] using an FFT based convolution."""
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    # real FFT is sufficient because the sequence is real.
    fa = np.fft.rfft(seq, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    # eliminate tiny negative round‑off errors
    conv = np.maximum(conv, 0.0)
    return float(conv.max())


# ----------------------------------------------------------------------
# Helper: evaluate the objective (identical to evaluate_sequence for
# non‑clipped, non‑negative vectors that sum to a positive value).
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    maxb = _max_convolution(seq)
    return float(2.0 * seq.shape[0] * maxb / (s * s))


# ----------------------------------------------------------------------
# Helper: project a non‑negative vector onto the simplex (sum = 1).
# Simple renormalisation is sufficient because we always keep the vector
# non‑negative.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / total


# ----------------------------------------------------------------------
# Random pairwise mass‑swap (keeps total mass exactly 1).
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    delta = (np.random.rand() - 0.5) * sigma
    # keep both entries non‑negative
    if delta > 0.0:
        delta = min(delta, seq[j])
    else:
        delta = -min(-delta, seq[i])

    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# Symmetric pair move targeting the current worst convolution index.
# It removes a small amount from a pair (i, j) with i + j = k_max and
# redistributes it elsewhere – this directly reduces the offending term.
# ----------------------------------------------------------------------
def _sym_pair_move(seq: np.ndarray, sigma: float, k_max: int) -> np.ndarray:
    n = seq.shape[0]
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    i = np.random.randint(lo, hi + 1)
    j = k_max - i

    # If both indices coincide, fall back to a simple pair move.
    if i == j:
        return _random_pair_move(seq, sigma)

    epsilon = np.random.rand() * sigma
    epsilon = min(epsilon, seq[i], seq[j])

    new_seq = seq.copy()
    new_seq[i] -= epsilon
    new_seq[j] -= epsilon

    # Choose two fresh locations for redistribution.
    candidates = [idx for idx in range(n) if idx not in (i, j)]
    dst1, dst2 = np.random.choice(candidates, size=2, replace=False)
    new_seq[dst1] += epsilon / 2.0
    new_seq[dst2] += epsilon / 2.0
    return new_seq


# ----------------------------------------------------------------------
# Linear‑programming step: for a fixed f and a bound M, maximise Σ g
# subject to (f * g) ≤ M and g ≥ 0.
# ----------------------------------------------------------------------
def _lp_max_mass(f: np.ndarray, M: float):
    """Return an optimal g (as a NumPy array) or None on failure."""
    if not _HAS_CVX:
        return None
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)

    # Convolution of the constant f with the variable g.
    conv_fg = cp.conv(cp.Constant(f), g)  # length 2n‑1
    constraints = [conv_fg <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=500, verbose=False)
    except Exception:  # pragma: no cover
        return None
    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    g_val = g.value
    if g_val is None:
        return None
    return np.asarray(g_val, dtype=float)

    start = time.time()

    # --------------------------------------------------------------

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    f = best_seq.copy()

    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
