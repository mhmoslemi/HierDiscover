Implementation of simplex projection:

```
def project_simplex(v):
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = len(v)
    u = np.sort(v)[::-1]   # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho: largest i where u_i - cssv_i / i > 0
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
propose_candidate
-----------------
A stochastic mirror‑descent optimiser that directly minimises the
peak of the self‑convolution.  The objective is smoothed with a
soft‑max (temperature α) so that the gradient is well‑behaved; α is
increased during the run (simulated‑annealing style).  Updates are
performed on the probability simplex by a projected Adam step.  The
algorithm works for any length n; we try a few lengths until the time
budget runs out and keep the best sequence found (L1‑norm = 1, all
entries non‑negative).  The final sequence is clipped to the limits
required by ``evaluate_sequence`` before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> list[float]:
    """
    Classical arcsine coefficients C_i = binom(2i,i)/4**i, normalised
    to have L1‑norm 1.
    """
    seq = []
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    seq = np.asarray(seq, dtype=np.float64) / total
    return seq.tolist()


# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i >= 0, sum x_i = 1}.  Implementation follows
    Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection works on 1‑D arrays")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the rho (largest i where u_i > cssv_i / i)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive → return the uniform vertex
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # numerical safety: renormalise exactly to 1
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(v, 1.0 / n)
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
def smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smooth‑max objective
        f_α(a) = (1/α) log Σ_k exp(α·(a∗a)_k)
    for a vector a (already normalised to sum 1).

    Returns a vector of length len(a) containing ∂f_α/∂a_i.
    """
    # soft‑max weights over convolution indices
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)         # safe, max(exp_scaled)=1
    w = exp_scaled / np.sum(exp_scaled)             # Σ w_k = 1

    # gradient = 2 * (a_rev ∗ w) restricted to the original length
    a_rev = a[::-1]
    conv2 = np.convolve(a_rev, w)                   # length = 3·n‑2
    n = a.shape[0]
    grad = 2.0 * conv2[n - 1:n - 1 + n]             # slice of length n
    return grad


# ----------------------------------------------------------------------
def raw_metric(a: np.ndarray) -> float:
    """
    Metric used in the original challenge for a normalised sequence:
        2·n·max(a∗a) .
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
def optimise_on_simplex(
    init_a: np.ndarray,
    deadline: float,
    alphas: list[float],
    max_iter_per_alpha: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Adam‑based mirror descent on the simplex, annealing the soft‑max
    temperature α according to the supplied schedule.  Returns the best
    sequence (as a NumPy array) and its raw metric.
    """
    a = init_a.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # learning‑rate (step‑size for the Adam update)
    lr = 0.5

    best_a = a.copy()
    best_score = raw_metric(a)

    # simple stagnation counter for occasional random jitter
    reject_streak = 0

    for alpha in alphas:
        # limit the number of iterations per temperature
        for _ in range(max_iter_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            t += 1

            # current convolution and metric
            conv = np.convolve(a, a)
            cur_score = 2.0 * n * np.max(conv)

            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                reject_streak = 0
                # occasional progress report
                if t % 200 == 0:
                    print(f"[n={n} α={alpha:.1f} iter={t}]  new best metric = {cur_score:.6f}")

            # compute smooth‑max gradient
            grad = smooth_gradient(a, conv, alpha)

            # Adam update (descent direction = -grad)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            cand = a - step
            cand = project_simplex(cand)

            cand_score = raw_metric(cand)

            if cand_score < cur_score:
                a = cand
                lr = min(lr * 1.1, 5.0)      # slightly enlarge step on success
                reject_streak = 0
                continue
            else:
                # reject – shrink step size
                lr = max(lr * 0.6, 1e-6)
                reject_streak += 1

                # every few rejections sprinkle a tiny random jitter to escape plateaux
                if reject_streak % 200 == 0:
                    i, j = np.random.choice(n, 2, replace=False)
                    delta = (np.random.rand() - 0.5) * 2e-3   # ≈ ±0.001
                    jitter = a.copy()
                    jitter[i] = max(0.0, jitter[i] + delta)
                    jitter[j] = max(0.0, jitter[j] - delta)
                    jitter = project_simplex(jitter)
                    if raw_metric(jitter) < cur_score:
                        a = jitter
                        lr = max(lr, 0.1)
                        reject_streak = 0

            # safety: if lr becomes absurdly tiny we abort the inner loop
            if lr < 1e-9:
                break

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑peak step‑function within the allotted time.
    The routine tries a handful of lengths, runs the simplex optimiser
    (with an annealing schedule on the soft‑max temperature) and keeps
    the best sequence discovered.  The returned list satisfies the
    non‑negative, bounded requirements of ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # tiny safety margin

    # lengths to investigate – longer vectors give a lower theoretical bound
    lengths = [2400, 2800, 3200, 3600, 4000]

    # annealing schedule for the smooth‑max temperature
    # start very smooth, finish very sharp (approaching the true max)
    alpha_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 30.0, 100.0, 300.0, 1000.0, 3000.0]

    best_seq = None
    best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # initialise – use the global best if it matches the required length,
        # otherwise fall back to the classic arcsine law
        if 'height_sequence_1' in globals():
            init = np.asarray(height_sequence_1, dtype=np.float64)
            if init.shape[0] != n:
                init = None
        else:
            init = None

        if init is None:
            init = np.asarray(arcsine_sequence(n), dtype=np.float64)
        else:
            init = init / init.sum()          # ensure exact L1‑norm = 1

        # give each length a share of the remaining time
        remaining = deadline - time.time()
        per_len_deadline = time.time() + max(5.0, remaining / (len(lengths) - lengths.index(n)))

        # run the optimiser
        seq, score = optimise_on_simplex(
            init_a=init,
            deadline=per_len_deadline,
            alphas=alpha_schedule,
            max_iter_per_alpha=4000,
        )

        print(f"Finished length {n}: best raw metric = {score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"*** New overall best (len={n}) score = {best_score:.6f}")

    # Fallback in the unlikely event that nothing was produced
    if best_seq is None:
        best_seq = np.asarray(arcsine_sequence(2000), dtype=np.float64)

    # Apply the final clipping required by ``evaluate_sequence``
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    print(f"\n--- FINAL RESULT ---\nBest length = {len(best_seq)} "
          f"with raw metric ≈ {best_score:.6f}")
    return best_seq
