```
def project_simplex(v):
    # Duchi et al.
    # returns w that sum=1, w>=0
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Fast construction of a near‑optimal step‑function.

The evaluation metric is
    2·n·max(conv(a,a)) / (sum(a))**2 .
For a non‑negative sequence with L1‑norm 1 this reduces to
    2·n·max(conv(a,a)) .

The continuous arcsine density 
   f(x)=1/(π√{x(1−x)}) ,  x∈(0,1)
minimises the supremum of its self‑convolution.
Its discretisation on {0,…,n‑1} gives the well‑known “arcsine
coefficients’’ C_i = binom(2i,i)/4^i.  After normalising so that
∑C_i=1 the metric converges to π/2≈1.570796 as n→∞.
Consequently, simply using a *large* value of n yields a strictly
smaller score than the 3400‑point construction (1.570912).

This implementation:
  1. Chooses a generous length (default = 50 000 – well within the
     1000 s budget; the convolution is performed by FFT internally).
  2. Builds the arcsine sequence in O(n) time and normalises it.
  3. (Optionally) performs a tiny mirror‑descent refinement while
     time permits – this never hurts the score.
  4. Returns the final sequence as a plain Python list of floats.

The returned sequence is guaranteed to satisfy all checks performed
by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> list[float]:
    """
    Return the first ``n`` coefficients of (1‑z)^‑1/2,
    i.e. C_i = binom(2*i, i) / 4**i, normalised to have sum = 1.
    The recurrence  C_{i+1}=C_i·(2i+1)/(2(i+1))  is used for stability.
    """
    seq = np.empty(n, dtype=np.float64)
    cur = 1.0               # C_0
    total = 0.0
    for i in range(n):
        seq[i] = cur
        total += cur
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    seq /= total            # normalise L1‑norm to 1
    return seq.tolist()


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum=1)."""
    # Duchi et al. (2008) – O(n log n) due to sorting.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _mirror_descent_step(a: np.ndarray, eta: float) -> np.ndarray:
    """
    A single (sub‑)gradient step for the *max* of the self‑convolution.
    The gradient is 2·a_{k*‑i} where k* is a maximising index.
    The update is multiplicative (mirror‑descent) and the result is
    re‑normalised to have unit L1‑norm.
    """
    conv = np.convolve(a, a)                 # length 2·n−1
    k_max = int(conv.argmax())
    # partner index for each i: k_max‑i (must lie in [0,n‑1])
    idx = np.arange(len(a))
    partner = k_max - idx
    mask = (partner >= 0) & (partner < len(a))
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[partner[mask]]
    # multiplicative update – keep exponents bounded
    upd = np.exp(-eta * grad)
    cand = a * upd
    cand_sum = cand.sum()
    if cand_sum <= 0.0:
        # numerical safety – fall back to uniform
        cand = np.full_like(a, 1.0 / len(a))
    else:
        cand /= cand_sum
    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Generate a high‑quality step‑function sequence.

    The algorithm builds a large‑scale discrete arcsine distribution,
    which asymptotically attains the theoretical lower bound π/2.
    A brief mirror‑descent polishing phase is performed if time permits.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # small safety margin

    # ------------------------------------------------------------------
    # 1) Choose a length large enough to make the discretisation error
    #    negligible while staying comfortably fast.  50 000 works well.
    #    If the remaining time is tiny we fall back to a modest size.
    preferred_n = 50000
    remaining = deadline - time.time()
    if remaining < 5.0:          # not enough time – use a tiny fallback
        n = 2000
    else:
        n = max(2000, min(preferred_n, int(remaining * 200)))  # rough scaling

    print(f"\n=== Building arcsine sequence of length {n} ===")
    a = np.array(arcsine_sequence(n), dtype=np.float64)

    # ------------------------------------------------------------------
    # 2) Optional short polishing phase (mirror‑descent) – it never
    #    worsens the metric and can squeeze a few 1e‑6 improvements.
    #    The phase is time‑bound; we stop as soon as the deadline is
    #    reached.
    eta = 0.5          # initial step size for the multiplicative update
    iteration = 0
    best_metric = 2.0 * n * np.convolve(a, a).max()
    best_a = a.copy()

    while time.time() < deadline:
        # perform a single mirror‑descent step
        a_candidate = _mirror_descent_step(a, eta)

        cur_metric = 2.0 * n * np.convolve(a_candidate, a_candidate).max()
        if cur_metric < best_metric - 1e-12:
            best_metric = cur_metric
            best_a = a_candidate.copy()
            print(f"[iter {iteration:6d}] new best metric {best_metric:.8f}")
            # success → slightly increase step size
            eta = min(eta * 1.1, 5.0)
            a = a_candidate
        else:
            # no improvement – shrink step
            eta *= 0.6
            # occasional random jitter to escape tiny plateaux
            if iteration % 200 == 0:
                i, j = np.random.choice(n, 2, replace=False)
                delta = (np.random.rand() - 0.5) * 2e-3   # ±0.001
                jitter = a.copy()
                jitter[i] = max(0.0, jitter[i] + delta)
                jitter[j] = max(0.0, jitter[j] - delta)
                jitter_sum = jitter.sum()
                if jitter_sum > 0:
                    jitter /= jitter_sum
                    jitter_metric = 2.0 * n * np.convolve(jitter, jitter).max()
                    if jitter_metric < cur_metric:
                        a = jitter
                        eta = max(eta, 0.1)
        iteration += 1
        if eta < 1e-10:          # step size negligible → stop early
            break

    # ------------------------------------------------------------------
    # 3) Final sanity – clip to the domain accepted by evaluate_sequence.
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_a.tolist()]

    # Report the final metric using the provided evaluator (if it exists)
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"\n--- FINAL RESULT ---\nLength = {len(final_seq)}  metric = {final_score:.9f}")
    except Exception as exc:   # guard against a missing import
        final_score = None
        print("\n--- FINAL RESULT ---")
        print(f"Length = {len(final_seq)}  (metric not evaluated: {exc})")

    return final_seq
