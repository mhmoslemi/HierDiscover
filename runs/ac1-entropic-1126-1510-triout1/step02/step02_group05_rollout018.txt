def linear_convolve_fft(x,y):
    n = len(x) + len(y) - 1
    size = 1 << (n-1).bit_length()
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    conv = np.fft.irfft(X * Y, size)[:n]
    return conv
```

def simplex_projection(v):
    """Projects a vector v onto the probability simplex (non-negative sum=1)."""
    # v may contain any real numbers
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    # find rho where u_i > cssv_i / i?
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w
```

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + max(0.0, budget_s - 2.0)

    # initialize length
    try:
        base = np.array(height_sequence_1, dtype=float)
        base = np.maximum(base, 0)
        if base.sum() <= 0: raise
        base = base / base.sum()
        n = len(base)
    except:
        n = 8000
        base = np.full(n, 1.0/n, dtype=float)

    a = base.copy()
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    print(f"[Init] score: {best_score:.6f}")

    # gradient descent parameters
    alpha = 5.0
    eta = 50.0
    alpha_mul = 1.05
    eta_mul = 0.995
    max_iter = 2000

    iter = 0
    while time.time() < deadline and iter < max_iter:
        iter += 1
        # compute convolution a*a
        b = linear_convolve_fft(a, a)
        M = b.max()
        # approximate score
        score = 2 * n * M
        # occasionally compute exact evaluate (costly) maybe every 50 iters
        if iter % 50 == 0:
            try:
                eval_score = evaluate_sequence(a.tolist())
            except:
                eval_score = float('inf')
            if eval_score < best_score:
                best_score = eval_score
                best_a = a.copy()
                print(f"[Iter {iter}] New best eval_score: {best_score:.6f}")

        # compute softmax weights
        # w = exp(alpha*(b - M)) / sum
        # using b - M <= 0
        exp_weights = np.exp(alpha * (b - M))
        w = exp_weights / exp_weights.sum()

        # gradient
        a_rev = a[::-1]
        conv_rev = linear_convolve_fft(a_rev, w)
        grad1 = conv_rev[n-1 : n-1 + n]

        conv2 = linear_convolve_fft(a, w)
        grad2 = conv2[:n]

        grad = grad1 + grad2

        # exponentiated gradient update
        a = a * np.exp(-eta * grad)
        a_sum = a.sum()
        if a_sum == 0 or np.isnan(a_sum):
            # reinitialize to uniform to avoid degenerate
            a = np.full(n, 1.0/n, dtype=float)
        else:
            a = a / a_sum

        # adapt alpha and eta
        alpha = min(alpha * alpha_mul, 200.0)
        eta = eta * eta_mul

        # Optionally print progress
        if iter % 200 == 0:
            print(f"[Iter {iter}] approx_score: {score:.6f}  alpha: {alpha:.2f}  eta: {eta:.2f}")

    # polishing with random pair moves
    # functions for random pair move defined earlier
    it = 0
    polish_iters = 5000
    while time.time() < deadline and it < polish_iters:
        it += 1
        sigma = 0.05 * (1 - it / polish_iters) + 0.005
        cand = _random_pair_move(best_a, sigma)
        try:
            cand_score = evaluate_sequence(cand.tolist())
        except:
            continue
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand
            print(f"[Polish {it}] new best score: {best_score:.6f}")

    # ensure non-negative
    best_a = np.maximum(best_a, 0.0)
    return best_a.tolist()

"""
Gradient‑descent on a smoothed version of the maximum autocorrelation.

The evaluation function is scale‑invariant, so we keep the coefficients on the
probability simplex (non‑negative, sum = 1).  Starting from the best known
construction (``height_sequence_1`` when present) we minimise a smooth
approximation of   max_k (f*f)_k :

    Φ(f) = (log∑_k exp(α·(f*f)_k)) / α      (α > 0)

Φ(f) → max_k (f*f)_k when α→∞.  Its gradient can be written in closed form
using two convolutions, which are computed efficiently with FFTs.
We perform exponentiated gradient updates
    f ← f * exp(‑η·∇Φ(f))   and renormalise to sum = 1.
The step size η and the smoothing parameter α are adapted during the run.
After the gradient phase we run a short random‑pair polishing (the same
routine used in the previous solution) to capture any remaining local
improvements.

The whole routine respects the 1000 s budget, prints progress, and returns
the best sequence found.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper: FFT‑based linear convolution (real input → real output)
# --------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    conv = np.fft.irfft(X * Y, size)[:n]
    return conv


# --------------------------------------------------------------
# Helper: cheap random pairwise move that keeps the sum exactly 1.
# --------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Transfer a small amount of mass between two distinct indices."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    delta = (np.random.rand() - 0.5) * sigma   # uniform in [‑σ/2, σ/2]

    if delta > 0:
        delta = min(delta, seq[j])            # take from j
    else:
        delta = -min(-delta, seq[i])          # take from i

    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq

    # ------------------------------------------------------------------
