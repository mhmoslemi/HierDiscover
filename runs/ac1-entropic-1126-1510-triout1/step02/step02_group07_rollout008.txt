def compute_conv_max(seq):
    n = len(seq)
    # compute minimal FFT length as next power of 2 >= 2*n-1
    n_fft = 1 << ((2*n - 2).bit_length())
    A = np.fft.rfft(seq, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[:2 * n - 1]
    return conv.max()

def compute_metric(seq):
    n = len(seq)
    maxc = compute_conv_max(seq)
    return 2.0 * n * maxc

def compute_conv_max(seq):
    n = seq.shape[0]
    n_fft = 1 << ((2*n - 2).bit_length())
    # rfft of real seq
    A = np.fft.rfft(seq, n=n_fft)
    # element-wise product
    conv = np.fft.irfft(A * A, n=n_fft)

"""
Propose a non‑negative step‑function sequence minimizing
    2·n·max(conv(f,f))/(∑f)²

Algorithm
---------

1. **Initialisation**
   * Use a previously found sequence (global variable
     ``height_sequence_1``) if it exists, otherwise start from the
     “arcsine’’ distribution.
   * Perform a cheap parametric sweep (power‑law and symmetric Beta
     families) to obtain a better starting point.

2. **Simulated‑Annealing (SA) refinement**
   * Each iteration proposes either a *local* mass transfer
     (move a random amount from index i to j) or a *global* Gaussian
     perturbation of the whole vector.
   * The new candidate is projected onto the probability simplex.
   * Acceptance follows the Metropolis rule with an exponentially
     decreasing temperature; every improvement updates the best
     solution seen so far.

3. **Gradient‑descent polishing**
   * When the SA phase ends, a few soft‑max gradient steps are taken
     (the gradient of the smooth maximum of the self‑convolution).
   * Steps are projected back onto the simplex.

All heavy operations (self‑convolution and convolutions needed for the
gradient) are performed with FFTs for speed.  The function respects the
time budget and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> np.ndarray:
    """Return a length‑n arcsine‑distribution (sum = 1)."""
    seq = np.empty(n, dtype=float)
    cur = 1.0
    for i in range(n):
        seq[i] = cur
        if i + 1 < n:
            cur = cur * (2.0 * i + 1.0) / (2.0 * (i + 1))
    seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("target sum s must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper returning the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def conv_max_fft(seq: np.ndarray) -> float:
    """Return max(self‑convolution) using FFT (seq assumed to sum to 1)."""
    n = seq.shape[0]
    n_fft = _next_pow2(2 * n - 1)
    # real‑valued FFT
    A = np.fft.rfft(seq, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
def metric_fft(seq: np.ndarray) -> float:
    """Scale‑invariant metric 2·n·max(conv).  Assumes seq ≥ 0, sum > 0."""
    n = seq.shape[0]
    maxc = conv_max_fft(seq)
    return 2.0 * n * maxc


# ----------------------------------------------------------------------
def full_conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Convolution of a and b using FFT (full length)."""
    n = a.shape[0]
    m = b.shape[0]
    n_fft = _next_pow2(n + m - 1)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    conv = np.fft.irfft(A * B, n=n_fft)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
def softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max over the convolution values (numerically stable)."""
    maxv = conv.max()
    exp_vals = np.exp(beta * (conv - maxv))
    return exp_vals / exp_vals.sum()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence with a low evaluation
    score.  Returns the best list of coefficients found within the time
    budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3  # safety margin

    # --------------------------------------------------------------
    # 1) Choose a starting length and initialise a sequence
    # --------------------------------------------------------------
    try:
        # use the globally saved sequence if it exists
        global height_sequence_1
        init_seq = np.asarray(height_sequence_1, dtype=float)
        if init_seq.ndim != 1 or init_seq.size == 0:
            raise ValueError
    except Exception:
        init_seq = None

    if init_seq is None:
        # previous best length was 3200; use that as a comfortable size
        n0 = 3200
        init_seq = arcsine_sequence(n0)

    init_seq = np.maximum(init_seq, 0.0)
    if init_seq.sum() == 0:
        init_seq = arcsine_sequence(init_seq.size)
    init_seq = init_seq / init_seq.sum()
    n = init_seq.shape[0]

    # --------------------------------------------------------------
    # 2) Fast parametric sweep to get a potentially better start
    # --------------------------------------------------------------
    best_seq = init_seq.copy()
    best_score = metric_fft(best_seq)

    # power‑law family a_i ∝ (i+1)^{-p}
    for p in np.linspace(0.10, 1.90, 45):   # 45 points, cheap
        cand = (np.arange(1, n + 1, dtype=float) ** (-p))
        cand /= cand.sum()
        sc = metric_fft(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand

    # symmetric Beta family a_i ∝ x^{α‑1}(1‑x)^{α‑1}
    xs = (np.arange(n, dtype=float) + 0.5) / n
    for alpha in np.linspace(0.10, 1.90, 45):
        cand = xs ** (alpha - 1) * (1.0 - xs) ** (alpha - 1)
        cand_sum = cand.sum()
        if cand_sum == 0:
            continue
        cand /= cand_sum
        sc = metric_fft(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand

    # --------------------------------------------------------------
    # 3) Simulated‑Annealing refinement
    # --------------------------------------------------------------
    current_seq = best_seq.copy()
    current_score = best_score

    # SA hyper‑parameters
    T = 1.0               # initial temperature
    T_min = 1e-6
    decay = 0.9997        # per‑iteration exponential cooling
    max_step_factor = 0.5 # maximal fraction of source mass to move
    sigma = 0.07          # std‑dev for global perturbations
    prob_local_move = 0.6  # probability to perform a local i→j move

    iter_cnt = 0
    last_print = time.time()

    while time.time() < deadline:
        iter_cnt += 1

        # ----------------------------------------------------------
        # propose a candidate
        # ----------------------------------------------------------
        if np.random.rand() < prob_local_move:
            # ----- local mass transfer -----
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            if current_seq[i] <= 1e-15:
                continue
            # allowed delta is a fraction of the source mass
            max_delta = current_seq[i]
            delta = max_delta * max_step_factor * np.random.rand()
            cand = current_seq.copy()
            cand[i] -= delta
            cand[j] += delta
        else:
            # ----- global Gaussian perturbation -----
            cand = current_seq + sigma * np.random.randn(n)
            cand = np.maximum(cand, 0.0)

        # re‑normalise to the simplex
        cand_sum = cand.sum()
        if cand_sum == 0:
            continue
        cand /= cand_sum

        # compute metric (fast FFT version)
        cand_score = metric_fft(cand)

        # ----------------------------------------------------------
        # acceptance test (Metropolis)
        # ----------------------------------------------------------
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            # probability = exp(-(Δ)/T)
            if T > 0:
                prob = math.exp(-(cand_score - current_score) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            current_seq = cand
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[SA] iter {iter_cnt:7d}  new best {best_score:.6f}")

        # temperature schedule
        T = max(T_min, T * decay)

        # occasional status print (once per ~5 s)
        if time.time() - last_print > 5.0:
            print(f"[SA] iter {iter_cnt:7d}  best {best_score:.6f}  T={T:.3e}")
            last_print = time.time()

    # --------------------------------------------------------------
    # 4) Gradient‑descent polishing (soft‑max smoothing)
    # --------------------------------------------------------------
    # use remaining time (if any)
    grad_step = 0.12
    beta = 5.0
    while time.time() < deadline - 0.5:
        conv = full_conv_fft(best_seq, best_seq)  # length 2n‑1
        maxc = conv.max()
        # soft‑max over the convolution entries
        w = softmax_weights(conv, beta)

        # gradient = 2 * (w * best_seq_rev) convolution, slice the relevant part
        rev = best_seq[::-1]
        grad_full = full_conv_fft(w, rev)  # length 2n‑1
        grad = 2.0 * grad_full[n - 1 : 2 * n - 1]

        cand = best_seq - grad_step * grad
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0:
            break
        cand /= cand_sum

        cand_score = metric_fft(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            print(f"[grad] improved to {best_score:.6f}")

        # decay step‑size and sharpen soft‑max
        grad_step *= 0.995
        beta = min(200.0, beta * 1.02)

        # early exit if step becomes negligible
        if grad_step < 1e-6:
            break

    # --------------------------------------------------------------
    # 5) Final sanity check and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total == 0:
        best_seq = arcsine_sequence(n)
    else:
        best_seq /= total

    # use the official evaluator for the final reported score
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Best sequence length = {len(best_seq)}  score = {final_score:.6f}")

    return best_seq.tolist()
