"""
Simulated‑annealing / random‑local‑search optimizer.

The evaluation metric for a non‑negative list a of length n (with
∑a = 1) is 2·n·max(conv(a,a)).  The goal is to make the maximal
self‑convolution as small as possible.  The algorithm proceeds as
follows:

1. Initialise a with the “arcsine’’ distribution (coefficients
   C_i = binom(2i,i)/4^i) – the known near‑optimal shape.
2. Repeatedly propose a tiny redistribution of mass between two
   randomly chosen indices while preserving the L1 norm.
   The new sequence is accepted if it improves the score,
   otherwise with a Metropolis probability that slowly cools.
3. The temperature and the size of the redistribution decay
   exponentially, giving a fine‑grained search near the optimum.
4. When the time budget is almost exhausted a few projected‑gradient
   steps (soft‑max of the convolution) are performed for polishing.
5. The best sequence seen during the run is returned.

All operations stay on the simplex (non‑negative, sum = 1) and the
convolution is evaluated by FFT for speed (O(n log n) per step).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> np.ndarray:
    """Return the normalised (sum = 1) arcsine coefficients C_i for i=0..n‑1.
    The recurrence C_{i+1}=C_i·(2i+1)/(2(i+1)) is used for stability.
    """
    seq = np.empty(n, dtype=np.float64)
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq[i] = cur
        total += cur
        if i + 1 < n:
            cur = cur * (2.0 * i + 1.0) / (2.0 * (i + 1))
    seq /= total
    return seq


# ----------------------------------------------------------------------
def conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using a real FFT."""
    n = a.shape[0]
    # next power of two for speed
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
def score_from_seq(a: np.ndarray) -> float:
    """Direct computation of the objective (identical to evaluate_sequence)."""
    n = a.shape[0]
    # keep guard against pathological zero vector
    s = a.sum()
    if s < 0.01:
        return float("inf")
    a = a / s
    m = conv_max(a)
    return float(2.0 * n * m / (s * s))


# ----------------------------------------------------------------------
def softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights approximating the arg‑max of conv."""
    max_val = conv.max()
    # numerical stability
    expv = np.exp(beta * (conv - max_val))
    w = expv / expv.sum()
    return w


# ----------------------------------------------------------------------
def gradient_step(a: np.ndarray, beta: float, step: float) -> np.ndarray:
    """One projected gradient step on the soft‑max approximation."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    w = softmax_weights(conv, beta)
    # gradient = 2 * (w * a_reversed) convolved, then slice central part
    grad = 2.0 * np.convolve(w, a[::-1])[n - 1 : 2 * n - 1]
    cand = a - step * grad
    # projection onto the probability simplex (non‑negative, sum = 1)
    # using the algorithm of Duchi et al.
    u = np.sort(cand)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(cand) + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = 0.0
    proj = np.maximum(cand - theta, 0.0)
    return proj


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence with a low
    evaluation score.  The routine runs for up to `budget_s` seconds
    and returns the best list found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # -------------------------------------------------------------
    # 1) Choose a starting length.  Moderate lengths keep the
    #    FFT cheap while still allowing a fine discretisation.
    n_candidates = [1500, 1800, 2000, 2200, 2500, 3000, 3200]
    best_seq = None
    best_sc = float("inf")
    best_n = None

    for n in n_candidates:
        if time.time() >= deadline:
            break
        a = arcsine_sequence(n)
        sc = score_from_seq(a)
        print(f"[init] n={n:<4d}  score={sc:.6f}")
        if sc < best_sc:
            best_sc = sc
            best_seq = a.copy()
            best_n = n

    # -------------------------------------------------------------
    # 2) If a previous global solution exists, try it as well.
    try:
        global height_sequence_1
        if isinstance(height_sequence_1, (list, tuple, np.ndarray)):
            a = np.asarray(height_sequence_1, dtype=np.float64)
            if a.size > 0:
                a = np.maximum(a, 0.0)
                if a.sum() > 0.0:

                n = a.shape[0]
                sc = score_from_seq(a)

                    best_seq = a.copy()
                    best_n = n
    except Exception:

    # -------------------------------------------------------------

        best_sc = score_from_seq(best_seq)

    # -------------------------------------------------------------

    n = best_seq.shape[0]
    cur_seq = best_seq.copy()
    cur_sc = best_sc

    # Annealing parameters – fairly gentle cooling so that we can
    # explore for a long time.
    temperature = 0.005          # initial acceptance temperature

        # pick a destination uniformly (different from src)

                print(f"[SA] iter={it:<6d}  new best {best_sc:.6f}")

        temperature = max(1e-6, temperature * temp_decay)
        step_scale = max(1e-8, step_scale * step_decay)

        # occasional quick check‑point to avoid overrunning the budget
        if it % 5000 == 0 and time.time() >= deadline:

    # -------------------------------------------------------------
    # 4) Gradient‑descent polishing (soft‑max of the convolution)
    #    This final stage often squeezes a little more out of the
    #    current best vector.
    grad_step = 0.05

        cand = gradient_step(best_seq, beta, grad_step)
        cand_sc = score_from_seq(cand)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_seq = cand
            print(f"[grad] polished best {best_sc:.6f}")
        # slowly tighten the approximation
        grad_step *= 0.98
        beta = min(200.0, beta * 1.02)

    # -------------------------------------------------------------
    # Final sanity – enforce non‑negativity and unit sum
    best_seq = np.maximum(best_seq, 0.0)
    if best_seq.sum() == 0.0:

    best_seq /= best_seq.sum()
