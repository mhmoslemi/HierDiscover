def _softmax_weights(conv, T):
   maxv = conv.max()
   shifted = conv - maxv
   exp_arg = shifted / T
   exp_arg = np.clip(exp_arg, -50, 0)
   w = np.exp(exp_arg)

       w.fill(1.0 / w.size)
   else:
       w /= w_sum
   return w

    deadline = time.time() + budget_s - 0.5

    # Determine n

        init_seq = np.array(height_sequence_1, dtype=float)

    except NameError:

    # Helper _score
    def _score(a):
        conv = _self_convolution(a, size_fft)

    # Evaluate reference if exists

        best_score = cur_score2
        best_a = a_ref.copy()
    else:
        # start with uniform
        a_uni = np.full(n, target_sum / n)

        a = _project_to_simplex(a, target_sum)
        sc = _score(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
    # double-ended inverse sqrt:
    for _ in range(150):
        beta = np.random.rand() * 20.0
        i = np.arange(n)
        a = (1.0 / np.sqrt(i + beta + 1.0) + 1.0 / np.sqrt((n-1-i) + beta + 1.0))
        a = _project_to_simplex(a, target_sum)
        sc = _score(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()

    # Power law:
    for _ in range(150):
        beta = np.random.rand() * 20.0

        a = _project_to_simplex(a, target_sum)
        sc = _score(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
    # exponential
    for _ in range(150):
        decay = np.random.rand() * 0.02 + 0.0001

        sc = _score(a)
        if sc < best_score:

            sc = _score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()

    # random block candidates
    for _ in range(500):
        nb = np.random.randint(2, 9)  # 2-8 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)

    print(f"[shape search] best score = {best_score:.6f}")

    # Stage 2: gradient descent with Adam and softmax

    a = best_a.copy()

    # Adam hyperparams

    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lr0 = target_sum * 0.3 / n   # base learning rate
    step = 0
    no_improve = 0
    # Keep track of best (as we might drop later)
    best_score = _score(a)
    best_seq = a.copy()

    # Determine after how many steps random perturbation is applied
    perturb_interval = 5000
    max_iter = 2000000   # large limit if time permits.

    while time.time() < deadline and step < max_iter:
        # compute conv and score
        conv = _self_convolution(a, size_fft)
        cur_score = conv.max()
        # update best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"[{step:07d}] new best {best_score:.6f}")
        else:
            no_improve += 1

        # softmax weights
        if T > 1e-5:
            # compute softmax weights
            maxv = conv.max()
            shifted = conv - maxv
            exp_arg = shifted / T
            exp_arg = np.clip(exp_arg, -50, 0)
            w = np.exp(exp_arg)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
        else:
            # hard max subgradient: uniform weight on argmax
            maxv = conv.max()
            inds = np.where(np.abs(conv - maxv) < 1e-12)[0]
            w = np.zeros_like(conv)
            w[inds] = 1.0 / inds.size

        # gradient from weights
        grad = _gradient_from_weights(a, w, size_fft)

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (step + 1))
        v_hat = v / (1 - beta2 ** (step + 1))

        step_lr = lr0 / (1 + 1e-6 * step)  # decreasing

        a_new = a - step_lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex to enforce constraints
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional random perturbation
        if (step % perturb_interval == 0 and step > 0) or no_improve >= 10000:
            noise = np.random.randn(n) * target_sum * 5e-4
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)
            no_improve = 0
            print(f"[{step:07d}] random perturb")

        a = a_new
        T *= 0.9997  # slowly cool
        step += 1

    # Stage 3: final random block refinement
    a = best_seq.copy()
    # we can do many random block trials
    for _ in range(1000):
        nb = np.random.randint(2, 9)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb) + 0.05
        cand = np.zeros(n)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[final block] improved to {best_score:.6f}")

    # Report final best
    print(f"[final] best score = {best_score:.6f}")

    # Return as list of floats
    return best_seq.tolist()

"""
Hybrid heuristic for minimizing 2·n·max(conv(a,a)) / (sum(a)^2).

1. Initialise a large pool of candidate shapes (inverse‑sqrt, power‑law,
   exponential, cosine, mixtures, Dirichlet draws, random block layouts)
   and keep the best (projected onto the simplex Σa = √(2n)).

2. Refine the best candidate with an Adam‑type projected gradient descent
   on a smooth approximation of the max‑convolution.  The smoothness is
   controlled by a temperature T which is slowly cooled, turning the loss
   into the true max at the end of the run.

3. When T becomes very small we switch to a pure sub‑gradient step on the
   actual max (uniform weights on the arg‑max positions).  This yields a
   fine‑grained final polishing.

4. A final random‑block search is performed to escape possible plateaus.

All updates respect the non‑negativity and sum constraints by Euclidean
projection onto the simplex.  The algorithm runs for at most the supplied
time budget and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using FFT (output length len(u)+len(v)-1)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full discrete convolution a*a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    w must sum to 1 and be non‑negative.
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)
    start = len(a) - 1
    grad = 2.0 * conv[start:start + len(a)]
    return grad
