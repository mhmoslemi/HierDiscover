"""
A fast, two‑phase optimiser for the step‑function score.

Phase 1 – cheap parametric sweep:
    • Try several sequence lengths (up to ~8 k).
    • For each length generate simple families:
        – uniform,
        – power‑law decay  p_i ∝ (i+1)^‑α,
        – mirrored power‑law (mass at the right side),
        – exponential decay  p_i ∝ r^i.
    • Keep the best candidate (objective = 2·n·max(conv)).

Phase 2 – targeted simulated‑annealing:
    • Work on the best sequence from phase 1.
    • Each iteration:
        – compute the full convolution by FFT,
        – locate the shift k_max where the convolution attains its maximum,
        – compute the contribution p_i·p_{k_max‑i} (zero outside the support).
        – move a small amount of mass from the index with the largest
          contribution (the “donor”) to an index with the smallest gradient
          (the “recipient”).  The move respects non‑negativity and preserves
          the L1 norm.
        – accept the move if it improves the score or with a probability
          exp(‑Δ/temperature) (temperature cools linearly).
    • Occasionally perform a completely random pair move to escape plateaux.
    • Track the best sequence seen so far.

The convolution is evaluated with an FFT (O(n log n)) which makes millions of
iterations feasible within the 1000 s budget.  The returned list satisfies all
security checks of ``evaluate_sequence`` (non‑negative, sum≈1)."""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: convolution via FFT (real input -> real output)
# ----------------------------------------------------------------------
def _conv_fft(p: np.ndarray) -> np.ndarray:
    """Return the full linear convolution p * p using a power‑of‑two FFT."""
    n = p.shape[0]
    nfft = 1 << int(np.ceil(np.log2(2 * n - 1)))
    ft = np.fft.rfft(p, n=nfft)
    conv = np.fft.irfft(ft * ft, n=nfft)[: 2 * n - 1]
    # Numerical noise may produce tiny negatives
    return np.maximum(conv, 0.0)


# ----------------------------------------------------------------------
# Helper: compute the evaluation function (2·n·max(conv))
# ----------------------------------------------------------------------
def _score(p: np.ndarray) -> float:
    """Score of a probability vector p (sum(p)=1)."""
    n = p.shape[0]
    conv = _conv_fft(p)
    M = conv.max()
    return float(2.0 * n * M)

            accept = True
        else:
            prob = math.exp(-(new_score - curr_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            p = p_new
            curr_score = new_score

        # ----- keep the overall best -----
        if new_score < best_score:
            best_score, best_seq = new_score, p_new.copy()
            print(f"[SA {iter_cnt:06d}] new best score={best_score:.6f}")

        # ----- occasional random perturbation to shake out of stagnation -----
        if iter_cnt % 10_000 == 0:
            # add a tiny uniform noise and re‑normalise
            noise = (np.random.rand(N) - 0.5) * 1e-3
            p = np.clip(p + noise, 0.0, None)
            if p.sum() == 0.0:
                p = np.full(N, 1.0 / N, dtype=float)
            else:
                p /= p.sum()
            curr_score = _score(p)

    # --------------------------------------------------------------
    # Post‑processing: trim negligible trailing entries and return.
    # --------------------------------------------------------------
    eps = 1e-12
    nz = np.where(best_seq > eps)[0]
    if nz.size > 0:
        best_seq = best_seq[nz[0] : nz[-1] + 1]
    else:
        # extremely unlikely, fallback to a uniform vector of length 1
        best_seq = np.array([1.0], dtype=float)

    # ensure non‑negative numbers (extra safety)
    best_seq = np.maximum(best_seq, 0.0)

    return best_seq.tolist()
