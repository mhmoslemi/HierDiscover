import time
import math
import numpy as np

def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum x = z}."""
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries will be zero, put all mass on the largest element
        w = np.zeros_like(v)
        w[np.argmax(v)] = z
        return w
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using FFT, output length len(u)+len(v)-1."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]

def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full convolution a*a."""
    return _conv_fft(a, a, size_fft)

def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    The formula is 2 * (w * a_rev)[n-1 : n-1+n]   (no extra subtraction needed).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)  # length = len(w)+len(a_rev)-1 = (2n-1)+n-1 = 3n-2
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad

    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries would become zero – put everything on the largest component
        w = np.zeros_like(v)
        w[np.argmax(v)] = z
        return w
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution u * v using FFT (output length len(u)+len(v)-1)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full discrete convolution a*a."""
    return _conv_fft(a, a, size_fft)


def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    Formula: 2 * (w * a_rev)[n‑1 : n‑1+n], where a_rev = a[::-1].
    """
    n = a.shape[0]
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)          # length = 3n‑2
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function (sequence) that minimizes
    `evaluate_sequence`.  The algorithm proceeds in three stages:

    1. **Broad parametric shape sampling** – families such as inverse‑sqrt,
       power‑law, exponential, cosine, sine‑squared, triangular, random
       Dirichlet and random block structures are sampled and projected onto
       the simplex {a_i ≥ 0, Σ a_i = √(2 n)}.

    2. **Refinement** – a smoothed‑max gradient descent using Adam and an
       exponentially decreasing temperature (soft‑max approximation of the
       max) is run.  The gradient is obtained analytically via FFT‑based
       convolutions; after each step the vector is projected back onto the
       simplex.  Random perturbations and plateau handling help escape local
       minima.

    3. **Final random‑block post‑processing** – a short sweep of block‑wise
       candidates finishes the search.

    The returned list of floats satisfies all security checks of
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # problem size
    try:
        n = len(height_sequence_1)   # use the reference sequence if present
    except Exception:
        n = 3000
    target_sum = math.sqrt(2.0 * n)

    # FFT size – must be at least 3·n‑2 for the weighted‑gradient step
    size_fft = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # quick scoring (max of the self‑convolution)
    def _score(a: np.ndarray) -> float:
        return _self_convolution(a, size_fft).max()

    # ------------------------------------------------------------------
    # 1. PARAMETRIC SHAPE SEARCH
    best_a = None
    best_score = float("inf")

    def _test_candidate(arr: np.ndarray):
        nonlocal best_score, best_a
        arr = _project_to_simplex(arr, target_sum)
        sc = _score(arr)
        if sc < best_score:
            best_score, best_a = sc, arr.copy()
            print(f"[shape] new best = {best_score:.6f}")

    # uniform
    a = np.ones(n)
    a = _project_to_simplex(a, target_sum)
    _test_candidate(a)

    # inverse‑sqrt families
    for _ in range(80):
        beta = np.random.rand() * 10.0
        a = 1.0 / np.sqrt(np.arange(n) + beta + 1.0)
        _test_candidate(a)

    # double‑ended inverse‑sqrt
    for _ in range(80):
        beta = np.random.rand() * 10.0
        i = np.arange(n)
        a = (1.0 / np.sqrt(i + beta + 1.0) +
             1.0 / np.sqrt((n - 1 - i) + beta + 1.0))
        _test_candidate(a)

    # power‑law
    for _ in range(80):
        beta = np.random.rand() * 10.0
        pexp = np.random.rand() * 1.7 + 0.3
        a = 1.0 / (np.arange(n) + beta + 1.0) ** pexp
        _test_candidate(a)

    # exponential decay
    for _ in range(80):
        decay = np.random.rand() * 0.01 + 0.0005
        a = np.exp(-decay * np.arange(n))
        _test_candidate(a)

    # cosine‑based (large at ends, small in the middle)
    for _ in range(80):
        p = np.random.rand() * 5.0 + 0.5
        a = np.abs(np.cos(np.pi * np.arange(n) / (n - 1))) ** p
        _test_candidate(a)

    # sine‑squared family
    for _ in range(80):
        p = np.random.rand() * 4.0 + 0.5
        a = np.sin(np.pi * np.arange(n) / (n - 1)) ** p
        _test_candidate(a)

    # triangular (tent) shape
    for _ in range(40):
        a = np.minimum(np.arange(n) + 1, n - np.arange(n)).astype(float)
        _test_candidate(a)

    # mixture of uniform and inverse‑sqrt
    for _ in range(80):
        beta = np.random.rand() * 10.0
        lam = np.random.rand()
        a = lam * np.ones(n) + (1.0 - lam) * (1.0 /
                np.sqrt(np.arange(n) + beta + 1.0))
        _test_candidate(a)

    # random Dirichlet draws (sparse candidates)
    for _ in range(120):
        concentration = np.random.rand() * 1.5 + 0.05   # in [0.05, 1.55]
        a = np.random.gamma(concentration, 1.0, size=n)
        _test_candidate(a)

    # random block candidates (few large blocks)
    for _ in range(200):
        nb = np.random.randint(2, 7)                     # 2‑6 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.zeros(n)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        _test_candidate(cand)

    print(f"[shape search] best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. REFINEMENT VIA SMOOTHED GRADIENT DESCENT (Adam + temperature)
    a = best_a.copy()
    conv = _self_convolution(a, size_fft)
    T = max(1e-6, (conv.max() - conv.min()) * 0.5)   # initial temperature

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr0 = 0.05                     # base learning rate
    it = 0
    no_improve = 0
    max_iters = 2_000_000

    while time.time() < deadline and it < max_iters:
        conv = _self_convolution(a, size_fft)
        cur_score = conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[{it:07d}] new best = {best_score:.6f}")
        else:
            no_improve += 1

        # soft‑max weights (temperature T)
        shift = conv - conv.max()
        if T < 1e-12:
            T = 1e-12
        exp_arg = shift / T
        exp_arg = np.clip(exp_arg, -50, 0)   # avoid extreme exponents
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a
        grad = _weighted_gradient(a, w, size_fft)

        # Adam update
        it1 = it + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it1)
        v_hat = v / (1.0 - beta2 ** it1)
        lr = lr0 / (1.0 + it * 1e-4)          # very gentle decay

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = _project_to_simplex(a_new, target_sum)

        # perturbation to escape plateaus
        if (it % 5000 == 0 and it > 0) or no_improve > 3000:
            scale = target_sum * (5e-3 if no_improve > 3000 else 1e-3)
            a_new = a_new + np.random.randn(n) * scale
            a_new = _project_to_simplex(a_new, target_sum)
            if no_improve > 3000:
                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0
                print(f"[{it:07d}] plateau perturbation")

        a = a_new
        T = T * 0.9995                     # cool down slowly
        it += 1

    # ------------------------------------------------------------------
    # 3. FINAL RANDOM BLOCK SEARCH (light local improvement)
    for _ in range(200):
        nb = np.random.randint(2, 7)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.zeros(n)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[block] improved to {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f} after {it} iterations")
    return best_a.tolist()
