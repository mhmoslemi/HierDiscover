"""
Propose a low‑score step‑function.

The algorithm works in three stages:

1. **Initial candidates** –  for a given length `n` we try a few
   symmetric Beta(α,α) densities (including the arcsine case α=½) and
   keep the one with the smallest raw `2·n·max(conv(a,a))` score.

2. **Adam optimisation on a smooth max** –  the discrete distribution
   `a` is represented by soft‑max logits `z` (so `a` is always
   non‑negative and sums to 1).  We minimise a smooth approximation of
   the peak of the self‑convolution

        F_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k)          (λ → ∞ ⇒ max)

   The gradient of `F_λ` w.r.t. `a` can be written as
   `g_a = 2·corr(w,a)` where `w` are the soft‑max weights of the
   convolution entries.  Using the chain rule we obtain the gradient
   w.r.t. the logits `z`.  An Adam update is applied, and `λ` is
   increased from a small value (0.5) to a larger final value (≈30) so
   that the optimisation gradually focuses on the true peak.

   The optimisation runs as long as time permits (up to thousands of
   iterations).  The best sequence seen during the run is retained.

3. **Multiple lengths / fall‑back** –  a few different lengths are
   explored (preferring medium sizes, which are cheap per iteration),
   and the globally best sequence is returned.  If a historic sequence
   `height_sequence_1` exists it is also considered.

The whole search respects the 1000 s budget, prints progress and
finally returns a plain Python list of non‑negative heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (α=½) on n points, Σa = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the grid x_i = (i+½)/n (Σ=1)."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, m)
    B = A * A
    c = np.fft.irfft(B, m)
    return c[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k‑i} (correlation) using FFT.
    w – weights of length 2·n‑1, a – current sequence (length n),
    m – FFT length (power of two, at least len(w)+len(a)-1).
    Returns the slice g[0…n‑1].
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start:start + len(a)]


# ----------------------------------------------------------------------
def _optimise_adam(
    n: int,
    deadline: float,
    init_seq: np.ndarray,
    max_iters: int = 50000,
    lr: float = 0.05,
    lambda_target: float = 30.0,
    lambda_ramp_iters: int = 1500,
) -> tuple[np.ndarray, float]:
    """
    Refine `init_seq` (already normalised to sum=1) by minimising a
    smooth approximation of the max‑convolution peak using Adam on the
    soft‑max logits.  Returns the best sequence seen and its (unsmoothed)
    score `2·n·max(conv(a,a))`.
    """
    eps = 1e-12
    z = np.log(init_seq + eps)           # logits, ensures a = softmax(z)
    m_fft = 1 << ((3 * n - 2).bit_length())   # FFT size for correlation

    # Adam state
    m1 = np.zeros_like(z)
    v1 = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8

    best_seq = None
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # --- a = softmax(z) -------------------------------------------------
        maxz = np.max(z)
        expz = np.exp(z - maxz)
        a = expz / np.sum(expz)

        # --- current (unsmoothed) score ------------------------------------
        b = _conv_full_fft(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} Adam iter={it:5d} score={best_score:.6f}")

        # --- smooth‑max weighting -------------------------------------------
        lam = min(lambda_target,
                  0.5 + (lambda_target - 0.5) * (it / lambda_ramp_iters))
        maxb = b.max()
        exp_arg = lam * (b - maxb)          # ≤0, safe
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.shape[0])
        else:
            w /= w_sum

        # --- gradient w.r.t. a ----------------------------------------------
        g_a = _gradient_via_correlation(w, a, m_fft)

        # --- gradient w.r.t. logits z (soft‑max Jacobian) --------------------
        dot = np.dot(g_a, a)                # a·g_a
        g_z = a * (g_a - dot)

        # --- Adam update ----------------------------------------------------
        m1 = beta1 * m1 + (1 - beta1) * g_z
        v1 = beta2 * v1 + (1 - beta2) * (g_z * g_z)
        m_hat = m1 / (1 - beta1 ** it)
        v_hat = v1 / (1 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # --- occasional random jitter (helps escape plateaus) ---------------
        if it % 500 == 0 and np.random.rand() < 0.2:
            z += np.random.randn(n) * 1e-4

    if best_seq is None:               # should never happen
        best_seq = init_seq.copy()
        best_score = 2.0 * n * _conv_full_fft(best_seq).max()
    return best_seq, best_score
