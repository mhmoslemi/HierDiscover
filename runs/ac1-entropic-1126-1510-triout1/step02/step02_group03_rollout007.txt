def _optimise_expgrad(a, n, deadline, max_iters, eta0, lambda0, lambda_max):
    # a is normalized.
    best_seq, best_score = a.copy(), compute_score(a)
    m = 1 << ((3*n-2).bit_length())
    lam = lambda0
    for it in range(1, max_iters+1):
        if time.time()>deadline: break
        b = _conv_full_fft(a)
        score = 2*n*b.max()
        if score<best_score:
            best_score, best_seq = score, a.copy()
        # compute w for softmax
        maxb = b.max()
        exp_arg = lam*(b - maxb)
        # to avoid overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum==0.0:
            w = np.full_like(b, 1.0/b.shape[0])
        else:
            w/=w_sum
        # gradient
        g = _gradient_via_correlation(w, a, m)
        eta = eta0/np.sqrt(it)
        a = a*np.exp(-eta*g)
        a = _project_onto_simplex(a)
        # increase lam
        lam = min(lambda_max, lam*1.01)
        # jitter
        if it%200==0 and np.random.rand()<0.2:
            a+=np.random.randn(n)*1e-4
            a = np.clip(a,0.0,None)
            a = _project_onto_simplex(a)
    return best_seq, best_score
```

def _optimise_maxsubgrad(a, n, deadline, max_iters):
    best_seq = a.copy()
    best_score = compute_score(a)
    m = 1 << ((3*n-2).bit_length())
    for it in range(1, max_iters+1):
        if time.time()>deadline: break
        b = _conv_full_fft(a)
        cur_max = b.max()
        score = 2*n*cur_max
        if score<best_score:
            best_score, best_seq = score, a.copy()
        mask = (b == cur_max)
        # handle multiple ties:
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum>0:
            w/=w_sum
        else:
            w = np.full_like(b, 1.0/b.shape[0])
        g = _gradient_via_correlation(w, a, m)
        eta = 0.1/np.sqrt(it)  # maybe smaller
        a = a - eta * g
        # ensure positivity and project to simplex
        a = _project_onto_simplex(a)
    return best_seq, best_score
```

"""
Propose a low‑score non‑negative step‑function.

The algorithm performs many short stochastic runs:
* pick a length n (several candidates up to 20000)
* build a random “seed” distribution (Beta, Dirichlet, mixture or historic)
* minimise a smooth approximation of the peak of the self‑convolution by
  exponentiated‑gradient (mirror descent on the simplex) while slowly
  annealing the soft‑max temperature λ.
* finish the run with a few projected sub‑gradient steps on the true
  max‑convolution (λ → ∞) to polish the result.
All runs are time‑boxed; the best sequence seen before the 1000 s budget
expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a power‑of‑two FFT.
    Returned length = 2·len(a)‑1.
    """
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, m)
    c = np.fft.irfft(A * A, m)
    return c[: 2 * n - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the grid x_i = (i+½)/n, Σa = 1."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a Dirichlet vector with a random concentration (∈[0.05,0.7])."""
    conc = np.random.uniform(0.05, 0.7)
    g = np.random.gamma(conc, 1.0, size=n)
    g /= g.sum()
    return g


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g = 2·Σ_k w_k·a_{k‑i} (correlation) using FFT.
    w – length 2·n‑1, a – length n, m – FFT size (power of two ≥ len(w)+len(a)-1).
    Returns the slice of length n.
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start:start + len(a)]


# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return the exact evaluate_sequence score for a normalised (sum=1) vector."""
    n = len(a)
    b = _conv_full_fft(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
def _optimise_expgrad(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 8000,
    eta0: float = 0.3,
    lambda0: float = 0.5,
    lambda_max: float = 80.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the soft‑max approximation of
    max(conv(a,a)).  Returns the best sequence seen and its *exact* score.
    """
    a = a0.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = _score(a)

    # FFT length sufficient for correlation of length (2n‑1) with a of length n
    m = 1 << ((3 * n - 2).bit_length())
    lam = lambda0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- current exact score (for bookkeeping) -----
        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} it={it:5d} score={best_score:.6f}")

        # ----- soft‑max weights -----
        b = _conv_full_fft(a)
        maxb = b.max()
        # compute exp(lam*(b-maxb)) safely (maxb subtracted to avoid overflow)
        w = np.exp(lam * (b - maxb))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.shape[0])
        else:
            w /= w_sum

        # ----- gradient (correlation) -----
        g = _gradient_via_correlation(w, a, m)

        # ----- exponentiated‑gradient step -----
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)          # stays non‑negative
        a = _project_onto_simplex(a)      # enforce Σa = 1

        # ----- occasional jitter to escape plateaus -----
        if it % 300 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a = _project_onto_simplex(a)

        # ----- anneal λ towards the hard max -----
        lam = min(lambda_max, lam * 1.01)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _optimise_hard_max(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution peak.
    Starts from a0 (already normalised).  Returns best sequence seen.
    """
    a = a0.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = _score(a)

    m = 1 << ((3 * n - 2).bit_length())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        b = _conv_full_fft(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} hard‑it={it:5d} score={best_score:.6f}")

        # weights that pick the currently maximal entries (average over ties)
        mask = (b == cur_max).astype(float)
        w_sum = mask.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.shape[0])
        else:
            w = mask / w_sum

        # sub‑gradient of the max (correlation)
        g = _gradient_via_correlation(w, a, m)

        eta = 0.15 / np.sqrt(it)          # smaller step than EG
        a = a - eta * g
        a = _project_onto_simplex(a)

        # tiny random jitter occasionally
        if it % 250 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-5
            a = np.clip(a, 0.0, None)
            a = _project_onto_simplex(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _initial_sequence(n: int) -> np.ndarray:
    """
    Produce a random start vector for length n.
    Choices:
        - Beta(α,α) with random α ∈ [0.02, 1.6]
        - Dirichlet with random concentration
        - Simple mixture of a Beta and the uniform distribution
        - Historic sequence if available
    """
    choice = np.random.choice(["beta", "dirichlet", "mix", "hist"], p=[0.4, 0.3, 0.25, 0.05])

    if choice == "beta":
        alpha = np.random.uniform(0.02, 1.6)
        return _beta_distribution(n, alpha)

    if choice == "dirichlet":
        return _random_dirichlet(n)

    if choice == "mix":
        alpha = np.random.uniform(0.1, 1.2)
        beta_vec = _beta_distribution(n, alpha)
        w = np.random.rand()
        mix = w * beta_vec + (1.0 - w) * np.full(n, 1.0 / n)
        return mix / mix.sum()

    # historic (fallback if not defined)
    if "height_sequence_1" in globals():
        hist = np.array(height_sequence_1, dtype=np.float64)
        hist = np.clip(hist, 0.0, None)
        if hist.size > 0 and hist.sum() > 0.0:
            if hist.size >= n:
                seq = hist[:n]
            else:
                pad = np.zeros(n - hist.size, dtype=np.float64)
                seq = np.concatenate([hist, pad])
            if seq.sum() == 0.0:
                seq = np.full(n, 1.0 / n)
            else:
                seq = seq / seq.sum()
            return seq

    # pure uniform as last resort
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a very low `evaluate_sequence` score.
    The routine runs many short stochastic optimisation runs (exp‑grad +
    hard‑max) for several lengths and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0                # safety margin

    # ------------------------------------------------------------------
    # global bookkeeping
    best_seq = None
    best_score = float("inf")

    # candidate lengths (balanced between speed and theoretical quality)
    candidate_lengths = [5000, 8000, 12000, 16000, 20000]

    # main stochastic loop – run until time runs out
    while time.time() < deadline:
        # pick a length (biased towards larger n when we have time)
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        # simple heuristic: pick larger n more often as we get deeper time
        probs = np.array(candidate_lengths, dtype=float)
        probs = probs / probs.sum()
        n = np.random.choice(candidate_lengths, p=probs)

        # ------------------------------------------------------------------
        # 1) build a random start vector
        a0 = _initial_sequence(n)

        # ------------------------------------------------------------------
        # 2) smooth optimisation (exponentiated gradient)
        # allocate a slice of the remaining time (about 60 %)
        slice_deadline = time.time() + min(6.0, remaining * 0.6)
        a_mid, score_mid = _optimise_expgrad(
            a0=a0,
            n=n,
            deadline=slice_deadline,
            max_iters=8000,
            eta0=0.3,
            lambda0=0.5,
            lambda_max=80.0,
        )

        # ------------------------------------------------------------------
        # 3) hard max polishing (projected sub‑gradient)
        if time.time() < deadline:
            slice_deadline2 = time.time() + min(3.0, deadline - time.time())
            a_refined, score_refined = _optimise_hard_max(
                a0=a_mid,
                n=n,
                deadline=slice_deadline2,
                max_iters=3000,
            )
        else:
            a_refined, score_refined = a_mid, score_mid

        # decide which score to keep
        if score_refined < best_score:
            best_score = score_refined
            best_seq = a_refined.copy()
            print(f"*** New global best: n={n} score={best_score:.6f}")

        # early exit if we have already beaten the target
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # fallback – guarantee a return value
    if best_seq is None:
        n = 8000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _score(best_seq)

    print(f"Final best: score={best_score:.6f} length={len(best_seq)}")
    # return as plain Python list
    return best_seq.tolist()
