"""
Propose a step‑function (list of non‑negative heights) with a low
`evaluate_sequence` score.

Algorithm
---------
* Work on a probability simplex: the objective is
      2·n·max(a∗a)   (because the sum of the coefficients is fixed to 1).
* Initialise the sequence with a useful shape (arcsine, previous best,
  uniform or random).
* Minimise the maximal self‑convolution by a *smooth* projected gradient
  descent:
    - Use a soft‑max on the convolution values with a temperature
      parameter α.  For small α the gradient is an average of many
      convolution entries, for large α it approaches the sub‑gradient
      of the true max.
    - The gradient of the soft‑max is 2·(w∗a_rev) where w are the
      soft‑max weights and a_rev is a reversed copy of a.  This
      convolution is evaluated with FFT for speed.
    - Momentum (β) and a diminishing step‑size η/√t are applied.
    - After each step we project back onto the simplex.
* α is increased geometrically, so the optimisation starts smooth and
  ends with a true sub‑gradient step.
* The procedure is run for several sequence lengths (derived from the
  previous best) and several random restarts within the overall time
  budget (default 1000 s).  The best sequence found is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_sequence(n: int, kind: str) -> np.ndarray:
    """Create an initial probability vector of length n."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif kind == "arcsine":
        a = _arcsine_distribution(n)
    elif kind == "prev":
        try:
            a = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if a.size != n:
                raise ValueError
            a = a / a.sum()
        except Exception:
            a = _arcsine_distribution(n)
    else:  # fallback
        a = _arcsine_distribution(n)

    # make sure it lies exactly on the simplex
    a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_softmax(a: np.ndarray,
                         b: np.ndarray,
                         alpha: float,
                         size_fft: int) -> np.ndarray:
    """
    Compute the gradient of  (1/α)·log∑_k exp(α·b_k)  w.r.t. a.
    For large α this reduces to the sub‑gradient of max(b).
    """
    n = a.shape[0]

    # --- case α large: treat the max entry as a one‑hot weight ----------
    if alpha > 2000:                 # heuristic – safe to use sub‑gradient
        k = int(np.argmax(b))
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]
        return g

    # --- otherwise compute the full soft‑max gradient --------------------
    shift = b - b.max()                 # numerical stability
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:                    # should not happen, but guard anyway
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    # convolution w * a_rev  (a_rev = a reversed)
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = n - 1
    g = 2.0 * conv[start:start + n]
    return g


# ----------------------------------------------------------------------
def _optimize_one_length(init_a: np.ndarray,
                         n: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent with a soft‑max smoothing of the max‑convolution.
    Returns the best sequence discovered on this length and its internal
    (2·n·max_b) score (i.e. the exact value returned by evaluate_sequence when the
    sum of the coefficients equals 1).
    """
    a = init_a.copy()
    best_seq = a.copy()
    b = _fft_self_convolve(a)
    best_score = 2.0 * n * b.max()

    # parameters
    beta = 0.9                # momentum
    eta0 = 0.8                # initial step size constant
    velocity = np.zeros_like(a)
    alpha = 1.0               # soft‑max temperature
    alpha_mul = 1.02          # geometric increase
    alpha_max = 20000.0
    size_fft = 1 << ((3 * n - 2).bit_length())   # enough for w (2n‑1) + a_rev

    it = 0
    while time.time() < deadline:
        it += 1

        # current convolution and score
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} score={best_score:.6f}")

        # gradient (soft‑max or sub‑gradient)
        grad = _gradient_via_softmax(a, b, alpha, size_fft)

        # momentum update
        velocity = beta * velocity + (1.0 - beta) * grad

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        # gradient step + projection
        a = a - eta * velocity
        a = _project_onto_simplex(a, s=1.0)

        # anneal α
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional tiny jitter to escape plateaus
        if it % 2000 == 0 and np.random.rand() < 0.15:
            a += np.random.randn(n) * 1e-4
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score
