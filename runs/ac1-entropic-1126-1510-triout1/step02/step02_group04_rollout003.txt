Define projection function:

def project_simplex(v):
    """Project onto simplex (nonnegative, sum=1) using standard algorithm."""
    # This method ensures optimum projection.
    # Implementation: sort descending, compute cumulative sum, find rho.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / (np.arange(1, len(v)+1)) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Gradient‑descent optimiser for the step‑function problem.

The algorithm works directly on the probability simplex (non‑negative,
entries that sum to 1).  At each iteration we:
  1. compute the self‑convolution a∗a,
  2. form a soft‑max weighting of the convolution entries
     (parameter β controls how close the weight is to the true max),
  3. obtain the gradient of the smooth approximation of max(a∗a) as
        g = 2·(w * a_rev)[n−1 : n−1+n]    (linear convolution)
     where w are the soft‑max weights and a_rev is a reversed copy of a,
  4. take a step opposite to the gradient,
  5. enforce symmetry (a ← (a+a[::-1])/2) and project back onto the
     simplex.

The soft‑max temperature β is slowly increased (annealed) so the
optimiser first follows a smooth landscape and later focuses on the
actual maximum.  Occasionally a small random perturbation and a simple
restart are performed to avoid stagnation.  The best sequence found
within the time budget is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (exact, O(n log n))
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex (sum=1, v_i≥0)."""
    # shift to ensure positivity
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / (np.arange(1, v.size + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear convolution via FFT (real inputs → real output)
# ----------------------------------------------------------------------
def _linconv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the linear convolution of real vectors x and y using FFT."""
    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    # tiny negative values due to floating point are removed
    conv[conv < 0.0] = 0.0
    return conv


# ----------------------------------------------------------------------
# Fast scoring (the same formula used by evaluate_sequence for a simplex)
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """
    Evaluate the objective for a probability vector `seq`.
    Equivalent to evaluate_sequence(seq.tolist()) for valid sequences.
    """
    n = seq.size
    conv = _linconv_fft(seq, seq)           # a * a
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# One gradient descent step towards reducing the (soft) maximum of the convolution
# ----------------------------------------------------------------------
def _gradient_step(
    a: np.ndarray, beta: float, step: float
) -> tuple[np.ndarray, float]:
    """
    Perform a single projected gradient step.

    Parameters
    ----------
    a : np.ndarray
        Current probability vector (sum = 1, non‑negative).
    beta : float
        Soft‑max temperature (larger → closer to true max).
    step : float
        Learning‑rate for the gradient step.

    Returns
    -------
    a_new : np.ndarray
        Updated probability vector (projected onto the simplex).
    new_score : float
        Objective value after the update.
    """
    n = a.size

    # --- 1. self‑convolution -------------------------------------------------
    conv = _linconv_fft(a, a)                # length 2n‑1
    max_c = conv.max()

    # --- 2. soft‑max weights over the convolution entries --------------------
    # w_k ∝ exp(beta * (conv_k - max_c))  → avoids overflow
    exp_terms = np.exp(beta * (conv - max_c))
    w = exp_terms / exp_terms.sum()          # length 2n‑1, non‑negative, sums to 1

    # --- 3. gradient of the smooth max --------------------------------------
    # g_i = 2 * Σ_k w_k * a_{k-i}
    a_rev = a[::-1]                          # reversed a
    full = _linconv_fft(w, a_rev)            # length 3n‑2
    g = 2.0 * full[n - 1 : n - 1 + n]        # slice of length n

    # --- 4. gradient step ----------------------------------------------------
    a_new = a - step * g

    # --- 5. enforce symmetry (optional, improves stability) ---------------
    a_new = 0.5 * (a_new + a_new[::-1])

    # --- 6. project back onto simplex ----------------------------------------
    a_new = _project_simplex(a_new)

    # --- 7. compute the (exact) new objective -------------------------------
    new_score = _score(a_new)
    return a_new, new_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimises
    `evaluate_sequence`.  The routine runs a soft‑max‑guided gradient
    descent with an annealing schedule for β and occasional random
    perturbations.  The best sequence encountered before the time budget
    expires is returned.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + max(0.0, budget_s - 0.5)   # small safety margin

    # ------------------------------------------------------------------
    # Initialise the candidate (use the previously discovered construction if
    # it exists, otherwise fall back to a uniform distribution).
    # ------------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)   # may be present
        base = _project_simplex(base)                     # enforce simplex
    except Exception:
        n = 8000
        base = np.full(n, 1.0 / n, dtype=float)

    n = base.size
    a = base.copy()

    best = a.copy()
    best_score = _score(best)
    print(f"[Init] n = {n}, initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters
    # ------------------------------------------------------------------
    beta_start = 5.0
    beta_end = 200.0               # final temperature (approaches true max)
    max_iters_estimate = 30000     # rough upper bound – the loop stops early by time
    step_initial = 0.5
    step_min = 0.01

    last_improve_iter = 0
    iteration = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # anneal β linearly (clamp at beta_end)
        prog = min(1.0, iteration / max_iters_estimate)
        beta = beta_start + prog * (beta_end - beta_start)

        # simple learning‑rate schedule
        step = max(step_min, step_initial / np.sqrt(iteration))

        # one gradient step
        a, cur_score = _gradient_step(a, beta, step)

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve_iter = iteration
            print(
                f"[Iter {iteration:6d}] New best score = {best_score:.6f} "
                f"(β={beta:.1f}, step={step:.4f})"
            )

        # ------------------------------------------------------------------
        # occasional random perturbation to escape shallow plateaus
        # ------------------------------------------------------------------
        if iteration % 200 == 0:
            pert = best + 0.005 * (np.random.rand(n) - 0.5)
            pert = _project_simplex(pert)
            pert_score = _score(pert)
            if pert_score < best_score:
                best_score = pert_score
                best = pert.copy()
                a = pert.copy()
                last_improve_iter = iteration
                print(
                    f"[Iter {iteration:6d}] Perturbation improved best → {best_score:.6f}"
                )

        # ------------------------------------------------------------------
        # simple stagnation restart: if we have not improved for a while,
        # replace the current point with a fresh random simplex point.
        # ------------------------------------------------------------------
        if iteration - last_improve_iter > 1500:
            a = _project_simplex(np.random.rand(n))
            last_improve_iter = iteration
            print(f"[Iter {iteration:6d}] Restart due to stagnation")

    # ----------------------------------------------------------------------
    # Return the best sequence as a plain Python list
    # ----------------------------------------------------------------------
    print(f"[Done] Best score after {iteration} iterations = {best_score:.6f}")
    return best.tolist()
