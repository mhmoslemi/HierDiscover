def _optimise_subgradient(n, deadline, init_seq, max_iters=3000, eta0=0.1):
    a = init_seq.copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = 2.0 * n * _conv_full_fft(a).max()
    m = 1 << ((3 * n - 2).bit_length())
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        b = _conv_full_fft(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
        # compute subgradient
        max_val = b.max()
        mask = (b == max_val).astype(np.float64)
        # In case multiple peaks, average mask
        w = mask / mask.sum()
        g = _gradient_via_correlation(w, a, m)
        # step size schedule
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum
        # optional jitter
        if it % 300 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-5
            a = np.clip(a, 0, None)
            a /= a.sum()
    return best_seq, best_score

"""
Propose a low‑score step‑function sequence.

The scored quantity is 2·n·max(conv(a,a)) with Σa = 1 (the
normalisation is done inside ``evaluate_sequence``).  The goal is to
minimise the peak of the self‑convolution.  The algorithm works in
three stages:

1.  **Initial families** –  for each candidate length we generate a
    handful of deterministic “nice’’ sequences (arcsine law, symmetric
    Beta(α,α) for many α) and a few random Dirichlet draws.

2.  **Smooth optimisation** –  starting from the best seed we minimise
    a smooth approximation of the max‑operator
        F(a)= (1/λ)·log Σ_k exp(λ·(a*a)_k) .
    The gradient is obtained by a single FFT‑based correlation and an
    exponentiated‑gradient (mirror‑descent on the simplex) update is
    applied.  The temperature λ is increased slowly up to a large
    value (≈ 70) so that the smoothed objective approaches the true
    peak.

3.  **Sub‑gradient refinement** –  after the smooth phase we switch
    to the genuine max‑objective.  The sub‑gradient of the peak
    convolution entry is computed (again by a correlation) and an
    exponentiated‑gradient step is taken.  Occasional jitter helps to
    escape tiny plateaus.

Both optimisation phases respect a global time budget (default 1000 s)
and keep the best sequence ever seen.  If a historic solution
``height_sequence_1`` is present it is used as an extra starter.
The function returns the best sequence found as a plain Python list.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a power‑of‑two FFT.
    Returned length = 2·len(a)‑1.
    """
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, m)
    c = np.fft.irfft(A * A, m)
    return c[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k‑i} (correlation) using FFT.
    w – weight vector (length 2·n‑1), a – current sequence (length n),
    m – FFT length (power of two, at least len(w)+len(a)-1).
    Returns the slice g[0…n‑1].
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start:start + len(a)]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (α=½) on n points, Σa = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the grid x_i = (i+½)/n, Σa = 1."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _optimise_smooth(
    n: int,
    deadline: float,
    init_seq: np.ndarray,
    max_iters: int = 4000,
    eta0: float = 0.2,
    lambda0: float = 0.5,
    lambda_max: float = 30.0,
) -> tuple[np.ndarray, float]:
    """
    Refine ``init_seq`` (already normalised to sum=1) by minimising a
    smooth approximation of the max‑convolution peak.
    Returns the best sequence seen and its (unsmoothed) score.
    """
    a = np.clip(init_seq.astype(np.float64), 0.0, None)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()

    best_seq = a.copy()
    best_score = float(2.0 * n * _conv_full_fft(a).max())

    # FFT size sufficient for correlation w (len=2n‑1) with a (len=n)
    m = 1 << ((3 * n - 2).bit_length())   # >= (2n‑1)+(n‑1)

    lamb = lambda0
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current (unsmoothed) score – keep best ever
        b = _conv_full_fft(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} it={it:5d} score={best_score:.6f}")

        # smooth‑max (soft‑max) weights
        maxb = b.max()
        exp_arg = lamb * (b - maxb)          # ≤ 0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.shape[0])
        else:
            w /= w_sum

        # gradient via correlation
        g = _gradient_via_correlation(w, a, m)

        # exponentiated‑gradient (mirror descent) step
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # occasional tiny jitter to escape plateaus
        if it % 800 == 0 and np.random.rand() < 0.15:
            a += np.random.randn(n) * 1e-5
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        # anneal the temperature
        lamb = min(lambda_max, lamb * 1.01)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _optimise_subgradient(
    n: int,
    deadline: float,
    init_seq: np.ndarray,
    max_iters: int = 3000,
    eta0: float = 0.1,
) -> tuple[np.ndarray, float]:
    """
    Refine ``init_seq`` by directly minimising the true peak
    max(conv(a,a)).  Uses a sub‑gradient (the gradient of the entry
    attaining the maximum) together with an exponentiated‑gradient
    update on the simplex.
    """
    a = np.clip(init_seq.astype(np.float64), 0.0, None)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()

    best_seq = a.copy()
    best_score = float(2.0 * n * _conv_full_fft(a).max())

    m = 1 << ((3 * n - 2).bit_length())   # FFT length for correlation

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        b = _conv_full_fft(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} sub‑it={it:5d} score={best_score:.6f}")

        maxb = b.max()
        # mask of indices attaining the maximum (handle numerical noise)
        mask = np.isclose(b, maxb, atol=1e-12).astype(np.float64)
        mask_sum = mask.sum()
        if mask_sum == 0.0:
            w = np.full_like(b, 1.0 / b.shape[0])
        else:
            w = mask / mask_sum

        # sub‑gradient of the max entry
        g = _gradient_via_correlation(w, a, m)

        # step size schedule
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # jitter every few hundred iterations
        if it % 600 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-5
            a = np.clip(a, 0.0, None)
            a /= a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a very low
    ``evaluate_sequence`` score.  The routine explores several lengths,
    a variety of initial families and then applies two optimisation
    phases (smooth → sub‑gradient).  The best sequence found before the
    time budget expires is returned as a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  Historic solution, if available
    try:
        hist = np.array(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        hist = np.clip(hist, 0.0, None)
        if hist.sum() > 0.0:
            hist = hist / hist.sum()
            hist_score = float(2.0 * len(hist) * _conv_full_fft(hist).max())
            overall_best_seq = hist.copy()
            overall_best_score = hist_score
            print(f"Historic sequence: length={len(hist)} score={hist_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Choose lengths (larger tends to lower the theoretical bound)
    candidate_lengths = [2000, 4000, 8000, 12000, 20000]

    for n in candidate_lengths:
        if time.time() >= deadline:
            break

        print(f"=== Exploring length n={n} ===")

        # ----- a) generate a pool of starting points --------------------
        seeds = []

        # arcsine law (always a good baseline)
        seeds.append(_arcsine_distribution(n))

        # deterministic Beta sweep
        alphas = np.linspace(0.15, 1.2, 9)
        alphas = np.concatenate([alphas, np.random.uniform(0.05, 1.5, size=6)])
        for alpha in alphas:
            seeds.append(_beta_distribution(n, float(alpha)))

        # a few random Dirichlet draws
        for _ in range(4):
            r = np.random.rand(n)
            seeds.append(r / r.sum())

        # track the best seed after a quick sweep
        best_seed = None
        best_seed_score = float("inf")
        for s in seeds:
            cur_score = float(2.0 * n * _conv_full_fft(s).max())
            if cur_score < best_seed_score:
                best_seed_score = cur_score
                best_seed = s.copy()
        print(f"Length {n:5d}: best pre‑optimisation score = {best_seed_score:.6f}")

        # ----- b) smooth optimisation -----------------------------------
        slice_deadline = time.time() + max(0.0, (deadline - time.time()) * 0.5)
        refined_seq, refined_score = _optimise_smooth(
            n=n,
            deadline=slice_deadline,
            init_seq=best_seed,
            max_iters=5000,
            eta0=0.25,
            lambda0=0.5,
            lambda_max=70.0,
        )
        print(f"After smooth phase: score={refined_score:.6f}")

        # ----- c) sub‑gradient refinement --------------------------------
        if time.time() < deadline:
            final_seq, final_score = _optimise_subgradient(
                n=n,
                deadline=deadline,
                init_seq=refined_seq,
                max_iters=6000,
                eta0=0.12,
            )
        else:
            final_seq, final_score = refined_seq, refined_score

        print(f"After sub‑gradient: score={final_score:.6f}")

        # ----- d) update global best ------------------------------------
        if final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = final_seq.copy()
            print(f"*** New global best → length={n}  score={overall_best_score:.6f}")

        if overall_best_score <= 1.503:
            # early exit – target met
            break

    # ------------------------------------------------------------------
    # 2️⃣  Fallback – ensure something is returned
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = float(2.0 * n * _conv_full_fft(overall_best_seq).max())
        print(f"Fallback: length={n} score={overall_best_score:.6f}")

    print(
        f"FINAL best: length={len(overall_best_seq)} score={overall_best_score:.6f}"
    )
    # convert to plain Python list for the external runner
    return overall_best_seq.tolist()
