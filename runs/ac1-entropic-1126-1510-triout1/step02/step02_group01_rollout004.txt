"""
Hybrid search for a step‑function with a low evaluation score.

The algorithm combines three ideas

1. **Projected Adam on a smooth‑max surrogate** –  we minimise the
   maximum value of the auto‑convolution  f∗f  by a soft‑max
   approximation.  The gradient is obtained by a single convolution,
   the update is performed with the Adam optimiser and the iterate is
   always projected onto the simplex {a≥0, Σa = √(2 n)}.

2. **Periodic LP‑updates** –  for the current iterate  a  we solve the
   linear programme  

       maximise   Σb_j
       subject to  (a∗b)_k ≤ M   for all k,
                   b_j ≥ 0,

   where  M = max_k (a∗a)_k  is the current worst convolution value.
   The solution  b₀  is then rescaled to satisfy  Σb = √(2 n)  and mixed
   into the current iterate (mixing factor  t≈0.1).  This step can
   dramatically pull down the peak.

3. **Multi‑restart + block‑wise random refinements** –  a few
   independent restarts (random starts or perturbations of the best
   solution) explore the landscape, and a cheap block‑search is run
   at the end to catch easy improvements.

The whole search respects the 1000 s time budget; if the budget runs
out the best sequence found so far is returned.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear auto‑convolution using FFT (real‐valued)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    c = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return c


# ----------------------------------------------------------------------
def max_conv(a: np.ndarray) -> float:
    """Maximum entry of a∗a."""
    return conv_fft(a).max()


# ----------------------------------------------------------------------
def grad_softmax(a: np.ndarray, conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate  τ·log Σ exp(conv_k/τ)  w.r.t. a.
    """
    n = a.shape[0]
    max_val = conv.max()
    # stable soft‑max
    diff = conv - max_val                 # ≤ 0
    w = np.exp(diff / tau)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[conv.argmax()] = 1.0
        w_sum = 1.0
    w /= w_sum

    # Compute Σ_k w_k * a_{k-i}   via a single convolution
    conv_wr = np.convolve(w, a[::-1])                 # length 3n‑2
    sum_term = conv_wr[n - 1 : n - 1 + n]            # size n
    grad = 2.0 * sum_term

    # diagonal correction (terms with i=j)
    double_idx = 2 * np.arange(n)
    mask = double_idx < conv.shape[0]
    grad[mask] -= w[double_idx[mask]] * a[mask]
    return grad


# ----------------------------------------------------------------------
def adam_optimize(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iter: int = 200_000,
    tau_start_factor: float = 1.0,
    tau_decay: float = 0.9995,
    t_min: float = 1e-6,
    lr_factor: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate with projection onto the simplex.
    Returns the final iterate and its max‑convolution value.
    """
    n = a0.shape[0]
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Adam step size  (≈ average entry)
    eta0 = lr_factor * target_sum / n

    # initial temperature = tau_start_factor * current max
    conv = conv_fft(a)
    tau = max(tau_start_factor * conv.max(), t_min)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(a)
        cur_max = conv.max()

        # gradient of smooth‑max (high temperature) or top‑k (low temperature)
        if tau > 0.02 * cur_max:          # smooth‑max regime
            grad = grad_softmax(a, conv, tau)
        else:                             # switch to hard‑max using top‑k = 5
            # top‑k sub‑gradient, similar to the reference implementation
            top_k = 5
            idxs = np.argpartition(-conv, top_k - 1)[:top_k]
            grad = np.zeros_like(a)
            i = np.arange(n)
            for k in idxs:
                j = k - i
                mask = (j >= 0) & (j < n)
                grad[mask] += 2.0 * a[j[mask]]
                diag = mask & (i == j)
                grad[diag] -= a[i[diag]]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 4000 == 0:
            a += np.random.randn(n) * target_sum * 1e-4
            a = project_to_simplex(a, target_sum)

        # cool temperature
        tau = max(tau * tau_decay, t_min)

    final_max = conv_fft(a).max()
    return a, final_max


# ----------------------------------------------------------------------
def lp_update(a: np.ndarray, cur_max: float, target_sum: float) -> tuple[np.ndarray, float]:
    """
    One LP‑based refinement step.
    Solves max Σ b_j s.t. (a∗b)_k ≤ cur_max, b≥0,
    rescales b to the correct ℓ¹‑norm and mixes it into a.
    Returns the new a and its max‑convolution.
    """
    n = a.shape[0]

    # Build the convolution matrix G where (G b)_k = (a∗b)_k
    rows = 2 * n - 1
    G = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        G[i : i + n, i] = a

    # Linear programme (maximise Σb  ≡  minimise –Σb)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(rows, cur_max, dtype=np.float64)

    res = linprog(
        c,
        A_ub=G,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True},
    )

    if not res.success:
        return a, cur_max

    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 < 1e-12:          # degenerate solution – nothing to do
        return a, cur_max

    # Rescale to the required ℓ¹‑norm
    scale = target_sum / sum_g0
    g = g0 * scale

    # Mix with the current iterate
    t_mix = 0.10                     # mixing factor
    a_new = (1.0 - t_mix) * a + t_mix * g
    a_new = np.maximum(a_new, 0.0)
    a_new = project_to_simplex(a_new, target_sum)

    new_max = conv_fft(a_new).max()
    return a_new, new_max
