def _simulated_annealing(a, n, deadline, max_iter=20000, init_temp=1e-2, final_temp=1e-6):
    best_a = a.copy()
    best_score = compute_score(a)
    cur_a = a.copy()
    cur_score = best_score
    T0 = init_temp
    T1 = final_temp
    it = 0
    while time.time() < deadline and it < max_iter:
        # geometric cooling schedule
        t_frac = it / max_iter
        T = T0 * (T1/T0)**t_frac
        # propose move
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i==j: continue
        # choose delta proportion of a[i]
        delta = np.random.rand() * 0.1 * cur_a[i]   # up to 10% of mass
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        if new_a[i] < 0: continue
        # re-normalize (should be already sum 1)
        # compute new score
        new_score = compute_score(new_a)
        if new_score < cur_score or np.random.rand() < np.exp(-(new_score-cur_score)/max(T,1e-12)):
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = new_a.copy()
        it += 1
    return best_a, best_score

def conv_max(a):
    # compute full linear convolution via FFT
    n = a.shape[0]
    m = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, m)
    c = np.fft.irfft(A*A, m)
    return c[:2*n-1].max()

"""
Propose a low‑score step‑function sequence.

The score is 2·n·max(conv(a,a)) while the ℓ₁‑mass is irrelevant
(because scaling cancels).  We therefore seek a probability vector
(a_i≥0, Σa_i=1) whose self‑convolution (distribution of the sum of two
independent draws) is as flat as possible.

Algorithm
---------
1. **Initial families** – a few hand‑crafted shapes (uniform,
   arcsine, symmetric Beta(α,α)) and random Dirichlet draws give a
   diverse set of starting points.

2. **Smooth optimisation** – minimise a smooth approximation of the
   max‑convolution:
        F_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k)
   The gradient can be obtained by a single FFT‑based correlation.
   An exponentiated‑gradient (mirror‑descent on the simplex) step is
   taken and the temperature λ is increased gradually.  This step
   quickly drives the sequence into a low‑peak region.

3. **Greedy simulated annealing** – from the best sequence obtained
   above we perform a simple SA on the simplex:
   repeatedly move a small amount of mass from a source index to a
   target index, accepting the move if it improves the true objective
   or with a probability that decays with a temperature schedule.
   This stochastic refinement often escapes the plateaus left by the
   smooth optimiser.

4. **Multiple lengths & restarts** – the whole pipeline is run for a
   handful of lengths (2000…10000).  The best sequence seen before the
   1000 s budget expires is returned.

All heavy work (convolution, correlation) uses NumPy FFTs, which are
fast enough to evaluate several‑hundred thousand candidates well within
the time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (α=½) on n points, Σa = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the grid x_i = (i+½)/n, Σa=1."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, m)
    c = np.fft.irfft(A * A, m)
    return c[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute  g_i = 2·Σ_k w_k·a_{k‑i}  (correlation) using FFT.
    w – weight vector of length 2·n‑1, a – current sequence (length n),
    m – FFT length (power of two, ≥ len(w)+len(a)-1).
    Returns an array of length n.
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start : start + len(a)]


# ----------------------------------------------------------------------
def _smooth_optimise(
    a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 3000,
    eta0: float = 0.2,
    lambda0: float = 0.5,
    lambda_max: float = 30.0,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the smoothed max‑convolution (soft‑max) objective.
    Returns the best sequence seen and its *real* (unsmoothed) score.
    """
    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()

    best_seq = a.copy()
    best_score = 2.0 * n * _conv_full_fft(a).max()

    # FFT length sufficient for correlation (len(w)=2n‑1, len(a)=n)
    m = 1 << ((3 * n - 2).bit_length())

    lamb = lambda0
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current (unsmoothed) score
        conv = _conv_full_fft(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} it={it:5d} score={best_score:.6f}")

        # soft‑max over convolution entries
        maxb = conv.max()
        exp_arg = lamb * (conv - maxb)               # ≤0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.shape[0]
        else:
            w /= w_sum

        # gradient (correlation) and exponentiated‑gradient step
        g = _gradient_via_correlation(w, a, m)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)                     # stays non‑negative
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum

        # occasionally inject a tiny jitter – helps escaping tiny plateaus
        if it % 400 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        # increase temperature λ slowly
        lamb = min(lambda_max, lamb * 1.01)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _simulated_annealing(
    a: np.ndarray,
    n: int,
    deadline: float,
    max_iter: int = 15000,
    init_temp: float = 0.02,
    final_temp: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    Greedy simulated‑annealing on the simplex.
    Moves a small amount of mass between two indices and accepts the
    move with the usual Metropolis probability.
    """
    cur = a.copy()
    cur_score = 2.0 * n * _conv_full_fft(cur).max()
    best = cur.copy()
    best_score = cur_score

    # pre‑compute cumulative distribution for source sampling
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # linear cooling schedule in the exponent of the temperature
        t_frac = it / max_iter
        temp = init_temp * (final_temp / init_temp) ** t_frac
        # pick source proportional to its current mass (biggest contributors)
        src = np.random.choice(n, p=cur / cur.sum())
        dst = np.random.randint(n)
        if src == dst:
            continue

        # propose moving a random fraction (up to 30%) of the source mass
        max_move = 0.3 * cur[src]
        if max_move <= 0.0:
            continue
        delta = np.random.rand() * max_move
        new_seq = cur.copy()
        new_seq[src] -= delta
        new_seq[dst] += delta

        # evaluate
        new_score = 2.0 * n * _conv_full_fft(new_seq).max()

        # acceptance test
        if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur = new_seq
            cur_score = new_score
            if new_score < best_score:
                best = new_seq.copy()
                best_score = new_score
                print(f"[{time.time():.1f}] n={n:5d} SA it={it:5d} score={best_score:.6f}")

    return best, best_score

    short simulated‑annealing refinement.  The best sequence seen
    before the time budget expires is returned.
    """

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # 0️⃣  historic sequence (if any)

    overall_best_score = float("inf")

            hist_score = 2.0 * len(hist) * _conv_full_fft(hist).max()
            overall_best_seq = hist.copy()
            overall_best_score = hist_score
            print(f"Historic sequence: length={len(hist)} score={hist_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  lengths to explore – a few values give a good trade‑off.
    lengths = [2000, 4000, 6000, 8000, 10000]

    # allocate a modest slice of the budget per length
    per_len_budget = (deadline - time.time()) / max(len(lengths), 1)

    for n in lengths:
        if time.time() >= deadline:
            break

        # ------------------------------------------------------------------
        # a) generate a small pool of diverse initial candidates
        init_candidates = []

        # uniform – the benchmark (score = 2)
        init_candidates.append(np.full(n, 1.0 / n))

        # arcsine law
        init_candidates.append(_arcsine_distribution(n))

        # a handful of symmetric beta shapes

        # ------------------------------------------------------------------
