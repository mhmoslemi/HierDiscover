"""
propose_candidate()
------------------

Search for a non‑negative step‑function (a list of heights) that minimises

    score = 2·n·max(conv(a, a)) / (sum(a)**2)

where conv(a, a) is the full linear convolution of a with itself.
The score is scale‑invariant, therefore the algorithm always works with
vectors whose ℓ1‑norm equals √(2 n).

The search consists of three stages:

1. **Extensive parametric scan** – families of monotone shapes
   (exponential decay, inverse‑sqrt, power‑law and mixtures of two
   exponentials) are generated, ℓ1‑normalised and scored.  The best
   candidate from this stage is kept.

2. **Projected sub‑gradient descent** – starting from the best candidate,
   the true non‑smooth objective is minimised by repeatedly moving
   opposite to a sub‑gradient of the active (max) convolution entry.
   An Adam‑style update is used and the iterate is projected onto the
   simplex {a ≥ 0, Σa = √(2 n)}.  Random perturbations are injected when
   progress stalls.

3. **Random‑block refinement** – a short uniform random‑block search is
   performed to escape possible local minima.

All operations obey the security checks performed by `evaluate_sequence`
(the optimiser never creates negative values and always keeps the
ℓ1‑norm fixed, so the final score is unchanged by the clipping steps).

The function runs for at most the allotted time (≈1000 s) and returns the
best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}.
    """
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    # Duchi et al. 2008 algorithm – uses a sort (O(n log n))
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would be projected to zero; put all mass on first entry
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution using FFT (output length len(u)+len(v)-1)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Return the full convolution a * a (via FFT)."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _score(a: np.ndarray, size_fft: int) -> float:
    """Fast score = max convolution (the scaling cancels out)."""
    conv = _self_convolution(a, size_fft)
    return float(conv.max())


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Return a list of heights with a low evaluation score."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1) problem size
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)            # ℓ1‑norm we keep
    size_fft = 1 << ((2 * n - 1).bit_length())   # FFT size ≥ 2n‑1

    best_a = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to test a candidate quickly
    def try_candidate(a_raw: np.ndarray, description: str = "") -> None:
        nonlocal best_a, best_score
        a = _project_to_simplex(a_raw, target_sum)
        sc = _score(a, size_fft)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{description}] new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Uniform baseline
    try_candidate(np.ones(n), "uniform")

    # ------------------------------------------------------------------
    # 2) Parametric families (dense grid)
    idx = np.arange(n, dtype=float)

    # ---- Exponential decay -------------------------------------------------
    lam_min, lam_max = 0.0001, 0.0050
    n_lam = 1200
    for lam in np.linspace(lam_min, lam_max, n_lam):
        raw = np.exp(-lam * idx)
        try_candidate(raw, f"exp lam={lam:.6f}")

    # ---- Inverse‑sqrt -------------------------------------------------------
    for beta in np.linspace(0.01, 10.0, 300):
        raw = 1.0 / np.sqrt(idx + beta + 1.0)
        try_candidate(raw, f"inv_sqrt β={beta:.4f}")

    # ---- Power‑law -----------------------------------------------------------
    for beta in np.linspace(0.0, 5.0, 150):
        for p in np.linspace(0.2, 2.5, 150):
            raw = 1.0 / (idx + beta + 1.0) ** p
            try_candidate(raw, f"power β={beta:.2f} p={p:.2f}")

    # ---- Mixture of two exponentials (random sampling) ----------------------
    n_mix = 800
    for _ in range(n_mix):
        lam1 = np.random.uniform(lam_min, lam_max)
        lam2 = np.random.uniform(lam_min, lam_max)
        w = np.random.rand()
        raw = w * np.exp(-lam1 * idx) + (1.0 - w) * np.exp(-lam2 * idx)
        try_candidate(raw, "mix_exp")

    # ------------------------------------------------------------------
    # 3) Refine the best candidate with projected sub‑gradient descent (Adam)
    if best_a is None:
        # Fallback – uniform if somehow nothing was kept
        best_a = _project_to_simplex(np.ones(n), target_sum)
        best_score = _score(best_a, size_fft)

    a = best_a.copy()
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_iter = 0

    # step‑size – a few times the typical entry size
    eta = target_sum * 0.5 / n
    no_improve = 0

    while time.time() < deadline and t_iter < 300_000:
        conv = _self_convolution(a, size_fft)
        cur_score = float(conv.max())

        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            no_improve = 0
            print(f"[{t_iter:07d}] new best = {best_score:.6f}")
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # Sub‑gradient of the active convolution entry (or average if tied)
        max_val = cur_score
        # tolerance to collect all (almost) active indices
        tol = max(1e-12, max_val * 1e-12)
        ks = np.where(conv >= max_val - tol)[0]

        grad = np.zeros_like(a)
        for k in ks:
            start_i = max(0, k - (n - 1))
            end_i = min(n - 1, k)
            idxs = np.arange(start_i, end_i + 1)
            grad[idxs] += 2.0 * a[k - idxs]
        grad /= max(1, len(ks))

        # --------------------------------------------------------------
        # Adam update (followed by simplex projection)
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a = _project_to_simplex(a_new, target_sum)

        # slowly decay the learning rate
        eta *= 0.99995

        # occasional random kick to escape plateaus
        if (t_iter % 8000 == 0) or no_improve > 5000:
            noise = np.random.randn(n) * target_sum * 1e-3
            a = _project_to_simplex(a + noise, target_sum)
            no_improve = 0
            print(f"[{t_iter:07d}] random perturbation")

    # ------------------------------------------------------------------
    # 4) Final random‑block search (few hundred trials)
    for _ in range(300):
        nb = np.random.randint(2, 7)
        cuts = np.sort(
            np.random.choice(n - 1, nb - 1, replace=False) + 1
        )
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.zeros(n)
        for blk, vblk in zip(blocks, vals):
            cand[blk] = vblk
        cand = _project_to_simplex(cand, target_sum)
        sc = _score(cand, size_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[block] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
