"""
Propose a step‑function (list of non‑negative heights) that minimizes the
`evaluate_sequence` objective.

Algorithm
---------
1.  Initialise a distribution on the simplex (uniform, arcsine, random or the
    previously found best sequence).
2.  Minimise a *smooth* approximation of the maximal self‑convolution value
    using an Adam‑style projected gradient descent.
    * The smooth max is the soft‑max:   w_k ∝ exp(α·b_k), where b = a*a.
    * The gradient of the smoothed objective is
          g = 2·conv(w, a_rev)[n‑1 : n‑1+n]   (a_rev = a reversed).
      Both convolutions are computed with a single FFT of size ≥ 3·n.
    * After each gradient step the vector is projected onto the probability
      simplex {x≥0, Σx=1}.
    * The temperature α starts small (α≈2) and is increased logarithmically
      towards a maximum (α≈200) so that the optimisation first smooths the
      landscape and later focuses on the true maximum.
3.  The inner loop runs until the allotted time slice expires, keeping the
    best sequence seen (according to the internal estimate
    `score = 2·n·max(b)`).
4.  Several lengths (≈500–1.5·base) and initialisations are tried; the overall
    best sequence is returned.
5.  Finally the best candidate is evaluated with the supplied black‑box
    `evaluate_sequence` to obtain the official score.

The method is purely gradient‑based (no integer/LP steps) and exploits the
FFT for O(n log n) per iteration, allowing many thousands of iterations
within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    rho_arr = np.where(u - cssv / ind > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine probability distribution of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_sequence(n: int, kind: str) -> np.ndarray:
    """Create an initial feasible sequence of length n."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif kind == "arcsine":
        a = _arcsine_distribution(n)
    elif kind == "prev":
        # Use the globally supplied sequence if it matches the length.
        try:
            a = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if a.size != n:
                raise ValueError
        except Exception:
            a = _arcsine_distribution(n)
        if a.sum() == 0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / a.sum()
    else:
        a = _arcsine_distribution(n)
    return _project_onto_simplex(a, s=1.0)


# ----------------------------------------------------------------------
def _smooth_max_optimisation(
    a0: np.ndarray,
    n: int,
    deadline: float,
    target_score: float = 1.503,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected gradient descent on the soft‑max of the self‑convolution.
    Returns the best sequence (as a NumPy array) and the corresponding internal
    score `2·n·max(b)`.
    """
    # FFT size large enough for convolutions of lengths (2n‑1) + n.
    fft_len = 1 << ((3 * n - 2 - 1).bit_length())

    a = a0.copy()
    best_seq = a.copy()
    # Initial internal score
    A = np.fft.rfft(a, n=fft_len)
    b_full = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    best_score = float(2.0 * n * b_full.max())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.10
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- convolution a*a -------------------------------------------------
        A = np.fft.rfft(a, n=fft_len)
        b_full = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
        max_b = float(b_full.max())
        cur_score = 2.0 * n * max_b

        # keep the best seen so far (cheap internal evaluation)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] n={n} it={it} score={best_score:.6f}"
            )
            if best_score <= target_score:
                # early exit from inner loop
                break

        # ----- soft‑max weights ------------------------------------------------
        # temperature schedule: start low, rise logarithmically, cap at 200
        if it < 1000:
            alpha = 2.0
        else:
            alpha = min(200.0, 2.0 * (1.0 + np.log(it / 1000.0 + 1.0)))

        # safe soft‑max (shift by max_b)
        w_raw = np.exp(alpha * (b_full - max_b))
        w = w_raw / w_raw.sum()  # Σw = 1

        # ----- gradient = 2·conv(w, a_rev) ------------------------------------
        W_fft = np.fft.rfft(w, n=fft_len)
        a_rev = a[::-1]
        A_rev_fft = np.fft.rfft(a_rev, n=fft_len)
        conv_full = np.fft.irfft(W_fft * A_rev_fft, n=fft_len)
        # slice the relevant n entries
        g = 2.0 * conv_full[n - 1 : n - 1 + n]

        # ----- Adam update ----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a, s=1.0)

        # occasional tiny random perturbation (helps escape flat plateaux)
        if it % 2000 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score
