def convolve_self(a):
    n = a.shape[0]

def gradient_softmax_max(a, alpha):
    conv = convolve_self(a)  # length 2n-1
    max_val = conv.max()
    # compute softmax weights
    # subtract max_val to avoid overflow
    exp_vals = np.exp(alpha * (conv - max_val))
    w = exp_vals / exp_vals.sum()
    # gradient = 2 * alpha * (w convolve a_rev)[n-1:n-1+n]
    a_rev = a[::-1]
    grad_full = convolve_w_rev(w, a_rev)
    grad = 2.0 * alpha * grad_full[(len(a)-1): (len(a)-1+len(a))]
    return grad

def simplex_proj(v):
    """Project v onto the simplex {x >=0, sum x =1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v):
    """Projection onto probability simplex (non-negative, sum=1)."""
    # Works for any shape 1D array.
    v = np.asarray(v)
    if v.ndim !=1:
        raise ValueError
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    # find rho = max {j | u_j - cssv_j / j > 0}
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non-positive; return uniform
        return np.full_like(v, 1.0/n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Optimise a non‑negative step‑function (a probability vector) for the
evaluation metric

    2·n·max(conv(a,a)) / (sum(a)^2)

The metric is invariant under scaling, so we keep the vector normalised
to sum = 1 (the probability simplex).  The optimisation uses a
smoothed version of the max‑convolution:  

    f(a) = log Σ_i exp(α·(a∗a)_i)

For α→∞ this equals the original max.  f is convex and its gradient
has a simple closed form that can be evaluated with three FFTs per
iteration.  We run a projected (additive) gradient descent with a
decreasing step‑size, gradually increasing α from 1 to 2000.  The
algorithm is run for several sequence lengths and several initial
vectors (arcsine, uniform, random Dirichlet, possibly a provided
`height_sequence_1`).  The best sequence found before the time budget
expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def convolve_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Return a∗a using an FFT of length a power‑of‑two.
    Result length is 2·len(a)‑1.
    """
    n = a.shape[0]
    N = next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    conv = np.fft.irfft(A * A, N)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def convolve_w_rev_fft(w: np.ndarray, a_rev: np.ndarray) -> np.ndarray:
    """
    Convolve w (length m) with the reversed vector a_rev (length n).
    Returns a vector of length m+n‑1.
    """
    m = w.shape[0]
    n = a_rev.shape[0]
    N = next_pow2(m + n - 1)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a_rev, N)
    conv = np.fft.irfft(W * A, N)[: m + n - 1]
    return conv


# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex_projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def compute_metric(seq: np.ndarray) -> float:
    """
    Metric for a vector already normalised to sum = 1:
        2·n·max(conv(seq,seq))
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def gradient_alpha(a: np.ndarray, conv_self: np.ndarray, alpha: float):
    """
    Gradient of  f(a)=log Σ_i exp(α·(a∗a)_i) .
    Returns the gradient vector (size n) and the softmax weights w.
    """
    n = a.shape[0]
    # softmax over the convolution entries – subtract max for stability
    max_c = conv_self.max()
    exp_vals = np.exp(alpha * (conv_self - max_c))
    w = exp_vals / exp_vals.sum()

    # g_k = Σ_i w_i a_{i‑k}  → 2·α·g
    a_rev = a[::-1]
    grad_full = convolve_w_rev_fft(w, a_rev)          # length 3·n‑2
    grad = 2.0 * alpha * grad_full[(n - 1) : (n - 1 + n)]
    return grad, w


# ----------------------------------------------------------------------
def _optimize_sequence(
    init_seq: list[float],
    deadline: float,
    max_iters: int = 20000,
) -> tuple[list[float], float]:
    """
    Projected gradient descent on the smoothed objective.
    Returns the best sequence (as a list) discovered before the deadline
    and its metric value.
    """
    a = np.array(init_seq, dtype=np.float64)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_metric(best_a)

    # Parameters for the continuation on α
    alpha_start = 1.0
    alpha_max = 2000.0

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- evaluate current point ---------------------------------
        conv_self = convolve_self_fft(a)
        cur_score = 2.0 * n * conv_self.max()

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()

        # ----- schedule α (linear increase) --------------------------
        alpha = alpha_start + (alpha_max - alpha_start) * min(it / max_iters, 1.0)

        # ----- gradient of the smoothed max ---------------------------
        grad, _ = gradient_alpha(a, conv_self, alpha)

        # ----- step size (decreasing) --------------------------------
        lr = 0.5 / np.sqrt(it)

        # ----- additive update + simplex projection --------------------
        a = a - lr * grad
        a = simplex_projection(a)

        # ----- occasional jitter to escape plateaus -------------------
        if it % 2000 == 0:
            # mix a tiny amount of the uniform distribution
            epsilon = 2e-4
            a = (1 - epsilon) * a + epsilon * (1.0 / n)
            a = simplex_projection(a)

        # ----- optional progress output --------------------------------
        if it % 5000 == 0:
            print(
                f"    iter {it:6d} | metric {cur_score:.6f} | α {alpha:.1f} | lr {lr:.5f}"
            )

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> list[float]:
    """
    First n coefficients of (1‑z)^‑1/2, normalised to L1‑norm = 1.
    """
    seq = []
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    seq = [x / total for x in seq]
    return seq
