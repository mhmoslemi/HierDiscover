def simplex_projection(v, s):
    if s < 0: raise error
    # shift to ensure sum = s
    # Based on algorithm by Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def project_onto_simplex(v, s):
    """Projects vector v onto nonnegative simplex sum=v_i = s."""
    # Implementation from Duchi 2008
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # Find rho
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all negative? set all zeros maybe
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    # Project onto simplex sum=S
    a = a / a.sum() * S

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Init score: {best_score}")
    # Set hyperparameters
    total_iters = 5000
    beta_start = 0.5
    beta_end = 40.0
    lr_start = 0.5
    lr_end = 0.02

    it = 0
    while time.time() < deadline and it < total_iters:
        it += 1
        # Compute convolution
        conv = np.convolve(a, a)
        max_conv = conv.max()
        # compute softmax weights
        beta = beta_start + (beta_end - beta_start) * (it / total_iters)
        # prevent overflow
        exp_vals = np.exp(beta * (conv - max_conv))
        w = exp_vals / exp_vals.sum()
        # compute gradient using correlation
        grad = 2.0 * np.correlate(w, a, mode='valid')
        # step size schedule
        lr = lr_start + (lr_end - lr_start) * (it / total_iters)
        a_new = a - lr * grad
        # Project onto simplex sum=S
        a_new = np.maximum(a_new, 0)
        # Simple projection:
        a_new = a_new / a_new.sum() * S
        a = a_new

        # evaluate
        if it % 10 == 0:  # evaluate less often for speed?
            try:

                best_seq = a.copy()

"""
Gradient‑descent + simplex projection + random‑pair polishing.
-----------------------------------------------------------------
The evaluation score is scale‑invariant, so we fix the ℓ1‑norm to
S = √(2·n).  The objective 2·n·max(conv)/S² reduces to simply
max(conv) when this normalization is used.

We minimise max(conv) by a smooth approximation:
    w_k ∝ exp(β·(conv_k – max(conv)) )
which concentrates weight on the current peak for large β.
The gradient of the soft‑max objective is
    ∇ = 2·corr(w, a)            (correlation, “valid” mode)
and we take a projected gradient step on the non‑negative simplex.
β and the step size are annealed during optimisation.
After the gradient phase a short random‑pair local search (mass
swaps preserving the ℓ1‑norm) is performed to polish the result.

The algorithm runs until the allotted time budget expires and returns
the best sequence found.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Helper: project a vector onto the non‑negative simplex with prescribed sum.
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Return the Euclidean projection of v onto {x≥0, Σx = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)
    # Duchi et al. (2008) algorithm.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Helper: cheap random pairwise mass‑swap (keeps sum exactly target_sum).
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Move a small amount of mass from one index to another (preserves sum)."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    # draw delta uniformly from (‑σ/2, σ/2)
    delta = (np.random.rand() - 0.5) * sigma
    # keep both entries non‑negative
    if delta > 0:
        delta = min(delta, seq[j])
    else:
        delta = -min(-delta, seq[i])

    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq

# ----------------------------------------------------------------------
# FFT‑based convolution (full) – returns a*a of length 2n‑1.
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, nfft)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------
# FFT‑based correlation: corr = np.correlate(w, a, mode='valid')
# ----------------------------------------------------------------------
def _fft_correlate(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = w.shape[0]                     # = 2n‑1
    a_rev = a[::-1]
    nfft = 1 << ((L + n - 1).bit_length())
    W = np.fft.rfft(w, nfft)
    A_rev = np.fft.rfft(a_rev, nfft)
    full = np.fft.irfft(W * A_rev, nfft)[: L + n - 1]
    start = n - 1
    return full[start : start + n]    # length n (valid correlation)

# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Optimise a non‑negative step‑function by gradient descent on a
    smooth approximation of max(conv) and a final random‑pair polishing.
    Returns the best sequence (list of floats) found before the time limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the harness
    deadline = start_time + max(0.0, budget_s - 2.0)

    # --------------------------------------------------------------
    # Determine problem size.
    # --------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        n0 = base.size
    except Exception:
        n0 = 2000

    # Use a larger length than the previous best – more degrees of freedom.
    N = min(20000, max(8000, n0 * 2))
    S = math.sqrt(2.0 * N)               # ℓ1‑norm we enforce (score = max(conv))

    # --------------------------------------------------------------
    # Initialise a feasible point (uniform distribution on the simplex).
    # --------------------------------------------------------------
    a = np.full(N, S / N, dtype=float)
    # add a tiny random perturbation to break symmetry
    a += 0.001 * S * np.random.rand(N) / N
    a = _project_onto_simplex(a, S)

    # keep the best candidate seen so far
    best_seq = a.copy()
    conv = _fft_convolve(a)
    best_score = conv.max()               # with sum = S, score == max(conv)
    print(f"[Init] length={N}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Gradient‑descent phase (soft‑max approximation of the max).
    # --------------------------------------------------------------
    max_iters = 20000                     # upper bound – early stop by time.
    beta_start = 0.5
    beta_end = 80.0
    lr_start = 0.5
    lr_end = 0.01

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # 1) convolution and soft‑max weights
        conv = _fft_convolve(a)
        max_conv = conv.max()

        # schedule parameters
        t_frac = it / max_iters
        beta = beta_start + (beta_end - beta_start) * t_frac
        lr = lr_start + (lr_end - lr_start) * t_frac

        # soft‑max (stable)
        exp_vals = np.exp(beta * (conv - max_conv))
        w = exp_vals / exp_vals.sum()

        # 2) gradient = 2 * corr(w, a)
        grad = 2.0 * _fft_correlate(w, a)

        # 3) take a step and re‑project onto the simplex
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)          # enforce non‑negativity
        a_new = _project_onto_simplex(a_new, S)

        a = a_new

        # occasional evaluation (every 10 iterations)
        if it % 10 == 0:
            conv = _fft_convolve(a)
            score = conv.max()
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[Iter {it:05d}] New best score: {best_score:.6f}")

    # --------------------------------------------------------------
    # Polishing phase – random pairwise mass swaps (simulated annealing).
    # --------------------------------------------------------------
    polish_iters = 8000
    it = 0
    while time.time() < deadline and it < polish_iters:
        it += 1
        # linear cooling for the swap amplitude
        sigma = 0.4 * (1.0 - it / polish_iters) + 0.02
        cand = _random_pair_move(best_seq, sigma)
        conv = _fft_convolve(cand)
        score = conv.max()
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[Polish {it:05d}] New best score: {best_score:.6f}")

    # --------------------------------------------------------------
    # Return the best sequence as a plain Python list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    return best_seq.tolist()
