"""
Propose a step‑function sequence with a small self‑convolution peak.

Algorithm
---------

1. **Explore simple parametric families** – for a few candidate lengths
   (800‒3000) we generate:
   * the classic arcsine distribution,
   * symmetric Beta‑distributions   a_i ∝ x^{α‑1}(1‑x)^{α‑1},
   * random Dirichlet draws.
   Each candidate is evaluated with ``evaluate_sequence`` and the best
   (lowest score) is kept as a starting point.

2. **Adam‑based smoothing optimisation** – starting from the best
   candidate we minimise the smoothed max of the autocorrelation.
   For a temperature β we form soft‑max weights
   ``w_k ∝ exp(β·(conv_k‑max(conv)) )`` and the gradient of this smooth
   objective is
   ``grad = 2·conv(w , a[::-1])[n‑1:2·n‑1]``.
   An Adam update (with projection onto the probability simplex) is
   performed.  β and the learning rate are gradually increased and
   decreased respectively, giving a smooth‑to‑hard transition.

3. **Occasional linear‑programming “fill‑the‑slack’’ step** – when the
   current maximal autocorrelation value is ``M``, we solve the LP

        maximise    1ᵀ g
        subject to  g ≥ 0
                    conv_f_g ≤ M

   (``conv_f_g`` is linear in ``g``).  The LP is solved with
   ``scipy.optimize.linprog`` (sparse matrix).  The optimal ``g`` is
   renormalised and mixed into the current sequence with a decaying
   factor ``t``.  This typically reduces the peak without increasing the
   overall score.

4. **Book‑keeping** – after each update the true score
   ``evaluate_sequence`` is computed; the best sequence ever seen is
   returned at the end of the allotted time (≈1000 s).

The implementation uses only NumPy, SciPy (optional) and no I/O.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> np.ndarray:
    """Arcsine coefficients C_i = binom(2i,i)/4^i, normalised to sum = 1."""
    seq = np.empty(n, dtype=float)
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq[i] = cur
        total += cur
        if i + 1 < n:
            cur = cur * (2.0 * i + 1.0) / (2.0 * (i + 1))
    seq /= total
    return seq


# ----------------------------------------------------------------------
def beta_sequence(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta distribution on {0,…,n‑1}:
        a_i ∝ ( (i+0.5)/n )^{α‑1} ( 1‑(i+0.5)/n )^{α‑1}
    Normalised to total mass 1.  ``alpha`` > 0.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    # protect against overflow/underflow for extreme alphas
    with np.errstate(divide='ignore', invalid='ignore'):
        vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    # replace NaNs (occurs when x==0 or 1) by a large number
    vals = np.where(np.isnan(vals), 0.0, vals)
    total = vals.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n, dtype=float)
    return vals / total


# ----------------------------------------------------------------------
def random_dirichlet_sequence(n: int, concentration: float, rng) -> np.ndarray:
    """
    Sample a random point from Dirichlet(α,…,α) with ``α = concentration``.
    Returns a normalised non‑negative vector summed to 1.
    """
    x = rng.gamma(concentration, 1.0, size=n)
    total = x.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n, dtype=float)
    return x / total


# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σ x = s}.
    Implementation of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Target simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using FFT (real‐to‑real).  Returns an array of length
    len(a)+len(b)-1.
    """
    na = a.shape[0]
    nb = b.shape[0]
    n = na + nb - 1
    N = 1 << ((n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:n]
    return conv


# ----------------------------------------------------------------------
def solve_lp(f: np.ndarray, M: float, max_time: float = 5.0):
    """
    Solve the linear programme
        maximise    1ᵀ g
        subject to  g ≥ 0,
                    conv_f_g ≤ M   (element‑wise)
    using ``scipy.optimize.linprog`` (highs).  Returns the optimal g
    (as a dense 1‑D ndarray) or ``None`` on failure.
    """
    try:
        import scipy.optimize
        import scipy.sparse as sp
    except Exception:
        return None

    n = f.shape[0]
    rows = 2 * n - 1

    # Build the convolution matrix A where (A @ g)_k = sum_i f_i * g_{k-i}
    # i.e. column i of A contains f_i on rows i … i+n‑1.
    A = sp.lil_matrix((rows, n), dtype=float)
    for i in range(n):
        A[i:i + n, i] = f[i]
    A = A.tocsc()

    c = -np.ones(n, dtype=float)                     # maximize sum(g)
    b_ub = M * np.ones(rows, dtype=float)
    bounds = [(0.0, None)] * n

    # Solve with a modest time limit to keep the whole program fast.
    opts = {"presolve": True, "time_limit": max_time}
    try:
        res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options=opts,
        )
    except Exception:
        return None

    if not res.success or res.x is None:
        return None

    g = res.x
    if g.sum() <= 0.0:
        return None
    return g


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises the
    evaluation metric defined by ``evaluate_sequence``.
    Returns the best list of coefficients found within the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1) QUICK PARAMETRIC SEARCH (≈10–15 % of the budget)
    # ------------------------------------------------------------------
    candidate_lengths = [800, 1200, 1500, 2000, 2500, 3000]
    alpha_grid = np.linspace(0.2, 2.0, 10)           # symmetric Beta parameters
    dirichlet_alphas = [0.2, 0.5, 1.0, 2.0]         # concentration for Dirichlet draws

    # Helper to evaluate and possibly update the global best
    def maybe_update(seq: np.ndarray):
        nonlocal best_seq, best_score, best_n
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            best_n = seq.shape[0]
            print(f"[search] n={best_n:<4d}  new best score={best_score:.6f}")

    for n in candidate_lengths:
        if time.time() >= deadline:
            break

        # (a) arcsine
        seq = arcsine_sequence(n)
        maybe_update(seq)

        # (b) symmetric Beta family
        for alpha in alpha_grid:
            if time.time() >= deadline:
                break
            seq = beta_sequence(n, alpha)
            maybe_update(seq)

        # (c) random Dirichlet draws (few per length)
        for _ in range(3):
            if time.time() >= deadline:
                break
            conc = rng.choice(dirichlet_alphas)
            seq = random_dirichlet_sequence(n, conc, rng)
            maybe_update(seq)

    # Fallback: if nothing succeeded (should never happen) use arcsine n=2000
    if best_seq is None:
        best_seq = arcsine_sequence(2000)
        best_score = evaluate_sequence(best_seq.tolist())
        best_n = best_seq.shape[0]
        print("[fallback] using default arcsine sequence")

    # ------------------------------------------------------------------
    # 2) ADAM‑DRIVEN SMOOTHED‑MAX OPTIMISATION
    # ------------------------------------------------------------------
    n = best_seq.shape[0]
    a = best_seq.astype(float)          # current iterate
    m = np.zeros_like(a)                # 1st moment
    v = np.zeros_like(a)                # 2nd moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t_iter = 0

    # Hyper‑parameters
    lr = 0.5               # base learning rate for Adam
    beta = 0.5             # initial soft‑max temperature
    lr_decay = 0.998
    beta_inc = 1.02
    mix_weight = 0.2       # LP mixing factor
    mix_min = 0.02
    mix_decay = 0.97
    lp_interval = 70       # attempt LP every this many iterations
    have_scipy = True
    try:
        import scipy.optimize   # just a quick import test
    except Exception:
        have_scipy = False

    # ------------------------------------------------------------------
    def current_score(arr):
        return evaluate_sequence(arr.tolist())

    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ------------------------------------------------------------------
        # Compute current convolution and its max
        conv = conv_fft(a, a)                     # length 2n‑1
        M = float(conv.max())

        # --------------------------------------------------------------
        # Occasionally solve the LP “fill‑the‑slack’’ sub‑problem
        if have_scipy and (t_iter % lp_interval == 0) and (n <= 2000):
            # guarantee we leave enough time for the LP
            if time.time() + 5.0 < deadline:
                g = solve_lp(a, M, max_time=4.0)
                if g is not None:
                    g_scaled = g / g.sum()
                    a = (1.0 - mix_weight) * a + mix_weight * g_scaled
                    a = project_simplex(a, s=1.0)
                    mix_weight = max(mix_min, mix_weight * mix_decay)
                    # recompute score after LP mix
                    sc = current_score(a)
                    if sc < best_score:
                        best_score = sc
                        best_seq = a.copy()
                        best_n = n
                        print(f"[LP] iter={t_iter:<5d}  improved to {sc:.6f}")
                    # skip gradient step this round – continue loop
                    continue

        # --------------------------------------------------------------
        # Gradient of the smooth approximation of max(conv)
        max_val = M
        # Soft‑max over the convolution entries (numerically stable)
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / exp_vals.sum()

        # Gradient: 2 * conv(w, a[::-1]) restricted to the central n entries
        grad_full = conv_fft(w, a[::-1])          # length 2n‑1
        grad = 2.0 * grad_full[n - 1:2 * n - 1]   # shape (n,)

        # --------------------------------------------------------------
        # Adam update
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = a - step
        cand = project_simplex(cand, s=1.0)

        # --------------------------------------------------------------
        # Evaluate candidate (cheap enough)
        cand_score = current_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            best_n = n
            a = cand
            print(f"[grad] iter={t_iter:<5d}  new best {cand_score:.6f}")
        else:
            # keep the old iterate (discard the non‑improving step)
            a = a

        # --------------------------------------------------------------
        # Decay learning‑rate and raise temperature gradually
        lr *= lr_decay
        beta = min(200.0, beta * beta_inc)

    # ------------------------------------------------------------------
    # Final sanity checks (project once more to guarantee feasibility)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    if best_seq.sum() == 0.0:
        best_seq = np.full(best_seq.shape, 1.0 / best_seq.shape[0])
    else:
        best_seq = best_seq / best_seq.sum()

    print(f"Best sequence length = {len(best_seq)}  final score = {best_score:.6f}")
    return best_seq.tolist()
