"""
propose_candidate
=================

A three‑phase optimizer for the max‑self‑convolution problem
(aka minimise 2·n·max(conv)/(∑a)²).  The objective reduces to the
maximum entry of the linear convolution `a*a` when the sum of the
coefficients is fixed to √(2 n); therefore the algorithm works in the
simplex {a≥0, Σa = √(2 n)}.

Phases
------
1. **Rich parametric shape search** – a large collection of analytic
   families (inverse‑sqrt, power‑law, exponential, cosine, triangular,
   quadratic, Gaussian, beta‑distribution, asymmetric mixtures and
   random block patterns) are sampled.  Each candidate is projected onto
   the simplex and scored via a fast FFT convolution.  The best candidate
   is kept.

2. **Adam‑based smoothed gradient descent** – the non‑smooth `max`
   is approximated by a soft‑max with a temperature `T`.  The gradient
   of the smoothed objective is computed analytically using FFTs.
   An Adam optimiser (momentum + adaptive step) moves the vector while
   staying on the simplex.  The temperature is slowly cooled; occasional
   random perturbations help to escape plateaus.

3. **Local pair‑wise hill‑climbing** – with the remaining time a simple
   random transfer of mass between two positions is tried; if it lowers
   the maximum convolution value the move is kept.  This often yields
   a final few‑percent improvement.

The whole routine respects the 1000 s budget, prints progress and
returns the best list of floats found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution using FFT (output length len(u)+len(v)-1)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.ifft(fu * fv).real
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full convolution of a with itself."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    Implements the analytic formula using FFTs.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)          # length = 3n-2
    start = n - 1
    grad = 2.0 * conv[start:start + n]            # 2·Σ_k w_k a_{k-i}
    # correct the diagonal double–counting (indices where k = 2i)
    diag_w = w[0:2 * n - 1:2]                     # w_{2i}, length n
    grad -= a * diag_w
    return grad

        if time.time() >= deadline: break
        decay = np.random.rand() * 0.012 + 0.001     # decay ∈ [0.001, 0.013]
        vec = np.exp(-decay * idx)
        _try_raw(vec)

    # 5) cosine‑based (large at ends, small in middle)
    for _ in range(80):
        if time.time() >= deadline: break
        p = np.random.rand() * 5.0 + 0.5
        vec = np.abs(np.cos(np.pi * idx / (n - 1))) ** p
        _try_raw(vec)

    # 6) triangular (linear ramp up/down) with optional exponent
    for _ in range(80):
        if time.time() >= deadline: break
        p = np.random.rand() * 3.0 + 0.5
        tri = 1.0 - np.abs(idx - (n - 1) / 2) / ((n - 1) / 2)
        vec = tri ** p
        _try_raw(vec)

    # 7) quadratic (parabolic) bump
    for _ in range(80):
        if time.time() >= deadline: break
        p = np.random.rand() * 4.0 + 0.5
        quad = 1.0 - ((idx - (n - 1) / 2) / ((n - 1) / 2)) ** 2
        vec = np.maximum(quad, 0.0) ** p
        _try_raw(vec)

    # 8) Gaussian bumps (centred)
    for _ in range(80):
        if time.time() >= deadline: break
        sigma = np.random.rand() * (n / 4) + n / 20   # σ ∈ [n/20 , n/5]
        mu = (n - 1) / 2
        vec = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
        _try_raw(vec)

    # 9) Beta‑distribution shaped (symmetric)
    for _ in range(80):
        if time.time() >= deadline: break
        alpha = 1.0 + np.random.rand() * 5.0
        beta = 1.0 + np.random.rand() * 5.0
        x = idx / (n - 1)
        vec = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1)) + 1e-6
        _try_raw(vec)

    # 10) Asymmetric power‑law (different left/right exponents)
    for _ in range(80):
        if time.time() >= deadline: break
        beta = np.random.rand() * 8.0 + 0.5
        pL = np.random.rand() * 2.0 + 0.2
        pR = np.random.rand() * 2.0 + 0.2
        left = 1.0 / (idx + beta + 1.0) ** pL
        right = 1.0 / ((n - 1 - idx) + beta + 1.0) ** pR
        vec = left + right
        _try_raw(vec)

    # 11) Random block candidates (few large blocks)
    for _ in range(200):
        if time.time() >= deadline: break
        nb = np.random.randint(2, 7)                 # 2‑6 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.zeros(n)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        _try_raw(cand)

    print(f"[shape search] best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  SMOOTHED ADAM GRADIENT DESCENT
    a = best_a.copy()
    conv = _self_convolution(a, size_fft)
    T = max(1e-4, (conv.max() - conv.min()) * 0.5)   # initial temperature
    min_T = 1e-7

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_iter = 0

    # base learning rate (empirically good for our scale)
    lr0 = target_sum * 0.4 / n

    while time.time() < deadline and t_iter < 800_000:
        # evaluate current vector
        conv = _self_convolution(a, size_fft)
        cur_score = float(conv.max())
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[{t_iter:07d}] new best = {best_score:.6f}")

        # soft‑max weights (smooth approximation of max)
        shifted = conv - conv.max()
        exp_arg = shifted / max(T, 1e-12)
        exp_arg = np.clip(exp_arg, -50, 0)   # avoid overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed objective
        grad = _weighted_gradient(a, w, size_fft)

        # Adam update
        t_iter += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v / (1 - beta2 ** t_iter)
        step = lr0 * m_hat / (np.sqrt(v_hat) + epsilon)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional random perturbation to jump out of plateaus
        if (t_iter % 6000 == 0) or (np.random.rand() < 0.001):
            noise = np.random.randn(n) * target_sum * 5e-4
            a_new = a_new + noise
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_to_simplex(a_new, target_sum)
            print(f"[{t_iter:07d}] perturbation")

        a = a_new
        T = max(T * 0.9989, min_T)   # slower cooling for stability

    print(f"[gradient] best score = {best_score:.6f} after {t_iter} iterations")

    # ------------------------------------------------------------------
    # 3️⃣  LOCAL PAIR‑WISE HILL‑CLIMBING
    attempts = 0
    while time.time() < deadline and attempts < 15_000:
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        # transfer up to 30 % of the source mass (but not more than available)
        max_delta = a[i] * 0.3
        if max_delta <= 0:
            attempts += 1
            continue
        delta = np.random.rand() * max_delta
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta
        # projection not needed – the sum is preserved and positivity kept
        sc = _score(a_candidate)
        attempts += 1
        if sc < best_score:
            best_score, best_a = sc, a_candidate.copy()
            a = a_candidate
            print(f"[local {attempts:05d}] improved to {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
