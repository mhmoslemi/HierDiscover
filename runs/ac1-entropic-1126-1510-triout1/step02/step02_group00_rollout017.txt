"""
search for a step‑function (list of non‑negative heights) that minimises
``evaluate_sequence``.

The algorithm explores several parametric families (arcsine, beta,
triangular, uniform, random, previous best) for a handful of lengths.
For each family it evaluates the simple bound
    score = 2·n·max(a∗a)   (a is normalised to sum = 1)
and keeps the best shape.  The best shapes are then refined with a
projected Adam optimiser that follows the sub‑gradient of the maximal
self‑convolution entry (using the top few active entries for a smoother
direction).  The optimiser works on the probability simplex, so the
score is invariant to scaling.  All work respects the overall time
budget (≈1000 s); the best sequence found is returned as a plain list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Discrete distributions ------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α)‑like shape (α>0)."""
    i = np.arange(n, dtype=np.float64)
    # (i+0.5)^{α‑1}·(n‑i‑0.5)^{α‑1}
    vals = (i + 0.5) ** (alpha - 1.0) * (n - i - 0.5) ** (alpha - 1.0)
    vals = np.maximum(vals, 0.0)
    total = vals.sum()
    if total == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return vals / total


def _triangle_distribution(n: int, edge_high: bool = False) -> np.ndarray:
    """Linear shape, either centre‑high (edge_high=False) or edge‑high."""
    i = np.arange(n, dtype=np.float64)
    if edge_high:
        vals = np.abs((2.0 * i - (n - 1)) / (n - 1))
    else:
        vals = 1.0 - np.abs((2.0 * i - (n - 1)) / (n - 1))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Fast self‑convolution (real FFT)
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Return a * a (real self‑convolution). Length = 2·len(a)‑1."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Simple score (scale‑invariant)
def _simple_score(a: np.ndarray) -> float:
    """Score = 2·n·max(a∗a) for a already normalised to sum = 1."""
    n = len(a)
    max_b = np.max(_fft_self_convolve(a))
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Adam optimiser on the simplex, using the top‑K active convolution entries
def _adam_optimize(a0: np.ndarray,
                   max_iters: int,
                   deadline: float,
                   top_k: int = 5,
                   base_lr: float = 0.5) -> tuple[np.ndarray, float]:
    a = _project_onto_simplex(a0, s=1.0)          # ensure feasibility
    n = len(a)
    best_seq = a.copy()
    best_score = _simple_score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()

        # pick the K largest convolution entries (including ties)
        if top_k < conv.size:
            top_idx = np.argpartition(-conv, top_k - 1)[:top_k]
        else:
            top_idx = np.arange(conv.size)

        # sub‑gradient (averaged over the selected entries)
        g = np.zeros_like(a)
        for k in top_idx:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1)
            idx_j = k - idx_i
            g[idx_i] += 2.0 * a[idx_j]
        if top_idx.size > 1:
            g *= 1.0 / top_idx.size

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = base_lr / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _simple_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            # occasional progress message (lightweight)
            print(f"[{time.time():.1f}] adam iter={it} score={cur_score:.6f}")

        # tiny random jitter every few thousand iterations to escape plateaux
        if it % 2000 == 0 and np.random.rand() < 0.05:
            a += np.random.randn(n) * 1e-4
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score

    np.random.seed(seed)
    # leave a couple of seconds for finalisation and printing
    total_deadline = time.time() + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Lengths to explore – moderate sizes are cheap yet expressive
    lengths = [2000, 4000, 6000, 8000]
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= total_deadline:
            break

        # reserve ~20 % of the remaining budget for this length
        slice_deadline = time.time() + max(0.1,
                                          (total_deadline - time.time()) * 0.2)

        # --------------------------------------------------------------
        # Build a catalogue of initial candidates for this length
        candidates = []

        # arcsine
        candidates.append(_arcsine_distribution(n))

        # uniform
        candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # triangular (centre‑high & edge‑high)
        candidates.append(_triangle_distribution(n, edge_high=False))
        candidates.append(_triangle_distribution(n, edge_high=True))

        # beta family (symmetric) – a few exponents around the critical 0.5
        for alpha in (0.30, 0.35, 0.40, 0.45, 0.50, 0.55, 0.60, 0.70, 0.80, 1.00):
            candidates.append(_beta_distribution(n, alpha))

        # random vector
        rnd = np.random.rand(n)
        candidates.append(rnd)

        # previous best (if the global variable exists)
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.size != n:
                # linear interpolation to the new size
                old_x = np.linspace(0.0, 1.0, prev.size)
                new_x = np.linspace(0.0, 1.0, n)
                prev = np.interp(new_x, old_x, prev)
            candidates.append(prev)
        except Exception:
            pass

        # --------------------------------------------------------------
        # Evaluate each raw candidate with the cheap bound and keep the
        # best few for a proper optimisation pass.
        prefiltered = []
        for raw in candidates:
            a = _project_onto_simplex(raw, s=1.0)
            sc = _simple_score(a)
            prefiltered.append((sc, a))
        # sort by the cheap score
        prefiltered.sort(key=lambda x: x[0])
        # keep at most 5 most promising raw candidates
        top_raw = [seq for _, seq in prefiltered[:5]]

        # --------------------------------------------------------------
        # Optimise each kept candidate with Adam
        for a0 in top_raw:
            if time.time() >= slice_deadline:
                break

            # rough iteration budget – the optimiser respects the deadline
            max_iters = 20000
            refined_seq, refined_score = _adam_optimize(a0,
                                                        max_iters,
                                                        slice_deadline)

            # final (more reliable) evaluation through the black‑box
            try:
                final_score = evaluate_sequence(refined_seq.tolist())
            except Exception:
                final_score = refined_score

            print(
                f"[{time.time():.1f}] length={n} refined_score={refined_score:.6f} "
                f"final_score={final_score:.6f}"
            )

            if final_score < best_score:
                best_score = final_score
                best_seq = refined_seq.copy()
                print(
                    f"*** New overall best → length={n} score={best_score:.6f}"
                )
                if best_score <= 1.503:
                    break

        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
        print("Fallback sequence used.")

    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
