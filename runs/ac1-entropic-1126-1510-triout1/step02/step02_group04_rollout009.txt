"""
Propose a step‑function minimizing `evaluate_sequence`.

Algorithm
---------
* Work on the probability simplex (non‑negative, sum = 1).
* Use a smooth approximation of the max‑convolution term:
      M_β(a) = (1/β)·log Σ exp(β·(a*a)_k)
  The gradient of 2·n·M_β is a simple cross‑correlation that can be
  evaluated with (real) FFTs.
* Perform many iterations of projected (additive) gradient descent
  with a decreasing step size, occasionally applying a *targeted move*
  that moves a small amount of mass from the pair that contributes most
  to the currently maximal convolution entry.
* If no improvement is seen for a long time, inject a fresh random
  point (simple stagnation restart).
* The whole loop runs until the 1000 s budget (minus a small safety
  margin) expires and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: project onto the probability simplex (clip negatives, renormalise)
# ----------------------------------------------------------------------
def _clip_and_normalise(v: np.ndarray) -> np.ndarray:
    """Clip negative entries to zero and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fallback: uniform distribution
        v = np.full_like(v, 1.0 / v.size)
    else:
        v = v / s
    return v


# ----------------------------------------------------------------------
# Fast scoring – same formula as evaluate_sequence when sum(seq)=1
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)).  Assumes seq is non‑negative and sums to 1."""
    n = seq.size
    conv = np.convolve(seq, seq)          # linear convolution, length 2n‑1
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Compute convolution of a with itself using a single FFT of length L
# ----------------------------------------------------------------------
def _self_convolution_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return linear convolution a*a (length 2·len(a)-1) using FFT length L."""
    Fa = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(Fa * Fa, n=L)
    n = a.size
    return conv_full[: 2 * n - 1]  # slice the linear part


# ----------------------------------------------------------------------
# Gradient of the smoothed max–convolution term.
# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, beta: float, L: int,
                         conv_full: np.ndarray) -> np.ndarray:
    """
    Compute the gradient (w.r.t. a) of 2·n·M_β(a),
    where M_β is the soft‑max of a*a with temperature 1/β.
    Returns a vector of length n.
    """
    n = a.size
    # soft‑max weights over the convolution entries
    max_c = conv_full.max()
    # for numerical stability subtract max before exponentiation
    exp_vals = np.exp(beta * (conv_full - max_c))
    w = exp_vals / exp_vals.sum()                     # shape (2n‑1,)

    # cross‑correlation a ⋆ w = sum_i a_i * w_{i+j}
    # can be obtained by convolving a with reversed w
    rev_w = w[::-1]                                   # reverse
    Fa = np.fft.rfft(a, n=L)                          # re‑use the same length L
    Fw = np.fft.rfft(rev_w, n=L)
    corr_full = np.fft.irfft(Fa * Fw, n=L)            # length L (≥3n‑2)
    # indices that give the correlation values for j = 0 … n‑1
    idx = (len(w) - 1) - np.arange(n)                 # (2n‑2) … (n‑1)
    grad = 2.0 * corr_full[idx]                       # gradient of M_β (scale 2·n omitted)
    return grad


# ----------------------------------------------------------------------
# Targeted move: steal a little mass from the larger of the two indices
# that most contribute to the current maximal convolution entry.
# ----------------------------------------------------------------------
def _targeted_move(seq: np.ndarray, conv_full: np.ndarray,
                   frac: float = 0.02) -> np.ndarray:
    """Return a new point after a small directed mass transfer."""
    n = seq.size
    k = int(conv_full.argmax())                     # lag with maximal conv entry

    i_arr = np.arange(n)
    j_arr = k - i_arr
    mask = (j_arr >= 0) & (j_arr < n)

    if not mask.any():
        # fallback: small random perturbation
        cand = seq + frac * (np.random.rand(n) - 0.5)
        return _clip_and_normalise(cand)

    # product a_i * a_j for each feasible pair
    contrib = np.zeros_like(seq)
    contrib[mask] = seq[i_arr[mask]] * seq[j_arr[mask]]
    i_max = int(np.argmax(contrib))                 # index with largest product
    j_max = k - i_max

    src = i_max if seq[i_max] >= seq[j_max] else j_max
    dst = np.random.randint(n)
    while dst == src:
        dst = np.random.randint(n)

    amount = frac * seq[src] * np.random.rand()
    cand = seq.copy()
    cand[src] -= amount
    cand[dst] += amount
    return _clip_and_normalise(cand)

    beta_max = 60.0          # final temperature

    best = a.copy()

    # ------------------------------------------------------------------
