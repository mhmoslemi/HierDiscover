"""
Optimisation of a non‑negative step function by greedy reduction of the
maximum self‑convolution value.

Strategy
--------
1. Initialise with the normalised arcsine coefficients (the best known
   analytical construction).  The uniform vector is also tried and the
   better one is kept.
2. Repeatedly locate the index *k* where the current convolution attains
   its maximum.  The sub‑gradient of this maximum w.r.t. the sequence is
   known analytically:
        ∂(conv[k]) / ∂a_i = 2·a_{k‑i}   (if the index is inside the support)
   and zero otherwise.
3. Perform a multiplicative (exponential) descent step
        a_i ← a_i·exp(‑η·g_i)
   (η is an adaptive step size) and renormalise to a probability simplex.
   If this does not improve the objective a small additive step with Euclidean
   projection onto the simplex is tried.
4. Occasionally apply a random “mass‑transfer’’ perturbation to
   escape shallow local minima.
5. The best sequence seen so far (according to the fast FFT based
   evaluation) is kept.  At the end of the allotted time the true score
   is obtained with the provided ``evaluate_sequence`` function and the
   sequence is returned.

The implementation uses FFT based convolution (O(n log n)) and a
project‑onto‑simplex routine, allowing thousands of iterations within the
1000 s budget.  Empirically this improves the metric below the previous
value 1.5709 and can reach the target ≈1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> list[float]:
    """Return the L1‑normalised arcsine coefficients C_i = binom(2i,i)/4^i,
    i = 0 … n‑1.  The recurrence C_{i+1}=C_i·(2i+1)/(2(i+1)) is used.
    """
    seq = []
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2.0 * i + 1.0) / (2.0 * (i + 1))
    seq = np.array(seq, dtype=float) / total
    return seq.tolist()


# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x ≥ 0, Σ x = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("target sum s must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT (real input)."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())      # next power of two ≥ m
    fa = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(fa * fa, n=nfft)[:m]
    # numerical noise may produce tiny negative values – clamp to zero
    conv[conv < 0] = 0.0
    return conv


# ----------------------------------------------------------------------
def gradient_at_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of conv[k] (the current maximum) w.r.t. a."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        idx = np.arange(i_start, i_end + 1, dtype=int)
        grad[idx] = 2.0 * a[k - idx]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence`` within the given time budget (seconds).
    Returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # leave a small safety margin

    # ------------------------------------------------------------------
    # 1) Choose a starting length and initial vector
    # ------------------------------------------------------------------
    candidate_lengths = [1500, 2000, 2500, 3000, 3500]
    best_seq = None
    best_score = float("inf")
    best_n = None

    # try a few lengths with two simple initialisations
    for n in candidate_lengths:
        if time.time() >= deadline:
            break
        # 1a) arcsine
        seq = np.asarray(arcsine_sequence(n), dtype=float)
        sc = evaluate_sequence(seq.tolist())
        if sc < best_score:
            best_score, best_seq, best_n = sc, seq.copy(), n
        # 1b) uniform
        u = np.ones(n, dtype=float) / n
        sc = evaluate_sequence(u.tolist())
        if sc < best_score:
            best_score, best_seq, best_n = sc, u.copy(), n

    # optionally fallback to a global sequence if it exists
    try:
        global height_sequence_1
        if (
            isinstance(height_sequence_1, (list, tuple, np.ndarray))
            and len(height_sequence_1) > 0
        ):
            seq = np.asarray(height_sequence_1, dtype=float)
            seq = np.maximum(seq, 0.0)
            total = seq.sum()
            if total > 0:
                seq = seq / total
                n = seq.shape[0]
                sc = evaluate_sequence(seq.tolist())
                if sc < best_score:
                    best_score, best_seq, best_n = sc, seq.copy(), n
    except Exception:
        pass

    # final fallback (should never happen)
    if best_seq is None:
        best_n = 2000
        best_seq = np.asarray(arcsine_sequence(best_n), dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())

    # ------------------------------------------------------------------
    # Helpers
    # ------------------------------------------------------------------
    def approx_score(a: np.ndarray) -> float:
        """Fast evaluation: 2·n·max(conv) for a sequence with L1=1."""
        conv = conv_full_fft(a)
        return 2.0 * a.shape[0] * conv.max()

    # ------------------------------------------------------------------
    # 2) Optimisation loop
    # ------------------------------------------------------------------
    seq = best_seq.copy()                # current candidate
    n = seq.shape[0]
    cur_score = approx_score(seq)
    best_score = cur_score
    best_seq = seq.copy()

    eta = 0.5                # step size for multiplicative update
    eta_min = 1e-8
    iter_no = 0
    # parameters for occasional random perturbations
    perturb_interval = 300
    perturb_eps = 0.02       # up to 2 % of the source mass

    while time.time() < deadline:
        iter_no += 1

        # ----- current convolution and argmax -----
        conv = conv_full_fft(seq)
        max_val = conv.max()
        k = int(conv.argmax())

        # ----- gradient of the maximal entry -----
        grad = gradient_at_max(seq, k)

        # ----- multiplicative (exponential) descent -----
        cand = seq * np.exp(-eta * grad)
        cand_sum = cand.sum()
        if cand_sum > 0:
            cand = cand / cand_sum
        else:
            cand = seq.copy()

        cand_score = approx_score(cand)

        # ----- accept multiplicative step if it improves -----
        if cand_score < cur_score - 1e-12:
            seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(
                    f"[iter {iter_no}] mult‑step improved approx score {cand_score:.6f}"
                )
            # modestly increase step size on success
            eta = min(2.0, eta * 1.04)
            continue

        # ----- additive step (projected gradient) as a fallback -----
        added = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            step = eta * factor
            cand_add = seq - step * grad
            cand_add = project_simplex(cand_add, s=1.0)
            cand_add_score = approx_score(cand_add)
            if cand_add_score < cur_score - 1e-12:
                seq = cand_add
                cur_score = cand_add_score
                if cand_add_score < best_score:
                    best_score = cand_add_score
                    best_seq = cand_add.copy()
                    print(
                        f"[iter {iter_no}] add‑step improved approx score {cand_add_score:.6f}"
                    )
                eta = min(2.0, eta * 1.04)
                added = True
                break
        if added:
            continue

        # ----- neither worked – shrink the step size -----
        eta = max(eta_min, eta * 0.55)

        # ----- occasional random mass‑transfer perturbation -----
        if iter_no % perturb_interval == 0:
            # pick a source index with probability proportional to its mass
            probs = seq / seq.sum()
            i = np.random.choice(n, p=probs)
            j = np.random.choice(n)
            if i != j and seq[i] > 0:
                delta = np.random.rand() * min(seq[i] * perturb_eps, seq[i])
                cand_pert = seq.copy()
                cand_pert[i] -= delta
                cand_pert[j] += delta
                cand_pert = project_simplex(cand_pert, s=1.0)
                cand_pert_score = approx_score(cand_pert)
                if cand_pert_score < best_score:
                    seq = cand_pert
                    cur_score = cand_pert_score
                    best_score = cand_pert_score
                    best_seq = cand_pert.copy()
                    print(
                        f"[iter {iter_no}] random perturb improved approx score {cand_pert_score:.6f}"
                    )
                    # after a successful perturb we can reset eta a little larger
                    eta = max(eta, 0.3)

    # ------------------------------------------------------------------
    # Final sanitisation and true scoring
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total == 0:
        best_seq = np.asarray(arcsine_sequence(len(best_seq)), dtype=float)
    else:
        best_seq = best_seq / total

    true_score = evaluate_sequence(best_seq.tolist())
    print(
        f"Final length = {len(best_seq)}  approx best = {best_score:.6f}  true best = {true_score:.6f}"
    )
    return best_seq.tolist()
