def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    lengths = [1800, 2200, 2600, 3000, 3400, 4000]
    best_seq = None
    best_metric = float('inf')
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"Optimizing length {n}")
        # initial candidates from param families
        candidates = []
        # arcsine baseline
        candidates.append(arcsine_sequence(n))
        # beta families param p values
        for p in [0.8, 1.0, 1.2, 1.5, 2.0, 3.0]:
            seq = beta_sym_sequence(n, p)
            candidates.append(seq)
        # random restarts
        for _ in range(3):
            init = np.random.rand(n)
            init = init / init.sum()
            candidates.append(init.tolist())
        # evaluate each candidate, keep best start
        for cand in candidates:
            metric = compute_metric_of(cand)
            if metric < best_metric:
                best_metric = metric
                best_seq = cand
        # refine best via gradient descent
        seq, metric = refine_with_adam(best_seq, deadline - time.time())
        if metric < best_metric:
            best_metric = metric
            best_seq = seq
        print(f"Length {n} best metric = {best_metric}")
    # final scaling
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return best_seq

def refine_with_adam(init_seq, remaining_time):
    n = len(init_seq)
    a = np.array(init_seq, dtype=np.float64)
    # convert to logits x = log(a) where a>0; fudge small epsilon for zeros
    eps = 1e-12
    x = np.log(a + eps)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1=0.9; beta2=0.999; epsilon=1e-8
    it = 0
    best_a = a.copy()
    best_metric = compute_metric_of(a)
    start = time.time()
    # schedule for alpha (softmax temperature)
    alpha = 1.0
    max_alpha = 30.0
    alpha_inc = (max_alpha - alpha) / (5000)   # approx increments
    while time.time() - start < remaining_time and it < 20000:
        it += 1
        a = np.exp(x - np.max(x))  # stable
        a = a / a.sum()
        # compute conv a*a via FFT
        L = 1 << ((2*n-1).bit_length())
        fa = np.fft.rfft(a, L) 
        conv = np.fft.irfft(fa * fa, L)[:2*n-1]
        # compute smooth weights
        # scale conv maybe large? conv values approx 1/n. Use alpha accordingly.
        maxc = conv.max()  # for numeric stability
        # we may subtract maxc to avoid overflow
        exp_terms = np.exp(alpha * (conv - maxc))
        w = exp_terms / exp_terms.sum()

        L2 = 1 << ((n + (2*n-1) - 1).bit_length())  # >=3n-2

            alpha = min(max_alpha, alpha + alpha_inc)
        # possibly break if no improvement for many steps

def beta_sym_sequence(n, p):
    """Generate symmetric beta shape with exponent p (p>0). For p=0.5 yields arcsine."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    x = idx / n
    raw = np.power(x, p-1) * np.power(1 - x, p-1)
    raw_sum = raw.sum()
    if raw_sum == 0:
        return np.full(n, 1.0 / n).tolist()
    seq = (raw / raw_sum).tolist()
    return seq
```

def refine_adam(init_seq, time_budget):
    a = np.asarray(init_seq, dtype=np.float64)
    n = a.shape[0]
    # ensure positive
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    # initialize logits
    x = np.log(a)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1=0.9
    beta2=0.999
    eps = 1e-8
    best_a = a.copy()
    best_metric = compute_metric(a)
    start = time.time()
    it = 0
    # pre compute FFT sizes
    L = 1 << ((2*n-1).bit_length())
    L2 = 1 << ((3*n-2).bit_length())
    alpha = 1.0
    max_alpha = 30.0
    # schedule: increase alpha every 500 iterations
    while time.time() - start < time_budget and it < 200000:
        it += 1
        # softmax a from logits
        max_x = np.max(x)
        expx = np.exp(x - max_x)
        a = expx / expx.sum()
        # compute convolution a * a
        fa = np.fft.rfft(a, L)
        conv = np.fft.irfft(fa * fa, L)[:2*n-1]
        # softmax weights
        maxc = conv.max()
        # Use stable softmax: subtract maxc
        exp_w = np.exp(alpha * (conv - maxc))
        w = exp_w / exp_w.sum()
        # correlation to compute gradient
        w_rev = w[::-1]
        fa = np.fft.rfft(a, L2)
        fw_rev = np.fft.rfft(w_rev, L2)
        conv_aw = np.fft.irfft(fa * fw_rev, L2)
        offset = 2*n - 2
        h = conv_aw[offset - np.arange(n)]
        grad_a = 4.0 * n * h
        # chain rule for logits
        avg = np.dot(grad_a, a)
        grad_x = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        lr = 0.5  # might tune
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        # evaluate real metric
        metric = compute_metric(a)
        if metric < best_metric:
            best_metric = metric
            best_a = a.copy()
        # increase alpha gradually
        if it % 200 == 0 and alpha < max_alpha:
            alpha = min(max_alpha, alpha * 1.2)
        # maybe break if stagnation
        if it % 1000 == 0 and time.time() - start > time_budget * 0.95:
            break
    return best_a.tolist(), best_metric

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    # choose length
    n = 3000
    # generate candidates
    candidates = []
    candidates.append(arcsine_sequence(n))

    for p in [0.4, 0.5, 0.6, 0.8, 1.0, 1.2, 1.5, 2.0, 3.0]:

        rand /= rand.sum()

        print(f"Refining best sequence for ~{remaining:.1f}s")
        seq_refined, metric_refined = refine_adam(best_seq, remaining)
        print(f"Refined metric: {metric_refined:.6f}")
        if metric_refined < best_metric:
            best_metric = metric_refined
            best_seq = seq_refined

"""
Optimise a non‑negative step‑function sequence for the evaluation metric.

The algorithm works in three stages:

1. **Initial candidates** – generate a few hand‑crafted families
   (arcsine, symmetric Beta‑type shapes with various exponents and a few
   random Dirichlet draws) and keep the best one according to a fast
   Fourier‑based metric (exactly the same quantity returned by
   `evaluate_sequence` for normalised sequences).

2. **Smooth‑max Adam optimisation** – minimise a smooth approximation of
   the peak of the self‑convolution.  The peak is approximated by a
   temperature‑controlled soft‑max; the gradient of the smoothed loss
   can be computed analytically by a correlation of the current sequence
   with the soft‑max weights.  Adam with a slowly increasing temperature
   steers the iterates away from the “max‑gradient” plateau that blocks
   the simple mirror‑descent performed in the reference code.

3. **Return the best sequence** – clip to the limits enforced by
   `evaluate_sequence` and hand it back.

The whole routine respects the 1000 s budget and aims at pushing the
metric below the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> list[float]:
    """First n coefficients of (1‑z)^-1/2, normalised to sum 1."""
    seq = []
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    seq = [x / total for x in seq]
    return seq


# ----------------------------------------------------------------------
def beta_sym_sequence(n: int, p: float) -> list[float]:
    """
    Symmetric Beta‑type shape.
    a_i ∝ ( (i+0.5)/n )^{p‑1} * (1‑(i+0.5)/n)^{p‑1}.
    p=0.5 reproduces the arcsine distribution.
    """
    idx = np.arange(n, dtype=np.float64) + 0.5
    x = idx / n
    # prevent 0**neg by adding a tiny epsilon
    eps = 1e-16
    raw = np.power(np.maximum(x, eps), p - 1.0) * np.power(np.maximum(1.0 - x, eps), p - 1.0)
    total = raw.sum()
    if total == 0.0:
        raw = np.ones_like(raw)
        total = raw.sum()
    seq = (raw / total).tolist()
    return seq


# ----------------------------------------------------------------------
def _next_pow_two(k: int) -> int:
    """Small helper – next power of two ≥ k."""
    return 1 << (k - 1).bit_length()


# ----------------------------------------------------------------------
def compute_metric(seq: list[float]) -> float:
    """
    Fast evaluation of 2·n·max(conv(seq,seq)) for a normalised (sum=1) sequence.
    Identical to what ``evaluate_sequence`` would return because the function
    is homogeneous of degree 0 (ignoring the clipping performed inside it).
    """
    a = np.asarray(seq, dtype=np.float64)
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    a = a / s
    n = a.shape[0]
    L = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    max_conv = conv.max()
    return float(2.0 * n * max_conv)


# ----------------------------------------------------------------------
def refine_adam(init_seq: list[float], time_budget: float) -> tuple[list[float], float]:
    """
    Adam optimiser on the smoothed max‑convolution objective.
    Returns the best sequence discovered within ``time_budget`` seconds.
    """
    a = np.asarray(init_seq, dtype=np.float64)
    n = a.shape[0]

    # initialise logits (log‑softmax parameters) – keep a strictly positive
    eps = 1e-12
    a = np.clip(a, eps, None)
    a /= a.sum()
    x = np.log(a)

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # FFT sizes (powers of two, large enough for linear convolutions)
    L_conv = _next_pow_two(2 * n - 1)          # for a*a
    L_corr = _next_pow_two(3 * n - 2)          # for a * w_rev

    # temperature schedule for the soft‑max (smooth max)
    alpha = 1.0
    max_alpha = 60.0      # final temperature – strongly approximates a hard max
    inc_every = 200       # iterations between temperature increases

    best_a = a.copy()
    best_metric = compute_metric(best_a.tolist())
    start = time.time()
    it = 0

    while time.time() - start < time_budget and it < 200_000:
        it += 1

        # ---------- softmax → probability vector ----------
        max_x = np.max(x)
        expx = np.exp(x - max_x)
        a = expx / expx.sum()

        # ---------- self‑convolution ----------
        fa = np.fft.rfft(a, L_conv)
        conv = np.fft.irfft(fa * fa, L_conv)[: 2 * n - 1]

        # ---------- smooth‑max weights ----------
        # use max(conv) for numeric stability
        maxc = conv.max()
        w_raw = np.exp(alpha * (conv - maxc))
        w = w_raw / w_raw.sum()

        # ---------- gradient w.r.t. a ----------
        w_rev = w[::-1]
        fa = np.fft.rfft(a, L_corr)
        fw = np.fft.rfft(w_rev, L_corr)
        corr = np.fft.irfft(fa * fw, L_corr)  # linear correlation a ⋆ w
        offset = 2 * n - 2
        h = corr[offset - np.arange(n)]       # h_i = Σ_k w_{i+k} a_k
        grad_a = 4.0 * n * h                  # ∂L/∂a_i

        # ---------- chain rule to logits ----------
        avg = np.dot(grad_a, a)
        grad_x = a * (grad_a - avg)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.5
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---------- bookkeeping ----------
        cur_metric = compute_metric(a.tolist())
        if cur_metric < best_metric:
            best_metric = cur_metric
            best_a = a.copy()
            # optional: report progress
            if it % 500 == 0:
                print(f"[Adam it {it:6d}] new best metric {best_metric:.6f}")

        # increase temperature slowly
        if it % inc_every == 0 and alpha < max_alpha:
            alpha = min(max_alpha, alpha * 1.2)

        # early stop if almost out of time
        if time.time() - start > time_budget * 0.95:
            break

    return best_a.tolist(), best_metric


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Main entry point – called by the scoring harness.
    Returns the best step‑function sequence found within ``budget_s`` seconds.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ---------------------------------------------------------------
    # 1. Choose a single length (larger n gives more flexibility, but
    #    the metric is essentially scale‑free, so n≈3000 is a good trade‑off)
    # ---------------------------------------------------------------
    n = 3000
    print(f"=== Optimising length {n} ===")

    # ---------------------------------------------------------------
    # 2. Build an initial pool of candidates
    # ---------------------------------------------------------------
    candidates = []

    # classic arcsine
    candidates.append(arcsine_sequence(n))

    # symmetric beta‑type shapes (p parameter controls endpoint singularity)
    for p in [0.4, 0.5, 0.6, 0.8, 1.0, 1.2, 1.5, 2.0, 3.0, 4.0]:
        candidates.append(beta_sym_sequence(n, p))

    # a few random Dirichlet draws
    for _ in range(3):
        rand = np.random.rand(n).astype(np.float64)
        rand /= rand.sum()
        candidates.append(rand.tolist())

    # ---------------------------------------------------------------
    # 3. Evaluate the pool and keep the best starting point
    # ---------------------------------------------------------------
    best_seq = None
    best_metric = float("inf")
    for seq in candidates:
        metric = compute_metric(seq)
        print(f"  candidate metric = {metric:.6f}")
        if metric < best_metric:
            best_metric = metric
            best_seq = seq

    # ---------------------------------------------------------------
    # 4. Refine the best candidate with the Adam optimiser
    # ---------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print(f"Refining the best candidate for ~{remaining:.1f}s")
        refined_seq, refined_metric = refine_adam(best_seq, remaining)
        print(f"Refined metric = {refined_metric:.6f}")
        if refined_metric < best_metric:
            best_metric = refined_metric
            best_seq = refined_seq
    else:
        print("Insufficient time left for refinement – returning initial best.")

    # ---------------------------------------------------------------
    # 5. Final clipping (mirrors the sanitisation inside evaluate_sequence)
    # ---------------------------------------------------------------
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    print(f"\n--- FINAL RESULT ---\nLength = {len(best_seq)}  metric = {best_metric:.6f}")
    return best_seq
