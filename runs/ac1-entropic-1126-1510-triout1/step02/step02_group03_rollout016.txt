def _optimize_with_smooth(initial_seq, n, deadline, max_iters=800,
                          eta0=0.3, lambda0=0.5, lambda_max=8.0):
    a = initial_seq.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = compute_score(a)
    m = 1 << ((3*n - 2).bit_length())
    lamb = lambda0
    it_last_improved = 0
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # Compute conv, score
        b = _conv_full_fft(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            it_last_improved = it
        # compute softmax weights
        maxb = b.max()
        exp_arg = lamb * (b - maxb)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(b, 1.0 / b.shape[0])
        else:
            w /= w_sum
        # gradient via correlation
        g = _gradient_via_correlation(w, a, m)
        # step size
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum
        # increase lambda
        lamb = min(lambda_max, lamb * 1.02)
        # optional jitter
        if it % 200 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
    return best_seq, best_score

def _hill_climb_local(initial_seq, n, deadline, max_steps=5000, step_frac=0.1):
    a = initial_seq.astype(np.float64).copy()
    a = a / a.sum()
    best_seq = a.copy()

    rng = np.random.default_rng()
    for step in range(max_steps):

            break

        if a[i] == 0:
            continue
        # delta as fraction of a[i], random between 0 and step_frac

        new_score = compute_score(a_new)
        if new_score < best_score:
            a = a_new
            best_seq = a_new.copy()
            best_score = new_score
            # reduce step size gradually? optional
    return best_seq, best_score

"""
Propose a low‑score step‑function sequence.

The algorithm proceeds in several layers:

1️⃣  Build a diverse pool of candidate distributions for a chosen length *n*:
    * uniform
    * discrete arcsine (β=½)
    * symmetric Beta(α,α) for many α ∈ (0.05, 1.3)
    * two‑edge block distributions (mass concentrated in short intervals at
      both ends)
    * convex mixtures of arcsine and uniform.

2️⃣  Keep the best raw candidate (by the exact objective).

3️⃣  Refine it with two mirror‑descent phases on the simplex:
    – a smooth “log‑sum‑exp” approximation of the max‑convolution,
    – a hard‑max (sub‑gradient) version.
   Both use exponentiated‑gradient updates with an adaptive step size and
   occasional jitter to avoid stagnation.

4️⃣  Perform a short stochastic hill‑climbing stage: repeatedly move a tiny
    amount of mass between two random indices and keep the move if it lowers
    the true objective.

All phases are time‑boxed; the best overall sequence found before the
budget expires is returned as a plain Python list.  The code works solely
with NumPy (FFT based convolutions) and respects the evaluation function’s
security checks.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the grid x_i = (i+½)/n, normalized."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1.0 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (β=½) on n points, Σa = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _block_distribution(n: int, frac: float) -> np.ndarray:
    """
    Two‑edge block distribution.
    `frac` ∈ (0, ½] denotes the proportion of points taken on each side.
    Half of the total mass is placed uniformly on the left block,
    the other half on the right block.
    """
    block_len = max(1, int(round(frac * n)))
    a = np.zeros(n, dtype=np.float64)
    half = 0.5
    left_weight = half / block_len
    right_weight = half / block_len
    a[:block_len] = left_weight
    a[-block_len:] = right_weight
    # normalise (overlap when 2*block_len > n is harmless)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a power‑of‑two FFT.
    Returned length = 2·len(a)‑1.
    """
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, m)
    B = A * A
    c = np.fft.irfft(B, m)
    return c[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k‑i} (correlation) using FFT.
    w – weight vector of length 2·n‑1,
    a – current sequence (length n),
    m – FFT length (power of two, ≥ len(w)+len(a)-1).
    Returns the slice g[0…n‑1].
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start:start + len(a)]


# ----------------------------------------------------------------------
def _score_of(a: np.ndarray) -> float:
    """Exact score 2·n·max(conv(a,a))."""
    n = len(a)
    b = _conv_full_fft(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
def _optimise_smooth(
    init_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 600,
    eta0: float = 0.3,
    lambda0: float = 0.5,
    lambda_max: float = 8.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on a smooth approximation of the max‑convolution.
    Returns the best sequence seen and its (unsmoothed) exact score.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = _score_of(a)

    # FFT length sufficient for a (len=n) and w (len=2n‑1)
    m = 1 << ((3 * n - 2).bit_length())
    lam = lambda0
    it_last_improve = 0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current exact score (for bookkeeping)
        cur_score = _score_of(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            it_last_improve = it

        # smooth max via soft‑max weights
        b = _conv_full_fft(a)
        maxb = b.max()
        exp_arg = lam * (b - maxb)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.shape[0])
        else:
            w /= w_sum

        # gradient
        g = _gradient_via_correlation(w, a, m)

        # step size (adaptive)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum

        # increase smoothing parameter
        lam = min(lambda_max, lam * 1.02)

        # occasional jitter to escape plateaus
        if it % 200 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 5e-5
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _optimise_hard_max(
    init_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300,
    eta0: float = 0.4,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent using a hard‑max (sub‑gradient) weight vector.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = _score_of(a)

    m = 1 << ((3 * n - 2).bit_length())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        b = _conv_full_fft(a)
        cur_score = _score_of(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()

        maxb = b.max()
        idxs = np.where(b >= maxb - 1e-12)[0]
        w = np.zeros_like(b)
        if idxs.size:
            w[idxs] = 1.0 / idxs.size

        g = _gradient_via_correlation(w, a, m)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum

        if it % 100 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 5e-5
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _hill_climb(
    init_seq: np.ndarray,
    n: int,
    deadline: float,
    max_steps: int = 5000,
    step_frac: float = 0.1,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing: move a random fraction of mass
    from one index to another and keep the move if the exact score improves.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = _score_of(a)
    rng = np.random.default_rng()

    for step in range(max_steps):
        if time.time() >= deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        if a[i] <= 0.0:
            continue
        delta = a[i] * rng.uniform(0.0, step_frac)
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # numerical safety
        if a_new[i] < 0.0:
            continue
        new_score = _score_of(a_new)
        if new_score < best_score:
            a = a_new
            best_seq = a_new.copy()
            best_score = new_score
    return best_seq, best_score
