def _optimise_smooth_advanced(...):

    best_seq = a.copy()
    best_score = 2.0 * n * _conv_full_fft(a).max()

    m = 1 << ((3 * n - 2).bit_length())
    lamb = lambda0
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        b = _conv_full_fft(a)
        cur_score = 2.0*n*b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n} iter={it} score={cur_score:.6f}")
        # compute softmax weights
        maxb = b.max()
        exp_arg = lamb * (b - maxb)
        # clip large negative to avoid underflow? not needed
        w = np.exp(exp_arg)
        w_sum = w.sum()

        else:
            w /= w_sum
        # gradient
        g = _gradient_via_correlation(w, a, m)
        # step
        eta = eta0 / np.sqrt(it)
        # clip gradient to avoid overflow
        step = -eta * g
        step = np.clip(step, -50, 50)  # for exponent
        a = a * np.exp(step)
        s = a.sum()
        if s == 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s
        # increase lambda
        lamb = min(lambda_max, lamb * 1.01)
        # occasional jitter
        if it % 500 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0, None)
            a = a / a.sum()

    for it in range(200):

            best_score = cur_score
            best_seq = a.copy()
            print(f"[Hard] n={n} hard_iter={it} score={cur_score:.6f}")

    return best_seq, best_score

"""
Propose a low‑score step‑function sequence.

The algorithm improves on the previous Beta‑sweep + smooth‑optimiser
by

1. trying many initial shapes (Beta, arcsine, Dirichlet) for several
   lengths,
2. using a more aggressive smooth‑max (λ grows up to 200) together
   with an exponentiated‑gradient mirror‑descent,
3. finishing with a few hard‑max sub‑gradient steps projected onto the
   simplex,
4. performing short random perturbations of the current best solution
   and re‑optimising them.

All heavy work (convolutions, correlations) uses power‑of‑two FFTs,
so even 20 000‑point sequences are handled quickly.  The routine stops
as soon as the 1000 s budget (minus a small safety margin) is exhausted
and returns the best sequence found.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on the grid x_i = (i+½)/n.
    Returns a probability vector (sum = 1).
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (α=½) on n points, Σa = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a power‑of‑two FFT.
    Returned length = 2·len(a)‑1.
    """
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, m)
    B = A * A
    c = np.fft.irfft(B, m)
    return c[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k‑i} (correlation) using FFT.
    w – weight vector of length 2·n‑1, a – current sequence (length n),
    m – FFT length (power of two, at least len(w)+len(a)-1).
    Returns the slice g[0…n‑1].
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start : start + len(a)]


# ----------------------------------------------------------------------
def _optimise_smooth_advanced(
    n: int,
    deadline: float,
    init_seq: np.ndarray,
    max_iters: int = 4000,
    eta0: float = 0.3,
    lambda0: float = 0.5,
    lambda_max: float = 200.0,
) -> tuple[np.ndarray, float]:
    """
    Refine `init_seq` (already normalised to sum=1) by minimising a
    smooth approximation of the ‖a*a‖∞ peak and then a few exact
    sub‑gradient steps.  Returns the best sequence seen and its
    (unsmoothed) score 2·n·max(conv(a,a)).
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = float(2.0 * n * _conv_full_fft(a).max())

    # FFT size sufficient for correlation (len(w)=2n‑1, len(a)=n)
    m = 1 << ((3 * n - 2).bit_length())   # >= (2n‑1)+(n‑1)

    lam = lambda0
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current score
        b = _conv_full_fft(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} iter={it:5d} score={cur_score:.6f}")

        # smooth‑max weights (soft‑max)
        maxb = b.max()
        exp_arg = lam * (b - maxb)                # shift, never positive
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.shape[0])
        else:
            w /= w_sum

        # gradient of the smooth objective
        g = _gradient_via_correlation(w, a, m)

        # exponentiated‑gradient (mirror descent on simplex)
        eta = eta0 / np.sqrt(it)
        step = -eta * g
        # avoid overflow in the exponential
        step = np.clip(step, -50, 50)
        a = a * np.exp(step)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # increase λ slowly, but never exceed λ_max
        lam = min(lambda_max, lam * 1.01)

        # occasional tiny jitter to escape plateaus
        if it % 500 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

    # ------------------------------------------------------------------
    # Hard‑max sub‑gradient refinement (λ → ∞)
    for it in range(200):
        if time.time() >= deadline:
            break
        b = _conv_full_fft(a)
        maxb = b.max()
        # indices where the max is attained (allow tiny tolerance)
        tol = maxb * 1e-12 + 1e-15
        max_idx = np.where(b >= maxb - tol)[0]
        w = np.zeros_like(b)
        w[max_idx] = 1.0 / len(max_idx)

        g = _gradient_via_correlation(w, a, m)

        # simple projected gradient step
        eta_h = 0.1 / np.sqrt(it + 1)
        a = a - eta_h * g
        a = _project_onto_simplex(a, s=1.0)

        cur_score = 2.0 * n * _conv_full_fft(a).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[hard] n={n:5d} hard_iter={it:3d} score={cur_score:.6f}")

    return best_seq, best_score
