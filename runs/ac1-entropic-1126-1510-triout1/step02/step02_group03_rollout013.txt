def refine_subgradient(seq, deadline, max_iters=2000):
    a = seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_score = 2.0 * n * _conv_full_fft(a).max()
    m = 1 << ((3 * n - 2).bit_length())
    # use lambda=0 for exact max (w is one-hot)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # compute convolution
        b = _conv_full_fft(a)
        max_val = b.max()
        k = b.argmax()
        # compute gradient via correlation with indicator at k
        w = np.zeros_like(b)
        w[k] = 1.0
        g = _gradient_via_correlation(w, a, m)
        # step size schedule
        eta = 0.5 / (np.sqrt(it+1) * (np.max(g)+1e-12))
        # gradient descent in Euclidean sense with projection onto simplex
        a = a - eta * g
        # project onto simplex
        a = _project_onto_simplex(a, s=1.0)
        # evaluate
        cur_score = 2.0 * n * _conv_full_fft(a).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] refined best {best_score:.6f} iter={it}")
    return best_seq, best_score

def refine_exponential(a, deadline, max_iters):

    best_score = 2.0 * n * _conv_full_fft(a).max()
    m = 1 << ((3 * n - 2).bit_length())
    for it in range(1, max_iters+1):

            break

        k = int(b.argmax())

        w = np.zeros_like(b)
        w[k] = 1.0
        g = _gradient_via_correlation(w, a, m)

            a = a / a_sum

            best_score = cur_score

def _refine_expgrad(a: np.ndarray, deadline: float, max_iters: int = 2000):
    """
    Refine a by exponentiated‑gradient steps that directly minimise the
    maximal self‑convolution entry (one‑hot weight).  Returns the best
    sequence seen before `deadline`.
    """
    n = len(a)
    best = a.copy()
    best_score = _score_from_seq(best)
    # FFT length sufficient for correlation of (2n‑1) and n vectors
    m = 1 << ((3 * n - 2).bit_length())
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        b = _conv_full_fft(a)
        k = int(b.argmax())
        w = np.zeros_like(b)
        w[k] = 1.0
        g = _gradient_via_correlation(w, a, m)
        eta = 0.2 / np.sqrt(it)          # simple diminishing step‑size
        a = a * np.exp(-eta * g)         # multiplicative update (always ≥0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s
        cur = _score_from_seq(a)
        if cur < best_score:
            best_score = cur
            best = a.copy()
            print(f"[{time.time():.1f}] refined score {best_score:.6f} (it={it})")
    return best, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The function respects the given time budget, keeps the best sequence
    discovered and finally returns it as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")

    # -------------------------------------------------
    # 1️⃣  Historic construction (if available)
    try:
        hist = np.array(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        hist = np.clip(hist, 0.0, None)
        if hist.sum() > 0.0:
            hist = hist / hist.sum()
            hist_score = _score_from_seq(hist)
            overall_best_seq = hist.copy()
            overall_best_score = hist_score
            print(f"Historic sequence: length={len(hist)} score={hist_score:.6f}")
    except Exception:
        pass

    # -------------------------------------------------
    # 2️⃣  Deterministic short sweep (fast initial candidates)
    candidate_lengths = [2048, 4096, 8192, 16384]
    candidate_lengths = [n for n in candidate_lengths if n >= 256]

    for n in candidate_lengths:
        if time.time() >= deadline:
            break

        # a few fixed alphas (Beta family)
        for alpha in np.linspace(0.2, 1.2, 11):
            a = _beta_distribution(n, alpha)
            a = (a + a[::-1]) * 0.5
            a /= a.sum()
            sc = _score_from_seq(a)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[init] n={n} α={alpha:.3f} → score={sc:.6f}")

        # random mixtures of Beta PDFs
        for _ in range(12):
            a = _random_beta_or_mix(n)
            sc = _score_from_seq(a)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[init‑mix] n={n} → score={sc:.6f}")

        # exponential‑family candidates
        for _ in range(8):
            a = _random_exp(n)
            sc = _score_from_seq(a)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[init‑exp] n={n} → score={sc:.6f}")

    # -------------------------------------------------
    # 3️⃣  Main random search
    pool_lengths = [4096, 8192, 12288, 16384]
    while time.time() < deadline:
        # gradually bias to shorter lengths when close to the deadline
        if time.time() > start_time + 0.85 * budget_s:
            n = np.random.choice(pool_lengths[:2])
        else:
            n = np.random.choice(pool_lengths)

        # pick a random family
        r = np.random.rand()
        if r < 0.45:
            a = _random_beta_or_mix(n)
        elif r < 0.75:
            a = _random_exp(n)
        else:
            a = _random_dirichlet(n)

        # occasional tiny jitter to help escape flat regions
        if np.random.rand() < 0.07:
            a += 1e-4 * np.random.randn(n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        score = _score_from_seq(a)
        if score < overall_best_score:
            overall_best_score = score
            overall_best_seq = a.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST n={n} score={score:.6f}")

        # a quick local refinement when we have a promising bound
        if overall_best_score < 1.55 and time.time() < deadline - 2.0:
            refine_deadline = min(deadline, time.time() + 0.35)
            refined_seq, refined_score = _refine_expgrad(overall_best_seq, refine_deadline,
                                                         max_iters=300)
            if refined_score < overall_best_score:
                overall_best_score = refined_score
                overall_best_seq = refined_seq.copy()
                print(f"[{time.time():.1f}] REFINED global best → score={refined_score:.6f}")

        # early stop if target already beaten
        if overall_best_score <= 1.503:
            print(f"Target reached (score={overall_best_score:.6f}), breaking.")
            break

    # -------------------------------------------------
    # 4️⃣  Final (long) refinement
    if overall_best_seq is not None and time.time() < deadline - 0.5:
        overall_best_seq, overall_best_score = _refine_expgrad(
            overall_best_seq, deadline, max_iters=2000
        )
        print(f"Final refined score: {overall_best_score:.6f}")

    # -------------------------------------------------
    # 5️⃣  Fallback (should never happen)
    if overall_best_seq is None:
        n = 1024
        overall_best_seq = _uniform_distribution(n)
        overall_best_score = _score_from_seq(overall_best_seq)
        print(f"Fallback uniform length={n} score={overall_best_score:.6f}")

    print(
        f"Returning best sequence of length {len(overall_best_seq)} "
        f"with score {overall_best_score:.6f}"
    )
    return overall_best_seq.tolist()
