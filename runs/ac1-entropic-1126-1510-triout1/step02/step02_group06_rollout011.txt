def optimize_via_lp(init_seq, time_deadline, max_iter=30, t=0.3):
    f = init_seq normalized sum = 1.
    n = len(f)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    for i in range(max_iter):
        if time.time() >= time_deadline:
            break
        M = np.max(np.convolve(f, f))
        # solve LP
        g = solve_lp(f, M)
        if g is None:
            break
        # rescale g to sum=1
        sum_g = np.sum(g)
        if sum_g <= 0:
            break
        g = g / sum_g
        # blend
        f_new = (1 - t) * f + t * g
        # ensure sum normalization (should sum to 1)

        # evaluate score
        cur_score = evaluate_sequence(f_new.tolist())

        if cur_score < best_score:

            best_f = f_new.copy()
            print(f"[LP iter {i}] new best score {best_score:.6f}")
        f = f_new
        # adapt t maybe decrease as improvement small
    return best_f.tolist(), best_score

def solve_lp_for_f(f, M, B_max=1e6):

    n = len(f)
    b = cp.Variable(n)
    constraints = [b >= 0, b <= B_max, cp.conv(cp.Constant(f), b) <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=1000, verbose=False)
    if prob.status not in ["optimal","optimal_inaccurate"]:
        return None
    return np.array(b.value).flatten()

def lp_optimize_length(n, deadline, t=0.2, max_iters=30):

    # init seq: arcsine normalized sum to 1
    f = np.array(arcsine_sequence(n))
    f = f / f.sum()
    best_seq = f.copy()
    best_score = evaluate_sequence(f.tolist())
    for it in range(max_iters):
        cur_time = time.time()
        if cur_time >= deadline:
            break
        M = np.max(np.convolve(f, f))
        g_raw = solve_lp_for_f(f, M, B_max=1e6)
        if g_raw is None:
            print("LP solve failed")
            break
        # rescale to sum 1
        g = g_raw / np.sum(g_raw) if np.sum(g_raw)>0 else g_raw
        # blend
        f_new = (1 - t) * f + t * g
        # renormalize
        f_new = f_new / np.sum(f_new)
        score = evaluate_sequence(f_new.tolist())
        if score < best_score:
            best_score = score
            best_seq = f_new.copy()
            print(f"[LP iteration {it}] new best {best_score:.6f}")
        f = f_new
        # reduce t if improvement small
        if it > 0 and best_score - score < 1e-6:
            t *= 0.9
    return best_seq.tolist(), best_score

   a = np.array(init_seq)
   best_a = a.copy()
   best_metric = metric_of(a)
   for eps in epsilon_list:
       eta = 1.0
       for it in range(max_iter_per_eps):
           if time.time() >= deadline:
               return best_a.tolist(), best_metric
           conv = np.convolve(a, a)
           curr_metric = 2*len(a)*conv.max()
           if curr_metric < best_metric:
               best_metric = curr_metric
               best_a = a.copy()
           # gradient
           w = np.exp(conv / eps)
           w_sum = w.sum()
           if w_sum == 0:

           rev_a = a[::-1]

           # update multiplicative

           # clip exponent

           upd = np.exp(exp_arg)
           cand = a * upd
           cand_sum = cand.sum()
           if cand_sum <= 0:

           else:
               cand = cand / cand_sum
           cand_metric = metric_of(cand)
           if cand_metric < curr_metric:
               a = cand
               eta = min(eta * 1.1, 5.0)

               eta *= 0.6

"""
Propose a step‑function (non‑negative sequence) that minimizes
``evaluate_sequence``.  

Algorithm
---------

1.  For several lengths (≈1200‑3600) generate a few different
    initial sequences:
        * the classic arcsine coefficients,
        * a lightly perturbed arcsine,
        * a random Dirichlet sample.

2.  Each initial sequence is refined by *smoothed mirror‑descent* on
    the simplex.
    •  The objective is the peak of the self‑convolution.
    •  Instead of the non‑smooth max we minimise a
       smooth approximation
       `log‑sum‑exp(conv/ε)`.  This yields a dense gradient that
       guides the update.
    •  The update is multiplicative (mirror‑descent) and the vector
       is re‑normalised to keep the L¹‑norm equal to 1.
    •  A schedule of decreasing temperatures ``ε`` (0.5 → 0.01)
       together with an adaptive step size explores the space
       efficiently.

3.  The best sequence seen across all lengths and starts (lowest
    metric) is returned.  The result is clipped to the limits
    enforced by ``evaluate_sequence`` (0 ≤ x ≤ 1000).

The whole optimisation respects the 1000 s wall‑time budget; the
inner loops stop early if the deadline is reached.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> list[float]:
    """First n coefficients of (1‑z)^‑½, normalised to sum 1."""
    seq = []
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    seq = np.array(seq) / total
    return seq.tolist()


# ----------------------------------------------------------------------
def metric_of(seq: np.ndarray) -> float:
    """
    Fast evaluation of the metric when sum(seq) == 1:
        2 * n * max(conv(seq, seq))
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
def smoothed_gradient(seq: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smooth approximation
        φ_ε(a) = ε·log( Σ_k exp( (a*a)_k / ε ) )
    with respect to a (keeping the L¹‑norm equal to 1).

    Returns a dense gradient vector of length n.
    """
    n = seq.shape[0]
    # self‑convolution
    conv = np.convolve(seq, seq)                # length 2n‑1
    # soft‑max weights
    w = np.exp(conv / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # grad_j = 2· Σ_i a_i·w_{i+j}
    # which is a sliding dot‑product → a convolution of rev(a) and w.
    rev_a = seq[::-1]                           # length n
    full = np.convolve(rev_a, w)               # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
def smoothed_mirror_descent(
    init_seq: list[float],
    deadline: float,
    eps_schedule: list[float],
    max_iter_per_eps: int = 600,
) -> tuple[list[float], float]:
    """
    Refines ``init_seq`` (already normalised to L¹‑norm 1) by
    smoothed mirror‑descent.  The routine stops when the deadline is
    reached or when the allotted iteration budget is exhausted.
    Returns the best sequence encountered and its metric.
    """
    a = np.array(init_seq, dtype=np.float64)
    n = a.shape[0]

    best_a = a.copy()
    best_score = metric_of(a)

    for eps in eps_schedule:
        # step size – will be adapted
        eta = 0.8
        it = 0
        while time.time() < deadline and it < max_iter_per_eps:
            it += 1

            # current metric
            cur_score = metric_of(a)
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_a = a.copy()

            # gradient of the smooth max
            grad = smoothed_gradient(a, eps)

            # multiplicative (mirror) update, with clipping for safety
            exp_arg = -eta * grad
            # guard against overflow/underflow of the exponent
            exp_arg = np.clip(exp_arg, -50.0, 50.0)
            upd = np.exp(exp_arg)

            cand = a * upd
            s = cand.sum()
            if s <= 0.0 or not np.isfinite(s):
                cand = np.full_like(a, 1.0 / n)
            else:
                cand = cand / s

            cand_score = metric_of(cand)

            # accept if we improve, otherwise shrink step size
            if cand_score < cur_score:
                a = cand
                eta = min(eta * 1.12, 5.0)
                # occasional larger step‑size increase after a good move
                if it % 250 == 0:
                    eta = min(eta * 1.3, 5.0)
            else:
                eta *= 0.6
                # occasional random jitter to escape flat plateaux
                if it % 200 == 0:
                    i, j = np.random.choice(n, 2, replace=False)
                    delta = (np.random.rand() - 0.5) * 2e-3   # ±0.001
                    jitter = a.copy()
                    jitter[i] = max(0.0, jitter[i] + delta)
                    jitter[j] = max(0.0, jitter[j] - delta)
                    jitter = jitter / jitter.sum()
                    jitter_score = metric_of(jitter)
                    if jitter_score < cur_score:
                        a = jitter
                        eta = max(eta, 0.1)

            # very small step size → stop early
            if eta < 1e-8:
                break

    return best_a.tolist(), best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Main entry point used by the evaluation harness.
    Searches for a good non‑negative step‑function within the
    allotted time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # lengths we will explore – a trade‑off between quality and speed
    lengths = [1200, 1800, 2400, 3000, 3600]

    best_seq = None
    best_score = float("inf")

    # temperature schedule for the smoothed max
    eps_schedule = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # --------------------------------------------------------------
        # 1) build a small pool of starting points
        start_pool = []

        # a) pure arcsine
        start_pool.append(arcsine_sequence(n))

        # b) arcsine with mild Gaussian perturbations
        base_arc = np.array(arcsine_sequence(n))
        for scale in [0.02, 0.05]:
            noisy = base_arc + scale * np.random.randn(n)
            noisy = np.clip(noisy, 0.0, None)
            if noisy.sum() == 0.0:
                noisy = base_arc.copy()
            noisy = noisy / noisy.sum()
            start_pool.append(noisy.tolist())

        # c) random Dirichlet (exponential) sample
        rand = np.random.exponential(1.0, size=n)
        rand = rand / rand.sum()
        start_pool.append(rand.tolist())

        # --------------------------------------------------------------
        # 2) run the optimiser for each start, respecting the remaining budget
        remaining = deadline - time.time()
        # split the remaining time evenly among the remaining lengths
        per_len_deadline = time.time() + max(5.0, remaining / (len(lengths) - lengths.index(n)))

        for idx, init in enumerate(start_pool):
            if time.time() >= per_len_deadline:
                break
            print(f"  start #{idx+1}/{len(start_pool)} (length {n})")
            seq, score = smoothed_mirror_descent(
                init_seq=init,
                deadline=per_len_deadline,
                eps_schedule=eps_schedule,
                max_iter_per_eps=600,
            )
            print(f"    candidate metric = {score:.6f}")

            if score < best_score:
                best_score = score
                best_seq = seq
                print(f"*** New overall best (len={n}) score = {best_score:.6f}")

    if best_seq is None:
        # fallback – should never happen, but stay safe
        best_seq = arcsine_sequence(2000)

    # Final safety: enforce the limits required by ``evaluate_sequence``
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    print(f"\n--- FINAL RESULT ---\nBest length = {len(best_seq)} "
          f"with metric = {best_score:.6f}")
    return best_seq
