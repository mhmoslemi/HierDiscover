"""
Hybrid optimizer for the step‑function minimisation problem.

The algorithm works in two phases for a fixed length *n* (chosen from the
previous best or a default size).

Phase 1 – Smooth Lₚ optimisation:
    * The objective  ‖a∗a‖ₚ  (p large) approximates the L∞ norm.
    * Adam‑style updates are performed on the probability simplex
      (non‑negative entries summing to 1) using a Euclidean projection.
    * The exponent *p* is increased gradually (4 → 8 → 16 → 32) so the
      optimisation first sees a smooth landscape and then focuses on the true
      maximum of the self‑convolution.

Phase 2 – Sub‑gradient refinement:
    * A classic projected sub‑gradient step on the exact max‑convolution
      entry (the same update used in the previous state‑of‑the‑art code) is
      applied for a few thousand iterations with momentum.
    * This eliminates any remaining “plateau” effects after the smooth phase.

The best sequence observed (according to the black‑box ``evaluate_sequence``)
is kept and finally returned.  All numbers are non‑negative and the sum is
forced to 1, which does not affect the evaluation score because the
objective is homogeneous.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine probability vector of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _fft_self_convolve(a: np.ndarray, size: int) -> np.ndarray:
    """
    Full self‑convolution a*a using FFT with a given zero‑padded size.
    Returns an array of length 2·len(a)‑1.
    """
    A = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(A * A, n=size)
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def _fft_convolve_valid(a: np.ndarray, b: np.ndarray, size: int) -> np.ndarray:
    """
    Return the *valid* convolution of a (len=n) with b (len=m≥n) using FFT.
    Result length = m‑n+1 (=len(a) if m = 2·len(a)‑1).
    """
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv_full = np.fft.irfft(A * B, n=size)
    n = a.shape[0]
    # valid part starts at index n‑1
    start = n - 1
    return conv_full[start : start + n]


def _adapt_sequence(seq, n: int) -> np.ndarray:
    """
    Take an existing sequence (list/ndarray) and adapt it to length n.
    If the sequence is shorter it is padded with zeros, if longer it is truncated.
    The result is normalised to sum to 1.
    """
    seq = np.asarray(seq, dtype=np.float64)
    if seq.size < n:
        seq = np.concatenate([seq, np.zeros(n - seq.size, dtype=np.float64)])
    elif seq.size > n:
        seq = seq[:n]
    total = seq.sum()
    if total <= 0.0:
        seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        seq = seq / total
    return seq


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a low‑score step‑function within the allotted time budget.
    Returns the best list of non‑negative heights found.
    """
    np.random.seed(seed)
    # reserve a couple of seconds for finalisation / printing
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # Choose a base length.  Prefer the previously discovered length,
    # otherwise fall back to a moderate size.
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
        if base_n < 500:
            raise ValueError
    except Exception:
        base_n = 8000

    # Try a few nearby lengths (the exact optimum is not known)
    lengths = [
        max(500, int(base_n * f))
        for f in (0.75, 1.0, 1.25, 1.5)
        if int(base_n * f) >= 500
    ]

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # allocate a share of the remaining time for this length
        slice_deadline = time.time() + max(
            0.1, (deadline - time.time()) * 0.3
        )  # about 30 % of what is left

        # ------------------------------------------------------------------
        # Initialise a feasible sequence
        try:
            # use the historic sequence if it has the right size,
            # otherwise adapt it.
            init_seq = _adapt_sequence(height_sequence_1, n)  # type: ignore[name-defined]
        except Exception:
            init_seq = _arcsine_distribution(n)

        a = init_seq.copy()  # current iterate – always a probability vector
        best_local_seq = a.copy()
        best_local_score = float("inf")

        # Pre‑compute FFT sizes (constant for a given n)
        size_self = _next_pow2(2 * n - 1)
        size_grad = _next_pow2(3 * n - 2)

        # ------------------------------------------------------------------
        # Phase 1 – smooth Lₚ optimisation with Adam
        # ------------------------------------------------------------------
        m = np.zeros_like(a)          # first moment
        v = np.zeros_like(a)          # second moment
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        eta0 = 0.25                    # base learning rate
        p = 4                           # start exponent
        iteration = 0

        while time.time() < slice_deadline:
            iteration += 1

            # ----- self‑convolution and current score (internal estimate) -----
            fft_a = np.fft.rfft(a, n=size_grad)
            conv_self_full = np.fft.irfft(fft_a * fft_a, n=size_grad)
            conv_self = conv_self_full[: 2 * n - 1]          # length 2n‑1
            max_self = conv_self.max()
            cur_score = 2.0 * n * max_self                    # sum(a)=1

            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local_seq = a.copy()
                print(
                    f"[{time.time():.1f}] len={n} phase=Lp iter={iteration} p={p} score={best_local_score:.6f}"
                )

            # ----- schedule for the exponent p (gradually sharpen) -----
            if iteration % 2000 == 0 and p < 32:
                p = min(p * 2, 32)

            # ----- gradient of (∥a∗a∥ₚ) -----
            # avoid negative numerical noise
            conv_self = np.maximum(conv_self, 0.0)
            y = conv_self ** (p - 1)               # shape 2n‑1
            fft_y = np.fft.rfft(y, n=size_grad)

            # valid convolution a * y (size n)
            conv_valid_full = np.fft.irfft(fft_a * fft_y, n=size_grad)
            grad = 2.0 * conv_valid_full[n - 1 : n - 1 + n]  # length n

            # ----- Adam update + projection onto the simplex -----
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** iteration)
            v_hat = v / (1.0 - beta2 ** iteration)
            eta = eta0 / np.sqrt(iteration)
            a = a - eta * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a, s=1.0)

        # ------------------------------------------------------------------
        # Phase 2 – classic sub‑gradient refinement (momentum)
        # ------------------------------------------------------------------
        velocity = np.zeros_like(a)
        beta = 0.9
        eta0_sub = 0.5
        sub_iter = 0
        while time.time() < deadline:
            sub_iter += 1
            # current self‑convolution
            conv_self = _fft_self_convolve(a, size_self)
            max_self = conv_self.max()
            cur_score = 2.0 * n * max_self
            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local_seq = a.copy()
                print(
                    f"[{time.time():.1f}] len={n} phase=Sub iter={sub_iter} score={best_local_score:.6f}"
                )

            # active index of the maximal convolution entry
            k_max = int(np.argmax(conv_self))
            idx = k_max - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            subgrad = np.zeros_like(a)
            subgrad[mask] = 2.0 * a[idx[mask]]

            # momentum and diminishing step size
            velocity = beta * velocity + (1.0 - beta) * subgrad
            eta = eta0_sub / np.sqrt(sub_iter)
            a = a - eta * velocity
            a = _project_onto_simplex(a, s=1.0)

            # occasional tiny random jitter to escape shallow plateaux
            if sub_iter % 500 == 0 and np.random.rand() < 0.07:
                a += np.random.randn(n) * 1e-4
                a = _project_onto_simplex(a, s=1.0)

            # stop early if we have exhausted the global deadline
            if time.time() >= deadline:
                break

        # ------------------------------------------------------------------
        # Final evaluation with the black‑box (more reliable than the internal estimate)
        # ------------------------------------------------------------------
        try:
            final_score = evaluate_sequence(best_local_seq.tolist())
        except Exception:
            final_score = best_local_score  # fallback

        if final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = best_local_seq.copy()
            print(
                f"*** New global best → len={n} score={overall_best_score:.6f}"
            )

        # early exit if we have already beaten the target
        if overall_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback in the unlikely case that nothing was produced
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())

    print(
        f"Final best score: {overall_best_score:.6f} (length = {len(overall_best_seq)})"
    )
    # Convert to plain Python list as required by the runner
    return overall_best_seq.tolist()
