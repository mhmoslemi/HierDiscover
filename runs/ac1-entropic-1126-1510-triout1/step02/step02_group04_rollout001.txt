def smooth_grad(a, alpha):
    conv = np.convolve(a, a)  # length 2n-1
    # stabilize by subtract max before exp
    maxc = conv.max()
    exp_term = np.exp(alpha * (conv - maxc))  # avoid overflow
    w = exp_term / exp_term.sum()
    # compute gradient
    w_rev = w[::-1]
    # convolve a (len n) with w_rev (len 2n-1)
    grad_full = np.convolve(a, w_rev)  # length n + (2n-1) - 1 = 3n - 2
    grad = 2.0 * grad_full[(len(w)-1):(len(w)-1)+len(a)]
    return grad
```

def _project_to_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Gradient‑descent on a smooth approximation of the ∞‑norm of a*a.

We work on the probability simplex (non‑negative entries, sum = 1).
The true objective is   score = 2·n·max_i (a∗a)[i] .
Because `max` is non‑smooth we minimise the log‑sum‑exp
    f(a)= (1/α)·log Σ_i exp(α·(a∗a)[i])
whose gradient is a weighted sum of the individual quadratic
gradients.  For a given α the gradient can be computed by a single
convolution:

    w_i = exp(α·((a∗a)[i]/max(a∗a) – 1))      (softmax on the normalised
                                                convolution)
    g = 2·conv(a, rev(w))   (rev = w reversed)

We perform a projected (clip‑and‑renormalise) gradient step with a
decreasing step size η and increase α during the run so that the
approximation becomes tighter.  Every few hundred iterations a tiny
random perturbation is added to avoid being trapped in a flat region.
The best (unsmoothed) sequence seen so far is kept and finally
returned.

The implementation uses only numpy, runs in ~O(n log n) per iteration
and comfortably fits into the 1000 s budget.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – guarantees non‑negative and sum=1
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # based on the algorithm of Wang & Carreira‑Perpiñán (2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    # find the rho where u_i > (cssv_i / i)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – return uniform point
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast scoring: 2·n·max(a∗a)  (sum(a)=1 → denominator = 1)
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.size
    conv = np.convolve(a, a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Compute the softmax weights for a given α.
# The weights sum to 1 and concentrate on the maximal convolution entry.
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Return w_i ∝ exp(α·(conv_i / max(conv) – 1))."""
    maxc = conv.max()
    if maxc <= 0.0:
        # degenerate case – uniform weights
        return np.full_like(conv, 1.0 / conv.size)
    # normalise to unit maximum, then shift so that the maximum gets exponent 0
    scaled = conv / maxc - 1.0          # max entry → 0, others ≤ 0
    # avoid overflow/underflow
    exp_term = np.exp(alpha * scaled)
    w = exp_term / exp_term.sum()
    return w


# ----------------------------------------------------------------------
# Gradient of the smooth max (log‑sum‑exp) w.r.t. the probability vector.
# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute ∇_a  (1/α)·log Σ_i exp(α·(a∗a)[i]).
    Implemented via two convolutions using the softmax weights.
    """
    n = a.size
    conv = np.convolve(a, a)                     # length 2n‑1
    w = _softmax_weights(conv, alpha)            # length 2n‑1, sum = 1
    # gradient = 2 * Σ_i w_i * D_i a  = 2 * conv(a, rev(w)) restricted to n entries
    w_rev = w[::-1]
    full = np.convolve(a, w_rev)                 # length n + (2n‑1) – 1 = 3n‑2
    start = len(w) - 1                           # = 2n‑2
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# A tiny random perturbation, kept on the simplex.
# ----------------------------------------------------------------------
def _random_perturb(a: np.ndarray, magnitude: float = 0.01) -> np.ndarray:
    """Add a small zero‑mean Gaussian perturbation and re‑project."""
    pert = a + magnitude * (np.random.rand(a.size) - 0.5)
    return _project_to_simplex(pert)
