   `max(conv) = 2*n*max(a*a)/(Σa)^2`.  The best shape is kept as a

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero, put everything on the first coordinate
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Linear convolution of u and v using FFT.
    Output length = len(u) + len(v) - 1.
    """
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Return the full linear convolution a*a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a  (w is assumed fixed and non‑negative).
    Implements the analytic expression using FFT for speed.
    """
    n = a.shape[0]
    # reverse a for convolution with the weight vector
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)          # length = (2n‑1) + n‑1 = 3n‑2
    start = n - 1                                 # position of the “zero‑shift”
    grad = 2.0 * conv[start:start + n]            # factor 2 for the two symmetric terms
    # correct the diagonal terms that were counted only once
    diag_w = w[0:2 * n - 1:2]                      # w_{2i}, i = 0,…,n‑1
    grad -= a * diag_w
    return grad

    conv = _self_convolution(a, _SIZE_FFT)   # _SIZE_FFT is set globally before use
    max_val = conv.max()
    # Because the denominator Σa² = (sqrt(2n))² = 2n after projection,
    # the official score equals the maximal convolution entry.
    return float(max_val)

        beta = np.random.rand() * 12.0
        i = np.arange(n, dtype=float)
        arr = (1.0 / np.sqrt(i + beta + 1.0) +
               1.0 / np.sqrt((n - 1 - i) + beta + 1.0))
        _sample_and_evaluate(arr)

    # power‑law
    for _ in range(120):
        beta = np.random.rand() * 12.0
        p = np.random.rand() * 1.8 + 0.2          # exponent ∈ [0.2, 2.0]
        arr = 1.0 / (np.arange(n) + beta + 1.0) ** p
        _sample_and_evaluate(arr)

    # exponential decay (symmetric)
    for _ in range(120):
        decay = np.random.rand() * 0.015 + 0.0003   # ∈ [3e‑4, 1.53e‑2]
        i = np.arange(n, dtype=float)
        arr = np.exp(-decay * i) + np.exp(-decay * (n - 1 - i))
        _sample_and_evaluate(arr)

    # cosine‑based shape (large at ends, small in middle)
    for _ in range(120):
        p = np.random.rand() * 6.0 + 0.3            # exponent ≥0.3
        arr = np.abs(np.cos(np.pi * np.arange(n) / (n - 1))) ** p
        _sample_and_evaluate(arr)

    # mixture of uniform and inverse‑sqrt
    for _ in range(120):
        beta = np.random.rand() * 12.0
        lam = np.random.rand()
        inv = 1.0 / np.sqrt(np.arange(n) + beta + 1.0)
        arr = lam * np.ones(n) + (1.0 - lam) * inv
        _sample_and_evaluate(arr)

    # random block candidates (few large constant blocks)
    for _ in range(300):
        nb = np.random.randint(2, 8)                     # 2‑7 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb) + 0.05                 # avoid exact zeros
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        _sample_and_evaluate(cand)

    print(f"[shape search] best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. REFINEMENT VIA PROJECTED ADAM ON THE EXACT MAX
    if best_a is None:
        # fallback (should never happen)
        a = np.ones(n) * target_sum / n
    else:
        a = best_a.copy()

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Base learning rate – about half of the average coordinate size
    base_lr = target_sum * 0.5 / n
    iteration = 0
    no_improve = 0

    while time.time() < deadline and iteration < 800_000:
        # compute convolution and current score
        conv = _self_convolution(a, _SIZE_FFT)
        cur_max = conv.max()
        cur_score = float(cur_max)          # because Σa = target_sum ⇒ denominator = 2n
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            no_improve = 0
            print(f"[{iteration:07d}] new best = {best_score:.6f}")
        else:
            no_improve += 1

        # identify the set of indices attaining the maximum (with tiny tolerance)
        tol = 1e-12
        mask = conv >= cur_max - tol
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            # numerical safety fallback
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = np.zeros_like(conv)
            w[mask] = 1.0 / cnt

        # gradient of the (sub‑)objective
        grad = _weighted_gradient(a, w, _SIZE_FFT)

        # Adam moment updates
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional random perturbation to escape plateaus
        if iteration % 4000 == 0 or no_improve > 4000:
            noise = np.random.randn(n) * target_sum * 5e-4
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)
            no_improve = 0
            print(f"[{iteration:07d}] random perturbation")

        a = a_new

    # --------------------------------------------------------------
    # 3. FINAL RANDOM BLOCK POLISHING
    for _ in range(250):
        nb = np.random.randint(2, 8)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb) + 0.05
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = _score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[block] improved to {best_score:.6f}")

    total_time = time.time() - start_time
    print(f"[final] best score = {best_score:.6f} after {iteration} iterations "
          f"({total_time:.1f}s)")

    # Return as a plain Python list of floats
    return best_a.tolist()
