"""
Propose a step‑function that minimizes ``evaluate_sequence``.

The optimizer works in three phases:

1. **Warm start** – use the best known construction (``height_sequence_1``)
   if it exists, otherwise a uniform vector.  The vector is normalised to
   have L1‑norm 1 (the evaluation is scale‑invariant).

2. **Projected sub‑gradient descent** – minimise the maximal convolution
   entry  M = max(a * a)  under the simplex constraint  a≥0 , Σa = 1.
   For the active shift(s) attaining M we compute a sub‑gradient
   (which for a shift k is simply ``a[k‑i]`` where defined) and take a
   step in the negative direction, projecting back onto the simplex.
   A decaying learning‑rate and a simple acceptance test are used.

3. **Random‑pair polishing** – a cheap simulated‑annealing style
   refinement that repeatedly moves a small amount of mass between two
   positions while preserving the simplex constraint.  The best sequence
   seen throughout the whole run is returned.

The implementation uses FFT based convolution (``np.fft``) for speed,
runs completely on a single thread, and falls back gracefully if
``cvxpy`` is unavailable.  All numbers are kept non‑negative and the
final list is a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: project a vector onto the probability simplex {x≥0, Σx = z}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = z}.
    Returns a new array with the same shape as ``v``.
    """
    if z < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # Duchi et al. 2008 implementation
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be zero
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1           # last true index (0‑based)
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w


# ----------------------------------------------------------------------
# Fast convolution using FFT (real‑valued inputs)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Return np.convolve(a, a) using FFT (real output)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[: 2 * n - 1]
    # Numerical noise may produce tiny negatives – clip to 0
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of the maximum convolution entry
# ----------------------------------------------------------------------
def _subgradient_max(conv: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of  M(a)=max_k (a*a)[k] .
    For every active shift k (where the max is attained) we add the
    vector g_i = a[k-i]  (if the index is in range).
    """
    n = a.shape[0]
    M = conv.max()
    # tolerance to capture ties caused by floating errors
    tol = 1e-12
    active = np.where(conv >= M - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        grad[i] += a[j]
    # factor 2 is irrelevant for the descent direction (constant scaling)
    return grad


# ----------------------------------------------------------------------
# A cheap random pairwise mass move (keeps the L1 sum exactly 1)
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Pick two distinct indices i,j and transfer a small amount of mass."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    delta = (np.random.rand() - 0.5) * sigma
    # Clip delta so we never go negative
    if delta > 0:
        delta = min(delta, seq[j])
    else:
        delta = -min(-delta, seq[i])

    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    # Projection is not needed – the operation already preserves sum and
    # non‑negativity (by clipping).
    return new_seq
