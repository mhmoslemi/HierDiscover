"""
Optimize a non‑negative step function (vector a, sum(a)=1) to minimise
    score = 2·n·max(a*a) / (sum(a)**2)   (the denominator is invariant to scaling)

The objective is equivalent to minimising the peak of the self‑convolution
(b = a * a).  We use a smoothed‑max surrogate:
    Lα(a) = (1/α)·log Σ_k exp(α·b_k)
Its gradient w.r.t. a can be written as a convolution of the reversed a
with the soft‑max weights w_k ∝ exp(α·(b_k−max(b))).  By slowly increasing
α we first move a according to a smooth objective, then focus on the true
peak.  Each step is projected back onto the probability simplex.
The method is a simple momentum gradient descent with an adaptive step‑size
and occasional tiny random jitter to escape plateaus.

The search is performed for several lengths (derived from the previously
found sequence if available) and for a few random initialisations.  The
best sequence found (according to the black‑box `evaluate_sequence`) is
returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Return a * a via real FFT (full length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT (real output)."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]
