"""
propose_candidate
-----------------

A hybrid search that combines an exponentiated‑gradient descent on the simplex
with a smooth‑max surrogate for the convolution peak, occasional random
re‑starts, and a cheap block‑search / local‑move refinement at the end.

Key ideas
~~~~~~~~~
* The objective of ``evaluate_sequence`` is scale‑invariant, so we keep the
  ℓ¹‑norm fixed to √(2 n).  With that normalisation the score equals the
  maximum entry of the self‑convolution.
* For a given sequence `a` the gradient of the (soft) maximum of the
  convolution is a weighted sum of shifted copies of `a`.  Using the
  soft‑max weights yields a smooth surrogate that is cheap to compute with a
  single convolution.
* When the temperature becomes very low we fall back to a true sub‑gradient
  that only uses the indices attaining the current maximum.
* The update is performed by exponentiated‑gradient (mirrored) descent:
  `a_i ← a_i·exp(‑η · grad_i)`, which automatically respects non‑negativity.
  After each step the vector is renormalised to the target ℓ¹‑norm.
* The algorithm runs for the whole time budget, performing several random
  restarts.  After the main loop we run a fast block‑search (piecewise‑constant
  candidates) and finally a few random local moves to polish the best
  solution found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, mu: float) -> np.ndarray:
    """Return soft‑max weights w_k ∝ exp((conv_k‑max)/mu)."""
    max_val = conv.max()
    # values are ≤ 0, safe exponential
    w = np.exp((conv - max_val) / mu)
    s = w.sum()
    if s == 0.0:                     # numeric safety
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= s
    return w


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate.
    g_i = 2 * Σ_k w_k * a_{k‑i}   (indices out of range are ignored).
    """
    n = a.size
    # convolution of the weight vector with the reversed a
    conv_wr = np.convolve(w, a[::-1])
    # central slice gives Σ_k w_k a_{k‑i}
    grad = 2.0 * conv_wr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _gradient_hardmax(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the true max (averaged over all maximal indices).
    For each maximal index k the contribution is 2·a_{k‑i}.
    """
    n = a.size
    max_val = conv.max()
    idxs = np.where(conv == max_val)[0]
    grad = np.zeros_like(a)
    for k in idxs:
        if k < n:
            # i = 0 … k
            grad[: k + 1] += 2.0 * a[k :: -1]
        else:
            # i = k-(n‑1) … n‑1
            off = k - (n - 1)
            # reversed slice a[n‑1 : off‑1 : -1]
            grad[off:] += 2.0 * a[n - 1 : off - 1 : -1]
    grad /= float(len(idxs))
    return grad
