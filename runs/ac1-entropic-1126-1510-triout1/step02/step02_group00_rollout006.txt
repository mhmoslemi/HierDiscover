def _fft_convolve(u, v):
    m = u.shape[0] + v.shape[0] - 1
    size = 1 << ((m - 1).bit_length())
    U = np.fft.rfft(u, n=size)
    V = np.fft.rfft(v, n=size)
    conv = np.fft.irfft(U*V, n=size)[:m]
    return conv

"""
Hybrid smoothed‑max gradient descent with simplex projection.

The objective is 
    score = 2·n·max(a∗a) / (sum(a))²,
but after projecting onto the probability simplex (sum=1) the denominator is
constant, so we minimise max(a∗a).  The algorithm repeatedly:

1. Computes the self‑convolution b = a∗a.
2. Forms a soft‑max weighting w = exp(α·(b−max(b))) / Σexp(...).
   For α→∞ this weight concentrates on the maximal entry, reproducing a
   sub‑gradient of max(b).  For moderate α the gradient is smoother,
   allowing the optimiser to escape plateaux.
3. Obtains the gradient of the smoothed objective:
      ∇J(a) = 2·corr(w, a)               (corr = cross‑correlation)
   which is evaluated efficiently by an FFT‑based convolution.
4. Updates a with a diminishing step size η = η₀/√t and projects back onto the
   probability simplex.
5. Increases the temperature α gradually (α←α·γ, γ≈1.001) so the optimisation
   focuses more on the true max later on.
6. After the main phase a short sub‑gradient refinement using the exact
   maximal convolution entry is performed.

Multiple lengths (around the previously found size) and several random
initialisations are tried within the allotted time budget.  The best
sequence found (according to the provided `evaluate_sequence`) is returned.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_sequence(n: int, kind: str) -> np.ndarray:
    """Return a probability vector of length n."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif kind == "arcsine":
        a = _arcsine_distribution(n)
    elif kind == "prev":
        try:
            a = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if a.size != n:
                raise ValueError
        except Exception:
            a = _arcsine_distribution(n)
        if a.sum() == 0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / a.sum()
    else:
        a = _arcsine_distribution(n)

    return _project_onto_simplex(a, s=1.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Return a * a using FFT (real input). Result length = 2·len(a)‑1."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())  # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=size)
    C = np.fft.irfft(A * A, n=size)
    return C[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Convolution of two real vectors via FFT. Length = len(u)+len(v)‑1."""
    m = u.shape[0] + v.shape[0] - 1
    size = 1 << ((m - 1).bit_length())
    U = np.fft.rfft(u, n=size)
    V = np.fft.rfft(v, n=size)
    conv = np.fft.irfft(U * V, n=size)[:m]
    return conv


# ----------------------------------------------------------------------
def _smoothed_max_gradient(a: np.ndarray, alpha: float):
    """
    Compute b = a*a and the gradient of the soft‑max approximation of max(b).

    Returns (b, grad) where grad = 2·corr(w, a) with
    w_i = exp(alpha·(b_i−max(b))) / Σ_j exp(...).
    """
    b = _fft_self_convolve(a)
    b_max = b.max()
    # soft‑max weights (stable subtraction of the maximum)
    exp_vals = np.exp(alpha * (b - b_max))
    w = exp_vals / exp_vals.sum()
    # correlation via convolution with reversed a
    a_rev = a[::-1]
    conv = _fft_convolve(w, a_rev)               # length = 3·n‑2
    start = len(a) - 1
    grad = 2.0 * conv[start:start + len(a)]
    return b, grad


# ----------------------------------------------------------------------
def _subgradient_refine(a: np.ndarray, n: int, time_budget: float) -> np.ndarray:
    """
    Short refinement using the exact max‑convolution sub‑gradient.
    Runs for at most `time_budget` seconds.
    """
    end = time.time() + time_budget
    velocity = np.zeros_like(a)
    beta = 0.9
    eta0 = 0.5
    it = 0
    while time.time() < end:
        it += 1
        b = _fft_self_convolve(a)
        k = int(np.argmax(b))
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]
        velocity = beta * velocity + (1.0 - beta) * g
        eta = eta0 / np.sqrt(it)
        a = a - eta * velocity
        a = _project_onto_simplex(a, s=1.0)
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000):
    """
    Search for a non‑negative step‑function with a low `evaluate_sequence`
    score.  The method combines a soft‑max gradient descent (with temperature
    annealing) and a final exact sub‑gradient polishing step.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # keep 2 s for final I/O

    # base length – reuse the historic construction if available
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # a modest set of lengths around the base (all ≥ 500)
    lengths = sorted(set(
        max(500, int(base_n * f)) for f in (0.75, 1.0, 1.25, 1.5, 2.0)
    ))

    best_seq = None
    best_score = float('inf')

    for n in lengths:
        if time.time() >= deadline:
            break

        # allocate a fraction of the remaining time to this length
        slice_deadline = time.time() + max(0.1, (deadline - time.time()) * 0.3)

        for init_kind in ('prev', 'arcsine', 'uniform', 'random'):
            if time.time() >= slice_deadline:
                break

            a = _initial_sequence(n, init_kind)

            # hyper‑parameters for the smoothed‑max phase
            alpha = 5.0
            alpha_max = 200.0
            alpha_growth = 1.001
            lr0 = 0.5
            it = 0

            # track best internal estimate (2·n·max(b))
            local_best_a = a.copy()
            local_best_est = float('inf')

            while time.time() < slice_deadline:
                it += 1
                b, grad = _smoothed_max_gradient(a, alpha)

                # diminishing step size η = η₀ / √t
                eta = lr0 / np.sqrt(it)
                a = a - eta * grad
                a = _project_onto_simplex(a, s=1.0)

                # anneal temperature
                alpha = min(alpha_max, alpha * alpha_growth)

                # occasional tiny jitter to avoid stagnation
                if it % 2000 == 0 and np.random.rand() < 0.1:
                    a += np.random.randn(n) * 1e-4
                    a = _project_onto_simplex(a, s=1.0)

                # periodic bookkeeping
                if it % 500 == 0:
                    cur_est = 2.0 * n * b.max()
                    if cur_est < local_best_est:
                        local_best_est = cur_est
                        local_best_a = a.copy()
                        print(f"[{time.time():.1f}] len={n} init={init_kind} "
                              f"it={it} est={local_best_est:.6f} alpha={alpha:.2f}")

            # short exact sub‑gradient polishing (if time permits)
            refine_time = max(0.0, slice_deadline - time.time())
            if refine_time > 2.0:
                a_refined = _subgradient_refine(local_best_a, n, refine_time)
            else:
                a_refined = local_best_a

            # final (expensive) evaluation with the supplied black‑box
            try:
                final_score = evaluate_sequence(a_refined.tolist())
            except Exception:
                final_score = float('inf')

            if final_score < best_score:
                best_score = final_score
                best_seq = a_refined.copy()
                print(f"*** New global best: len={n} init={init_kind} "
                      f"score={best_score:.6f}")

            # early exit if target achieved
            if best_score <= 1.503:
                break
        if best_score <= 1.503:
            break

    # fallback in the unlikely case of no candidate
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())

    print(f"Final best score: {best_score:.6f} (length = {len(best_seq)})")
    return best_seq.tolist()
