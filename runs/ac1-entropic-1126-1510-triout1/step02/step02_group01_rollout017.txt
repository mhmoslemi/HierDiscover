"""
Propose candidate via softmax-parameterized quasi‑Newton optimization.

The sequence a is represented as a simplex point a = S·softmax(y).
For a temperature μ we minimise the smooth approximation
    f_μ(y) = max_k (a*a)[k] + μ·log Σ exp((a*a)[k]/μ)
whose gradient w.r.t. y can be written analytically.
A continuation schedule (μ → 0) together with L‑BFGS‑B
produces a high‑quality local minimum.
Several initialisations (uniform, V‑shaped, random, reference)
are tried and the best feasible sequence is returned.
A cheap block‑search at the end may capture tiny remaining improvements.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()

# ----------------------------------------------------------------------
def _compute_f_and_grad(y: np.ndarray, mu: float, n: int,
                       S: float) -> tuple[float, np.ndarray]:
    """
    Smooth max objective and its gradient w.r.t. the softmax parameters y.

    Parameters
    ----------
    y : np.ndarray
        Real‑valued vector of length n (free parameters).
    mu : float
        Temperature. mu>0 gives a smooth approximation, mu≈0 gives the hard max.
    n : int
        Length of the underlying sequence.
    S : float
        Target ℓ¹‑norm (sqrt(2·n)).

    Returns
    -------
    f_val : float
        Smooth objective value.
    grad_y : np.ndarray
        Gradient ∂f/∂y (same shape as y).
    """
    # probability simplex
    p = _softmax(y)               # Σp = 1, p_i ≥ 0
    a = S * p                     # actual non‑negative heights

    # full convolution a*a
    conv = np.convolve(a, a)      # length 2·n‑1
    max_c = conv.max()

    # soft‑max weights w_k ∝ exp((conv_k‑max)/mu)
    if mu > 0.0:
        diff = (conv - max_c) / mu
        # clip to avoid overflow in exp()
        diff = np.clip(diff, -700.0, 700.0)
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        # hard max – distribute weight uniformly among all maximisers
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. the heights a
    # g_a_i = 2 * Σ_k w_k a_{k‑i}
    conv_wr = np.convolve(w, a[::-1])          # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]       # slice length n

    # chain rule: ∂f/∂y_i = S·p_i·(g_a_i – Σ_j p_j g_a_j)
    inner = np.dot(g_a, p)
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)

# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable object required by scipy.optimize.minimize."""

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        f, g = _compute_f_and_grad(y, self.mu, self.n, self.S)
        return f, g

"""
Softmax‑parameterised quasi‑Newton optimisation for the step‑function problem.

The heights a_i are expressed as a point on the simplex:
    a = S·softmax(y)          with  S = sqrt(2·n).

For a temperature μ we minimise the smooth surrogate of the
maximum convolution entry

    f_μ(y) = max_k (a*a)[k] + μ·log Σ_k exp((a*a)[k]/μ),

which is differentiable.  Its gradient w.r.t. y is obtained analytically.
A continuation schedule (μ → 0) together with L‑BFGS‑B yields a high‑quality
local minimum.  Several initialisations (uniform, V‑shaped, random,
reference) are tried, a cheap random block‑search finishes the run and a
few projected‑gradient steps on the hard max further polish the result.
The best feasible sequence found within the time budget is returned.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _compute_f_and_grad(y: np.ndarray, mu: float, n: int,
                       S: float) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective and gradient w.r.t. the softmax parameters y.

    Parameters
    ----------
    y : np.ndarray
        Free parameters (real vector of length n).
    mu : float
        Temperature (μ > 0 gives a smooth surrogate, μ≈0 gives the hard max).
    n : int
        Length of the sequence.
    S : float
        Desired ℓ¹‑norm (sqrt(2·n)).

    Returns
    -------
    f_val : float
        Value of the smooth surrogate.
    grad_y : np.ndarray
        Gradient ∂f/∂y (shape (n,)).
    """
    # simplex probabilities and actual heights
    p = _softmax(y)          # Σp = 1, p_i ≥ 0
    a = S * p                # Σa = S

    # full convolution a*a
    conv = np.convolve(a, a)      # length 2·n‑1
    max_c = conv.max()

    # soft‑max weights w_k ∝ exp((conv_k – max)/μ)
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        # hard max: distribute weight uniformly among all maximisers
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. the heights a
    # g_a[i] = 2 * Σ_k w_k * a_{k‑i}
    conv_wr = np.convolve(w, a[::-1])          # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]       # slice length n

    # chain rule to obtain gradient w.r.t. y
    inner = np.dot(g_a, p)                     # scalar = Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """
    Callable wrapper required by scipy.optimize.minimize.
    Returns (objective, gradient) when called.
    """

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        f, g = _compute_f_and_grad(y, self.mu, self.n, self.S)
        return f, g

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5     # tiny safety margin

    # ------------------------------------------------------------------
    # problem size
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    S = math.sqrt(2.0 * n)           # ℓ¹‑norm that makes the score = max·conv

    # ------------------------------------------------------------------
    # best solution bookkeeping
    best_a = np.zeros(n, dtype=float)
    best_score = float("inf")

    # ------------------------------------------------------------------
    # generate initial y‑vectors (free parameters)
    init_y = []

    # 0. uniform (y = 0)
    init_y.append(np.zeros(n, dtype=np.float64))

    # 1. V‑shaped (large at ends, tiny in the centre)
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    vshape = np.abs(idx - mid) + 1e-3
    init_y.append(np.log(vshape))

    # 2. random Gaussian
    init_y.append(np.random.randn(n) * 0.5)

    # 3. reference sequence if available
    if have_ref:
        a_ref = np.asarray(height_sequence_1, dtype=float)
        a_ref = _project_to_simplex(a_ref, S)
        eps = 1e-12
        init_y.append(np.log(a_ref + eps))

    mu_factor = 0.5            # geometric cooling factor
    mu_min = 1e-6
    maxiter_per_mu = 150       # L‑BFGS‑B iterations for each temperature

        if time.time() > deadline:
            break

        y = y0.copy()

            mu *= mu_factor

    # ------------------------------------------------------------------
    # if nothing succeeded yet, fall back to a uniform candidate
    if best_score == float("inf"):
        best_a = S * np.ones(n) / n
        best_score = evaluate_sequence(best_a.tolist())
        print("[fallback] using uniform distribution")

    # cheap random block search (as in the reference implementation)
    remaining = deadline - time.time()
    if remaining > 2.0:
        block_trials = int(min(2000, remaining * 100))
        print(f"[block] performing {block_trials} random block trials")
        for _ in range(block_trials):
            nb = np.random.randint(2, 7)                     # 2‑6 blocks
            cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = np.random.rand(nb)
            cand = np.zeros(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, S)
            sc = evaluate_sequence(cand.tolist())
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
                print(f"[block improve] new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # final projected‑gradient polishing on the hard max (μ = 0)
    # simple sub‑gradient descent with decreasing step size
    a = best_a.copy()
    eta0 = S * 0.5 / n        # initial step size, roughly the average entry
    decay = 1e-4
    it = 0

        max_c = conv.max()

        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        conv_wr = np.convolve(w, a[::-1])

        eta = eta0 / (1.0 + decay * it)

        a_new = a - eta * g_a
        a_new = _project_to_simplex(a_new, S)

        sc = evaluate_sequence(a_new.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            print(f"[grad refine] new best = {best_score:.6f}")

        a = a_new

        if it > 5000:          # safety cap

    # ------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
