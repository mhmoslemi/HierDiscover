def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1

    # lengths list (choose moderate sizes)
    lengths = [1500, 2000, 2500, 3000, 3500, 4000]

    best_seq = None
    best_score = float('inf')
    best_n = None

    # Parameter grids
    p_vals = np.linspace(0.6, 1.4, 17)  # e.g., 0.6, 0.66, ..., 1.4
    r_vals = np.linspace(0.6, 1.0, 9)   # 0.6, 0.65, ...

    for n in lengths:
        if time.time() >= deadline: break
        # compute arcsine base sequence
        base = arcsine_sequence(n)
        # compute L for FFT once
        L = 1 << ((2*n - 1).bit_length())
        # Precompute exponent vectors for r^i
        exponents = np.arange(n, dtype=np.float64)

        for p in p_vals:
            # compute power transformed sequence
            seq_p = np.power(base, p)
            seq_p_sum = seq_p.sum()
            if seq_p_sum == 0:
                continue
            seq_p = seq_p / seq_p_sum
            # we can compute metric for this p (r=1)
            metric = _metric_fft(seq_p, L) # compute metric
            if metric < best_score:
                best_score = metric
                best_seq = seq_p.tolist()
                best_n = n

            # loop over r
            for r in r_vals:
                seq_pr = seq_p * np.power(r, exponents)
                s = seq_pr.sum()
                if s == 0:
                    continue
                seq_pr = seq_pr / s
                metric = _metric_fft(seq_pr, L)
                if metric < best_score:
                    best_score = metric
                    best_seq = seq_pr.tolist()
                    best_n = n

        print(f"Length {n}: best metric so far {best_score:.6f}")

    # Now SA refinement on the best length
    if best_seq is None:
        # fallback
        best_seq = arcsine_sequence(2000)
        best_n = len(best_seq)

    # SA parameters
    seq = np.array(best_seq, dtype=np.float64)
    n = best_n
    L = 1 << ((2*n - 1).bit_length())
    cur_metric = _metric_fft(seq, L)
    best_metric = cur_metric
    best_seq_arr = seq.copy()

    # SA schedule
    T0 = 0.01 * cur_metric  # initial temperature
    Tmin = 1e-6
    decay = 0.9995
    temperature = T0
    step_frac_initial = 0.3
    step_frac = step_frac_initial
    iter_count = 0

    while time.time() < deadline:
        iter_count += 1

        # random pair
        i, j = np.random.choice(n, 2, replace=False)

        a_i = seq[i]
        a_j = seq[j]
        min_ij = min(a_i, a_j)

"""
Propose a high‑quality step‑function sequence via
(1) a coarse grid search over simple parametric deformations
    of the classic arcsine distribution and (2) a fast
    simulated‑annealing refinement on the simplex.
The metric is evaluated with an FFT‑based convolution to keep
the inner loop cheap and fits comfortably into the 1000 s budget.
"""

import time
import numpy as np

# ---------------------------------------------------------
def arcsine_sequence(n: int) -> list[float]:
    """Return the first ``n`` coefficients of the series of (1‑z)^‑1/2,
    normalised to L1‑norm 1."""
    seq = []
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    # normalise to sum 1
    seq = np.array(seq, dtype=np.float64) / total
    return seq.tolist()


# ---------------------------------------------------------
def _metric_fft(seq_arr: np.ndarray, fft_len: int = None) -> float:
    """Fast evaluation of ``2 * n * max(conv(seq,seq))`` for a sequence
    with L1‑norm 1.  Uses an FFT of length ``fft_len`` (next power of two)."""
    n = seq_arr.shape[0]
    if fft_len is None:
        fft_len = 1 << ((2 * n - 1).bit_length())
    # real FFT of the zero‑padded sequence
    fa = np.fft.rfft(seq_arr, n=fft_len)
    # convolution via pointwise product
    conv = np.fft.irfft(fa * fa, n=fft_len)
    # keep only the linear part
    conv = conv[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b)


# ---------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimises the
    evaluation metric.  The method consists of:

    1. a cheap grid search over two simple deformation parameters
       (a power transformation and an exponential tilt) applied to the
       classic arcsine coefficients for several lengths;
    2. a simulated‑annealing hill‑climb that moves mass between two
       randomly chosen positions while keeping the sequence on the
       probability simplex;  the acceptance probability follows a
       standard annealing schedule.
    The best sequence found before the time budget expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1    # safety margin

    # ---------------------------------------------------------
    # Phase 1 – parametric grid search
    # ---------------------------------------------------------
    lengths = [1500, 2000, 2500, 3000, 3500, 4000]
    p_vals = np.linspace(0.7, 1.3, 13)      # 0.7 … 1.3 in steps of 0.05
    r_vals = np.linspace(0.7, 1.0, 7)       # 0.7 … 1.0

    best_seq = None
    best_score = float("inf")
    best_n = None
    best_fft_len = None

    for n in lengths:
        if time.time() >= deadline:
            break

        print(f"\n=== Grid search for length {n} ===")
        base = np.array(arcsine_sequence(n), dtype=np.float64)
        fft_len = 1 << ((2 * n - 1).bit_length())
        exponents = np.arange(n, dtype=np.float64)

        # baseline (pure arcsine)
        score = _metric_fft(base, fft_len)
        if score < best_score:
            best_score = score
            best_seq = base.copy()
            best_n = n
            best_fft_len = fft_len
            print(f"  baseline arcsine score {score:.6f}")

        # power/exponential deformations
        for p in p_vals:
            # power transform – keep the shape but flatten/steepen it
            pow_seq = np.power(base, p)
            pow_sum = pow_seq.sum()
            if pow_sum == 0.0:
                continue
            pow_seq /= pow_sum

            # evaluate without the exponential tilt
            score = _metric_fft(pow_seq, fft_len)
            if score < best_score:
                best_score = score
                best_seq = pow_seq.copy()
                best_n = n
                best_fft_len = fft_len
                print(f"  new best (p={p:.2f}) score {score:.6f}")

            # now try exponential tilts on top of the power transform
            for r in r_vals:
                tilted = pow_seq * np.power(r, exponents)
                s = tilted.sum()
                if s == 0.0:
                    continue
                tilted /= s
                score = _metric_fft(tilted, fft_len)
                if score < best_score:
                    best_score = score
                    best_seq = tilted.copy()
                    best_n = n
                    best_fft_len = fft_len
                    print(f"  new best (p={p:.2f}, r={r:.2f}) score {score:.6f}")

        print(f"Finished length {n}: current overall best metric = {best_score:.6f}")

    # ---------------------------------------------------------
    # Phase 2 – simulated‑annealing refinement
    # ---------------------------------------------------------
    if best_seq is None:
        # fall back to a safe default – arcsine of length 2000
        best_n = 2000
        best_seq = np.array(arcsine_sequence(best_n), dtype=np.float64)
        best_fft_len = 1 << ((2 * best_n - 1).bit_length())
        best_score = _metric_fft(best_seq, best_fft_len)

    seq = best_seq.copy()
    n = best_n
    fft_len = best_fft_len

    cur_score = _metric_fft(seq, fft_len)
    best_local_seq = seq.copy()
    best_local_score = cur_score

    # annealing parameters
    T0 = 0.005 * cur_score          # initial temperature (roughly 0.5 % of the score)
    Tmin = 1e-7
    decay = 0.9997                  # geometric cooling factor per iteration
    temperature = T0
    step_frac_initial = 0.3         # maximal fraction of the smaller entry that may be moved

    iterations = 0
    print("\n=== Simulated annealing refinement ===")
    while time.time() < deadline:
        iterations += 1

        # pick two distinct indices
        i, j = np.random.choice(n, 2, replace=False)
        ai, aj = seq[i], seq[j]
        if ai == 0.0 and aj == 0.0:
            continue
        # the amount that can be moved without violating non‑negativity
        move_limit = min(ai, aj)
        if move_limit == 0.0:
            # we are allowed to move all of the non‑zero one towards the zero one
            move_limit = max(ai, aj)
            if move_limit == 0.0:
                continue

        # step size slowly shrinks together with the temperature
        step_frac = step_frac_initial * (temperature / T0)
        delta = (np.random.rand() - 0.5) * 2.0 * step_frac * move_limit

        # propose the new sequence (mass transferred from i to j)
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # evaluate
        new_score = _metric_fft(new_seq, fft_len)
        delta_E = new_score - cur_score

        # Metropolis acceptance test
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / max(temperature, 1e-12)):
            seq = new_seq
            cur_score = new_score
            # update the best ever seen
            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local_seq = seq.copy()
                if iterations % 5000 == 0:
                    print(f"[iter {iterations:7d}] new best metric {best_local_score:.6f}")

        # geometric cooling
        temperature = max(Tmin, temperature * decay)

        # occasional re‑normalisation to fight accumulated rounding error
        if iterations % 20000 == 0:
            s = seq.sum()
            if abs(s - 1.0) > 1e-12:
                seq /= s

    # ---------------------------------------------------------
    # Final sanity check and return
    # ---------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, v))) for v in best_local_seq.tolist()]

    # optional: obtain the true evaluation value (for logging)
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"\n--- FINAL RESULT ---\nLength = {len(final_seq)}  metric ≈ {final_score:.6f}")
    except Exception:
        print(f"\n--- FINAL RESULT ---\nLength = {len(final_seq)}  metric ≈ {best_local_score:.6f}")

    return final_seq

            if pow_sum == 0.0:
                continue
            pow_seq /= pow_sum

            # without tilt
            score = _metric_fft(pow_seq, fft_len)
            if score < best_score:
                best_score = score
                best_seq = pow_seq.copy()
                best_n = n
                best_fft_len = fft_len
                print(f"  new best (p={p:.2f}) score {score:.6f}")

            # with tilt
            for r in r_vals:
                tilted = pow_seq * np.power(r, exponents)
                s = tilted.sum()
                if s == 0.0:
                    continue
                tilted /= s
                score = _metric_fft(tilted, fft_len)
                if score < best_score:
                    best_score = score
                    best_seq = tilted.copy()
                    best_n = n
                    best_fft_len = fft_len
                    print(f"  new best (p={p:.2f}, r={r:.2f}) score {score:.6f}")

        print(f"Finished length {n}: current overall best metric = {best_score:.6f}")

    # -------------------- Phase 2 : simulated annealing ----------------
    if best_seq is None:
        # fallback – safe default
        best_n = 2000
        best_seq = np.array(arcsine_sequence(best_n), dtype=np.float64)
        best_fft_len = 1 << ((2 * best_n - 1).bit_length())
        best_score = _metric_fft(best_seq, best_fft_len)

    seq = best_seq.copy()
    n = best_n
    fft_len = best_fft_len

    cur_score = _metric_fft(seq, fft_len)
    best_local_seq = seq.copy()
    best_local_score = cur_score

    # annealing schedule
    T0 = 0.005 * cur_score               # start temperature (~0.5 % of score)
    Tmin = 1e-7
    decay = 0.9997                        # geometric cooling per iteration
    temperature = T0
    step_frac_initial = 0.3               # maximal fraction of the smaller entry moved

    it = 0
    print("\n=== Simulated annealing refinement ===")
    while time.time() < deadline:
        it += 1

        # pick two distinct indices
        i, j = np.random.choice(n, 2, replace=False)
        ai, aj = seq[i], seq[j]

        # determine the amount we may transfer while staying non‑negative
        move_limit = min(ai, aj)
        if move_limit == 0.0:
            # allow moving the whole mass of the non‑zero entry
            move_limit = max(ai, aj)
            if move_limit == 0.0:
                continue

        # step size shrinks together with temperature
        step_frac = step_frac_initial * (temperature / T0)
        delta = (np.random.rand() - 0.5) * 2.0 * step_frac * move_limit

        # propose new sequence (mass transferred from i to j)
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_score = _metric_fft(new_seq, fft_len)
        dE = new_score - cur_score

        # Metropolis acceptance
        if dE < 0.0 or np.random.rand() < np.exp(-dE / max(temperature, 1e-12)):
            seq = new_seq
            cur_score = new_score
            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local_seq = seq.copy()
                if it % 5000 == 0:
                    print(f"[iter {it:7d}] new best metric {best_local_score:.6f}")

        # cool down
        temperature = max(Tmin, temperature * decay)

        # occasional renormalisation (tiny drift from floating‑point ops)
        if it % 20000 == 0:
            s = seq.sum()
            if abs(s - 1.0) > 1e-12:
                seq /= s

    # -------------------- Return the result ---------------------------
    final_seq = [float(max(0.0, min(1000.0, v))) for v in best_local_seq.tolist()]

    # optional sanity check using the provided evaluator (only for logging)
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"\n--- FINAL RESULT ---\nLength = {len(final_seq)}  metric ≈ {final_score:.6f}")
    except Exception:
        print(f"\n--- FINAL RESULT ---\nLength = {len(final_seq)}  metric ≈ {best_local_score:.6f}")

    return final_seq
