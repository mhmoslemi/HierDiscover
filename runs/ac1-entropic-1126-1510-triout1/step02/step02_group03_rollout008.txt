def adam_optimize_seq(n, a0, time_deadline, max_iter=20000):

    v = np.log(np.clip(a0, 1e-12, None))  # safe

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = None
    best_score = np.inf

    m_conv = next_pow2(2*n-1)
    m_corr = next_pow2(3*n-2)

    lambda_val = 1.0
    lambda_max = 100.0
    lambda_growth = 1.001

    for t in range(1, max_iter+1):
        if time.time()>time_deadline:
            break

        # compute a
        v_max = v.max()
        exp_v = np.exp(v - v_max)
        a = exp_v / exp_v.sum()

        # compute conv and score
        A_fft = np.fft.rfft(a, m_conv)
        b = np.fft.irfft(A_fft * A_fft, m_conv)[:2*n-1]
        maxb = b.max()
        score = 2 * n * maxb
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n} iter={t} score={best_score:.6f}")

        # compute w
        exp_arg = lambda_val * (b - maxb)  # <=0
        # to avoid underflow, subtract max (which is 0) already satisfied
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(b, 1.0 / b.size)
        else:
            w /= w_sum

        # gradient via correlation
        W_fft = np.fft.rfft(w, m_corr)
        A_fft_corr = np.fft.rfft(a, m_corr)
        corr = np.fft.irfft(np.conj(W_fft) * A_fft_corr, m_corr)
        start = n - 1
        g = 2.0 * corr[start:start+n]

        # grad_v
        a_dot_g = np.dot(a, g)
        grad_v = a * (g - a_dot_g)

        # Adam update
        m = beta1 * m + (1-beta1) * grad_v
        v2 = beta2 * v2 + (1-beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1**t)
        v2_hat = v2 / (1 - beta2**t)
        # step size: maybe lr_t = 0.2 / np.sqrt(t) (or constant)
        lr = 0.1 / np.sqrt(t)  # decaying

        # occasional jitter

"""
Optimise a non‑negative step‑function (sequence) to minimise the
``evaluate_sequence`` score.

The idea:
1. Parameterise the sequence by a log‑softmax vector (a_i = softmax(v_i)),
   which automatically satisfies a_i ≥ 0 and Σ a_i = 1.
2. Approximate the max‑convolution peak with a smooth soft‑max:
       w_k ∝ exp(λ·(b_k – max b)),   b = a * a.
   As λ → ∞, w becomes a Dirac at the maximal entry, reproducing the
   true sub‑gradient of max b.
3. Compute the gradient
       g = 2·corr(w, a)                               (FFT based)
   and transform it to the log‑softmax space:
       ∇_v = a * (g – (a·g)).
4. Apply Adam (adaptive‑momentum) on v, increasing λ gradually.
5. Run the optimiser for several lengths (4000‑16000) and several
   initialisations (arcsine, uniform, random Dirichlet).  The best
   sequence found before the time budget expires is returned.

All heavy work (convolution, correlation) uses NumPy FFTs; the routine
checks the global time budget (default 1000 s) and aborts early if the
target 1.503 is reached.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a via FFT. Length = 2·len(a)‑1."""
    n = a.shape[0]
    m = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, m)
    c = np.fft.irfft(A * A, m)
    return c[: 2 * n - 1]


# ----------------------------------------------------------------------
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2· Σ_k w_k·a_{k‑i} (cross‑correlation) using FFT.
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    m = _next_pow2(3 * n - 2)          # ≥ (2n‑1)+(n‑1)
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = n - 1
    return 2.0 * corr[start:start + n]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Symmetric or asymmetric Beta(α,β) density on points x_i = (i+½)/n.
    Returns a probability vector (sum = 1).
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(½,½)) on n points, Σ = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _resample_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """
    Linear resampling of a probability vector to a new length.
    The input must sum to 1; the output is re‑normalised to sum to 1.
    """
    old_n = seq.shape[0]
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_len) + 0.5) / new_len
    new_seq = np.interp(new_x, old_x, seq, left=0.0, right=0.0)
    new_seq = np.clip(new_seq, 0.0, None)
    total = new_seq.sum()
    if total == 0.0:
        new_seq = np.full(new_len, 1.0 / new_len)
    else:
        new_seq /= total
    return new_seq


# ----------------------------------------------------------------------
def _adam_optimize(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 20000,
    lr0: float = 0.2,
    lambda_start: float = 1.0,
    lambda_max: float = 80.0,
    lambda_growth: float = 1.001,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max of the self‑convolution.
    Returns the best sequence (sum = 1) and its unsmoothed score.
    """
    # log‑softmax parameters
    eps_init = 1e-12
    v = np.log(np.clip(init_a, eps_init, None))

    m = np.zeros_like(v)       # first moment
    v2 = np.zeros_like(v)      # second moment

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # FFT sizes
    m_conv = _next_pow2(2 * n - 1)
    m_corr = _next_pow2(3 * n - 2)

    best_seq = None
    best_score = np.inf

    lam = lambda_start

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- build probability vector a from v -----
        v_max = v.max()
        exp_v = np.exp(v - v_max)
        a = exp_v / exp_v.sum()

        # ----- current (true) score -----
        conv_full = _conv_full_fft(a)
        maxb = conv_full.max()
        cur_score = 2.0 * n * maxb
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} it={it:5d} score={best_score:.6f}")

        # ----- smooth max weights w -----
        # w_k ∝ exp(λ (b_k – maxb))
        exp_arg = lam * (conv_full - maxb)          # ≤ 0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv_full, 1.0 / conv_full.size)
        else:
            w /= w_sum

        # ----- gradient w.r.t. a (correlation) -----
        g = _correlation_fft(w, a)   # length n

        # ----- transform gradient to log‑softmax space -----
        a_dot_g = np.dot(a, g)
        grad_v = a * (g - a_dot_g)

        # ----- Adam update on v -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        # decaying learning rate
        lr = lr0 / np.sqrt(it)

        v -= lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

        # ----- temperature schedule -----
        lam = min(lambda_max, lam * lambda_growth)

        # ----- occasional jitter to escape tiny plateaus -----
        if it % 2000 == 0 and np.random.rand() < 0.2:
            v += np.random.randn(n) * 0.02

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a low ``evaluate_sequence`` score.
    The algorithm explores several lengths, initialises from arcsine /
    uniform / historic sequences, and refines each with an Adam optimiser
    on a smooth‑max surrogate.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # --------------------------------------------------------------
    # 0️⃣  Load historic construction if present
    historic_seq = None
    try:
        hist = np.array(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        hist = np.clip(hist, 0.0, None)
        if hist.sum() > 0.0:
            hist /= hist.sum()
            historic_seq = hist
            hist_score = 2.0 * len(hist) * _conv_full_fft(hist).max()
            print(f"Historic sequence: len={len(hist)} score≈{hist_score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Choose a set of lengths
    candidate_lengths = [4000, 6000, 8000, 10000, 12000, 16000]
    # make sure we have at least some time for each length
    overall_best_seq = None
    overall_best_score = np.inf

    # --------------------------------------------------------------
    for n in candidate_lengths:
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        # allocate a modest slice per length (≈15 % of remaining, min 15 s)
        slice_time = max(15.0, 0.15 * remaining)

        # ----- build a list of initial sequences for this length -----
        init_candidates = []

        # a) arcsine (theoretically good)
        init_candidates.append(_arcsine_distribution(n))

        # b) uniform
        init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # c) a few symmetric Beta shapes
        for alpha in [0.4, 0.45, 0.55, 0.6]:
            init_candidates.append(_beta_distribution(n, alpha, alpha))

        # d) random Dirichlet draws (soft & spiky)
        for conc in [0.3, 1.0]:
            init_candidates.append(np.random.dirichlet(np.full(n, conc)))

        # e) historic sequence resampled to this length (if available)
        if historic_seq is not None:
            init_candidates.append(_resample_sequence(historic_seq, n))

        # ----------------------------------------------------------
        # 2️⃣  Optimise each candidate (time‑sliced)
        for idx, a0 in enumerate(init_candidates):
            sub_deadline = time.time() + slice_time
            # limit the number of iterations by a generous upper bound
            max_iters = 25000
            refined_seq, refined_score = _adam_optimize(
                n=n,
                init_a=a0,
                deadline=sub_deadline,
                max_iters=max_iters,
                lr0=0.2,
                lambda_start=1.0,
                lambda_max=80.0,
                lambda_growth=1.001,
            )
            if refined_score < overall_best_score:
                overall_best_score = refined_score
                overall_best_seq = refined_seq.copy()
                print(
                    f"*** New global best: len={n} trial={idx} score={overall_best_score:.6f}"
                )
            # early exit if we already beat the target
            if overall_best_score <= 1.503:
                break
        if overall_best_score <= 1.503:
            break

    # --------------------------------------------------------------
    # 3️⃣  Fallback – uniform if nothing was produced
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = 2.0 * n * _conv_full_fft(overall_best_seq).max()
        print("Fallback to uniform sequence.")

    print(
        f"Final best score≈{overall_best_score:.6f}  length={len(overall_best_seq)}"
    )
    # The runner expects a plain Python list
    return overall_best_seq.tolist()
