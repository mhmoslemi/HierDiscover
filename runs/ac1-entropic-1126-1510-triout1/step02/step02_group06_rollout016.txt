def project_simplex(v):
    """Projection of v onto the probability simplex."""
    # Duchi et al. algorithm.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)

def anneal(p, n, deadline, max_iter):
    best_p = p.copy()
    best_score = metric_of(p)
    cur_p = p.copy()
    cur_score = best_score

    it = 0
    T0 = 0.1   # initial temperature
    # we may set temperature schedule based on iteration fraction
    while time.time() < deadline and it < max_iter:
        it += 1
        # temperature schedule: linearly decrease
        T = T0 * (1 - it / max_iter)
        # select random i != j
        i, j = np.random.choice(n, 2, replace=False)
        # decide delta
        max_move = min(cur_p[i], 0.1)  # limit magnitude
        if max_move <= 0:
            continue
        delta = (np.random.rand() * 2 - 1) * max_move   # random within [-max_move, max_move]? but we want move from i to j: delta positive indicates moving from i to j
        # Actually we want to move mass from i to j, so delta = random fraction * cur_p[i]
        delta = np.random.rand() * cur_p[i] * 0.2  # up to 20% of mass at i
        # propose
        new_p = cur_p.copy()
        new_p[i] -= delta
        new_p[j] += delta
        # compute metric
        new_score = metric_of(new_p)
        delta_score = new_score - cur_score
        if delta_score < 0 or np.random.rand() < np.exp(-delta_score / max(T, 1e-9)):
            cur_p = new_p
            cur_score = new_score
            # update best
            if new_score < best_score:
                best_score = new_score
                best_p = new_p.copy()
        # optional: occasional random resampling or jitter
    return best_p, best_score
```

"""
Propose a step‑function sequence with a low  ``evaluate_sequence``  value.

Algorithm
---------

* Several families of analytic probability vectors are sampled:
    – the classical arcsine law,
    – power‑law shapes  p_i ∝ (i+1)^‑α  (α∈[0.3, 1.2]),
    – exponential shapes   p_i ∝ exp(-λ i)  (λ∈[2·10⁻⁴, 1·10⁻²]),
    – a simple triangular distribution.

  Every candidate is normalised to L1‑norm = 1 (a probability vector).

* For each length `n` a short *Linear‑Program* refinement is attempted
  (if SciPy is available).  Starting from the candidate `f`,
  we maximise the total mass of a secondary vector `g` under the constraint
  `f * g ≤ M` where `M = max(f*f)`.  The optimiser is a plain LP
  (``scipy.optimize.linprog``).  After solving we scale `g` to have the same
  L1‑norm as `f` and mix the two vectors:
  `f ← (1‑t)·f + t·g` (default `t = 0.5`).  This step provably does not
  increase `M` and often decreases it.

* After the LP step a *simulated‑annealing* phase is run:
  – a random pair of indices `(i, j)` is chosen,
  – a tiny amount of probability mass is transferred from `i` to `j`,
  – the new metric is evaluated,
  – the move is accepted if it improves the metric, otherwise with a
    temperature‑dependent probability.
  The temperature decays linearly to zero.
  The move always conserves the total mass, so the vector stays on the
  probability simplex automatically.

* The whole search is time‑boxed (default 1000 s).  The available budget
  is split over a few different lengths (≈1500–3000).  The best sequence
  (smallest metric) found during the run is finally returned.

The code uses only NumPy, SciPy (optional) and the standard library,
conforms to the required signature and runs comfortably inside the
time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional import of SciPy – the LP based refinement is a huge
# help but the code works without it.
try:
    from scipy.optimize import linprog
    _SCIPY_OK = True
except Exception:               # pragma: no cover
    _SCIPY_OK = False

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    # Duchi, Shalev‑Shwartz, Singer, Chandra (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – return the uniform point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _metric_of(p: np.ndarray) -> float:
    """Return the evaluation metric for a probability vector `p`."""
    n = p.shape[0]
    conv = np.convolve(p, p)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> np.ndarray:
    """First n coefficients of (1‑z)^‑½, normalised to sum = 1."""
    seq = np.empty(n, dtype=np.float64)
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq[i] = cur
        total += cur
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    seq /= total
    return seq


# ----------------------------------------------------------------------
def powerlaw_sequence(n: int, alpha: float) -> np.ndarray:
    """p_i ∝ (i+1)^‑α, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-alpha)
    seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
def exponential_sequence(n: int, lam: float) -> np.ndarray:
    """p_i ∝ exp(‑λ·i), normalised to sum = 1."""
    idx = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * idx)
    seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
def triangular_sequence(n: int) -> np.ndarray:
    """Symmetric triangle: p_i ∝ min(i+1, n‑i)."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    seq = np.minimum(idx, n + 1 - idx)
    seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray, max_iter: int, time_deadline: float) -> np.ndarray:
    """
    Perform a few LP‑based refinements of the probability vector `f`.
    The vector is assumed to have L1‑norm = sqrt(2·n) (the scaling used
    in the original paper).  The routine returns the improved vector;
    if the LP solver is not available or a step fails the original
    vector is returned unchanged.
    """
    if not _SCIPY_OK:
        return f

    n = f.shape[0]
    target_norm = math.sqrt(2.0 * n)

    # Normalise to the required L1‑norm for the LP formulation
    f = f * (target_norm / f.sum())

    for _ in range(max_iter):
        if time.time() >= time_deadline:
            break

        # current bound M
        conv_f = np.convolve(f, f)
        M = float(conv_f.max())

        # Build the constraint matrix A_ub such that (f * g)_k ≤ M.
        # Row k corresponds to a shift by k.
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        for i in range(n):
            A[i:i + n, i] = f[i]

        # Linear programme: maximise sum(g)  ⇔ minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)
        b = np.full(rows, M, dtype=np.float64)

        res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method="highs")
        if res.status != 0:               # LP failed – give up
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break

        # Scale g back to the same L1‑norm as f
        scale = target_norm / sum_g
        g_scaled = g * scale

        # Mix the two vectors (t=0.5 works well in practice)
        t = 0.5
        f_new = (1.0 - t) * f + t * g_scaled

        # Evaluate new bound
        conv_new = np.convolve(f_new, f_new)
        M_new = float(conv_new.max())

        # Accept the move only if it improves the bound
        if M_new < M:
            f = f_new
        else:
            # no improvement – stop the LP loop
            break

    # Return to probability scale (sum = 1) for the outer optimiser
    return f / f.sum()


# ----------------------------------------------------------------------
def _simulated_annealing(p0: np.ndarray,
                         time_deadline: float,
                         max_iter: int = 200_000) -> np.ndarray:
    """
    Simple annealing on the probability simplex.
    Moves consist of transferring a small amount of mass from a randomly
    chosen index `i` to another index `j`.  The total mass is preserved,
    negativity is never introduced.
    """
    n = p0.shape[0]
    best = p0.copy()
    best_score = _metric_of(best)
    cur = p0.copy()
    cur_score = best_score

    it = 0
    while time.time() < time_deadline and it < max_iter:
        it += 1

        # Linear cooling schedule
        T = max(1e-8, 0.1 * (1.0 - it / max_iter))

        # choose two distinct indices
        i, j = np.random.choice(n, 2, replace=False)

        # move a random fraction (up to 20%) of the mass at i to j
        if cur[i] == 0.0:
            continue
        delta = np.random.rand() * 0.2 * cur[i]

        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        # evaluate new candidate
        new_score = _metric_of(new)
        diff = new_score - cur_score

        if diff < 0 or np.random.rand() < math.exp(-diff / T):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

    return best


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence``.  The method explores several sequence lengths,
    several analytic initial families and refines each candidate by a
    linear‑programming step (when SciPy is available) followed by a
    simulated‑annealing local search.  The best sequence found before the
    time budget expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1    # safety margin

    # lengths that give a reasonable trade‑off between speed and quality
    lengths = [1500, 2000, 2500, 3000]
    best_seq = None
    best_score = float("inf")

    # families of analytic starters (functions that return a probability vector)
    starters = [
        lambda n: arcsine_sequence(n),
        lambda n: triangular_sequence(n),
        lambda n: powerlaw_sequence(n, alpha=0.5),
        lambda n: powerlaw_sequence(n, alpha=0.8),
        lambda n: powerlaw_sequence(n, alpha=1.0),
    ]

    # add a few random power‑law / exponential shapes
    for _ in range(3):
        a = np.random.uniform(0.3, 1.2)
        starters.append(lambda n, a=a: powerlaw_sequence(n, alpha=a))
    for _ in range(3):
        lam = np.random.uniform(2e-4, 1e-2)
        starters.append(lambda n, lam=lam: exponential_sequence(n, lam=lam))

    # Main loop over the chosen lengths
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # --- pick a starter, evaluate and improve it ---------------------------------
        for start_fn in starters:
            if time.time() >= deadline:
                break

            # initial probability vector
            p = start_fn(n)

            # --- optional LP refinement -------------------------------------------------
            # We work with a scaled version (L1 = sqrt(2n)) because the LP is
            # defined for that scaling.
            p_lp = _lp_refine(p, max_iter=3,
                              time_deadline=min(deadline, time.time() + 5.0))
            # after LP we keep the best of the two candidates
            if _metric_of(p_lp) < _metric_of(p):
                p = p_lp

            # --- simulated annealing ----------------------------------------------------
            time_left = deadline - time.time()
            per_restart_budget = max(5.0, time_left / 4.0)
            p_opt = _simulated_annealing(p,
                                          time_deadline=time.time() + per_restart_budget,
                                          max_iter=100_000)

            score = _metric_of(p_opt)

            # keep the overall best
            if score < best_score:
                best_score = score
                best_seq = p_opt.copy()
                print(f"*** New best (len={n}) metric = {best_score:.6f}")

        # clean‑up: guarantee that we have a non‑empty result for this length
        if best_seq is None:
            # fallback – uniform distribution (very unlikely to be used)
            best_seq = np.full(n, 1.0 / n, dtype=np.float64)
            best_score = _metric_of(best_seq)

    # ----------------------------------------------------------------------
    # Final safety: clip to the limits enforced by ``evaluate_sequence``.
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    print(f"\n--- FINAL RESULT ---\nBest length = {len(best_seq)} "
          f"with metric = {best_score:.6f}")
    return best_seq
