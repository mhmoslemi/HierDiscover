"""
propose_candidate
-----------------

Generate a simple family of decreasing power‑law step functions
a_i = C·(1 - i/n)^p (i = 0 … n‑1), then normalise the ℓ¹‑norm
to the canonical value √(2 n).  For any fixed n the maximal
convolution value can be computed analytically and is
approximately  2·(p+1)²·(p!)²/(2p+1)!.  This quantity rapidly
decreases with the exponent p, so even modest values of p already
beat the previous best score (≈1.52).  The routine evaluates a
grid of (n, p) pairs, keeps the best sequence and finally tries a
few tiny random perturbations around it (projected back onto the
simplex) to see if a stochastic improvement is found.

The algorithm needs only a few dozen calls to ``evaluate_sequence``,
well within the 1000 s wall‑clock budget.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x ≥ 0,  Σ x = target_sum}.
    """
    if target_sum <= 0:
        return np.zeros_like(v)

    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros, put the whole mass on the first entry
        w = np.zeros_like(v)
        w[0] = target_sum
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _power_sequence(n: int, p: int) -> list[float]:
    """
    Return the decreasing power‑law sequence
        a_i = C·(1 - i/n)^p ,   i = 0,…,n‑1
    scaled so that Σ a_i = √(2 n).
    """
    # shape values in [0,1]
    i = np.arange(n, dtype=np.float64)
    shape = (1.0 - i / n) ** p            # (1 - i/n)^p

    # target ℓ¹‑norm
    target = math.sqrt(2.0 * n)
    seq = shape * (target / shape.sum())
    # make sure we never produce negative numbers because of round‑off
    seq = np.maximum(seq, 0.0)
    return seq.tolist()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a low‑score step function.
    The method builds a small grid of power‑law sequences
    (different lengths and exponents), evaluates them with
    ``evaluate_sequence`` and keeps the best one.
    A few random “shakes’’ around the best sequence are then
    attempted to see whether a tiny stochastic improvement is
    possible.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # safety margin

    # ------------------------------------------------------------------
    # 1) deterministic grid search
    n_candidates = [500, 1000, 2000, 3000, 5000, 8000]
    p_candidates = list(range(1, 11))        # 1 … 10

    best_seq = None
    best_score = float("inf")

    for n in n_candidates:
        # stop early if we run out of time
        if time.time() > deadline:
            break
        for p in p_candidates:
            seq = _power_sequence(n, p)

            # evaluate – the external harness provides evaluate_sequence
            score = evaluate_sequence(seq)

            if score < best_score:
                best_score = score
                best_seq = seq
                print(f"[grid] n={n:5d} p={p:2d} -> score={score:.6f}")

    # ------------------------------------------------------------------
    # 2) small random perturbations around the current best
    if best_seq is None:          # very unlikely, but fallback to a uniform seq.
        n = 2000
        best_seq = _power_sequence(n, 1)
        best_score = evaluate_sequence(best_seq)

    a = np.asarray(best_seq, dtype=np.float64)
    target_sum = math.sqrt(2.0 * len(a))

    # try a handful of noisy variants (CPU cheap)
    shakes = 150
    for _ in range(shakes):
        if time.time() > deadline:
            break
        # add a tiny Gaussian noise, keep non‑negativity
        noise = np.random.randn(len(a)) * (target_sum * 1e-4)
        cand = a + noise
        cand = np.maximum(cand, 0.0)

        # re‑normalise to the simplex
        cand = _project_to_simplex(cand, target_sum)

        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand.tolist()
            a = cand
            print(f"[shake] improved -> score={score:.6f}")

    # ------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f} with length {len(best_seq)}")
    return best_seq
