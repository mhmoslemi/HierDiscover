def _fft_convolve(x, y):
    size = 1 << ((len(x)+len(y)-2).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X*Y, n=size)
    return conv[:len(x)+len(y)-1]

"""
Propose a step‑function (list of non‑negative heights) that drives
`evaluate_sequence` as low as possible.

Algorithm
---------
1.  Initialise a sequence of a given length on the probability simplex
    (uniform / arcsine / random / previous best).
2.  Warm‑up phase – a smoothed version of the objective is minimised.
    * The max‑convolution `b = a*a` is turned into a soft‑max with a
      temperature τ that shrinks like 1/√t.
    * The gradient of the smoothed objective is
          g = 2 * (w  *  a_rev)   (convolution),
      where w are the soft‑max weights and a_rev = a[::-1].
    * An exponentiated‑gradient (mirror descent) step
          a ← a·exp(−η·g)          (element‑wise)
      is performed and the result is projected back onto the simplex.
3.  Refinement phase – the true non‑smooth objective is tackled.
    * The active index k where the convolution attains its maximum is
      identified.
    * The sub‑gradient is `g_i = 2·a_{k−i}` (zero outside the support).
    * A Polyak step size
          η = (current_score − best_score) / ‖g‖²
      is used (fallback to a diminishing 1/√t step if the denominator
      is tiny or the step would be too large).
    * After the gradient step the vector is again projected onto the
      simplex.
4.  The above two phases are executed for several initialisations and
    several candidate lengths (derived from the previously found best).
    The best sequence seen (according to the official `evaluate_sequence`
    call) is kept and returned.

The code respects the 1000 s budget, prints progress, and never makes
any filesystem or network accesses.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex {x ≥ 0, sum(x)=s}
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]                 # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Generate an arcsine‑type probability vector of length n
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

# Choose an initial distribution
def _initial_sequence(n: int, kind: str) -> np.ndarray:
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif kind == "arcsine":
        a = _arcsine_distribution(n)
    elif kind == "prev":
        try:
            a = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if a.size != n:
                raise ValueError
        except Exception:
            a = _arcsine_distribution(n)
        if a.sum() == 0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / a.sum()
    else:                                   # fallback
        a = _arcsine_distribution(n)

    a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
# FFT‑based convolution for two real vectors
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Convolution of real vectors x and y using FFT (real‑valued)."""
    n = x.shape[0]
    m = y.shape[0]
    size = 1 << ((n + m - 2).bit_length())          # next power of two
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[: n + m - 1]


# ----------------------------------------------------------------------
# Self‑convolution (a * a) – convenience wrapper
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    return _fft_convolve(a, a)

                # temperature τ shrinks like 1/√t but never below min factor
                tau = max(max_b * max(tau_min_factor, 1.0 / np.sqrt(it)), 1e-14)

                # soft‑max weights
                exps = np.exp((b - max_b) / tau)          # max_b gives stability
                w = exps / exps.sum()

                # gradient of the smoothed objective:
                #   g = 2 * (w  *  a_rev)   (convolution)
                a_rev = a[::-1]
                conv_wa = _fft_convolve(w, a_rev)
                g = 2.0 * conv_wa[n - 1 : n - 1 + n]      # slice to length n

                # exponentiated‑gradient update
                eta = eta0 / np.sqrt(it)
                a = a * np.exp(-eta * g)                 # element‑wise
                a = _project_onto_simplex(a, s=1.0)

                # evaluate current score (cheap estimate)
                b = _fft_self_convolve(a)
                cur_score = 2.0 * n * b.max()
                if cur_score + 1e-9 < best_local_score:
                    best_local_score = cur_score
                    best_local_seq = a.copy()
                    print(
                        f"[{time.time():.1f}] len={n} init={init_kind} "
                        f"warm‑iter={it} score={best_local_score:.6f}"
                    )

                # if we have already beaten the global target we may exit early
                if best_local_score <= 1.503:
                    break

            # ------------------------------------------------------------------
            # 2) Refinement with true max‑convolution (sub‑gradient + Polyak step)
            # ------------------------------------------------------------------
            beta = 0.9          # momentum for a *very* short window (optional)
            velocity = np.zeros_like(a)
            it_refine = 0
            while time.time() < slice_deadline:
                it_refine += 1
                # current convolution & max index
                b = _fft_self_convolve(a)
                max_b = b.max()
                cur_score = 2.0 * n * max_b
                k_max = int(np.argmax(b))

                # sub‑gradient for the active constraint
                idx = k_max - np.arange(n)
                mask = (idx >= 0) & (idx < n)
                g = np.zeros_like(a)
                g[mask] = 2.0 * a[idx[mask]]

                # momentum (helps escape shallow plateaux)
                velocity = beta * velocity + (1.0 - beta) * g

                # Polyak step size: (current − best) / ||g||²
                norm2 = np.dot(velocity, velocity) + 1e-15
                step = (cur_score - best_local_score) / norm2
                # safeguard: fall back to a small diminishing step if needed
                if step <= 0.0 or step > 1.0:
                    step = 0.5 / np.sqrt(it_refine)

                # gradient step + projection onto the simplex
                a = a - step * velocity
                a = _project_onto_simplex(a, s=1.0)

                # re‑evaluate
                b = _fft_self_convolve(a)
                cur_score = 2.0 * n * b.max()
                if cur_score + 1e-9 < best_local_score:
                    best_local_score = cur_score
                    best_local_seq = a.copy()
                    print(
                        f"[{time.time():.1f}] len={n} init={init_kind} "
                        f"refine‑iter={it_refine} score={best_local_score:.6f}"
                    )
                    # update global best as soon as we improve locally
                    if best_local_score < overall_best_score:
                        overall_best_score = best_local_score
                        overall_best_seq = best_local_seq.copy()
                        print(
                            f"*** New global best → len={n} init={init_kind} "
                            f"score={overall_best_score:.6f}"
                        )
                    if overall_best_score <= 1.503:
                        break

                # occasional tiny random jitter to get off plateaux
                if it_refine % 700 == 0 and np.random.rand() < 0.05:
                    a += np.random.randn(n) * 1e-4
                    a = _project_onto_simplex(a, s=1.0)

                # early exit when target met
                if overall_best_score <= 1.503:
                    break

            # ------------------------------------------------------------------
            # Final verification with the official black‑box evaluator
            # ------------------------------------------------------------------
            try:
                final_score = evaluate_sequence(best_local_seq.tolist())
            except Exception:
                final_score = best_local_score

            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = best_local_seq.copy()
                print(
                    f"*** Updated global best after black‑box eval → len={n} "
                    f"init={init_kind} score={overall_best_score:.6f}"
                )

            if overall_best_score <= 1.503:
                break

        if overall_best_score <= 1.503 or time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Fallback – should never be needed but guarantees a return value.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())

    print(
        f"Final best score: {overall_best_score:.6f} "
        f"(length = {len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
