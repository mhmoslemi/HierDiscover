"""
Multi‑strategy search for a low‑self‑convolution step function.

The evaluation metric is scale‑invariant, so all sequences are normalised to
sum = 1.  The search proceeds as follows

1.  Initialise a pool of candidates:
    – the classic “arcsine’’ sequence,
    – a family of Beta‑distribution shapes a_i ∝ x_i^{α‑1}(1‑x_i)^{α‑1},
    – a previously found global sequence (if the variable ``height_sequence_1''
      exists).

2.  Repeatedly improve the best candidate using three complementary moves
    (chosen randomly each iteration):
        * sub‑gradient descent on the true max‑convolution,
        * random Gaussian perturbations followed by projection onto the simplex,
        * an optional linear‑programming refinement that maximises the L₁‑norm
          under the linear constraints (f∗g) ≤ M (solved with SciPy’s
          ``linprog`` when available).

   The sub‑gradient of max(conv) is computed exactly by taking a uniform
   weight on all indices attaining the maximum.  After each move the new
   sequence is projected onto the probability simplex and evaluated; any
   improvement is kept.

3.  Simple exponential decay of the step‑size / perturbation scale and a
    progressive reduction of the probability of taking an LP step keep the
    search focused as time runs out.

The function returns the best sequence found before the time budget expires
(≈ 1000 s).  All numbers are non‑negative and the sequence is normalised to
unit L₁‑norm, satisfying the requirements of ``evaluate_sequence``.

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> list[float]:
    """Return a normalised arcsine distribution (sum = 1) of length n."""
    seq = []
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2.0 * i + 1.0) / (2.0 * (i + 1))
    seq = np.array(seq, dtype=float)
    seq = seq / seq.sum()
    return seq.tolist()


# ----------------------------------------------------------------------
def beta_sequence(alpha: float, n: int) -> list[float]:
    """
    Discrete Beta(α,α) density on n points.  Handles α<1 by using logarithms.
    The points are taken at the centre of each bin: x_i = (i+0.5)/n.
    """
    i = np.arange(n, dtype=float) + 0.5
    x = i / n
    eps = np.finfo(float).eps
    x = np.clip(x, eps, 1.0 - eps)
    # log‑form to avoid overflow when α is far from 1
    log_seq = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    seq = np.exp(log_seq)
    seq = seq / seq.sum()
    return seq.tolist()


# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("target sum s must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def subgradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    One sub‑gradient descent step that reduces the maximal entry of
    the self‑convolution.  The sub‑gradient is computed on the exact set of
    maximal indices.
    """
    conv = np.convolve(a, a)
    max_val = conv.max()
    # indices attaining the maximum (tolerance guards against floating point noise)
    tol = 1e-12
    mask = conv >= max_val - tol
    w = mask.astype(float)
    w_sum = w.sum()
    if w_sum == 0:
        # fallback to a tiny soft‑max (should never happen)
        beta = 5.0
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / exp_vals.sum()
    else:
        w = w / w_sum
    # gradient = 2 * (w * a_rev) convolution, sliced to length n
    n = a.shape[0]
    grad = 2.0 * np.convolve(w, a[::-1])[n - 1:2 * n - 1]
    cand = a - step * grad
    cand = project_simplex(cand, s=1.0)
    return cand


# ----------------------------------------------------------------------
def random_perturbation(a: np.ndarray, sigma: float) -> np.ndarray:
    """
    Add i.i.d. Gaussian noise (scale sigma) to all coordinates, clip to
    non‑negative and project back onto the simplex.
    """
    noise = np.random.randn(a.shape[0]) * sigma
    cand = a + noise
    cand = np.maximum(cand, 0.0)
    cand = project_simplex(cand, s=1.0)
    return cand


# ----------------------------------------------------------------------
def solve_lp_max_sum(a: np.ndarray, M: float, max_time: float = 5.0):
    """
    Solve the linear program:
        maximise    Σ g_i
        subject to  g_i ≥ 0  ,   (a ∗ g)[k] ≤ M  ∀k
    Returns a normalised feasible vector (sum = 1) or None on failure.
    """
    try:
        from scipy.optimize import linprog
    except Exception:
        return None

    n = a.shape[0]
    # Build convolution matrix A (shape (2n‑1) × n) so that A @ g = a ∗ g
    A = np.zeros((2 * n - 1, n), dtype=float)
    for j in range(n):
        A[j:j + n, j] = a  # column j contains a shifted copy of a

    c = -np.ones(n)                     # minimise –Σ g_i  → maximise Σ g_i
    bounds = [(0.0, None)] * n
    b_ub = M * np.ones(2 * n - 1)

    # Solver options: use HiGHS (fast) and enforce a modest time limit
    opts = {"presolve": True, "time_limit": max_time}
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options=opts)

    if not res.success or res.x is None:
        return None
    g = res.x
    s = g.sum()
    if s <= 0:
        return None
    g_norm = g / s
    return g_norm

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` under the given time budget (seconds).
    Returns the best list found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_score = float('inf')
    best_n = None

    # ------------------------------------------------------------
    # 1) Initialise from a variety of shapes
    candidate_lengths = [1200, 1500, 1800, 2000, 2300]
    alphas = [0.30, 0.40, 0.45, 0.50, 0.55, 0.60, 0.80, 1.0, 1.5, 2.0]

    # arcsine (β=0.5) is already covered in alphas but we keep explicit entry
    for n in candidate_lengths:
        if time.time() >= deadline:
            break
        seq = np.array(arcsine_sequence(n), dtype=float)
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            best_n = n
        # try a few Beta shapes for the same length
        for alpha in alphas:
            seq_beta = np.array(beta_sequence(alpha, n), dtype=float)
            score_beta = evaluate_sequence(seq_beta.tolist())
            if score_beta < best_score:
                best_score = score_beta
                best_seq = seq_beta.copy()
                best_n = n

    # ------------------------------------------------------------
    # 2) Global sequence (if supplied) – use only a manageable length
    try:
        global height_sequence_1
        if isinstance(height_sequence_1, (list, tuple, np.ndarray)):
            seq_glob = np.asarray(height_sequence_1, dtype=float)
            if seq_glob.size > 0:
                seq_glob = np.maximum(seq_glob, 0.0)
                if seq_glob.sum() > 0:
                    seq_glob = seq_glob / seq_glob.sum()
                n_glob = seq_glob.shape[0]
                # Trim very long globals to a reasonable size (≤2500)
                if n_glob > 2500:
                    # simple down‑sampling by averaging blocks
                    block = n_glob // 2500
                    seq_glob = seq_glob[: block * 2500].reshape(-1, block).mean(axis=1)
                    n_glob = seq_glob.shape[0]
                    seq_glob = seq_glob / seq_glob.sum()
                score_glob = evaluate_sequence(seq_glob.tolist())
                if score_glob < best_score:
                    best_score = score_glob
                    best_seq = seq_glob.copy()
                    best_n = n_glob
                    print(f"[global] n={n_glob:<4d}  score={score_glob:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------
    # Fallback (should never happen)
    if best_seq is None:
        best_n = 2000
        best_seq = np.array(arcsine_sequence(best_n), dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] n={best_n}  score={best_score:.6f}")

    # ------------------------------------------------------------
    # Helper to evaluate candidate safely
    def safe_score(vec):
        sc = evaluate_sequence(vec.tolist())
        if math.isfinite(sc):
            return sc
        return float('inf')

    # ------------------------------------------------------------
    # Parameters for the iterative improvement
    step_size = 0.25          # initial sub‑gradient step
    step_decay = 0.9995
    sigma = 0.02              # initial Gaussian perturbation scale
    sigma_decay = 0.999
    prob_lp = 0.12            # probability of attempting an LP step each iteration
    prob_grad = 0.66          # probability for a gradient step (rest → random)
    min_lp_interval = 5.0     # seconds to wait between two LP attempts
    last_lp_time = -float('inf')
    have_lp = True            # will be set to False if linprog is unavailable

    # ------------------------------------------------------------
    # Main optimisation loop
    while time.time() < deadline:
        now = time.time()
        n = best_seq.shape[0]

        # ----- decide which move to perform ---------------------------------
        r = np.random.rand()
        performed_lp = False

        # 1) Linear‑programming refinement (occasionally)
        if have_lp and r < prob_lp and (now - last_lp_time) > min_lp_interval:
            # recompute the current maximal self‑convolution value M
            M = np.convolve(best_seq, best_seq).max()
            g_norm = solve_lp_max_sum(best_seq, M, max_time=4.0)
            last_lp_time = now
            if g_norm is not None:
                cand_score = safe_score(g_norm)
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = g_norm.copy()
                    best_n = n
                    print(f"[lp] new best {cand_score:.6f} (n={n})")
                performed_lp = True

        if performed_lp:
            continue   # skip other moves this iteration

        # 2) Sub‑gradient descent step
        if r < prob_lp + prob_grad:
            cand = subgradient_step(best_seq, step_size)
            cand_score = safe_score(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                best_n = n
                print(f"[grad] new best {cand_score:.6f} (n={n})")
        else:
            # 3) Random perturbation (hill‑climbing)
            cand = random_perturbation(best_seq, sigma)
            cand_score = safe_score(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                best_n = n
                print(f"[rand] new best {cand_score:.6f} (n={n})")

        # ----- decay schedules ------------------------------------------------
        step_size = max(1e-5, step_size * step_decay)
        sigma = max(1e-5, sigma * sigma_decay)

        # occasional check for the time budget and break early if needed
        if time.time() + 0.5 >= deadline:
            break

    # ------------------------------------------------------------
    # Final sanity check – enforce non‑negativity and unit L₁‑norm
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        # fallback to uniform if something went wrong
        best_seq = np.full(best_seq.shape, 1.0 / best_seq.shape[0])

    print(f"Best sequence length = {len(best_seq)}  score = {best_score:.6f}")
    return best_seq.tolist()
