    # Helper compute score
    def _score(a):
        conv = _self_convolution_fft(a, size_fft)
        return 2.0 * n * conv.max() / (a.sum() ** 2)

    best_a = init_seq.copy()
    best_score = _score(best_a)
    print(f"[init] n={n}, init score={best_score:.6f}")

    # parameters
    alpha0 = target_sum * 0.2 / n  # step base
    decay = 5e-5
    momentum = 0.9
    vel = np.zeros_like(best_a)

    no_improve = 0

    deadline = time.time() + budget_s - 0.5

    while time.time() < deadline and it < 3_000_000:
        # compute conv
        conv = _self_convolution_fft(best_a, size_fft)
        max_b = conv.max()
        k_max = conv.argmax()
        # compute gradient of objective O
        S = best_a.sum()

        grad = factor * (grad_c - (2.0 * max_b / S) * best_a)
        # step size
        step = alpha0 / (1.0 + decay * it)
        # momentum update
        vel = momentum * vel - step * grad
        a_new = best_a + vel

        # evaluate new score
        cur_score = _score(a_new)

        # update best record (the best across whole run)

            best_a = a_new.copy()

"""
Propose a step‑function (non‑negative sequence) with a low evaluation score.

Algorithm
---------
1. **Initialisation** – Determine the problem size `n`.  If a previously
   discovered sequence `height_sequence_1` exists it is used, otherwise a
   uniform start is taken.  All candidates are projected onto the simplex
   `{a ≥ 0, Σa = √(2 n)}`.

2. **Exploratory shape search** – A few hundred cheap parametric families
   (inverse‑sqrt, power‑law, exponential, cosine, mixtures and random blocks)
   are sampled, projected and evaluated.  The best among them becomes the
   starting point for the refinement phase.

3. **Refinement – sub‑gradient descent on the true objective**  
   The objective is  
        `F(a) = 2·n·max_k (a*a)_k / (Σa)²`.  
   At each iteration the convolution `c = a*a` is computed with an FFT,
   the index `k*` where `c` attains its maximum is identified, and the
   sub‑gradient of `F` at `a` is obtained analytically:
        `∂F/∂a_i = (2·n)/(S²)·( 2·a_{k*-i} – 2·c_{k*}/S·a_i )`,
   where `S = Σa`.  A Nesterov‑type momentum update with a decaying step
   size is performed, followed by a Euclidean projection onto the simplex.
   Periodic random perturbations and occasional “random‑block” restarts
   help escape plateaus.

4. **Final random‑block polishing** – After the time budget is almost
   exhausted a handful of fresh random block candidates are evaluated;
   any improvement is kept.

The routine respects the security checks of `evaluate_sequence`,
returns the best sequence found before the 1000 s budget expires and prints
progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (the simplex)."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _self_convolution_fft(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution a * a using FFT.  Returns the full 2·n‑1 vector."""
    fu = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fu * fu, n=size_fft)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
def _score(a: np.ndarray, n: int) -> float:
    """Exact evaluation used internally (no security checks)."""
    conv = _self_convolution_fft(a, 1 << ((2 * n - 1).bit_length()))
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, conv: np.ndarray, k_max: int,
                 n: int, S: float) -> np.ndarray:
    """
    Sub‑gradient of F(a) = 2·n·max(conv)/(S²) w.r.t. a.
    Parameters
    ----------
    a      : current vector (length n)
    conv   : full self‑convolution a*a (length 2·n‑1)
    k_max  : index where conv attains its maximum
    n      : length of a
    S      : sum of a (positive)
    Returns
    -------
    grad   : ndarray of shape (n,)
    """
    # gradient of the max‑convolution term
    grad_c = np.zeros_like(a)
    i = np.arange(n)
    j = k_max - i
    mask = (j >= 0) & (j < n)
    grad_c[mask] = 2.0 * a[j[mask]]          # 2·a_{k*-i}
    factor = (2.0 * n) / (S * S)
    # account for the denominator (S²) via chain rule
    grad = factor * (grad_c - (2.0 * conv[k_max] / S) * a)
    return grad
