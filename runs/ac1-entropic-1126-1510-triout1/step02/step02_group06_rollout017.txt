def metric(seq):
    a = np.asarray(seq, dtype=np.float64)
    # ensure nonnegative and sum>0
    n = a.shape[0]
    m = 1 << (2*n -1).bit_length() # next power of 2 length
    fft_a = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fft_a * fft_a, n=m)[:2*n-1]
    max_b = conv.max()
    sum_a = a.sum()
    return 2.0 * n * max_b / (sum_a**2)

import time, math, numpy as np, random

def metric(seq):
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    # compute convolution via FFT
    m = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa*fa, n=m)[:2*n-1]
    maxb = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float('inf')
    return 2.0 * n * maxb / (sum_a**2)

def grad_max(seq):
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    m = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa*fa, n=m)[:2*n-1]
    k = int(np.argmax(conv))
    # gradient w.r.t a
    idx = np.arange(n)
    partner = k - idx
    mask = (partner >= 0) & (partner < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[partner[mask]]
    return grad, k, conv.max()

def mirror_descent_step_exp(a, m_t, v_t, t, beta1=0.9, beta2=0.999, eps=1e-8):
    # compute gradient subgradient
    grad, _, _ = grad_max(a)
    # update moments
    m_t = beta1 * m_t + (1-beta1) * grad
    v_t = beta2 * v_t + (1-beta2) * (grad**2)
    # bias correction
    m_hat = m_t / (1 - beta1**t)
    v_hat = v_t / (1 - beta2**t)
    # compute step size factor (like Adam)
    step = m_hat / (np.sqrt(v_hat) + eps)
    # treat step as 'learning rate' times direction gradient; we use multiplicative update:
    # a_i <- a_i * exp(-eta * step_i)
    eta = 0.5  # base step size
    factor = np.exp(-eta * step)
    a_new = a * factor
    s = a_new.sum()
    if s > 0:
        a_new = a_new / s
    else:
        a_new = np.full_like(a, 1.0 / len(a))
    return a_new, m_t, v_t

def refine_sequence(seq_init, deadline):
    a = np.array(seq_init, dtype=np.float64)
    n = len(a)
    best_a = a.copy()
    best_score = metric(a)
    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.5  # step size factor
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _, _ = grad_max(a)
        # update moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = m_hat / (np.sqrt(v_hat) + eps)
        # multiplicative update
        factor = np.exp(-base_lr * step)
        a_new = a * factor
        total = a_new.sum()
        if total > 0:
            a_new = a_new / total
        else:
            a_new = np.full_like(a, 1.0 / n)
        cur_score = metric(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        # acceptance: if improvement, assign a = a_new else keep old but maybe random jitter
        if cur_score <= metric(a):
            a = a_new
        else:
            # random jitter occasionally
            if t % 300 == 0:
                i, j = np.random.randint(0, n, size=2)
                delta = (np.random.rand() - 0.5) * 2e-3
                a_jitter = a.copy()
                a_jitter[i] = max(0.0, a_jitter[i] + delta)
                a_jitter[j] = max(0.0, a_jitter[j] - delta)
                a_jitter = a_jitter / a_jitter.sum()
                if metric(a_jitter) < cur_score:
                    a = a_jitter
        # early stop if step sizes extremely small?
        if base_lr * np.max(np.abs(step)) < 1e-12:
            break
    return best_a, best_score

"""
Propose a high‑quality step‑function sequence.

The method combines many diverse initialisations (arcsine, power‑law
families and random Dirichlet draws) with a fast mirror‑descent /
Adam‑style optimisation on the simplex.  Convolutions are evaluated by
FFT (O(n log n)) so thousands of iterations fit easily in the 1000 s
budget.  The best sequence found over all restarts is returned.
"""

import time
import math
import random
import numpy as np

# -------------------------------------------------------------
# Helper to compute the evaluation metric (no clipping – it is safe
# because all generated values stay well below 1000).
def _metric(a: np.ndarray) -> float:
    """2·n·max(a∗a) / (sum(a)²) for a non‑negative array a."""
    n = a.shape[0]
    # length of FFT – next power of two ≥ 2·n‑1 for speed and exactness
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


# -------------------------------------------------------------
def _conv_and_max(a: np.ndarray):
    """Return the full linear convolution a∗a and its maximal value."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv, conv.max()


# -------------------------------------------------------------
def _subgradient(a: np.ndarray):
    """
    Sub‑gradient of the peak of a∗a.
    Returns (grad, k_max) where k_max is the index attaining the maximum.
    """
    n = a.shape[0]
    conv, _ = _conv_and_max(a)
    k = int(conv.argmax())
    idx = np.arange(n)
    partner = k - idx
    mask = (partner >= 0) & (partner < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[partner[mask]]
    return grad, k


# -------------------------------------------------------------
def _mirrored_adam_step(a: np.ndarray, m: np.ndarray, v: np.ndarray,
                        t: int, base_lr: float = 0.5,
                        beta1: float = 0.9, beta2: float = 0.999,
                        eps: float = 1e-12):
    """One Adam‑style mirror‑descent step on the probability simplex."""
    grad, _ = _subgradient(a)
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = m_hat / (np.sqrt(v_hat) + eps)          # Adam direction
    upd = np.exp(-base_lr * step)                  # multiplicative move
    cand = a * upd
    cand_sum = cand.sum()
    if cand_sum > 0.0:
        cand /= cand_sum
    else:                                          # safety fallback
        cand = np.full_like(a, 1.0 / a.shape[0])
    return cand, m, v


# -------------------------------------------------------------
def _refine_sequence(start_seq: np.ndarray, deadline: float):
    """
    Run the Adam‑style mirror descent until ``deadline``.
    Returns the best sequence (as a list) and its metric.
    """
    a = start_seq.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _metric(a)

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    t = 0
    base_lr = 0.5
    no_improve = 0

    while time.time() < deadline:
        t += 1
        cand, m, v = _mirrored_adam_step(a, m, v, t, base_lr)

        cand_score = _metric(cand)
        cur_score = _metric(a)

        # keep the overall best
        if cand_score < best_score - 1e-14:
            best_score = cand_score
            best_a = cand.copy()
            no_improve = 0
            if t % 1000 == 0:
                print(f"[iter {t:6d}] new best metric {best_score:.6f}")

        # accept improvement, otherwise keep the old point
        if cand_score <= cur_score:
            a = cand
            base_lr = min(base_lr * 1.08, 5.0)   # gently increase step size
            no_improve = 0
        else:
            base_lr *= 0.65                      # shrink step size on failure
            no_improve += 1

            # occasional random jitter to escape flat plateaux
            if no_improve % 250 == 0:
                i, j = np.random.choice(n, 2, replace=False)
                delta = (np.random.rand() - 0.5) * 2e-3     # ±0.001
                jitter = a.copy()
                jitter[i] = max(0.0, jitter[i] + delta)
                jitter[j] = max(0.0, jitter[j] - delta)
                jitter /= jitter.sum()
                jitter_score = _metric(jitter)
                if jitter_score < cur_score:
                    a = jitter
                    base_lr = max(base_lr, 0.1)

        # safety break when the update becomes astronomically tiny
        if base_lr < 1e-9:
            break

    return best_a.tolist(), float(best_score)

def arcsine_sequence(n: int):
    """Discrete (1‑z)^‑1/2 coefficients, L1‑normalised to 1."""
    seq = []
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    seq = np.array(seq, dtype=np.float64)
    seq /= seq.sum()
    return seq.tolist()


# -------------------------------------------------------------
def power_law_sequence(n: int, alpha: float):
    """
    Symmetric power‑law shape:
        a_i ∝ (i+1)^(-α) + (n-i)^(-α) .
    The result is normalised to L1 = 1.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = (i ** (-alpha) + (n - i + 1) ** (-alpha))
    seq /= seq.sum()
    return seq.tolist()

def propose_candidate(seed: int = 42, budget_s: int = 1000):
    """
    Search for the smallest possible evaluation score within the time budget.
    Several families of starting sequences are tried; each is refined by a
    mirror‑descent / Adam optimiser.  The best sequence found is returned.
    """
    random.seed(seed)
    np.random.seed(seed)

    deadline = time.time() + budget_s - 0.1   # small safety margin

    # Choose a single length – larger values give a finer discretisation.
    n = 4000
    print(f"\n=== Optimisation started (length = {n}) ===")

    # -----------------------------------------------------------------
    # Build a pool of diverse initial sequences.
    initial_pool = []

    # 1) classic arcsine
    initial_pool.append(arcsine_sequence(n))

    # 2) a few power‑law families (α around the arcsine exponent 0.5)
    for a in [0.30, 0.35, 0.40, 0.45, 0.48, 0.50, 0.52, 0.55]:
        initial_pool.append(power_law_sequence(n, a))

    # 3) several random Dirichlet draws (sparseness encourages low peaks)
    for _ in range(6):
        # concentration < 1 gives a sparse distribution
        dirichlet_vec = np.random.dirichlet(np.full(n, 0.4)).tolist()
        initial_pool.append(dirichlet_vec)

    # -----------------------------------------------------------------
    # Allocate time roughly equally to each start point.
    per_start = max(5.0, (deadline - time.time()) / max(1, len(initial_pool)))
    best_seq = None
    best_score = float("inf")

    for idx, init in enumerate(initial_pool):
        if time.time() >= deadline:
            break
        sub_deadline = time.time() + per_start
        print(f"\n--- Restart {idx + 1}/{len(initial_pool)} (time slice {per_start:.1f}s) ---")
        seq_opt, score_opt = _refine_sequence(np.asarray(init, dtype=np.float64), sub_deadline)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt
            print(f"*** New overall best score {best_score:.6f}")

    # -----------------------------------------------------------------
    # Final safety: enforce the limits used inside evaluate_sequence.
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    print(f"\n--- FINAL RESULT ---\nBest length = {len(best_seq)}  metric = {best_score:.6f}\n")
    return best_seq
