"""
Propose a step‑function (probability) sequence that minimizes the
evaluation metric 2·n·max(conv)/(∑a)².

Algorithm
---------

1. **Initial candidates** – generate several simple families (arcsine,
   power‑law, geometric, random Dirichlet) for a moderate length
   (≈2000).  Keep the best of them.

2. **Projected sub‑gradient descent** – the objective is
   f(a)=max_k (a*a)_k (the L∞‑norm of the self‑convolution).  For a
   fixed a the sub‑gradient of f is
        g = 2·conv(w, a[::-1])[n‑1:2·n‑1],
   where w is the uniform weight on the indices attaining the maximal
   convolution value.  We move a in the opposite direction,
        a ← a – η·g,
   and re‑project onto the probability simplex (non‑negative, sum = 1).
   The step size η starts relatively large (≈0.05) and decays slowly.

3. **Occasional LP refinement** – if `cvxpy` with the SCS solver is
   available we occasionally (≈ every 30 seconds) solve the linear program

        maximise    1ᵀ g
        subject to  g ≥ 0
                    f ∗ g ≤ M ,

   where M = max(f*f).  The solution g is rescaled to unit L₁‑norm and
   mixed with the current f:
        f ← (1‑τ)·f + τ·g
   with a mixing weight τ that slowly shrinks.

4. **Random restarts** – from time to time a small random perturbation
   (Dirichlet‑draw) is examined; if it improves the score it becomes the
   new current iterate.

5. **Book‑keeping** – the true score is obtained via `evaluate_sequence`
   after each update; the best sequence seen so far is returned at the
   end of the allotted 1000 s budget.

All operations respect the time limit, use only NumPy / cvxpy (SCS) and
contain no I/O.  The returned list contains non‑negative numbers that
sum to 1 (the metric is scale‑invariant).  
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> np.ndarray:
    """Discrete arcsine coefficients C_i = binom(2i,i)/4^i, L1‑normalised."""
    seq = np.empty(n, dtype=float)
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq[i] = cur
        total += cur
        if i + 1 < n:
            cur = cur * (2.0 * i + 1.0) / (2.0 * (i + 1))
    seq /= total
    return seq


# ----------------------------------------------------------------------
def power_law_sequence(n: int, exponent: float = -0.5) -> np.ndarray:
    """a_i ∝ (i+1)^exponent, then normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=float)
    seq = idx ** exponent
    seq = np.maximum(seq, 0.0)
    seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
def geometric_sequence(n: int, c: float = 1.255) -> np.ndarray:
    """
    Truncated geometric distribution a_i = (1‑r) r^i with r = 1‑c/n,
    then normalised to sum = 1.  The parameter c≈1.255 minimises the
    asymptotic evaluation constant for this family.
    """
    eps = c / n
    r = 1.0 - eps
    i = np.arange(n, dtype=float)
    seq = (1.0 - r) * (r ** i)
    # Renormalise because the tail is cut off at n
    seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
def random_dirichlet_sequence(n: int, alpha: float = 0.5) -> np.ndarray:
    """Sample a random probability vector from Dirichlet(alpha)."""
    g = np.random.gamma(alpha, 1.0, size=n)
    seq = g / g.sum()
    return seq


# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σ x = s}.
    Duchi‑et‑al. algorithm (2008).
    """
    if s < 0:
        raise ValueError("target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of   f(a)=max_k (a*a)_k   .
    The gradient is 2·conv(w, a[::-1])[n‑1:2n‑1] where w is the uniform
    weight on the arg‑max positions of the convolution.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_val = conv.max()
    # use a small tolerance to collect all maximal positions
    tol = max_val * 1e-12
    max_idx = np.where(conv >= max_val - tol)[0]
    w = np.zeros_like(conv, dtype=float)
    if max_idx.size > 0:
        w[max_idx] = 1.0 / max_idx.size
    grad = 2.0 * np.convolve(w, a[::-1])[n - 1:2 * n - 1]
    return grad


# ----------------------------------------------------------------------
def solve_lp_refinement(a: np.ndarray, M: float):
    """
    Solve  maximise Σ g   s.t.   g ≥ 0,   a ∗ g ≤ M .
    Returns the normalised optimiser g (L1‑norm = 1) or None on failure.
    """
    n = a.shape[0]
    try:
        import cvxpy as cp
    except Exception:
        return None

    # Build the convolution matrix A (shape (2n‑1, n))
    A = np.zeros((2 * n - 1, n), dtype=float)
    for i in range(n):
        A[i:i + n, i] = a[i]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M * np.ones(2 * n - 1)]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        # SCS is a first‑order conic solver that is usually available.
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=2000,
                   eps=1e-6, verbose=False)
    except Exception:
        return None

    if g.value is None:
        return None
    g_opt = np.asarray(g.value, dtype=float)
    total = g_opt.sum()
    if total <= 0:
        return None
    # Normalise to a probability vector
    return g_opt / total


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000):
    """
    Search for a non‑negative step‑function sequence with the lowest
    evaluation score within the given time budget (seconds).  Returns a
    list of coefficients (they will sum to 1).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin
    best_seq = None
    best_score = float('inf')
    n = 2200                     # moderate length, good trade‑off

    # ------------------------------------------------------------------
    # 1) generate a handful of diverse starting points
    starters = []

    # arcsine (the classical baseline)
    starters.append(arcsine_sequence(n))

    # power‑law variants
    for exp in (-0.45, -0.55):
        starters.append(power_law_sequence(n, exponent=exp))

    # geometric family (parameter tuned to the asymptotic optimum)
    starters.append(geometric_sequence(n, c=1.255))

    # random Dirichlet draws
    for _ in range(5):
        starters.append(random_dirichlet_sequence(n, alpha=0.5))

    # evaluate all starters, keep the best
    for seq in starters:
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[init] score={score:.6f}")

    # ------------------------------------------------------------------
    # Helper to compute current score (avoid repeated imports)
    def cur_score(a):
        return evaluate_sequence(a.tolist())

    # ------------------------------------------------------------------
    # Simple momentumless projected sub‑gradient descent parameters
    step = 0.05                # initial step size
    step_decay = 0.9995        # exponential decay
    momentum = np.zeros_like(best_seq)  # optional momentum buffer
    use_momentum = True
    momentum_beta = 0.9

    # LP refinement schedule
    last_lp_time = time.time()
    lp_interval = 30.0          # try an LP every ~30 s
    mix_weight = 0.5            # initial mixing coefficient for LP update
    mix_decay = 0.97
    mix_min = 0.02

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        n = best_seq.shape[0]

        # -----------------------------------------------------------
        # 2) Sub‑gradient descent step
        grad = subgradient_max_conv(best_seq)

        if use_momentum:
            momentum = momentum_beta * momentum - step * grad
            cand = best_seq + momentum
        else:
            cand = best_seq - step * grad

        cand = project_simplex(cand, s=1.0)

        cand_score = cur_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[grad] iter={iteration:04d}  score={cand_score:.6f}")

        # decay step size (but keep it above a tiny floor)
        step = max(1e-6, step * step_decay)

        # -----------------------------------------------------------
        # 3) Random exploration (small Dirichlet perturbation)
        if iteration % 200 == 0:
            pert = random_dirichlet_sequence(n, alpha=0.3) * 0.05
            cand = best_seq * 0.95 + pert
            cand = project_simplex(cand, s=1.0)
            cand_score = cur_score(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[pert] iter={iteration:04d}  score={cand_score:.6f}")

        # -----------------------------------------------------------
        # 4) LP refinement (if enough time has passed)
        now = time.time()
        if now - last_lp_time > lp_interval:
            last_lp_time = now
            conv = np.convolve(best_seq, best_seq)
            M = float(conv.max())
            g_opt = solve_lp_refinement(best_seq, M)
            if g_opt is not None:
                # Mix the new direction with the current iterate
                new_f = (1.0 - mix_weight) * best_seq + mix_weight * g_opt
                new_f = project_simplex(new_f, s=1.0)
                new_score = cur_score(new_f)
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new_f.copy()
                    print(f"[LP ] iter={iteration:04d}  score={new_score:.6f}")
                mix_weight = max(mix_min, mix_weight * mix_decay)

        # -----------------------------------------------------------
        # 5) early‑exit safety check – if we are already extremely close
        if best_score < 1.51:      # far beyond the target of 1.503
            # continue a little longer in case we can push it further
            pass

    # ------------------------------------------------------------------
    # Final sanitisation (project once more to guarantee feasibility)
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = best_seq / best_seq.sum()
    print(f"Best sequence length = {len(best_seq)}  score = {best_score:.6f}")
    return best_seq.tolist()
