def _softmax_gradient(a, gamma):
    n = a.shape[0]
    b = _fft_self_convolve(a)  # length 2n-1
    max_b = b.max()
    # compute softmax weights
    # w_i = exp(gamma * (b_i - max_b))  # to avoid overflow
    w = np.exp(gamma * (b - max_b))
    w_sum = w.sum()
    p = w / w_sum
    # compute gradient g = 2 * conv(p, a_rev)[n-1:n-1+n]
    a_rev = a[::-1]
    # compute convolution of p (len=2n-1) with a_rev (len=n) using FFT with appropriate size

    conv = np.fft.irfft(P * A_rev, n=size)
    # slice
    g = 2 * conv[n-1: n-1 + n]  # length n
    return g, max_b

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using FFT (real input). Result length = 2*len(a)-1."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()     # next power of two
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1} normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # avoid overflow for extremely small α; clip to a safe interval
    alpha = max(0.05, min(alpha, 5.0))
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / total
    return a


# ----------------------------------------------------------------------
def _score_of_alpha(alpha: float, n: int) -> float:
    """Internal (fast) score 2·n·max(a*a) for a Beta(α,α) distribution."""
    a = _beta_distribution(n, alpha)
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
def _search_alpha(n: int, deadline: float) -> tuple[float, float]:
    """
    Find a near‑optimal α for length n.
    Returns (best_alpha, best_score).  Stops when time runs out.
    """
    # coarse grid
    alphas = np.linspace(0.2, 0.9, 15)
    best_alpha = alphas[0]
    best_score = float('inf')
    for a in alphas:
        if time.time() > deadline:
            break
        sc = _score_of_alpha(a, n)
        if sc < best_score:
            best_score, best_alpha = sc, a

    # refined golden‑section search around the best grid point
    left = max(0.1, best_alpha - 0.15)
    right = min(1.0, best_alpha + 0.15)
    phi = (1.0 + 5.0 ** 0.5) / 2.0

    while time.time() < deadline and (right - left) > 1e-5:
        c = right - (right - left) / phi
        d = left + (right - left) / phi
        sc = _score_of_alpha(c, n)
        sd = _score_of_alpha(d, n)
        if sc < sd:
            right = d
            if sc < best_score:
                best_score, best_alpha = sc, c
        else:
            left = c
            if sd < best_score:
                best_score, best_alpha = sd, d
    return best_alpha, best_score


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, gamma: float) -> tuple[np.ndarray, float]:
    """
    Compute the gradient of the smoothed objective
        Lγ(a) = (1/γ)·log Σ_i exp(γ·(a*a)_i)
    Returns (g, max_b) where g is the gradient and max_b = max_i (a*a)_i.
    """
    n = a.shape[0]
    # 1) self‑convolution
    b = _fft_self_convolve(a)                     # length 2n‑1
    max_b = b.max()

    # 2) soft‑max weights (stable)
    w = np.exp(gamma * (b - max_b))
    w_sum = w.sum()
    p = w / w_sum                                 # length 2n‑1, Σp = 1

    # 3) gradient = 2 * (p * a_rev)  (cross‑correlation)
    a_rev = a[::-1]
    size = 1 << ((len(p) + n - 1) - 1).bit_length()
    P = np.fft.rfft(p, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(P * A_rev, n=size)
    # slice the central part that corresponds to indices j = 0…n‑1
    g = 2.0 * conv[n - 1:n - 1 + n]
    return g, max_b


# ----------------------------------------------------------------------
def _refine_with_gradient(
    a0: np.ndarray,
    deadline: float,
    init_name: str = "beta",
) -> tuple[np.ndarray, float]:
    """
    Run the soft‑max gradient descent starting from a0 until `deadline`.
    Returns the best sequence found (as ndarray) and its internal score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    # initial score
    b = _fft_self_convolve(a)
    best_score = float(2.0 * n * b.max())

    # optimiser state
    velocity = np.zeros_like(a)
    beta = 0.9
    eta0 = 0.4
    gamma = 0.5                     # start smooth, increase later

    it = 0
    while time.time() < deadline:
        it += 1

        # anneal gamma slowly (more focus on true max later)
        if it % 300 == 0:
            gamma = min(20.0, gamma * 1.15)

        # gradient of the smoothed max
        g, cur_max = _softmax_gradient(a, gamma)

        # momentum update
        velocity = beta * velocity + (1.0 - beta) * g

        # diminishing step size
        eta = eta0 / math.sqrt(it)

        # gradient step + projection onto simplex
        a = a - eta * velocity
        a = _project_onto_simplex(a, s=1.0)

        # occasional small jitter to hop out of tiny plateaus
        if it % 800 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

        cur_score = float(2.0 * n * cur_max)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] len={n} init={init_name} iter={it} "
                f"score={best_score:.6f}"
            )
        # early stop if we are already very good
        if best_score < 1.35:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    # keep a safe margin for final printing / return
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # Determine a base length (reuse previous solution if available)
    try:
        base_n = len(height_sequence_1)    # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # A modest set of lengths to explore (ensuring at least 500 entries)
    lengths = sorted(
        {
            max(500, int(base_n * f))
            for f in (0.75, 1.0, 1.25, 1.5)
            if int(base_n * f) >= 500
        }
    )
    if not lengths:
        lengths = [8000]

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    for n in lengths:
        # slice of remaining time for this length (≈30 % of what is left)
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        slice_deadline = time.time() + remaining * 0.33

        # ---- 1) pick a good α for the Beta family -------------------
        alpha_deadline = time.time() + (slice_deadline - time.time()) * 0.3
        best_alpha, alpha_score = _search_alpha(n, alpha_deadline)

        # initialise from the Beta distribution
        a0 = _beta_distribution(n, best_alpha)

        # optionally incorporate the historic solution if it matches the length
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.size == n and prev.sum() > 0:
                a_prev = prev / prev.sum()
                a0 = (a0 + a_prev) / 2.0
                a0 = _project_onto_simplex(a0, s=1.0)
        except Exception:
            pass

        # ---- 2) smooth gradient refinement ---------------------------
        refine_seq, refine_score = _refine_with_gradient(
            a0, slice_deadline, init_name=f"beta_n{n}"
        )

        # ---- 3) final black‑box evaluation ----------------------------
        try:
            final_score = evaluate_sequence(refine_seq.tolist())
        except Exception:
            final_score = refine_score

        if final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = refine_seq.copy()
            print(
                f"*** New global best → len={n} alpha={best_alpha:.4f} "
                f"score={overall_best_score:.6f}"
            )

        # early exit if we have already reached the target bound
        if overall_best_score <= 1.503:
            break

        # stop if overall deadline is approaching
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Fallback in case nothing succeeded
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())

    print(
        f"Final best score: {overall_best_score:.6f} "
        f"(length = {len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
