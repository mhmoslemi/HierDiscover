"""
Hill‑climbing with incremental convolution updates.

The evaluation function is scale‑invariant, so we optimise the shape of the
step‑function on the probability simplex (Σ a_i = 1, a_i ≥ 0).  For a
fixed shape the objective reduces to

    O = 2·n·max_k (a * a)[k]               (1)

where n = len(a).  The algorithm starts from the best known construction
(`height_sequence_1` if present) and repeatedly moves a small amount of
mass from a position that contributes heavily to the current maximal
autocorrelation towards a low‑mass position.  Convolution updates are
performed incrementally in O(n) time using the identity

    (a+δa)*(a+δa) = a*a + 2·(a*δa) + (δa*δa),

where δa has only two non‑zero entries (source and destination).  This
allows millions of cheap proposals inside the 1000‑second budget.
The best sequence seen so far is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Incremental convolution update for a mass transfer src → dst of size δ.
# ----------------------------------------------------------------------
def _apply_move(seq: np.ndarray, conv: np.ndarray, src: int, dst: int,
                delta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Apply a move that transfers ``delta`` mass from ``src`` to ``dst``.
    Returns the new sequence and the updated full convolution (length 2n‑1).
    The update uses the exact formula for (a+δa)*(a+δa) and runs in O(n).
    """
    n = seq.shape[0]
    # Prepare new objects (copies) – cheap because they are small NumPy arrays.
    new_seq = seq.copy()
    new_conv = conv.copy()

    # δa: -δ at src, +δ at dst
    delta_src = -delta
    delta_dst = +delta

    # 2·(a*δa) part – slice updates.
    # a is the old sequence, required for the linear part.
    new_conv[src:src + n] += 2.0 * delta_src * seq
    new_conv[dst:dst + n] += 2.0 * delta_dst * seq

    # (δa*δa) part – three scalar updates.
    idx2src = 2 * src
    idx2dst = 2 * dst
    idxsum = src + dst

    new_conv[idx2src] += delta_src * delta_src          # (=δ²)
    new_conv[idx2dst] += delta_dst * delta_dst          # (=δ²)
    new_conv[idxsum] += 2.0 * delta_src * delta_dst     # (=‑2δ²)

    # Update the sequence.
    new_seq[src] += delta_src
    new_seq[dst] += delta_dst

    # Guard against tiny negative values caused by rounding.
    if new_seq[src] < 0.0:
        new_seq[src] = 0.0
    if new_seq[dst] < 0.0:
        new_seq[dst] = 0.0

    return new_seq, new_conv


# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that tries to minimise
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a small safety margin for the harness (2 s)
    deadline = start + max(0.0, budget_s - 2.0)

    # --------------------------------------------------------------
    # Initialise a starting sequence (prefer the best known one).
    # --------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)       # type: ignore
        if base.size == 0:
            raise ValueError
        base = np.maximum(base, 0.0)                          # safety
    except Exception:
        # Fallback: a modest uniform vector.
        base = np.full(2000, 1.0 / 2000.0, dtype=float)

    # Normalise to the probability simplex (sum = 1).  Scaling does not
    # affect the evaluation score because it is homogeneous of degree 0.
    base_sum = base.sum()
    if base_sum == 0.0:
        base = np.full_like(base, 1.0 / base.size)
    else:
        base = base / base_sum

    a = base.copy()
    n = a.shape[0]

    # Initial convolution and objective.
    conv = np.convolve(a, a)
    M = conv.max()
    best_seq = a.copy()
    best_score = 2.0 * n * M  # because sum(a)=1

    # Occasionally recompute the convolution from scratch to avoid drift.
    recompute_every = max(5000, n // 2)

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # Periodic full recomputation for numerical stability.
        if iteration % recompute_every == 0:
            conv = np.convolve(a, a)
            M = conv.max()

        # ----------------------------------------------------------
        # Identify the shift that attains the current maximum.
        # ----------------------------------------------------------
        max_idx = int(conv.argmax())
        M = float(conv[max_idx])

        # ----------------------------------------------------------
        # Heuristic to pick a source index that contributes most to
        # the maximal convolution entry.
        # ----------------------------------------------------------
        low = max(0, max_idx - (n - 1))
        high = min(n - 1, max_idx)
        i_range = np.arange(low, high + 1, dtype=int)
        j_range = max_idx - i_range
        contrib = a[i_range] * a[j_range]

        # Choose the index with the largest product a_i * a_j.
        best_pair = int(i_range[np.argmax(contrib)])
        partner = int(max_idx - best_pair)

        # Pick the heavier side of the pair as the source.
        src = best_pair if a[best_pair] >= a[partner] else partner

        # ----------------------------------------------------------
        # Destination – pick a low‑mass index (bottom 30 % quantile).
        # ----------------------------------------------------------
        thresh = np.percentile(a, 30)
        low_mass_idxs = np.where(a <= thresh)[0]
        if low_mass_idxs.size == 0:
            dst = np.random.randint(n)
        else:
            dst = int(np.random.choice(low_mass_idxs))
        # Ensure src ≠ dst.
        if dst == src:
            # fall back to a random different index
            candidates = np.arange(n)
            candidates = candidates[candidates != src]
            dst = int(np.random.choice(candidates))

        # ----------------------------------------------------------
        # Step size schedule (larger early, tiny later).
        # ----------------------------------------------------------
        elapsed = time.time() - start
        time_frac = elapsed / budget_s
        # start with up to 20 % of the source mass, shrink to 0.1 % near the end.
        max_fraction = 0.20 * (1.0 - time_frac) + 0.001 * time_frac
        delta = min(a[src] * max_fraction, a[src] * 0.5)
        if delta <= 0.0:
            continue  # nothing to move

        # ----------------------------------------------------------
        # Apply the move (incremental convolution update).
        # ----------------------------------------------------------
        a_candidate, conv_candidate = _apply_move(a, conv, src, dst, delta)

        # New maximal autocorrelation after the move.
        M_new = float(conv_candidate.max())

        # ----------------------------------------------------------
        # Acceptance rule (simulated‑annealing style).
        # ----------------------------------------------------------
        accept = False
        if M_new < M:
            accept = True
        else:
            # Temperature proportional to the current maximum, shrinking linearly.
            temp_start = 0.05 * M
            temp_end = 0.0001 * M
            temperature = temp_start * (1.0 - time_frac) + temp_end * time_frac
            prob = np.exp(-(M_new - M) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_candidate
            conv = conv_candidate
            M = M_new

        # ----------------------------------------------------------
        # Record the best sequence seen so far.
        # ----------------------------------------------------------
        current_score = 2.0 * n * M
        if current_score < best_score:
            best_score = current_score
            best_seq = a.copy()
            print(f"[Iter {iteration:07d}] New best score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence (as a plain Python list) after the budget.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)      # safety
    return best_seq.tolist()
```

This implementation performs a large number of cheap, directed
mass‑transfer proposals while keeping the convolution up‑to‑date in linear
time.  By always moving mass away from the indices that dominate the
current maximal autocorrelation, the algorithm continuously pushes the
value `max_k (a*a)[k]` down, which directly reduces the evaluation
score `2·n·max_k (a*a)[k]`.  The gradual cooling schedule allows occasional
up‑hill moves to escape shallow local minima, and periodic full
re‑convolutions eliminate accumulated numerical drift.  The method works
for any sequence length (the previous best construction of length ~7300
is used as a warm start) and respects all constraints of
`evaluate_sequence`.
