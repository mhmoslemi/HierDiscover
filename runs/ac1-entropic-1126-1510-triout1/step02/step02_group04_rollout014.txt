import numpy as np

# ----------------------------------------------------------------------
# Helper: project a vector onto the probability simplex (non‑negative,
# sum = 1).  Used after every mutation / gradient step.
# ----------------------------------------------------------------------
def _clip_and_normalise(v: np.ndarray) -> np.ndarray:
    """Clip negatives to 0 and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        v = np.full_like(v, 1.0 / v.size)
    else:
        v = v / s
    return v


# ----------------------------------------------------------------------
# Fast convolution (linear, not circular) using FFT.
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length a power of two.
    Returns an array of length 2*len(a)-1.
    """
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())          # smallest power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Scoring function (identical to evaluate_sequence() for sum = 1).
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv) for a probability vector `seq`."""
    n = seq.size
    conv = _conv_fft(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Gradient step that tries to reduce the *k* largest convolution entries.
# ----------------------------------------------------------------------
def _multi_lag_gradient_step(seq: np.ndarray, eta: float, top_k: int = 3) -> np.ndarray:
    """
    One projected gradient step on the sum of the `top_k` largest
    convolution entries.
    """
    n = seq.size
    conv = _conv_fft(seq)

    # indices of the k largest entries (ties broken arbitrarily)
    top_idxs = np.argpartition(-conv, np.arange(top_k))[:top_k]

    g = np.zeros_like(seq)
    for k in top_idxs:
        # contribution of a_i to conv[k] is 2·a_{k‑i} (if inside the array)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = k - i
        # add 2·a_j to the gradient at position i
        g[i] += 2.0 * seq[j]

    new_seq = seq - eta * g
    return _clip_and_normalise(new_seq)


# ----------------------------------------------------------------------
# Targeted move: move a tiny fraction of mass from the pair that
# contributes most to the maximal convolution entry.
# ----------------------------------------------------------------------
def _targeted_move(seq: np.ndarray, frac: float = 0.03) -> np.ndarray:
    """
    Identify the lag with maximal convolution value, locate the pair
    (i, k‑i) that gives the largest product a_i·a_{k‑i} and move a
    small fraction of the larger of the two to a random location.
    """
    n = seq.size
    conv = _conv_fft(seq)
    k = int(conv.argmax())

    i_arr = np.arange(n, dtype=int)
    j_arr = k - i_arr
    mask = (j_arr >= 0) & (j_arr < n)
    if not np.any(mask):
        # Very unlikely, but fall back to a random perturbation.
        cand = seq + frac * (np.random.rand(n) - 0.5)
        return _clip_and_normalise(cand)

    contributions = np.zeros_like(seq)
    contributions[mask] = seq[i_arr[mask]] * seq[j_arr[mask]]

    i_max = int(np.argmax(contributions))
    j_max = k - i_max

    # choose the larger coefficient as the source of the transfer
    src = i_max if seq[i_max] >= seq[j_max] else j_max
    dst = np.random.randint(n)
    while dst == src:
        dst = np.random.randint(n)

    # transfer up to `frac`·seq[src] (scaled by an extra random factor)
    amount = frac * seq[src] * np.random.rand()
    cand = seq.copy()
    cand[src] -= amount
    cand[dst] += amount
    return _clip_and_normalise(cand)

    lam = 80              # number of offspring each generation
    sigma = 0.10          # initial mutation strength (std. dev.)

    # initialise parent population: start from the base and add a few
    # random simplex points for diversity.
    parents = np.empty((mu, n), dtype=float)
    parents[0] = base.copy()
    for i in range(1, mu):
        rnd = np.random.rand(n)
        parents[i] = _clip_and_normalise(rnd)

    parent_scores = np.array([_score(p) for p in parents])
    best_idx = int(parent_scores.argmin())
    best = parents[best_idx].copy()
    best_score = float(parent_scores[best_idx])
    print(f"[Init] n = {n}, initial best score = {best_score:.6f}")

    generation = 0
    last_improve = generation

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        generation += 1

        # ---- recombination ------------------------------------------------
        mean = parents.mean(axis=0)

        # ---- generate offspring with antithetic Gaussian mutations -----------
        offspring = np.empty((lam, n), dtype=float)
        # half of them are antithetic pairs
        half = lam // 2
        for i in range(half):
            z = np.random.randn(n)
            cand1 = mean + sigma * z
            cand2 = mean - sigma * z
            offspring[2 * i] = _clip_and_normalise(cand1)
            offspring[2 * i + 1] = _clip_and_normalise(cand2)

        # ---- deterministic problem‑specific candidates ----------------------
        # 1) targeted move based on the current best
        offspring[0] = _targeted_move(best, frac=0.03)
        # 2) symmetrised version (average with its reverse)
        rev = best[::-1]
        offspring[1] = _clip_and_normalise(0.5 * (best + rev))

        # ---- evaluation ----------------------------------------------------
        scores = np.empty(lam, dtype=float)
        for i, cand in enumerate(offspring):
            scores[i] = _score(cand)

        # ---- update global best --------------------------------------------
        min_off = int(scores.argmin())
        if scores[min_off] < best_score:
            best_score = float(scores[min_off])
            best = offspring[min_off].copy()
            last_improve = generation
            print(f"[Gen {generation:5d}] New best score = {best_score:.6f}")

        # ---- survivor selection (μ best of parents+offspring) ---------------
        combined = np.vstack([parents, offspring])
        combined_scores = np.concatenate([parent_scores, scores])
        keep = combined_scores.argsort()[:mu]
        parents = combined[keep]
        parent_scores = combined_scores[keep]

        # ---- adapt step size (1/5‑rule) ------------------------------------
        # success = fraction of offspring that are better than the current mean
        # (approximated by the best parent after selection)
        success = np.mean(scores < parent_scores[0] + 1e-12)
        if success > 0.2:
            sigma = min(sigma * 1.2, 0.5)
        else:
            sigma = max(sigma * 0.85, 1e-5)

        # ---- every few generations perform a few fast gradient refinements --
        if generation % 20 == 0:
            tmp = best.copy()
            for step in range(3):
                eta = 0.5 / (generation + step + 1.0)
                tmp = _multi_lag_gradient_step(tmp, eta=eta, top_k=4)
            tmp_score = _score(tmp)
            if tmp_score < best_score:
                best_score = float(tmp_score)
                best = tmp
                print(f"[Gen {generation:5d}] Gradient‑refined best = {best_score:.6f}")

        # ---- simple stagnation restart --------------------------------------
        if generation - last_improve > 600:
            # replace the worst parent by a fresh random point
            new_rand = _clip_and_normalise(np.random.rand(n))
            parents[-1] = new_rand
            parent_scores[-1] = _score(new_rand)
            sigma = 0.10
            last_improve = generation
            print(f"[Gen {generation:5d}] Restart (stagnation)")

    # ------------------------------------------------------------------
    # Return the best candidate found.
    # ------------------------------------------------------------------
    print(f"[Done] Best score after {generation} generations = {best_score:.6f}")
    return best.tolist()
