# ----------------------------------------------------------------------
# Projection onto the probability simplex {x ≥ 0, sum(x)=s}
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast convolution of a real vector with itself using FFT
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Return a * a using FFT (real input). Length = 2·len(a)‑1."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Discrete arcsine probability vector (used for a good start)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Linear interpolation / up‑sampling of a sequence
def _interpolate(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_new = np.linspace(0.0, 1.0, new_n)
    x_old = np.linspace(0.0, 1.0, old_n)
    return np.interp(x_new, x_old, seq)

            a = a - eta * velocity
            a = _project_onto_simplex(a, s=1.0)

            # 6) enforce symmetry after projection
            a = (a + a[::-1]) / 2.0
            a = _project_onto_simplex(a, s=1.0)

            # 7) occasional tiny random jitter to escape plateaux
            if it % 2000 == 0 and np.random.rand() < 0.09:
                a += np.random.randn(n) * 1e-4
                a = _project_onto_simplex(a, s=1.0)

        # ------------------------------------------------------------------
        # Final (expensive) evaluation of the best sequence for this length
        try:
            final_score = evaluate_sequence(best_local_seq.tolist())
        except Exception:
            final_score = best_local_score

        if final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = best_local_seq.copy()
            print(
                f"*** Global best updated: len={n} "
                f"score={overall_best_score:.6f}"
            )

        # Store for the next length (up‑sample)
        prev_seq = best_local_seq.copy()

        # early exit if we have already reached the target bound
        if overall_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # If time is still left, try a short refinement on the current global best.
    time_left = deadline - time.time()
    if overall_best_seq is not None and time_left > 8.0:
        a = overall_best_seq.copy()
        n = len(a)
        velocity = np.zeros_like(a)
        beta = 0.9
        eta0 = 0.4
        it = 0
        refinement_deadline = time.time() + min(8.0, time_left * 0.5)
        while time.time() < refinement_deadline:
            it += 1
            b = _fft_self_convolve(a)
            max_b = b.max()
            cur_score = 2.0 * n * max_b
            # gradient from top‑3 entries (as before)
            top_M = 3
            top_idx = np.argpartition(b, -top_M)[-top_M:]
            g = np.zeros_like(a)
            for k in top_idx:
                idx = k - np.arange(n)
                mask = (idx >= 0) & (idx < n)
                g[mask] += 2.0 * a[idx[mask]]
            g = g / float(top_M)

            velocity = beta * velocity + (1.0 - beta) * g
            eta = eta0 / np.sqrt(it)
            a = a - eta * velocity
            a = _project_onto_simplex(a, s=1.0)
            a = (a + a[::-1]) / 2.0
            a = _project_onto_simplex(a, s=1.0)

        try:
            refined_score = evaluate_sequence(a.tolist())
        except Exception:
            refined_score = float("inf")
        if refined_score < overall_best_score:
            overall_best_score = refined_score
            overall_best_seq = a.copy()
            print(
                f"*** After final refinement: score={overall_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Fallback – should never be needed but guarantees a return value.
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())

    print(
        f"Final best score: {overall_best_score:.6f} "
        f"(length = {len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
