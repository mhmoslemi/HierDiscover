The goal is to minimise  

\[
E(a)=\frac{2\,n\,\max\limits_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\]

where `a*a` denotes the full linear convolution of the non‑negative step‑function
`a`.  
The evaluation routine already projects the vector onto the feasible set
(`a_i≥0` and caps each entry at 1000).  The optimiser therefore works **only**
with vectors that satisfy the constraints; the projection onto the simplex
\(\{a\ge 0,\;\sum a_i=S\}\) (with the normalising sum \(S=\sqrt{2n}\)) guarantees
feasibility after every update.

The optimiser consists of three stages:

1. **Large‑scale parametric search** – a wide collection of analytical families
   (inverse‑√, power‑law, exponential, cosine, logarithmic, mixed, random block
   patterns…) is sampled, projected onto the simplex and the best candidate is
   kept.

2. **Smoothed projected Adam optimisation** – starting from the best candidate
   we minimise a smooth approximation of the max‑convolution:
   \[
   F_T(a)=\sum_k w_k(T)\,(a*a)_k,\qquad
   w_k(T)=\frac{\exp\bigl(( (a*a)_k-\max(a*a))/T\bigr)}{\sum_j\exp(\cdot)}.
   \]
   The temperature `T` is slowly reduced, turning `F_T` into the true max.
   Gradients are obtained analytically (convolution with the weight vector) and
   an Adam step is taken, followed by Euclidean projection onto the simplex.
   Random perturbations and occasional Adam‑state resets help to escape plateaus.

3. **Final random‑block refinement** – a short burst of random block candidates
   (few large constant blocks) is evaluated; any improvement replaces the best
   solution.

All heavy work (convolutions, gradients) is performed with FFTs, giving
`O(n log n)` per iteration.  The whole search respects the 1000 s budget and
prints progress information.

"""
Propose a step‑function (non‑negative sequence) with a low evaluation score.

The algorithm proceeds in three phases:

1.  Broad sampling of parametric families (inverse‑sqrt, power‑law, exponential,
    cosine, logarithmic, mixtures, random block patterns).  Each candidate is
    projected onto the simplex Σa = √(2·n) and the best one is kept.

2.  Smoothed projected Adam optimisation.  The smooth max is realised by a
    soft‑max over the convolution entries with a temperature that is slowly
    decreased.  The gradient of the weighted sum is obtained analytically via
    FFT‑based convolutions.  After each Adam update the vector is projected back
    onto the simplex.  Random perturbations and Adam‑state resets are applied
    periodically to avoid stagnation.

3.  A short random‑block search that may further improve the solution.

The function returns the best sequence found before the time limit expires.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using real FFT (output length len(u)+len(v)-1)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[:len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full convolution a*a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·(a*a)_k  w.r.t. a   (weights w are fixed).
    Uses the identity   ∂/∂a_i Σ_k w_k (a*a)_k = 2 Σ_k w_k a_{k-i}
    and corrects the double‑counted diagonal terms.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)          # length = 3n‑2
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    # diagonal correction (pairs i=j counted only once)
    diag_w = w[0:2 * n - 1:2]                      # w_{2i}, length n
    grad -= a * diag_w
    return grad


# ----------------------------------------------------------------------
def _score(a: np.ndarray, size_fft: int) -> float:
    """Exact evaluation score (no security checks)."""
    conv = _self_convolution(a, size_fft)
    return 2.0 * a.shape[0] * conv.max() / (a.sum() ** 2)

    # Determine length n (use the reference sequence if available)
    try:
        n = len(height_sequence_1)               # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)               # normalisation used in literature

    # FFT size – must be >= 3*n‑2 for the gradient computation
    size_fft = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------------
    # Helper to keep the best solution found so far
    best_a = None
    best_score = float("inf")

    def _update_best(candidate: np.ndarray):
        nonlocal best_a, best_score
        sc = _score(candidate, size_fft)
        if sc < best_score:
            best_score = sc
            best_a = candidate.copy()
            print(f"[shape] new best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1. PARAMETRIC SHAPE SEARCH
    # ------------------------------------------------------------------
    # Uniform
    a = np.ones(n)
    a = _project_to_simplex(a, target_sum)
    _update_best(a)

    # Helper to generate and test a candidate shape
    def _test_shape(arr: np.ndarray):
        arr = _project_to_simplex(arr, target_sum)
        _update_best(arr)

    # ---- families -------------------------------------------------
    # inverse‑sqrt
    for _ in range(150):
        beta = np.random.rand() * 20.0 + 1e-3   # avoid zero
        arr = 1.0 / np.sqrt(np.arange(n, dtype=float) + beta)
        _test_shape(arr)

    # double‑ended inverse‑sqrt
    for _ in range(150):
        beta = np.random.rand() * 20.0 + 1e-3
        i = np.arange(n, dtype=float)
        arr = (1.0 / np.sqrt(i + beta) +
               1.0 / np.sqrt((n - 1 - i) + beta))
        _test_shape(arr)

    # power‑law
    for _ in range(150):
        beta = np.random.rand() * 20.0 + 1e-3
        p = np.random.rand() * 1.7 + 0.3            # exponent in [0.3, 2.0]
        arr = 1.0 / (np.arange(n, dtype=float) + beta) ** p
        _test_shape(arr)

    # exponential decay
    for _ in range(150):
        decay = np.random.rand() * 0.02 + 1e-4
        arr = np.exp(-decay * np.arange(n, dtype=float))
        _test_shape(arr)

    # cosine‑power (large at the ends, small in the middle)
    for _ in range(150):
        p = np.random.rand() * 6.0 + 0.5
        arr = np.abs(np.cos(np.pi * np.arange(n, dtype=float) / (n - 1))) ** p
        _test_shape(arr)

    # logarithmic decay
    for _ in range(120):
        beta = np.random.rand() * 10.0 + 1.0
        arr = 1.0 / (np.log(np.arange(n, dtype=float) + beta + 1.0) + 1.0)
        _test_shape(arr)

    # mixture of two random families
    for _ in range(130):
        mix = np.random.rand()
        # pick two families at random
        if np.random.rand() < 0.5:
            # inverse‑sqrt + exponential
            beta = np.random.rand() * 20.0 + 1e-3
            decay = np.random.rand() * 0.02 + 1e-4
            a1 = 1.0 / np.sqrt(np.arange(n, dtype=float) + beta)
            a2 = np.exp(-decay * np.arange(n, dtype=float))
        else:
            # power‑law + cosine‑power
            beta = np.random.rand() * 20.0 + 1e-3
            p1 = np.random.rand() * 1.7 + 0.3
            p2 = np.random.rand() * 6.0 + 0.5
            a1 = 1.0 / (np.arange(n, dtype=float) + beta) ** p1
            a2 = np.abs(np.cos(np.pi * np.arange(n, dtype=float) / (n - 1))) ** p2
        arr = mix * a1 + (1 - mix) * a2
        _test_shape(arr)

    # random block candidates (few blocks, large values)
    for _ in range(400):
        nb = np.random.randint(2, 9)                 # number of blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.zeros(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        _test_shape(cand)

    print(f"[phase 1] best score after shape search = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. SMOOTHED PROJECTED ADAM OPTIMISATION
    if best_a is None:
        # fallback to uniform if somehow no candidate was accepted
        best_a = _project_to_simplex(np.ones(n), target_sum)

    a = best_a.copy()

    # Initial temperature – a fraction of the current spread of the convolution
    conv = _self_convolution(a, size_fft)
    T = max(1e-4, (conv.max() - conv.min()) * 0.5)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.02 * target_sum / n                # base step size (≈0.5·average entry)
    lr = lr0

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = beta1
    beta2_pow = beta2

    it = 0
    max_iters = 400_000                         # safety cap

    while time.time() < deadline and it < max_iters:
        # ---- evaluate current vector ----
        conv = _self_convolution(a, size_fft)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{it:07d}] improvement → {best_score:.6f}")

        # ---- soft‑max weights (smooth max) ----
        max_c = conv.max()
        shifted = conv - max_c
        # avoid overflow / underflow
        exp_arg = shifted / max(T, 1e-12)
        exp_arg = np.clip(exp_arg, -50, 0)      # values ≤0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the smooth objective ----
        grad = _weighted_gradient(a, w, size_fft)

        # ---- Adam update ----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # ---- projection onto the simplex (feasibility) ----
        a = _project_to_simplex(a, target_sum)

        # ---- schedule updates ----
        beta1_pow *= beta1
        beta2_pow *= beta2
        lr *= 0.9999                # very gentle decay
        T *= 0.9995                # cool down

        # ---- occasional random perturbation to escape plateaus ----
        if it % 8000 == 0:
            noise = np.random.randn(n) * target_sum * 1e-3
            a = _project_to_simplex(a + noise, target_sum)
            # reset Adam moments to avoid carrying stale directions
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = beta1
            beta2_pow = beta2
            print(f"[{it:07d}] random perturbation")

    print(f"[phase 2] best score after Adam optimisation = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. FINAL RANDOM BLOCK REFINEMENT
    for _ in range(300):
        nb = np.random.randint(2, 7)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.zeros(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = _score(cand, size_fft)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[block] improved to {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f} (elapsed {time.time() - (deadline - budget_s + 0.5):.1f}s)")
    return best_a.tolist()
