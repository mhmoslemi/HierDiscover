"""
Propose a step‑function that improves the current best bound.

Algorithm
---------
1. Initialise a population on the probability simplex.
   If a previous construction `height_sequence_1` is present it is used,
   otherwise a uniform vector of length `n≈8000` is taken.
2. Run a (μ/λ)‑ES with μ=5, λ=30.  Offspring are created by:
      * Gaussian mutation + simplex projection,
      * a targeted move that reduces the currently maximal convolution entry,
      * an aggressive move that removes mass from the pair giving the largest
        contribution to that entry,
      * a reverse‑symmetrised copy,
      * a few multi‑lag gradient steps (sub‑gradient of the top‑k convolution
        peaks).
3. The best individual is kept (elitism).  A 1/5‑rule adapts the Gaussian
   step size.
4. Every 20 generations a short gradient‑refinement of the current best
   is performed (top‑k = 15, decreasing learning rate).
5. If the search stagnates for 500 generations the worst parent is
   replaced by a fresh random point (restart).

All vectors are kept on the simplex (non‑negative, sum = 1) by simple
clipping and renormalisation.  The fast score implementation
`_score(seq) = 2·n·max(conv(seq,seq))` is used during the optimisation;
the final list is returned for the external `evaluate_sequence` check.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (non‑negative, sum=1)
# ----------------------------------------------------------------------
def _clip_and_normalise(v: np.ndarray) -> np.ndarray:
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fallback to uniform point
        v = np.full_like(v, 1.0 / v.size)
    else:
        v = v / s
    return v


# ----------------------------------------------------------------------
# Fast scoring (exactly the same as evaluate_sequence for simplex vectors)
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    n = seq.size
    conv = np.convolve(seq, seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Multi‑lag gradient step (sub‑gradient of the top‑k convolution peaks)
# ----------------------------------------------------------------------
def _multi_lag_gradient_step(seq: np.ndarray, eta: float,
                             top_k: int = 10) -> np.ndarray:
    n = seq.size
    conv = np.convolve(seq, seq)
    # indices of the k largest values (unsorted)
    top_idxs = np.argpartition(-conv, range(top_k))[:top_k]

    g = np.zeros_like(seq)
    idx = np.arange(n)
    for k in top_idxs:
        j = k - idx
        mask = (j >= 0) & (j < n)
        g[mask] += 2.0 * seq[j[mask]]

    new_seq = seq - eta * g
    return _clip_and_normalise(new_seq)


# ----------------------------------------------------------------------
# Targeted move – reduces the single maximal convolution entry
# ----------------------------------------------------------------------
def _targeted_move(seq: np.ndarray, frac: float = 0.05) -> np.ndarray:
    n = seq.size
    conv = np.convolve(seq, seq)
    k = int(conv.argmax())

    # contribution a_i * a_{k-i}
    i_arr = np.arange(n)
    j = k - i_arr
    mask = (j >= 0) & (j < n)

    if not np.any(mask):
        # fall back to a tiny random jitter
        cand = seq + frac * (np.random.rand(n) - 0.5)
        return _clip_and_normalise(cand)

    contrib = np.zeros_like(seq)
    contrib[mask] = seq[i_arr[mask]] * seq[j[mask]]

    i_max = int(np.argmax(contrib))
    j_max = k - i_max

    src = i_max if seq[i_max] >= seq[j_max] else j_max
    dst = np.random.randint(n)
    while dst == src:
        dst = np.random.randint(n)

    amount = frac * seq[src] * np.random.rand()
    cand = seq.copy()
    cand[src] -= amount
    cand[dst] += amount
    return _clip_and_normalise(cand)


# ----------------------------------------------------------------------
# Aggressive move – removes mass from both members of the largest pair
# ----------------------------------------------------------------------
def _aggressive_move(seq: np.ndarray, frac: float = 0.07) -> np.ndarray:
    n = seq.size
    conv = np.convolve(seq, seq)
    k = int(conv.argmax())

    i_arr = np.arange(n)
    j = k - i_arr
    mask = (j >= 0) & (j < n)

    if not np.any(mask):
        # fallback: tiny random jitter
        cand = seq + frac * (np.random.rand(n) - 0.5)
        return _clip_and_normalise(cand)

    contrib = np.zeros_like(seq)
    contrib[mask] = seq[i_arr[mask]] * seq[j[mask]]

    i_max = int(np.argmax(contrib))
    j_max = k - i_max

    # transfer a fraction from both i_max and j_max to random new locations
    amount_i = frac * seq[i_max] * np.random.rand()
    amount_j = frac * seq[j_max] * np.random.rand()

    dst_i = np.random.randint(n)
    while dst_i == i_max:
        dst_i = np.random.randint(n)
    dst_j = np.random.randint(n)
    while dst_j == j_max:
        dst_j = np.random.randint(n)

    cand = seq.copy()
    cand[i_max] -= amount_i
    cand[dst_i] += amount_i
    cand[j_max] -= amount_j
    cand[dst_j] += amount_j
    return _clip_and_normalise(cand)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Evolution‑Strategy optimiser with problem‑specific mutations.
    Returns a list of non‑negative heights (sum > 0) that aims to minimise
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + max(0.0, budget_s - 0.5)

    # --------------------------------------------------------------
    # Initialise dimension and base solution
    # --------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        base = _clip_and_normalise(base)
    except Exception:
        n = 8000
        base = np.full(n, 1.0 / n, dtype=float)

    # ES parameters
    mu = 5          # parents retained
    lam = 30        # offspring per generation
    sigma = 0.08    # mutation strength (std‑dev)

    # initialise parent population (first parent = base, others random)
    parents = np.empty((mu, n), dtype=float)
    parents[0] = base.copy()
    for i in range(1, mu):
        parents[i] = _clip_and_normalise(np.random.rand(n))

    parent_scores = np.array([_score(p) for p in parents])
    best_idx = int(parent_scores.argmin())
    best = parents[best_idx].copy()
    best_score = float(parent_scores[best_idx])

    generation = 0
    last_improve_gen = 0

    print(f"[Init] n = {n}, best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Evolution loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        generation += 1
        mean = parents.mean(axis=0)

        # --------------------------- offspring generation ---------------------------
        offspring = np.empty((lam, n), dtype=float)

        # first few are deterministic, problem‑specific moves
        offspring[0] = _targeted_move(best, frac=0.04)
        offspring[1] = _aggressive_move(best, frac=0.06)
        offspring[2] = _clip_and_normalise(0.5 * (best + best[::-1]))
        # three gradient‑based perturbations of the best
        tmp = best.copy()
        for step in range(3):
            eta = 0.6 / (generation + step + 1.0)
            tmp = _multi_lag_gradient_step(tmp, eta=eta, top_k=12)
        offspring[3] = tmp

        # remaining are Gaussian‑mutated individuals
        for i in range(4, lam):
            cand = mean + sigma * np.random.randn(n)
            offspring[i] = _clip_and_normalise(cand)

        # --------------------------- evaluation ------------------------------------
        scores = np.empty(lam, dtype=float)
        for i, cand in enumerate(offspring):
            scores[i] = _score(cand)

        # --------------------------- best update ---------------------------------
        min_off_idx = int(scores.argmin())
        if scores[min_off_idx] < best_score:
            best_score = float(scores[min_off_idx])
            best = offspring[min_off_idx].copy()
            last_improve_gen = generation
            print(f"[Gen {generation:5d}] New best = {best_score:.6f}")

        # --------------------------- selection ------------------------------------
        combined = np.vstack([parents, offspring])
        combined_scores = np.concatenate([parent_scores, scores])
        keep_idx = combined_scores.argsort()[:mu]
        parents = combined[keep_idx]
        parent_scores = combined_scores[keep_idx]

        # --------------------------- sigma adaptation (1/5 rule) ------------------
        # success = proportion of offspring better than current best
        success = np.mean(scores < best_score + 1e-12)
        if success > 0.2:
            sigma = min(sigma * 1.2, 0.5)
        else:
            sigma = max(sigma * 0.85, 1e-5)

        # --------------------------- occasional gradient refinement ---------------
        if generation % 20 == 0:
            refined = best.copy()
            for step in range(5):
                eta = 0.5 / (generation + step + 1.0)
                refined = _multi_lag_gradient_step(refined, eta=eta, top_k=15)
            refined_score = _score(refined)
            if refined_score < best_score:
                best_score = float(refined_score)
                best = refined
                print(f"[Gen {generation:5d}] Gradient‑refined best = {best_score:.6f}")

        # --------------------------- stagnation restart --------------------------
        if generation - last_improve_gen > 500:
            # replace worst parent by a fresh random point
            new_rand = _clip_and_normalise(np.random.rand(n))
            parents[-1] = new_rand
            parent_scores[-1] = _score(new_rand)
            last_improve_gen = generation
            sigma = 0.08
            print(f"[Gen {generation:5d}] Restart due to stagnation")

    print(f"[Done] Best score after {generation} generations = {best_score:.6f}")
    # Return plain Python list (the external harness will call evaluate_sequence)
    return best.tolist()
