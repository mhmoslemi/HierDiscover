def generate_sidon_set(n, target_m=None):
    used = set()
    positions = []
    available = list(range(n))
    np.random.shuffle(available)
    for pos in available:
        # check if adding pos would cause any sum collision
        collides = False
        for p in positions:
            s = pos + p
            if s in used:
                collides = True
                break
        if not collides:
            # also check self sum
            if pos + pos in used:
                continue
            # accept
            positions.append(pos)
            for p in positions:
                used.add(pos + p)   # but this adds also self sum
            if target_m and len(positions) >= target_m:
                break
    # sort positions for deterministic order
    return sorted(positions)

"""
Hybrid optimizer for the step‑function evaluation problem.

The objective to minimise is
    score = 2 * n * max_k (a * a)[k] / (sum a)^2 .
It is homogeneous, therefore we normalise every candidate to L1‑norm 1
and minimise the quantity  n * max_k (a * a)[k].

Strategy
---------
1. **Random Sidon‑like support** – we greedily construct a set of indices
   with few sum‑collisions (a weak Sidon set).  A uniform weight on this
   support already gives a good starting value because the maximal
   convolution is roughly the number of collisions times (1/m)^2.

2. **LP refinement** – with a fixed support ``f`` we solve the linear
   program
        maximise   Σ g_j
        subject to (f * g)_k ≤ M   for all k,
                    g_j ≥ 0,
   where ``M`` is the current maximum autocorrelation of ``f``.
   The LP tries to increase the total L1 mass while keeping the same
   bound on the convolution.  We convex‑combine the solution with ``f``
   (mixing coefficient adapts automatically).  Several iterations of
   this step quickly lower the maximal convolution.

3. **Gradient‑inspired mass moves** – after the LP stage we locate the
   index ``k*`` that attains the current maximum and move a small amount
   of mass away from the most offending positions.  This is a cheap
   sub‑gradient step for the ``max`` objective.

4. **Simulated‑annealing polishing** – a long random‑pair “mass‑swap”
   phase with a linear cooling schedule finalises the solution and
   explores the neighbourhood of the best candidate found so far.

The whole procedure is repeated with several random seeds (restarts) and
the best sequence seen before the time budget expires is returned.
All heavy computation (convolution, maximum index) is performed with an
FFT‑based routine for speed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional cvxpy import – core of the LP refinement.
# ----------------------------------------------------------------------
try:
    import cvxpy as cp  # type: ignore
    _HAS_CVX = True
except Exception:  # pragma: no cover
    _HAS_CVX = False

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, fast for n up to a few 10 000).
# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray) -> float:
    """Return max_k (a * a)[k] using an FFT based convolution."""
    n = a.shape[0]
    # length of full convolution = 2*n-1, choose next power of two.
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return float(conv.max())


def _max_convolution_and_index(a: np.ndarray):
    """Return (max value, argmax index) of the full convolution a*a."""
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    idx = int(conv.argmax())
    return float(conv.max()), idx


# ----------------------------------------------------------------------
# Random greedy construction of a weak Sidon set (few collisions).
# ----------------------------------------------------------------------
def _greedy_sidons(n: int, target_m: int = None, rng: np.random.Generator = None):
    """
    Build a set of positions S ⊂ {0,…,n‑1} such that sums i+j are
    rarely repeated.  The greedy algorithm shuffles the candidate
    indices and inserts them if they do not introduce a new sum conflict.
    """
    if rng is None:
        rng = np.random.default_rng()
    used = set()
    S = []
    # random order of candidates
    cand = rng.permutation(n)
    max_m = target_m if target_m is not None else int(np.sqrt(2 * n))
    for pos in cand:
        # quick stop if we have enough points
        if len(S) >= max_m:
            break
        # check collisions with existing points (including self)
        conflict = False
        for q in S:
            if (pos + q) in used:
                conflict = True
                break
        if conflict:
            continue
        # also check the self‑sum
        if (pos + pos) in used:
            continue
        # accept the point
        S.append(pos)
        for q in S:
            used.add(pos + q)
        if target_m is not None and len(S) >= target_m:
            break
    S.sort()
    return S


# ----------------------------------------------------------------------
# Linear programming sub‑routine (identical to the reference implementation).
# ----------------------------------------------------------------------
def _lp_max_mass(f: np.ndarray, M: float):
    """Solve   maximise Σ g   subject to   (f * g) ≤ M , g ≥ 0.
    Returns the optimal g as a NumPy array, or None on failure."""
    if not _HAS_CVX:
        return None
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)
    conv_fg = cp.conv(cp.Constant(f), g)  # length 2n‑1
    constraints = [conv_fg <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=500, verbose=False)
    except Exception:  # pragma: no cover
        return None
    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    g_val = g.value
    if g_val is None:
        return None
    return np.asarray(g_val, dtype=float)


# ----------------------------------------------------------------------
# Small gradient‑inspired mass redistribution step.
# ----------------------------------------------------------------------
def _gradient_step(seq: np.ndarray, step: float) -> np.ndarray:
    """
    One sub‑gradient iteration:
    * locate the index k where the convolution is maximal,
    * move a small amount of mass away from the locations that most
      contribute to this max (those with large partner weight).
    """
    max_val, k = _max_convolution_and_index(seq)
    n = seq.shape[0]
    idx = np.arange(n)
    partner_idx = k - idx
    mask = (partner_idx >= 0) & (partner_idx < n)
    partner = np.zeros_like(seq)
    partner[mask] = seq[partner_idx[mask]]
    # contribution of each position to the maximal sum
    contrib = seq * partner
    # pick the most offending position
    i = int(contrib.argmax())
    if contrib[i] == 0.0:
        return seq  # nothing to do
    delta = min(step * seq[i], seq[i] * 0.5)  # move at most 50 % of the mass
    if delta <= 0.0:
        return seq
    # choose a random target (prefer a zero entry)
    zero_candidates = np.where(seq == 0.0)[0]
    if zero_candidates.size > 0:
        j = int(np.random.choice(zero_candidates))
    else:
        j = int(np.random.randint(n))
    new_seq = seq.copy()
    new_seq[i] -= delta
    new_seq[j] += delta
    # renormalise to unit L1 norm (preserves the homogeneity of the score)
    total = new_seq.sum()
    if total <= 0.0:
        return seq
    new_seq /= total
    return new_seq


# ----------------------------------------------------------------------
# Random pairwise mass‑swap used in the polishing phase.
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Pick two distinct indices i, j and exchange a small amount of mass."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    # Symmetric uniform proposal of width sigma
    delta = (np.random.rand() - 0.5) * sigma
    if delta > 0:
        delta = min(delta, seq[j])
        new_seq = seq.copy()
        new_seq[i] += delta
        new_seq[j] -= delta
    else:
        delta = -min(-delta, seq[i])
        new_seq = seq.copy()
        new_seq[i] += delta
        new_seq[j] -= delta
    return new_seq

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that strives to minimise
    ``evaluate_sequence`` under the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a small safety margin
    deadline = start + max(0.0, budget_s - 2.0)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate quickly (uses the official evaluator)
    # ------------------------------------------------------------------
    def _eval(seq):
        try:
            return evaluate_sequence(seq.tolist())
        except Exception:
            return float("inf")

    # ------------------------------------------------------------------
    # Main optimisation loop (multiple restarts)
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline:
        restart += 1
        # ------------------------------------------------------------------
        # 1. Choose a problem size (n) – a compromise between richness and speed.
        # ------------------------------------------------------------------
        n = int(rng.integers(2500, 12000))
        # ------------------------------------------------------------------

        # ------------------------------------------------------------------

        # 3. Initialise a uniform weight on the support.
        # ------------------------------------------------------------------
        a = np.zeros(n, dtype=float)

        # ------------------------------------------------------------------
        # 4. Local LP refinement (few iterations, fast convergence).
        # ------------------------------------------------------------------
        f = a.copy()
        current_M = _max_convolution(f)
        mix_coef = 0.5
        lp_iters = 0
        max_lp_iters = 60
        while time.time() < deadline and lp_iters < max_lp_iters:
            lp_iters += 1
            # recompute the bound M
            current_M = _max_convolution(f)
            g_raw = _lp_max_mass(f, current_M)
            if g_raw is None:
                break
            sum_g = g_raw.sum()
            if sum_g <= 1.0 + 1e-12:  # no progress possible
                break
            g_norm = g_raw / sum_g
            f_candidate = (1.0 - mix_coef) * f + mix_coef * g_norm
            # evaluate the candidate
            cand_score = _eval(f_candidate)
            # keep global best
            if cand_score < best_score:
                best_score = cand_score
                best_seq = f_candidate.copy()
                print(f"[Restart {restart:02d} LP {lp_iters:02d}] "

            # accept if improvement, otherwise shrink mixing coefficient
            f_score = _eval(f)
            if cand_score < f_score:

                mix_coef = max(0.1, mix_coef * 0.55)

        # ------------------------------------------------------------------
        # 6. Polishing phase – random pair swaps with linear cooling.
        # ------------------------------------------------------------------
        polish_iters = 4000
        it = 0
        while time.time() < deadline and it < polish_iters:
            it += 1

            sigma = 0.25 * (1.0 - temp_frac) + 0.02 * temp_frac
            cand = _random_pair_move(f, sigma)
            cand_score = _eval(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[Restart {restart:02d} polish {it:04d}] "

    # ------------------------------------------------------------------

    best_seq = np.maximum(best_seq, 0.0)  # safety
    return best_seq.tolist()
