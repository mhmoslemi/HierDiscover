"""
Propose a low‑score step‑function.

The algorithm works in three stages:

1. **Sampling** – for each chosen length `n` we generate a few
   candidate shapes (symmetric Beta densities, random Dirichlet vectors
   and an up‑sampled best‑so‑far shape).  The best of these elementary
   candidates is taken as a starter.

2. **Smooth optimisation** – we minimise a smooth approximation of the
   ``max`` in the objective:
       F_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k) .
   Its gradient can be written as
       ∇F_λ(a) = 2·conv(w, a)  with  w_i = softmax_λ((a*a)_i) .
   An Adam optimiser (with projection onto the simplex after every step)
   is used.  λ is increased slowly so that the optimisation first smooths
   the landscape and finally concentrates on the true peak.

3. **Length escalation** – after a successful optimisation for a given
   length we up‑sample the best shape to the next (larger) length and
   repeat the process.  Larger lengths give a lower theoretical bound
   (≈ 1) while the optimisation from a good initial shape needs only a
   few thousand iterations.

All work is done with NumPy FFTs; the time budget (≈1000 s) is split
automatically over the different lengths and restarts.  The best sequence
found before the deadline is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the grid x_i = (i+½)/n, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (α = ½) on n points, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float = 0.5) -> np.ndarray:
    """Draw a random point on the simplex via Gamma(α,1) with concentration `conc`."""
    g = np.random.gamma(conc, 1.0, size=n)
    if g.sum() == 0.0:
        g = np.ones(n)
    return g / g.sum()


# ----------------------------------------------------------------------
def _interpolate_seq(old_seq: np.ndarray, n_new: int) -> np.ndarray:
    """Upsample an old sequence to a new length by linear interpolation."""
    n_old = len(old_seq)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    new_seq = np.interp(x_new, x_old, old_seq)
    new_seq /= new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray, m: int) -> np.ndarray:
    """Full linear convolution a*a using FFT of length m (m ≥ 2·len(a)-1)."""
    A = np.fft.rfft(a, m)
    c = np.fft.irfft(A * A, m)
    return c[: 2 * len(a) - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """Compute 2·Σ_k w_k·a_{k‑i} (correlation) using FFT of length m."""
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start : start + len(a)]


# ----------------------------------------------------------------------
def _score_from_seq(a: np.ndarray, m: int) -> float:
    """Exact score 2·n·max(conv(a,a)) for a normalised to sum = 1."""
    n = len(a)
    b = _conv_full_fft(a, m)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _optimise_adam(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 20000,
    alpha0: float = 0.10,
    lambda0: float = 0.5,
    lambda_max: float = 50.0,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Refine a normalised sequence using an Adam optimiser on the
    smooth log‑sum‑exp approximation of the max‑convolution peak.
    Returns the best sequence seen and its exact (unsmoothed) score.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()                       # safety normalisation

    m = 1 << ((3 * n - 2).bit_length())   # FFT length (≥ (2n‑1)+(n‑1))

    best_seq = a.copy()
    best_score = _score_from_seq(a, m)

    # Adam state
    mt = np.zeros_like(a)
    vt = np.zeros_like(a)

    lamb = lambda0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- evaluate exact score (for bookkeeping) ----
        cur_score = _score_from_seq(a, m)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] n={n:5d} it={it:5d} score={best_score:.6f}"
            )

        # ---- compute convolution and soft‑max weights ----
        A_fft = np.fft.rfft(a, m)
        b_full = np.fft.irfft(A_fft * A_fft, m)[: 2 * n - 1]
        max_b = b_full.max()

        # soft‑max (stable)
        exp_arg = lamb * (b_full - max_b)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b_full, 1.0 / b_full.size)
        else:
            w = w / w_sum

        # ---- gradient w.r.t. a (via correlation) ----
        g = _gradient_via_correlation(w, a, m)

        # ---- Adam update ----
        mt = beta1 * mt + (1.0 - beta1) * g
        vt = beta2 * vt + (1.0 - beta2) * (g * g)

        mt_hat = mt / (1.0 - beta1 ** it)
        vt_hat = vt / (1.0 - beta2 ** it)

        step = alpha0 / (np.sqrt(it) + 1.0)

        a = a - step * mt_hat / (np.sqrt(vt_hat) + eps)

        # ---- project back onto the simplex ----
        a = _project_onto_simplex(a, s=1.0)

        # ---- increase λ slowly ----
        lamb = min(lambda_max, lamb * 1.015)

        # ---- occasional jitter to escape tiny plateaus ----
        if it % 1500 == 0:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score

            break

        print(f"\n=== Optimising length {n} ===")

        # ----- a) generate a collection of start vectors -----
        starters = []

        # 1. up‑sample previous best (if any)
        if prev_best_seq is not None:
            up = _interpolate_seq(prev_best_seq, n)
            starters.append(up)

        # 2. symmetric Beta sweeps (few α values)
        alphas = np.concatenate(
            [
                np.array([0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0]),
                np.random.uniform(0.2, 2.0, size=6),
            ]
        )
        for α in alphas:
            starters.append(_beta_distribution(n, α))

        # 3. random Dirichlet draws
        for _ in range(4):
            starters.append(_random_dirichlet(n, conc=0.6))

        # Keep only a manageable number (avoid too many FFTs)
        starters = starters[:12]

        # ----- b) keep the best starter according to the exact score -----
        best_start = None
        best_start_score = float("inf")
        m_fft = 1 << ((3 * n - 2).bit_length())
        for s in starters:
            sc = _score_from_seq(s, m_fft)
            if sc < best_start_score:
                best_start_score = sc
                best_start = s

        print(f"Best start score (exact) = {best_start_score:.6f}")

        # ----- c) perform several Adam restarts -----
        # Allocate a slice of the remaining time for this length.
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # Heuristic: give longer runs to larger n
        time_per_restart = max(5.0, remaining / (len(length_schedule) * 3))

        for attempt in range(3):
            if time.time() >= deadline:
                break

            # Initialise from the best start (perturbed a little)
            init = best_start.copy()
            if attempt > 0:
                # add a tiny random perturbation to diversify
                init += np.random.randn(n) * 1e-4
                init = np.clip(init, 0.0, None)
                init /= init.sum()

            sub_deadline = min(deadline, time.time() + time_per_restart)

            refined_seq, refined_score = _optimise_adam(
                n=n,
                init_seq=init,
                deadline=sub_deadline,
                max_iters=20000,
                alpha0=0.12,
                lambda0=0.5,
                lambda_max=30.0,
            )

            if refined_score < overall_best_score:
                overall_best_score = refined_score
                overall_best_seq = refined_seq.copy()
                print(
                    f"*** New global best → length={n}  score={overall_best_score:.6f}"
                )

        # store the best for the next size
        if overall_best_seq is not None and len(overall_best_seq) == n:
            prev_best_seq = overall_best_seq.copy()
        else:
            # up‑sample whatever we have
            if overall_best_seq is not None:
                prev_best_seq = _interpolate_seq(overall_best_seq, n)
            else:
                prev_best_seq = _arcsine_distribution(n)

        # early exit if we already beat the target
        if overall_best_score <= 1.503:
            print("Target reached – exiting early.")
            break

    # ------------------------------------------------------------------
    # Fallback – ensure something is returned
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = _score_from_seq(overall_best_seq, 1 << ((3 * n - 2).bit_length()))
        print(f"Fallback sequence produced score {overall_best_score:.6f}")

    print(
        f"\nFinal best (exact) score: {overall_best_score:.6f}  length={len(overall_best_seq)}"
    )

    # Return a plain Python list as required by the runner
    return overall_best_seq.tolist()
