"""
Improved step‑function search.

The evaluation metric after normalising Σa = 1 is
        score = 2·n·max(a*a) .
Hence we must minimise the peak of the self‑convolution.

Approach
--------

1. **Generation of starter shapes**
   - Symmetric Beta(α,α) densities on a uniform grid for a wide
     range of α (including very small values ≪ 0.5 to obtain
     strongly edge‑biased shapes).
   - Random Dirichlet draws (uniform on the simplex) for diversity.
   - The historic sequence `height_sequence_1` is also tried if present.

2. **Smooth max optimisation**
   - We minimise a smooth approximation of the max–operator:
         Fλ(a) = max(b) + (1/λ)·log Σk exp(λ·(b_k–max(b))) ,
     where b = a*a (full linear convolution).
   - The gradient of Fλ w.r.t. a is 2·corr(w,a) with
         w_k = softmax(λ·b_k) .
     Correlation is evaluated by a single FFT, giving O(n log n) per
     iteration.
   - An exponentiated‑gradient (mirror‑descent on the simplex) update
     is performed:
         a ← a·exp(−η·∇Fλ) ,   a ← a / Σa .
   - The temperature λ is increased geometrically from a modest start
     (λ≈1) up to a large value (λ≈200) so that the optimisation first
     smooths the objective and finally concentrates on the true peak.
   - The step‑size η decays as η₀/√t, with η₀ chosen generously.
   - Random jitter is injected occasionally to escape tiny plateaux.

3. **Multi‑restart, multi‑length search**
   - Several candidate lengths (2 k … 32 k) are examined; for each length
     the best few initialisations are refined.
   - The whole budget (≈1000 s) is split dynamically; as soon as a
     sequence with score ≤ 1.503 is found the search terminates early.

All helper functions are top‑level, there are no closures or lambdas,
and the only external import is NumPy (and `time`).  The routine
returns a plain Python list suitable for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on grid x_i = (i+½)/n, normalised to Σ=1.
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a power‑of‑two FFT.
    Returned length = 2·len(a)‑1.
    """
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, m)
    c = np.fft.irfft(A * A, m)
    return c[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k‑i} (correlation) using FFT.
    w – weight vector of length 2·n‑1,
    a – current sequence (length n),
    m – FFT length (power of two, ≥ len(w)+len(a)-1).
    Returns the slice g[0…n‑1].
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start:start + len(a)]


# ----------------------------------------------------------------------
def _optimise_smooth(
    n: int,
    deadline: float,
    init_seq: np.ndarray,
    max_iters: int = 20000,
    eta0: float = 0.5,
    lambda0: float = 1.0,
    lambda_max: float = 200.0,
    jitter_prob: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Refine a normalised start sequence by minimising a smoothed max‑convolution.
    Returns the best sequence observed (still normalised to sum=1) and its
    unsmoothed score 2·n·max(conv).
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = float(2.0 * n * _conv_full_fft(a).max())

    # FFT size sufficient for correlation w (len=2n‑1) with a (len=n)
    m = 1 << ((3 * n - 2).bit_length())   # ≥ (2n‑1)+(n‑1)

    lamb = lambda0
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current score (unsmoothed)
        b = _conv_full_fft(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} it={it:5d} score={best_score:.6f}")

        # ----- smooth‑max (soft‑max) -----
        maxb = b.max()
        exp_arg = lamb * (b - maxb)          # ≤ 0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:                     # numerically impossible with our scaling,
            w = np.full_like(b, 1.0 / b.shape[0])
        else:
            w /= w_sum

        # ----- gradient (correlation) -----
        g = _gradient_via_correlation(w, a, m)

        # ----- exponentiated‑gradient step -----
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)             # stays non‑negative
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum

        # occasional jitter to escape plateaus
        if np.random.rand() < jitter_prob and it % 200 == 0:
            a += np.random.randn(n) * 5e-5
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        # increase temperature λ geometrically
        lamb = min(lambda_max, lamb * 1.02)

    return best_seq, best_score

        # b) smooth optimisation from each starter
        # allocate a slice of the remaining time for this length
        slice_deadline = time.time() + max(0.5, (deadline - time.time()) * 0.5)

        for idx, starter in enumerate(top_initial, 1):
            if time.time() >= slice_deadline:
                break
            print(f"-- Run {idx}/{len(top_initial)} for n={n}")
            refined_seq, refined_score = _optimise_smooth(
                n=n,
                deadline=slice_deadline,
                init_seq=starter,
                max_iters=15000,
                eta0=0.5,
                lambda0=1.0,
                lambda_max=200.0,
                jitter_prob=0.09,
            )
            if refined_score < overall_best_score:
                overall_best_score = refined_score
                overall_best_seq = refined_seq.copy()
                print(f"*** New global best → n={n} score={overall_best_score:.6f}")

            # early exit if we have already reached the required target
            if overall_best_score <= 1.503:
                break
        if overall_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 2️⃣  Fallback – guarantee a result
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = 2.0 * n * _conv_full_fft(overall_best_seq).max()
        print("Fallback uniform sequence used.")

    print(
        f"\nFinal best (unsmoothed) score: {overall_best_score:.6f}  length={len(overall_best_seq)}"
    )
    return overall_best_seq.tolist()
