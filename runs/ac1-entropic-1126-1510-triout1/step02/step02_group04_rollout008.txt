def simplex_projection(v):
    # returns projection onto simplex {x >=0, sum x = 1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v):
    # v is 1-d array; projects onto simplex {x >= 0, sum x = 1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Sub‑gradient / exponentiated‑gradient optimiser for the “minimise the
self‑convolution peak” problem.

The algorithm works entirely on the probability simplex (non‑negative,
sum = 1).  Starting from either the previously‑found construction (if
available) or a uniform distribution, it repeatedly

1. computes the linear convolution a∗a by FFT,
2. extracts the few lags with the largest convolution values,
3. builds the sub‑gradient of the maximal entry (the sum of the
   shifted copies of a that contribute to those lags),
4. performs an exponentiated‑gradient step
   a ← a·exp(−η·grad)  and renormalises back onto the simplex,
5. adds a tiny random jitter to keep the search stochastic,
6. occasionally mirrors the sequence (a←½(a+a[::-1])) and restarts
   from a fresh random point if no improvement has been seen for a
   while.

A simple decreasing schedule η = η₀/√t is used.  The score is
computed analytically as 2·n·max(a∗a) (the denominator of the official
`evaluate_sequence` equals 1 because we stay on the simplex).  The best
sequence seen during the run is returned as a plain Python list.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper: fast linear convolution using a power‑of‑two FFT
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a*a (length 2·len(a)-1)."""
    n = a.size
    m = 2 * n - 1
    # next power of two ≥ m  (FFT size)
    size = 1 << ((m - 1).bit_length())
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)[:m]
    return conv


# ----------------------------------------------------------------------
# Helper: normalise a non‑negative vector onto the probability simplex
# ----------------------------------------------------------------------
def _normalise_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x≥0, sum x = 1}.  Assumes v is non‑negative."""
    s = v.sum()
    if s <= 0.0:
        # fallback – uniform distribution
        v = np.full_like(v, 1.0 / v.size)
    else:
        v = v / s
    return v


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Gradient‑based optimiser for the step‑function problem.

    Returns a list of non‑negative heights that (approximately) minimise
    `evaluate_sequence`.  The routine runs for at most `budget_s` seconds.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + max(0.0, budget_s - 0.5)

    # --------------------------------------------------------------
    # Choose a problem size (number of coefficients)
    # --------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        n = base.size
        # normalise onto the simplex (the score is scale‑invariant)
        a = _normalise_to_simplex(np.maximum(base, 0.0))
    except Exception:
        # fallback – a reasonable length, uniform start
        n = 8000
        a = np.full(n, 1.0 / n, dtype=float)

    # --------------------------------------------------------------
    # Optimisation parameters
    # --------------------------------------------------------------
    eta0 = 0.02            # initial step size (will be divided by √t)
    top_k = 7              # how many largest convolution lags to use
    jitter_scale = 1e-6    # magnitude of random jitter added each step
    restart_patience = 3000  # generations without improvement before a restart
    symmetrise_every = 500   # apply a↔reverse averaging periodically

    best_a = a.copy()
    conv = _conv_fft(a)
    best_score = 2.0 * n * conv.max()
    no_improve = 0
    iteration = 0
    last_report = 0

    print(f"[Init] n = {n}, initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ----- convolution and current objective -----
        conv = _conv_fft(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[Iter {iteration:6d}] New best score = {best_score:.6f}")
        else:
            no_improve += 1

        # ----- identify the most offending lags (sub‑gradient) -----
        # take the `top_k` entries of conv (ties broken arbitrarily)
        if top_k >= conv.size:
            top_idxs = np.arange(conv.size)
        else:
            top_idxs = np.argpartition(-conv, range(top_k))[:top_k]

        # compute sub‑gradient of the max‑entry (sum of shifted copies)
        grad = np.zeros_like(a)
        idx = np.arange(n)
        for lag in top_idxs:
            j = lag - idx
            mask = (j >= 0) & (j < n)
            grad[mask] += 2.0 * a[j[mask]]

        # ----- exponentiated‑gradient step -----
        eta = eta0 / np.sqrt(iteration)          # decreasing step size
        a = a * np.exp(-eta * grad)              # multiplicative update
        # tiny random jitter to keep the dynamics stochastic
        a += jitter_scale * (np.random.rand(n) - 0.5)
        # re‑normalise onto the simplex
        a = _normalise_to_simplex(np.maximum(a, 0.0))

        # ----- occasional symmetrisation (helps flatten the autocorrelation) -----
        if iteration % symmetrise_every == 0:
            a = _normalise_to_simplex(0.5 * (a + a[::-1]))

        # ----- stagnation restart -----
        if no_improve >= restart_patience:
            a = _normalise_to_simplex(np.random.rand(n))
            no_improve = 0
            print(f"[Iter {iteration:6d}] Restart due to stagnation")

        # ----- lightweight progress report (once per second) -----
        if time.time() - last_report > 1.0:
            last_report = time.time()
            # (no extra work – just a silent heartbeat)

    # --------------------------------------------------------------
    # Return the best sequence discovered
    # --------------------------------------------------------------
    print(f"[Done] Best score after {iteration} iterations = {best_score:.6f}")
    return best_a.tolist()
