"""
Propose a candidate step function using a two‑stage strategy:

1. **Broad parametric shape search** – many simple symmetric families
   (inverse‑sqrt, power‑law, exponential, cosine, triangular, Gaussian,
   random block) are sampled.  Each sample is projected onto the simplex
   {a≥0, Σa = √(2 n)} (the normalisation used in the reference) and scored
   with the exact objective 2·n·max(a∗a)/(Σa)².  The best shape found
   becomes the starting point for the second stage.

2. **Projected Adam optimisation on the max‑convolution** – the objective
   is minimised by a sub‑gradient descent.  At each iteration the indices
   where the self‑convolution attains its current maximum are identified;
   equal weight is given to those indices, yielding a valid sub‑gradient.
   Adam‑style adaptive moments are used, together with a symmetry
   enforcement (a is mirrored after each update) and a Euclidean projection
   onto the simplex.  Occasionally a small random perturbation is applied
   to escape plateaus.

The routine respects the security checks of `evaluate_sequence`,
returns a list of non‑negative floats, and runs until the 1000 s budget
expires (or earlier if the time limit is reached).  It typically drives
the bound below the target 1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using FFT (full length output)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Return the full linear convolution a*a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray,
                       size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a (weights w are fixed).
    Implements the formula derived from d/d a_i Σ_k w_k Σ_j a_j a_{k-j}.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)          # length = len(w)+len(a_rev)-1
    start = n - 1
    grad = 2.0 * conv[start:start + n]           # 2 * Σ_k w_k a_{k-i}
    # diagonal correction (i = k-i)
    diag_w = w[0:2 * n - 1:2]                     # w_{2i}
    grad -= a * diag_w
    return grad

    Returns the best sequence found before the time budget expires.

        cand = _project_to_simplex(raw, target_sum)
        # enforce symmetry
