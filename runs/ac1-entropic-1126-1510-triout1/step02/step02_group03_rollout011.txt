       maxb = b.max()
       # weight for max indices
       max_idxs = np.where(b >= maxb - 1e-12)[0]

def _adam_optimize(n, init_seq, deadline, max_iters=2000, eta=0.1, lambda0=0.5, lambda_max=200, lambda_factor=1.02):
    a = init_seq.astype(np.float64)
    a_sum = a.sum()
    if a_sum <= 0: a = np.full(n, 1.0/n)
    else: a = a / a_sum
    y = np.log(a + 1e-12)  # shape (n)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9; beta2 = 0.999; eps = 1e-8
    best_seq = a.copy()
    best_score = _score(a)
    lambda_cur = lambda0
    # compute FFT length for correlation and maybe conv
    m_fft = 1 << ((3 * n - 2).bit_length())
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # compute a from y
        maxy = y.max()
        expy = np.exp(y - maxy)
        a = expy / expy.sum()
        # compute convolution
        b = _conv_full_fft(a)   # length 2n-1
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            # print progress
            print(f"[{time.time():.1f}] n={n:5d} it={it:5d} score={cur_score:.6f} best={best_score:.6f}")
        # compute smoothed weights
        maxb = cur_max
        # Avoid overflow: use maxb shift
        w = np.exp(lambda_cur * (b - maxb))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.shape[0])
        else:
            w /= w_sum
        # gradient w.r.t a
        g_a = _gradient_via_correlation(w, a, m_fft)
        # gradient w.r.t y
        # compute dot = np.dot(g_a, a)
        dot_ga_a = np.dot(g_a, a)
        g_y = a * (g_a - dot_ga_a)
        # Adam updates
        m = beta1 * m + (1 - beta1) * g_y
        v = beta2 * v + (1 - beta2) * (g_y ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        # step
        y = y - eta * m_hat / (np.sqrt(v_hat) + eps)
        # lambda schedule
        lambda_cur = min(lambda_max, lambda_cur * lambda_factor)
        # optional jitter
        if it % 500 == 0 and np.random.rand() < 0.1:
            y += np.random.randn(n) * 1e-4
    # after loop, optionally do subgradient refinement
    # compute leftover time
    time_left = deadline - time.time()
    if time_left > 0.5:
        # refine
        ref_seq, ref_score = _subgradient_refine(best_seq, n, deadline, max_iters=2000, eta0=0.05)
        if ref_score < best_score:
            best_seq = ref_seq
            best_score = ref_score
    return best_seq, best_score

def _upscale(seq, new_n):
    n = len(seq)
    x_old = np.arange(n) / (n - 1) if n > 1 else np.array([0])
    x_new = np.arange(new_n) / (new_n - 1)
    vals = np.interp(x_new, x_old, seq)
    # adjust due to sum? Since interpolation preserves integral maybe not, but we can renormalize
    vals = np.maximum(vals, 0)
    vals_sum = vals.sum()
    if vals_sum == 0:
        return np.full(new_n, 1.0 / new_n)
    else:
        return vals / vals_sum

"""
Propose a step‑function sequence with a low evaluate_sequence score.

Algorithm
---------
1.  Try to reuse a historic construction (global variable `height_sequence_1`).
2.  Sweep several lengths (2000 … 20000).  For each length we generate
    a handful of symmetric starters: arcsine law, several Beta(α,α)
    densities (α∈[0.3,1.5]), random Dirichlet draws and – if available –
    an up‑scaled version of the current global best.
3.  Each starter is refined by an Adam‑type mirror descent on the
    simplex.  The objective is a smooth log‑sum‑exp approximation of the
    peak of the self‑convolution; the smoothing parameter λ is increased
    slowly so that the method finally optimises the true max‑function.
    All updates are performed in the log‑softmax (entropy‑mirror) space,
    guaranteeing non‑negative entries that sum to one.
4.  When λ is large the soft‑max weight collapses onto the active
    convolution indices, i.e. we are effectively performing sub‑gradient
    descent on the original max‑objective.  A short sub‑gradient phase
    with a decreasing step size is applied after the Adam loop.
5.  The best sequence over all lengths and initialisations is returned.
   The implementation relies only on NumPy and respects the 1000 s
   time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the grid x_i = (i+½)/n (ℓ₁‑normalised)."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (α=½) on n points (ℓ₁‑normalised)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT (real → rfft)."""
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())          # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, m)
    c = np.fft.irfft(A * A, m)
    return c[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g = 2·Σ_k w_k·a_{k‑i} (correlation) using FFT.
    w – weight vector (length 2·n‑1), a – current sequence (length n),
    m – FFT length (power of two, at least len(w)+len(a)-1).
    Returns g of length n.
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start:start + len(a)]


# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Exact evaluate_sequence score (without the evaluate_sequence wrapper)."""
    n = len(a)
    b = _conv_full_fft(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
def _upscale(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a longer length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    # interpolate on the unit interval [0,1]
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    vals = np.interp(x_new, x_old, seq)
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s == 0.0:
        return np.full(new_n, 1.0 / new_n, dtype=np.float64)
    return vals / s


# ----------------------------------------------------------------------
def _adam_optimize(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 4000,
    eta: float = 0.08,
    lambda0: float = 0.1,
    lambda_max: float = 200.0,
    lambda_factor: float = 1.02,
) -> tuple[np.ndarray, float]:
    """
    Refine `init_seq` (already ℓ₁‑normalised) with Adam on the
    smooth‑max objective.  Returns the best sequence seen and its
    exact (unsmoothed) score.
    """
    a = init_seq.astype(np.float64).copy()
    a_sum = a.sum()
    if a_sum <= 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= a_sum

    # log‑softmax parameters (entropy mirror)
    y = np.log(a + 1e-12)
    m_adam = np.zeros_like(y)
    v_adam = np.zeros_like(y)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    best_seq = a.copy()
    best_sc = _score(a)

    # FFT length for correlation (w length = 2n‑1, a length = n)
    m_fft = 1 << ((3 * n - 2).bit_length())

    lam = lambda0
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- recompute a from y (softmax) and enforce exact sum = 1 -----
        maxy = y.max()
        expy = np.exp(y - maxy)
        a = expy / expy.sum()

        # ----- evaluate current (exact) score -----
        cur_sc = _score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] n={n:5d} it={it:5d} score={cur_sc:.6f} best={best_sc:.6f}"
            )
            # early exit if we have hit the target bound
            if best_sc <= 1.503:
                break

        # ----- smooth max (log‑sum‑exp) -----
        b = _conv_full_fft(a)
        b_max = b.max()
        # safe soft‑max
        w = np.exp(lam * (b - b_max))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.size)
        else:
            w /= w_sum

        # ----- gradient wrt a (correlation) -----
        g_a = _gradient_via_correlation(w, a, m_fft)

        # ----- gradient wrt y (entropy mirror) -----
        dot = np.dot(g_a, a)
        g_y = a * (g_a - dot)

        # ----- Adam update on y -----
        m_adam = beta1 * m_adam + (1.0 - beta1) * g_y
        v_adam = beta2 * v_adam + (1.0 - beta2) * (g_y ** 2)
        m_hat = m_adam / (1.0 - beta1 ** it)
        v_hat = v_adam / (1.0 - beta2 ** it)
        y -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # ----- increase smoothing parameter λ -----
        lam = min(lambda_max, lam * lambda_factor)

        # ----- occasional jitter to escape plateaus -----
        if it % 500 == 0 and np.random.rand() < 0.1:
            y += np.random.randn(n) * 1e-4

    # ----- final sub‑gradient refinement (hard max) -----
    time_left = deadline - time.time()
    if time_left > 1.0:
        best_seq, best_sc = _subgradient_refine(
            best_seq, n, deadline, max_iters=800, eta0=0.05
        )
    return best_seq, best_sc


# ----------------------------------------------------------------------
def _subgradient_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800,
    eta0: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent on the exact max‑convolution objective.
    Uses the current max indices as the active set.
    """
    best_seq = a.copy()
    best_sc = _score(a)

    # FFT length for correlation (same as in Adam)
    m_fft = 1 << ((3 * n - 2).bit_length())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        b = _conv_full_fft(a)
        cur_max = b.max()
        # active set: indices attaining (within tolerance) the maximum
        tol = 1e-12
        active = np.where(b >= cur_max - tol)[0]
        w = np.zeros_like(b)
        w[active] = 1.0 / len(active)

        g = _gradient_via_correlation(w, a, m_fft)

        step = eta0 / np.sqrt(it)
        a = a - step * g
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        cur_sc = _score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] subgrad n={n:5d} it={it:5d} score={cur_sc:.6f}"
            )
            if best_sc <= 1.503:
                break

    return best_seq, best_sc
