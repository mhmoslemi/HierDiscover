def linear_convolve(x, y):
    m = len(x) + len(y) - 1
    N = 1 << ((m - 1).bit_length())
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    Z = X * Y
    conv = np.fft.irfft(Z, N)[:m]
    return conv
```

"""
Propose a step‑function minimising ``evaluate_sequence``.

The optimiser works in two phases:

1. **Soft‑max gradient descent** – the maximal convolution entry is
   approximated by a smooth soft‑max (temperature β).  The gradient of
   the smoothed objective is obtained by a single linear convolution,
   and the iterate is projected onto the probability simplex.
   β is increased gradually so the optimiser finally concentrates on
   the true peak.

2. **Compact (μ/λ)‑ES refinement** – the best vector from phase 1 seeds a
   small evolution‑strategy that mutates the population with Gaussian
   noise, adds two deterministic candidates (a targeted mass move and a
   symmetrised version) and occasionally performs a few projected‑gradient
   steps on the most offending lags.  The ES runs until the time budget
   expires, always keeping the current best solution.

Both phases respect the non‑negative, sum‑to‑one constraints required by
``evaluate_sequence``.  The code uses FFT based convolutions for speed
and works for any length (the historic ``height_sequence_1`` is used as
a starting point when available).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: clip to the simplex (non‑negative entries, sum = 1)
# ----------------------------------------------------------------------
def _clip_and_normalise(v: np.ndarray) -> np.ndarray:
    """Clip negatives to zero and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        v = np.full_like(v, 1.0 / v.size)
    else:
        v = v / s
    return v


# ----------------------------------------------------------------------
# Linear convolution via FFT (returns full linear convolution)
# ----------------------------------------------------------------------
def _linear_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    m = x.size + y.size - 1
    N = 1 << ((m - 1).bit_length())            # next power of two
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    Z = X * Y
    conv = np.fft.irfft(Z, N)[:m]
    return conv


# ----------------------------------------------------------------------
# Fast scoring (same formula as evaluate_sequence but without checks)
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """
    Return the evaluation score for a probability vector ``seq``.
    Equivalent to ``evaluate_sequence(seq.tolist())``.
    """
    n = seq.size
    conv = _linear_convolve(seq, seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Gradient step that reduces the (soft‑max) maximal convolution entry
# ----------------------------------------------------------------------
def _grad_step(a: np.ndarray, beta: float, eta: float):
    """
    One projected gradient descent step.
    ``beta`` is the soft‑max temperature (larger → sharper focus on the
    true maximum).  ``eta`` is the step size.
    Returns the new probability vector and the current maximal convolution.
    """
    n = a.size
    # current convolution and its maximum
    conv = _linear_convolve(a, a)
    maxb = conv.max()

    # soft‑max weights over the convolution entries
    # (subtract max for numerical stability)
    w = np.exp(beta * (conv - maxb))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    # gradient of the smoothed objective:
    #   g = 2 * (a_rev * w)  (restricted to the n original indices)
    a_rev = a[::-1]
    conv_grad = _linear_convolve(a_rev, w)          # length 3n‑2
    grad = 2.0 * conv_grad[n - 1 : n - 1 + n]       # slice of length n

    # gradient step + projection onto simplex
    a_new = a - eta * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new = np.full_like(a, 1.0 / n)
    else:
        a_new = a_new / s
    return a_new, maxb


# ----------------------------------------------------------------------
# Targeted move: shift a small fraction of mass away from the most
#               contributing pair for the currently maximal convolution.
# ----------------------------------------------------------------------
def _targeted_move(a: np.ndarray, frac: float = 0.03) -> np.ndarray:
    """
    Identify the lag with maximal convolution value, locate the pair
    (i, k‑i) contributing most to it and move a fraction of the larger
    entry to a random destination.
    """
    n = a.size
    conv = _linear_convolve(a, a)
    k = int(conv.argmax())

    # contributions for this lag
    i_arr = np.arange(n)
    j_arr = k - i_arr
    mask = (j_arr >= 0) & (j_arr < n)
    prod = np.zeros_like(a)
    prod[mask] = a[i_arr[mask]] * a[j_arr[mask]]

    if not np.any(mask):
        # fallback to a tiny random perturbation
        cand = a + frac * (np.random.rand(n) - 0.5)
        return _clip_and_normalise(cand)

    i_max = int(np.argmax(prod))
    j_max = k - i_max

    src = i_max if a[i_max] >= a[j_max] else j_max
    dst = np.random.randint(n)
    while dst == src:
        dst = np.random.randint(n)

    amount = frac * a[src] * np.random.rand()
    cand = a.copy()
    cand[src] -= amount
    cand[dst] += amount
    return _clip_and_normalise(cand)


# ----------------------------------------------------------------------
# Gradient step that reduces several large convolution peaks at once.
# ----------------------------------------------------------------------
def _multi_lag_gradient_step(a: np.ndarray, eta: float, top_k: int = 3) -> np.ndarray:
    """
    One projected gradient step using the sum of the sub‑gradients of the
    ``top_k`` largest convolution entries.
    """
    n = a.size
    conv = _linear_convolve(a, a)
    top_idxs = np.argpartition(-conv, range(top_k))[:top_k]

    g = np.zeros_like(a)
    idx = np.arange(n)
    for k in top_idxs:
        j = k - idx
        mask = (j >= 0) & (j < n)
        g[mask] += 2.0 * a[j[mask]]

    a_new = a - eta * g
    return _clip_and_normalise(a_new)
