"""
Propose a step‑function sequence minimizing
    2·n·max(conv(a,a)) / (sum(a))²

The algorithm consists of three stages:

1.  **Broad parametric search** – many families (inverse‑sqrt,
    power‑law, exponentials, cosine, beta‑shapes, mixtures and random
    block/Diri‑chlet samples) are generated, projected onto the simplex
    Σa = √(2n) and evaluated.  The best candidate is kept.

2.  **Refinement by Adam‑based projected sub‑gradient descent** – a
    smooth approximation of the max is obtained by a soft‑max with a
    slowly decreasing temperature.  The true sub‑gradient of the weighted
    sum of convolutions is computed analytically via FFT (no diagonal
    correction – this fixes a bug in the previous implementation).  An
    Adam update with projection onto the simplex drives the max‑convolution
    down.  Random perturbations and occasional restarts avoid stagnation.

3.  **Final random‑block polishing** – a short sweep of random block
    candidates tries to improve the best solution found.

The whole routine stops after the allotted time budget (≈ 1000 s) and
returns the best sequence (as a plain Python list).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}.
    """
    if z < 0:
        raise ValueError("target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Linear convolution of u and v using FFT.
    Output length = len(u) + len(v) – 1.
    """
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Return a * a (full linear convolution)."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    The correct analytical expression is
        grad_i = 2 Σ_k w_k a_{k-i},
    which is obtained by a convolution of w with the reversed a.
    No diagonal correction is needed (the previous code subtracted it
    erroneously).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)            # length = len(w)+len(a_rev)-1
    start = n - 1                                   # slice giving Σ_k w_k a_{k-i}
    grad = 2.0 * conv[start:start + n]
    return grad

        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb) + 0.05
        cand = np.zeros(n)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[block] improved best = {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f} (max convolution)")

    # Return as plain list for the evaluator
    return best_a.tolist()
