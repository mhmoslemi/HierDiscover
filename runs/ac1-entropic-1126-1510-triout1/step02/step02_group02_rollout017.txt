"""
Propose a step‑function sequence that minimizes ``evaluate_sequence``.
The algorithm runs in three stages:

1. **Parametric shape search** – random samples from several simple families
   (inverse‑sqrt, power‑law, exponential, cosine, Gaussian, beta‑distribution,
   mixed, random block patterns). Each candidate is projected onto the simplex
   {a≥0, Σa = √(2 n)} and its max‑convolution is measured. The best candidate
   is retained.

2. **Refinement** – starting from the best candidate we minimise the smooth
   objective  Σ_k w_k·(a∗a)_k where the weights ``w`` are a soft‑max of the
   convolution entries.  An Adam‑style projected gradient descent with a
   temperature schedule (soft‑max → max) is used.  After each step the iterate
   is symmetrised (a←(a+a[::-1])/2) and projected back onto the simplex.
   Random perturbations are added periodically to escape plateaus.

3. **Final block search** – a short burst of random block candidates is tried
   to squeeze any remaining improvement.

The returned list is guaranteed to be non‑negative; its length is the same as
the reference sequence ``height_sequence_1`` when available, otherwise 3000.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """ Euclidean projection of v onto {x ≥ 0, Σx = z}. """
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """ Linear convolution of u and v using FFT (full length returned). """
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """ Full convolution a*a (length 2·n‑1). """
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·(a*a)_k with respect to a.
    The formula reduces to 2·conv[:n][::-1] where conv = w * a_rev.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    conv = _conv_fft(w, rev_a, size_fft)          # length = 3·n‑2
    grad = 2.0 * conv[:n][::-1]                   # reverse first n entries
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative sequence with low ``evaluate_sequence`` score."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # problem size
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        have_ref = True
        ref_seq = np.array(height_sequence_1, dtype=float)
    except Exception:
        n = 3000
        have_ref = False
        ref_seq = None

    target_sum = math.sqrt(2.0 * n)          # the normalisation used in the paper
    size_fft = 1 << ((3 * n - 2).bit_length())   # power of two ≥ 3n‑2

    # ------------------------------------------------------------------
    # helper to evaluate a candidate (fast, no security checks)
    def _score(a: np.ndarray) -> float:
        conv = _self_convolution(a, size_fft)
        # because Σa = target_sum, the evaluate_sequence value equals max(conv)
        return float(conv.max())

    # ------------------------------------------------------------------
    # 1. PARAMETRIC SHAPE SEARCH
    best_a = None
    best_score = float("inf")

    # uniform
    a = np.ones(n)
    a = _project_to_simplex(a, target_sum)
    sc = _score(a)
    best_score, best_a = sc, a.copy()
    print(f"[init] uniform score = {sc:.6f}")

    # helper to sample and test a shape
    def _try_candidate(arr: np.ndarray):
        nonlocal best_score, best_a
        arr = _project_to_simplex(arr, target_sum)
        sc = _score(arr)
        if sc < best_score:
            best_score, best_a = sc, arr.copy()
            print(f"[shape] new best = {best_score:.6f}")

    rng = np.random.default_rng(seed)

    # inverse‑sqrt families
    for _ in range(80):
        beta = rng.random() * 10.0
        arr = 1.0 / np.sqrt(np.arange(n) + beta + 1.0)
        _try_candidate(arr)

    # double‑ended inverse‑sqrt
    for _ in range(80):
        beta = rng.random() * 10.0
        i = np.arange(n)
        arr = (1.0 / np.sqrt(i + beta + 1.0) +
               1.0 / np.sqrt((n - 1 - i) + beta + 1.0))
        _try_candidate(arr)

    # power‑law
    for _ in range(80):
        beta = rng.random() * 10.0
        pexp = rng.random() * 1.7 + 0.3
        arr = 1.0 / (np.arange(n) + beta + 1.0) ** pexp
        _try_candidate(arr)

    # exponential decay
    for _ in range(80):
        decay = rng.random() * 0.01 + 0.0005
        arr = np.exp(-decay * np.arange(n))
        _try_candidate(arr)

    # cosine‑based (large at ends, small in middle)
    for _ in range(80):
        p = rng.random() * 5.0 + 0.5
        arr = np.abs(np.cos(np.pi * np.arange(n) / (n - 1))) ** p
        _try_candidate(arr)

    # Gaussian bumps (centered)
    for _ in range(80):
        sigma = rng.random() * (n / 8) + n / 30
        arr = np.exp(-0.5 * ((np.arange(n) - (n - 1) / 2) / sigma) ** 2)
        _try_candidate(arr)

    # Beta‑distribution shape on [0,1]
    for _ in range(80):
        a_alpha = rng.random() * 4.0 + 0.5
        b_beta = rng.random() * 4.0 + 0.5
        x = np.linspace(0, 1, n)
        arr = (x ** a_alpha) * ((1 - x) ** b_beta)
        _try_candidate(arr)

    # mixture of uniform and inverse‑sqrt
    for _ in range(80):
        beta = rng.random() * 10.0
        lam = rng.random()
        arr = lam * np.ones(n) + (1.0 - lam) * (
            1.0 / np.sqrt(np.arange(n) + beta + 1.0))
        _try_candidate(arr)

    # random block candidates (few large blocks)
    for _ in range(200):
        nb = rng.integers(2, 7)                     # 2‑6 blocks
        cuts = np.sort(rng.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = rng.random(nb) + 0.1
        cand = np.zeros(n)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        _try_candidate(cand)

    print(f"[shape search] best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. REFINEMENT VIA SMOOTHED GRADIENT DESCENT (Adam)
    a = best_a.copy()
    conv = _self_convolution(a, size_fft)
    # initial temperature – a fraction of the spread of the convolution
    T = max(1e-3, (conv.max() - conv.min()) * 0.5)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    beta1_pow, beta2_pow = beta1, beta2
    eps = 1e-8
    lr0 = 0.5 * target_sum / n          # base learning rate

    it = 0
    no_improve = 0

    while time.time() < deadline and it < 3_000_000:
        # evaluate current iterate
        conv = _self_convolution(a, size_fft)
        max_conv = conv.max()
        cur_score = float(max_conv)   # equal to evaluate_sequence because Σa = √(2n)

        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            no_improve = 0
            print(f"[{it:07d}] new best = {best_score:.6f}")
        else:
            no_improve += 1

        # --- soft‑max weights (smooth max) ---
        shifted = conv - max_conv                     # ≤ 0
        T = max(T, 1e-12)                             # avoid division by zero
        exp_arg = shifted / T
        exp_arg = np.clip(exp_arg, -50, 0)            # stability
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # --- gradient of the smoothed objective ---
        grad = _weighted_gradient(a, w, size_fft)

        # --- Adam update ---
        it += 1
        m = beta1 * (m if it > 1 else np.zeros_like(a)) + (1 - beta1) * grad
        v = beta2 * (v if it > 1 else np.zeros_like(a)) + (1 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step

        # symmetrisation (helps to keep the solution balanced)
        a_new = (a_new + a_new[::-1]) * 0.5

        # project back onto simplex
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional random perturbation to escape plateaus
        if (it % 5000 == 0) or (no_improve > 5000):
            noise = rng.standard_normal(n) * target_sum * 1e-3
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)
            no_improve = 0
            print(f"[{it:07d}] random perturbation")

        a = a_new

        # geometric cooling of the temperature
        T *= 0.9999

        # update bias‑correction factors
        beta1_pow *= beta1
        beta2_pow *= beta2

    # ------------------------------------------------------------------
    # 3. FINAL RANDOM BLOCK SEARCH
    for _ in range(200):
        nb = rng.integers(2, 7)
        cuts = np.sort(rng.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = rng.random(nb) + 0.1
        cand = np.zeros(n)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = _score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[block] improved to {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f} after {it} iterations")
    return best_a.tolist()
