def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time, numpy as np

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2

    # choose length based on time left after init (maybe 1 sec)
    # For now set n = 4000 or a bit less.
    candidate_lengths = [2000, 3000, 4000, 5000]
    best_seq = None; best_score = float('inf'); best_n=None

    for n in candidate_lengths:
        if time.time() >= deadline: break
        seq = arcsine_sequence(n) # returns sum=1
        score = 2.0 * n * np.max(np.convolve(seq, seq))
        if score < best_score:
            best_score = score
            best_seq = np.array(seq)
            best_n = n
        print(f"[init len {n}] score {score:.6f}")

    # optionally incorporate height_sequence_1 (if exists)
    try:
        global height_sequence_1
        if isinstance(height_sequence_1, (list, np.ndarray)):
            s = np.asarray(height_sequence_1, dtype=float)
            if s.size>0:
                s = np.maximum(s, 0.0)
                total=s.sum()
                if total>0:
                    s = s/total
                n=s.shape[0]
                score = 2.0 * n * np.max(np.convolve(s, s))
                if score < best_score:
                    best_score=score; best_seq=s; best_n=n
                print(f"[global] n={n}, score={score:.6f}")
    except Exception:
        pass

    # Ensure best_seq exists
    if best_seq is None:
        best_n = 4000
        best_seq = np.asarray(arcsine_sequence(best_n), dtype=float)
        best_score = 2.0 * best_n * np.max(np.convolve(best_seq, best_seq))

    # Adam optimizer hyperparameters
    a = best_seq.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05  # initial learning rate
    # schedule
    decay_lr = 0.9995
    beta_soft = 2.0
    beta_factor = 1.02
    max_beta = 200.0
    it = 0
    while time.time() < deadline:
        it += 1
        # compute conv
        conv = np.convolve(a, a)
        M = conv.max()
        # compute softmax weights
        exp_vals = np.exp(beta_soft * (conv - M))
        w = exp_vals / exp_vals.sum()
        # compute gradient
        grad = 2.0 * np.convolve(w, a[::-1])[n-1:2*n-1]
        # adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # projection onto simplex (sum=1)
        a_new = project_simplex(a_new, s=1.0)
        a = a_new
        # compute score
        conv = np.convolve(a, a)
        M = conv.max()
        score = 2.0 * n * M
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[iter {it}] new best score {score:.6f} (beta={beta_soft:.2f})")
        # update schedules
        lr *= decay_lr
        beta_soft = min(max_beta, beta_soft * beta_factor)
        # occasionally random perturbation
        if it % 1000 == 0 and time.time() < deadline - 0.5:
            # random shuffle small mass
            for _ in range(5):
                i, j = np.random.randint(0, n, size=2)
                delta = 0.001 * np.random.rand()
                if a[i] >= delta:
                    a[i] -= delta
                    a[j] += delta
            a = project_simplex(a, s=1.0)
    # final scaling to ensure sum close to 1
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = best_seq / best_seq.sum()

    # final evaluation
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score {final_score:.6f} (length {len(best_seq)})")
    except Exception:
        final_score = None
        print("Evaluation failed.")

    return best_seq.tolist()

"""
Propose a step‑function sequence that minimizes the evaluation metric.

Algorithm
---------

1. **Initialisation** – try a few lengths (2000 … 5000) and start from the
   normalised “arcsine’’ distribution (C_i = binom(2i,i)/4^i).  Keep the best
   among them and also the global variable ``height_sequence_1`` if it exists.

2. **Projected Adam optimisation** – the objective for a unit‑sum sequence
   `a` is `score = 2·n·max(conv(a,a))`.  The max is smoothed with a
   soft‑max (log‑sum‑exp) controlled by a temperature `beta`.  The gradient
   of the smoothed max is `grad = 2 * conv(w, a[::-1])` where `w` are the
   soft‑max weights.  Adam (with bias‑correction) is used for the update,
   followed by a Euclidean projection onto the probability simplex
   (non‑negative, sum = 1).

3. **Schedule** – start with a modest learning rate and `beta≈2`.  After each
   iteration `beta` is multiplied by 1.02 (capped at 200) to make the
   approximation sharper, while the learning rate decays slightly.  Occasionally
   a tiny random mass transfer is performed to help escape flat regions.

4. **Book‑keeping** – the best sequence (lowest true `evaluate_sequence`
   score) found so far is stored.  The loop stops before the time budget
   expires and the best sequence is returned.

The method stays completely in‑memory, uses only NumPy, and respects the
1000‑second limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> list[float]:
    """Return the normalised (sum = 1) arcsine coefficients for i=0..n‑1.
    Recurrence: C_{i+1}=C_i·(2i+1)/(2(i+1)).
    """
    seq = []
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2.0 * i + 1.0) / (2.0 * (i + 1))
    seq = np.array(seq, dtype=float)
    seq /= seq.sum()                # L1‑normalise to 1
    return seq.tolist()


# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σ x = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("target sum s must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence with a low evaluation
    score within the given time budget (seconds).  Returns the best list of
    coefficients found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # safety margin

    best_seq = None
    best_score = float('inf')
    best_n = None

    # -------------------------------------------------------------
    # 1) Choose a promising length and starting point
    candidate_lengths = [2000, 3000, 4000, 5000]
    for n in candidate_lengths:
        if time.time() >= deadline:
            break
        seq = np.asarray(arcsine_sequence(n), dtype=float)   # sum = 1
        conv = np.convolve(seq, seq)
        score = 2.0 * n * conv.max()                         # exact score for sum=1
        print(f"[init] n={n:4d}  score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            best_n = n

    # -------------------------------------------------------------
    # 2) If a previous global solution exists, try it as well
    try:
        global height_sequence_1
        if isinstance(height_sequence_1, (list, tuple, np.ndarray)):
            seq = np.asarray(height_sequence_1, dtype=float)
            if seq.size > 0:

                if total > 0:

                conv = np.convolve(seq, seq)
                score = 2.0 * n * conv.max()
                print(f"[global] n={n:4d}  score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    best_n = n
    except Exception:
        pass

    # -------------------------------------------------------------
    # Fallback – should never be needed
    if best_seq is None:

        print(f"[fallback] n={best_n}  score={best_score:.6f}")

    # -------------------------------------------------------------
    # 3) Projected Adam optimiser on the smoothed max‑convolution objective
    a = best_seq.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05           # initial learning rate
    decay_lr = 0.9995   # slow decay

    # Soft‑max temperature schedule
    beta_soft = 2.0
    beta_factor = 1.02
    max_beta = 200.0

    it = 0
    while time.time() < deadline:
        it += 1

        # convolution a * a
        conv = np.convolve(a, a)
        M = conv.max()

        # soft‑max weights over the full conv vector
        # shift by M to avoid overflow (exp(β*(x‑M)) ≤ 1)

        w_sum = w.sum()

        else:
            w /= w_sum

        # gradient of the smoothed max
        grad = 2.0 * np.convolve(w, a[::-1])[n - 1:2 * n - 1]

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = project_simplex(a_new, s=1.0)

        a = a_new

        # exact score for the current iterate (sum=1 ⇒ score = 2·n·max(conv))
        conv = np.convolve(a, a)
        M = conv.max()
        cur_score = 2.0 * n * M

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[iter {it:5d}] new best score {cur_score:.6f} (β={beta_soft:.2f})")

        # schedule updates
        lr *= decay_lr
        beta_soft = min(max_beta, beta_soft * beta_factor)

        # occasional tiny random mass transfer to escape plateaus
        if it % 1000 == 0:

                    a[j] += delta

    except Exception as exc:
