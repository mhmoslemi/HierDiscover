"""
Propose a low‑score step‑function.

The objective (after normalising Σa = 1) is

    score = 2·n·max_k (a*a)_k .

We try to *flatten* the self‑convolution, i.e. make the peak as close
as possible to the uniform value 1/(2n‑1).  The algorithm works as
follows:

1.  For several lengths `n` we generate a few deterministic starters:
    – the discrete arcsine law,
    – a dense sweep of symmetric Beta(α,α) densities,
    – a few random Dirichlet samples.
2.  Each starter is refined by a *mirror‑descent* (exponentiated‑gradient)
    loop that directly minimises the quantity  max(b) – min(b) where
    b = a*a.  The sub‑gradient of the max‑term and the min‑term are
    obtained in closed form by a single FFT‑based correlation.
    A temperature parameter `lam` is increased slowly so that the algorithm
    first smooths the max/min and finally concentrates on the truly
    extreme entries.
3.  The best sequence found for any length is kept.  A short final
    smoothing step (soft‑max of the peak) is applied to polish the
    result.
The whole search respects the 1000 s budget and always returns a
non‑negative simplex‑normalised sequence.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete symmetric arcsine law on n points, Σ=1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the grid x_i = (i+½)/n, Σ=1."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())       # next power of two
    A = np.fft.rfft(a, m)
    c = np.fft.irfft(A * A, m)
    return c[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k‑i} (correlation) using FFT.
    w – weight vector of length 2·n‑1, a – current sequence (length n),
    m – FFT length (power of two, ≥ len(w)+len(a)-1).
    Returns g[0…n‑1].
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)          # correlation
    start = a.shape[0] - 1
    return 2.0 * corr[start:start + a.shape[0]]


# ----------------------------------------------------------------------
def _optimise_flatten(
    n: int,
    deadline: float,
    init_seq: np.ndarray,
    max_iters: int = 8000,
    eta0: float = 0.25,
    lam0: float = 0.5,
    lam_max: float = 200.0,
    lam_factor: float = 1.02,
) -> Tuple[np.ndarray, float]:
    """
    Mirror‑descent optimiser that reduces max(b)-min(b) where b = a*a.
    Returns the best sequence seen and its (unsmoothed) score.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()                         # guard against rounding errors

    best_seq = a.copy()
    b = _conv_full_fft(a)
    best_score = 2.0 * n * b.max()

    # FFT size sufficient for correlation (len(w)=2n‑1, len(a)=n)
    m = 1 << ((3 * n - 2).bit_length())   # >= (2n‑1)+(n‑1)

    lam = lam0
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- current convolution and score ---------------------------------
        b = _conv_full_fft(a)
        cur_max = b.max()
        cur_min = b.min()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] n={n:5d} iter={it:5d} score={best_score:.6f}"
            )

        # ----- construct weights for max and min -----------------------------
        # soft‑max on the peak
        w_max = np.exp(lam * (b - cur_max))          # max‑term (max exponent = 0)
        w_max_sum = w_max.sum()
        if w_max_sum == 0.0:
            w_max = np.full_like(b, 1.0 / b.size)
        else:
            w_max /= w_max_sum

        # soft‑min on the trough
        w_min = np.exp(lam * (cur_min - b))          # min‑term (max exponent = 0)
        w_min_sum = w_min.sum()
        if w_min_sum == 0.0:
            w_min = np.full_like(b, 1.0 / b.size)
        else:
            w_min /= w_min_sum

        # combined weight: push down the max and push up the min
        w = w_max - w_min   # sums to zero

        # ----- gradient (correlation) ----------------------------------------
        g = _gradient_via_correlation(w, a, m)

        # ----- exponentiated‑gradient step ------------------------------------
        eta = eta0 / np.sqrt(it)          # diminishing step size
        delta = -eta * g
        # clip to avoid overflow in exp
        delta = np.clip(delta, -30.0, 30.0)
        a = a * np.exp(delta)

        # ----- re‑normalise to the simplex ------------------------------------
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # ----- occasional jitter (helps escape tiny plateaus) ---------------
        if it % 500 == 0 and np.random.rand() < 0.15:
            a += np.random.randn(n) * 1e-5
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        # ----- increase temperature -------------------------------------------
        lam = min(lam_max, lam * lam_factor)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _smooth_refine(
    n: int,
    deadline: float,
    init_seq: np.ndarray,
    max_iters: int = 2000,
    eta0: float = 0.2,
    lam0: float = 0.5,
    lam_max: float = 30.0,
) -> Tuple[np.ndarray, float]:
    """
    Small final polishing using the soft‑max of the peak (the routine
    from the original baseline).  Returns the best sequence and its
    unsmoothed score.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = 2.0 * n * _conv_full_fft(a).max()
    m = 1 << ((3 * n - 2).bit_length())

    lam = lam0
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        A_fft = np.fft.rfft(a, m)
        b = np.fft.irfft(A_fft * A_fft, m)[: 2 * n - 1]
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
        # soft‑max over the peak
        maxb = b.max()
        exp_arg = lam * (b - maxb)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.size)
        else:
            w /= w_sum
        # gradient via correlation
        g = _gradient_via_correlation(w, a, m)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum
        lam = min(lam_max, lam * 1.01)
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    score.  The routine explores several lengths, builds a few deterministic
    starters (arcsine, beta sweep, random Dirichlet) and refines each with a
    specialised mirror‑descent that flattens the self‑convolution.
    The best sequence found before the time budget expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 4.0          # a tiny safety margin

    # ------------------------------------------------------------------
    # 0️⃣  Load historic candidate if available
    overall_best_seq = None
    overall_best_score = float("inf")
    try:
        hist = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        hist = np.clip(hist, 0.0, None)
        if hist.sum() > 0.0:
            hist /= hist.sum()
            hist_score = 2.0 * len(hist) * _conv_full_fft(hist).max()
            overall_best_seq = hist.copy()
            overall_best_score = hist_score
            print(f"Historic sequence: length={len(hist)} score={hist_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Lengths to explore (larger lengths usually give lower constants)
    candidate_lengths = [4000, 8000, 12000, 18000, 24000]
    candidate_lengths = [n for n in candidate_lengths if n >= 100]

    for n in candidate_lengths:
        if time.time() >= deadline:
            break

        print(f"\n=== Optimising length n = {n} ===")
        # --------------------------------------------------------------
        # a) a modest sweep of symmetric Beta densities (dense grid)
        alphas = np.linspace(0.15, 1.45, 30)
        alphas = np.concatenate([alphas, np.random.uniform(0.2, 1.4, size=10)])
        best_local_seq = None
        best_local_score = float("inf")

        for alpha in alphas:
            a = _beta_distribution(n, float(alpha))
            s = 2.0 * n * _conv_full_fft(a).max()
            if s < best_local_score:
                best_local_score = s
                best_local_seq = a.copy()

        # include the arcsine law – it is a strong baseline
        a_arc = _arcsine_distribution(n)
        s_arc = 2.0 * n * _conv_full_fft(a_arc).max()
        if s_arc < best_local_score:
            best_local_score = s_arc
            best_local_seq = a_arc.copy()

        # add a few random Dirichlet starters (different concentration)
        for conc in [0.3, 1.0, 3.0]:
            a_rand = np.random.dirichlet(np.full(n, conc))
            s_rand = 2.0 * n * _conv_full_fft(a_rand).max()
            if s_rand < best_local_score:
                best_local_score = s_rand
                best_local_seq = a_rand.copy()

        print(f"Length {n:5d}: best after sweep = {best_local_score:.6f}")

        # --------------------------------------------------------------
        # b) flattening optimisation from the best starter
        slice_deadline = time.time() + min(0.35, (deadline - time.time()) * 0.35)

        refined_seq, refined_score = _optimise_flatten(
            n=n,
            deadline=slice_deadline,
            init_seq=best_local_seq,
            max_iters=8000,
            eta0=0.25,
            lam0=0.5,
            lam_max=200.0,
            lam_factor=1.015,
        )

        # optional short soft‑max polishing
        polish_deadline = time.time() + min(0.10, (deadline - time.time()) * 0.10)
        refined_seq, refined_score = _smooth_refine(
            n=n,
            deadline=polish_deadline,
            init_seq=refined_seq,
            max_iters=2000,
            eta0=0.15,
            lam0=0.5,
            lam_max=30.0,
        )

        # --------------------------------------------------------------
        if refined_score < overall_best_score:
            overall_best_score = refined_score
            overall_best_seq = refined_seq.copy()
            print(f"*** New global best → n={n:5d} score={overall_best_score:.6f}")

        # early exit if we have already passed the target bound
        if overall_best_score <= 1.503:
            print("Target achieved – exiting early.")
            break

    # ------------------------------------------------------------------
    # 2️⃣  Fallback – guarantee a return value
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = 2.0 * n * _conv_full_fft(overall_best_seq).max()

    print(
        f"\nFinal best score (unsmoothed) = {overall_best_score:.6f} "
        f"with length = {len(overall_best_seq)}"
    )
    # Return plain Python list as required by the runner
    return overall_best_seq.tolist()
