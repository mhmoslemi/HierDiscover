"""
A two‑phase optimizer for the “step‑function” challenge.

Phase 1 – Adam‑style projected Adam on a smooth‐max surrogate of the
convolution peak.  The surrogate uses a soft‑max weighting of the
convolution entries, temperature μ is geometrically cooled and the
gradient is obtained with a single extra convolution, so each iteration
is O(n log n).

Phase 2 – Targeted pairwise mass‑transfer moves (simulated‑annealing style).
The current maximal convolution offset k* is identified, the two
indices that contribute most/least to that offset are found and a
fraction of mass is moved from the former to the latter.  Moves that
increase the score are occasionally accepted according to a temperature
schedule, allowing the search to escape shallow local minima.

Finally a cheap block‑search explores many piecewise‑constant patterns;
each pattern is locally refined by a few top‑k gradient steps.

All iterates are Euclidean‑projected onto the simplex
{a≥0, Σa = √(2 n)} so the evaluation score is simply max(conv(a,a)).
The routine respects the 1000 s wall‑clock budget and prints progress.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _gradient_topk(a: np.ndarray, conv: np.ndarray, top_k: int = 5) -> np.ndarray:
    """Sub‑gradient of the maximum of the convolution using the `top_k` largest entries."""
    n = a.shape[0]
    idxs = np.argpartition(-conv, top_k - 1)[:top_k]
    grad = np.zeros_like(a)
    i = np.arange(n)
    for k in idxs:
        j = k - i
        mask = (j >= 0) & (j < n)
        grad[mask] += 2.0 * a[j[mask]]
        diag = mask & (i == j)
        grad[diag] -= a[i[diag]]
    return grad

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate wall‑clock time budget (seconds).

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """

    deadline = time.time() + budget_s - 0.15  # tiny safety margin

    # Length of the problem (use the provided reference if available)
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
        have_ref = True
    except Exception:

        have_ref = False

    target_sum = math.sqrt(2.0 * n)          # the normalisation used by the score

    # --------------------------------------------------------------
    # Starting point (reference sequence with probability 0.6)
    if have_ref and np.random.rand() < 0.6:
        a = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
    else:
        a = np.random.rand(n)
    a = _project_to_simplex(a, target_sum)

    # Global best tracking
    best_a = a.copy()
    best_score = np.max(np.convolve(best_a, best_a))
    print(f"[init] n={n}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # ----- Phase 1 : Adam on smooth‑max ---------------------------------
    # hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step_count = 0

    mu_decay = 0.9992

        delta = a[src] * 0.25          # transfer 25 % of the source entry
        if delta > 0:

            new_max = np.convolve(a_new, a_new).max()

        # temperature decay
        T = max(T * T_decay, T_min)
        move_iter += 1

        # occasional global shake

            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a, target_sum)
            cur_score = np.convolve(a, a).max()

    # --------------------------------------------------------------
    # ----- Phase 3 : cheap block‑search with local refinement ------------
    while time.time() < deadline:
        nb = np.random.randint(2, 12)                     # 2‑11 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)

            a_blk = _project_to_simplex(a_blk, target_sum)

        if blk_score < best_score:
            best_score = blk_score
            best_a = a_blk.copy()
            print(f"[block] new best = {best_score:.6f}")

    # Final projection (ensures feasibility) and return
    best_a = np.clip(best_a, 0.0, None)
