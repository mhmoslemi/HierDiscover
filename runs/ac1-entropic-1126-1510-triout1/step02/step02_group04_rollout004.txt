"""
Simple mirror‑descent optimiser for the step‑function problem.

The algorithm works on the probability simplex (non‑negative entries,
sum = 1).  Starting from a known construction (or a uniform vector) it
repeatedly reduces the largest convolution entries by a projected
multiplicative (mirror) gradient step.  Every few iterations a
targeted move is applied (mass is shifted away from the pair that
contributes most to the current maximal lag) and the vector is
symmetrised (a ↔ a[::-1]) – the optimum is known to be symmetric.
A small amount of random noise is injected periodically and full
restarts are performed if the search stagnates.

The routine runs for at most ``budget_s`` seconds and returns the best
sequence found (as a plain Python list of floats).  All numbers are
non‑negative and the vector is normalised to sum = 1, which makes the
score identical to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: clip to the simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _clip_and_normalise(v: np.ndarray) -> np.ndarray:
    """Clip negative entries to zero and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:                     # fallback to uniform point
        v = np.full_like(v, 1.0 / v.size)
    else:
        v = v / s
    return v


# ----------------------------------------------------------------------
# Fast scoring (same formula as evaluate_sequence but without the
# safety wrapper – all our vectors are already on the simplex).
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Return the evaluation score for a probability vector ``seq``."""
    n = seq.size
    conv = np.convolve(seq, seq)          # linear convolution
    max_conv = conv.max()
    return float(2.0 * n * max_conv)      # denominator = 1 because sum(seq)=1


# ----------------------------------------------------------------------
# Gradient contribution of a single lag (mirrored indices)
# ----------------------------------------------------------------------
def _lag_gradient(seq: np.ndarray, lag: int) -> np.ndarray:
    """Return the gradient of the convolution entry at index ``lag``."""
    n = seq.size
    idx = np.arange(n)
    j = lag - idx
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(seq)
    # d/da_i (sum a_i a_{lag-i}) = 2 * a_{lag-i}
    g[mask] = 2.0 * seq[j[mask]]
    return g


# ----------------------------------------------------------------------
# Mirror‑descent step that reduces several of the largest convolution peaks.
# ----------------------------------------------------------------------
def _multi_lag_mirror_step(seq: np.ndarray, eta: float, top_k: int = 10) -> np.ndarray:
    """
    Perform a projected mirror‑descent step using the gradients of the
    ``top_k`` largest convolution entries.
    """
    conv = np.convolve(seq, seq)
    # indices of the k largest entries (ties broken arbitrarily)
    top_idxs = np.argpartition(-conv, range(top_k))[:top_k]

    # accumulate gradient
    g = np.zeros_like(seq)
    for lag in top_idxs:
        g += _lag_gradient(seq, lag)

    # multiplicative (mirror) update: a_i <- a_i * exp(-eta * g_i)
    # to avoid overflow we clip g to a reasonable range
    g = np.clip(g, 0.0, 50.0 / max(eta, 1e-12))
    a_new = seq * np.exp(-eta * g)
    a_new = _clip_and_normalise(a_new)
    return a_new


# ----------------------------------------------------------------------
# Targeted move: move a small fraction of mass away from the pair that
# contributes most to the current maximal convolution entry.
# ----------------------------------------------------------------------
def _targeted_move(seq: np.ndarray, frac: float = 0.02) -> np.ndarray:
    """
    Identify the lag with maximal convolution value, pick the pair
    (i, lag‑i) that contributes most to it and move a small fraction
    of mass from the larger index to a random location.
    """
    conv = np.convolve(seq, seq)
    lag = int(conv.argmax())

    # contributions a_i * a_{lag-i}
    i_arr = np.arange(seq.size)
    j_arr = lag - i_arr
    mask = (j_arr >= 0) & (j_arr < seq.size)
    contrib = np.zeros_like(seq)
    contrib[mask] = seq[i_arr[mask]] * seq[j_arr[mask]]

    if not np.any(mask):
        # fallback – random perturbation
        cand = seq + frac * (np.random.rand(seq.size) - 0.5)
        return _clip_and_normalise(cand)

    i_max = int(np.argmax(contrib))
    j_max = lag - i_max

    # move mass from the larger entry of the pair
    src = i_max if seq[i_max] >= seq[j_max] else j_max
    dst = np.random.randint(seq.size)
    while dst == src:
        dst = np.random.randint(seq.size)

    amount = frac * seq[src] * np.random.rand()
    cand = seq.copy()
    cand[src] -= amount
    cand[dst] += amount
    return _clip_and_normalise(cand)
