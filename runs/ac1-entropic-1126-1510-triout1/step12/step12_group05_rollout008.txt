    deadline = time.time() + budget_s - 2.0

    base_n = 8000
    # generate seeds

    # Evaluate seeds
    elite = []
    overall_best = None
    overall_best_score = float('inf')
    for cand in seeds:
        # ensure nonneg
        cand = np.maximum(cand, 0.0)
        # symmetrize
        cand = symmetrize(cand)
        cand /= cand.sum()
        sc = _objective(cand)
        elite.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best = cand.copy()
    # Keep top K

    elite = elite[:8]

    remaining = deadline - time.time()

"""
Hybrid symmetric search for the self‑convolution L∞ minimisation.

*Symmetrisation* – the optimal step function is known (or strongly believed)
to be symmetric, i.e. a_i = a_{n-1-i}.  All candidates and all updates are
forced to stay symmetric which reduces the search space and improves
numerical stability.

*Two‑stage optimisation* – a large family of analytic and random symmetric
distributions is generated for a moderate length (≈8000).  The best few
are refined by an exponentiated‑gradient descent on a soft‑max
approximation of the max‑convolution entry, followed by a hard‑max
sub‑gradient phase.  Afterwards the best distribution is up‑sampled to a
larger length (≈20000) and refined again.

*Polishing* – a cheap random mass‑transfer walk (accept‑only‑improving)
and a short simulated‑annealing walk are applied as final polishing.

The routine respects the provided time budget and returns the best
distribution (list of non‑negative floats) found.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force symmetry: v[i] = v[n‑1‑i]."""
    return (v + v[::-1]) / 2.0


def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (sum(a) > 0)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Sub‑gradient 2·a⋆w for a weight vector w (len = 2n‑1, sum(w)=1)."""
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.size - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


def _gradient_max_index(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the single maximal convolution entry conv[k]."""
    n = a.shape[0]
    g = np.zeros_like(a)

    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i = np.arange(i_min, i_max + 1, dtype=np.int64)
    j = k - i

    eq_mask = (i == j)
    if np.any(eq_mask):
        g[i[eq_mask]] += 2.0 * a[i[eq_mask]]
    if np.any(~eq_mask):
        ii = i[~eq_mask]
        jj = j[~eq_mask]
        g[ii] += a[jj]
        g[jj] += a[ii]

    return g

# Analytic families (all produce non‑negative vectors)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    eps = 1e-15
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * x / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals

def _initial_candidates_sym(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Broad pool of symmetric seeds for a given length."""
    cand = []
    # analytic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_sine(n))
    cand.append(_distribution_two_blocks(n, block_frac=0.07))

    # symmetric beta on a grid
    for p in np.linspace(0.05, 1.0, 12):
        cand.append(_distribution_beta_sym(n, p))

    # power‑law families
    for p in (0.5, 1.0, 2.0, 3.0):
        cand.append(_distribution_power_increasing(n, p))

    # exponential families
    for lam in (0.001, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential_increasing(n, lam))

    # random Dirichlet draws
    for a in (0.1, 0.3, 0.5, 1.0, 2.0):
        for _ in range(2):
            cand.append(_distribution_dirichlet(n, rng, alpha=a))

    # sparse random draws
    for dens in (0.005, 0.01, 0.02):
        cand.append(_distribution_random_sparse(n, rng, density=dens))

    # asymmetric beta + symmetrisation
    for _ in range(4):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        tmp = _distribution_beta_asym(n, alpha, beta)
        cand.append(_symmetrize(tmp))

    # ensure symmetry for everything (harmless double‑symmetrisation)
    cand = [_symmetrize(np.maximum(c, 0.0)) for c in cand]
    return cand

def _softmax_descent(a: np.ndarray, deadline: float,
                     best_seq: np.ndarray, best_score: float,
                     lr0: float = 1.0,
                     max_iter: int = 80000,
                     eval_interval: int = 400,
                     alpha_start: float = 0.5,
                     alpha_end: float = 80.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a soft‑max approximation of the maximal
    convolution entry.  Symmetry is enforced after each update.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        # linear temperature schedule
        alpha = alpha_start + (alpha_end - alpha_start) * t / max_iter
        # stable soft‑max
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()
        grad = _gradient_from_weights(a, w)
        # enforce symmetry on the gradient
        grad = (grad + grad[::-1]) / 2.0
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = _symmetrize(a)
        total = a.sum()
        if total > 0.0:
            a /= total
        else:
            a = np.full(n, 1.0 / n)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] soft‑max it={t} new best {best_score:.6f}")

    return best_seq, best_score


def _hardmax_descent(a: np.ndarray, deadline: float,
                     best_seq: np.ndarray, best_score: float,
                     lr0: float = 0.5,
                     max_iter: int = 60000,
                     eval_interval: int = 400) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent using the true maximal convolution entry.
    Symmetry is enforced after each update.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())
        grad = _gradient_max_index(a, k)
        grad = (grad + grad[::-1]) / 2.0
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = _symmetrize(a)
        total = a.sum()
        if total > 0.0:
            a /= total
        else:
            a = np.full(n, 1.0 / n)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hard‑max it={t} idx={k} new best {best_score:.6f}")

    return best_seq, best_score

def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer walk (accept‑only‑improving).  Symmetry is kept.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    cur_obj = _objective(a)
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue
        # move up to 30 % of source mass
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_a = _symmetrize(new_a)
        new_a = np.maximum(new_a, 0.0)
        total = new_a.sum()
        if total > 0.0:
            new_a /= total
        else:
            new_a = np.full(n, 1.0 / n)
        new_obj = _objective(new_a)
        if new_obj < cur_obj - 1e-14:
            a = new_a
            cur_obj = new_obj
            if new_obj < best_score - 1e-14:
                best_score = new_obj
                best_seq = a.copy()
                print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score


def _anneal_polish(a: np.ndarray, deadline: float,
                   best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing walk that can accept worsening moves early.
    Symmetry is reinstated after each proposal.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    cur_obj = _objective(a)
    start = time.time()
    max_frac = 0.20  # max fraction of source mass to move
    T0 = 0.008       # initial temperature (tuned empirically)

    while time.time() < deadline:
        elapsed = time.time() - start
        T = max(T0 * (1.0 - elapsed / (deadline - start)), 1e-9)

        # pick a source with non‑trivial mass
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue
        delta = a[i] * rng.random() * max_frac
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_a = _symmetrize(new_a)
        new_a = np.maximum(new_a, 0.0)
        total = new_a.sum()
        if total > 0.0:
            new_a /= total
        else:
            new_a = np.full(n, 1.0 / n)

        new_obj = _objective(new_a)

        if new_obj < cur_obj or rng.random() < math.exp(-(new_obj - cur_obj) / T):
            a = new_a
            cur_obj = new_obj
            if new_obj < best_score - 1e-14:
                best_score = new_obj
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _upsample_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a symmetric distribution to a larger size.
    """
    n = a.shape[0]
    if new_n == n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    interpolated = np.interp(new_x, old_x, a)
    # enforce symmetry again (tiny asymmetries from interpolation)
    interpolated = _symmetrize(interpolated)
    return interpolated

def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Main entry point.  Generates symmetric candidates, refines the most
    promising ones with a two‑phase exponentiated‑gradient descent,
    up‑samples to a larger length, refines again and finishes with cheap
    local/annealing polishing.  Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # keep a safety margin

    # ------------------------------------------------------------------
    # 1. Base length (moderate size for fast iterations)
    base_n = 8000
    seeds = _initial_candidates_sym(base_n, rng)

    # Incorporate a previously found solution if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = _symmetrize(np.maximum(prev, 0.0))
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            prev /= prev.sum()
            seeds.append(prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2. Evaluate all seeds, keep a small elite set
    elite: list[tuple[float, np.ndarray]] = []
    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")

    for cand in seeds:
        if time.time() >= deadline:
            break
        cand = _symmetrize(np.maximum(cand, 0.0))
        cand /= cand.sum()
        sc = _objective(cand)
        elite.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = cand.copy()
            print(f"[{time.time():.1f}] !!! new global best len={base_n} score={sc:.6f}")

    # keep only the top few (balance exploration / exploitation)
    elite.sort(key=lambda x: x[0])
    elite = elite[:10]

    # ------------------------------------------------------------------
    # 3. Two‑phase refinement on elite seeds (soft‑max then hard‑max)
    remaining = deadline - time.time()
    if remaining > 30.0:
        per_seed = max(0.5, remaining / (2.0 * len(elite)))  # split time between two phases
        for _, seed_seq in elite:
            if time.time() >= deadline:
                break
            # Phase A – soft‑max
            sub_dead = time.time() + per_seed
            seq_a, sc_a = _softmax_descent(
                seed_seq.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=1.0,
                max_iter=120000,
                eval_interval=500,
                alpha_start=0.4,
                alpha_end=90.0,
            )
            if sc_a < overall_best_score:
                overall_best_score = sc_a
                overall_best_seq = seq_a.copy()
                print(f"[{time.time():.1f}] after soft‑max new best {overall_best_score:.6f}")

            # Phase B – hard‑max
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            seq_b, sc_b = _hardmax_descent(
                seq_a.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.5,
                max_iter=100000,
                eval_interval=500,
            )
            if sc_b < overall_best_score:
                overall_best_score = sc_b
                overall_best_seq = seq_b.copy()
                print(f"[{time.time():.1f}] after hard‑max new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Upsample to larger lengths and evaluate – the bound improves with n
    candidate_lengths = [base_n, 12000, 20000]  # increasing granularity
    for n in candidate_lengths:
        if time.time() >= deadline:
            break
        if n == base_n:
            continue
        up_seq = _upsample_distribution(overall_best_seq, n)
        up_seq = _symmetrize(up_seq)
        up_seq /= up_seq.sum()
        sc = _objective(up_seq)
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = up_seq.copy()
            print(f"[{time.time():.1f}] upsampled to n={n} improved score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final refinement on the (now larger) best length
    if time.time() < deadline - 5.0:
        # Soft‑max phase again (shorter budget)
        sub_dead = time.time() + (deadline - time.time()) * 0.45
        seq, sc = _softmax_descent(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.8,
            max_iter=80000,
            eval_interval=400,
            alpha_start=0.5,
            alpha_end=80.0,
        )
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = seq.copy()
            print(f"[{time.time():.1f}] final soft‑max new best {overall_best_score:.6f}")

        # Hard‑max phase again
        if time.time() < deadline - 2.0:
            sub_dead = time.time() + (deadline - time.time()) * 0.45
            seq, sc = _hardmax_descent(
                overall_best_seq.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.5,
                max_iter=80000,
                eval_interval=400,
            )
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = seq.copy()
                print(f"[{time.time():.1f}] final hard‑max new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Polishing – cheap swaps + simulated annealing
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _anneal_polish(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
