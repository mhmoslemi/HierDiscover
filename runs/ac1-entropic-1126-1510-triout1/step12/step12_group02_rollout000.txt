def _smooth_gradient_descent(a: np.ndarray, deadline: float, best_seq, best_score, alpha0=5.0, alpha_max=200.0, eta0=0.5, eval_every=200):
    t = 0
    n = a.shape[0]
    while time.time() < deadline:
        t += 1
        # compute conv
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        # compute weight vector
        cmax = max_conv
        # temperature for smoothing schedule: increase alpha with t
        # compute target alpha
        # maybe linearly increase to alpha_max in first 10000 iters.
        if t < 2000:
            alpha = alpha0 + (alpha_max - alpha0) * t / 2000.0
        else:
            alpha = alpha_max

        # compute w
        # subtract cmax for numerical stability
        w_exp = np.exp(alpha * (conv - cmax))
        Z = w_exp.sum()
        # compute gradient using convolution of w with reversed a
        a_rev = a[::-1]
        # need to ensure length of fft appropriate
        size = 1 << ((len(w_exp) + n - 1 - 1).bit_length())
        W = np.fft.rfft(w_exp, n=size)
        A_rev = np.fft.rfft(a_rev, n=size)
        corr = np.fft.irfft(W * A_rev, n=size)
        grad = 2.0 * corr[n-1:n-1+n] / Z

        # exponentiated gradient update
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0/n)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = 2.0 * n * conv.max()
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth GD it={t} new best {best_score:.6f}")

    return best_seq, best_score

def _fft_convolve(x, y):
    n = len(x)
    m = len(y)
    size = 1 << ((n + m - 1) - 1).bit_length()
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[: n + m -1]

This implementation searches for a non‑negative step function (a probability vector) that minimises  

\[
\text{score}(a)=\frac{2\,n\;\max (a*a)}{\bigl(\sum a\bigr)^2}.
\]

Because the score is homogeneous, we can restrict ourselves to vectors whose
entries sum to 1 – the objective then simplifies to `2·n·max(conv)`.
The algorithm consists of three stages:

1. **Seed generation** – a small, diverse pool of analytic distributions
   (uniform, arcsine, symmetric beta, edge‑weighted, generalized‑normal,
   random Dirichlet, sparse) for a single length (default 8000).  
   The best few seeds are kept.

2. **Smoothed mirror‑descent** – instead of using only the gradient of the
   currently maximal convolution entry, we minimise a *smooth* approximation of
   the max using the log‑sum‑exp trick.  
   For a temperature `α` we define  

   \[
   \phi_\alpha(a)=\frac{1}{\alpha}\log\!\Bigl(\sum_k e^{\alpha\,c_k}\Bigr),
   \qquad
   c = a*a .
   \]

   The gradient of `φ_α` can be written as a convolution of the weight
   vector `w_k = e^{α(c_k‑c_max)}` with the reversed probability vector.
   Using FFTs the update is `a ← a·exp(−η·∇φ_α)` followed by normalisation
   onto the simplex (exponentiated‑gradient / mirror descent).  
   `α` is increased from a small value to a large one (≈ 150) so that the
   algorithm first explores a smooth landscape and then focuses on the true
   `max` as the optimisation proceeds.

3. **Final pair‑wise annealing** – a cheap simulated‑annealing polish that
   moves a random fraction of mass from one bin to another.  Improvements are
   always kept; early on worse moves are occasionally accepted with a linear
   cooling schedule.

The whole search respects the 1000 s budget (a safety margin of 2 s).  
If a previous construction `height_sequence_1` is present it is added to the
seed pool.  The best sequence found is returned as a plain list of `float`s.

```python
"""
Propose a non‑negative step function that (almost) minimises
``evaluate_sequence``.  The optimiser works in three phases:

1.  Generate a small pool of analytic seeds (uniform, arcsine, beta,
    edge‑weighted, generalized‑normal, Dirichlet, sparse) for a single
    length (default 8000).  Keep the best few.

2.  Run a *smoothed* mirror‑descent on the simplex:
    minimise a log‑sum‑exp surrogate of the max‑convolution.
    The gradient is obtained by a single FFT‑based convolution per
    iteration, and the update is an exponentiated‑gradient step
    (projection onto the simplex is free).

3.  Polish the result with a long‑run pair‑wise simulated‑annealing
    (mass transfers with linear cooling).

All computation is done with NumPy and the FFT; the code stays well
within the 1000 s budget and returns a list[float] ready for
``evaluate_sequence``.
"""

import math
import time
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    n = x.shape[0]
    m = y.shape[0]
    size = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[: n + m - 1]


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective (the quantity returned by ``evaluate_sequence``)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Compute the exact value of ``evaluate_sequence`` for a given *a*.
    The vector is assumed to be non‑negative; sum does not have to be 1.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:  # matching the security guard in the provided function
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the grid { (k+0.5)/n }."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (dist_to_left * dist_to_right)^{-p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    w /= w.sum()
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(-|x‑½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    if w.sum() == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with the given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.random(k)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    if a.sum() == 0.0:
        a[0] = 1.0
    a /= a.sum()
    return a


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smoothed mirror‑descent (log‑sum‑exp surrogate of the max)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    alpha_start: float = 10.0,
    alpha_end: float = 150.0,
    warmup_iters: int = 2000,
    eta0: float = 0.5,
    eval_every: int = 200,
) -> Tuple[np.ndarray, float]:
    """
    Perform exponentiated‑gradient descent on the smoothed objective

        φ_α(a) = (1/α)·log Σ_k exp(α·c_k) ,   c = a*a.

    The gradient of φ_α is obtained via a single FFT‑based convolution.
    The update is   a ← a·exp(−η·∇φ_α)   followed by normalisation on the
    probability simplex.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1

        # self‑convolution and its maximum
        conv = _fft_self_convolve(a)
        c_max = conv.max()

        # schedule for the smoothing parameter α
        if t <= warmup_iters:
            alpha = alpha_start + (alpha_end - alpha_start) * t / warmup_iters
        else:
            alpha = alpha_end

        # soft‑max weights (stabilised)
        w = np.exp(alpha * (conv - c_max))
        Z = w.sum()

        # gradient = (2 / Z) * (w  *  a_rev)  (cross‑correlation)
        a_rev = a[::-1]
        corr = _fft_convolve(w, a_rev)                     # length 3·n‑2
        grad = 2.0 * corr[n - 1 : n - 1 + n] / Z           # slice = n entries

        # exponentiated‑gradient step
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # keep positivity and renormalise
        a = np.maximum(a, 1e-16)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = 2.0 * n * conv.max()
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth‑GD it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise simulated annealing (final polishing)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> Tuple[np.ndarray, float]:
    """
    Randomly transfer a fraction of mass between two bins.
    Accept improvements always; early on also accept worsening moves
    with a linearly decreasing temperature.
    """
    n = a.shape[0]
    start = time.time()
    total = deadline - start

    while time.time() < deadline:
        # source (must have some mass)
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        # destination (different index)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move up to 40 % of the source mass
        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur_score = _objective(new_a)

        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.01

        accept = False
        if cur_score < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur_score - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation while staying on the simplex)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= a_new.sum()
    return a_new

            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after annealing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)/sum²): {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
```
