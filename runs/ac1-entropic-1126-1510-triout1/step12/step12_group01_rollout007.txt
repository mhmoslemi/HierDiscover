def _fft_cross_convolve(a,w):
    size = 1 << ((len(a)+len(w)-1 -1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(w, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[:len(a)+len(w)-1]

def _full_softmax_descent(a, deadline, best_seq, best_score, eta0=0.6, tau0=2.0, tau_factor=1.002, eval_interval=200, patience=2000):
    t=0; no_improve=0; tau=tau0; rng = np.random.default_rng()
    while time.time()<deadline and t<500_000:
        t+=1
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        # compute weights
        w_raw = np.exp(tau * (conv - max_conv))
        w = w_raw / w_raw.sum()
        # gradient via cross convolution with w_rev
        w_rev = w[::-1]
        cross = _fft_cross_convolve(a, w_rev)  # length n + L -1
        # L = len(w)
        L = len(w)
        # gradient size n: cross index L-1 - j
        grad = 2.0 * cross[L-1 - np.arange(len(a))]
        # exp gradient update
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        # simplex projection
        a = _project_onto_simplex(a, s=1.0)
        # update tau
        tau *= tau_factor

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] descent it={t} new best {best_score:.6f}")

                no_improve += 1

            jitter = rng.exponential(scale=0.005, size=len(a))
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

def _upsample_sequence(a, new_n):
    """Upsample sequence a to length new_n using linear interpolation."""
    old_n = len(a)
    # positions for old at midpoints: (i+0.5)/old_n
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    new_a = np.interp(new_x, old_x, a, left=0.0, right=0.0)
    # The interpolation may not preserve sum, renormalise
    total = new_a.sum()
    if total <= 0:
        new_a = np.full(new_n, 1.0/new_n)
    else:
        new_a /= total
    return new_a

"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.

Algorithm
---------
1. **Generate a diverse pool** – analytic families (uniform, arcsine,
   Beta‑sym, edge‑weighted, sine‑weighted, exponential), random Dirichlet
   and sparse vectors for several lengths.  Keep the ~30 best seeds.

2. **Full soft‑max exponential descent** – for each promising seed run a
   convex optimisation of the smooth surrogate
   S_τ(a)= (1/τ)·log Σ_i exp(τ·(a*a)_i) .
   The gradient is a weighted sum of the usual convolution gradients,
   computed efficiently with FFTs.  The update is multiplicative
   (exponentiated‑gradient) followed by Euclidean projection onto the
   probability simplex.  The temperature τ grows slowly, turning the
   surrogate into the true max.

3. **Upsample promising solutions** – the best sequence is linearly
   interpolated to larger sizes (≈1.5× and 2×) and a short descent run is
   performed on each; larger n often yields a lower peak.

4. **Local refinement** – cheap pairwise mass transfers and a final
   simulated‑annealing walk improve the result further.

The routine respects the supplied time budget (default 1000 s) and returns
a plain Python list; ``evaluate_sequence`` will perform the final checks.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


def _fft_cross_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a*b using a real FFT (length len(a)+len(b)‑1)."""
    n = a.shape[0]
    m = b.shape[0]
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective used by evaluate_sequence when Σa = 1:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Beta(a,b) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ sin(π (i+½)/n)^p."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential(n: int, gamma: float) -> np.ndarray:
    """Monotonically decaying exponential on the left side."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    w /= w.sum()
    return w


def _distribution_exponential_symmetric(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential peaked in the centre."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    w = np.exp(-gamma * np.abs(i - center))
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a to length new_n, then re‑normalise."""
    old_n = a.shape[0]
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    new_a = np.interp(new_x, old_x, a, left=0.0, right=0.0)
    tot = new_a.sum()
    if tot <= 0.0:
        new_a = np.full(new_n, 1.0 / new_n, dtype=np.float64)
    else:
        new_a /= tot
    return new_a

# Full soft‑max exponential descent ------------------------------------
def _full_softmax_descent(
    a: np.ndarray,
    deadline: float,
    eta0: float = 0.5,
    tau0: float = 2.0,
    tau_factor: float = 1.001,
    eval_interval: int = 100,
    patience: int = 2500,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the smooth surrogate
    S_τ(a) = (1/τ)·log Σ_i exp(τ·conv_i).  The temperature τ is
    increased slowly, turning the surrogate into the true max.
    Returns the best vector seen during the run and its (exact) objective.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    tau = tau0

    best_local = a.copy()
    best_score = _objective(a)

    while time.time() < deadline and t < 300_000:
        t += 1

        # self‑convolution and soft‑max weights
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        w_raw = np.exp(tau * (conv - max_conv))          # avoid overflow
        w = w_raw / w_raw.sum()

        # gradient = 2 * (a * w_rev) convolution slice
        w_rev = w[::-1]
        cross = _fft_cross_convolve(a, w_rev)            # length n+L‑1
        L = len(w)
        grad = 2.0 * cross[L - 1 - np.arange(n)]

        # multiplicative update + projection onto simplex
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = _project_onto_simplex(a, s=1.0)

        # temperature schedule
        tau *= tau_factor

        # periodic exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_local = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] descent it={t} new best {best_score:.6f}")
            else:
                no_improve += 1

        # jitter on long stagnation
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.005, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    # final check
    final_score = _objective(a)
    if final_score < best_score:
        best_score, best_local = final_score, a.copy()
    return best_local, best_score

# Pairwise swap (deterministic local refinement) ------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Transfer a small amount of mass; keep the move only if it improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_score = _objective(a)

    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(best_seq > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = best_seq[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta

        cur = _objective(cand)
        if cur < best_score - 1e-14:
            best_seq, best_score = cand, cur
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style local search ---------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    init_temp: float = 1e-2,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_score = _objective(a)
    T = init_temp

    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = best_seq[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta

        cur = _objective(cand)
        if cur < best_score - 1e-14:
            best_seq, best_score = cand, cur
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                best_seq = cand
        T *= decay

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget (default
    1000 s) and returns a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 3.0   # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Build a diverse seed pool (multiple lengths & families)
    lengths = [2500, 4000, 6000, 9000, 14000, 20000]
    p_grid = np.concatenate(
        [
            np.logspace(-2, -1, 9),      # 0.01 … 0.1
            np.logspace(-1, 0, 9),       # 0.1 … 1
            np.logspace(0, 0.5, 7),      # 1 … 3.16
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 7)          # 0 … 1
    gamma_grid = np.logspace(-3, 0.5, 7)        # 0.001 … 3.16

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # parametrised families + mixing with uniform
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            sine = _distribution_sine_weighted(n, p)
            expo = _distribution_exponential(n, p)          # reuse p as gamma
            expo_sym = _distribution_exponential_symmetric(n, p)

            for base in (beta, edge, sine, expo, expo_sym):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # asymmetric beta random sampling
        for _ in range(4):
            if time.time() >= deadline:
                break
            a_param = 10 ** rng.uniform(-2, 0)   # 0.01 … 1
            b_param = 10 ** rng.uniform(-2, 0)
            cand = _distribution_beta_asym(n, a_param, b_param)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best asym‑beta len={n} a={a_param:.3g} b={b_param:.3g} score={sc:.6f}"
                )

        # random Dirichlet seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(
                    f"[{time.time():.1f}] new global best Dirichlet len={n} score={sc:.6f}"
                )

        # sparse random seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(
                    f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}"
                )

        # keep only the most promising seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2. Include a previously discovered construction if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Guard against empty pool
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Full soft‑max descent on the most promising seeds
    remaining = deadline - time.time()
    if remaining > 30 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] starting full‑softmax descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            best_seq_local, best_score_local = _full_softmax_descent(
                seed_seq.copy(),
                sub_deadline,
                eta0=0.55,
                tau0=2.0,
                tau_factor=1.0015,
                eval_interval=120,
                patience=2000,
            )
            if best_score_local < overall_best_score:
                overall_best_score, overall_best_seq = best_score_local, best_seq_local.copy()
                print(
                    f"[{time.time():.1f}] after descent seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5. Upsample the current best and give it a short refinement
    if time.time() < deadline:
        base_n = len(overall_best_seq)
        for factor in (1.5, 2.0):
            new_n = int(base_n * factor)
            if new_n <= base_n or new_n > 20000:
                continue
            up_seq = _upsample_sequence(overall_best_seq, new_n)
            up_score = _objective(up_seq)
            print(f"[{time.time():.1f}] upsampled to {new_n}, score={up_score:.6f}")
            sub_deadline = time.time() + min(25.0, deadline - time.time())
            best_up, best_up_score = _full_softmax_descent(
                up_seq.copy(),
                sub_deadline,
                eta0=0.5,
                tau0=2.0,
                tau_factor=1.001,
                eval_interval=100,
                patience=1800,
            )
            if best_up_score < overall_best_score:
                overall_best_score, overall_best_seq = best_up_score, best_up.copy()
                print(
                    f"[{time.time():.1f}] new global best from upsample ({new_n}) score={overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 6. Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            max_trials=250_000,
        )

    # ------------------------------------------------------------------
    # 7. Final stochastic local search (simulated annealing style)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final random‑local search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list – evaluate_sequence will perform its own checks.
    return overall_best_seq.tolist()
