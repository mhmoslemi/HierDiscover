def _adam_descent(a, deadline, best_seq, best_score,
                  lr0=0.2, beta1=0.9, beta2=0.999, eps=1e-8,
                  active_frac_start=0.1, active_frac_end=0.01,
                  max_active=30, eval_interval=200,
                  patience=2000):
    n = len(a)
    m = np.zeros_like(a)  # first moment
    v = np.zeros_like(a)  # second moment
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    while time.time() < deadline and t < 200000:
        t += 1
        # compute convolution
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        # compute active fraction linearly decreasing
        frac = active_frac_start * ((active_frac_end/active_frac_start) ** (t / 200000))
        thresh = max_c * (1 - frac)
        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            top_idx = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top_idx]
        # compute subgradient
        grad = np.zeros_like(a)
        ind_grid = np.arange(n, dtype=np.int32)
        for idx in act:
            ind = idx - ind_grid
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * a[ind[mask]]
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step  # gradient descent (since we want minimize)
        # ensure nonnegativity via projection onto simplex
        a = _project_onto_simplex(a, s=1.0)
        # evaluate periodically
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(...)
            else:
                no_improve += 1
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.01, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a, 1.0)
            no_improve = 0
    return best_seq, best_score

"""
Improved search for a non‑negative step function minimizing
evaluate_sequence.

The algorithm consists of three stages:

1. **Diverse seed generation** – Analytic families (uniform,
   arcsine, symmetric beta, edge‑weighted, edge‑inverse‑power,
   exponential, linear, logistic), random Dirichlet and sparse
   vectors are sampled for several lengths (up to 20 000).  The best
   ~20 seeds (lowest exact objective 2·n·max(conv(a,a))) are kept.

2. **Adam‑style projected sub‑gradient descent** – For each seed we
   minimise   f(a)=2·n·max(conv(a,a))   under a∈Δ₁ (non‑negative,
   sum = 1).  In every iteration the active set consists of the
   convolution entries whose value lies within a diminishing
   fraction of the current maximum.  The sub‑gradient is assembled
   from those active indices, an Adam update is applied and the
   iterate is projected back onto the simplex.  Learning‑rate
   decay and occasional exponential jitter help escape plateaus.

3. **Local refinement** – A greedy pairwise mass‑transfer
   optimisation followed by a cheap simulated‑annealing search
   cleans up remaining irregularities.

The whole routine respects a user‑supplied time budget (default
1000 s) and returns a plain Python list suitable for
``evaluate_sequence``.  All numbers are non‑negative; the
search never writes to disk or uses external services.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())          # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped: w_i ∝ ((i+½)*(n‑i‑½))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    """Edge‑inverse‑power: w_i ∝ (i+½)^(−p)+(n‑i‑½)^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(‑c·i) ,  c>0."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i)
    w /= w.sum()
    return w


def _distribution_linear_decay(n: int) -> np.ndarray:
    """a_i ∝ (n‑i) (strictly decreasing linear)."""
    i = np.arange(n, dtype=np.float64)
    w = (n - i)
    w /= w.sum()
    return w


def _distribution_logistic(n: int, s: float) -> np.ndarray:
    """Sigmoidal shape: a_i ∝ 1/(1+exp((i‑n/2)/s))."""
    i = np.arange(n, dtype=np.float64) - n / 2.0
    w = 1.0 / (1.0 + np.exp(i / s))
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    """Sparse random vector with given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet with a random concentration."""
    alpha = rng.uniform(0.01, 3.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U+(1‑lam)·base (U uniform)."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Adam‑style projected sub‑gradient descent ----------------------------
def _adam_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.2,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    active_frac_start: float = 0.12,
    active_frac_end: float = 0.008,
    max_active: int = 30,
    max_iters: int = 200_000,
    eval_interval: int = 200,
    patience: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Adam‑type projected sub‑gradient method for
    f(a)=2·n·max(conv(a,a)), a∈Δ₁.
    """
    n = a.shape[0]
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    ind_grid = np.arange(n, dtype=np.int32)

    while time.time() < deadline and t < max_iters:
        t += 1

        # full convolution
        conv = _fft_self_convolve(a)
        max_c = conv.max()

        # diminishing active fraction (geometric schedule)
        frac = active_frac_start * ((active_frac_end / active_frac_start) ** (t / max_iters))
        thresh = max_c * (1.0 - frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            # keep only the largest contributors
            top_idx = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top_idx]

        # sub‑gradient from the active set
        grad = np.zeros_like(a)
        for idx in act:
            ind = idx - ind_grid
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * a[ind[mask]]

        # Adam update (minimisation → step opposite to gradient)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step                # gradient descent step

        # project back onto simplex (guarantees non‑negativity & Σ=1)
        a = _project_onto_simplex(a, s=1.0)

        # periodic exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(
                    f"[{time.time():.1f}] it={t:,} new global best {best_score:.6f}"
                )
            else:
                no_improve += 1

        # jitter if stalled
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.01, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a, s=1.0)
            no_improve = 0

    return best_seq, best_score

# Deterministic pairwise refinement ------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Greedy transfer of a random fraction of mass from one index to another."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.30
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score

# Cheap simulated‑annealing style local search -------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.25
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score
