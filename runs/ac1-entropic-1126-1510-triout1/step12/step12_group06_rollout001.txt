"""
Hybrid optimiser for the max‑self‑convolution problem.

Key ideas
---------
* Generate a very diverse set of seed probability vectors for a few
  lengths (including classic families, power‑law / exponential tails,
  mixtures of arcsine & uniform, Dirichlet draws and sparse patterns).
* For each seed run a three‑phase local optimiser:
    1. Smooth‑max Adam descent (temperature slowly increased) – fast
       exploration of the landscape.
    2. Exact max‑sub‑gradient descent (exponentiated‑gradient) – fine‑tune.
    3. Random Gaussian perturbations with simplex projection – escape
       shallow local minima.
* Keep the best sequence found across all lengths and seeds.
* All operations stay on the probability simplex, so the objective is
  scale‑invariant and the global budget (≈1000 s) is respected.
"""

import time
import math
import numpy as np
from typing import List, Tuple


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_i w_i·conv_i(a)  w.r.t. a.
    Implemented via FFT cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a))/sum(a)^2.
    The code keeps sum(a)=1, thus it reduces to 2·n·max(conv).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# ---- Seed families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discretised unit interval."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(alpha,beta) on the discretised unit interval."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_powerlaw_edges(n: int, e: float) -> np.ndarray:
    """
    Symmetric power‑law w_i ∝ (i+0.5)^e + (n-i-0.5)^e.
    Negative exponents concentrate mass near the ends,
    positive ones pull mass towards the centre.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    w = (i ** e) + ((n - i) ** e)
    w = np.maximum(w, 0.0)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exp_edges(n: int, lam: float) -> np.ndarray:
    """
    Symmetric exponential tails: w_i ∝ exp(-lam * min(i, n-1-i)).
    Larger lam → sharper drop towards the centre.
    """
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    w = np.exp(-lam * d)
    w /= w.sum()
    return w


def _distribution_mixed_arcsine_uniform(n: int, mix: float) -> np.ndarray:
    """mix * arcsine + (1-mix) * uniform, mix∈[0,1]."""
    a = _distribution_arcsine(n)
    u = _distribution_uniform(n)
    w = mix * a + (1.0 - mix) * u
    w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Few non‑zero coordinates, random values."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals

def _initial_candidates(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    """Return a rich collection of seed vectors for a given length."""
    cand: List[np.ndarray] = []

    # classic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # symmetric Beta exponents (log‑spaced, including very small exponents)
    for p in np.geomspace(0.01, 5.0, num=13):
        cand.append(_distribution_beta_sym(n, float(p)))

    # asymmetric Beta draws
    for _ in range(6):
        alpha = 10 ** rng.uniform(-3, 1)   # 0.001 … 10
        beta = 10 ** rng.uniform(-3, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # power‑law edge families
    for e in (-2.0, -1.5, -1.2, -1.0, -0.8, -0.6, -0.5, -0.4, -0.2, 0.0, 0.2):
        cand.append(_distribution_powerlaw_edges(n, e))

    # exponential edge families
    for lam in (0.001, 0.003, 0.006, 0.01, 0.02, 0.04):
        cand.append(_distribution_exp_edges(n, lam))

    # mixtures of arcsine and uniform
    for mix in (0.2, 0.4, 0.6, 0.8):
        cand.append(_distribution_mixed_arcsine_uniform(n, mix))

    # Dirichlet draws with varying concentration
    for a in (0.05, 0.1, 0.2, 0.5, 0.9):
        cand.append(_distribution_dirichlet(n, rng, alpha=a))

    # Sparse random patterns
    cand.append(_distribution_random_sparse(n, rng, density=0.005))
    cand.append(_distribution_random_sparse(n, rng, density=0.02))

    # few completely random vectors
    for _ in range(4):
        v = rng.random(n)
        v = np.maximum(v, 0.0)
        v /= v.sum()
        cand.append(v)

    return cand

def _smooth_max_adam_descent(a: np.ndarray,
                             deadline: float,
                             best_seq: np.ndarray,
                             best_score: float,
                             eta0: float = 0.6,
                             beta1: float = 0.9,
                             beta2: float = 0.999,
                             eps: float = 1e-8,
                             alpha0: float = 0.1,
                             alpha_mul: float = 1.12,
                             alpha_max: float = 200.0,
                             max_iter: int = 120000,
                             eval_interval: int = 400) -> Tuple[np.ndarray, float]:
    """
    Adam‑style exponentiated‑gradient on the smooth‑max surrogate
    f_α(a) = (1/α)·log Σ_i exp(α·conv_i(a)).
    Temperature α is increased slowly (α←α·α_mul every few hundred steps).
    """
    n = a.shape[0]
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        # self‑convolution
        conv = _fft_self_convolve(a)

        # soft‑max weights
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient
        grad = _gradient_from_weights(a, w)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # multiplicative update (exponentiated‑gradient)
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(-step)

        # project onto the simplex
        a = _project_onto_simplex(a, s=1.0)

        # temperature schedule
        if t % 300 == 0 and alpha < alpha_max:
            alpha = min(alpha * alpha_mul, alpha_max)

        # occasional exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f}  best={best_score:.6f}")

    return best_seq, best_score

def _subgradient_descent(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         lr0: float = 0.6,
                         max_iter: int = 80000,
                         eval_interval: int = 500) -> Tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent on the exact max‑convolution objective.
    Uses an exponentiated‑gradient update with a decaying step size.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        # active set (indices attaining the max, up to numerical tolerance)
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)          # diminishing step size
        a = a * np.exp(-eta * grad)       # multiplicative EG step
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Subgrad it={t} best={best_score:.6f}")

    return best_seq, best_score

def _random_gaussian_perturb(a: np.ndarray,
                             rng: np.random.Generator,
                             sigma: float) -> np.ndarray:
    """Additive Gaussian noise followed by simplex projection."""
    b = a + rng.normal(scale=sigma, size=a.shape)
    b = np.maximum(b, 0.0)
    b = _project_onto_simplex(b, s=1.0)
    return b


def _perturbation_phase(a: np.ndarray,
                        rng: np.random.Generator,
                        deadline: float,
                        best_seq: np.ndarray,
                        best_score: float,
                        start_sigma: float = 0.005,
                        min_sigma: float = 1e-5,
                        decay: float = 0.999) -> Tuple[np.ndarray, float]:
    """
    Stochastic Gaussian perturbations with a simple cooling schedule.
    Accepts a move if it improves the objective; otherwise discards.
    """
    sigma = start_sigma
    cur_a = a.copy()
    cur_score = best_score
    while time.time() < deadline and sigma > min_sigma:
        cand = _random_gaussian_perturb(cur_a, rng, sigma)
        cand_score = _objective(cand)
        if cand_score < cur_score - 1e-13:
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score - 1e-13:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time():.1f}] Perturb sigma={sigma:.2e} new best={best_score:.6f}")
        sigma *= decay
    return best_seq, best_score

def _interpolate_resize(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    vals = np.interp(x_new, x_old, seq)
    vals = np.maximum(vals, 0.0)
    vals = _project_onto_simplex(vals, s=1.0)
    return vals

            top_candidates.append((sc, cand.copy()))

        # keep only a handful of promising seeds overall

                prev = _project_onto_simplex(prev, s=1.0)

                    overall_best_score = sc
                    overall_best_seq = prev.copy()
                    print(f"[{time.time():.1f}] Previous solution improved score={sc:.6f}")
                top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Fallback (should never trigger)
    if overall_best_seq is None:
        n0 = 4096
        overall_best_seq = _distribution_uniform(n0)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] Fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. First optimisation phase – Adam smooth‑max on a few top seeds
    remaining = deadline - time.time()
    if remaining > 60.0:
        # select the best few seeds (including the global best)
        seeds = [overall_best_seq.copy()] + [
            seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:4]
        ]
        for seed_seq in seeds:
            if time.time() >= deadline:
                break
            # allocate roughly equal remaining budget to each seed
            sub_deadline = time.time() + (deadline - time.time()) / max(1, len(seeds))
            seq_out, score_out = _smooth_max_adam_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.6,
                beta1=0.9,
                beta2=0.999,
                eps=1e-8,
                alpha0=0.1,
                alpha_mul=1.12,
                alpha_max=200.0,
                max_iter=120000,
                eval_interval=500,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] After Adam new best={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Second optimisation phase – exact sub‑gradient descent
    remaining = deadline - time.time()
    if remaining > 40.0:
        seeds = [overall_best_seq.copy()] + [
            seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:2]
            if not np.may_share_memory(seq, overall_best_seq)
        ]
        for seed_seq in seeds:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + (deadline - time.time()) / max(1, len(seeds))
            seq_out, score_out = _subgradient_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.6,
                max_iter=80000,
                eval_interval=500,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] After sub‑grad new best={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Optional upscale & refinement (if time allows)
    if time.time() < deadline - 5.0:
        # try to enlarge the distribution while preserving its shape
        target_n = max(lengths) + 2000          # a modest increase
        up_seq = _interpolate_resize(overall_best_seq, target_n)
        up_deadline = deadline - 3.0
        # a short Adam run on the upscaled vector
        up_seq, up_score = _smooth_max_adam_descent(
            up_seq,
            up_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.5,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            alpha0=0.1,
            alpha_mul=1.12,
            alpha_max=200.0,
            max_iter=50000,
            eval_interval=500,
        )
        if up_score < overall_best_score:
            overall_best_score = up_score
            overall_best_seq = up_seq.copy()
            print(f"[{time.time():.1f}] Upscaled refinement best={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final polishing – stochastic Gaussian perturbations
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _perturbation_phase(
            overall_best_seq.copy(),
            rng,
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            start_sigma=0.004,
            min_sigma=1e-6,
            decay=0.998,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
