def _distribution_exponential_decreasing(n, lam=0.005):

def _distribution_power_decreasing(n, p):

    vals = i**(-p)
    vals /= vals.sum()
    return vals

def _distribution_random_decreasing(n, rng):
    v = rng.random(n)
    v = np.sort(v)[::-1]  # descending
    v /= v.sum()

def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:

#!/usr/bin/env python
"""
Hybrid optimiser for the max‑convolution step‑function problem.

Main ideas
~~~~~~~~~~
* Start with a modest length (≈2000) to explore many analytic families
  (including new decreasing families) and cheap random permutations.
* Keep the ten most promising normalised seeds and refine them with an
  Adam‑style mirror‑descent on a smooth‑max surrogate.  The temperature
  α is increased to a large value (≈800) so that the optimisation finally
  targets the true L∞‑objective.
* Upsample the current best solution to larger lengths (4000, 8000, 12000)
  via linear interpolation, normalise again and run a short Adam phase
  for each length.  Upsampling preserves the overall shape while giving
  the optimiser more degrees of freedom.
* Finish with a classic sub‑gradient descent (multimax) and a cheap
  greedy pairwise mass‑transfer polishing step.
* All updates stay on the probability simplex (sum = 1, non‑negative)
  using Euclidean projection.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·conv_k(a)  w.r.t. a.
    Implemented via FFT cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (sum = 1 ⇒ 2·n·max)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic families (expanded)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


# ---- new decreasing families ------------------------------------------------
def _distribution_exponential_decreasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * i)
    vals /= vals.sum()
    return vals


def _distribution_power_decreasing(n: int, p: float = 1.0) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _distribution_random_decreasing(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    v = np.sort(v)[::-1]               # descending order
    v /= v.sum()
    return v


# ----------------------------------------------------------------------
def _initial_candidates_extended_fine(n: int,
                                     rng: np.random.Generator) -> List[np.ndarray]:
    """Rich set of seed vectors for a given length (including decreasing families)."""
    cand = []

    # Base families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # Symmetric Beta – log‑spaced exponents
    for p in np.geomspace(0.01, 5.0, num=12):
        cand.append(_distribution_beta_sym(n, float(p)))

    # Random asymmetric Beta draws
    for _ in range(6):
        alpha = 10 ** rng.uniform(-3, 1)
        beta = 10 ** rng.uniform(-3, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # Power‑law increasing
    for p in (0.5, 1.0, 2.0, 3.0):
        cand.append(_distribution_power_increasing(n, p))

    # Exponential increasing
    for lam in (0.001, 0.003, 0.005, 0.01):
        cand.append(_distribution_exponential_increasing(n, lam))

    # Two‑block spikes
    for bf in (0.03, 0.05, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac=bf))

    # Dirichlet draws
    for a in (0.1, 0.3, 0.7, 1.0):
        cand.append(_distribution_dirichlet(n, rng, alpha=a))

    # Random sparse patterns
    cand.append(_distribution_random_sparse(n, rng, density=0.01))
    cand.append(_distribution_random_sparse(n, rng, density=0.03))

    # Completely random probability vectors
    for _ in range(4):
        v = rng.random(n)
        v = np.maximum(v, 0.0)
        v /= v.sum()
        cand.append(v)

    # ---- new decreasing families --------------------------------------------
    for lam in (0.0005, 0.001, 0.002, 0.003, 0.005):
        cand.append(_distribution_exponential_decreasing(n, lam))
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        cand.append(_distribution_power_decreasing(n, p))
    for _ in range(3):
        cand.append(_distribution_random_decreasing(n, rng))

    return cand

def _smooth_max_adam_descent(a: np.ndarray,
                             deadline: float,
                             best_seq: np.ndarray,
                             best_score: float,
                             eta0: float = 0.6,
                             beta1: float = 0.9,
                             beta2: float = 0.999,
                             eps: float = 1e-8,
                             alpha0: float = 0.2,
                             alpha_mul: float = 1.4,
                             alpha_max: float = 800.0,
                             max_iter: int = 50000,
                             eval_interval: int = 400) -> Tuple[np.ndarray, float]:
    """Adam‑style mirror‑descent on the soft‑max surrogate of the max‑conv."""
    n = a.shape[0]
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        # Full self‑convolution
        conv = _fft_self_convolve(a)

        # Soft‑max weights (log‑sum‑exp trick)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # Gradient of smooth surrogate
        grad = _gradient_from_weights(a, w)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # Bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Multiplicative update (mirror descent on simplex)
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(-step)

        # Projection back onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # Temperature schedule
        if t % 300 == 0 and alpha < alpha_max:
            alpha = min(alpha * alpha_mul, alpha_max)

        # Periodic exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.6,
                                 max_iter: int = 35000,
                                 eval_interval: int = 500) -> Tuple[np.ndarray, float]:
    """Classic sub‑gradient descent on the true L∞‑objective."""
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()

        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)

        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Multimax it={t} new best {best_score:.6f}")

    return best_seq, best_score

def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Greedy pairwise mass‑transfer polishing."""
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        # distinct destination
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # random transfer amount (up to 30 % of source mass)
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) to new_n, keeping non‑negativity."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    new_seq = np.maximum(new_seq, 0.0)
    return new_seq

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects a time budget (default 1000 s) and returns the best
    coefficient list found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Initial exploration on a modest length
    base_n = 2000
    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: List[Tuple[float, np.ndarray]] = []

    print(f"[{time.time():.1f}] Generating seed vectors of length {base_n}")
    cand_list = _initial_candidates_extended_fine(base_n, rng)

    for cand in cand_list:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s
        sc = _objective(cand)
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = cand.copy()
            print(f"[{time.time():.1f}] **** new global best(len={base_n}) score={sc:.6f}")
        top_candidates.append((sc, cand.copy()))

    # Keep a small elite set
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:12]

    # ------------------------------------------------------------------
    # 2. Adam‑style smooth‑max descent on the elite seeds
    if time.time() < deadline:
        seeds = [overall_best_seq.copy()] + [
            seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:4]
        ]
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + (deadline - time.time()) / max(1, len(seeds) - idx)
            seq_out, score_out = _smooth_max_adam_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.5,
                beta1=0.9,
                beta2=0.999,
                eps=1e-8,
                alpha0=0.15,
                alpha_mul=1.5,
                alpha_max=800.0,
                max_iter=40000,
                eval_interval=400,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after Adam seed {idx} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Upsample to larger lengths and re‑optimise
    larger_lengths = [4000, 8000, 12000]
    for n in larger_lengths:
        if time.time() >= deadline:
            break
        if overall_best_seq is None:
            continue
        up_seq = _upsample_sequence(overall_best_seq, n)
        up_seq = np.maximum(up_seq, 0.0)
        s = up_seq.sum()
        if s == 0.0:
            continue
        up_seq /= s
        # a short Adam phase on the larger problem
        sub_deadline = time.time() + (deadline - time.time()) / (len(larger_lengths) - larger_lengths.index(n) + 1)
        seq_out, score_out = _smooth_max_adam_descent(
            up_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.4,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            alpha0=0.1,
            alpha_mul=1.4,
            alpha_max=600.0,
            max_iter=30000,
            eval_interval=400,
        )
        if score_out < overall_best_score:
            overall_best_score = score_out
            overall_best_seq = seq_out.copy()
            print(f"[{time.time():.1f}] upsampled len={n} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Classic multimax sub‑gradient descent for fine‑tuning
    if time.time() < deadline and overall_best_seq is not None:
        seq_out, score_out = _subgradient_multimax_descent(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.5,
            max_iter=35000,
            eval_interval=500,
        )
        if score_out < overall_best_score:
            overall_best_score = score_out
            overall_best_seq = seq_out.copy()
            print(f"[{time.time():.1f}] after multimax final best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Greedy pairwise polishing
    if time.time() < deadline and overall_best_seq is not None:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
