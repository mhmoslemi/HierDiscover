"""
Simple simulated‑annealing plus subgradient polishing search for the
step‑function minimisation problem.  The algorithm generates a few
analytical candidates for several moderate lengths, then improves each
candidate by repeated random mass‑transfer moves on the probability
simplex (accepting improvements and occasional uphill moves).  After a
fixed budget of moves the current best is polished with the multimax
sub‑gradient descent from the previous solution.  The best vector found
is returned as a plain list of floats.
"""

import time
import math
from typing import List, Optional
import numpy as np

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x>=0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # Fast path for already feasible vector
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of 2
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2; assumes a is non‑negative."""
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    # We keep a on the simplex, i.e. s≈1, so we can skip division.
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

# Analytic families -------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals

def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals

def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals

def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a

def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals

def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals

def _distribution_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.minimum(i + 1, n - i)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals

def _distribution_gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    vals /= vals.sum()
    return vals

# -------------------------------------------------------------------------
def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.6,
                                 max_iter: int = 30000,
                                 eval_interval: int = 300) -> tuple[np.ndarray, float]:
    """Simple multimax sub‑gradient descent on the simplex (as in the
    reference implementation)."""
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt
        # gradient wrt a: 2 * a ⋆ w (cross‑correlation)
        # Implement via FFT
        a_rev = a[::-1]
        size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
        W = np.fft.rfft(w, n=size)
        A_rev = np.fft.rfft(a_rev, n=size)
        conv_grad = np.fft.irfft(W * A_rev, n=size)
        start = n - 1
        grad = 2.0 * conv_grad[start:start + n]

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] multimax it={t} new best {best_score:.6f}")
    return best_seq, best_score

# -------------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Simulated‑annealing search for a probability vector that minimises
    ``evaluate_sequence``.  The routine builds a few analytic seeds for
    several moderate lengths, then improves each seed by many random
    mass‑transfer moves on the simplex (accepting improvements and a
    temperature‑controlled amount of uphill moves).  The current best
    vector is finally polished with a short multimax sub‑gradient
    descent.  The best sequence found before the deadline is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1.  Determine lengths to explore.
    lengths = [4000, 6000, 8000]   # moderate sizes that still run fast
    overall_best_seq: Optional[np.ndarray] = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 2.  Build seed pool for each length.
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Exploring length {n}")

        seeds = [
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_sine(n),
            _distribution_triangular(n),
            _distribution_gaussian(n),
            _distribution_two_blocks(n, block_frac=0.05),
            _distribution_random_sparse(n, rng, density=0.01),
        ]

        # symmetric beta family
        for p in (0.2, 0.4, 0.6, 1.0, 2.0):
            seeds.append(_distribution_beta_sym(n, p))

        # power‑law families
        for p in (0.5, 1.0, 2.0, 3.0):
            seeds.append(_distribution_power_increasing(n, p))

        # exponential families
        for lam in (0.001, 0.003, 0.005, 0.01):
            seeds.append(_distribution_exponential_increasing(n, lam))

        # random Dirichlet draws
        for _ in range(3):
            alpha = 0.1 + 0.9 * rng.random()
            seeds.append(_distribution_dirichlet(n, rng, alpha=alpha))

        # include previous solution if compatible
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.ndim == 1 and prev.size > 0:
                # resize or trim/pad to current length
                if prev.size < n:
                    # pad with zeros
                    pad = np.zeros(n - prev.size, dtype=np.float64)
                    prev = np.concatenate([prev, pad])
                elif prev.size > n:
                    prev = prev[:n]
                prev = np.maximum(prev, 0.0)
                s = prev.sum()
                if s == 0.0:
                    prev = _distribution_uniform(n)
                else:
                    prev = prev / s
                seeds.append(prev)
        except Exception:
            pass

        # ------------------------------------------------------------------
        # 3.  Evaluate seeds, keep a few promising ones.
        evaluated = []
        for svec in seeds:
            svec = np.maximum(svec, 0.0)
            svec = _project_onto_simplex(svec)   # guarantee sum=1
            sc = _objective(svec)
            evaluated.append((sc, svec))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = svec.copy()
                print(f"[{time.time():.1f}] **** new global best (len={n}) score={sc:.6f}")

        evaluated.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in evaluated[:3]]   # three best seeds for this length

        # ------------------------------------------------------------------
        # 4.  Local stochastic improvement for each top seed.
        for seed_vec in top_seeds:
            if time.time() >= deadline:
                break
            a = seed_vec.copy()
            cur_score = _objective(a)
            # Ensure we work with a normalized vector
            a = _project_onto_simplex(a)

            # Parameters for the simulated‑annealing walk
            sigma_start = 0.25   # start moving up to 25 % of the source mass
            sigma_end   = 0.001  # final step size
            T_start = 0.05
            T_end   = 1e-6
            max_iter = 200000

            start_time = time.time()
            for it in range(1, max_iter + 1):
                if time.time() >= deadline:
                    break

                # linearised fraction of the schedule
                frac = it / max_iter
                sigma = sigma_start * (sigma_end / sigma_start) ** frac
                T = T_start * (T_end / T_start) ** frac

                # ---- propose a new vector by moving a random fraction of mass ----
                # Choose source index weighted by current mass (more likely to pick a large entry)
                i = rng.choice(len(a), p=a)
                # Choose a distinct target index uniformly
                j = rng.integers(len(a) - 1)
                if j >= i:
                    j += 1
                # Amount of mass to move
                delta = a[i] * sigma * rng.random()
                if delta <= 0.0:
                    continue
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                # (no negative entries by construction)
                cand_score = _objective(cand)

                # ---- Metropolis acceptance ----
                if cand_score < cur_score:
                    a = cand
                    cur_score = cand_score
                    if cur_score < overall_best_score:
                        overall_best_score = cur_score
                        overall_best_seq = a.copy()
                        print(f"[{time.time():.1f}] SA iter={it} len={n} new best {overall_best_score:.6f}")
                else:
                    # uphill move with probability exp(-(Δ)/T)
                    if rng.random() < math.exp(-(cand_score - cur_score) / T):
                        a = cand
                        cur_score = cand_score

                # Optional periodic polishing with sub‑gradient (lightweight)
                if it % 25000 == 0:
                    sub_deadline = time.time() + min(5.0, deadline - time.time())
                    a, cur_score = _subgradient_multimax_descent(
                        a.copy(),
                        sub_deadline,
                        best_seq=overall_best_seq,
                        best_score=overall_best_score,
                        lr0=0.4,
                        max_iter=8000,
                        eval_interval=400,
                    )
                    if cur_score < overall_best_score:
                        overall_best_score = cur_score
                        overall_best_seq = a.copy()
                        print(f"[{time.time():.1f}] post‑sgd polish len={n} best {overall_best_score:.6f}")

            # End of SA loop for this seed – give one final polishing step
            sub_deadline = time.time() + min(10.0, deadline - time.time())
            a, cur_score = _subgradient_multimax_descent(
                a.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.3,
                max_iter=12000,
                eval_interval=500,
            )
            if cur_score < overall_best_score:
                overall_best_score = cur_score
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] final polish after SA len={n} best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Fallback if nothing was found (should not happen)
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"Fallback uniform seq, score={overall_best_score}")

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()

"""
Simple simulated‑annealing + multimax polishing for the step‑function
minimisation task.

The routine builds a handful of analytic probability vectors for a few
moderate lengths (4000, 6000, 8000).  For each length the best seed is
improved by a large number of random mass‑transfer moves on the simplex
(Metropolis‑style simulated annealing).  The current global best is
periodically polished with a short multimax sub‑gradient descent (the
method used in the reference implementation).  The best sequence found
before the time budget expires is returned as a plain list of floats.
"""

import time
import math
from typing import List, Optional
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a))/sum(a)^2 (assumes a ≥ 0).
    The vector is usually kept on the simplex, i.e. sum(a)≈1.
    """
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

# Analytic families (probability vectors) --------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.minimum(i + 1, n - i)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.6,
                                 max_iter: int = 30000,
                                 eval_interval: int = 300) -> tuple[np.ndarray, float]:
    """
    Classic multimax sub‑gradient descent (exponentiated‑gradient on the
    simplex).  Used as a polishing step after the stochastic walk.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        # gradient = 2 * a ⋆ w   (cross‑correlation)
        a_rev = a[::-1]
        size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
        W = np.fft.rfft(w, n=size)
        A_rev = np.fft.rfft(a_rev, n=size)
        conv_grad = np.fft.irfft(W * A_rev, n=size)
        start = n - 1
        grad = 2.0 * conv_grad[start:start + n]

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] multimax it={t} new best {best_score:.6f}")
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Simulated‑annealing search for a probability vector that minimises
    ``evaluate_sequence``.  The routine builds analytic seeds for several
    moderate lengths, improves each seed with many random mass‑transfer
    moves on the simplex (accepting improvements and a few uphill moves
    controlled by a temperature schedule), and finally polishes the
    current best with a short multimax sub‑gradient descent.  The best
    sequence found before the deadline is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a few moderate lengths.
    lengths = [4000, 6000, 8000]

    overall_best_seq: Optional[np.ndarray] = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Exploring length {n}")

        # ------------------------------------------------------------------
        # 2. Assemble a modest seed set.
        seeds = [
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_sine(n),
            _distribution_triangular(n),
            _distribution_gaussian(n),
            _distribution_two_blocks(n, block_frac=0.05),
            _distribution_random_sparse(n, rng, density=0.01),
        ]

        # symmetric beta family
        for p in (0.2, 0.4, 0.6, 1.0, 2.0):
            seeds.append(_distribution_beta_sym(n, p))

        # power‑law families
        for p in (0.5, 1.0, 2.0, 3.0):
            seeds.append(_distribution_power_increasing(n, p))

        # exponential families
        for lam in (0.001, 0.003, 0.005, 0.01):
            seeds.append(_distribution_exponential_increasing(n, lam))

        # random Dirichlet draws
        for _ in range(3):
            alpha = 0.1 + 0.9 * rng.random()
            seeds.append(_distribution_dirichlet(n, rng, alpha=alpha))

        # include previous solution if available
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.ndim == 1 and prev.size > 0:
                if prev.size < n:
                    prev = np.concatenate([prev, np.zeros(n - prev.size)])
                elif prev.size > n:
                    prev = prev[:n]
                prev = np.maximum(prev, 0.0)
                s = prev.sum()
                if s == 0.0:
                    prev = _distribution_uniform(n)
                else:
                    prev = prev / s
                seeds.append(prev)
        except Exception:
            pass

        # ------------------------------------------------------------------
        # 3. Evaluate seeds and keep the best one.
        evaluated = []
        for svec in seeds:
            svec = np.maximum(svec, 0.0)
            svec = _project_onto_simplex(svec)
            sc = _objective(svec)
            evaluated.append((sc, svec))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = svec.copy()
                print(f"[{time.time():.1f}] **** new global best (len={n}) score={sc:.6f}")

        evaluated.sort(key=lambda x: x[0])
        best_seed = evaluated[0][1]   # the single most promising seed for this length

        # ------------------------------------------------------------------
        # 4. Simulated‑annealing (mass‑transfer) improvement of this seed.
        a = best_seed.copy()
        cur_score = _objective(a)
        a = _project_onto_simplex(a)   # safety

        # annealing schedule parameters
        sigma_start = 0.25   # up‑to 25 % of source mass may be moved initially
        sigma_end   = 0.001  # final step size
        T_start = 0.05
        T_end   = 1e-6
        max_iter = 120_000   # enough iterations for a good search

        for it in range(1, max_iter + 1):
            if time.time() >= deadline:
                break

            # linear schedule fraction
            frac = it / max_iter
            sigma = sigma_start * (sigma_end / sigma_start) ** frac
            T = T_start * (T_end / T_start) ** frac

            # ---- propose a new vector by moving a random fraction of mass ----
            i = rng.choice(len(a), p=a)                     # heavy entries more likely as source
            j = rng.integers(len(a) - 1)                    # uniform target index
            if j >= i:

            delta = a[i] * sigma * rng.random()
            if delta <= 0.0:

            cand_score = _objective(cand)

                if cur_score < overall_best_score - 1e-13:

                    overall_best_seq = a.copy()
                    print(f"[{time.time():.1f}] SA iter={it} len={n} new best {overall_best_score:.6f}")

                    a = cand
                    cur_score = cand_score
