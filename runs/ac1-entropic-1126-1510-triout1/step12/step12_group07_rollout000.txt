"""
Hybrid random‑seed + smooth‑max Adam optimiser with extended families.

The algorithm works in four stages within the allotted time budget:

1. **Broad seeding** – many analytic families (uniform, arcsine, beta,
   power‑law, exponential, sine, triangular, Gaussian, two‑block,
   random Dirichlet, sparse spikes) are generated for several lengths.
   The best few candidates (by the exact objective) are retained.

2. **Adam smooth‑max descent** – each retained seed is optimised with an
   Adam optimiser on the log‑sum‑exp surrogate of the max‑convolution.
   The temperature α is gradually increased to a large value so the
   surrogate converges to the true max.  The best sequence across all
   runs is kept.

3. **Multimax sub‑gradient polishing** – a classic exponentiated‑gradient
   (sub‑gradient) descent on the exact max‑convolution is applied to the
   current best sequence for a short period.

4. **Greedy pairwise‑swap walk** – a cheap local search that moves a
   small amount of mass from one coordinate to another, accepting only
   improvements.  This often yields tiny final gains.

All vectors are kept on the probability simplex (non‑negative,
sum = 1); scaling is irrelevant for the objective because it is
homogeneous of degree 0.  The routine respects the user‑supplied time
budget and returns the best sequence found as a plain ``list[float]``.
"""

import time
import math
from typing import List, Tuple, Optional

import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient 2·a⋆w where w (length 2·n‑1) sums to 1.
    Implemented via FFT cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (sum(a)>0)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0)
    eps = 1e-15
    w = np.exp((p - 1) * np.log(np.clip(x, eps, 1.0))
               + (p - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(alpha,beta) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    """a_i ∝ i^p, i=1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_power_decreasing(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n-i+1)^p, i=1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = (n - i + 1) ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_exponential_decreasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * i)
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.minimum(i + 1, n - i)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    vals /= vals.sum()
    return vals

def _initial_candidates_extended(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    """Collect a wide variety of seed vectors for length n."""
    cand = []

    # deterministic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_sine(n))
    cand.append(_distribution_triangular(n))
    cand.append(_distribution_gaussian(n))

    # Symmetric beta with a fine grid (including very spiky)
    for p in (0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        cand.append(_distribution_beta_sym(n, p))

    # Asymmetric beta draws
    for _ in range(6):
        alpha = 10 ** rng.uniform(-3, 1)   # allow more extreme tails
        beta = 10 ** rng.uniform(-3, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # Power‑law families (both increasing and decreasing)
    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))
        cand.append(_distribution_power_decreasing(n, p))

    # Exponential families (both directions)
    for lam in (0.001, 0.003, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential_increasing(n, lam))
        cand.append(_distribution_exponential_decreasing(n, lam))

    # Two‑block spikes with varying fractions
    for block_frac in (0.02, 0.04, 0.07, 0.10, 0.15):
        cand.append(_distribution_two_blocks(n, block_frac))

    # Random Dirichlet draws with varied concentration
    for _ in range(8):
        alpha = 10 ** rng.uniform(-2, 0)   # 0.01 … 1.0
        cand.append(_distribution_dirichlet(n, rng, alpha=alpha))

    # Random sparse patterns of various densities
    for density in (0.001, 0.003, 0.005, 0.01):
        cand.append(_distribution_random_sparse(n, rng, density=density))

    # Completely random probability vectors
    for _ in range(5):
        v = rng.random(n)
        v = np.maximum(v, 0.0)
        v /= v.sum()
        cand.append(v)

    return cand

def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector."""
    shift = np.max(z)
    exp_z = np.exp(z - shift)
    s = exp_z.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / len(z))
    return exp_z / s

def _adam_optimize(z0: np.ndarray,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   alpha0: float = 1.0,
                   alpha_max: float = 1000.0,
                   growth_factor: float = 1.02,
                   lr: float = 0.12,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   max_iter: int = 300000,
                   eval_interval: int = 300) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate (log‑sum‑exp).
    The temperature α grows geometrically up to alpha_max.
    """
    n = z0.shape[0]
    z = z0.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        # probability vector on simplex
        a = _softmax(z)

        # smooth‑max weights
        conv = _fft_self_convolve(a)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # exact gradient of smooth surrogate
        grad_a = _gradient_from_weights(a, w)

        # chain‑rule through softmax Jacobian
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        z = z - step

        # temperature schedule
        if t % 500 == 0:
            alpha = min(alpha * growth_factor, alpha_max)

        # periodic exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score

def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.6,
                                 max_iter: int = 60000,
                                 eval_interval: int = 300) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on exact max‑convolution.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)

        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] multimax it={t} new best {best_score:.6f}")

    return best_seq, best_score

def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer local walk: move a random fraction of mass
    from one index to another, accepting only improvements.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        # choose a distinct destination
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction of the source mass (up to 30%)
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Main entry point.  Runs the hybrid optimisation within the time budget
    and returns the best found non‑negative step‑function (list of floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # 1. Broad seeding
    top_seeds, best_score, best_seq = _collect_initial_seeds(rng, deadline)

    if best_seq is None:
        # fallback to uniform distribution of a modest length
        n = 4000
        best_seq = _distribution_uniform(n)
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Adam smooth‑max descent on a selection of seeds
    remaining = deadline - time.time()
    if remaining > 30.0:
        # Distribute remaining time roughly equally among the seeds
        per_seed_time = max(10.0, remaining / max(1, len(top_seeds)))
        for idx, seed_seq in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            slice_deadline = time.time() + min(per_seed_time,
                                               deadline - time.time())
            # initialise Adam parameters on the softmax‐space
            z0 = np.log(np.maximum(seed_seq, 1e-300))

            seq_out, score_out = _adam_optimize(
                z0,
                slice_deadline,
                best_seq=best_seq,
                best_score=best_score,
                alpha0=0.5,
                alpha_max=800.0,
                growth_factor=1.025,
                lr=0.15,
                max_iter=250000,
                eval_interval=250,
            )
            if score_out < best_score:
                best_score = score_out
                best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after Adam (seed {idx}) new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Multimax sub‑gradient polishing (exponentiated gradient)
    remaining = deadline - time.time()
    if remaining > 20.0:
        sub_deadline = time.time() + 0.6 * remaining
        best_seq, best_score = _subgradient_multimax_descent(
            best_seq.copy(),
            sub_deadline,
            best_seq=best_seq,
            best_score=best_score,
            lr0=0.6,
            max_iter=80000,
            eval_interval=400,
        )
        print(f"[{time.time():.1f}] after multimax polishing best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final greedy pairwise‑swap walk
    if time.time() < deadline:
        best_seq, best_score = _pairwise_swap_local(
            best_seq.copy(),
            deadline,
            best_seq=best_seq,
            best_score=best_score,
        )
        print(f"[{time.time():.1f}] after final swaps best {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    # Ensure Python list of plain floats (no np.float64 objects)
    return [float(x) for x in best_seq.tolist()]
