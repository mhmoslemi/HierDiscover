def _smooth_mirror_descent(a0, deadline, best_score, max_iters=10**7):
    a = a0.copy()
    n = a.shape[0]
    t = 0
    best_seq = a.copy()
    best_local_score = _objective(a)
    # hyperparameters
    # schedule for smoothing parameter alpha: start small and multiply.
    alpha = 0.5
    # schedule for learning rate eta: 0.5 / sqrt(t+1)
    while time.time() < deadline:
        t += 1
        # maybe gradually increase alpha every some iterations
        if t % 20000 == 0:
            alpha = min(alpha * 1.5, 30.0)  # cap
        # compute conv via FFT
        conv = _fft_self_convolve(a)
        # compute softmax weights
        # to improve numeric, subtract max
        maxv = conv.max()
        exp_vals = np.exp(alpha * (conv - maxv))
        w = exp_vals / exp_vals.sum()
        # compute gradient via correlation
        # use FFT correlation
        size = 1 << ((3 * n - 2 - 1).bit_length())
        A = np.fft.rfft(a, n=size)
        W = np.fft.rfft(w, n=size)
        corr_full = np.fft.irfft(A * np.conj(W), n=size)
        # correlation result length = size, but we need first (3n-2) entries valid
        # slice
        corr = corr_full[: (3 * n - 2)]
        # gradient indices offset
        start = 2 * n - 2
        grad = 2.0 * corr[start : start + n]
        # update with EG
        eta = 0.5 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        # avoid all zeros
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s
        # evaluate occasionally
        if t % 5000 == 0:
            cur_score = _objective(a)
            if cur_score < best_local_score - 1e-12:
                best_local_score = cur_score
                best_seq = a.copy()
                # also update global best
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    # the caller will capture best_seq after returning.
                print(f"[{time.time():.1f}] iter={t} smooth best {cur_score:.6f}")
        # optional early break if improvement stagnates
        if t > 2000000 and time.time() + 0.1 > deadline:
            break
    return best_seq, best_local_score

"""
Hybrid Construction for Low‑Autocorrelation Step Function

The algorithm proceeds in three stages:

1.  Analytic seeding – a rich palette of parametric families (uniform,
    arcsine, beta, Gaussian, power‑inverse‑sqrt, two‑block, sparse,
    Dirichlet) is instantiated for several lengths.  All candidates are
    normalised to sum = 1 and the exact objective
        2·n·max(conv(a,a))/sum(a)²
    is evaluated.  The best few seeds are retained.

2.  Smoothed mirror‑descent – the non‑smooth L∞ objective is replaced by
    a soft‑max (log‑sum‑exp) with a temperature α that is slowly increased.
    The gradient of the smoothed objective is obtained analytically as a
    cross‑correlation and an exponentiated‑gradient (entropy mirror) step
    is taken, followed by projection onto the probability simplex.
    This stage refines each elite seed.

3.  Greedy polishing – cheap random mass‑transfer moves are accepted only
    if they improve the true objective, removing remaining spikes.

The best sequence found (a plain list of non‑negative floats) is returned.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)²."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine_offset(n: int, c: float) -> np.ndarray:
    """Arcsine density with offset c>0."""
    i = np.arange(n, dtype=np.float64)
    left = i + c
    right = (n - 1 - i) + c
    vals = 1.0 / np.sqrt(left * right)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        return _distribution_uniform(n)
    return vals / total


def _distribution_power_inv_sqrt(n: int, p: float) -> np.ndarray:
    """Generalised arcsine family w_i ∝ ((i+0.5)*(n‑i‑0.5))^{-p/2}."""
    i = np.arange(n, dtype=np.float64)
    left = i + 0.5
    right = (n - 1 - i) + 0.5
    vals = (left * right) ** (-p / 2.0)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        return _distribution_uniform(n)
    return vals / total


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int,
                                rng: np.random.Generator,
                                density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    return vals / vals.sum()


def _distribution_dirichlet(n: int,
                            rng: np.random.Generator,
                            alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    return vals / vals.sum()


def _distribution_beta(n: int, a: float, b: float) -> np.ndarray:
    """Symmetric (or not) beta density on {0,…,n‑1}."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    vals = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        return _distribution_uniform(n)
    return vals / total


def _distribution_gaussian(n: int, sigma_frac: float = 0.2) -> np.ndarray:
    """Centered Gaussian, sigma expressed as a fraction of the length."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    vals = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    if vals.sum() == 0.0:
        return _distribution_uniform(n)
    return vals / vals.sum()

def _smooth_mirror_descent(a0: np.ndarray,
                           deadline: float,
                           global_best_score: float) -> tuple[np.ndarray, float]:
    """
    Smoothed L∞ mirror descent.
    Uses a soft‑max approximation of max(conv) with a temperature α that
    grows during the run.  Exponentiated‑gradient updates keep the iterate
    on the probability simplex.
    Returns the best sequence seen and its exact objective.
    """
    a = a0.copy()
    n = a.shape[0]
    # FFT size sufficient for a (length n) and w (length 2n‑1)
    max_len = 3 * n - 2
    size = 1 << ((max_len - 1).bit_length())

    best_seq = a.copy()
    best_score = _objective(a)

    t = 0
    alpha = 0.5          # initial smoothing
    next_alpha_inc = 20000

    while time.time() < deadline:
        t += 1

        # ----- FFT of the current distribution -----
        A = np.fft.rfft(a, n=size)

        # ----- exact self‑convolution (needed for the soft‑max) -----
        conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]

        # ----- soft‑max weights over the convolution entries -----
        m = conv.max()
        exp_vals = np.exp(alpha * (conv - m))          # stable soft‑max
        w = exp_vals / exp_vals.sum()

        # ----- correlation a ⋆ w (gradient of the smoothed max) -----
        W = np.fft.rfft(w, n=size)
        corr_full = np.fft.irfft(A * np.conj(W), n=size)[:max_len]
        start = 2 * n - 2
        grad = 2.0 * corr_full[start:start + n]        # g_i = 2·∑_j a_j w_{i+j}

        # ----- exponentiated‑gradient step (mirror descent) -----
        eta = 0.5 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # ----- projection onto the simplex (simple renormalisation) -----
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # ----- schedule for α (more aggressive as we converge) -----
        if t == next_alpha_inc:
            alpha = min(alpha * 1.2, 30.0)   # cap to avoid overflow
            next_alpha_inc += 20000

        # ----- occasional exact evaluation -----
        if t % 5000 == 0:
            cur_score = _objective(a)
            if cur_score < best_score - 1e-13:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}]   smooth iter={t} best={best_score:.6f}")
            # early exit if we have already beaten the global bound by a good margin
            if best_score < global_best_score - 1e-6:
                global_best_score = best_score

    return best_seq, best_score

def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Random mass‑transfer walk that accepts only improving moves."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue
        delta = a[i] * rng.random() * 0.30               # up to 30 % of source
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur_score = _objective(new_a)
        if cur_score < best_score - 1e-13:
            a = new_a
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}]   swap {i}->{j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score

            seeds_this_len.append(_distribution_beta(n, a, a))
        for _ in range(3):
            a = 0.1 + 1.9 * rng.random()
            b = 0.1 + 1.9 * rng.random()
            seeds_this_len.append(_distribution_beta(n, a, b))

        # Gaussian shapes
        for sigma in (0.12, 0.18, 0.25, 0.35):
            seeds_this_len.append(_distribution_gaussian(n, sigma))

        # random sparse and Dirichlet draws
        seeds_this_len.append(_distribution_random_sparse(n, rng, density=0.008))
        for _ in range(2):
            seeds_this_len.append(_distribution_dirichlet(n, rng,
                                                          alpha=0.2 + 0.8 * rng.random()))

        # previously known solution (if available) – try to reuse its shape
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
            if prev.ndim == 1 and prev.size > 0:
                if prev.shape[0] != n:
                    # resample to the current length by simple interpolation
                    x_old = np.linspace(0, 1, prev.shape[0])
                    x_new = np.linspace(0, 1, n)
                    prev = np.interp(x_new, x_old, prev)
                prev = np.maximum(prev, 0.0)
                if prev.sum() == 0.0:
                    prev = _distribution_uniform(n)
                prev = prev / prev.sum()
                seeds_this_len.append(prev)
        except Exception:
            pass

        # ------------------------------------------------------------------
        # Evaluate every seed, keep the elite ones
        for cand in seeds_this_len:
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                cand = _distribution_uniform(n)
                s = cand.sum()
            cand = cand / s
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] !!! new global best len={n} score={sc:.6f}")
            global_top.append((sc, cand.copy()))

        # keep only a small elite set globally
        global_top.sort(key=lambda x: x[0])
        global_top = global_top[:12]

    # ------------------------------------------------------------------
    # 2.  Smoothed mirror‑descent refinement of elite seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and global_top:
        # allocate time roughly equally among the elite seeds
        elite = [seq for _, seq in global_top[:6]]
        per_seed = max(5.0, remaining / len(elite))
        for seed_seq in elite:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            refined_seq, refined_score = _smooth_mirror_descent(
                seed_seq.copy(),
                sub_deadline,
                overall_best_score,
            )
            if refined_score < overall_best_score:
                overall_best_score = refined_score
                overall_best_seq = refined_seq.copy()
                print(f"[{time.time():.1f}] after smooth new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Final greedy polishing
    if overall_best_seq is not None and time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            overall_best_seq.copy(),
            overall_best_score,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # make sure the returned list respects the non‑negative requirement
    result = np.maximum(overall_best_seq, 0.0).tolist()
    return result
