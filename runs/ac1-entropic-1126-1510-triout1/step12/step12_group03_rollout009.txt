                  lr=0.5, beta_start=0.5, beta_end=20.0, eval_every=200):
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t=0
    while time.time() < time_limit:
        t+=1
        beta = beta_start + (beta_end - beta_start) * min(1.0, t / 5000)
        # compute smooth objective and gradient
        conv = _fft_self_convolve(a)
        z = beta * conv
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)
        sum_exp = np.sum(exp_z)
        p = exp_z / sum_exp
        # gradient via correlation
        a_rev = a[::-1]
        size = 1 << ((len(p) + len(a_rev) - 1 - 1).bit_length())
        fft_p = np.fft.rfft(p, n=size)
        fft_a_rev = np.fft.rfft(a_rev, n=size)
        corr = np.fft.irfft(fft_p * fft_a_rev, n=size)
        grad = 4.0 * n * corr[:n][::-1]
        # Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

"""
search for a non‑negative step‑function minimizing evaluate_sequence().
The pipeline:

1. Scan a few analytic families (uniform, arcsine, beta‑type, edge‑weighted,
   sine, cosine, quadratic) on several lengths and keep the best few seeds.
2. Refine each seed with a smooth “log‑sum‑exp” Adam descent.
   The smoothed objective approximates 2·n·max(conv) and provides a
   differentiable surrogate whose gradient is computed by a single FFT‑based
   correlation (O(n log n)).
3. Finish with a cheap greedy pairwise mass‑transfer hill‑climb.
All work respects the supplied time budget (default 1000 s) and returns a
plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1.
    The input must be non‑negative; the function does **not** perform the
    safety clipping present in evaluate_sequence().
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the grid {0,…,n‑1} (p>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0) when p<1
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """
    Edge‑weighted family: w_i ∝ ((i+½)*(n‑i‑½))^{‑p}.
    For p>0 the distribution is heavy at the two ends.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    """Parabolic shape peaking in the middle."""
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.01) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Smoothed Adam descent -------------------------------------------------
def _adam_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr: float = 0.4,
    beta_start: float = 0.5,
    beta_end: float = 20.0,
    eval_every: int = 300,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑sum‑exp smooth surrogate of the max‑convolution.
    Returns the best sequence seen (exact objective).
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    while time.time() < time_limit:
        t += 1

        # --- smoothness schedule ---
        beta = beta_start + (beta_end - beta_start) * min(1.0, t / 4000)

        # self‑convolution
        conv = _fft_self_convolve(a)                     # length 2n‑1
        z = beta * conv
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)                       # stability
        sum_exp = np.sum(exp_z)
        p = exp_z / sum_exp                              # softmax weights

        # gradient = 4·n·corr(p, a) (cross‑correlation)
        a_rev = a[::-1]
        conv_len = len(p) + len(a_rev) - 1               # = 3n‑2
        size = 1 << ((conv_len - 1).bit_length())
        fft_p = np.fft.rfft(p, n=size)
        fft_a_rev = np.fft.rfft(a_rev, n=size)
        corr = np.fft.irfft(fft_p * fft_a_rev, n=size)   # real
        grad = 4.0 * n * corr[:n][::-1]                  # first n values, reversed

        # Adam updates (minimise → move against gradient)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # project back onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # periodic exact evaluation
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] adam it={t} new global best {best_score:.6f}")

    return best_seq, best_score

# Greedy pairwise mass transfer ----------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from one index to another.
    Keep the move only if it strictly improves the exact objective.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score

                best_score=overall_best_score,
                lr=0.45,
                beta_start=0.4,
                beta_end=25.0,
                eval_every=250,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] after Adam seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Final greedy pairwise polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy swap polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (2·n·max(conv)): {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list – evaluate_sequence will handle its own clipping.
    return overall_best_seq.tolist()
