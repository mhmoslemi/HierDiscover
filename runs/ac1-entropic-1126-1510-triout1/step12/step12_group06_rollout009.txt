    conv_f = _fft_self_convolve(f)
    M = conv_f.max()

    if not active:

    # ensure we have at most some limit of rows
    max_active = 80
    # main loop: solve LP and add violated constraints
    for _ in range(8):
        rows = sorted(active)
        m = len(rows)
        # Build dense A matrix (m x n)
        A = np.zeros((m, n), dtype=np.float64)
        for r, k in enumerate(rows):
            # vectorized fill
            j = np.arange(n)
            i = k - j
            mask = (i >= 0) & (i < n)
            A[r, mask] = f[i[mask]]

#!/usr/bin/env python
"""
Hybrid optimiser with an active‑set linear‑programming refinement.

Strategy
--------
1.  Generate a rich set of initial probability vectors (uniform,
    arcsine, Beta families, Dirichlet draws, sparse patterns and the
    previous best solution if available) for a fixed length `n`.
2.  Run a few iterations of Adam on the smooth‑max surrogate of the
    convolution maximum to obtain a decent starting point.
3.  Repeatedly solve a tiny linear programme that maximises the total
    mass of a new vector `b` under the *active* convolution constraints
    `conv(f,b) ≤ M` (where `M` is the current maximal convolution of `f`).
    Only the constraints that are close to being active are kept,
    turning an O(n²) problem into a handful of rows.
    The solution `b` is scaled to a probability vector `g` and mixed with
    the current iterate `f ← (1‑t)·f + t·g`.  This step provably never
    increases the maximal convolution and quickly flattens it.
4.  After the LP phase a cheap greedy pair‑wise mass‑transfer polishing
    step is applied.
5.  The best vector found is returned as a plain list of floats.

All updates keep the vector on the simplex (non‑negative, sum = 1),
which makes the evaluation function scale‑invariant.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from typing import List, Tuple

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_cross_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full cross‑convolution a*b using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact evaluation of the target function.
    The vector is assumed to be normalised (sum = 1), which makes the
    expression simply 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic families for seeding ------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discretised unit interval."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(alpha,beta) on the discretised unit interval."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Few non‑zero coordinates, random values."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    """Two symmetric spikes of equal mass."""
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _initial_candidates_extended_fine(n: int,
                                      rng: np.random.Generator) -> List[np.ndarray]:
    """
    Produce a diverse collection of seed vectors for a given length.
    The set is deliberately richer than the original implementation.
    """
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    for p in np.geomspace(0.01, 5.0, num=15):
        cand.append(_distribution_beta_sym(n, float(p)))

    for _ in range(8):
        alpha = 10 ** rng.uniform(-3, 1)
        beta = 10 ** rng.uniform(-3, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))

    for lam in (0.001, 0.003, 0.005, 0.01):
        cand.append(_distribution_exponential_increasing(n, lam))

    for bf in (0.03, 0.05, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac=bf))

    for a in (0.1, 0.3, 0.7, 1.0):
        cand.append(_distribution_dirichlet(n, rng, alpha=a))

    cand.append(_distribution_random_sparse(n, rng, density=0.01))
    cand.append(_distribution_random_sparse(n, rng, density=0.03))

    for _ in range(6):
        v = rng.random(n)
        v /= v.sum()
        cand.append(v)

    return cand

# Adam optimisation on the smooth‑max surrogate --------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute 2·a⋆w where w (length 2·n‑1) sums to 1.
    Implemented via FFT cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


def _smooth_max_adam_descent(a: np.ndarray,
                             deadline: float,
                             best_seq: np.ndarray,
                             best_score: float,
                             eta0: float = 0.6,
                             beta1: float = 0.9,
                             beta2: float = 0.999,
                             eps: float = 1e-8,
                             alpha0: float = 0.3,
                             alpha_mul: float = 1.25,
                             alpha_max: float = 150.0,
                             max_iter: int = 60000,
                             eval_interval: int = 500) -> Tuple[np.ndarray, float]:
    """
    Adam‑style exponentiated‑gradient descent on the smooth‑max surrogate
    f_α(a) = (1/α)·log Σ_i exp(α·conv_i(a)).
    """
    n = a.shape[0]
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)

        # soft‑max weights (stable)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        grad = _gradient_from_weights(a, w)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(-step)

        # project back onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % 300 == 0 and alpha < alpha_max:
            alpha = min(alpha * alpha_mul, alpha_max)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score

# Linear‑programming refinement (active‑set) -----------------------------------
def _lp_improve(f: np.ndarray,
                t: float,
                deadline: float) -> Tuple[np.ndarray, float]:
    """
    Perform one active‑set LP refinement step.
    Maximises Σ b_j subject to  b ≥ 0 and  conv(f,b) ≤ M.
    The solution is scaled to a probability vector `g` and mixed with `f`
    using weight `t`.
    """
    n = f.shape[0]
    conv_f = _fft_self_convolve(f)
    M = conv_f.max()

    # initialise active set with indices very close to the maximum
    thresh = M * 0.02   # 2 % band
    active = set(np.where(conv_f >= M - thresh)[0].tolist())
    if not active:
        active.add(int(np.argmax(conv_f)))

    max_active_rows = 80   # never let the LP grow too big
    for _ in range(8):     # at most a few re‑solves
        rows = sorted(active)
        m = len(rows)

        # Build dense matrix A (m × n) – tiny compared with n²
        A = np.zeros((m, n), dtype=np.float64)
        for r, k in enumerate(rows):
            j = np.arange(n)
            i = k - j
            mask = (i >= 0) & (i < n)
            A[r, mask] = f[i[mask]]

        c = -np.ones(n, dtype=np.float64)   # maximise Σb  →  minimise –Σb
        bounds = [(0, None)] * n

        # Solve LP; the “highs” method is fast and accepts dense A_ub.
        res = linprog(c,
                      A_ub=A,
                      b_ub=M * np.ones(m),
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True})
        if not res.success:
            break

        b = res.x

        # check all constraints (full convolution) – add any violators
        conv_fb = _fft_cross_convolve(f, b)
        violating = np.where(conv_fb > M + 1e-12)[0]
        if violating.size == 0:
            # feasible solution found
            break

        # add up to five most violated constraints
        excess = conv_fb[violating] - M
        worst = violating[np.argsort(-excess)[:5]]
        active.update(int(k) for k in worst)

        # prune if we exceed the maximum rows budget
        if len(active) > max_active_rows:
            # keep the original rows plus the largest violations only
            active = set(rows)
            extra = sorted(violating,
                           key=lambda k: conv_fb[k],
                           reverse=True)[:max_active_rows - len(rows)]
            active.update(extra)

    sum_b = b.sum()
    if sum_b <= 0.0:
        return f, _objective(f)

    # scale `b` down to a probability vector (guaranteed not to violate constraints)
    if sum_b > 1.0:
        g = b / sum_b
    else:
        # sum(b) ≤ 1 would require scaling up, which could break the constraints.
        # In this unlikely case we simply keep `b` as‑is and renormalise after mixing.
        g = b / max(sum_b, 1e-12)

    new_f = (1.0 - t) * f + t * g
    new_f = np.maximum(new_f, 0.0)
    s = new_f.sum()
    if s == 0.0:
        new_f = np.full(n, 1.0 / n)
    else:
        new_f = new_f / s

    if time.time() > deadline:
        # safety – return the old vector if we are out of time
        return f, _objective(f)

    new_score = _objective(new_f)
    return new_f, new_score

# Greedy local polishing ------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Greedy pair‑wise mass transfer that accepts improving moves."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = int(rng.choice(src))
        j = int(rng.integers(0, n - 1))
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Main entry point used by the evaluation harness.
    Returns a list of non‑negative heights that (hopefully) minimises
    ``evaluate_sequence`` within the allotted time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a single length that balances flexibility and speed.
    # 3268 is the length of the previously best construction.
    n = 3268
    if budget_s < 300:
        n = 2000   # smaller when the budget is tight

    # ------------------------------------------------------------------
    # 1.  Build a pool of seed vectors.
    seed_vecs = _initial_candidates_extended_fine(n, rng)

    # optionally add a few completely random vectors
    for _ in range(4):
        v = rng.random(n)
        v /= v.sum()
        seed_vecs.append(v)

    # incorporate previous best if present
    try:
        prev = globals().get("height_sequence_1")
        if isinstance(prev, (list, np.ndarray)):
            prev = np.asarray(prev, dtype=np.float64)
            if prev.ndim == 1 and prev.size > 0:
                prev = np.maximum(prev, 0.0)
                prev_sum = prev.sum()
                if prev_sum > 0.0:
                    prev /= prev_sum
                seed_vecs.append(prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2.  Pick the best seed according to the exact objective.
    best_f = None
    best_score = float("inf")
    for v in seed_vecs:
        if time.time() >= deadline:
            break
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s == 0.0:
            continue
        v = v / s
        sc = _objective(v)
        if sc < best_score:
            best_score = sc
            best_f = v.copy()
            print(f"[{time.time():.1f}] initial best {best_score:.6f}")

    if best_f is None:
        best_f = np.full(n, 1.0 / n)
        best_score = _objective(best_f)

    # ------------------------------------------------------------------
    # 3.  Light Adam smoothing – helps to leave obvious bad local minima.
    best_f, best_score = _smooth_max_adam_descent(best_f.copy(),
                                                  deadline,
                                                  best_seq=best_f,
                                                  best_score=best_score,
                                                  max_iter=25000,
                                                  eval_interval=500)

    # ------------------------------------------------------------------
    # 4.  Active‑set linear‑programming refinement loop.
    #    We keep trying decreasing mixing weights `t` until no improvement.
    while time.time() < deadline - 5.0:
        improvement = False
        for t in (0.5, 0.3, 0.2, 0.1):
            if time.time() >= deadline:
                break
            new_f, new_score = _lp_improve(best_f.copy(), t, deadline)
            if new_score < best_score - 1e-12:
                best_f, best_score = new_f, new_score
                print(f"[{time.time():.1f}] LP step t={t:.2f} new best {best_score:.6f}")
                improvement = True
                break
        if not improvement:
            # No LP move could improve – exit the loop.
            break

    # ------------------------------------------------------------------
    # 5.  Final cheap greedy polishing.
    if time.time() < deadline:
        best_f, best_score = _pairwise_swap_local(best_f.copy(),
                                                  deadline,
                                                  best_seq=best_f,
                                                  best_score=best_score)

    print(f"Final best score: {best_score:.6f} (length={len(best_f)})")
    # Return a plain Python list as required by the harness.
    return best_f.tolist()
