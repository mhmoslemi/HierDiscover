"""
Hybrid optimizer for the self‑convolution max‑ratio problem.

The algorithm follows three stages within the allowed time budget:

1. **Broad seed generation** – a variety of analytic families (uniform,
   arcsine, Beta, Dirichlet, exponential, power‑law, sparse, two‑block,
   sine) are instantiated for several lengths.  All seeds are normalised
   to sum = 1 and their exact objective 2·n·max(conv(a,a)) is evaluated.
   The best few (elite) seeds are retained.

2. **Deterministic refinement** – each elite seed undergoes
   exponentiated‑gradient descent.
   *Phase A* uses a soft‑max approximation of the maximal convolution entry,
   *Phase B* uses the true hard‑max sub‑gradient.  Both phases work on the
   probability simplex via multiplicative updates and renormalisation.

3. **Stochastic polishing** – the current global best is further improved
   by a simulated‑annealing walk that repeatedly transfers a random
   fraction of mass between two coordinates.  Moves that improve the
   objective are always accepted; worsening moves are accepted with a
   temperature‑controlled probability, allowing the search to escape local
   minima.

The routine respects the `budget_s` argument (default 1000 s), prints
intermediate progress and finally returns the best sequence found as a
plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective   2·n·max(conv(a,a)) / (sum(a)**2).
    The vector a is assumed non‑negative.
    """
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
def _gradient_max_index(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the single maximal convolution entry conv[k]."""
    n = a.shape[0]
    g = np.zeros_like(a)

    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i = np.arange(i_min, i_max + 1, dtype=np.int64)
    j = k - i

    eq_mask = (i == j)
    if np.any(eq_mask):
        g[i[eq_mask]] += 2.0 * a[i[eq_mask]]

    if np.any(~eq_mask):
        ii = i[~eq_mask]
        jj = j[~eq_mask]
        g[ii] += a[jj]
        g[jj] += a[ii]

    return g


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient for a soft‑max weight vector w (sum(w)=1).
    Implements 2·a⋆w using FFT‑based correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.size - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _softmax_grad_descent(a: np.ndarray, deadline: float,
                          best_seq: np.ndarray, best_score: float,
                          lr0: float = 0.9,
                          max_iter: int = 60000,
                          eval_interval: int = 500,
                          alpha_start: float = 0.5,
                          alpha_end: float = 80.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a soft‑max approximation of
    max(conv(a,a)).
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        # linear temperature schedule
        alpha = alpha_start + (alpha_end - alpha_start) * t / max_iter

        # stable soft‑max
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # renormalise onto the simplex
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= total

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] soft‑max it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _max_grad_descent(a: np.ndarray, deadline: float,
                      best_seq: np.ndarray, best_score: float,
                      lr0: float = 0.5,
                      max_iter: int = 80000,
                      eval_interval: int = 500) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent using the true maximal convolution entry.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())

        grad = _gradient_max_index(a, k)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= total

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hard‑max it={t} idx={k} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float,
                         initial_temp: float = 2e-2,
                         final_temp: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Temperature‑controlled random mass‑transfer walk.
    Occasionally accepts uphill moves to escape local minima.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = best_score
    best_local_seq = best_seq.copy()
    best_local_score = best_score
    rng = np.random.default_rng()

    start_time = time.time()
    total_time = deadline - start_time
    if total_time <= 0:
        return best_seq, best_score

    while time.time() < deadline:
        # linearised progress fraction for temperature schedule
        frac = (time.time() - start_time) / total_time
        temp = initial_temp * (final_temp / initial_temp) ** frac
        # adapt max move size to temperature (larger jumps early)
        max_move = min(0.3, temp * 10.0 + 0.01)

        # pick a source with non‑trivial mass
        src_candidates = np.where(cur_a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        j = rng.integers(n)
        if j == i:
            continue

        delta = cur_a[i] * rng.random() * max_move
        if delta < 1e-14:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        # renormalise numerically (sum should stay 1 but floating error may accumulate)
        s = new_a.sum()
        if s <= 0:
            continue
        new_a /= s

        new_score = _objective(new_a)

        if new_score < cur_score:
            cur_a = new_a
            cur_score = new_score
            if new_score < best_local_score - 1e-14:
                best_local_score = new_score
                best_local_seq = new_a.copy()
                print(f"[{time.time():.1f}] SA new best {best_local_score:.6f}")
        else:
            # Metropolis acceptance
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                cur_a = new_a
                cur_score = new_score

    return best_local_seq, best_local_score

# Analytic families (identical to the previous state‑of‑the‑art code)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * x / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Broad pool of analytic seeds for a given length."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_sine(n))
    cand.append(_distribution_two_blocks(n, block_frac=0.07))

    for p in np.linspace(0.05, 1.0, 12):
        cand.append(_distribution_beta_sym(n, p))

    for _ in range(8):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    for p in (0.5, 1.0, 2.0, 3.0):
        cand.append(_distribution_power_increasing(n, p))

    for lam in (0.001, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential_increasing(n, lam))

    for dens in (0.005, 0.01, 0.02):
        cand.append(_distribution_random_sparse(n, rng, density=dens))

    for a in (0.1, 0.3, 0.5, 1.0):
        cand.append(_distribution_dirichlet(n, rng, alpha=a))

    return cand
