"""
Propose a low‑autocorrelation (low‑max‑convolution) non‑negative step function.

Algorithm
---------
1.  Seed generation
    * Several analytic families (uniform, arcsine with offset, power‑inverse‑sqrt,
      two‑block, sparse random, Dirichlet) are instantiated for a handful of
      lengths (≈2000–6000).  For the arcsine and power families a fine grid of
      the shape parameter is scanned.  All candidates are normalised to sum = 1
      and evaluated with the exact objective.
    * The best candidate (lowest objective) is kept as the start point.
    * If a previous solution `height_sequence_1` exists it is also considered.

2.  Simulated annealing (global search)
    * Starting from the best seed we repeatedly move a random amount of mass
      from a source index to a destination index, keeping the total sum exactly
      one.
    * The move is accepted if it improves the objective or with probability
      exp(‑Δ/ T) where Δ is the increase and T is a temperature that follows a
      geometric cooling schedule (T₀≈0.1 → T_end≈1e‑6).
    * The temperature schedule is tied to the elapsed fraction of the total
      time budget, allowing large exploratory moves early and fine‑grained
      moves later.

3.  Greedy polishing (local optimisation)
    * After the annealing phase a cheap hill‑climbing pass is performed:
      random mass‑transfer moves are accepted only when they strictly improve
      the objective.  This removes tiny spikes that the stochastic search may
      have missed.

The whole procedure respects the time budget (`budget_s` seconds).  The
returned list contains non‑negative floats; the value of the objective is
scale‑invariant, so the sequence is normalised to sum = 1 throughout the
search.  The implementation uses only NumPy (FFT based convolution) and
standard library modules.
"""

import time
import math

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)² (sum is assumed >0)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine_offset(n: int, c: float) -> np.ndarray:
    """Arcsine pdf with offset c>0 (c controls the “flatness”)."""
    i = np.arange(n, dtype=np.float64)
    left = i + c
    right = (n - 1 - i) + c
    vals = 1.0 / np.sqrt(left * right)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _distribution_power_inv_sqrt(n: int, p: float) -> np.ndarray:
    """
    Generalised arcsine family:
        w_i ∝ ((i+0.5)*(n-i-0.5))^{-p/2}.
    p=1 yields the classic arcsine.
    """
    i = np.arange(n, dtype=np.float64)
    left = i + 0.5
    right = (n - 1 - i) + 0.5
    vals = (left * right) ** (-p / 2.0)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    """Two blocks of equal mass at both ends of the array."""
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Place a small random set of non‑zero entries."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    """Draw a Dirichlet vector with concentration α."""
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals

def _best_seed_across_lengths(rng: np.random.Generator,
                              lengths: list[int]) -> tuple[np.ndarray, float]:
    """Exhaustively scan a few analytic families for each length and return
    the best (lowest objective) vector together with its score."""
    best_seq = None
    best_score = float("inf")

    for n in lengths:
        # Uniform
        cand = _distribution_uniform(n)
        sc = _objective(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc

        # Arcsine offset – dense grid
        for c in np.linspace(0.05, 5.0, 101):
            cand = _distribution_arcsine_offset(n, float(c))
            sc = _objective(cand)
            if sc < best_score:
                best_seq, best_score = cand, sc

        # Power‑inverse‑sqrt family – dense grid
        for p in np.linspace(0.4, 1.6, 25):
            cand = _distribution_power_inv_sqrt(n, float(p))
            sc = _objective(cand)
            if sc < best_score:
                best_seq, best_score = cand, sc

        # Two‑block shapes
        for bf in np.linspace(0.01, 0.20, 20):
            cand = _distribution_two_blocks(n, float(bf))
            sc = _objective(cand)
            if sc < best_score:
                best_seq, best_score = cand, sc

        # Random sparse draws
        for _ in range(5):
            cand = _distribution_random_sparse(n, rng, density=0.01)
            sc = _objective(cand)
            if sc < best_score:

def _simulated_annealing(start_seq: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Global stochastic search.
    Moves mass between random indices; Metropolis acceptance with exponential
    cooling.  Returns the best sequence found.
    """
    cur = start_seq.copy()
    cur_score = _objective(cur)

    best_seq = cur.copy()
    best_score = cur_score

    n = cur.shape[0]

    # Annealing parameters
    T0 = 0.12          # start temperature
    T_end = 1e-6       # final temperature
    start_time = time.time()
    total = deadline - start_time

    # maximum fraction of source mass that may be moved in a single step
    max_frac = 0.35

    while time.time() < deadline:
        # temperature based on elapsed fraction (geometric cooling)
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / total)
        T = T0 * (T_end / T0) ** frac

        # pick a source that actually has mass
        src_candidates = np.where(cur > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        j = rng.integers(n)
        if j == i:
            continue

        # Amount of mass to transfer – up to max_frac of the source
        delta = cur[i] * max_frac * rng.random()
        if delta < 1e-15:
            continue

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # Numerical safety – renormalise very rarely
        if abs(new_seq.sum() - 1.0) > 1e-12:
            new_seq /= new_seq.sum()

        new_score = _objective(new_seq)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis acceptance
            exp_arg = -(new_score - cur_score) / max(T, 1e-12)
            if exp_arg > -700:               # avoid underflow
                if rng.random() < math.exp(exp_arg):
                    accept = True

        if accept:
            cur = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_seq = new_seq.copy()
                best_score = new_score
                print(f"[{time.time():.1f}] SA new best {best_score:.6f}")

    return best_seq, best_score

def _greedy_polish(start_seq: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing: random mass transfers that are kept only if they
    strictly improve the objective.
    """
    cur = start_seq.copy()
    cur_score = _objective(cur)
    best_seq = cur.copy()
    best_score = cur_score
    n = cur.shape[0]

    while time.time() < deadline:
        src_candidates = np.where(cur > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        j = rng.integers(n)
        if j == i:
            continue
        # small transfer – up to 10 % of source
        delta = cur[i] * 0.10 * rng.random()
        if delta < 1e-15:
            continue
        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if abs(new_seq.sum() - 1.0) > 1e-12:
            new_seq /= new_seq.sum()
        new_score = _objective(new_seq)
        if new_score < cur_score - 1e-14:
            cur = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_seq = new_seq.copy()
                best_score = new_score
                print(f"[{time.time():.1f}] polish new best {best_score:.6f}")

    return best_seq, best_score

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 1e-12:
        # fallback to uniform if something went terribly wrong
        best_seq = _distribution_uniform(best_seq.shape[0])

    print(f"Returning sequence of length {best_seq.shape[0]} with score {best_score:.6f}")
    return best_seq.tolist()
