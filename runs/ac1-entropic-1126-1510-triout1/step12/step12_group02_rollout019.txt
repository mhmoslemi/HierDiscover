def _projected_gradient_descent(a, deadline, best_seq, best_score):
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        k = int(np.argmax(conv))
        M = conv[k]
        S = a.sum()
        # compute gradient of J
        grad = np.empty_like(a)
        # compute shift of a to align with k
        if k < n:
            # valid indices 0..k inclusive
            grad[:k+1] = a[k::-1] - M / S
            grad[k+1:] = -M / S
        else:
            # k >= n
            start = k - (n - 1)
            grad[:start] = -M / S
            grad[start:] = a[start:][::-1] - M / S
        # update additive
        eta = 0.1 / np.sqrt(t)   # schedule
        a = a - eta * grad
        # clip negative values
        a = np.maximum(a, 0.0)
        # project onto simplex sum=1
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / len(a))
        if t % 100 == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"{time.time():.1f} GD new best {best_score:.6f}")
    return best_seq, best_score
```

def _grad_softmax(a, conv, beta):
    # conv: self-convolution of a (length 2n-1)
    max_c = conv.max()
    # compute softmax probabilities
    p = np.exp(beta * (conv - max_c))
    p_sum = p.sum()
    if p_sum == 0:
        p = np.full_like(p, 1.0 / p.shape[0])
    else:
        p /= p_sum
    # compute weighted avg L = sum(p * conv)
    L = np.dot(p, conv)
    # correlation grad = 2 * corr - L
    # compute corr = sum_j p_j * a_{i+j}
    # convolution of p with a_rev
    n = a.shape[0]
    size = 1 << ((3 * n - 2) - 1).bit_length()
    P = np.fft.rfft(p, n=size)
    A_rev = np.fft.rfft(a[::-1], n=size)
    corr_full = np.fft.irfft(P * A_rev, n=size)
    # corr_full length = size, but needed slice [n-1: n-1+n]
    corr = corr_full[n-1:n-1+n][::-1]  # slice length n, reverse
    grad = 2.0 * corr - L
    return grad

"""
Propose a non‑negative step function with a low
evaluate_sequence() value.

The algorithm is a three‑stage optimisation:

1. **Seed generation** – use the analytic “arcsine” family,
   which is known to flatten the self‑convolution, for several
   lengths (2000 … 4000).  Keep the best few seeds.

2. **Smooth projected gradient descent** – minimise the
   smoothed objective
        Lβ(a) = (1/β)·log Σk exp(β·c_k)    (c = a*a)
   whose gradient is a simple weighted correlation.
   The temperature β is increased gradually so that the
   optimisation first follows a smooth landscape and later
   focuses on the true maximum.  After each update the
   vector is projected onto the probability simplex.

3. **Final pairwise annealing** – a cheap stochastic polishing
   that moves a small fraction of mass between two bins and
   accepts improving moves (and occasional worsening moves
   early on).

All steps respect the 1000 s budget and use only NumPy
(FFT‑based convolutions) – no external LP solvers are needed.
The returned list can be fed directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast linear self‑convolution using real FFT
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective (same as evaluate_sequence but much faster)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) / (sum(a)²)  –  a may be any non‑negative vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_arcsine(n: int) -> np.ndarray:
    """Arcsine‑like shape: w_i ∝ 1/√((i+½)(n‑i‑½))."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Smoothed gradient of the max‑convolution (softmax surrogate)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of Lβ(a) = (1/β)·log Σ exp(β·c_k)  where c = a*a.
    Result is 2·corr – L, where corr_i = Σ_k p_k·a_{k-i}
    and p_k = softmax(β·c_k).  Computed via FFT for speed.
    """
    # softmax probabilities (stable)
    max_c = conv.max()
    prob = np.exp(beta * (conv - max_c))
    prob_sum = prob.sum()
    if prob_sum == 0.0:
        prob = np.full_like(prob, 1.0 / prob.shape[0])
    else:
        prob /= prob_sum

    # L = Σ p_k·c_k
    L = float(np.dot(prob, conv))

    # compute corr_i = Σ_k p_k·a_{k-i}
    n = a.shape[0]
    # convolution of prob with reversed a
    size = 1 << ((3 * n - 2) - 1).bit_length()          # enough for length 3n‑2
    P = np.fft.rfft(prob, n=size)
    A_rev = np.fft.rfft(a[::-1], n=size)
    corr_full = np.fft.irfft(P * A_rev, n=size)
    # slice the centre part (indices n‑1 … n‑1+n‑1)
    corr = corr_full[n - 1 : n - 1 + n][::-1]          # length n, reversed to match a_i
    grad = 2.0 * corr - L
    return grad


# ----------------------------------------------------------------------
# Projected gradient descent with a growing soft‑max temperature
# ----------------------------------------------------------------------
def _smooth_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Minimise the true objective using a smooth approximation of the max.
    The update is:
        a ← a – η·grad  (grad from the soft‑max surrogate)
    followed by Euclidean projection onto the probability simplex.
    """
    n = a.shape[0]
    t = 0
    beta = 0.5                # start with a smooth landscape
    eta0 = 0.5                # initial step size

    # pre‑compute the FFT size for the correlation step (depends on n only)
    fft_size = 1 << ((3 * n - 2) - 1).bit_length()

    while time.time() < deadline:
        t += 1
        # current self‑convolution
        conv = _fft_self_convolve(a)

        # gradient of the soft‑max surrogate
        grad = _softmax_gradient(a, conv, beta)

        # decreasing step size
        eta = eta0 / math.sqrt(t)
        a = a - eta * grad

        # enforce non‑negativity and renormalise to a probability vector
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # slowly increase the temperature – after a few hundred iterations
        # we are close to the true max.
        if t % 200 == 0:
            beta = min(beta * 1.07, 80.0)   # cap to avoid overflow

        # occasional real‑objective evaluation
        if t % 300 == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] GD it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise mass‑transfer annealing (same idea as the previous runner)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins."""
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    trials = 0
    while time.time() < deadline:
        trials += 1
        src = rng.choice(np.where(a > 1e-12)[0])
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1
        delta = a[src] * rng.random() * 0.35  # up to 35 % of source mass
        if delta < 1e-15:
            continue
        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta
        cur = _objective(new_a)

        # simple linear cooling schedule
        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            prob = math.exp(-(cur - best_score) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal src={src}→dst={dst} δ={delta:.3e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point -----------------------------------------------------
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the 1000 s budget
    (a small safety margin is kept) and returns a plain ``list[float]``.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0      # safety margin

    # ------------------------------------------------------------------
    # 1.  Seed generation – a few arcsine vectors of moderate length.
    # ------------------------------------------------------------------
    lengths = [2000, 2500, 3000, 3500, 4000]
    best_seq: np.ndarray | None = None
    best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        a = _distribution_arcsine(n).astype(np.float64)
        sc = _objective(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time():.1f}] seed length={n} score={sc:.6f}")

        # ------------------------------------------------------------------
        # 2.  Smooth projected gradient descent on this seed.
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.45   # give ~45 % of the budget to GD
        a_opt, sc_opt = _smooth_gradient_descent(
            a.copy(),
            sub_deadline,
            best_seq=best_seq,
            best_score=best_score,
        )
        if sc_opt < best_score - 1e-12:
            best_score, best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after GD length={n} new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Fallback – if for any reason nothing was produced.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _distribution_arcsine(2500).astype(np.float64)
        best_score = _objective(best_seq)

    # ------------------------------------------------------------------
    # 4.  Final annealing polish (the remaining time).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing polish (length={len(best_seq)})")
        best_seq, best_score = _pairwise_annealing(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            rng,
        )
        print(f"[{time.time():.1f}] after annealing best score={best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Returning sequence length={len(best_seq)} with objective≈{best_score:.6f}"
    )
    return best_seq.tolist()
