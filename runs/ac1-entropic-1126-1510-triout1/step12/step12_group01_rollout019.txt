def _lp_optimize(given_f, M):
    n = len(given_f)
    # Build A matrix
    # We'll create a list of rows using list comprehension then convert to sparse matrix for speed
    rows = []
    for k in range(2*n-1):
        row = np.zeros(n)
        # j runs from max(0, k-(n-1)) to min(n-1, k)
        j_start = max(0, k-(n-1))
        j_end = min(n-1, k)
        # for j in range(j_start, j_end+1):
        #   row[j] = given_f[k - j]
        # This can be slice: row[j_start:j_end+1] = given_f[k - np.arange(j_start, j_end+1)]
        idx = np.arange(j_start, j_end+1)
        row[idx] = given_f[k - idx]
        rows.append(row)
    A_ub = np.stack(rows)   # shape (2n-1, n)
    b_ub = np.full(2*n-1, M)
    # initial objective: minimize -sum(g)
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        g0 = res.x
        return g0
    else:
        return None

"""
Search for a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.  The objective is
    2·n·max(conv(a,a)) / (sum(a)²)
where ``conv`` is the full self‑convolution.  Since the expression is
homogeneous of degree 0 we can (and do) keep the vector normalised to
sum = 1 during optimisation.

The algorithm has three phases:

1. **Diversified seeding** – generate a pool of candidate vectors from a
   wide range of analytic families (uniform, Beta, arcsine, sine‑weighted,
   triangular, edge‑weighted, random Dirichlet, sparse) for several
   lengths.  Keep the ~30 best seeds.

2. **Exact projected sub‑gradient descent** – for each promising seed run
   a deterministic sub‑gradient method on the exact objective.
   The sub‑gradient of the max‑convolution is computed analytically as
   ``grad = 2·(a ⋆ m)`` where ``m`` is the 0/1 mask of the active (max)
   convolution entries and ``⋆`` denotes correlation.  The correlation
   is evaluated with a single FFT, so each iteration costs only
   ``O(n log n)``.  After the gradient step the vector is projected back
   onto the probability simplex (non‑negative and sum = 1).  A decaying
   step‑size schedule and occasional jitter avoid stagnation.

3. **Local refinement** – a cheap pairwise‑mass‑swap routine followed by
   a simulated‑annealing walk finishes the optimisation.

All phases respect a total wall‑time budget (default 1000 s).  The
function returns a plain ``list[float]``; ``evaluate_sequence`` will
apply its own final clipping / normalisation.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1.
    The function is homogeneous of degree 0, therefore we keep the sum
    normalised to 1 during optimisation.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric beta (Beta(a,b)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight proportional to sin(π (i+½)/n)^p, vanishing at the ends."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    """Linear ramp up to the centre and back down (triangular shape)."""
    mid = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    w = 1.0 - np.abs(i - mid) / mid
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Exact sub‑gradient descent (projected onto simplex) --------------------
def _gradient_maxconv(a: np.ndarray, mask: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of  Σ_k mask_k · (a*a)[k]  w.r.t. ``a``.
    The analytical result is   grad = 2·(a ⋆ mask)   (correlation).
    ``mask`` is a 0/1 vector of length 2n‑1 indicating the active
    convolution indices.  The correlation is evaluated using FFT for speed.
    """
    n = a.shape[0]
    need = n + mask.shape[0] - 1
    size = 1 << ((need - 1).bit_length())

    A = np.fft.rfft(a, n=size)
    M = np.fft.rfft(mask, n=size)

    # correlation = ifft( A * conj(M) )
    corr = np.fft.irfft(A * np.conj(M), n=size)[:need]

    start = mask.shape[0] - 1  # lag = 0 corresponds to this index
    grad = 2.0 * corr[start:start + n]
    return grad


def _subgradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_eta: float = 0.6,
    eta_decay: float = 0.9995,
    eval_interval: int = 200,
    max_iters: int = 1_000_000,
    tol: float = 1e-12,
) -> Tuple[np.ndarray, float]:
    """
    Deterministic projected sub‑gradient descent on the exact objective.
    ``mask`` selects all indices attaining the current maximum (within ``tol``).
    The update is   a ← a − η·grad , projected onto the simplex.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    eta = init_eta
    no_improve = 0
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()

        # 0/1 mask of the active (max) entries
        mask = np.zeros_like(conv)
        mask[conv >= max_c - tol] = 1.0

        # exact sub‑gradient
        grad = _gradient_maxconv(a, mask)

        # additive step, projection onto simplex
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a_new /= s

        # occasional evaluation
        if it % eval_interval == 0:
            cur = _objective(a_new)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a_new.copy()
                a = a_new
                eta = init_eta          # reset step size after a success
                no_improve = 0
                print(f"[{time.time():.1f}] subgrad it={it} new best {best_score:.6f}")
            else:
                no_improve += 1
                a = a_new
        else:
            a = a_new

        eta *= eta_decay

        # jitter on prolonged stagnation
        if no_improve >= 2500:
            jitter = rng.exponential(scale=0.008, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            eta = init_eta
            no_improve = 0

    return best_seq, best_score

# Greedy pairwise refinement --------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> Tuple[np.ndarray, float]:
    """Randomly move a small amount of mass; keep move only if objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score

# Simulated‑annealing style local search ---------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9995,
) -> Tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Main entry point used by the evaluation harness.
    Returns a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the allotted time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 3.0   # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: List[Tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Build a diverse pool of analytic / random seeds
    lengths = [3000, 5000, 8000, 12000, 20000]
    p_grid = np.concatenate(
        [
            np.logspace(-2, -1, 9),   # 0.01 .. 0.1
            np.logspace(-1, 0, 9),    # 0.1 .. 1
            np.logspace(0, 0.5, 7),   # 1 .. 3.16
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 7)  # mixing with uniform

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n), _distribution_triangular(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # parametrised families + mixing
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            sine = _distribution_sine_weighted(n, p)

            for base in (beta, edge, sine):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # asymmetric beta family – random draws
        for _ in range(4):
            if time.time() >= deadline:
                break
            a_param = 10 ** rng.uniform(-2, 0)   # 0.01 … 1
            b_param = 10 ** rng.uniform(-2, 0)
            cand = _distribution_beta_asym(n, a_param, b_param)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best asym‑beta len={n} a={a_param:.3g} b={b_param:.3g} score={sc:.6f}"
                )

        # Dirichlet draws (both sparse and dense)
        for _ in range(8):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(
                    f"[{time.time():.1f}] new global best Dirichlet len={n} α={alpha:.3f} score={sc:.6f}"
                )

        # Sparse random vectors
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(
                    f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}"
                )

        # keep only the best few seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:40]

    # ------------------------------------------------------------------
    # 2. Add the previous solution if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Guard – should never be needed
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Exact sub‑gradient descent on the most promising seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:8]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] starting sub‑gradient descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _subgradient_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                init_eta=0.7,
                eta_decay=0.9997,
                eval_interval=150,
                max_iters=500_000,
                tol=1e-12,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after sub‑gradient seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5. Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 6. Final stochastic local search (simulated annealing)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final random‑local search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
