"""
Hybrid search with smoothed max‑convolution gradient descent.

The core idea is to replace the pure sub‑gradient step (which only reacts
to the single largest convolution entry) by a *soft‑max* version.
For a temperature λ we set weights

    w_k ∝ exp( λ·( (a*a)_k – max_k(a*a)_k ) ),

so that all large convolution entries receive a non‑zero weight.
The gradient of the smoothed objective is then a weighted sum of the
usual gradients, which can be computed efficiently by a single extra
FFT convolution.  λ is increased during the run (annealing) while the
step size follows a 1/√t schedule.  After the smoothed phase we fall
back to the original max‑subgradient descent and finish with a short
greedy mass‑transfer polishing.

The routine respects the overall time budget (default 1000 s) and
returns the best sequence (list of non‑negative floats) found.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    n = x.shape[0]
    m = y.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    z = np.fft.irfft(X * Y, n=size)
    return z[: n + m - 1]


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)² (a is assumed non‑negative)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine_offset(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = i + c
    right = (n - 1 - i) + c
    vals = 1.0 / np.sqrt(left * right)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _distribution_power_inv_sqrt(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = i + 0.5
    right = (n - 1 - i) + 0.5
    vals = (left * right) ** (-p / 2.0)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Generate a small yet diverse pool of analytic seeds for length n."""
    cand = []

    cand.append(_distribution_uniform(n))

    for c in (0.2, 0.5, 1.0, 2.0):
        cand.append(_distribution_arcsine_offset(n, c))

    for p in (0.8, 1.0, 1.2):
        cand.append(_distribution_power_inv_sqrt(n, p))

    cand.append(_distribution_two_blocks(n, 0.05))
    cand.append(_distribution_two_blocks(n, 0.10))

    cand.append(_distribution_random_sparse(n, rng, density=0.01))

    for _ in range(2):
        cand.append(_distribution_dirichlet(
            n, rng, alpha=0.3 + 0.7 * rng.random()))

    return cand

def _smooth_subgrad_descent(a0: np.ndarray, deadline: float,
                            best_seq: np.ndarray, best_score: float,
                            lr0: float = 0.5,
                            lambda0: float = 0.2,
                            lambda_max: float = 30.0) -> tuple[np.ndarray, float]:
    """
    Smoothed max‑convolution descent.

    At iteration t we compute weights w_k ∝ exp(λ·(conv_k – max_conv)),
    where λ grows like λ0·√t up to λ_max.
    The gradient of the smoothed objective is
        g = 2 * (a * w_rev)_{shift}
    and we perform an exponentiated‑gradient update
        a ← a · exp(–η·g),   η = lr0/√t,
    followed by projection onto the probability simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    # pre‑compute index array needed for extracting the gradient
    idx = (2 * n - 2) - np.arange(n)

    while time.time() < deadline:
        t += 1
        # self‑convolution
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] smooth iter={t} new best {best_score:.6f}")

        # temperature schedule
        lam = min(lambda_max, lambda0 * math.sqrt(t))

        # stable exponent computation
        exp_input = lam * (conv - max_conv)          # ≤ 0
        exp_input = np.clip(exp_input, -50.0, 0.0)   # avoid underflow/overflow
        w = np.exp(exp_input)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient = 2 * (a * w_rev) shifted appropriately
        w_rev = w[::-1]
        conv_aw = _fft_convolve(a, w_rev)           # length = 3n‑2
        grad = 2.0 * conv_aw[idx]

        # step size
        eta = lr0 / math.sqrt(t)

        # multiplicative update + simplex projection
        a = a * np.exp(-eta * grad)
        a = _project_onto_simplex(a, 1.0)

    return best_seq, best_score


def _max_subgrad_descent(a0: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float,
                         lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Classical max‑subgradient (only the worst convolution index) descent.
    Kept as a fallback / final refinement.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        k = int(np.argmax(conv))
        max_conv = conv[k]

        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad iter={t} new best {best_score:.6f}")

        # gradient for the maximal index
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1)
        idx_j = k - idx_i
        grad = np.zeros_like(a)
        grad[idx_i] = 2.0 * a[idx_j]

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # projection onto simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s
    return best_seq, best_score


def _pairwise_swap_local(a0: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Greedy random mass‑transfer polishing."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur_score = _objective(new_a)
        if cur_score < best_score - 1e-13:
            a = new_a
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}→{j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Entry point for the optimizer.

    * A diverse set of analytic seeds is generated for several lengths.
    * The best few seeds are refined with the smoothed sub‑gradient descent.
    * A short max‑subgradient phase and greedy polishing finish the search.
    * The function respects the total time budget and returns the best
      sequence (list of floats) discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1.  Seed generation (multiple lengths)
    lengths = [1800, 2500, 3200, 4000, 5000]   # a good trade‑off between quality & speed
    top_seeds: list[tuple[float, np.ndarray]] = []   # (score, array)
    overall_best_seq = None
    overall_best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] --- scanning length {n} ---")
        for cand in _initial_candidates(n, rng):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] !!! new global best len={n} score={sc:.6f}")
            top_seeds.append((sc, cand.copy()))

        # retain only the elite across all processed lengths
        top_seeds.sort(key=lambda x: x[0])
        top_seeds = top_seeds[:12]

    # ------------------------------------------------------------------
    # 2.  Incorporate any previously known solution (if present)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best improved: {sc:.6f}")
            top_seeds.append((sc, prev.copy()))
            top_seeds.sort(key=lambda x: x[0])
            top_seeds = top_seeds[:12]
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Main optimisation – smoothed max‑subgradient descent
    remaining = deadline - time.time()
    if remaining > 20.0 and top_seeds:
        # allocate roughly equal time to the elite seeds
        seeds_to_use = [seq for _, seq in top_seeds[:5]]
        per_seed = max(1.0, remaining / len(seeds_to_use))

        for seed_seq in seeds_to_use:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            # smoothed phase
            seq_out, score_out = _smooth_subgrad_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.6,
                lambda0=0.15,
                lambda_max=25.0,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after smoothed phase new best {overall_best_score:.6f}")

            # short pure max‑subgradient refinement (≈ 30 % of the per‑seed budget)
            extra_deadline = time.time() + 0.3 * per_seed
            seq_out, score_out = _max_subgrad_descent(
                overall_best_seq.copy(),
                extra_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.5,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after pure subgrad new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Final greedy polishing
    if overall_best_seq is not None and time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # Ensure the returned list respects the non‑negative requirement.
    final = np.maximum(overall_best_seq, 0.0).tolist()
    return final
