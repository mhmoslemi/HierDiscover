def _adam_optimize(a0, deadline, current_best_score, max_iter=200000, eval_interval=200):

    n = a0.shape[0]
    # initial logits
    eps = 1e-12
    x = np.log(np.maximum(a0, eps))
    # maintain best local
    a = a0.copy()
    best_local_seq = a.copy()
    best_local_score = _objective(a)
    # Adam parameters
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.05
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        # compute a via softmax
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)  # shift for numerical stability
        a = exp_x / np.sum(exp_x)

        # conv
        conv = _fft_self_convolve(a)
        max_idx = int(conv.argmax())
        # gradient wrt a
        w = np.zeros_like(conv)
        w[max_idx] = 1.0
        grad_a = _gradient_from_weights(a, w)  # shape n
        # transform to grad wrt logits x
        dot_ag = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot_ag)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        # update logits
        x = x - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        if t % eval_interval == 0:
            # compute exact objective
            score = _objective(a)
            if score < best_local_score - 1e-14:
                best_local_score = score
                best_local_seq = a.copy()
                # optionally report improvement relative to global best
                if score < current_best_score - 1e-12:
                    print(f"[{time.time():.1f}] Adam it={t} new local best {score:.6f}")
    return best_local_seq, best_local_score

    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.03
    best_seq = None
    best_score = float('inf')
    # initial evaluate
    cur_a = a0.copy()
    cur_score = _objective(cur_a)
    if cur_score < best_score:
        best_score = cur_score
        best_seq = cur_a.copy()
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / np.sum(exp_x)
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())
        # compute gradient w.r.t a
        grad_a = _grad_max_entry(a, k)   # shape n
        # convert to gradient wrt logits
        dot = np.dot(a, grad_a)

            max_x = np.max(x)

def propose_candidate(seed=123, budget_s=1000):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.0

    best_seq = None
    best_score = float('inf')

    # add prev if exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            prev = prev / prev.sum()
            score = _objective(prev)
            if score < best_score:
                best_score = score
                best_seq = prev.copy()
                print(f"Initial prev sequence score {score:.6f}")
    except Exception:
        pass

    lengths = [2000, 3000, 4000, 6000, 8000]
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] === processing length {n} ===")
        # generate seeds
        seeds = _initial_candidates_extended(n, rng)
        # add random Dirichlet seeds:
        for _ in range(8):
            conc = 10**rng.uniform(-2, 1)   # random concentration
            vals = rng.gamma(conc, size=n)
            vals = np.maximum(vals, 0)
            if vals.sum() == 0.0:
                vals = np.full(n, 1.0 / n)
            else:
                vals = vals / vals.sum()
            seeds.append(vals)
        # add alternating
        seeds.append(_distribution_alternating(n))
        # Evaluate quickly and keep top seeds
        scored = []
        for s in seeds:
            if time.time() >= deadline:
                break
            s = np.maximum(s, 0.0)
            if s.sum() == 0.0:
                continue
            s = s / s.sum()
            sc = _objective(s)
            scored.append((sc, s))
        if not scored:
            continue
        scored.sort(key=lambda x: x[0])
        top_seeds = [seq for _, seq in scored[:6]]   # keep best 6

        # Optimize each seed
        for seed_seq in top_seeds:
            if time.time() >= deadline:
                break
            remaining = deadline - time.time()
            sub_dead = time.time() + min(30.0, remaining * 0.5)  # allocate up to 30 secs or half remaining
            opt_seq, opt_score = _adam_optimize(seed_seq.copy(), sub_dead, best_score)
            if opt_score < best_score:
                best_score = opt_score
                best_seq = opt_seq.copy()
                print(f"[{time.time():.1f}] New global best after Adam: len={n} score={best_score:.6f}")
            # optional local polish
            if time.time() < deadline:
                pol_seq, pol_score = _pairwise_swap_local(opt_seq.copy(), deadline, best_seq, best_score)
                if pol_score < best_score:
                    best_score = pol_score
                    best_seq = pol_seq.copy()
                    print(f"[{time.time():.1f}] New global best after polishing: len={n} score={best_score:.6f}")

    # final polishing
    if best_seq is not None and time.time() < deadline:
        best_seq, best_score = _pairwise_swap_local(best_seq.copy(), deadline, best_seq, best_score)

    print(f"Final best score: {best_score:.6f} length={len(best_seq) if best_seq is not None else 0}")
    return best_seq.tolist() if best_seq is not None else []

"""
AdaBoost‑style Adam optimiser for the self‑convolution max‑ratio problem.

The optimiser works as follows:

1. **Seed generation** – a wide collection of analytic families (uniform,
   arcsine, Beta, Dirichlet, exponential, power‑law, sparse, two‑block,
   alternating) together with many random Dirichlet draws are produced for a
   handful of lengths (≈2k‑8k).  All seeds are normalised to sum = 1 and the
   best few are kept.

2. **Adam optimisation on the true maximum entry** – the objective after
   normalising `sum(a)=1` is simply `2·n·max(conv(a,a))`.  The gradient of the
   maximal convolution entry with respect to the probabilities `a` is computed
   analytically in O(n) time.  The variables are re‑parameterised as
   soft‑max logits `x` (`a_i = exp(x_i)/∑exp(x)`); Adam updates are applied
   directly to `x`, guaranteeing positivity and the simplex constraint.

3. **Greedy polishing** – after the Adam phase a cheap random mass‑transfer
   walk (pairwise swaps) is performed until the time budget expires.

The routine respects the provided time budget (`budget_s` seconds) and
returns the best sequence it found as a plain list of floats.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective `2·n·max(conv(a,a))/sum(a)^2`.
    The vector a is assumed to be non‑negative.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
def _grad_max_entry(a: np.ndarray, k: int) -> np.ndarray:
    """
    Exact gradient of the maximal convolution entry `conv[k]`
    with respect to a (full factor 2 for both diagonal and off‑diagonal).
    O(n) implementation.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    if i_max >= i_min:
        idx = np.arange(i_min, i_max + 1, dtype=np.int64)
        grad[idx] = 2.0 * a[k - idx]
    return grad


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   global_best_score: float,
                   max_iter: int = 150_000,
                   eval_interval: int = 200) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex (via soft‑max logits).
    Stops when the time deadline is reached or `max_iter` iterations have been
    performed.
    """
    n = a0.shape[0]
    eps = 1e-15
    # initialise logits
    x = np.log(np.clip(a0, eps, None))

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.03
    best_seq = a0.copy()
    best_score = _objective(best_seq)

    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1

        # soft‑max to obtain probabilities
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / np.sum(exp_x)

        # convolution and index of the current max entry
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())

        # gradient wrt a (full factor 2)
        grad_a = _grad_max_entry(a, k)

        # chain rule to obtain gradient wrt logits x
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # periodic exact evaluation
        if t % eval_interval == 0:
            max_x = np.max(x)
            exp_x = np.exp(x - max_x)
            a = exp_x / np.sum(exp_x)
            score = _objective(a)
            if score < best_score - 1e-14:
                best_score = score
                best_seq = a.copy()
                if best_score < global_best_score - 1e-12:
                    print(f"[{time.time():.1f}] Adam it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """
    Cheap random mass‑transfer walk.  Randomly moves a fraction of mass from
    one index to another and keeps the change only if the true objective
    improves.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue

        # move up to 30 % of source mass
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * x / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::2] = 1.0
    a /= a.sum()
    return a


def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Broad pool of analytic seeds for a given length."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_sine(n))
    cand.append(_distribution_two_blocks(n, block_frac=0.07))
    cand.append(_distribution_alternating(n))

    # symmetric beta on a fine grid
    for p in np.linspace(0.05, 1.0, 12):
        cand.append(_distribution_beta_sym(n, p))

    # random asymmetric beta
    for _ in range(8):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # power law
    for p in (0.5, 1.0, 2.0, 3.0):
        cand.append(_distribution_power_increasing(n, p))

    # exponential
    for lam in (0.001, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential_increasing(n, lam))

    # sparse random draws
    for dens in (0.005, 0.01, 0.02):
        cand.append(_distribution_random_sparse(n, rng, density=dens))

    # Dirichlet draws with various concentrations
    for a in (0.1, 0.3, 0.5, 1.0):
        cand.append(_distribution_dirichlet(n, rng, alpha=a))

    return cand
