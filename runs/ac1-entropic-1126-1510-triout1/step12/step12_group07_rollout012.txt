def _lp_max_sum(f):
    n = len(f)
    conv_ff = _fft_self_convolve(f)
    M = conv_ff.max()
    T = _conv_matrix_sparse(f) # shape (2n-1, n)
    A_ub = T
    b_ub = np.full(2*n-1, M * (1 + 1e-12))
    c = -np.ones(n)  # maximize sum => minimize -sum
    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        # fallback: return f unchanged
        return f.copy()
    g = res.x
    return g

def propose_candidate(seed=42, budget_s=1000):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0
    overall_best_seq = None
    overall_best_score = float('inf')
    candidates = []

    lengths = [1800,2000,2200,2500]
    for n in lengths:
        if time.time() >= deadline: break
        print(f"[{time.time():.1f}] Generating candidates for length {n}")
        cand_list = _initial_candidates_extended(n, rng)
        for cand in cand_list:
            if time.time() >= deadline: break
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0:
                continue
            cand = cand / s
            sc = _objective(cand)
            candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] Global best updated: len={n} score={sc:.6f}")

    # keep top seeds across all lengths
    candidates.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in candidates[:8]]  # up to 8
    if overall_best_seq is None:
        # fallback uniform
        n = lengths[0] if lengths else 2000
        overall_best_seq = np.full(n, 1.0/n)
        overall_best_score = _objective(overall_best_seq)

    # LP refinement for each top seed
    for idx, seed_seq in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        n = seed_seq.shape[0]
        f = _normalize_to_sqrt2n(seed_seq)
        # ensure non-negative
        f = np.maximum(f, 0.0)
        # run LP refinement
        max_iters = 12
        t_mix = 0.3
        best_local = f.copy()
        best_local_score = _objective(best_local)
        for it in range(max_iters):
            if time.time() >= deadline:
                break
            g = _lp_max_sum(f)   # solves LP using f as coefficients
            # if LP unbounded or fails, fall back to f
            if g is None:
                g = f.copy()
            # mixing
            f = (1 - t_mix) * f + t_mix * g
            # f automatically retains sum sqrt(2n) because both have that sum
            cur_score = _objective(f)
            if cur_score + 1e-12 < best_local_score:
                best_local_score = cur_score
                best_local = f.copy()
                print(f"[{time.time():.1f}] Seed {idx} LP it={it} new best {best_local_score:.6f}")
            # optionally reduce mixing step
            t_mix *= 0.9
        # after LP, run short Adam to fine-tune
        if time.time() < deadline:
            # initialize Adam with log of best_local
            z0 = np.log(np.maximum(best_local, 1e-300))
            sub_deadline = time.time() + 30.0  # allocate 30 seconds per seed
            best_local, best_local_score = _adam_optimize(
                z0,
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                alpha0=1.0,
                alpha_max=200.0,
                growth_factor=1.03,
                lr=0.12,
                max_iter=150000,
                eval_interval=200,
            )
        # compare with global best
        if best_local_score < overall_best_score:
            overall_best_score = best_local_score
            overall_best_seq = best_local.copy()
            print(f"[{time.time():.1f}] Global best improved after seed {idx}: score={overall_best_score:.6f}")

    # final polishing steps if time remains
    remaining = deadline - time.time()
    if remaining > 20.0:

def _lp_max_sum(f):
    n = f.shape[0]
    conv_f_f = _fft_self_convolve(f)
    M = conv_f_f.max()
    # Build convolution matrix
    rows = 2*n - 1
    # Using dense approach
    T = np.empty((rows, n), dtype=np.float64)
    for i in range(n):
        T[i:i+n, i] = f
    # set up linprog
    import scipy.optimize
    c = -np.ones(n)  # maximize sum
    b = np.full(rows, M * (1.0 + 1e-12))
    bounds = [(0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=T, b_ub=b, bounds=bounds, method='highs')
    except Exception as e:
        # fallback
        return None
    if res.success and res.status == 0:
        g = res.x
        # scale to have sum sqrt(2n)
        target = math.sqrt(2.0 * n)
        s = g.sum()
        if s > 0:
            g = g * (target / s)
        else:
            g = f.copy()
        return g
    else:
        # unbounded or failed
        return None
```

"""
Hybrid optimizer for the max‑convolution step‑function problem.

The search consists of:
1.  Broad seeding using many analytic families and random Dirichlet draws.
2.  Selecting the best few seeds and refining each of them by an LP
    (maximize the ℓ₁‑norm under the current max‑convolution bound).
3.  A short Adam smooth‑max descent starting from the best LP‑refined
    vector.
4.  Final multimax sub‑gradient polishing and a cheap greedy pairwise‑swap
    walk.

All vectors are kept on the Euclidean simplex (sum ≥ 0, scale‑invariant),
and the routine respects a user‑supplied time budget.
"""

import time
import math
from typing import List, Tuple, Optional

import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of *v* onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate: 2·a⋆w, where *w* (length 2·n‑1)
    sums to 1.  Implemented via FFT cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    shift = np.max(z)
    exp_z = np.exp(z - shift)
    s = exp_z.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / len(z))
    return exp_z / s


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (sum>0)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# ---------------------  Analytic families -----------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.minimum(i + 1, n - i)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_candidates_extended(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    """Collect a broad variety of seeds for a given length."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_sine(n))
    cand.append(_distribution_triangular(n))
    cand.append(_distribution_gaussian(n))

    # symmetric Beta family
    for p in (0.05, 0.1, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0, 1.5, 2.0):
        cand.append(_distribution_beta_sym(n, p))

    # random asymmetric Beta draws
    for _ in range(4):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # power‑law families
    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))

    # exponential families
    for lam in (0.001, 0.003, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential_increasing(n, lam))

    # two‑block spikes
    for block_frac in (0.05, 0.10, 0.15):
        cand.append(_distribution_two_blocks(n, block_frac))

    # random Dirichlet draws
    for _ in range(4):
        alpha = 0.1 + 0.9 * rng.random()
        cand.append(_distribution_dirichlet(n, rng, alpha=alpha))

    # random sparse patterns
    cand.append(_distribution_random_sparse(n, rng, density=0.01))

    # a few completely random probability vectors
    for _ in range(3):
        v = rng.random(n)
        v = np.maximum(v, 0.0)
        if v.sum() == 0.0:
            v[0] = 1.0
        v /= v.sum()
        cand.append(v)

    return cand


# ----------------------------------------------------------------------
def _normalize_to_sqrt2n(a: np.ndarray) -> np.ndarray:
    """Rescale *a* so that sum(a) = sqrt(2·len(a))."""
    n = a.shape[0]
    target = math.sqrt(2.0 * n)
    s = a.sum()
    if s == 0.0:
        return np.full_like(a, target / n)
    return a * (target / s)


# ----------------------------------------------------------------------
def _conv_matrix_dense(f: np.ndarray) -> np.ndarray:
    """Full convolution matrix T such that conv(f,g) = T @ g."""
    n = f.shape[0]
    rows = 2 * n - 1
    T = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        T[i:i + n, i] = f
    return T


# ----------------------------------------------------------------------
def _lp_max_sum(f: np.ndarray) -> Optional[np.ndarray]:
    """
    Solve the LP max Σ g_j  s.t. g ≥ 0 and conv(f,g) ≤ M,
    where M = max(conv(f,f)).  Returns the scaled vector g with
    sum(g) = sqrt(2·n) if feasible, otherwise None.
    """
    n = f.shape[0]
    conv_ff = _fft_self_convolve(f)
    M = conv_ff.max()

    # Build convolution matrix (dense – still modest for n≈2000)
    T = _conv_matrix_dense(f)

    # RHS with a tiny slack to avoid numerical issues
    b = np.full(2 * n - 1, M * (1.0 + 1e-12), dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)          # maximise sum(g)  →  minimise -sum(g)
    bounds = [(0.0, None) for _ in range(n)]

    try:
        res = scipy.optimize.linprog(
            c, A_ub=T, b_ub=b, bounds=bounds, method="highs"
        )
    except Exception:  # any unexpected error → treat as failure
        return None

    # status 0 = optimal, 3 = unbounded, 2 = infeasible etc.
    if res.success and res.status == 0:
        g = res.x
        # Rescale to the canonical sum √(2n)
        target = math.sqrt(2.0 * n)
        s = g.sum()
        if s > 0.0:
            g = g * (target / s)
        else:
            return None
        return g
    else:
        # LP unbounded or failed – do not use the result
        return None


# ----------------------------------------------------------------------
def _adam_optimize(z0: np.ndarray,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   alpha0: float = 1.0,
                   alpha_max: float = 500.0,
                   growth_factor: float = 1.02,
                   lr: float = 0.1,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   max_iter: int = 200000,
                   eval_interval: int = 200) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    """
    n = z0.shape[0]
    z = z0.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        a = _softmax(z)                       # a lies on the simplex

        conv = _fft_self_convolve(a)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        grad_a = _gradient_from_weights(a, w)

        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        z = z - step

        if t % 500 == 0:
            alpha = min(alpha * growth_factor, alpha_max)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.5,
                                 max_iter: int = 20000,
                                 eval_interval: int = 200) -> Tuple[np.ndarray, float]:
    """
    Classic multimax sub‑gradient descent (exponentiated‑gradient on the simplex).
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)

        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] multimax it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Greedy mass‑transfer walk, accepting only improving moves."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the supplied time budget and returns the best
    sequence found (as a plain list of floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1.   Broad seeding – a moderate set of lengths (kept relatively small
    #      for the subsequent LP phase).
    lengths = [1800, 2000, 2200, 2500]

    overall_best_seq: Optional[np.ndarray] = None
    overall_best_score = float("inf")
    candidates: List[Tuple[float, np.ndarray]] = []

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating candidates for length {n}")

        for cand in _initial_candidates_extended(n, rng):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                continue
            cand = cand / s                                   # sum = 1
            sc = _objective(cand)
            candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] **** new global best (len={n}) score={sc:.6f}")

    # retain only the most promising seeds (across all lengths)
    candidates.sort(key=lambda x: x[0])
    candidates = candidates[:8]      # at most eight top seeds

    # ------------------------------------------------------------------
    # 2.   LP‑based refinement for the top seeds
    # ------------------------------------------------------------------
    for idx, (seed_score, seed_vec) in enumerate(candidates):
        if time.time() >= deadline:
            break
        n = seed_vec.shape[0]
        # bring the seed to the canonical sum √(2 n)
        f = _normalize_to_sqrt2n(seed_vec)
        best_local = f.copy()
        best_local_score = _objective(best_local)
        t_mix = 0.35                     # initial mixing parameter

        for it in range(12):            # fixed number of LP iterations
            if time.time() >= deadline:
                break
            g = _lp_max_sum(f)
            if g is None:               # LP failed / unbounded → skip this iteration
                g = f.copy()
            # mixing step (both f and g already have the correct sum)
            f = (1.0 - t_mix) * f + t_mix * g
            cur_score = _objective(f)
            if cur_score + 1e-13 < best_local_score:
                best_local_score = cur_score
                best_local = f.copy()
                print(f"[{time.time():.1f}] seed {idx} LP it={it} new best {best_local_score:.6f}")
            t_mix *= 0.85                # gradually diminish the step size

        # ------------------------------------------------------------------
        # 3.   Short Adam smooth‑max descent starting from the LP‑refined point
        # ------------------------------------------------------------------
        if time.time() < deadline:
            z0 = np.log(np.maximum(best_local, 1e-300))
            sub_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.4)
            best_local, best_local_score = _adam_optimize(
                z0,
                sub_deadline,
                best_seq=overall_best_seq if overall_best_seq is not None else best_local,
                best_score=overall_best_score,
                alpha0=1.0,
                alpha_max=300.0,
                growth_factor=1.03,
                lr=0.12,
                max_iter=150000,
                eval_interval=250,
            )

        # --------------------------------------------------------------
        # 4.   Keep the best solution overall
        # --------------------------------------------------------------
        if best_local_score < overall_best_score:
            overall_best_score = best_local_score
            overall_best_seq = best_local.copy()
            print(f"[{time.time():.1f}] **** global best improved by seed {idx} → {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.   Final polishing (multimax + greedy swaps) if time permits
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0 and overall_best_seq is not None:
        overall_best_seq, overall_best_score = _subgradient_multimax_descent(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.6,
            max_iter=25000,
            eval_interval=300,
        )
    if time.time() < deadline and overall_best_seq is not None:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    # 6.   Return the best sequence found
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        # fallback – uniform distribution of a modest size
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # Clip to the limits imposed by ``evaluate_sequence``
    overall_best_seq = np.clip(overall_best_seq, 0.0, 1000.0)

    return overall_best_seq.tolist()
