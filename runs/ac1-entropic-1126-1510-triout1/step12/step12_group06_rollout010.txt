"""
Hybrid optimiser for the max‑convolution problem (enhanced).

Key ideas
~~~~~~~~~
*  a very broad seed set (uniform, arcsine, many Beta families, Dirichlet,
   sparse patterns, plus the previous solution if any);
*  a long Adam‑style smooth‑max descent on the best seed (high temperature,
   aggressive steps, and most of the remaining time is spent here);
*  a classic multimax (sub‑gradient) descent for fine‑tuning;
*  a cheap greedy pair‑wise mass‑transfer polishing step;
*  a small “restart‑with‑noise + short Adam” loop that tries to escape
   local minima.

All updates keep the vector on the probability simplex (non‑negative,
sum = 1) by normalising after each multiplicative update.
The routine respects the 1000 s budget, prints intermediate scores and
returns the best coefficient list found.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of    f(a) = Σ_k w_k (a*a)_k   = 2·a⋆w,
    where w (length 2·n‑1) sums to 1.
    Implemented via FFT cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2; we keep sum(a)=1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discretised unit interval."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(alpha,beta) on the discretised unit interval."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Few non‑zero entries, random values."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    """Two symmetric spikes of equal mass."""
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _smooth_max_adam_descent(a: np.ndarray,
                             deadline: float,
                             best_seq: np.ndarray,
                             best_score: float,
                             eta0: float = 0.6,
                             beta1: float = 0.9,
                             beta2: float = 0.999,
                             eps: float = 1e-8,
                             alpha0: float = 0.3,
                             alpha_mul: float = 1.25,
                             alpha_max: float = 150.0,
                             max_iter: int = 60000,
                             eval_interval: int = 500) -> Tuple[np.ndarray, float]:
    """
    Adam‑style exponentiated‑gradient descent on the smooth‑max surrogate
    f_α(a) = (1/α)·log Σ_i exp(α·conv_i(a)).
    """
    n = a.shape[0]
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        # full self‑convolution
        conv = _fft_self_convolve(a)

        # soft‑max weights (stable)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad = _gradient_from_weights(a, w)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # multiplicative update
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(-step)

        # project onto the simplex (sum = 1)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # temperature schedule
        if t % 300 == 0 and alpha < alpha_max:
            alpha = min(alpha * alpha_mul, alpha_max)

        # periodic exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.6,
                                 max_iter: int = 35000,
                                 eval_interval: int = 400) -> Tuple[np.ndarray, float]:
    """
    Classic exponentiated‑gradient descent where the sub‑gradient is the
    average over all active indices attaining the maximal convolution value.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()

        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)

        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Multimax it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Greedy pairwise mass‑transfer: move a small amount from i to j if it improves the objective."""
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        # distinct destination
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # random transfer amount (up to 30 % of the source mass)
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _restart_with_noise(a: np.ndarray,
                       rng: np.random.Generator,
                       deadline: float,
                       best_seq: np.ndarray,
                       best_score: float) -> Tuple[np.ndarray, float]:
    """Create a noisy copy of ``a`` and run a short Adam run; keep any improvement."""
    n = a.shape[0]
    while time.time() < deadline:
        # generate a perturbed copy
        noisy = a + rng.normal(scale=0.01, size=n)
        noisy = np.maximum(noisy, 0.0)
        s = noisy.sum()
        if s == 0.0:
            continue
        noisy /= s

        # short Adam run (few thousand iterations, modest schedule)
        sub_deadline = time.time() + min(6.0, deadline - time.time())
        noisy, _ = _smooth_max_adam_descent(
            noisy,
            sub_deadline,
            best_seq=noisy,
            best_score=_objective(noisy),
            eta0=0.5,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            alpha0=0.2,
            alpha_mul=1.15,
            alpha_max=80.0,
            max_iter=20000,
            eval_interval=200,
        )
        cur = _objective(noisy)
        if cur < best_score - 1e-13:
            best_score = cur
            best_seq = noisy.copy()
            a = noisy.copy()
            print(f"[{time.time():.1f}] noise‑restart improved to {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects ``budget_s`` seconds and returns the best sequence
    found (as a plain list of floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # 1. Very fast scan of a wide collection of analytic families.
    #    Keep only a few promising seeds.
    lengths = [3000, 6000, 12000]          # a handful of lengths
    top_n = 12                            # keep this many candidates overall
    top_candidates: List[Tuple[float, np.ndarray]] = []
    best_seq: np.ndarray | None = None
    best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        cand_list: List[np.ndarray] = []

        # basic families
        cand_list.append(_distribution_uniform(n))
        cand_list.append(_distribution_arcsine(n))

        # symmetric Beta – finer grid
        for p in np.geomspace(0.01, 10.0, num=30):
            cand_list.append(_distribution_beta_sym(n, float(p)))

        # random asymmetric Beta draws
        for _ in range(12):
            alpha = 10 ** rng.uniform(-3, 1)
            beta = 10 ** rng.uniform(-3, 1)
            cand_list.append(_distribution_beta_asym(n, alpha, beta))

        # Dirichlet draws with varied concentration
        for a in (0.2, 0.5, 0.9, 1.5, 3.0):
            cand_list.append(_distribution_dirichlet(n, rng, a))

        # Random sparse patterns
        for dens in (0.005, 0.02):
            cand_list.append(_distribution_random_sparse(n, rng, density=dens))

        # A few fully dense random vectors
        for _ in range(6):
            v = rng.random(n)
            v = np.maximum(v, 0.0)
            v /= v.sum()
            cand_list.append(v)

        # evaluate candidates
        for cand in cand_list:
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                continue
            cand = cand / s
            sc = _objective(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time():.1f}] **** new global best len={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))

        # keep only the best few seeds seen so far
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:top_n]

    # ------------------------------------------------------------------
    # 2. Include previous run if available (helps continuation across sessions).
    try:
        prev = globals().get("height_sequence_1")
        if isinstance(prev, (list, np.ndarray)):
            prev = np.asarray(prev, dtype=np.float64)
            if prev.ndim == 1 and prev.size > 0:
                prev = np.maximum(prev, 0.0)
                s = prev.sum()
                if s == 0.0:
                    prev = _distribution_uniform(prev.shape[0])
                    s = prev.sum()
                prev = prev / s
                sc = _objective(prev)
                if sc < best_score:
                    best_score = sc
                    best_seq = prev.copy()
                    print(f"[{time.time():.1f}] previous solution improved: {sc:.6f}")
                top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback – should never be needed.
    if best_seq is None:
        n = 4000
        best_seq = _distribution_uniform(n)
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Long Adam‑style smooth‑max descent on the *currently best* seed.
    #    The majority of the remaining budget is spent here.
    remaining = deadline - time.time()
    if remaining > 10:
        adam_deadline = time.time() + remaining * 0.78   # ~78 % of what is left
        best_seq, best_score = _smooth_max_adam_descent(
            best_seq.copy(),
            adam_deadline,
            best_seq,
            best_score,
            eta0=0.9,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            alpha0=0.2,
            alpha_mul=1.30,
            alpha_max=300.0,
            max_iter=300000,
            eval_interval=200,
        )
        print(f"[{time.time():.1f}] after intensive Adam best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Short but aggressive multimax (sub‑gradient) refinement.
    remaining = deadline - time.time()
    if remaining > 8:
        multimax_deadline = time.time() + remaining * 0.65
        best_seq, best_score = _subgradient_multimax_descent(
            best_seq.copy(),
            multimax_deadline,
            best_seq,
            best_score,
            lr0=0.5,
            max_iter=120000,
            eval_interval=250,
        )
        print(f"[{time.time():.1f}] after multimax best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Greedy pairwise swaps for cheap polishing.
    remaining = deadline - time.time()
    if remaining > 2:
        best_seq, best_score = _pairwise_swap_local(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
        )
        print(f"[{time.time():.1f}] after pairwise swap best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Optional noisy restarts – try to escape any remaining local trap.
    remaining = deadline - time.time()
    if remaining > 5:
        best_seq, best_score = _restart_with_noise(
            best_seq.copy(),
            rng,
            deadline,
            best_seq,
            best_score,
        )
        print(f"[{time.time():.1f}] after noise restarts best {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    # ``evaluate_sequence`` will clip negatives and enforce sum≥0.01; we keep a proper probability vector.
    return best_seq.tolist()
