def _smooth_gradient(a, beta):
    conv = _fft_self_convolve(a)
    # to get w: softmax of beta * conv
    max_val = np.max(beta * conv)
    exp_vals = np.exp(beta * conv - max_val)
    w = exp_vals / np.sum(exp_vals)
    # compute convolution of w with reversed a
    a_rev = a[::-1]
    conv_w_ar = _fft_convolve(w, a_rev)
    # slice appropriate part
    n = len(a)
    g = conv_w_ar[n-1 : n-1+n]
    # gradient = 2*n * g
    return 2.0 * n * g

"""
Propose a non‑negative step function that (almost) minimizes
``evaluate_sequence``.

The optimiser works in four phases:

1️⃣  Generate a diverse pool of probability vectors (uniform,
    arcsine, symmetric Beta, edge‑weighted, generalized‑normal,
    random Dirichlet, sparse).  All vectors are normalised to sum‑1.

2️⃣  Refine the most promising seeds with *smooth* exponential /
    projected‑gradient descent.  Rather than using the raw
    ``max(conv)`` we minimise a smooth approximation
    ``(1/β)·log Σ exp(β·conv)``.
    For each β we compute the exact gradient via FFTs
    (convolution of the soft‑max weights with the reversed vector)
    and perform additive updates with a diminishing step‑size,
    projecting back onto the probability simplex.

3️⃣  If time permits we up‑sample the current best vector (linear
    interpolation while preserving the simplex) and repeat the
    smooth descent on the larger dimension.  Larger n allows the
    maximal convolution value to shrink further.

4️⃣  A final long‑run pairwise‑mass‑transfer annealing step
    polishes the best candidate (temperature decays linearly,
    acceptance of worsening moves is allowed early on).

The whole routine respects the 1000 s budget (2 s safety margin) and
returns a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolutions
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Full linear convolution of u and v (both real) using a real FFT."""
    size = 1 << ((len(u) + len(v) - 1) - 1).bit_length()
    U = np.fft.rfft(u, n=size)
    V = np.fft.rfft(v, n=size)
    conv = np.fft.irfft(U * V, n=size)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
# Exact objective used by ``evaluate_sequence`` (sum already = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a probability vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    if max_b < 0:
        max_b = 0.0
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, sum=1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination of ``base`` with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

    w = kaiser(n, beta, sym=True).astype(np.float64)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) gradient of the objective
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smoothed objective

        f_β(a) = (2·n/β)·log Σ_k exp(β·c_k)
    where c = a*a (full convolution).

    Returns ∇f_β, shape (n,).  ``a`` must be a probability vector.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)               # length 2n‑1
    scaled = beta * conv
    # stable soft‑max
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w /= w.sum()                               # weights over 2n‑1 entries

    # g_i = Σ_k w_k·a_{k‑i}
    a_rev = a[::-1]
    conv_w_rev = _fft_convolve(w, a_rev)       # length 3n‑2
    g = conv_w_rev[n - 1 : n - 1 + n]          # slice to length n
    # ∇f_β = 2·n·g
    return 2.0 * n * g


# ----------------------------------------------------------------------
# Smooth exponential / projected gradient descent
# ----------------------------------------------------------------------
def _smooth_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    betas: list[float],
    eta0: float = 0.4,
    eval_every: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Perform projected gradient descent on the smoothed objective.
    The schedule runs through the list ``betas`` (increasing),
    performing many tiny additive steps with a 1/√t step‑size.
    """
    t = 0
    n = a.shape[0]
    for beta in betas:
        # a fixed number of inner updates for each β; we stop early if time runs out
        max_inner = 6000
        for _ in range(max_inner):
            if time.time() >= deadline:
                break
            t += 1
            grad = _smooth_gradient(a, beta)
            eta = eta0 / math.sqrt(t)
            a = a - eta * grad
            a = _project_onto_simplex(a)  # keep Σa = 1, a≥0
            if t % eval_every == 0:
                cur = _objective(a)
                if cur < best_score - 1e-14:
                    best_score = cur
                    best_seq = a.copy()
                    print(
                        f"[{time.time():.1f}] smooth β={beta:.0f} it={t} new best {best_score:.6f}"
                    )
        if time.time() >= deadline:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple pairwise mass‑transfer annealing (as before, but with higher temp)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins, accept if better
    or with a temperature‑dependent probability."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.45  # up to 45 % of source mass
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)

        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.15

        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation while preserving the simplex)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= a_new.sum()
    return a_new

    return overall_best_seq.tolist()
