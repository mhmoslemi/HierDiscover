"""
A fast L∞‑convolution optimiser.

The objective of ``evaluate_sequence`` for a non‑negative vector ``a`` of
length ``n`` is

    f(a) = 2·n·max(conv(a,a)) / (∑a)² .

Since the value is invariant under scaling, we work on the probability
simplex {a_i≥0, ∑a_i=1}.  The naïve uniform distribution gives f=2, but
the theoretical lower bound (perfectly flat self‑convolution) is
≈2n/(2n−1) → 1 as *n* grows.  The optimiser has three stages:

1. **Seed generation** – a few analytic families (uniform, arcsine,
   symmetric Beta, edge‑weighted, exponential, generalized‑Normal,
   random Dirichlet, sparse) are created for several lengths.  The best
   seeds are retained.

2. **Smoothed‑max gradient descent** – instead of the sub‑gradient of the
   raw max we use a soft‑max (log‑sum‑exp) with an increasing
   temperature.  The gradient is computed via an FFT‑based convolution,
   and updates are performed with exponentiated‑gradient steps followed
   by Euclidean projection onto the simplex.

3. **Pairwise annealing polish** – a long‑run simulated‑annealing phase
   that swaps small amounts of mass between two coordinates.  It accepts
   improving moves always and occasionally accepts worsening moves
   according to a linear cooling schedule.

The whole routine respects the 1000 s budget (a 2 s safety margin).  The
best sequence found is returned as a plain ``list[float]`` ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean, onto {x≥0, Σx = 1})
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project onto the simplex {x≥0, Σx = s}."""
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolutions
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (a and b may have different lengths)."""
    n = a.shape[0]
    m = b.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    C = A * B
    conv = np.fft.irfft(C, n=size)
    return conv[: n + m - 1]


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (shortcut for the self‑case)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective (identical to evaluate_sequence for a simplex‑normalised vector)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    # ``sum_a`` equals 1 after our normalisation, but keep the formula generic
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, Σ=1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_generalised_normal(n: int, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_exponential(n: int, r: float) -> np.ndarray:
    """Geometric/exponential shape: a_i ∝ r**i, r∈(0,1]."""
    if r <= 0.0:
        r = 1e-12
    vals = r ** np.arange(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smoothed‑max gradient (soft‑max with temperature β)
# ----------------------------------------------------------------------
def _grad_smooth_max(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smoothed objective
        S_β(a) = (1/β)·log(∑_k exp(β·b_k)),   b = a*a.
    Returns ∇S_β(a) (length n) without the outer factor 2·n, because the
    optimiser works directly on the inner max‑term.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)                     # length 2n‑1
    max_b = b.max()
    # soft‑max weights (numerically stable)
    w = np.exp(beta * (b - max_b))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum                                 # ∑ w_k = 1

    # gradient = 2 * Σ_k w_k * a_{k‑i}  (with out‑of‑range ignored)
    a_rev = a[::-1]
    conv = _fft_convolve(a_rev, w)                # length 3n‑2
    grad = 2.0 * conv[n - 1 : n - 1 + n]           # slice of length n
    return grad


# ----------------------------------------------------------------------
# Smoothed‑max exponentiated‑gradient descent
# ----------------------------------------------------------------------
def _smooth_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    beta0: float = 0.5,
    beta_max: float = 30.0,
    eta0: float = 0.8,
    eval_every: int = 60,
) -> tuple[np.ndarray, float]:
    """
    Minimises the true objective using a smoothed max.  The update rule
    is a multiplicative (exponentiated‑gradient) step followed by a
    projection onto the probability simplex.
    """
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it += 1
        # temperature schedule – grow √(it) and clamp
        beta = min(beta_max, beta0 * math.sqrt(it))

        grad = _grad_smooth_max(a, beta)          # non‑negative
        eta = eta0 / math.sqrt(it)                # decreasing step‑size

        # multiplicative update stays non‑negative automatically
        a = a * np.exp(-eta * grad)

        # back to the simplex (Σa = 1)
        a = _project_onto_simplex(a, s=1.0)

        if it % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] smooth‑GD it={it} β={beta:.2f} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise mass‑transfer annealing (same as the previous implementation)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)

        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.01

        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsample (linear interpolation while staying on the simplex)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger length, followed by simplex normalisation."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that (almost) minimises
    ``evaluate_sequence``.  The routine respects the 1000 s total budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Build a diverse, inexpensive pool of analytic seeds.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 16000]            # moderate sizes; later stages may double them
    beta_grid = np.logspace(-2.0, -0.2, 12)       # for the exponential family
    p_grid = np.concatenate(
        [
            np.logspace(-2.0, -0.5, 12),
            np.logspace(-0.5, 0.0, 12),
            np.logspace(0.0, 0.7, 10),
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 5)
    q_grid = np.linspace(0.5, 3.0, 6)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families -------------------------------------------------
        for cand in (_distribution_uniform(n), _distribution_arcsine(n), _distribution_generalised_normal(n, 2.0)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best (len={n}) {sc:.6f}")

        # exponential (geometric) family -----------------------------------------
        for r in beta_grid:
            cand = _distribution_exponential(n, r)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best exp r={r:.4g} len={n} {sc:.6f}")

        # beta‑symmetric and edge‑weighted families --------------------------------
        for p in p_grid:
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            for base in (beta, edge):
                for lam in lam_grid:
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(
                            f"[{time.time():.1f}] new best p={p:.3g} lam={lam:.2f} len={n} {sc:.6f}"
                        )

        # generalized‑Normal shapes ------------------------------------------------
        for q in q_grid:
            cand = _distribution_generalised_normal(n, q)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best gn q={q:.2f} len={n} {sc:.6f}")

        # random Dirichlet seeds ---------------------------------------------------
        for _ in range(6):
            alpha = rng.uniform(0.01, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best Dirichlet len={n} {sc:.6f}")

        # sparse random seeds -------------------------------------------------------
        for _ in range(6):
            cand = _distribution_random_sparse(n, rng, density=0.01)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best sparse len={n} {sc:.6f}")

        # keep only the absolute best 30 seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2.  Include the previous construction (if present) ----------------
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Guard against an empty result.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Smoothed‑max gradient descent on the most promising seeds.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 40.0 and top_candidates:
        seed_seqs = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seed_seqs))
        print(f"[{time.time():.1f}] smooth‑GD on {len(seed_seqs)} seeds (≈{per_seed:.1f}s each)")
        for idx, seed_seq in enumerate(seed_seqs):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_gradient_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                beta0=0.5,
                beta_max=30.0,
                eta0=0.9,
                eval_every=80,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] after smooth‑GD seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Upsample (if time permits) and polish the upsampled version.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n <= 64000:
            print(f"[{time.time():.1f}] upsampling {cur_n} → {target_n}")
            up_seq = _upsample(overall_best_seq, target_n)
            up_score = _objective(up_seq)
            if up_score < overall_best_score:
                overall_best_score = up_score
                overall_best_seq = up_seq
                print(f"[{time.time():.1f}] upsampled new best {overall_best_score:.6f}")

            # a brief smooth‑GD burst on the upsampled vector
            sub_deadline = time.time() + (deadline - time.time()) * 0.3
            a_opt, sc_opt = _smooth_gradient_descent(
                overall_best_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                beta0=0.6,
                beta_max=20.0,
                eta0=0.7,
                eval_every=60,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] post‑upsample GD new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Final long‑run annealing polish.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing polish")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
            max_trials=None,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after annealing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (2·n·max(conv)/sum²) = {overall_best_score:.6f} "
        f"with length = {len(overall_best_seq)}"
    )
    return overall_best_seq.tolist()
