def _solve_lp_for_b(f, M, rng, deadline):
    from scipy.optimize import linprog
    n = len(f)
    # Build constraint matrix A_ub (size (2n-1, n))
    # This can be built once per f.
    # Use vectorized approach: Use Toeplitz with 0 padding.
    # We'll construct A using loops to avoid memory blow up.
    # Alternative: Use scipy.sparse for efficiency.
    from scipy.sparse import lil_matrix
    rows = 2*n - 1
    A = lil_matrix((rows, n))
    # fill
    for k in range(rows):
        i_min = max(0, k-(n-1))
        i_max = min(k, n-1)
        # i iterates f_i from i_min to i_max
        # corresponding b_j where j=k-i
        # b index j from k-i_max to k-i_min, reversed order
        # fill with f[i]
        # we will fill for each i: j = k - i
        for i in range(i_min, i_max+1):
            j = k - i
            A[k, j] = f[i]
    # Convert to csr for linprog? SciPy's linprog can accept sparse A_ub with method='highs'.
    A = A.tocsr()
    b_ub = np.full(rows, M)
    # objective maximize sum(b) -> minimize -sum(b)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    # Use highs method; set options maybe large runtimes.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})
    if res.success:
        return res.x
    else:
        return None

"""
Hybrid search for a low‑max‑convolution step‑function.

The routine combines three ideas:

1. **LP polishing on moderate lengths** – for n ≤ ~2000 we solve a
   linear program that maximises the ℓ1‑norm of a new vector `b` under the
   constraint `conv(f,b) ≤ M` where `M` is the current worst convolution
   value of `f`.  Adding a convex combination of the solution to `f`
   steadily reduces the peak.  This implements the “extreme point” idea
   described in the statement.

2. **Adam smooth‑max descent** – the best vectors from the LP step (and a
   few analytic families) are refined with an Adam optimiser on the
   smooth‑max surrogate `log‑sum‑exp`.  The temperature parameter is
   increased up to a large value so the surrogate converges to the true
   max.

3. **Final polishing** – a short multimax sub‑gradient descent on the
   exact objective followed by a greedy pairwise‑swap local search.

The algorithm respects the time budget (default 1000 s) and returns the
best probability vector found (scaled to sum = 1, i.e. a valid step‑
function).  The objective value `2·n·max(conv(a,a))/sum(a)^2` is printed
during optimisation.
"""

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full discrete self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  a·w  where w has length 2·n‑1 and sums to 1.
    Returns 2·a⋆w (cross‑correlation) as an ndarray of length n.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective used for internal ranking (sum(a) is assumed =1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    # protect against division by zero (should not happen)
    if a.sum() < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (a.sum() ** 2)

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp(
        (alpha - 1) * np.log(np.clip(x, eps, 1.0))
        + (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0))
    )
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.minimum(i + 1, n - i)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    vals /= vals.sum()
    return vals

def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    shift = np.max(z)
    exp_z = np.exp(z - shift)
    s = exp_z.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / len(z))
    return exp_z / s

def _adam_optimize(z0: np.ndarray,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   alpha0: float = 1.0,
                   alpha_max: float = 2000.0,
                   growth_factor: float = 1.03,
                   lr: float = 0.1,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   max_iter: int = 300000,
                   eval_interval: int = 300) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    The surrogate is `f_α(a) = (1/α)·log Σ_i exp(α·conv_i(a))`.
    The input `z0` is the logit vector such that a = softmax(z0).
    The function returns the best vector seen (on the exact objective)
    together with its score.
    """
    n = z0.shape[0]
    z = z0.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        # probability vector on the simplex
        a = _softmax(z)

        # convolution and smooth‑max weights
        conv = _fft_self_convolve(a)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient wrt a (exact for the smooth surrogate)
        grad_a = _gradient_from_weights(a, w)

        # chain rule to obtain gradient wrt z (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        z = z - step

        # temperature schedule
        if t % 500 == 0:
            alpha = min(alpha * growth_factor, alpha_max)

        # periodic exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.7,
                                 max_iter: int = 50000,
                                 eval_interval: int = 400) -> Tuple[np.ndarray, float]:
    """
    Multimax sub‑gradient descent (exponentiated‑gradient on the simplex).
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)

        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] multimax it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Greedy mass‑transfer walk, accepting only improving moves."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _solve_lp_for_b(f: np.ndarray,
                    M: float,
                    deadline: float) -> Optional[np.ndarray]:
    """
    Solve   maximise Σ b_j   subject to  b ≥ 0 and  conv(f,b) ≤ M.
    Returns the optimal b (not yet normalised) or None on failure.
    """
    try:
        from scipy.optimize import linprog
        from scipy.sparse import lil_matrix
    except Exception as e:
        # SciPy unavailable – skip LP stage.
        return None

    n = f.shape[0]
    rows = 2 * n - 1

    # Build sparse constraint matrix A_ub where each row corresponds to
    # (conv(f,b))_k = Σ_i f_i·b_{k-i}
    A = lil_matrix((rows, n), dtype=np.float64)

    for k in range(rows):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # for each i in [i_min,i_max] set A[k, k-i] = f[i]
        for i in range(i_min, i_max + 1):
            j = k - i
            A[k, j] = f[i]

    A = A.tocsr()
    b_ub = np.full(rows, M, dtype=np.float64)

    # Objective: maximise sum(b)  →  minimise -sum(b)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None) for _ in range(n)]

    # Allocate remaining time to the solver (highs respects a time limit)
    remaining = max(0.0, deadline - time.time())
    options = {"presolve": True, "time_limit": remaining}

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method="highs", options=options)

    if res.success:
        return res.x
    else:
        return None


# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving the ordering, then re‑normalise."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(x_new, x_old, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s == 0.0:
        up = np.full(new_n, 1.0 / new_n)
    else:
        up /= s
    return up

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine mixes a short LP‑polishing stage
    (for moderate lengths) with an Adam‑based smooth‑max descent,
    finishing with multimax sub‑gradient polishing and a greedy local
    walk.  The best vector found within the time budget is returned
    as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    overall_best_seq: Optional[np.ndarray] = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣  LP‑based refinement on moderate lengths (≤2000)
    lengths_lp = [600, 1000, 1500, 2000]
    for n in lengths_lp:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] LP stage length={n}")

        # a few diverse initialisations
        init_seqs = [
            _distribution_arcsine(n),
            _distribution_beta_sym(n, 0.4),
            _distribution_random_sparse(n, rng, density=0.03),
        ]

        for init in init_seqs:
            if time.time() >= deadline:
                break
            a = np.maximum(init, 0.0)
            a_sum = a.sum()
            if a_sum == 0.0:
                continue
            a /= a_sum

            # perform a few LP polishing iterations
            for it in range(8):
                if time.time() >= deadline:
                    break
                conv_a = _fft_self_convolve(a)
                M = conv_a.max()

                b_opt = _solve_lp_for_b(a, M, deadline)
                if b_opt is None:
                    break

                # Normalise b_opt to a probability vector
                b_opt = np.maximum(b_opt, 0.0)
                s = b_opt.sum()
                if s == 0.0:
                    break
                b_opt /= s

                # convex update (t = 0.2 works well)
                t_mix = 0.2
                a = (1.0 - t_mix) * a + t_mix * b_opt
                a = _project_onto_simplex(a)   # keep on simplex

                cur_score = _objective(a)
                if cur_score < overall_best_score:
                    overall_best_score = cur_score
                    overall_best_seq = a.copy()
                    print(f"[{time.time():.1f}] LP len={n} it={it} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Adam‑based optimisation on larger lengths
    lengths_adam = [2000, 4000, 8000, 12000]
    for n in lengths_adam:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Adam stage length={n}")

        # generate a pool of seeds
        seed_pool: List[np.ndarray] = [
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_sine(n),
            _distribution_triangular(n),
            _distribution_gaussian(n),
        ]

        # add up‑sampled version of the current overall best (if any)
        if overall_best_seq is not None:
            seed_pool.append(_upsample(overall_best_seq, n))

        # a few random Dirichlet draws
        for _ in range(3):
            seed_pool.append(_distribution_dirichlet(n, rng, alpha=rng.uniform(0.1, 0.9)))

        # evaluate each seed cheaply and retain the top few
        scored_seeds: List[Tuple[float, np.ndarray]] = []
        for seq in seed_pool:
            a = np.maximum(seq, 0.0)
            a_sum = a.sum()
            if a_sum == 0.0:
                continue
            a /= a_sum
            sc = _objective(a)
            scored_seeds.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] raw seed len={n} score={sc:.6f}")

        scored_seeds.sort(key=lambda x: x[0])
        top_seeds = [seq for _, seq in scored_seeds[:3]]

        # run Adam on each top seed (time‑sliced)
        for seed_seq in top_seeds:
            if time.time() >= deadline:
                break
            slice_time = min(80.0,
                             (deadline - time.time()) / max(1, len(top_seeds)))
            sub_deadline = time.time() + slice_time

            z0 = np.log(np.maximum(seed_seq, 1e-300))

            seq_out, score_out = _adam_optimize(
                z0,
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                alpha0=1.0,
                alpha_max=2000.0,
                growth_factor=1.03,
                lr=0.12,
                max_iter=300000,
                eval_interval=300,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after Adam n={n} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final polishing (multimax + greedy swaps)
    if overall_best_seq is None:
        # fallback – uniform distribution of a reasonable length
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)

    if time.time() < deadline - 5.0:
        sub_deadline = time.time() + (deadline - time.time()) * 0.5
        overall_best_seq, overall_best_score = _subgradient_multimax_descent(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.7,
            max_iter=60000,
            eval_interval=400,
        )
        print(f"[{time.time():.1f}] after multimax best {overall_best_score:.6f}")

    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] after final swaps best {overall_best_score:.6f}")

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
