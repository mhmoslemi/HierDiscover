def _smooth_gradient(a, beta):
    n = len(a)
    conv = _fft_self_convolve(a) # length 2n-1
    maxc = conv.max()
    # shift to avoid overflow
    shifted = conv - maxc
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    w = w / w_sum
    a_rev = a[::-1]
    size = 1 << ((3*n - 2) - 1).bit_length()
    # FFT of w and a_rev both real
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv2 = np.fft.irfft(W * A_rev, n=size)
    # conv2 length size, but we need indices from n-1 to 2n-2 inclusive (i.e., n elements)
    grad = 2.0 * conv2[n-1 : n-1 + n]  # slice appropriate length

"""
search for a non‑negative step function (a list of heights) that
minimises ``evaluate_sequence``.

The algorithm consists of three phases:

1. **Seed generation** – a diverse small pool of probability vectors is
   built from analytic windows (uniform, arcsine, beta, edge‑weighted,
   Hann, Kaiser, Chebyshev, generalized normal, random Dirichlet,
   sparse) for a modest length (≈4000).  The best few seeds are kept.

2. **Smooth‑gradient descent** – instead of using the sub‑gradient of the
   *active* convolution entry we minimise a *soft‑max* surrogate of the
   max‑convolution:
        F(a) = 2·n·log‑sum‑exp(β·(a∗a)) / (∑a)² .
   For a given β the gradient can be computed in O(n log n) via FFT:
   the weights wₖ ∝ exp(β·cₖ) (c = a∗a) are normalised, then
   ∇F(a) = 2·(w ∗ a_rev) (restricted to the original indices), where
   a_rev is a reversed copy of a.  An exponentiated‑gradient update
   (a ← a·exp(−η·∇F)) followed by simplex normalisation is performed.
   β is increased gradually (β≈1 → β≈40) so the optimisation first
   smoothes the landscape and then focuses on the true max.

3. **Polishing** – the best vector is up‑sampled (linear interpolation)
   to a larger length, a short second smooth‑gradient phase is run,
   and finally a long‑run pairwise‑mass‑transfer simulated‑annealing
   polishing step is applied.

The whole procedure respects the 1000 s time budget (a 2 s safety
margin is kept).  The function returns the best list of floats found.
"""

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, linear convolution)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective used by ``evaluate_sequence`` (sum fixed to 1 for
# optimisation, scaling does not change the objective).
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic families (return a probability vector, sum=1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) density on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{-p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(-|x-0.5|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# ----------------------------------------------------------------------
# Smoothed gradient of the soft‑max surrogate of max(conv)
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute ∇_a  log‑sum‑exp(β·(a*a))  for a fixed sum (the factor 2·n/∑a²
    is ignored because we keep ∑a = 1).  The result is a vector of length n.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)                     # length 2n‑1
    max_c = conv.max()
    # soft‑max weights (numerically stable)
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.shape[0])
    else:
        w /= w_sum

    # gradient = 2 * (w * a_rev)  (restricted to indices n‑1 … 2n‑2)
    a_rev = a[::-1]

    size = 1 << ((len(w) + len(a_rev) - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A = np.fft.rfft(a_rev, n=size)
    conv2 = np.fft.irfft(W * A, n=size)

    # slice the relevant part (i + n‑1 for i=0..n‑1)
    grad = 2.0 * conv2[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the smoothed objective
# ----------------------------------------------------------------------
def _smooth_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    beta0: float = 1.0,
    beta_max: float = 40.0,
    beta_inc: float = 1.02,
    eval_every: int = 50,
) -> tuple[np.ndarray, float]:
    """
    Perform EG descent on a soft‑max surrogate.
    The schedule slowly raises β so we first smooth the objective then
    focus on the true max.  After each update the vector is projected back
    onto the probability simplex.
    """
    n = a.shape[0]
    t = 0
    beta = beta0
    while time.time() < deadline:
        t += 1
        grad = _smooth_gradient(a, beta)

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        beta = min(beta_max, beta * beta_inc)

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] smooth‑GD it={t} β={beta:.2f} "
                    f"new best {best_score:.6f} (len={n})"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style pairwise mass transfers (unchanged)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)

        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal i={i}→j={j} δ={delta:.3e} "
                    f"new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation while preserving the simplex)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= a_new.sum()
    return a_new

                break
            alpha = rng.uniform(0.01, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best Dirichlet score={sc:.6f}")

        # Sparse random seeds
        for _ in range(8):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.01)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best sparse score={sc:.6f}")

        # keep only the best ~30 seeds
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # --------------------------------------------------------------
    # 2. Include a previously‑found construction if available
    # --------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = np.full_like(prev, 1.0 / prev.size)
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3. Guarantee a non‑empty result
    # --------------------------------------------------------------
    if overall_best_seq is None:
        overall_best_seq = _distribution_uniform(n_base)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Smooth gradient descent on the few most promising seeds
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seed_seqs = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seed_seqs))
        print(f"[{time.time():.1f}] smooth‑GD on {len(seed_seqs)} seeds")
        for idx, seed_seq in enumerate(seed_seqs):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_gradient_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.9,
                beta0=1.0,
                beta_max=40.0,
                beta_inc=1.02,
                eval_every=80,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] after smooth‑GD seed {idx} new best {overall_best_score:.6f}"
                )

    # --------------------------------------------------------------
    # 5. Upsample and refine (if time allows)
    # --------------------------------------------------------------
    if time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n <= 64000:
            print(f"[{time.time():.1f}] upsampling from {cur_n} → {target_n}")
            up_seq = _upsample(overall_best_seq, target_n)
            up_score = _objective(up_seq)
            if up_score < overall_best_score:
                overall_best_score = up_score
                overall_best_seq = up_seq
                print(f"[{time.time():.1f}] upsampled new best {overall_best_score:.6f}")

            # short smooth‑GD burst on the upsampled vector
            sub_deadline = time.time() + (deadline - time.time()) * 0.3
            a_opt, sc_opt = _smooth_gradient_descent(
                overall_best_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.7,
                beta0=5.0,
                beta_max=80.0,
                beta_inc=1.015,
                eval_every=60,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] post‑upsample smooth‑GD new best {overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Final long‑run annealing polish
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing polish")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
            max_trials=None,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after annealing new best {overall_best_score:.6f}")

    # --------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)/sum²): {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
