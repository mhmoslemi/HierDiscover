"""
Hybrid optimiser with an additional linear‑programming refinement.

The routine proceeds in four stages:

1. **Broad analytic seeding** – a large collection of one‑parameter families
   (uniform, arcsine, symmetric Beta, power‑law, two‑block, Dirichlet, …) is
   instantiated for a fixed length *n* (≈ 4000) and the exact objective
   `2·n·max(conv)/sum²` is evaluated.  The best seed becomes the current
   incumbent.

2. **Exponentiated‑gradient descent** – a soft‑max stage followed by a hard‑max
   stage (identical to the previous state‑of‑the‑art) refines the incumbent
   for many iterations, using a decaying step size and a temperature ramp.

3. **Linear‑programming boost** – given a current vector *f* we compute
   `M = max(f*f)`.  The linear program

        max   Σ b_j
        s.t.  (f * b)_k ≤ M   for all k,
              b_j ≥ 0

   finds a new distribution *b* with a larger ℓ₁‑norm while keeping the same
   convolution bound.  After normalising *b* to unit sum we mix it with *f*
   (mixing factor `t≈0.15`).  Repeating this step strictly decreases the
   objective and quickly escapes shallow local minima.

4. **Cheap greedy polishing** – a random mass‑transfer walk accepts moves only
   when the true objective improves.

The whole procedure respects the given time budget (default 1000 s) and
returns the best sequence found as a plain Python list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of *v* onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a))/sum(a)².
    a is assumed non‑negative.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Sub‑gradient 2·a⋆w for a weight vector w (length 2n‑1, sum(w)=1)."""
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.size - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _gradient_max_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the single maximal convolution entry ``conv[k]``.
    Returns a non‑negative vector of size len(a).
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i = np.arange(i_min, i_max + 1, dtype=np.int64)
    j = k - i

    eq_mask = (i == j)
    if np.any(eq_mask):
        g[i[eq_mask]] += 2.0 * a[i[eq_mask]]

    if np.any(~eq_mask):
        ii = i[~eq_mask]
        jj = j[~eq_mask]
        g[ii] += a[jj]
        g[jj] += a[ii]

    return g


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((p - 1) * np.log(np.clip(x, eps, 1.0)) +
               (p - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _softmax_grad_descent(a: np.ndarray, deadline: float,
                          best_seq: np.ndarray, best_score: float,
                          lr0: float = 0.9,
                          max_iter: int = 120000,
                          eval_interval: int = 400,
                          alpha_start: float = 0.4,
                          alpha_end: float = 120.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a temperature‑controlled soft‑max
    approximation of the maximal convolution entry.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_conv = conv.max()

        # linear schedule for temperature α
        alpha = alpha_start + (alpha_end - alpha_start) * t / max_iter

        # soft‑max weights (stable)
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # renormalise onto the simplex
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] soft‑max it={t} α={alpha:.1f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _max_grad_descent(a: np.ndarray, deadline: float,
                      best_seq: np.ndarray, best_score: float,
                      lr0: float = 0.5,
                      max_iter: int = 120000,
                      eval_interval: int = 400) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent using the true maximal entry of the convolution.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())

        grad = _gradient_max_index(a, k)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hard‑max it={t} idx={k} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer walk; only keep moves that strictly improve the objective.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue

        delta = a[i] * rng.random() * 0.30   # move at most 30 % of source mass
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _lp_refine_step(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the LP  max Σ b_j  s.t.  (f*b)_k ≤ M,  b ≥ 0  and return the
    normalised solution g (sum(g)=1).  The Toeplitz constraint matrix is
    built explicitly; the function is deliberately kept small because it
    is called only a handful of times.
    """
    n = f.shape[0]

    # Build the Toeplitz matrix for the convolution constraints
    col = np.concatenate([f, np.zeros(n - 1, dtype=np.float64)])
    row = np.concatenate([f[:1], np.zeros(n - 1, dtype=np.float64)])
    A = toeplitz(col, row)[: 2 * n - 1, :]   # shape (2n‑1, n)

    # Right‑hand side (all constraints share the same bound M)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Linear programme: maximise Σ b_j  ⇔ minimise −Σ b_j
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})

    if not res.success:
        # fallback: keep the original vector
        return f.copy()

    b = res.x
    if b.sum() == 0.0:
        return f.copy()

    # Normalise to unit ℓ₁‑norm (the rest of the code works with sum=1)
    g = b / b.sum()
    return g


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Entry point required by the judge.  The algorithm follows the four‑stage
    pipeline described in the module docstring and returns the best sequence
    (plain ``list[float]``) found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # small safety margin

    # --------------------------------------------------------------
    # 1. Length choice – a medium size that balances quality and LP cost.
    n = 4000

    # --------------------------------------------------------------
    # 2. Generate a wide pool of analytic seeds.
    candidates = []

    candidates.append(_distribution_uniform(n))
    candidates.append(_distribution_arcsine(n))
    # symmetric Beta on a fine log‑grid (covers very heavy tails)
    for p in np.logspace(-3, 0, 30):         # p ∈ [0.001, 1]
        candidates.append(_distribution_beta_sym(n, p))
    # power‑law increasing shapes
    for p in np.linspace(0.2, 4.0, 15):
        candidates.append(_distribution_power_increasing(n, p))
    # a few random Dirichlet draws (different concentrations)
    for _ in range(8):
        alpha = 0.1 + 1.9 * rng.random()
        candidates.append(_distribution_dirichlet(n, rng, alpha=alpha))
    # two‑block patterns with varying block size
    for frac in np.linspace(0.01, 0.20, 10):
        candidates.append(_distribution_two_blocks(n, block_frac=frac))

    # --------------------------------------------------------------
    # 3. Incorporate the previously produced sequence, if any.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            prev = prev / prev.sum()
            candidates.append(prev)
    except Exception:
        pass

    # --------------------------------------------------------------
    # 4. Scan all seeds and keep the best one.
    best_seq = None
    best_score = float("inf")
    for cand in candidates:
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()
        sc = _objective(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time():.1f}] initial new best score {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Two‑phase exponentiated‑gradient descent (soft‑max → hard‑max).
    def descend(seq: np.ndarray, time_budget: float) -> tuple[np.ndarray, float]:
        seq, sc = _softmax_grad_descent(seq, time_budget, seq, best_score,
                                        lr0=0.9, max_iter=120000,
                                        eval_interval=400,
                                        alpha_start=0.4, alpha_end=120.0)
        seq, sc = _max_grad_descent(seq, time_budget, seq, sc,
                                    lr0=0.5, max_iter=120000,
                                    eval_interval=400)
        return seq, sc

    # Use a few random perturbations of the current incumbent.
    while time.time() < deadline - 15.0:
        # a slight random jitter helps to escape plateaus
        jitter = 0.02 * (rng.random(best_seq.shape[0]) - 0.5)
        pert = best_seq * (1.0 + jitter)
        pert = np.maximum(pert, 0.0)
        pert /= pert.sum()
        sub_deadline = min(deadline, time.time() + 30.0)
        seq, sc = descend(pert, sub_deadline)
        if sc < best_score - 1e-12:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] after gradient descent new best {best_score:.6f}")

        # stop early if we already beat the target
        if best_score <= 1.503:
            break

    # --------------------------------------------------------------
    # 6. Linear‑programming boost.
    while time.time() < deadline - 8.0:
        conv = _fft_self_convolve(best_seq)
        M = conv.max()
        g = _lp_refine_step(best_seq, M)

        # mixing factor – small enough to keep progress monotone
        t_mix = 0.15
        new_seq = (1 - t_mix) * best_seq + t_mix * g
        new_seq = _project_onto_simplex(new_seq, s=1.0)

        new_score = _objective(new_seq)
        if new_score < best_score - 1e-12:
            best_score, best_seq = new_score, new_seq.copy()
            print(f"[{time.time():.1f}] after LP step new best {best_score:.6f}")
        else:
            # No improvement – break to avoid wasting time on further LPs
            break

        if best_score <= 1.503:
            break

    # --------------------------------------------------------------
    # 7. Final cheap polishing.
    best_seq, best_score = _pairwise_swap_local(best_seq.copy(),
                                                deadline,
                                                best_seq,
                                                best_score)

    # --------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    # Clip to the bounds enforced by ``evaluate_sequence`` before returning.
    best_seq = np.clip(best_seq, 0.0, 1000.0).tolist()
    return best_seq
