assistantfinal**Solution Overview**

The goal is to return a non‑negative vector `a` that minimizes  

```
score(a) = 2·n·max(conv(a,a)) / (∑a)²      (n = len(a))
```

The evaluation is performed by the supplied `evaluate_sequence` function, which
clips every entry to the interval [0, 1000] and rejects empty or invalid inputs.
The problem is a non‑convex optimisation on the probability simplex
(`a ≥ 0`, `∑a = 1`).  The implementation below follows a *multi‑start* strategy
combined with three optimisation phases:

1. **Broad seeding** – a large collection of analytic families (uniform,
   beta, power‑law, exponential, sinusoidal, Gaussian, two‑block,
   random‐sparse, Dirichlet) together with a *random‑phase* spectral
   construction.  All seeds are projected onto the simplex and the `score`
   (computed via `evaluate_sequence`) is measured.  The 12 best seeds are kept.

2. **Smooth‑max Adam optimisation** – each seed is refined with Adam on the
   smooth approximation  

   ```
   φ_α(a) = (1/α)·log( Σ_i exp(α·conv_i(a)) )
   ```

   where `conv_i(a)` denotes the i‑th entry of the self‑convolution.
   The temperature `α` is increased slowly from 1 to 10 000, which makes the
   surrogate converge to the true max‑convolution.  The variables are kept on
   the simplex through a soft‑max parametrisation `a = softmax(z)`.  The best
   vector found in this phase is retained.

3. **Polishing** – a short exponentiated‑gradient (multimax) sub‑gradient
   descent followed by a *simulated‑annealing* pair‑wise mass‑transfer walk.
   Both stages use the exact objective (`evaluate_sequence`) for acceptance,
   guaranteeing that the final result is never worse than the best vector from
   phase 2.

All steps obey a global time budget (`budget_s`).  The routine always returns
a valid sequence – in the worst case a uniform distribution – so the
submission never triggers the “no‑output” penalty.

"""
Hybrid multi‑start optimiser for the step‑function max‑convolution problem.

The algorithm proceeds as follows:
  1. Generate a wide variety of analytic and random seeds for several

  2. Refine every retained seed with Adam on a smooth‑max surrogate

  3. Polishing phase – exponentiated‑gradient sub‑gradient descent and a
     simulated‑annealing pairwise‑swap walk.
All vectors stay on the probability simplex (non‑negative, sum = 1).

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # Fast path: already feasible
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution a*a using a real FFT.
    Length of the result = 2·n‑1 (n = len(a)).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    shift = np.max(z)
    exp_z = np.exp(z - shift)
    s = exp_z.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / len(z))
    return exp_z / s


# ----------------------------------------------------------------------
def _smooth_max(conv: np.ndarray, alpha: float) -> Tuple[float, np.ndarray]:
    """
    Compute a smooth approximation of max(conv) together with the
    normalised weights w_i = exp(alpha·conv_i) / Σ exp(alpha·conv_i).
    """
    shift = conv.max()
    w = np.exp(alpha * (conv - shift))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum
    smoothed = (w * conv).sum() + shift  # = (1/α)·log Σ exp(α·conv_i)
    return smoothed, w

def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Exact gradient of the smooth‑max surrogate w.r.t. a.
    It equals 2·a ⋆ w where '⋆' denotes the (linear) cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    corr = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * corr[start:start + n]
    return grad

# Analytic families (all return probability vectors, i.e. sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((a - 1) * np.log(np.clip(x, eps, 1.0)) +
               (b - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.minimum(i + 1, n - i)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_random_phase(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Spectral construction: random phases, unit magnitude spectrum.
    The inverse real FFT is taken, absolute values are used and the result
    is projected onto the simplex.
    """
    # Number of non‑redundant frequencies in rfft
    m = n // 2 + 1
    # Random phases for the complex components (except DC & Nyquist)
    phases = rng.uniform(0, 2 * np.pi, size=m)
    spec = np.empty(m, dtype=np.complex128)

    # DC component must be real → choose a small random amplitude
    spec[0] = rng.uniform(0.0, 1.0)

    # Nyquist term for even n must also be real
    if n % 2 == 0:
        spec[-1] = rng.uniform(0.0, 1.0)
        upper = m - 1
    else:
        upper = m

    # Fill the remaining bins with unit magnitude and random phase
    spec[1:upper] = np.exp(1j * phases[1:upper])

    # Inverse FFT -> real signal
    raw = np.fft.irfft(spec, n=n)

    # Ensure non‑negative entries, then project onto the simplex
    raw = np.maximum(raw, 0.0)
    raw = _project_onto_simplex(raw, s=1.0)
    return raw

def _adam_refine(z0: np.ndarray,
                 deadline: float,
                 start_score: float,
                 start_seq: np.ndarray,
                 alpha_max: float = 1200.0,
                 growth: float = 1.015,
                 lr: float = 0.12,
                 max_iters: int = 300000,
                 eval_every: int = 250) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    Parameters:
        - z0          : initial logits (real vector, size = n)
        - deadline    : absolute time after which we stop
        - start_score : best score known before this run (for early exit)
        - start_seq   : the corresponding probability vector
    Returns the best probability vector found and its exact score.
    """
    n = z0.shape[0]
    z = z0.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    best_seq = start_seq.copy()
    best_score = start_score

    t = 0
    alpha = 1.0
    while t < max_iters and time.time() < deadline:
        t += 1

        # Soft‑max → probability vector
        a = _softmax(z)

        # Full convolution of the current iterate
        conv = _fft_self_convolve(a)

        # Smooth‑max surrogate and its weight vector
        _, w = _smooth_max(conv, alpha)

        # Gradient of the surrogate w.r.t. a, then back‑prop to z
        grad_a = _gradient_from_weights(a, w)
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = 0.9 * m + 0.1 * grad_z
        v = 0.999 * v + 0.001 * (grad_z ** 2)

        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        z -= step

        # Temperature schedule
        if t % 400 == 0:
            alpha = min(alpha * growth, alpha_max)

        # Periodic exact evaluation
        if t % eval_every == 0:
            cur_score = _objective_exact(a)
            if cur_score < best_score - 1e-13:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _multimax_subgradient(a: np.ndarray,
                          deadline: float,
                          best_seq: np.ndarray,
                          best_score: float,
                          lr0: float = 0.6,
                          max_iters: int = 40000,
                          eval_every: int = 300) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the true max‑convolution.
    """
    n = a.shape[0]
    t = 0
    while t < max_iters and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        # Uniform weights on all tight constraints
        mask = (conv >= max_val - 1e-14)
        cnt = mask.sum()
        w = mask.astype(np.float64) / max(cnt, 1)

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_every == 0:
            cur_score = _objective_exact(a)
            if cur_score < best_score - 1e-13:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] multimax it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         initial_temp: float = 0.3,
                         cooling: float = 0.995,
                         max_trials: int = 250000) -> Tuple[np.ndarray, float]:
    """
    Simple pairwise mass‑transfer walk with Metropolis acceptance.
    At each step a small amount of probability mass δ is moved from i → j.
    The move is accepted if it improves the objective or with probability
    exp( -Δ / T ).
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    T = initial_temp
    cur = a.copy()
    cur_score = _objective_exact(cur)

    for step in range(1, max_trials + 1):
        if time.time() >= deadline:
            break

        # pick a source index with positive mass
        src_candidates = np.where(cur > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        # pick a destination (different) index uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # propose a transfer: δ ∈ (0, 0.3·cur[i])
        delta = cur[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        # evaluate
        cand_score = _objective_exact(cand)

        improve = cand_score < cur_score - 1e-13
        if improve or rng.random() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur = cand
            cur_score = cand_score
            if cur_score < best_score - 1e-13:
                best_score = cur_score
                best_seq = cur.copy()
                print(f"[{time.time():.1f}] SA step={step} i→j {i}->{j} δ={delta:.3g} new best {best_score:.6f}")

        # temperature schedule
        if step % 1000 == 0:
            T *= cooling
            if T < 1e-6:
                T = 1e-6

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function minimizing ``evaluate_sequence``.
    The routine respects the supplied time budget and returns the best sequence
    as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # a small safety margin

    # ------------------------------------------------------------------
    # 1. Choose a set of lengths (moderate n keeps each iteration cheap)
    lengths = [1500, 3000, 4500, 6000]   # up to ~6 k entries – enough flexibility

    overall_best_seq: Optional[np.ndarray] = None
    overall_best_score = float("inf")
    elite: List[Tuple[float, np.ndarray]] = []   # (score, vector)

    # ------------------------------------------------------------------
    # 2. Broad‐seed generation & cheap evaluation
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating seeds of length {n}")

        for cand in _initial_seeds(n, rng):
            if time.time() >= deadline:
                break
            # ensure feasibility
            cand = np.maximum(cand, 0.0)
            cand = _project_onto_simplex(cand, s=1.0)
            score = _objective_exact(cand)
            elite.append((score, cand.copy()))
            if score < overall_best_score:
                overall_best_score = score
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] **** new global best len={n} score={score:.6f}")

        # keep only the most promising seeds (across all lengths)
        elite.sort(key=lambda x: x[0])
        elite = elite[:12]   # 12 best seeds go to the refinement stage

    # ------------------------------------------------------------------
    # 3. If a previous solution is available, add it to the pool
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev = _project_onto_simplex(prev, s=1.0)
            sc = _objective_exact(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous solution improved: {sc:.6f}")
            elite.append((sc, prev.copy()))
            elite.sort(key=lambda x: x[0])
            elite = elite[:12]
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Adam refinement on the elite seeds
    if overall_best_seq is None:
        # fallback – uniform distribution on the smallest length
        overall_best_seq = _distribution_uniform(lengths[0])
        overall_best_score = _objective_exact(overall_best_seq)

    remaining = deadline - time.time()
    if remaining > 30.0:
        top_seeds = [vec for _, vec in elite[:8]]
        for seed_vec in top_seeds:
            if time.time() >= deadline:
                break
            # allocate a slice of the remaining budget (at least 8 s per seed)
            slice_time = min(45.0, (deadline - time.time()) / max(1, len(top_seeds)))
            sub_deadline = time.time() + slice_time

            # Initialise Adam on log‑space (soft‑max parametrisation)
            z0 = np.log(np.maximum(seed_vec, 1e-300))

            seq, sc = _adam_refine(z0,
                                   sub_deadline,
                                   overall_best_score,
                                   overall_best_seq,
                                   alpha_max=1200.0,
                                   growth=1.02,
                                   lr=0.13,
                                   max_iters=250000,
                                   eval_every=300)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = seq.copy()
                print(f"[{time.time():.1f}] after Adam new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Multimax sub‑gradient polishing
    remaining = deadline - time.time()
    if remaining > 25.0:
        sub_deadline = time.time() + remaining * 0.55
        overall_best_seq, overall_best_score = _multimax_subgradient(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.65,
            max_iters=35000,
            eval_every=400,
        )
        print(f"[{time.time():.1f}] after multimax best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Simulated‑annealing walk (final fine‑tuning)
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _simulated_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            initial_temp=0.5,
            cooling=0.997,
            max_trials=300000,
        )
        print(f"[{time.time():.1f}] after SA final best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Finished – best score {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
