        # Adam update

"""
propose_candidate
-----------------

A fast “soft‑max‑gradient” optimiser for the non‑negative step‑function problem.

* Initialise a small, diverse set of seeds (uniform, arcsine, several
  beta‑type families, random Dirichlet and a previous solution if present).
* For each seed run an Adam‑style gradient descent on a **smooth approximation**
  of the objective:
      smooth_maxβ(a) = (1/β)·log ∑ exp(β·(a*a)),
  whose gradient can be written as a correlation of the soft‑max weights with
  the current vector.  The gradient is evaluated via real FFTs in O(n log n).
* The temperature β is increased gradually (β≈0.5 → 80) so the optimisation
  first flattens the whole convolution and then focuses on the true maximum.
* After the descent we polish the result with the deterministic pairwise
  mass‑transfer and a cheap simulated‑annealing local search (the same utilities
  used in the previous state‑of‑the‑art code).
* The whole routine respects the 1000 s budget and returns a plain list of
  heights ready for ``evaluate_sequence``.
"""
import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    """Second edge family: w_i ∝ (i+0.5)^(−p) + (n−i−0.5)^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (-p) + j ** (-p))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight proportional to sin(π (i+½)/n)^p, vanishing at the ends."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet sample with a random concentration."""
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
def _interpolate_to_length(old: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a distribution to a new length."""
    if old.size == new_n:
        return old.copy()
    old_x = np.linspace(0.0, 1.0, old.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    new = np.interp(new_x, old_x, old)
    new = np.maximum(new, 0.0)
    new /= new.sum()
    return new


# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights for a given temperature β."""
    max_c = conv.max()
    # subtract max for numerical stability
    exp_vals = np.exp(beta * (conv - max_c))
    Z = exp_vals.sum()
    return exp_vals / Z


# ----------------------------------------------------------------------
def _grad_smooth_max(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth max (log‑sum‑exp) of the self‑convolution.
    Implements   ∇  (1/β) log Σ exp(β·(a*a))  = 2·corr(w, a)
    where w are the soft‑max weights.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)               # length 2n‑1
    w = _smooth_weights(conv, beta)            # same shape as conv

    # correlation w ⋆ a (valid part, length n)
    size = 1 << ((len(w) + n - 1).bit_length())   # power of two ≥ len(w)+len(a)−1
    Wf = rfft(w, n=size)
    Af = rfft(a, n=size)
    corr_full = irfft(Wf * np.conj(Af), n=size)   # real result, length = size
    start = n - 1
    end = len(w)                                 # 2n‑1
    grad = 2.0 * corr_full[start:end]            # length n
    return grad


# ----------------------------------------------------------------------
def _smooth_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style descent on the smooth max of the convolution.
    The temperature β is increased geometrically (0.5 → 80) so the algorithm
    starts with a very smooth objective and ends by approximating the true
    max.  The iterate is projected onto the probability simplex after each
    step.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta = 0.5
    beta_max = 80.0
    lr0 = 0.02
    it = 0
    no_improve = 0

    while time.time() < deadline and it < 2_000_000:
        it += 1

        # increase temperature slowly
        if it % 500 == 0 and beta < beta_max:
            beta = min(beta * 1.5, beta_max)

        grad = _grad_smooth_max(a, beta)

        # Adam update (with bias correction)
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad ** 2)
        m_hat = m / (1.0 - b1 ** it)
        v_hat = v / (1.0 - b2 ** it)
        eta = lr0 / math.sqrt(it)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a, s=1.0)

        # occasional exact evaluation
        if it % 200 == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={it:,} β={beta:.2f} new best {best_score:.6f}")
            else:
                no_improve += 1

        # jitter to escape flat plateaus
        if no_improve >= 4000:
            rng = np.random.default_rng()
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small mass; keep the move only if the objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score
