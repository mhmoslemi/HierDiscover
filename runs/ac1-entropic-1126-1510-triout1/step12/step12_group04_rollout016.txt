def _conv_fft(a, b):
    n = len(a) + len(b) - 1
    size = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[:n]

def _smooth_gradient(a, beta):
    conv = _fft_self_convolve(a) # length 2n-1
    maxc = conv.max()
    # shift conv to compute w = exp(beta*(conv - maxc)), normalized.
    exp_shift = np.exp(beta * (conv - maxc))
    w = exp_shift / exp_shift.sum()
    # compute term2 = conv(w, a) (full) -> length 3n-2

def _smooth_descent(a, deadline, best_seq,best_score, eta0=0.5, beta_start=0.5, beta_max=20., beta_tot_iters=50000):

    t = 0
    beta_inc = (beta_max - beta_start) / max(beta_tot_iters,1)
    while time.time() < deadline:
        # compute beta: 
        beta = min(beta_max, beta_start + beta_inc * t)
        grad, conv = _smooth_gradient(a, beta)
        max_conv = conv.max()
        cur_score = 2.0 * len(a) * max_conv / (a.sum()**2)
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] iter={t} beta={beta:.2f} new best {best_score:.6f}")

        # compute step size normalized
        grad_max = grad.max()
        if grad_max == 0:
            # no gradient, break
            break
        step = eta0 / ((np.sqrt(t+1)) * (grad_max + 1e-12))
        a = a * np.exp(- step * grad)

        # project onto simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / len(a))
        else:
            a = a / s

        t += 1

    return best_seq, best_score

def _greedy_polish(a, deadline, best_seq, best_score):
    n = len(a)
    rng = np.random.default_rng()
    while time.time() < deadline:
        # choose src index with mass > eps

        i = rng.choice(indices)

"""
smooth‑descent step‑function optimizer

The algorithm explores several lengths (2000‒8000) and a variety of
analytic seeds (uniform, arcsine, power‑inverse‑sqrt, two‑block,
sparse, Dirichlet).  For each seed we run a *continuation* gradient
descent: the objective 2·n·max(conv(a,a)) is approximated by a smooth
log‑sum‑exp of the convolution, whose gradient can be obtained by FFT
in O(n log n).  The smoothing parameter β starts small (global view) and
is increased linearly to a large value (≈20) so that the method first
moves in a smoothed landscape and later focuses on the true maximum.
Updates are performed with an exponentiated‑gradient (mirror descent on
the simplex) and a step‑size η≈η₀/√t, normalised by the gradient’s
∞‑norm.  After the descent a cheap greedy polishing (random mass
transfer) is applied.  The best sequence seen across all lengths and
seeds is returned.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of x and y using real FFT."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)²."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine_offset(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = i + c
    right = (n - 1 - i) + c
    vals = 1.0 / np.sqrt(left * right)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _distribution_power_inv_sqrt(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = i + 0.5
    right = (n - 1 - i) + 0.5
    vals = (left * right) ** (-p / 2.0)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the smoothed objective (log‑sum‑exp of the convolution).
    Returns (grad, conv) where conv is the exact self‑convolution.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)                # length 2n‑1
    maxc = conv.max()

    # soft‑max weights (stable because we subtract maxc)
    exp_vals = np.exp(beta * (conv - maxc))
    w = exp_vals / exp_vals.sum()               # length 2n‑1, non‑negative, sum=1

    # term2 = (w * a)  (linear convolution)
    term2_full = _conv_fft(w, a)                # length 3n‑2
    term2 = term2_full[:n]

    # term1 = (w * a_rev)  ; then offset by n‑1
    a_rev = a[::-1]
    term1_full = _conv_fft(w, a_rev)            # length 3n‑2
    term1 = term1_full[n - 1 : n - 1 + n]

    grad = term1 + term2
    return grad, conv

def _smooth_descent(a: np.ndarray, deadline: float,
                    best_seq: np.ndarray, best_score: float,
                    eta0: float = 0.5,
                    beta_start: float = 0.5,
                    beta_max: float = 20.0,
                    beta_tot_iters: int = 30000) -> tuple[np.ndarray, float]:
    """
    Continuation gradient descent.
    Executes until `deadline`.  Returns the best sequence seen.
    """
    n = a.shape[0]
    t = 0
    beta_inc = (beta_max - beta_start) / max(beta_tot_iters, 1)

    while time.time() < deadline:
        beta = min(beta_max, beta_start + beta_inc * t)

        grad, conv = _smooth_gradient(a, beta)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] iter={t:5d} beta={beta:5.2f} "
                  f"new best {best_score:.6f}")

        # normalise step size by the ∞‑norm of the gradient
        grad_max = grad.max()
        if grad_max == 0:
            break
        step = eta0 / ((math.sqrt(t + 1)) * (grad_max + 1e-12))

        # exponentiated‑gradient update on the simplex
        a = a * np.exp(-step * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        t += 1

    return best_seq, best_score


# ----------------------------------------------------------------------
def _greedy_polish(a: np.ndarray, deadline: float,
                  best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Random mass‑transfer walk that accepts only improving moves."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue
        delta = a[i] * rng.random() * 0.40            # up to 40 % of source
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur_score = _objective(new_a)
        if cur_score < best_score - 1e-13:
            a = new_a
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] polish {i}->{j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Entry point for the optimisation.
    Returns a list of non‑negative heights (the best sequence found).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # incorporate any previously known construction, if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                print(f"[{time.time():.1f}] start from known sequence score={sc:.6f}")
        # keep it as a seed for later lengths
        known_seed = prev
    except Exception:
        known_seed = None

    # ------------------------------------------------------------------
    # candidate lengths (trade‑off between quality and speed)
    lengths = [2000, 3000, 4000, 5000, 8000]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] --- exploring length {n} ---")

        # ---- generate a modest pool of analytic seeds -----------------
        seeds = []

        seeds.append(_distribution_uniform(n))
        for c in (0.2, 0.5, 1.0, 2.0):
            seeds.append(_distribution_arcsine_offset(n, c))
        for p in (0.7, 0.9, 1.0, 1.1, 1.3):
            seeds.append(_distribution_power_inv_sqrt(n, p))
        seeds.append(_distribution_two_blocks(n, 0.05))
        seeds.append(_distribution_two_blocks(n, 0.10))

        # random sparse / Dirichlet draws
        for _ in range(2):
            seeds.append(_distribution_random_sparse(n, rng, density=0.01))
        for _ in range(3):
            seeds.append(_distribution_dirichlet(n, rng,
                                                alpha=0.3 + 0.7 * rng.random()))

        # possibly reuse the known solution (truncated / padded)
        if known_seed is not None:
            tmp = known_seed.copy()
            if tmp.size > n:
                tmp = tmp[:n]
            elif tmp.size < n:
                tmp = np.pad(tmp, (0, n - tmp.size), "constant")
            tmp = np.maximum(tmp, 0.0)
            if tmp.sum() > 0:
                tmp = tmp / tmp.sum()
                seeds.append(tmp)

        # ---- evaluate all seeds and keep the best few ----------------
        scored_seeds = []
        for s in seeds:
            if time.time() >= deadline:
                break
            s = np.maximum(s, 0.0)
            ssum = s.sum()
            if ssum == 0.0:
                continue
            s = s / ssum
            sc = _objective(s)
            scored_seeds.append((sc, s))
            if sc < best_score:
                best_score = sc
                best_seq = s.copy()
                print(f"[{time.time():.1f}] init seed len={n} score={sc:.6f}")

        # keep only the elite seeds for the intensive phase
        scored_seeds.sort(key=lambda x: x[0])
        elite = [seq for _, seq in scored_seeds[:3]]

        # ---- allocate time for local optimisation --------------------
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # give each elite seed roughly the same share (but keep some for polishing)
        per_seed_time = remaining * 0.65 / max(len(elite), 1)

        for seed_seq in elite:
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed_time
            best_seq, best_score = _smooth_descent(
                seed_seq.copy(),
                deadline=sub_dead,
                best_seq=best_seq,
                best_score=best_score,
                eta0=0.5,
                beta_start=0.5,
                beta_max=20.0,
                beta_tot_iters=40000,
            )
            if best_seq is not None:
                print(f"[{time.time():.1f}] after descent best {best_score:.6f}")

        # ---- final greedy polishing for this length ------------------
        if best_seq is not None and time.time() < deadline:
            best_seq, best_score = _greedy_polish(
                best_seq.copy(),
                deadline,
                best_seq,
                best_score,
            )

    # ------------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    # the evaluator clips values to [0,1000]; our numbers are already safe.
    return best_seq.tolist()
