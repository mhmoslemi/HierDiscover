def _optimize_sequence(seed, sub_deadline, best_seq, best_score):

   n = len(seed)

   epsilon = 1e-12

   x = np.log(seed + epsilon)  # shape n.

   m = np.zeros_like(x)

   v = np.zeros_like(x)

   beta1 = 0.9

   beta2 = 0.999

   eps = 1e-8

   t_iter = 0

   beta = 5.0

   beta_max = 300.0

   inc = 1.01  # Increase factor per iteration maybe

   eval_interval = 50

   while time.time() < sub_deadline:

         t_iter += 1

         # compute a (softmax)
         max_x = x.max()
         exp_x = np.exp(x - max_x)
         a = exp_x / exp_x.sum()

         # compute conv = a*a via FFT
         conv = _fft_self_convolve(a)

         # compute softmax weights w
         rc = beta * conv
         rc_max = rc.max()
         exp_rc = np.exp(rc - rc_max)
         Z = exp_rc.sum()
         w = exp_rc / Z

         # correlation to compute grad_a
         a_rev = a[::-1]
         corr_full = _fft_convolve(w, a_rev)
         # get grad_a segment
         n_len = n
         grad_a = 2.0 * corr_full[n_len - 1: n_len - 1 + n_len]  # length n

         # chain rule to grad_x
         grad_x = a * (grad_a - np.dot(grad_a, a))

         # Adam update
         m = beta1 * m + (1 - beta1) * grad_x
         v = beta2 * v + (1 - beta2) * (grad_x ** 2)
         m_hat = m / (1 - beta1 ** t_iter)
         v_hat = v / (1 - beta2 ** t_iter)
         lr = 0.2 / np.sqrt(t_iter)  # decaying step
         step = lr * m_hat / (np.sqrt(v_hat) + eps)
         x -= step

         # update beta
         if beta < beta_max:
            beta *= inc
            if beta > beta_max:
                beta = beta_max

         # periodic exact evaluation
         if t_iter % eval_interval == 0:
            # compute exact objective
            cur_obj = _objective(a)  # since sum=1
            if cur_obj < best_score - 1e-14:
                best_score = cur_obj
                best_seq = a.copy()
                print(f"[{time.time():.1f}] iter {t_iter} new best {best_score:.6f}")
   return best_seq, best_score

"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.

Algorithm
---------

1. **Seed generation** – build a small, diverse pool of normalised
   distributions (uniform, arcsine, several analytic “U‑shaped’’ families,
   a mixed‑uniform variant, random Dirichlet and sparse samples).  Every
   candidate is projected onto the probability simplex (sum = 1) and its
   exact objective `2·n·max(conv(a,a))` is computed.  The best ~8 seeds are
   retained.

2. **Soft‑max mirror descent** – each retained seed is refined by a
   gradient‑based optimiser that works on the *log‑it* parameters
   `x_i = log a_i`.  The objective used for the descent is the smooth
   approximation

        sm(β) = (1/β)·log Σ_k exp(β·(a*a)_k)

   whose gradient can be written in closed form using FFT‑based
   convolutions.  The optimisation employs an Adam‑style update,
   a geometric increase of the temperature `β` (starting from 5 up to
   300) and a decaying learning‑rate.  The exact (non‑smooth) objective
   is evaluated every few iterations and the best sequence is kept.

3. **Local polishing** – after the main optimisation a cheap greedy
   pairwise mass‑transfer (and a short simulated‑annealing) are applied
   to squeeze a little more value out of the final candidate.

The whole routine respects the supplied time budget (`budget_s`,
default 1000 s) and returns a plain Python list of floats.  The returned
sequence already satisfies the non‑negativity and normalisation required
by ``evaluate_sequence`` (the latter will re‑normalise it anyway).
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())          # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _fft_convolve(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    need = u.shape[0] + v.shape[0] - 1
    size = 1 << ((need - 1).bit_length())
    U = rfft(u, n=size)
    V = rfft(v, n=size)
    conv = irfft(U * V, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (-p) + j ** (-p))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, U uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Soft‑max mirror descent optimiser ------------------------------------
def _softmax_optimize(seed: np.ndarray,
                      sub_deadline: float,
                      best_seq: np.ndarray,
                      best_score: float,
                      init_lr: float = 0.2,
                      beta_start: float = 5.0,
                      beta_max: float = 300.0,
                      beta_inc: float = 1.01,
                      eval_interval: int = 50) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation on the log‑it parameters using a soft‑max
    approximation of the max‑convolution.
    """
    n = seed.shape[0]
    eps = 1e-12
    # initialise log‑it variables (soft‑max guarantees Σa = 1)
    x = np.log(np.clip(seed, eps, None))

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    beta = beta_start
    t = 0

    while time.time() < sub_deadline:
        t += 1

        # ---- a = softmax(x) ------------------------------------------------
        max_x = x.max()
        ex = np.exp(x - max_x)
        a = ex / ex.sum()

        # ---- self‑convolution ------------------------------------------------
        conv = _fft_self_convolve(a)

        # ---- soft‑max weights for the smooth max ----------------------------
        rc = beta * conv
        rc_max = rc.max()
        exp_rc = np.exp(rc - rc_max)           # stable
        Z = exp_rc.sum()
        w = exp_rc / Z                         # shape = len(conv)

        # ---- gradient w.r.t a (2 * Σ w_k * a_{k‑j}) -------------------------
        a_rev = a[::-1]
        corr_full = _fft_convolve(w, a_rev)   # length = 3n‑2
        grad_a = 2.0 * corr_full[n - 1:n - 1 + n]

        # ---- chain rule → gradient w.r.t x (soft‑max Jacobian) -------------
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)

        # ---- Adam update ----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = init_lr / math.sqrt(t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-12)
        x -= step

        # ---- temperature schedule -------------------------------------------
        if beta < beta_max:
            beta = min(beta_max, beta * beta_inc)

        # ---- occasional exact evaluation ------------------------------------
        if t % eval_interval == 0:
            cur = _objective(a)               # exact objective (Σa = 1)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] it={t:,} β={beta:.1f} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple local refinements ---------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         max_trials: int | None = None) -> tuple[np.ndarray, float]:
    """
    Randomly move a small amount of mass from i→j. Keep move only if it improves.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.30
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


def _random_local_search(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         init_temp: float = 0.008,
                         decay: float = 0.9997) -> tuple[np.ndarray, float]:
    """
    Very cheap simulated‑annealing style mass transfer.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Main entry point – returns a list of non‑negative heights.
    The function respects the given time budget and prints progress.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # 1. Build a small pool of diverse seeds
    lengths = [2000, 4000, 6000, 8000, 12000]   # increasing sizes
    p_vals = np.concatenate(
        [
            np.logspace(-3, -1, 7),            # 0.001 … 0.1
            np.logspace(-1, 0, 7),             # 0.1 … 1
            np.logspace(0, 0.5, 5),            # 1 … 3.16
        ]
    )
    lam_vals = np.linspace(0.0, 0.8, 5)

    top_candidates: list[tuple[float, np.ndarray]] = []
    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] exploring length {n}")

        # deterministic families (no mixing)
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (len {n}) {sc:.6f}")

        # parametrised families + mixing with uniform
        for p in p_vals:
            if time.time() >= deadline:
                break
            families = (
                _distribution_beta_sym(n, p),
                _distribution_edge_weighted(n, p),
                _distribution_edge_inversepow(n, p),
                _distribution_sine_weighted(n, p),
            )
            for base in families:
                for lam in lam_vals:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best (len {n}) p={p:.3g} lam={lam:.2f} {sc:.6f}"
                        )

        # random Dirichlet seeds
        for _ in range(5):
            if time.time() >= deadline:
                break
            cand = _distribution_dirichlet(n, rng)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best Dirichlet (len {n}) {sc:.6f}")

        # sparse random seeds
        for _ in range(5):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best sparse (len {n}) {sc:.6f}")

        # keep only the strongest ~10 seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:10]

    # --------------------------------------------------------------
    # 2. Incorporate a previously discovered construction if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3. Fallback guard
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Soft‑max mirror descent on the most promising seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(
            f"[{time.time():.1f}] start optimisation on {len(seeds)} seeds, ≈{per_seed:.1f}s each"
        )
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            a_opt, sc_opt = _softmax_optimize(
                seed_seq.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after optimisation seed {idx} new global best {overall_best_score:.6f}"
                )

    # --------------------------------------------------------------
    # 5. Greedy pairwise refinement (fast)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=200_000,
        )

    # --------------------------------------------------------------
    # 6. Final cheap simulated‑annealing style local search
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # --------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return plain Python list – evaluate_sequence will do final clipping/normalisation.
    return overall_best_seq.tolist()
