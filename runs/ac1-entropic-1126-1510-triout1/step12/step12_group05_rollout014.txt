def add_gradient_descent(a, deadline, best_seq, best_score, lr0=0.2, max_iter=50000, eval_interval=200):

    n = len(a)
    t = 0

    while t < max_iter and time.time() < deadline:
        t += 1

        # compute conv and max index
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())

        # gradient for max entry
        g = _gradient_max_index(a, k)

        # subtract mean to keep sum zero?
        # Actually additive update a' = a - η * (g - mean(g)) ensures sum unchanged.
        # Let's compute delta = -η * (g - avg_g) where avg_g = np.mean(g).
        avg_g = np.mean(g)
        delta = - (lr0 / np.sqrt(t)) * (g - avg_g)

        # apply step, but ensure non-negative: a + delta >= 0; we clip negative values to 0 and renorm.
        a_new = a + delta
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full(n, 1.0 / n)
        else:
            a_new /= s

        # evaluate
        if t % eval_interval == 0:
            cur = _objective(a_new)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a_new.copy()
                print("[{: .1f}] add‑grad it={} obj={:.6f}".format(time.time(), t, best_score))

        a = a_new

    return best_seq, best_score

def _guided_swap_local(a, deadline, best_seq, best_score, max_iter=200000, max_fraction=0.2, K=30):
    n = len(a)
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        # compute grad for max conv
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())
        g = _gradient_max_index(a, k)
        # get indices sorted
        sorted_idx = np.argsort(g) # ascending
        low_idxs = sorted_idx[:K]
        high_idxs = sorted_idx[-K:]

        # choose i from high, j from low
        i = rng.choice(high_idxs)
        j = rng.choice(low_idxs)
        if a[i] < 1e-12:
            continue
        # move delta amount
        delta = a[i] * rng.random() * max_fraction
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # renorm not needed as total unchanged
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print("[{: .1f}] guided swap i={}→j={} δ={:.3g} new best {:.6f}".format(time.time(), i, j, delta, best_score))
    return best_seq, best_score

def _additive_grad_descent(a, deadline, best_seq, best_score,
                          lr0=0.3, max_iter=80000, eval_interval=500):
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())
        grad = _gradient_max_index(a, k)
        # compute zero-sum direction
        grad_mean = np.mean(grad)
        direction = -(grad - grad_mean)
        eta = lr0 / math.sqrt(t)
        a_new = a + eta * direction
        # enforce nonnegativity by clipping then renorm
        a_new = np.maximum(a_new, 0.0)
        total = a_new.sum()
        if total == 0.0:
            a_new = np.full(n, 1.0/n)
        else:
            a_new /= total
        if t % eval_interval == 0:
            cur = _objective(a_new)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a_new.copy()
                print(f"[{time.time():.1f}] add‑grad it={t} idx={k} new best {best_score:.6f}")
        a = a_new
    return best_seq, best_score

def _guided_swap_local(a, deadline, best_seq, best_score,
                       max_iter=200000, max_fraction=0.2, K=30):
    n = a.shape[0]
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())
        grad = _gradient_max_index(a, k)
        # get sorted indices
        sorted_idx = np.argsort(grad)  # ascending
        low = sorted_idx[:K]
        high = sorted_idx[-K:]

        i = rng.choice(high)
        if a[i] < 1e-12:
            continue
        j = rng.choice(low)
        if i == j:
            continue

        delta = a[i] * rng.random() * max_fraction
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] guided swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score

"""
Hybrid optimizer for the self‑convolution max‑ratio problem.

The algorithm builds on the previous state‑of‑the‑art but adds:
* An additive (Euclidean‑mirror) gradient descent phase.
* A guided mass‑transfer local search that always moves mass from
  indices with the largest sub‑gradient (i.e. most responsible for the
  maximal convolution entry) to indices with the smallest one.
* A dynamic time allocation to ensure the whole search fits inside the
  provided budget (default 1000 s).

The workflow is:
1. Generate a modest pool of analytic/semi‑random seeds for several
   lengths and keep the best few (the *elite*).
2. For each elite seed run three descent phases:
   – Soft‑max (exponentiated‑gradient) on a temperature‑controlled
     approximation of the max.
   – Hard‑max sub‑gradient descent on the true maximal entry.
   – Additive (projected) gradient descent on the same sub‑gradient.
3. Finish with a guided swap walk that uses the sub‑gradient to decide
   which source/sink pairs to move mass between.
4. Return the best sequence found as a plain list of floats.
"""

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a))/sum(a)^2.
    The vector a is assumed to be non‑negative and sum(a)>0.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient 2·a⋆w for a weight vector w (length 2n‑1, sum(w)=1).
    Implemented using FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.size - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad

def _gradient_max_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the single maximal convolution entry `conv[k]`.
    Returns a vector of size len(a) with non‑negative entries.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i = np.arange(i_min, i_max + 1, dtype=np.int64)
    j = k - i

    eq_mask = (i == j)
    if np.any(eq_mask):
        g[i[eq_mask]] += 2.0 * a[i[eq_mask]]

    if np.any(~eq_mask):
        ii = i[~eq_mask]
        jj = j[~eq_mask]
        g[ii] += a[jj]
        g[jj] += a[ii]

    return g

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * x / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Broad pool of analytic seeds for a given length."""
    cand = []
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_sine(n))
    # several block fractions
    for bf in (0.03, 0.07, 0.12):
        cand.append(_distribution_two_blocks(n, block_frac=bf))

    for p in np.linspace(0.05, 1.0, 12):
        cand.append(_distribution_beta_sym(n, p))

    for _ in range(8):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    for p in (0.5, 1.0, 2.0, 3.0):
        cand.append(_distribution_power_increasing(n, p))

    for lam in (0.001, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential_increasing(n, lam))

    for dens in (0.005, 0.01, 0.02):
        cand.append(_distribution_random_sparse(n, rng, density=dens))

    for a in (0.1, 0.3, 0.5, 1.0):
        cand.append(_distribution_dirichlet(n, rng, alpha=a))

    return cand

def _softmax_grad_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    max_iter: int = 80000,
    eval_interval: int = 400,
    alpha_start: float = 0.5,
    alpha_end: float = 80.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a soft‑max approximation of the
    maximal convolution entry.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        # linear schedule for temperature α
        alpha = alpha_start + (alpha_end - alpha_start) * t / max_iter
        # stable soft‑max
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()
        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        total = a.sum()
        a = a / total if total > 0 else np.full(n, 1.0 / n)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] soft‑max it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score

def _hardmax_grad_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.5,
    max_iter: int = 80000,
    eval_interval: int = 400,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent using the true maximal convolution entry.
    The update is exponentiated‑gradient with a decaying step size.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())
        grad = _gradient_max_index(a, k)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        total = a.sum()
        a = a / total if total > 0 else np.full(n, 1.0 / n)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hard‑max it={t} idx={k} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _additive_grad_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.3,
    max_iter: int = 60000,
    eval_interval: int = 400,
) -> tuple[np.ndarray, float]:
    """
    Projected Euclidean gradient descent on the true maximal entry.
    The gradient is made zero‑sum (by subtracting its mean) so that the
    simplex constraint is respected; after each step we clip negatives
    and renormalise.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())
        grad = _gradient_max_index(a, k)

        # make the direction zero‑sum
        grad_centered = grad - grad.mean()
        eta = lr0 / math.sqrt(t)
        a = a - eta * grad_centered

        a = np.maximum(a, 0.0)
        total = a.sum()
        a = a / total if total > 0 else np.full(n, 1.0 / n)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] additive it={t} idx={k} new best {best_score:.6f}")

    return best_seq, best_score

def _guided_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_iter: int = 200000,
    max_fraction: float = 0.20,
    K: int = 30,
) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer walk guided by the sub‑gradient of the maximal
    convolution entry.  In each step we move a random fraction (up to
    `max_fraction`) of mass from a high‑gradient index to a low‑gradient
    one.  Improvements are kept immediately.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _fft_self_convolve(a)
        k = int(conv.argmax())
        grad = _gradient_max_index(a, k)

        # indices with extreme gradients
        idx_sorted = np.argsort(grad)          # ascending
        low = idx_sorted[:K]                    # smallest gradients
        high = idx_sorted[-K:]                 # largest gradients

        i = rng.choice(high)
        if a[i] < 1e-12:
            continue
        j = rng.choice(low)
        if i == j:
            continue

        delta = a[i] * rng.random() * max_fraction
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] guided swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score
