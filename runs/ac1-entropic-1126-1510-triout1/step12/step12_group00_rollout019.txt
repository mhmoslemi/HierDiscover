def _cross_corr_fft(a, w):
    n = a.shape[0]
    need = n + (2*n - 1) - 1
    size = 1 << ((need - 1).bit_length())

    W = rfft(w, n=size)

def _smooth_descent(a, deadline, best_seq, best_score,
                    lr0=0.7, beta=0.9,
                    alpha0=5.0, alpha_inc_factor=1.02,
                    eval_interval=100,
                    patience=2000):
    n = a.shape[0]
    momentum = np.zeros_like(a)
    t = 0
    no_improve = 0
    alpha = alpha0
    rng = np.random.default_rng()
    while time.time() < deadline and t < 500_000:
        t += 1
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # compute softmax weights
        # subtract maxc to avoid overflow
        exps = np.exp(alpha * (conv - maxc))
        w = exps / exps.sum()
        # compute gradient via cross-correlation using FFT
        # compute grad = 2 * cross_corr(a, w)
        grad = _gradient_cross_corr(a, w) # function using FFT

        # momentum
        momentum = beta * momentum + (1 - beta) * grad

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * momentum)

        # project onto simplex sum=1
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t} new best {cur:.6f}")
            else:
                no_improve += 1

        # adapt alpha
        if t % eval_interval == 0:
            alpha *= alpha_inc_factor

        if no_improve >= patience:
            # jitter
            jitter = rng.exponential(scale=0.005, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0

    return best_seq, best_score

def _grad_cross_corr(a, w):
    # a: (n,)
    # w: (2n-1,)
    n = a.shape[0]
    rev_a = a[::-1]
    # compute full linear convolution rev_a * w using FFT
    need = n + (2*n - 1) - 1   # = 3n-2
    size = 1 << ((need - 1).bit_length())
    A_rev = rfft(rev_a, n=size)
    W = rfft(w, n=size)
    conv = irfft(A_rev * W, n=size)
    conv = conv[:need]  # linear convolution result
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad

    n = a.shape[0]

def _smooth_descent(a, deadline, best_seq, best_score,
                    lr0=0.7, beta=0.9,
                    alpha0=3.0, alpha_inc_factor=1.015,
                    eval_interval=200, patience=2000):
    n = a.shape[0]
    t = 0
    momentum = np.zeros_like(a)
    no_improve = 0
    alpha = alpha0
    rng = np.random.default_rng()
    while time.time() < deadline and t < 400_000:
        t += 1
        # compute conv
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # softmax weights
        exps = np.exp(alpha * (conv - maxc))
        w = exps / exps.sum()
        # gradient
        grad = _grad_cross_corr(a, w)
        # momentum update
        momentum = beta * momentum + (1 - beta) * grad
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * momentum)
        # projection onto simplex (sum = 1)

"""
Non‑negative step‑function search.

1. Generate a diverse set of seeds using analytic families (uniform,
   arcsine, symmetric beta, edge‑weighted, edge‑inverse‑power, sine‑weighted)
   together with random Dirichlet and sparse vectors.
   A fine grid of the shape parameter `p` (≈0.05 … 2) is explored on a
   modest size (2000) and the best few (≈10) are kept.

2. Each kept seed is optionally up‑sampled to larger lengths (3000, 5000,
   8000) by linear interpolation of its CDF, because a finer discretisation
   can lower the peak of the self‑convolution.

3. For every seed we run a *smoothed‑max* gradient descent.
   The max in the objective  `2·n·max(conv(a,a))/sum(a)**2` is approximated
   by a soft‑max with temperature parameter `α`.  The gradient of the
   smoothed objective is the cross‑correlation of the current vector with the
   soft‑max weights and is computed by an FFT (O(n log n)).
   An exponentiated‑gradient update with momentum and a decaying learning
   rate is applied, followed by a Euclidean projection onto the probability
   simplex (sum = 1).  The temperature `α` is slowly increased so that the
   optimisation first sees a smooth landscape and finally focuses on the
   true maximum.

4. After the continuous optimisation a cheap greedy pairwise mass‑transfer
   and a lightweight simulated‑annealing local search are performed to polish
   the solution.

The whole routine respects the supplied time budget (default 1000 s) and
returns a plain Python list of non‑negative heights.  The returned list can
be fed directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (real FFT, length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    sum_a = a.sum()
    if sum_a == 0.0:
        return np.inf
    return 2.0 * n * conv.max() / (sum_a ** 2)

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = n - np.arange(n, dtype=np.float64) - 0.5
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = n - np.arange(n, dtype=np.float64) - 0.5
    w = (i ** (-p) + j ** (-p))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
def _interp_resize(a: np.ndarray, m: int) -> np.ndarray:
    """
    Upsample / downsample a probability vector by linear interpolation of its CDF.
    Returns a new vector of length ``m`` that sums to one.
    """
    n = a.shape[0]
    cdf = np.concatenate([[0.0], np.cumsum(a)])
    xs = np.arange(n + 1) / n
    xs_new = np.arange(m + 1) / m
    cdf_new = np.interp(xs_new, xs, cdf)
    a_new = np.diff(cdf_new)
    total = a_new.sum()
    if total == 0.0:
        return np.full(m, 1.0 / m, dtype=np.float64)
    return a_new / total


# ----------------------------------------------------------------------
def _grad_cross_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·conv_k(a)  w.r.t. a.
    For each index i, grad_i = 2·∑_j a_j·w_{i+j}.
    Computed via FFT as the cross‑correlation of a with w.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    need = n + (2 * n - 1) - 1            # 3·n‑2
    size = 1 << ((need - 1).bit_length())  # next power of two
    A_rev = rfft(rev_a, n=size)
    W = rfft(w, n=size)
    conv = irfft(A_rev * W, n=size)
    conv = conv[:need]                   # linear convolution result
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad

def _smooth_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.7,
    beta: float = 0.9,
    alpha0: float = 3.0,
    alpha_inc_factor: float = 1.015,
    eval_interval: int = 200,
    patience: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a soft‑max approximation of the max‑convolution.
    Uses an exponentiated‑gradient step, momentum and projection onto the
    simplex.  ``alpha`` (temperature) is increased gradually.
    """
    n = a.shape[0]
    t = 0
    momentum = np.zeros_like(a)
    no_improve = 0
    alpha = alpha0
    rng = np.random.default_rng()
    while time.time() < deadline and t < 400_000:
        t += 1

        conv = _fft_self_convolve(a)
        maxc = conv.max()

        # soft‑max weights
        exps = np.exp(alpha * (conv - maxc))
        w = exps / exps.sum()

        # gradient of the smoothed objective
        grad = _grad_cross_corr(a, w)

        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # learning rate (decays like 1/√t)
        eta = lr0 / math.sqrt(t)

        # exponentiated‑gradient update (clip to avoid overflow)
        step = -eta * momentum
        step = np.clip(step, -50.0, 50.0)
        a = a * np.exp(step)

        # projection onto simplex (sum = 1)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # periodic evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] descent it={t:,} new best {best_score:.6f}")
            else:
                no_improve += 1
            # increase temperature
            alpha *= alpha_inc_factor

        # jitter to escape plateaus
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.005, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small amount of mass between two positions."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}->{j} Δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9998,
) -> tuple[np.ndarray, float]:
    """Very cheap simulated‑annealing style local search."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] Dirichlet seed score={sc:.6f}")

    # random sparse seeds
    for _ in range(12):
        if time.time() >= deadline:
            break
        seq = _distribution_random_sparse(n0, rng, density=0.004)
        sc = _objective(seq)
        candidates.append((sc, seq.copy()))
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] sparse seed score={sc:.6f}")

    # previous best (if available)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s == 0.0:
                prev = np.full_like(prev, 1.0 / prev.size)
            else:
                prev = prev / s
            sc = _objective(prev)
            candidates.append((sc, prev.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 2. Keep the strongest seeds, optionally up‑sample them
    candidates.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in candidates[:8]]

    # up‑sample to a few larger lengths (finer discretisation can improve)
    large_lengths = [3000, 5000, 8000]
    for seq in top_seeds:
        for n_big in large_lengths:
            if time.time() >= deadline:
                break
            up = _interp_resize(seq, n_big)
            sc = _objective(up)
            candidates.append((sc, up.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, up.copy()
                print(f"[{time.time():.1f}] up‑sampled to {n_big} score={sc:.6f}")

    candidates.sort(key=lambda x: x[0])
    refinement_seeds = [seq for _, seq in candidates[:10]]

    # --------------------------------------------------------------
    # 3. Smoothed‑max gradient descent on the selected seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and refinement_seeds:
        per_seed = remaining / len(refinement_seeds)
        print(f"[{time.time():.1f}] smoothed descent on {len(refinement_seeds)} seeds, ≈{per_seed:.1f}s each")
        for idx, seed_seq in enumerate(refinement_seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.7,
                beta=0.9,
                alpha0=3.0,
                alpha_inc_factor=1.015,
                eval_interval=200,
                patience=2000,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after descent seed {idx} new best {overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # --------------------------------------------------------------
    # 5. Final cheap simulated‑annealing style local search
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final random‑local search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9998,
        )

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
