```
def build_conv_matrix(a, indices):
    rows = []
    cols = []
    data = []
    n = len(a)
    for row_ind, i in enumerate(indices):
        start = max(0, i - (n - 1))
        end = min(i, n - 1)
        # j runs start..end
        # g index k = i - j
        # coefficient a[j]
        for j in range(start, end+1):
            k = i - j
            rows.append(row_ind)
            cols.append(k)
            data.append(a[j])
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(indices), n))
    return A
```

```
from scipy.optimize import linprog
import scipy.sparse as sp

def solve_lp_max_sum(a, M, active_idx, epsilon=1e-12):
    n = len(a)
    # Build sparse matrix A for active conv constraints:
    rows = []
    cols = []
    data = []
    for idx_row, i in enumerate(active_idx):
        start = max(0, i - (n - 1))
        end = min(i, n - 1)
        # iterate j from start to end, coefficient a[j] for g_{i - j}
        # g_idx = i - j
        for j in range(start, end+1):
            g_idx = i - j
            rows.append(idx_row)
            cols.append(g_idx)
            data.append(a[j])
    A = sp.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
    # Linear programming: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)  # minimizing negative sum => maximizing sum
    b_ub = np.full(len(active_idx), M + 1e-12)  # add small slack
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    g = res.x
    return g
```

```
def lp_refinement(a, deadline):
    n = len(a)
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    best_score = _objective(a)
    best_a = a.copy()
    while time.time() < deadline:
        conv = _fft_self_convolve(a)
        M = conv.max()
        # choose active indices: top K conv values where conv close to M
        top_K = min(20, len(conv))
        # get indices where conv within e.g., 0.999 * M
        # Let's get K largest values:
        idxs = np.argpartition(conv, -top_K)[-top_K:]
        # Solve LP:
        g = solve_lp_max_sum(a, M, idxs)
        if g is None:
            break
        # Normalize g
        sumg = g.sum()
        if sumg == 0:
            break
        g_norm = g / sumg
        # combine with a
        # try a few mixing rates
        improved = False
        for lam in [0.5, 0.25, 0.125]: # optionally add 1.0 to just adopt g_norm if better
            new_a = (1 - lam) * a + lam * g_norm
            new_a = np.maximum(new_a, 0)
            new_a /= new_a.sum()
            new_score = _objective(new_a)
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = new_a.copy()
                a = new_a
                improved = True
                print(f"[{time.time():.1f}] LP refine lam={lam:.3f} score={best_score:.6f}")
                break
        if not improved:
            # if no improvement, break out
            break
    return best_a, best_score
```

Algorithm:
1. Assemble a diverse pool of analytic families (uniform, arcsine,
   symmetric beta, edge‑weighted, sine‑weighted) plus random Dirichlet and
   sparse vectors for a few different lengths.
2. Keep the best few candidates (according to the exact objective
   2·n·max(conv) with sum normalized to 1).
3. Refine the current best by an *active‑set linear‑program*:
   – fix the current max‑convolution value M,
   – maximise the ℓ₁‑norm of a new vector g under the linear constraints
     (a * g)≤M at the currently most‑binding convolution indices,
   – replace a by a convex combination with the normalised g.
   Repeating this step quickly drives the max‑convolution down.
4. Return the best vector found as a plain Python list.
The code respects the supplied time budget (default 1000 s) and works
entirely with numpy/scipy; no file or network I/O is performed.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x≥0, Σx=s} (Euclidean projection)."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full (a * a) using a real FFT; length = 2·len(a)-1."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (all return a probability vector, i.e. sum≈1)

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    # symmetric Beta(p,p) on a regular grid
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑space to avoid overflow for extreme p
    logw = (p - 1) * (np.log(x) + np.log(1.0 - x))
    # shift for numerical stability
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family w_i ∝ ((i+½)(n‑i‑½))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ sin(π (i+½)/n)^p, vanishing at the endpoints."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture lam·U + (1‑lam)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

def _solve_lp_max_sum(a: np.ndarray, M: float, active_idx: np.ndarray) -> np.ndarray | None:
    """
    Solve  max ∑g_j   s.t.   a * g ≤ M  on the rows ``active_idx``.
    Returns the optimal g (may be dense) or None on failure.
    """
    n = a.shape[0]
    rows = []
    cols = []
    data = []
    for row, i in enumerate(active_idx):
        # convolution row i: (a * g)[i] = Σ_k a[i‑k]·g_k
        start = max(0, i - (n - 1))
        end = min(i, n - 1)
        # j = i‑k runs from start .. end
        # coefficient for g_k is a[j] where j = i‑k
        for j in range(start, end + 1):
            k = i - j
            rows.append(row)
            cols.append(k)
            data.append(a[j])
    A = sp.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
    # Linear program: maximise sum(g)  <=> minimise –sum(g)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(len(active_idx), M * (1.0 + 1e-12), dtype=np.float64)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs", options={"presolve": True})
    except Exception:
        return None
    if not res.success:
        return None
    g = res.x
    return g


def _lp_refine(a: np.ndarray, deadline: float, max_iters: int = 8) -> tuple[np.ndarray, float]:
    """
    Active‑set linear‑program refinement.
    Repeatedly:
        * compute current max‑convolution M,
        * keep the indices with the largest values (active set),
        * solve the LP that maximises ℓ₁‑norm under (a * g)≤M on the active set,
        * replace a by a convex combination with the normalised solution.
    Stops after ``max_iters`` or when the time budget expires.
    """
    n = a.shape[0]
    # ensure full support – tiny epsilon prevents unbounded LP
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    best_a = a.copy()
    best_score = _objective(best_a)

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _fft_self_convolve(best_a)
        M = conv.max()
        # pick a modest active set (the largest coefficients)
        top_k = min(20, conv.size)
        active = np.argpartition(conv, -top_k)[-top_k:]

        g = _solve_lp_max_sum(best_a, M, active)
        if g is None:
            break
        s = g.sum()
        if s <= 0.0:
            break
        g_norm = g / s
        # convex combination – we keep the mixing weight modest to stay safe
        lam = 0.5
        new_a = (1.0 - lam) * best_a + lam * g_norm
        new_a = np.maximum(new_a, 0.0)
        new_a /= new_a.sum()
        new_score = _objective(new_a)
        if new_score < best_score - 1e-14:
            best_a, best_score = new_a, new_score
            print(f"[{time.time():.1f}] LP‑refine iter={it} → score={best_score:.6f}")
        else:
            # no improvement – stop early
            break
    return best_a, best_score

```python
def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family w_i ∝ ((i+½)(n‑i‑½))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ sin(π (i+½)/n)^p, vanishing at the endpoints."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int,
    rng: np.random.Generator,
    density: float = 0.01,
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Linear‑program helper (active‑set)

def _solve_lp_max_sum(a: np.ndarray, M: float, active_idx: np.ndarray) -> np.ndarray | None:
    """
    Solve  max ∑ g_j   s.t.   a * g ≤ M  on the rows ``active_idx``.
    Returns the optimal g (dense) or None on failure.
    """
    n = a.shape[0]
    rows, cols, data = [], [], []
    for row, i in enumerate(active_idx):
        # (a * g)[i] = Σ_k a[i‑k]·g_k  (with out‑of‑range terms = 0)
        start = max(0, i - (n - 1))
        end = min(i, n - 1)
        for j in range(start, end + 1):
            k = i - j
            rows.append(row)
            cols.append(k)
            data.append(a[j])
    A = sp.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
    c = -np.ones(n, dtype=np.float64)                     # minimise –Σg_j
    b_ub = np.full(len(active_idx), M * (1.0 + 1e-12), dtype=np.float64)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method="highs", options={"presolve": True})
    except Exception:
        return None
    if not res.success:
        return None
    return res.x


def _lp_refine(a: np.ndarray, deadline: float, max_iters: int = 8) -> tuple[np.ndarray, float]:
    """
    Active‑set LP refinement.
    Repeatedly:
        * compute current max‑convolution M,
        * pick the indices with the largest values (active set),
        * solve the LP that maximises ℓ₁‑norm under (a * g)≤M on the active set,
        * replace a by a convex combination with the normalised solution.
    Stops after ``max_iters`` or when the time budget expires.
    """
    n = a.shape[0]
    a = np.maximum(a, 1e-12)          # guarantee full support → bounded LP
    a /= a.sum()
    best_a, best_score = a.copy(), _objective(a)

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _fft_self_convolve(best_a)
        M = conv.max()
        top_k = min(20, conv.size)
        active = np.argpartition(conv, -top_k)[-top_k:]

        g = _solve_lp_max_sum(best_a, M, active)
        if g is None or g.sum() <= 0.0:
            break
        g_norm = g / g.sum()
        lam = 0.5                      # modest mixing to stay within the feasible region
        new_a = (1.0 - lam) * best_a + lam * g_norm
        new_a = np.maximum(new_a, 0.0)
        new_a /= new_a.sum()
        new_score = _objective(new_a)
        if new_score < best_score - 1e-14:
            best_a, best_score = new_a, new_score
            print(f"[{time.time():.1f}] LP‑refine iter={it} → score={best_score:.6f}")
        else:
            break
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function respects the supplied time budget (default 1000 s) and
    returns a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Create a diverse pool of seeds
    lengths = [2000, 3000, 4000]               # long enough to improve the bound
    p_vals = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    lam_vals = [0.0, 0.33, 0.66, 1.0]

    top_candidates: list[tuple[float, np.ndarray]] = []
    best_seq: np.ndarray | None = None
    best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break

        # basic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()

        # parametrised families + mixing with uniform
        for p in p_vals:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            sine = _distribution_sine_weighted(n, p)
            for base in (beta, edge, sine):
                for lam in lam_vals:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < best_score:
                        best_score, best_seq = sc, cand.copy()

        # a few random Dirichlet vectors
        for _ in range(4):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()

        # sparse random vectors
        for _ in range(4):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()

    # keep the most promising seeds
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2. Optionally include a previously discovered construction
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < best_score:
                best_score, best_seq = sc, prev.copy()
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Refine the global best using the active‑set LP
    if best_seq is not None:
        best_seq, best_score = _lp_refine(best_seq.copy(), deadline, max_iters=12)

    # ------------------------------------------------------------------
    # 4. Second pass: try the same refinement on a few other top seeds
    for idx, (_, seq) in enumerate(top_candidates[:10]):
        if time.time() >= deadline:
            break
        if best_seq is not None and np.allclose(seq, best_seq, atol=1e-12):
            continue
        a_opt, sc_opt = _lp_refine(seq.copy(), deadline, max_iters=8)
        if sc_opt < best_score - 1e-12:
            best_seq, best_score = a_opt, sc_opt
            print(f"[{time.time():.1f}] new global best from seed {idx} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Fallback safety (should never be needed)
    if best_seq is None:
        n = 2000
        best_seq = _distribution_uniform(n)
        best_score = _objective(best_seq)

    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
```
