def _softmax_step(a0, deadline, best_seq, best_score):
   a = a0.copy()
   n = a.shape[0]
   t =0
   beta = 1.0
   while time.time() < deadline:
       t+=1
       conv = _fft_self_convolve(a)   # length 2n-1
       # compute softmax weights
       max_c = conv.max()
       # increase beta gradually
       beta = min(beta * 1.01, 2000)   # max large to approach max
       w = np.exp(beta * (conv - max_c))
       w_sum = w.sum()
       if w_sum == 0.0:
           w = np.full_like(w, 1.0/(len(w)))
       else:
           w /= w_sum

       # compute gradient
       # cross-correlation a ⋆ w
       size = 1 << ((n + (2*n-1) -1).bit_length())
       A_fft = np.fft.rfft(a, n=size)
       W_fft = np.fft.rfft(w, n=size)
       corr = np.fft.irfft(A_fft * np.conj(W_fft), n=size)
       start = (2*n-2)
       grad = 2.0 * corr[start:start + n]

       # EG update
       eta = 0.5 / np.sqrt(t)  # step size
       a = a * np.exp(-eta * grad)
       a /= a.sum()

       if t % 200 == 0:
           cur = _objective(a)
           if cur < best_score - 1e-14:
               best_score = cur
               best_seq = a.copy()
               print(f"[{time.time():.1f}] softmax iteration {t} new best {best_score:.6f}")

   return best_seq, best_score

"""
Propose a step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.
The algorithm consists of three stages:

1️⃣  Scan a modest set of analytic families (uniform, arcsine,
    Gaussian, beta‑type, edge‑weighted etc.) on a few lengths and keep
    the best few seeds.

2️⃣  Refine each of the top seeds with a *soft‑max mirror descent*.
    Instead of the crude “top‑k” sub‑gradient, the current iterate
    computes a softmax over **all** convolution entries (parameter β)
    and takes the exact gradient of that smooth surrogate.
    The update is an exponentiated‑gradient step followed by Euclidean
    projection onto the probability simplex (∑a = 1, a ≥ 0).

3️⃣  Finish with a fast greedy pairwise mass‑transfer hill‑climb
    (the same “swap” routine used previously) to polish any remaining
    improvements.

The whole routine respects the 1000 s time budget, prints progress,
and finally returns a plain Python list which `evaluate_sequence`
will evaluate.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = v.copy()
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1.
    The input must be non‑negative.  No clipping is performed – the
    evaluation is identical to ``evaluate_sequence`` after the
    scale‑invariant normalisation.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Edge‑heavy family w_i ∝ ((i+½)*(n‑i‑½))^{‑p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_gaussian(n: int, sigma_factor: float) -> np.ndarray:
    """Discrete Gaussian centred at the middle, σ = sigma_factor·n."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = max(1e-12, sigma_factor * n)
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
def _cross_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the cross‑correlation (a ⋆ w)[j] = Σ_t a[t]·w[t+j],
    for j = 0…len(a)-1.  Both inputs are real, a is length n,
    w is length 2·n‑1.  Returned array has length n.
    """
    n = a.shape[0]
    size = 1 << ((n + (2 * n - 1) - 1).bit_length())  # ≥ 3n‑2
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w, n=size)
    corr_full = np.fft.irfft(A * np.conj(W), n=size)
    start = 2 * n - 2                       # index of zero lag
    return corr_full[start:start + n]


# ----------------------------------------------------------------------
def _softmax_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    beta0: float = 1.0,
    beta_max: float = 200.0,
    beta_mul: float = 1.02,
    lr0: float = 0.8,
    eval_every: int = 150,
) -> tuple[np.ndarray, float]:
    """
    Mirror descent on the *soft‑max* surrogate of max(conv).
    We maintain a probability vector a (sum = 1, a ≥ 0) and repeatedly
    perform an exponentiated‑gradient step using the exact gradient of
    the surrogate, followed by simplex projection.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    beta = beta0

    # pre‑compute FFT size that will stay constant for this n
    size = 1 << ((3 * n - 2).bit_length())

    while time.time() < time_limit and t < 500_000:
        t += 1

        # --- self‑convolution (exact) ---------------------------------
        A = np.fft.rfft(a, n=size)
        conv_full = np.fft.irfft(A * A, n=size)          # length = size
        conv = conv_full[: 2 * n - 1]                    # needed part

        # --- soft‑max weights over the convolution entries ------------
        cmax = conv.max()
        w_exp = np.exp(beta * (conv - cmax))            # safe, max term = 1
        w = w_exp / w_exp.sum()                         # ∑ w = 1

        # --- gradient of the surrogate (exact) -----------------------
        # grad = 2 * (a ⋆ w)
        grad = 2.0 * _cross_correlation(a, w)

        # optional clipping to avoid overflow in the exponentials
        grad = np.clip(grad, -50.0, 50.0)

        # --- EG update + simplex projection ---------------------------
        eta = lr0 / math.sqrt(t)                        # decaying stepsize
        a = a * np.exp(-eta * grad)
        a_sum = a.sum()
        if a_sum == 0.0:
            # fallback – uniform distribution
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / a_sum

        # --- anneal β -------------------------------------------------
        if beta < beta_max:
            beta = min(beta * beta_mul, beta_max)

        # --- occasional exact evaluation -------------------------------
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] softmax it={t} β={beta:.2f} "
                    f"new global best {best_score:.6f}"
                )
    return best_seq, best_score

def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from one index to another.
    The move is accepted only if it strictly improves the exact
    objective.  This is a very cheap hill‑climbing polish.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} "
                f"new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.
    The routine:
      1. scans a few analytic families on several lengths,
      2. refines the most promising seeds with a soft‑max mirror descent,
      3. finishes with a greedy pairwise polishing.
    The full budget (≈1000 s) is respected; progress is printed.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0        # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_seeds: list[np.ndarray] = []

    # ------------------------------------------------------------------
    # 1️⃣  Analytic family scan (moderate breadth)
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000, 20000]

    sigma_factors = [0.05, 0.10, 0.15, 0.20, 0.30]   # for Gaussian
    p_grid = np.concatenate(
        [
            np.logspace(-2, -0.3, 9),      # 0.01 … 0.5
            np.linspace(0.5, 2.0, 7),      # 0.5 … 2
            np.linspace(2.0, 5.0, 4),      # 2 … 5
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 5)           # 0, .25, .5, .75, 1

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # ----- deterministic families ---------------------------------
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
        ):
            cand = base_fn(n)
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best {sc:.6f} (len={n})")
            top_seeds.append(cand)

        # ----- Gaussian family ----------------------------------------
        for sf in sigma_factors:
            cand = _distribution_gaussian(n, sf)
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] gauss len={n} σf={sf:.2f} score={sc:.6f}")
            top_seeds.append(cand)

        # ----- beta / edge‑weighted families with mixing ---------------
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta_seq = _distribution_beta_sym(n, p)
            edge_seq = _distribution_edge_weighted(n, p)

            for base in (beta_seq, edge_seq):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(
                            f"[{time.time():.1f}] len={n} p={p:.3g} lam={lam:.2f} "
                            f"score={sc:.6f}"
                        )
                    top_seeds.append(cand)

        # ----- random Dirichlet seeds ---------------------------------
        for _ in range(3):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.05, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} score={sc:.6f}")
            top_seeds.append(cand)

    # retain only a small number of distinct seeds
    # (duplicates are harmless – we will later deduplicate)
    top_seeds = sorted(top_seeds, key=_objective)[:10]

    # ------------------------------------------------------------------
    # 2️⃣  Include any previously discovered construction, if present.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s > 0:
                prev = prev / s
            else:
                prev = _distribution_uniform(prev.size)
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
            top_seeds.append(prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3️⃣  Fallback (should never happen)
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 5000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(
            f"[{time.time():.1f}] fallback uniform len={n} score={overall_best_score:.6f}"
        )
        top_seeds = [overall_best_seq]

    # ------------------------------------------------------------------
    # 4️⃣  Soft‑max mirror‑descent refinement on the top seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_seeds:
        per_seed = remaining / max(1, len(top_seeds))
        print(f"[{time.time():.1f}] start softmax descent on {len(top_seeds)} seeds")
        for idx, seed_seq in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            sg_deadline = time.time() + per_seed
            a_opt, sc_opt = _softmax_descent(
                seed_seq,
                sg_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                beta0=1.0,
                beta_max=300.0,
                beta_mul=1.02,
                lr0=0.9,
                eval_every=200,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] after descent seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5️⃣  Final greedy pairwise polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (2·n·max(conv)): {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
