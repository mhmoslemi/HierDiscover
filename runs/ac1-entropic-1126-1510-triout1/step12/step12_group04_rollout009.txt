"""
Hybrid optimizer for the step‑function autocorrelation bound.

The algorithm proceeds in three phases:

1. **Broad seeding** – a rich collection of analytic families
   (uniform, arcsine with offset, power‑inverse‑sqrt, two‑block,
   sparse Dirichlet) is instantiated for several lengths.
   Each candidate is normalised (sum = 1) and its exact objective
   `2·n·max(conv)/sum(a)²` is evaluated by FFT.  The best few seeds
   across all lengths are retained.

2. **Active‑set exponentiated‑gradient descent** – starting from each
   elite seed we repeatedly:
   * compute the self‑convolution,
   * form an *active set* of indices whose convolution value is within
     a tiny tolerance of the current maximum,
   * build the sub‑gradient `g_i = 2·∑_{k∈active} a_{k‑i}` (efficiently
     via a few shifted copies of `a`),
   * update `a ← a·exp(-η·g)` with a step‑size `η = η₀/√t`,
   * project back onto the probability simplex.
   The best iterate seen during the run is kept.

3. **Greedy polishing** – cheap random mass‑transfer moves are
   accepted only if they improve the exact objective.  This removes
   remaining tiny spikes.

The routine respects the total time budget (default 1000 s) and
returns the best sequence found as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)²."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine_offset(n: int, c: float) -> np.ndarray:
    """Arcsine‑type density with a tunable offset `c>0`."""
    i = np.arange(n, dtype=np.float64)
    left = i + c
    right = (n - 1 - i) + c
    vals = 1.0 / np.sqrt(left * right)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _distribution_power_inv_sqrt(n: int, p: float) -> np.ndarray:
    """Generalised arcsine: w_i ∝ ((i+0.5)*(n‑i‑0.5))^{‑p/2}."""
    i = np.arange(n, dtype=np.float64)
    left = i + 0.5
    right = (n - 1 - i) + 0.5
    vals = (left * right) ** (-p / 2.0)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _active_set_descent(a: np.ndarray,
                        deadline: float,
                        best_seq: np.ndarray,
                        best_score: float,
                        lr0: float = 0.5,
                        rel_tol: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent using the *active set* of maximal
    convolution indices.  The gradient for a given active index `k` is
    `g_i = 2·a_{k‑i}` (zero outside the overlap).  The step size
    decays as η = lr0 / √t.  Returns the best vector seen.
    """
    n = a.shape[0]
    t = 0
    max_exp = 700.0  # safety for np.exp

    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] iter={t} new best {best_score:.6f}")

        # ---- active set -------------------------------------------------
        threshold = max_conv * (1.0 - rel_tol)
        active = np.where(conv >= threshold)[0]
        # limit size of active set to keep work cheap
        if active.size > 20:
            active = active[:20]

        # ---- gradient from active set ----------------------------------
        grad = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1)
            idx_j = k - idx_i
            grad[idx_i] += 2.0 * a[idx_j]

        # ---- exponentiated‑gradient update -------------------------------
        eta = lr0 / math.sqrt(t)
        # clip gradient to avoid overflow in the exponential
        grad_clipped = np.clip(grad, -max_exp / eta, max_exp / eta)
        a = a * np.exp(-eta * grad_clipped)

        # ---- project back onto probability simplex -----------------------
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Greedy random mass‑transfer polishing."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue
        delta = a[i] * rng.random() * 0.30  # move up to 30 % of source
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur_score = _objective(new_a)
        if cur_score < best_score - 1e-13:
            a = new_a
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}→{j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score

def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Main entry point: searches for a step‑function (list of non‑negative
    heights) that minimises the evaluation function.  The routine obeys
    the given time budget and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    elite_seeds: list[tuple[float, np.ndarray]] = []   # (score, vector)

    # -----------------------------------------------------------------
    # 1. Analytic seeding across several lengths
    lengths = [2000, 3000, 4000, 5000, 8000, 12000]
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] --- generating seeds for length {n} ---")
        # uniform
        cand = _distribution_uniform(n)
        cand = cand / cand.sum()
        sc = _objective(cand)
        elite_seeds.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = cand.copy()
            print(f"[{time.time():.1f}] !!! new global best len={n} score={sc:.6f}")

        # arcsine with many offsets
        for c in np.linspace(0.05, 3.0, 15):
            if time.time() >= deadline:
                break
            cand = _distribution_arcsine_offset(n, c)
            cand = cand / cand.sum()
            sc = _objective(cand)
            elite_seeds.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] !!! new global best len={n} c={c:.2f} score={sc:.6f}")

        # power‑inverse‑sqrt family
        for p in np.linspace(0.8, 1.2, 9):
            if time.time() >= deadline:
                break
            cand = _distribution_power_inv_sqrt(n, p)
            cand = cand / cand.sum()
            sc = _objective(cand)
            elite_seeds.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] !!! new global best len={n} p={p:.2f} score={sc:.6f}")

        # two‑block (few different block fractions)
        for bf in (0.05, 0.10, 0.15):
            if time.time() >= deadline:
                break
            cand = _distribution_two_blocks(n, block_frac=bf)
            cand = cand / cand.sum()
            sc = _objective(cand)
            elite_seeds.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] !!! new global best len={n} bf={bf:.2f} score={sc:.6f}")

        # sparse random draws
        for _ in range(2):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.01)
            cand = cand / cand.sum()
            sc = _objective(cand)
            elite_seeds.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] !!! new global best (sparse) len={n} score={sc:.6f}")

        # Dirichlet draws
        for _ in range(2):
            if time.time() >= deadline:
                break
            alpha = 0.3 + 0.7 * rng.random()
            cand = _distribution_dirichlet(n, rng, alpha=alpha)
            cand = cand / cand.sum()
            sc = _objective(cand)
            elite_seeds.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] !!! new global best (dirichlet) len={n} α={alpha:.3f} score={sc:.6f}")

    # -----------------------------------------------------------------
    # 2. Incorporate any previously known solution (if present)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            prev = prev / prev.sum()
            sc = _objective(prev)
            elite_seeds.append((sc, prev.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best improved: {sc:.6f}")
    except Exception:
        pass

    # Keep only a modest elite set
    elite_seeds.sort(key=lambda x: x[0])
    elite_seeds = elite_seeds[:20]

    # -----------------------------------------------------------------
    # 3. Local optimisation (active‑set descent) on elite seeds
    remaining = deadline - time.time()
    if remaining > 20.0 and elite_seeds:
        # allocate roughly equal time to each seed
        per_seed = max(1.0, remaining / len(elite_seeds))
        for idx, (sc, seed_seq) in enumerate(elite_seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            print(f"[{time.time():.1f}] descent on seed {idx+1}/{len(elite_seeds)} (len={len(seed_seq)})")
            seq_out, score_out = _active_set_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.5,
                rel_tol=1e-7,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after descent new best {overall_best_score:.6f}")

    # -----------------------------------------------------------------
    # 4. Greedy polishing
    if overall_best_seq is not None and time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # -----------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
