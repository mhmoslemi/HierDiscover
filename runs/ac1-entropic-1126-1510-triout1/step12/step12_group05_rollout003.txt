"""
Hybrid multi‑start + parametric search for the self‑convolution max‑ratio
problem.

The algorithm proceeds in three stages:

1.  **Broad sampling** – for several lengths we generate many analytic
    families (uniform, arcsine, Beta, power‑law, edge‑power,
    arcsine^(q), two‑blocks, Dirichlet, sparse, mixtures).  All candidates
    are normalised to sum = 1 and evaluated by the exact objective
    2·n·max(conv(a,a)).  The best ~20 candidates are kept in an
    “elite” list.

2.  **Focused refinement** – each elite seed is improved by two
    exponentiated‑gradient phases:
       * a soft‑max approximation of the max‑convolution (temperature
         α rising from a small start to ≈80);
       * a hard‑max sub‑gradient descent.
    After each exponential‑gradient update we (a) project onto the
    simplex, (b) enforce symmetry (a[i]=a[n‑1‑i]) and (c) renormalise.
    The total number of iterations per seed is adapted to the
    remaining time budget.

3.  **Final polishing** – a simulated‑annealing style random‑mass‑transfer
    walk accepts improving moves and, with a linearly decaying temperature,
    also occasionally uphill moves.  This finishes the optimisation
    within the allotted time.

The returned sequence is a plain list of non‑negative floats (already
normalised; scaling does not affect the objective because the evaluation
function is homogeneous).
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    # Take the linear part (discard zero‑padding tail)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)), assuming sum(a)=1.
    The sequence a is non‑negative.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    # sum is 1 by construction; guard against numerical drift
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the soft‑max weighted sum Σ w_k * conv_k.
    Returns 2·a⨂w (cross‑correlation) as a vector of length n.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.size - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    tmp = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * tmp[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _gradient_max_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the single maximal convolution entry conv[k].
    Returns a non‑negative vector of length n.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i = np.arange(i_min, i_max + 1, dtype=np.int64)
    j = k - i

    eq = (i == j)
    if np.any(eq):
        g[i[eq]] += 2.0 * a[i[eq]]

    neq = ~eq
    if np.any(neq):
        ii = i[neq]
        jj = j[neq]
        g[ii] += a[jj]
        g[jj] += a[ii]

    return g

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_arcsine_power(n: int, q: float) -> np.ndarray:
    """arcsine density raised to power q >=0."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    base = 1.0 / np.sqrt(x * (1.0 - x))
    vals = base ** q
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use log‑space to avoid under/overflow
    logw = (p - 1) * (np.log(x) + np.log(1.0 - x))
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    logw = (alpha - 1) * np.log(np.clip(x, eps, 1.0)) + \
           (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0))
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * x / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_edge_power(n: int, p: float) -> np.ndarray:
    """Weight ~ (distance to nearest edge)^(-p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    rev = n - i
    vals = (i ** -p) + (rev ** -p)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals

def _softmax_phase(a: np.ndarray, deadline: float,
                   best_seq: np.ndarray, best_score: float,
                   lr0: float = 0.9,
                   max_iter: int = 30000,
                   eval_interval: int = 250,
                   alpha_start: float = 0.4,
                   alpha_end: float = 80.0) -> tuple[np.ndarray, float]:
    """Exponentiated gradient on a soft‑max approximation of the max‑conv."""
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        # linear temperature schedule
        alpha = alpha_start + (alpha_end - alpha_start) * (t / max_iter)
        # stable soft‑max
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()
        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # enforce non‑negativity, symmetry and simplex
        a = np.maximum(a, 0.0)
        a = (a + a[::-1]) * 0.5          # symmetric
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] soft‑max it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score

def _hardmax_phase(a: np.ndarray, deadline: float,
                   best_seq: np.ndarray, best_score: float,
                   lr0: float = 0.5,
                   max_iter: int = 40000,
                   eval_interval: int = 250) -> tuple[np.ndarray, float]:
    """Exponentiated gradient on the true max‑conv (hard‑max)."""
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())
        grad = _gradient_max_index(a, k)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        a = np.maximum(a, 0.0)
        a = (a + a[::-1]) * 0.5          # keep symmetry
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hard‑max it={t} idx={k} new best {best_score:.6f}")

    return best_seq, best_score

def _annealing_polish(a: np.ndarray, deadline: float,
                      best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing‑style random mass‑transfer walk.
    Accepts improving moves, and also uphill moves with probability
    exp(-Δ/temperature).  Temperature decays linearly to zero.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    start = time.time()
    while time.time() < deadline:
        # linear temperature: high at start, zero at deadline
        frac = (deadline - time.time()) / (deadline - start)
        temp = 0.01 * frac          # small temperature scale; can be tuned

        # pick source with non‑zero mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        j = rng.integers(n)
        if j == i:
            continue

        # move a random fraction of source mass
        delta = a[i] * rng.random() * 0.40   # up to 40 %
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        # maintain symmetry (mirrored indices)
        new_a = (new_a + new_a[::-1]) * 0.5
        new_a = np.maximum(new_a, 0.0)
        new_a /= new_a.sum()

        cur = _objective(new_a)
        diff = cur - best_score
        if diff < -1e-14 or rng.random() < math.exp(-diff / max(temp, 1e-12)):
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal move i={i}->j={j} Δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                max_iter=60000,
                eval_interval=300,
                alpha_start=0.3,
                alpha_end=80.0,
            )
            if score_a < overall_best_score:
                overall_best_score = score_a
                overall_best_seq = seq_a.copy()
                print(f"[{time.time():.1f}] after soft‑max new best {overall_best_score:.6f}")

            # Phase B – hard‑max sub‑gradient descent
            if time.time() >= deadline:
                break
            sub_deadline_b = time.time() + per_seed
            seq_b, score_b = _hardmax_phase(
                seq_a.copy(),
                sub_deadline_b,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.5,
                max_iter=60000,
                eval_interval=300,
            )
            if score_b < overall_best_score:
                overall_best_score = score_b
                overall_best_seq = seq_b.copy()
                print(f"[{time.time():.1f}] after hard‑max new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final polishing with simulated‑annealing random mass transfers.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] Starting final annealing polish.")
        overall_best_seq, overall_best_score = _annealing_polish(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    # 6.  Return the best sequence found.
    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # Convert to plain Python list of floats as required by the checker.
    return overall_best_seq.tolist()
