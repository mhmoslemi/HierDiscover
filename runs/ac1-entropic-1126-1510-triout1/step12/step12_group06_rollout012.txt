def max_convolution(a):
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    conv = conv[:2*n-1]
    return conv.max()
```

def generate_seeds(n, rng):
    seeds = []
    # uniform
    seeds.append(np.full(n, 1.0 / n))
    # arcsine
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i+0.5)*(n-i-0.5))
    vals /= vals.sum()
    seeds.append(vals)
    # symmetric beta exponents
    for p in np.geomspace(0.01, 5.0, num=7):
        x = (np.arange(n)+0.5)/n
        w = np.exp((p-1)*np.log(x) + (p-1)*np.log(1-x))
        w /= w.sum()
        seeds.append(w)
    # random asymmetric beta
    for _ in range(5):
        a = 10**rng.uniform(-3,1)
        b = 10**rng.uniform(-3,1)
        x = (np.arange(n)+0.5)/n
        w = np.exp((a-1)*np.log(np.clip(x,1e-12,1)) + (b-1)*np.log(np.clip(1-x,1e-12,1)))
        w /= w.sum()
        seeds.append(w)
    # exponential decreasing
    for lam in (0.001,0.003,0.005,0.01):
        i = np.arange(n)
        w = np.exp(-lam*i)   # decreasing
        w = np.maximum(w, 1e-12)
        w /= w.sum()
        seeds.append(w)
    # dirichlet draws
    for a in (0.1,0.3,0.7,1.0,2.0):
        w = rng.gamma(a, size=n)
        w /= w.sum()
        seeds.append(w)
    # random sparse
    for density in (0.01,0.03,0.05):
        k = max(2, int(n*density))
        idx = rng.choice(n, size=k, replace=False)
        w = np.zeros(n)
        w[idx] = rng.random(k)
        w /= w.sum()
        seeds.append(w)

#!/usr/bin/env python
"""
Hybrid LP‑+‑gradient optimiser for the step‑function max‑convolution problem.

Strategy
--------
* Choose a modest length (4000–8000) and normalise every candidate so that
  sum(a)=sqrt(2·n).  With that normalisation the evaluation metric reduces to
  ``M = max(conv(a,a))`` – the value we try to minimise.
* For a fixed vector `f` and its current worst‑case convolution value `M`,
  the feasible set for a new vector `g` is linear:
      (f * g)[k] = Σ_i f[i]·g[k‑i] ≤ M   for all shifts k.
  The linear programme   maximise Σ_j g[j]   subject to the above constraints
  yields a vector `g₀` that carries as much total mass as possible without
  increasing the worst convolution.
* Scale `g₀` to the required sum, then mix it with `f`:
      f ← (1‑t)·f + t·g   (t∈(0,1) and the sum stays constant).
  The mix strictly reduces the max‑convolution or leaves it unchanged.
* Repeat the LP‑mix step a few times (the “alternating‑LP” loop).  Afterwards
  perform a short Adam‑style smooth‑max descent and a greedy pairwise‑swap
  polishing phase (identical to the reference implementation) to squeeze out
  the last bits.
* Several random seeds (uniform, arcsine, Beta, Dirichlet, sparse, exponential)
  are tried for each length; the best result overall is returned.

The whole routine respects the 1000 s budget, prints intermediate scores
and finally returns the best coefficient list found.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt
from typing import List, Tuple

# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _max_convolution(a: np.ndarray) -> float:
    """Return max(conv(a,a)).  ``a`` is assumed non‑negative."""
    return float(_fft_self_convolve(a).max())


# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Construct the (2n‑1)×n matrix `T_f` such that T_f @ g = conv(f,g).
    Row k (0≤k<2n‑1) contains the coefficients f[i] for all i with 0≤i<n
    and 0≤k‑i<n, i.e. the diagonal with offset i is filled with f[i].
    """
    n = f.shape[0]
    # each diagonal i (offset i) consists of n copies of f[i]
    data = np.vstack([np.full(n, f_i) for f_i in f])
    offsets = np.arange(n)               # 0,1,…,n‑1
    A = sp.dia_matrix((data, offsets), shape=(2 * n - 1, n))
    return A.tocsr()


def _solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray:
    """Linear programme:
           maximise Σ g_j
           s.t.   T_f @ g ≤ M·1
                  g ≥ 0
       Returns the optimal `g` (unscaled)."""
    n = f.shape[0]
    A = _build_conv_matrix(f)                     # (2n‑1)×n
    b = np.full(2 * n - 1, M, dtype=np.float64)  # RHS
    c = -np.ones(n, dtype=np.float64)             # minimise −Σg  ≡ maximise Σg
    bounds = [(0.0, None) for _ in range(n)]

    # HiGHS (the default interior‑point of scipy) works well with sparse data.
    res = opt.linprog(
        c, A_ub=A, b_ub=b, bounds=bounds,
        method='highs',
        options={'presolve': True, 'time_limit': 30}
    )
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


# ----------------------------------------------------------------------
def _generate_seeds(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    """Return a diverse collection of probability vectors of length n."""
    seeds: List[np.ndarray] = []

    # uniform
    seeds.append(np.full(n, 1.0 / n))

    # arcsine‑law
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    seeds.append(vals)

    # symmetric Beta(p,p) – log‑spaced p
    for p in np.geomspace(0.01, 5.0, num=8):
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
        w /= w.sum()
        seeds.append(w)

    # random asymmetric Beta draws
    for _ in range(6):
        a = 10 ** rng.uniform(-3, 1)      # 0.001 … 10
        b = 10 ** rng.uniform(-3, 1)
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        eps = 1e-15
        w = np.exp((a - 1) * np.log(np.clip(x, eps, 1.0)) +
                   (b - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
        w /= w.sum()
        seeds.append(w)

    # decreasing exponential families
    for lam in (0.001, 0.003, 0.005, 0.010):
        i = np.arange(n, dtype=np.float64)
        w = np.exp(-lam * i)
        w /= w.sum()
        seeds.append(w)

    # Dirichlet draws with various concentrations
    for alpha in (0.1, 0.3, 0.7, 1.0, 2.0):
        w = rng.gamma(alpha, size=n)
        w /= w.sum()
        seeds.append(w)

    # random sparse patterns
    for density in (0.01, 0.03, 0.05):
        k = max(2, int(n * density))
        idx = rng.choice(n, size=k, replace=False)
        w = np.zeros(n, dtype=np.float64)
        w[idx] = rng.random(k)
        w /= w.sum()
        seeds.append(w)

    return seeds


# ----------------------------------------------------------------------
def _lp_alternating_loop(
    f: np.ndarray,
    target_sum: float,
    max_iter: int,
    deadline: float,
    best_M: float,
    verbose_prefix: str = ""
) -> Tuple[np.ndarray, float]:
    """
    Repeated LP‑mix steps:
        * solve LP for the current worst‑case value M,
        * scale the solution to the required sum,
        * mix it into the current vector (mixing weight t decays).
    Returns the best vector seen in this block and its max‑convolution.
    """
    n = f.shape[0]
    cur_f = f.copy()
    cur_M = best_M
    best_f = cur_f.copy()
    best_local_M = cur_M

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # 1) solve the LP with the *current* worst value
        try:
            g0 = _solve_lp_for_g(cur_f, cur_M)
        except Exception as e:
            print(f"{verbose_prefix}[{time.time():.1f}] LP failed at it={it}: {e}")
            break

        # 2) scale to the required sum (keeps the mixing sum constant)
        g_sum = g0.sum()
        if g_sum == 0.0:
            break
        scale = target_sum / g_sum
        g = g0 * scale

        # 3) mixing – start with fairly large t and then decay
        t = 0.4 * (0.85 ** it)          # decays to ≈0.05 after ~30 iters
        t = max(t, 0.05)
        cur_f = (1.0 - t) * cur_f + t * g

        # 4) evaluate new max‑convolution
        cur_M = _max_convolution(cur_f)

        if cur_M < best_local_M - 1e-12:
            best_local_M = cur_M
            best_f = cur_f.copy()
            print(f"{verbose_prefix}[{time.time():.1f}] LP it={it} new best M={best_local_M:.6f}")

        # stop early if improvement stalls
        if cur_M >= best_M - 1e-10:
            # allow a few more iterations in case of minor fluctuations
            pass

    return best_f, best_local_M


# ----------------------------------------------------------------------
def _adam_fine_tune(
    a: np.ndarray,
    target_sum: float,
    deadline: float,
    best_M: float,
    verbose_prefix: str = ""
) -> Tuple[np.ndarray, float]:
    """
    A short Adam‑style smooth‑max descent (same as the reference code)
    that respects the simplex (sum = target_sum).  The temperature α is
    increased to a high value (here 300) to approximate the true max.
    """
    n = a.shape[0]
    # bring to probability simplex (sum = 1) for the smooth‑max code
    a = a / a.sum()
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)

    # hyper‑parameters tuned for speed
    eta0 = 0.7
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha = 0.3
    alpha_mul = 1.5
    alpha_max = 300.0

    t = 0
    best_local_a = a.copy()
    best_local_M = best_M

    while time.time() < deadline and t < 30000:
        t += 1
        conv = _fft_self_convolve(a)

        # smooth‑max weights
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient of smooth surrogate
        # using cross‑correlation via FFT (same trick as in the reference)
        a_rev = a[::-1]
        size = 1 << ((n + w.size - 1) - 1).bit_length()
        W = np.fft.rfft(w, n=size)
        A_rev = np.fft.rfft(a_rev, n=size)
        conv_grad = np.fft.irfft(W * A_rev, n=size)
        start = n - 1
        grad = 2.0 * conv_grad[start:start + n]

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # multiplicative update, then simplex projection
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(-step)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum

        # temperature schedule
        if t % 250 == 0 and alpha < alpha_max:
            alpha = min(alpha * alpha_mul, alpha_max)

        # periodic exact check
        if t % 1000 == 0:
            cur_M = _max_convolution(a)          # a is on the simplex → M = objective / (2n)
            if cur_M < best_local_M - 1e-12:
                best_local_M = cur_M
                best_local_a = a.copy()
                print(f"{verbose_prefix}[{time.time():.1f}] Adam it={t} new best M={best_local_M:.6f}")

    # rescale back to the original sum convention
    best_local_a = best_local_a * target_sum
    return best_local_a, best_local_M


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_M: float,
    verbose_prefix: str = ""
) -> Tuple[np.ndarray, float]:
    """Cheap greedy mass‑transfer moves that accept any strict improvement."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_M = best_M
    while time.time() < deadline:
        # pick a source index with non‑negligible mass
        src_candidates = np.where(cur_a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)

        # pick a different destination
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # propose a random transfer amount (up to 30 % of source)
        delta = cur_a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        new_M = _max_convolution(new_a)
        if new_M < cur_M - 1e-12:
            cur_a = new_a
            cur_M = new_M
            print(f"{verbose_prefix}[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best M={cur_M:.6f}")

    return cur_a, cur_M

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects ``budget_s`` seconds and returns the best coefficient
    list found (as a plain Python list of floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a handful of lengths – larger lengths give more freedom.
    #    We use three lengths and allocate time roughly proportional to n.
    lengths = [4000, 6000, 8000]
    overall_best_vec: np.ndarray | None = None
    overall_best_M = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] === Starting length {n} ===")
        target_sum = math.sqrt(2.0 * n)               # normalisation: sum = sqrt(2n)

        # ----------------------------------------------------------------
        # 2. Generate a moderate number of diverse seeds.
        seeds = _generate_seeds(n, rng)
        # add a few pure Dirichlet draws for extra randomness
        for _ in range(4):
            alpha = 10 ** rng.uniform(-2, 1)          # 0.01 … 10
            v = rng.gamma(alpha, size=n)
            v /= v.sum()
            seeds.append(v)

        # ----------------------------------------------------------------
        # 3. Scan the seed pool (cheap exact evaluation only).
        candidate_vecs: List[Tuple[float, np.ndarray]] = []
        for idx, s in enumerate(seeds):
            if time.time() >= deadline:
                break
            s = np.maximum(s, 0.0)
            s = s / s.sum() * target_sum                # enforce our normalisation
            M = _max_convolution(s)
            candidate_vecs.append((M, s))
            if M < overall_best_M:
                overall_best_M = M
                overall_best_vec = s.copy()
                print(f"[{time.time():.1f}] ** new global best (seed) M={M:.6f}")

        # keep the most promising few for the expensive LP stage
        candidate_vecs.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in candidate_vecs[:5]]

        # ----------------------------------------------------------------
        # 4. Alternating‑LP refinement on each top seed.
        for seed_id, f0 in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            # allocate a slice of remaining time to this seed
            per_seed_deadline = time.time() + (deadline - time.time()) / max(1, len(top_seeds) - seed_id)
            # start with the current max‑convolution as the bound
            M0 = _max_convolution(f0)
            # run the LP‑mix loop (max 20 iterations per seed)
            f_opt, M_opt = _lp_alternating_loop(
                f0, target_sum, max_iter=20,
                deadline=per_seed_deadline,
                best_M=M0,
                verbose_prefix=f"[len={n} seed={seed_id}] "
            )
            if M_opt < overall_best_M:
                overall_best_M = M_opt
                overall_best_vec = f_opt.copy()
                print(f"[{time.time():.1f}] *** after LP seed {seed_id} new global best M={overall_best_M:.6f}")

        # ----------------------------------------------------------------
        # 5. Short Adam‑style fine‑tuning of the current global best for this length.
        if overall_best_vec is not None and time.time() < deadline:
            remaining = deadline - time.time()
            # give a few seconds to fine‑tune
            fine_deadline = time.time() + min(30.0, remaining)
            f_fine, M_fine = _adam_fine_tune(
                overall_best_vec,
                target_sum,
                fine_deadline,
                overall_best_M,
                verbose_prefix=f"[len={n} Adam] "
            )
            if M_fine < overall_best_M:
                overall_best_M = M_fine
                overall_best_vec = f_fine.copy()
                print(f"[{time.time():.1f}] !!! Adam fine‑tune improved M={overall_best_M:.6f}")

        # ----------------------------------------------------------------
        # 6. Final greedy pairwise polishing (very cheap).
        if overall_best_vec is not None and time.time() < deadline:
            overall_best_vec, overall_best_M = _pairwise_swap_local(
                overall_best_vec,
                deadline,
                overall_best_M,
                verbose_prefix=f"[len={n} swap] "
            )
            print(f"[{time.time():.1f}] After swaps length {n} best M={overall_best_M:.6f}")

    # ----------------------------------------------------------------------
    # 7. Return the best sequence found (scaled back to ordinary sum, not sqrt(2n)).
    if overall_best_vec is None:
        # fall back to uniform – should never happen
        n = 4000
        overall_best_vec = np.full(n, 1.0 / n)
        overall_best_M = _max_convolution(overall_best_vec)

    # convert to ordinary scale (the evaluation function rescales internally)
    final_vec = overall_best_vec.tolist()
    print(f"Final best objective ≈ {overall_best_M:.6f} (length={len(final_vec)})")
    # expose the sequence for potential warm‑starts in future runs
    global height_sequence_1
    height_sequence_1 = final_vec
    return final_vec
