```
            else:
                # if no improvement, reduce step size
                t_cur = max(0.02, t_cur * 0.7)
                if t_cur < 0.02:
                    break
    else:
        print(f"[{time.time():.1f}] scipy linprog not available; skipping LP refinement")

    # final gradient descent for fine tuning
    if time.time() < deadline - 5:
        print(f"[{time.time():.1f}] final gradient descent")
        best_seq, best_score = _gradient_descent(
            best_seq,
            deadline,
            best_seq,
            best_score,
            eta0=0.3,
            eval_every=100,
        )

    # final annealing
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing")
        best_seq, best_score = _pairwise_annealing(
            best_seq,
            deadline,
            best_seq,
            best_score,
            np.random.default_rng(seed+999),
            max_trials=None,
        )

    print(f"Best objective {best_score:.6f} length {len(best_seq)}")
    # convert to list of floats
    return best_seq.tolist()
```

The new `propose_candidate` builds on the previous three‑stage approach but adds a **quadratic‑linear relaxation** step that directly reduces the worst autocorrelation value.  

1. **Initial pool** – a handful of analytic families and random Dirichlet draws are generated and run for a few hundred sub‑gradient iterations. The best few are kept.  
2. **LP‑refinement** – given a current candidate `f` we compute its worst autocorrelation `M = max(conv(f,f))`.  
   * The constraint `conv(f,g) ≤ M` is linear in `g`.  
   * We solve a linear program *maximising* the total mass `∑g_j` under those linear constraints (`g ≥ 0`).  
   * The solution `g` is normalised to a probability simplex and mixed with `f` (`f ← (1‑t)f + t g`).  
   * The step size `t` starts relatively large and is shrunk whenever an iteration does not improve the objective.  
   This is exactly the “alternating LP” idea described in the statement, but implemented explicitly with `scipy.optimize.linprog`.  
3. **Fine‑grained polishing** – a short exponentiated‑gradient descent, followed by a long pairwise‑mass‑transfer annealing, finishes the optimisation.  

The whole pipeline respects the 1000 s budget, prints progress, and always returns a non‑negative list ready for `evaluate_sequence`.  

```python
"""
Propose a non‑negative step function that (almost) minimises
``evaluate_sequence``.  The algorithm consists of:

1.  A seed‑generation phase (analytic families + random Dirichlet).
2.  A *linear‑programming refinement* phase that solves
    max Σ g_j  s.t.  f∗g ≤ M ,  g ≥ 0  (M = max(f∗f)).  The LP‑solution g
    is convex‑combined with the current f.
3.  A short exponentiated‑gradient descent on the exact objective.
4.  A long‑run pairwise‑mass‑transfer simulated‑annealing polish.

All work fits into the 1000 s time budget, prints intermediate scores,
and returns a plain ``list[float]`` suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, linear convolution)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective used by ``evaluate_sequence``.
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return the exact value of ``evaluate_sequence`` (scale‑invariant)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, sum=1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Sub‑gradient utilities (exponentiated‑gradient descent)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """Gradient of the currently maximal convolution entry."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))
    grad = np.zeros_like(a)
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


def _gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    eval_every: int = 50,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the exact objective."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] GD it={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style pairwise mass transfers
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Random mass moves with a simple linear cooling schedule."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation while preserving the simplex)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the total budget (≈1000 s) and returns the best
    sequence found.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0               # small safety margin

    # ------------------------------------------------------------------
    # 0.  Load previous construction if it exists.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim != 1:
            raise ValueError
        prev = prev / prev.sum()
        best_seq = prev.copy()
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] start from previous best {best_score:.6f}")
    except Exception:
        # fallback: uniform of moderate length
        n0 = 3000
        best_seq = _distribution_uniform(n0)
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform start score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1.  Generate a modest pool of diverse seeds and run a short GD.
    # ------------------------------------------------------------------
    seed_lengths = [1800, 2600, 3400]                # keep them manageable
    for n in seed_lengths:
        if time.time() >= deadline:
            break

        families = [
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_hann(n),
            _distribution_generalized_normal(n, q=1.2),
        ]

        # parameter sweeps
        for p in np.logspace(-1, 0.4, 7):
            families.append(_distribution_beta_sym(n, p))
            families.append(_distribution_edge_weighted(n, p))

        # random Dirichlet draws
        for _ in range(4):
            alpha = rng.uniform(0.01, 2.0)
            families.append(rng.dirichlet(np.full(n, alpha)))

        for base in families:
            if time.time() >= deadline:
                break
            # small mixing with uniform to increase diversity
            for lam in (0.0, 0.3, 0.7):
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                if sc < best_score:
                    best_seq = cand.copy()
                    best_score = sc
                    print(f"[{time.time():.1f}] new seed best len={n} score={sc:.6f}")

            # short exponentiated‑gradient refinement (few thousand iters)
            if time.time() + 3.0 > deadline:
                continue
            cand_opt, cand_sc = _gradient_descent(
                cand.copy(),
                time.time() + 3.0,
                best_seq,
                best_score,
                eta0=0.6,
                eval_every=100,
            )
            if cand_sc < best_score:
                best_seq = cand_opt.copy()
                best_score = cand_sc
                print(f"[{time.time():.1f}] after quick GD new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  LP‑based refinement (core of the new contribution).
    # ------------------------------------------------------------------
    try:
        from scipy.optimize import linprog

        lp_available = True
    except Exception:
        lp_available = False

    if lp_available:
        # make sure the length is not too large for the LP (memory ≈ n²)
        n = best_seq.shape[0]
        if n > 3500:
            # down‑sample a bit to keep the LP tractable
            best_seq = _upsample(best_seq, 3200)
            n = best_seq.shape[0]
        best_seq = best_seq / best_seq.sum()               # ensure simplex

        print(f"[{time.time():.1f}] starting LP refinement (n={n})")
        t_mix = 0.25                                        # initial mixing weight

        # give the LP stage a decent chunk of the remaining budget
        while time.time() < deadline - 8.0:
            conv = _fft_self_convolve(best_seq)
            M = conv.max()

            # ----- build the (2n‑1)×n constraint matrix A  -----
            #   conv_f_g[k] = Σ_i f_i * g_{k‑i}
            #   for fixed i the coefficient f_i appears in rows i,…,i+n‑1
            A = np.zeros((2 * n - 1, n), dtype=np.float64)
            for i in range(n):
                A[i : i + n, i] = best_seq[i]

            # ----- solve max Σ g_j  s.t. A·g ≤ M , g ≥ 0 -----
            c = -np.ones(n, dtype=np.float64)               # maximise Σ g_j
            b = np.full(2 * n - 1, M, dtype=np.float64)

            res = linprog(
                c,
                A_ub=A,
                b_ub=b,
                bounds=[(0, None)] * n,
                method="highs",
                options={"presolve": True},
            )
            if not res.success:
                print(f"[{time.time():.1f}] LP failed: {res.message}")
                break

            g = res.x
            if g.sum() == 0.0:
                break
            g = g / g.sum()                                 # bring back to simplex

            # ----- convex combination and evaluate -----
            new_seq = (1.0 - t_mix) * best_seq + t_mix * g
            new_seq = _project_onto_simplex(new_seq, s=1.0)
            new_score = _objective(new_seq)

            if new_score < best_score - 1e-12:
                best_seq = new_seq.copy()
                best_score = new_score
                print(f"[{time.time():.1f}] LP step improved to {best_score:.6f}")
                # gradually tighten the step size for finer adjustments
                t_mix = max(0.02, t_mix * 0.92)
            else:
                # no improvement → shrink the mixing factor
                t_mix = max(0.02, t_mix * 0.74)
                if t_mix <= 0.02:
                    break
    else:
        print(f"[{time.time():.1f}] scipy.linprog not available – skipping LP phase")

    # ------------------------------------------------------------------
    # 3.  Fine‑grained exponentiated‑gradient descent (short horizon)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final gradient descent")
        best_seq, best_score = _gradient_descent(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            eta0=0.35,
            eval_every=80,
        )

    # ------------------------------------------------------------------
    # 4.  Long‑run annealing / pairwise polishing (as in the previous code)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing polish")
        best_seq, best_score = _pairwise_annealing(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            rng=rng,
            max_trials=None,
        )

    # ------------------------------------------------------------------
    # 5.  Report and return
    # ------------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)/sum²) = {best_score:.6f} "
        f"(length={len(best_seq)})"
    )
    # ``evaluate_sequence`` caps values at 1000 and clips negatives – our vector
    # already respects those constraints.
    return best_seq.tolist()
```
