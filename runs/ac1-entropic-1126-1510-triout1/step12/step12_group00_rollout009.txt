def _smooth_adam_descent(a, deadline, best_seq, best_score):
    n = a.shape[0]
    a = a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    t=0
    while time.time()<deadline:
        t+=1
        # conv a*a
        conv = _fft_self_convolve(a)
        # choose alpha based on iteration
        if t < 2000:
            alpha = 5.0
        elif t < 8000:
            alpha = 20.0
        elif t < 20000:
            alpha = 80.0
        else:
            alpha = 200.0
        # compute smooth max and weights
        max_conv = conv.max()
        # stability shift
        # we will compute w = exp(alpha*(conv - max_conv))/sum
        exps = np.exp(alpha * (conv - max_conv))
        sum_exps = exps.sum()
        w = exps / sum_exps
        # compute gradient of smooth max
        a_rev = a[::-1]
        # We need full conv of w and a_rev
        conv_w_rev = _fft_full_convolve(w, a_rev)  # shape 3n-2

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = 0.1 / np.sqrt(t)  # decaying
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # enforce positivity
        a = np.maximum(a, 0.0)
        # symmetrize
        a = (a + a[::-1]) / 2.0
        # renormalize onto simplex
        a = _project_onto_simplex(a, s=1.0)
        # periodic evaluation
        if t % 200 == 0:
            cur = _objective(a)
            if cur < best_score - 1e-12:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] itr={t} new best {best_score:.6f}")
    return best_seq, best_score

def _fft_full_convolve(x, y):
    need = len(x) + len(y) - 1
    size = 1 << ((need-1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:need]

"""
Propose a non‑negative step function minimizing ``evaluate_sequence``.

Approach
--------
1. Build a diverse pool of symmetric seed vectors using analytic families
   (uniform, arcsine, edge‑weighted, edge‑exponential) and random symmetric
   Dirichlet samples.
2. For each seed run a smooth‑max Adam descent:
   * Replace the exact max of the self‑convolution by a soft‑max
     (log‑sum‑exp) with a gradually increasing temperature (α).
   * Compute the gradient of the smooth objective via FFT‑based
     convolution (full convolution of the soft‑max weights with the reversed
     sequence).  Adam updates are applied, the iterate is projected onto the
     probability simplex and symmetrised after each step.
   * Periodically evaluate the true objective (2·n·max(conv)) and keep the
     best ever seen.
3. Finish with the cheap pairwise mass‑transfer refinement and a short
   simulated‑annealing local search (the same routines used in the reference
   implementation).
4. Return the best sequence (list of floats).  All numbers are non‑negative
   and sum to one, which is scale‑invariant for the evaluation function.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _fft_full_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors via FFT."""
    need = len(x) + len(y) - 1
    size = 1 << ((need - 1).bit_length())
    X = rfft(x, n=size)
    Y = rfft(y, n=size)
    conv = irfft(X * Y, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) under the convention sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families (all symmetric by construction)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w /= w.sum()
    return w


def _distribution_edge_exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    rev_i = i[::-1]
    w = np.exp(-rate * i) + np.exp(-rate * rev_i)
    w /= w.sum()
    return w


def _distribution_random_symmetric_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a symmetric Dirichlet sample (mirror the first half)."""
    half = (n + 1) // 2
    a_half = rng.dirichlet(np.ones(half))
    if n % 2 == 0:
        a_full = np.concatenate([a_half, a_half[::-1]])
    else:
        a_full = np.concatenate([a_half, a_half[-2::-1]])  # omit duplicated middle entry
    a_full = a_full[:n]  # safety
    a_full /= a_full.sum()
    return a_full


# ----------------------------------------------------------------------
# Smooth‑max Adam descent on the exact shape (sum fixed to 1)
def _smooth_adam_descent(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """
    Minimise the true max‑convolution via a smooth‑max surrogate.
    Uses Adam with a schedule for the soft‑max temperature α.
    """
    n = a.shape[0]
    a = a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    no_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1

        # exact self‑convolution
        conv = _fft_self_convolve(a)

        # temperature schedule for smooth max
        if t < 2000:
            alpha = 5.0
        elif t < 8000:
            alpha = 20.0
        elif t < 20000:
            alpha = 80.0
        else:
            alpha = 200.0

        # soft‑max weights (stable w.r.t. max)
        max_conv = conv.max()
        w_exp = np.exp(alpha * (conv - max_conv))
        sum_exp = w_exp.sum()
        w = w_exp / sum_exp  # shape (2n‑1,)

        # gradient of smooth‑max (2·(w * a_rev) slice)
        a_rev = a[::-1]
        conv_w_rev = _fft_full_convolve(w, a_rev)      # length 3n‑2
        grad = 2.0 * conv_w_rev[n - 1:n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.1 / math.sqrt(t)     # decaying learning rate
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step                # gradient descent step

        # enforce feasibility
        a = np.maximum(a, 0.0)
        a = (a + a[::-1]) / 2.0    # keep symmetry
        a = _project_onto_simplex(a, s=1.0)

        # periodic exact evaluation
        if t % 200 == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} new global best {best_score:.6f}")
            else:
                no_improve += 1

        # jitter to escape plateaus
        if no_improve >= 4000:
            jitter = rng.exponential(scale=0.005, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = (a + a[::-1]) / 2.0
            a = _project_onto_simplex(a, s=1.0)
            no_improve = 0

    return best_seq, best_score

# Deterministic pairwise swap refinement (same as reference)
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         max_trials: int | None = None) -> tuple[np.ndarray, float]:
    """Randomly move a small amount of mass; keep it if the true objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score

# Cheap simulated‑annealing style local search (same as reference)
def _random_local_search(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         init_temp: float = 0.008,
                         decay: float = 0.9997) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score
