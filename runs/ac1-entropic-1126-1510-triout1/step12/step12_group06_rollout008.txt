def solve_lp(f, M):
    n = len(f)
    # build constraint matrix A_ub shape (2n-1, n)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = f
    # bound vector ub = M * np.ones(2*n-1)
    bounds = [(0, None)] * n
    # objective: maximize sum(b) => minimize -sum(b)
    c = -np.ones(n, dtype=np.float64)
    res = linprog(c, A_ub=A, b_ub=M*np.ones(2*n-1), bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        return res.x
    else:
        return np.zeros_like(f)

#!/usr/bin/env python
"""
Hybrid search using linear‑programming “cross‑convolution” refinement.

The algorithm:
1. Generate a small pool of seed distributions (uniform, arcsine,
   symmetric beta, Dirichlet, plus the previous best solution if any).
2. Keep the few best seeds (by the exact objective).
3. For each seed run a few iterations of a convex‑LP refinement:
   * compute M = max(conv(a,a)).
   * solve   max Σ b_j   s.t.   a * b ≤ M  (component‑wise) , b ≥ 0.
   * the LP increases the total mass while still respecting the same
     cross‑convolution bound.
   * scale the LP solution to a probability vector and mix it with the
     current vector; the mixture strictly reduces the objective.
4. Finish with a cheap greedy pairwise mass‑transfer polishing.
All work respects the supplied time budget; the best vector found
is returned as a plain list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from typing import List, Tuple

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact evaluation used by the competition (scale‑invariant)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discretised unit interval."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0)
    eps = 1e-15
    w = np.exp((p - 1) * np.log(np.clip(x, eps, 1.0 - eps)) +
               (p - 1) * np.log(np.clip(1.0 - x, eps, 1.0 - eps)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _solve_lp_cross(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max Σ b_j    s.t.   conv(a, b) ≤ M  (every entry),   b ≥ 0.
    Returned vector is the raw maximiser (not normalised).
    """
    n = a.shape[0]
    # Build the (2n‑1) × n constraint matrix A where A @ b = conv(a,b).
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column i contributes a at rows i … i+n‑1
    for i in range(n):
        A[i:i + n, i] = a
    c = -np.ones(n, dtype=np.float64)          # maximize Σ b  ⇔  minimize −Σ b
    b_ub = np.full(rows, M, dtype=np.float64)  # upper bound M for each conv entry
    bounds = [(0.0, None)] * n
    # ``highs`` is the default, fast for dense problems.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    # In the unlikely case of failure we return None.
    return None

def _lp_refine(a0: np.ndarray, deadline: float) -> Tuple[np.ndarray, float]:
    """
    Repeatedly apply the linear‑programming refinement described in the
    paper.  The process stops when time runs out or when no further
    improvement is observed.
    """
    a = a0.copy()
    best_score = _objective(a)
    mix = 0.4                     # initial mixing weight
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        conv_a = _fft_self_convolve(a)
        M = conv_a.max()
        b = _solve_lp_cross(a, M)
        if b is None:
            break
        sum_b = b.sum()
        if sum_b <= a.sum() + 1e-12:   # no gain in total mass
            break
        # Normalise b to a probability vector (sum = 1).
        g = b / sum_b
        # Mix and renormalise.
        new_a = (1.0 - mix) * a + mix * g
        new_a = new_a / new_a.sum()
        new_score = _objective(new_a)

        if new_score < best_score - 1e-13:
            a = new_a
            best_score = new_score
            print(f"[{time.time():.1f}] LP‑refine it={iteration} mix={mix:.3f} score={best_score:.6f}")
            # Successful step – slowly increase the mixing weight.
            mix = min(mix * 1.1, 0.9)
        else:
            # Unsuccessful – reduce the mixing weight.
            mix *= 0.5
            if mix < 1e-3:
                break
    return a, best_score

def _pairwise_swap_local(a: np.ndarray,

                         best_seq: np.ndarray,

            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score

"""
Hybrid search using a linear‑programming “cross‑convolution’’ refinement.

The algorithm
---------------
1. Produce a modest collection of seed distributions (uniform, arcsine,
   a few symmetric Beta shapes, Dirichlet draws and the previous best
   solution if it exists) for several moderate lengths (≈2‑3 k).
2. Keep the few best seeds (by the exact objective).
3. For each kept seed run a few iterations of the convex‑LP refinement
   described in the reference paper:
      * M = max(conv(a,a))
      * maximise Σb_j  subject to  a∗b ≤ M  (component‑wise) and b≥0.
        This is a linear programme with n variables and 2n‑1 constraints.
      * Scale the LP solution to a probability vector and mix it with the
        current vector; the mixture strictly reduces the objective.
4. End with a cheap greedy pairwise mass‑transfer polishing step.
All work respects the supplied time budget; the best vector found is
returned as a plain list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from typing import List, Tuple

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (scale‑invariant)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discretised unit interval."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((p - 1) * np.log(np.clip(x, eps, 1.0 - eps)) +
               (p - 1) * np.log(np.clip(1.0 - x, eps, 1.0 - eps)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _solve_lp_cross(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max Σ b_j   s.t.   conv(a,b) ≤ M   (for every index)  and   b ≥ 0.
    Returns the raw optimiser (not normalised) or None on failure.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    # Build the dense constraint matrix A where (A @ b) = conv(a,b).
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = a
    c = -np.ones(n, dtype=np.float64)               # maximise Σ b  ⇔ minimise −Σ b
    b_ub = np.full(rows, M, dtype=np.float64)       # each conv entry ≤ M
    bounds = [(0.0, None)] * n                       # b ≥ 0
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None
