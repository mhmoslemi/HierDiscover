"""
Low‑autocorrelation step‑function search.

The algorithm consists of three phases:

1.  Analytic seeding – a diverse collection of closed‑form families
    (uniform, arcsine, power‑inverse‑sqrt, two‑block, triangular,
    cosine‑shaped, logistic, sparse Dirichlet draws) is instantiated for
    several lengths.  All candidates are normalised to sum=1 and the
    exact objective 2·n·max(conv(a,a)) is evaluated.  The best few seeds
    are retained.

2.  Multi‑peak smooth sub‑gradient descent – we minimise the true
    non‑smooth objective by repeatedly moving mass away from the *most
    offending* convolution peaks.  At each iteration the convolution is
    computed (FFT), the top‑K peak indices are selected, and a softmax
    weighting (log‑sum‑exp smoothing) is used to aggregate their
    sub‑gradients.  An exponentiated‑gradient (mirror‑descent with KL)
    update is performed and the iterate is projected back onto the
    probability simplex.  The learning‑rate follows ηₜ = η₀/√t.  This
    smooths the sub‑gradient and usually pushes the solution past local
    minima that trap a single‑peak approach.

3.  Greedy polishing – cheap random mass‑transfer moves are accepted
    iff they improve the true objective.  Occasionally a worse move is
    accepted with a tiny Metropolis probability (simulated‑annealing
    style) to escape shallow traps.

The best sequence found (as a plain list of floats) is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear full convolution a*a via real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)²)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine_offset(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = i + c
    right = (n - 1 - i) + c
    vals = 1.0 / np.sqrt(left * right)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _distribution_power_inv_sqrt(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = i + 0.5
    right = (n - 1 - i) + 0.5
    vals = (left * right) ** (-p / 2.0)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = (i + 1) * (n - i)
    vals = np.where(vals > 0, vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_power(n: int, p: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = (1.0 - np.cos(np.pi * i / (n - 1))) ** p
    vals = np.where(np.isfinite(vals), vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_logistic(n: int, scale: float = 0.2) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) / (n - 1)
    vals = 1.0 / (1.0 + np.exp(-(i - 0.5) / scale))
    vals = np.where(np.isfinite(vals), vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a modest pool of diverse analytic seeds for length n."""
    cand = []

    cand.append(_distribution_uniform(n))

    # arcsine with several offsets
    for c in (0.2, 0.5, 1.0, 2.0):
        cand.append(_distribution_arcsine_offset(n, c))

    # power‑inverse‑sqrt family
    for p in (0.8, 1.0, 1.2):
        cand.append(_distribution_power_inv_sqrt(n, p))

    cand.append(_distribution_two_blocks(n, block_frac=0.05))
    cand.append(_distribution_two_blocks(n, block_frac=0.10))

    cand.append(_distribution_triangular(n))
    cand.append(_distribution_cosine_power(n, p=1.0))
    cand.append(_distribution_cosine_power(n, p=2.0))

    cand.append(_distribution_logistic(n, scale=0.2))
    cand.append(_distribution_logistic(n, scale=0.1))

    cand.append(_distribution_random_sparse(n, rng, density=0.005))
    cand.append(_distribution_random_sparse(n, rng, density=0.02))

    # a few Dirichlet draws with random α
    for _ in range(3):
        alpha = 0.1 + 0.9 * rng.random()
        cand.append(_distribution_dirichlet(n, rng, alpha=alpha))

    return cand

def _smooth_subgrad_descent(a: np.ndarray,
                            deadline: float,
                            best_seq: np.ndarray,
                            best_score: float,
                            lr0: float = 0.8,
                            top_k: int = 5,
                            gamma: float = 30.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent using a smooth (soft‑max) aggregation of the
    top‑K convolution peaks.  Returns the best sequence seen.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        M = conv.max()

        cur_score = 2.0 * n * M / (a.sum() ** 2)
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] iter={t} new best {best_score:.6f}")

        # ---- identify the top‑K peaks ----
        k = min(top_k, conv.size)
        idxs = np.argpartition(-conv, k - 1)[:k]  # unsorted top‑k
        # weight by a soft‑max around the maximum
        M_val = conv[idxs].max()
        exp_vals = np.exp(gamma * (conv[idxs] - M_val))
        w_sum = exp_vals.sum()
        if w_sum == 0.0:
            weights = np.full_like(exp_vals, 1.0 / k)
        else:
            weights = exp_vals / w_sum

        # ---- accumulate sub‑gradient ----
        grad = np.zeros_like(a)
        for w, kk in zip(weights, idxs):
            i_start = max(0, kk - (n - 1))
            i_end = min(n - 1, kk)
            idx_i = np.arange(i_start, i_end + 1, dtype=np.int64)
            idx_j = kk - idx_i
            grad[idx_i] += w * 2.0 * a[idx_j]

        # optional clipping to keep the exponent numerically sane
        grad = np.clip(grad, 0.0, 10.0)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)      # mirror‑descent step
        a = _project_onto_simplex(a)      # back to simplex

        # occasional small random mixing with uniform to avoid dead‑ends
        if t % 500 == 0:
            if np.random.rand() < 0.2:
                uniform = _distribution_uniform(n)
                a = 0.99 * a + 0.01 * uniform
                a = _project_onto_simplex(a)

    return best_seq, best_score

def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """
    Greedy (optionally simulated‑annealing) polishing using random mass
    transfers.  Accepts a move if it improves the objective, otherwise
    accepts with probability exp(-(Δ)/T) where T slowly cools.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    start_time = time.time()
    while time.time() < deadline:
        # linear cooling schedule
        elapsed = time.time() - start_time
        total = deadline - start_time
        T = 1e-2 * (1.0 - elapsed / total) + 1e-5   # between 1e-2 and 1e-5

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        j = rng.integers(n)
        if j == i:
            continue
        delta = a[i] * rng.random() * 0.30   # move up to 30 % of source mass
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # maintain feasibility (simplex) – projection not needed because sum unchanged
        cur_score = _objective(new_a)
        if cur_score < best_score - 1e-13:
            a = new_a
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}→{j} δ={delta:.3g} new best {best_score:.6f}")
        else:
            # accept worse move with small probability to escape shallow traps
            if rng.random() < math.exp(-(cur_score - best_score) / T):
                a = new_a
    return best_seq, best_score

def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """Entry point – searches a low‑autocorrelation step function."""
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5    # tiny safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_seeds: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Analytic seeding across several lengths
    lengths = [2000, 4000, 8000, 12000]  # growing lengths give better asymptotics
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] --- generating seeds for length {n} ---")
        for cand in _initial_candidates(n, rng):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] !!! new global best (len={n}) score={sc:.6f}")
            top_seeds.append((sc, cand.copy()))

        # keep a modest global elite set
        top_seeds.sort(key=lambda x: x[0])
        top_seeds = top_seeds[:20]

    # ------------------------------------------------------------------
    # 2.  Incorporate any previously known solution
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best improved: {sc:.6f}")
            top_seeds.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Main optimisation – smooth multi‑peak sub‑gradient descent
    remaining = deadline - time.time()
    if remaining > 30.0 and top_seeds:
        top_seeds.sort(key=lambda x: x[0])
        seeds_to_use = [seq for _, seq in top_seeds[:7]]
        per_seed = max(1.0, remaining / len(seeds_to_use))
        for idx, seed_seq in enumerate(seeds_to_use, 1):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            print(f"[{time.time():.1f}] subgradient run {idx}/{len(seeds_to_use)} (len={len(seed_seq)})")
            seq_out, score_out = _smooth_subgrad_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.8,
                top_k=5,
                gamma=30.0,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after subgrad new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Final greedy/annealing polishing
    if overall_best_seq is not None and time.time() < deadline:
        print(f"[{time.time():.1f}] polishing phase (len={len(overall_best_seq)})")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
