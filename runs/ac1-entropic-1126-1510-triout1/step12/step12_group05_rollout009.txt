"""
Search for a non‑negative step‑function that minimises
    2·n·max(self‑convolution) / (sum²)

The algorithm

1. builds a modest pool of analytic/draw seeds (length ≈ 4000);
2. evaluates them and keeps the few best (the “elite”);
3. refines the best elite seed with two Adam‑optimised phases:
   – a soft‑max approximation of the max‑convolution,
   – the exact hard‑max sub‑gradient;
4. finishes with a cheap random mass‑transfer walk (pairwise swaps).

All operations stay on the simplex by representing a as a soft‑max of free
log‑weights, thus automatically respecting the non‑negativity and unit‑sum
constraints.  The routine respects the given time budget and returns the
best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)² (a is assumed non‑negative)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient 2·a⋆w for a weight vector w (len 2·n‑1, sum(w)=1).
    Implemented using FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.size - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _gradient_max_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the single maximal convolution entry conv[k].
    Returns a non‑negative vector of the same size as a.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i = np.arange(i_min, i_max + 1, dtype=np.int64)
    j = k - i

    eq_mask = (i == j)
    if np.any(eq_mask):
        g[i[eq_mask]] += 2.0 * a[i[eq_mask]]

    if np.any(~eq_mask):
        ii = i[~eq_mask]
        jj = j[~eq_mask]
        g[ii] += a[jj]
        g[jj] += a[ii]

    return g

# Analytic families used as seeds ------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * x / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_candidates_small(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """A compact yet diverse pool of analytic and random seeds."""
    cand = []

    # deterministic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_sine(n))
    cand.append(_distribution_two_blocks(n, block_frac=0.07))

    # symmetric beta on a modest grid
    for p in (0.1, 0.3, 0.5, 0.8, 1.0):
        cand.append(_distribution_beta_sym(n, p))

    # random asymmetric beta draws
    for _ in range(5):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # power‑law families
    for p in (0.5, 1.0, 2.0):
        cand.append(_distribution_power_increasing(n, p))

    # exponential families
    for lam in (0.001, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential_increasing(n, lam))

    # sparse random draws
    for dens in (0.005, 0.01):
        cand.append(_distribution_random_sparse(n, rng, density=dens))

    # Dirichlet draws
    for a in (0.1, 0.3, 0.5, 1.0):
        cand.append(_distribution_dirichlet(n, rng, alpha=a))

    return cand

def _softmax_adam(a_init: np.ndarray, deadline: float,
                  best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Adam on the soft‑max surrogate of the max‑convolution."""
    x = np.log(a_init + 1e-12)                     # free logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05                                     # base learning rate
    t = 0
    warmup = 5000                                 # temperature ramp length

    while time.time() < deadline:
        t += 1
        # recover a from logits (soft‑max)
        max_x = np.max(x)
        e = np.exp(x - max_x)
        a = e / e.sum()

        conv = _fft_self_convolve(a)
        max_conv = conv.max()

        # linear temperature schedule (0.5 → 80)
        if t <= warmup:
            alpha = 0.5 + (80.0 - 0.5) * t / warmup
        else:
            alpha = 80.0

        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()
        grad_a = _gradient_from_weights(a, w)

        # gradient w.r.t. logits (chain rule through soft‑max)
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 500 == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] soft‑adam it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _hardmax_adam(a_init: np.ndarray, deadline: float,
                  best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Adam on the exact max‑convolution (hard‑max) sub‑gradient."""
    x = np.log(a_init + 1e-12)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    t = 0

    while time.time() < deadline:
        t += 1
        max_x = np.max(x)
        e = np.exp(x - max_x)
        a = e / e.sum()

        conv = _fft_self_convolve(a)
        k = int(conv.argmax())
        grad_a = _gradient_max_index(a, k)

        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 500 == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hard‑adam it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Cheap stochastic walk: move a random fraction of mass from i to j."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue

        delta = a[i] * rng.random() * 0.30   # up to 30 % of source mass
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Main entry point for the competition.  The routine respects the
    supplied time budget, builds a small pool of analytic seeds (length≈4000),
    refines the best seed with two Adam‑optimised phases (soft‑max → hard‑max)
    and finishes with a cheap pairwise‑swap polishing.  The best sequence
    (as a plain list) is returned.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0          # small safety margin

    # ------------------------------------------------------------------
    # 1. generate a modest collection of seeds (fixed length)
    n = 4000
    seeds = _initial_candidates_small(n, rng)

    # include the previously known solution if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            # adapt size to n
            if prev.size > n:
                prev = prev[:n]
            elif prev.size < n:
                prev = np.pad(prev, (0, n - prev.size), constant_values=0.0)
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(n)
            prev = prev / prev.sum()
            seeds.append(prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2. evaluate all seeds, keep the top few (elite)
    elite = []
    overall_best_seq = None
    overall_best_score = float("inf")

    for cand in seeds:
        cand = np.maximum(cand, 0.0)
        if cand.sum() == 0.0:
            continue
        cand = cand / cand.sum()
        sc = _objective(cand)
        elite.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = cand.copy()
            print(f"[{time.time():.1f}] initial best score {sc:.6f}")

    elite.sort(key=lambda x: x[0])
    elite = elite[:5]          # keep only the five best seeds

    # ------------------------------------------------------------------
    # 3. allocate remaining budget – most of it on the champion seed
    time_left = deadline - time.time()
    if time_left <= 0.0:
        return overall_best_seq.tolist()

    # 40 % soft‑max, 40 % hard‑max, remaining 20 % polishing
    soft_deadline = time.time() + 0.4 * time_left
    hard_deadline = time.time() + 0.4 * time_left
    polish_deadline = deadline

    # start refining from the global best seen so far
    seed_seq = overall_best_seq.copy()

    # ------------------------------------------------------------------
    # Soft‑max Adam phase
    seed_seq, overall_best_score = _softmax_adam(
        seed_seq,
        soft_deadline,
        best_seq=overall_best_seq,
        best_score=overall_best_score,
    )
    overall_best_seq = seed_seq.copy()

    # ------------------------------------------------------------------
    # Hard‑max Adam phase
    seed_seq, overall_best_score = _hardmax_adam(
        seed_seq,
        hard_deadline,
        best_seq=overall_best_seq,
        best_score=overall_best_score,
    )
    overall_best_seq = seed_seq.copy()

    # ------------------------------------------------------------------
    # Final cheap polishing
    overall_best_seq, overall_best_score = _pairwise_swap_local(
        overall_best_seq,
        polish_deadline,
        best_seq=overall_best_seq,
        best_score=overall_best_score,
    )

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
