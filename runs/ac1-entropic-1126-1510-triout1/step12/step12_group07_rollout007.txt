def _distribution_sidon(n, rng, m=None):
    if m is None: m = int(np.sqrt(n))
    pos = []
    diff_set = set()
    attempts = 0
    while len(pos) < m and attempts < n*10:
        candidate = rng.integers(n)
        if candidate in pos: continue
        # compute difference with each existing pos
        good = True
        for p in pos:
            d = abs(candidate - p)
            if d in diff_set or d == 0:
                good = False
                break
        if good:
            for p in pos:
                diff_set.add(abs(candidate-p))
            pos.append(candidate)
        attempts += 1
    # assign equal weight to each selected position
    a = np.zeros(n)
    if len(pos)==0:
        a[0]=1.0
    else:
        a[pos]=1.0/len(pos)
    return a

"""
Hybrid search for the min‑max self‑convolution step function.

The idea is to (i) generate a broad set of analytic and random seeds,
including sparse Sidon‑like patterns that have few repeated
differences, (ii) run a long Adam optimisation on the smooth‑max
surrogate (soft‑max parametrisation keeps the vector on the simplex),
(iii) polish the best result with a multimax sub‑gradient descent,
and finally (iv) a cheap greedy pairwise‑swap walk.  The routine
adapts the allocated time budget (default 1000 s) and returns the
best sequence found as a plain list of floats.
"""
import time
import math
from typing import List, Tuple, Optional

import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of   Σ_k w_k·conv_k(a,a)  w.r.t. a.
    Implements 2·a⋆w where ⋆ is cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact evaluate_sequence objective (scale‑invariant, sum a = 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (s * s))

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """k = round(density·n) non‑zero entries, Dirichlet‑like masses."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_sidon_like(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Greedy construction of a Sidon‑like set (no repeated absolute
    differences).  The set size is roughly √n which is close to the
    theoretical maximum.  Equal weight is assigned on the selected
    positions.
    """
    target = int(np.sqrt(n))          # desired size
    positions: List[int] = []
    used_diffs = set()
    attempts = 0
    max_attempts = n * 20

    while len(positions) < target and attempts < max_attempts:
        cand = rng.integers(n)
        if cand in positions:
            attempts += 1
            continue
        good = True
        for p in positions:
            d = abs(cand - p)
            if d == 0 or d in used_diffs:
                good = False
                break
        if good:
            for p in positions:
                used_diffs.add(abs(cand - p))
            positions.append(cand)
        attempts += 1

    if len(positions) == 0:          # fallback to uniform if we failed
        return _distribution_uniform(n)

    a = np.zeros(n, dtype=np.float64)
    a[positions] = 1.0 / len(positions)
    return a


# ----------------------------------------------------------------------
def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    shift = np.max(z)
    exp_z = np.exp(z - shift)
    s = exp_z.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / len(z))
    return exp_z / s


# ----------------------------------------------------------------------
def _adam_optimize(z0: np.ndarray,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   alpha0: float = 1.0,
                   alpha_max: float = 2000.0,
                   growth_factor: float = 1.05,
                   lr: float = 0.12,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   max_iter: int = 500000,
                   eval_interval: int = 200) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    The temperature α grows geometrically up to αmax.
    """
    n = z0.shape[0]
    z = z0.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        # probability vector a on the simplex
        a = _softmax(z)

        # convolution and smooth‑max weights
        conv = _fft_self_convolve(a)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient wrt a (exact for the smooth surrogate)
        grad_a = _gradient_from_weights(a, w)

        # chain rule (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        z = z - step

        # temperature schedule
        if t % 500 == 0:
            alpha = min(alpha * growth_factor, alpha_max)

        # periodic exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.6,
                                 max_iter: int = 100000,
                                 eval_interval: int = 300) -> Tuple[np.ndarray, float]:
    """
    Classic multimax sub‑gradient descent (exponentiated‑gradient on the
    simplex).  Used for polishing after Adam.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)

        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] multimax it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Greedy pairwise‑swap walk – only accepts objective‑improving moves."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction (up to 30 %) of mass from i → j
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the supplied time budget (default 1000 s) and
    returns the best sequence encountered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # a tiny safety margin

    # ------------------------------------------------------------------
    # 1.  choose a handful of lengths (larger n gives finer constants)
    lengths = [4000, 8000, 12000, 16000]

    overall_best_seq: Optional[np.ndarray] = None
    overall_best_score = float("inf")
    candidates: List[Tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 2.  generate diverse seeds for each length (cheap exact evaluations)
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        def _record(cand_vec: np.ndarray):
            nonlocal overall_best_seq, overall_best_score
            cand_vec = np.maximum(cand_vec, 0.0)
            s = cand_vec.sum()
            if s == 0.0:
                return
            cand_vec = cand_vec / s
            score = _objective(cand_vec)
            candidates.append((score, cand_vec.copy()))
            if score < overall_best_score:
                overall_best_score = score
                overall_best_seq = cand_vec.copy()
                print(f"[{time.time():.1f}] **** new global best len={n} score={score:.6f}")

        # analytic families
        _record(_distribution_uniform(n))
        _record(_distribution_arcsine(n))
        for p in (0.1, 0.2, 0.4, 0.8, 1.0, 1.5, 2.0):
            _record(_distribution_beta_sym(n, p))

        # random sparse seeds (different densities)
        for dens in (0.005, 0.01, 0.02):
            _record(_distribution_random_sparse(n, rng, density=dens))

        # Dirichlet draws (small alphas give sparsity)
        for _ in range(3):
            alpha = 0.05 + 0.45 * rng.random()
            _record(_distribution_dirichlet(n, rng, alpha=alpha))

        # Sidon‑like pattern – the most promising deterministic sparse seed
        _record(_distribution_sidon_like(n, rng))

        # keep a bounded list of promising candidates
        candidates.sort(key=lambda x: x[0])
        candidates = candidates[:20]

    # ------------------------------------------------------------------
    # 3.  incorporate a previously computed solution if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s == 0.0:
                prev = _distribution_uniform(prev.shape[0])
                s = prev.sum()
            prev = prev / s
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best improved: {sc:.6f}")
            candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4.  fallback (should never happen)
    if overall_best_seq is None:
        n = lengths[0]
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Adam smooth‑max descent on the best few seeds
    top_seeds = [seq for _, seq in sorted(candidates, key=lambda x: x[0])[:8]]
    for seed_seq in top_seeds:
        if time.time() >= deadline:
            break
        # allocate a slice of the remaining time (at least 30 s if possible)
        remaining = deadline - time.time()
        slice_time = min(80.0, remaining / max(1, len(top_seeds)))
        sub_deadline = time.time() + slice_time

        z0 = np.log(np.maximum(seed_seq, 1e-300))
        seq_out, score_out = _adam_optimize(
            z0,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            alpha0=1.0,
            alpha_max=2000.0,
            growth_factor=1.05,
            lr=0.12,
            max_iter=500000,
            eval_interval=250,
        )
        if score_out < overall_best_score:
            overall_best_score = score_out
            overall_best_seq = seq_out.copy()
            print(f"[{time.time():.1f}] after Adam new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  longer multimax sub‑gradient polishing
    if time.time() < deadline - 5:
        sub_deadline = time.time() + (deadline - time.time()) * 0.5
        overall_best_seq, overall_best_score = _subgradient_multimax_descent(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.6,
            max_iter=120000,
            eval_interval=300,
        )
        print(f"[{time.time():.1f}] after multimax best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  final greedy pairwise‑swap walk
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] after final swaps best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
