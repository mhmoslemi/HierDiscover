def _simulated_annealing(a, deadline, current_best_score):
    n = a.shape[0]; rng = np.random.default_rng()
    best_seq = a.copy(); best_score = current_best_score
    cur_seq = a.copy(); cur_score = _objective(cur_seq)
    # initial temperature such that some probability acceptance
    start_temp = max(1e-2, cur_score) # maybe
    end_temp = 1e-6
    # compute total steps estimated: maybe 10000 steps, but we time-limit.
    while time.time() < deadline:
        # compute proportion of time passed
        t_frac = (time.time() - start) / (deadline - start)
        temp = start_temp * (1 - t_frac) + end_temp * t_frac
        # propose move
        i = rng.choice(np.where(cur_seq > 1e-12)[0])
        j = rng.integers(n)
        if i==j:
            continue
        delta = cur_seq[i] * rng.random() * 0.5 # up to 50% move
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # renorm not needed (sum unchanged)
        new_score = _objective(new_seq)
        if new_score < cur_score:
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"[{time.time():.1f}] SA new best {best_score:.6f}")
        else:
            # accept with prob exp(-(new-cur)/temp)
            if temp > 0:
                prob = math.exp(-(new_score - cur_score) / temp)
                if rng.random() < prob:
                    cur_seq = new_seq
                    cur_score = new_score
    return best_seq, best_score

"""
Hybrid optimizer for the self‑convolution max‑ratio problem.

The routine works in three stages:

1. **Broad candidate generation** – analytic families (uniform, arcsine,
   sine, two‑block) together with a dense sweep of the symmetric Beta
   distribution (≈ 40 values of its shape parameter p) are created for a
   few different lengths (8000‑20000).  All candidates are normalised to
   sum = 1 and their exact objective 2·n·max(conv(a,a)) is evaluated.
   The best few (by objective value) are kept as elite seeds.

2. **Intensive refinement** – the current global best seed is refined
   with a long soft‑max exponentiated‑gradient descent (temperature
   α growing from 0.1 to 80) followed by a hard‑max exponentiated‑gradient
   descent (sub‑gradient of the true maximal convolution entry).  Both
   phases work on the probability simplex (multiplicative updates + a
   renormalisation).

3. **Global polishing** – a simulated‑annealing walk (random mass
   transfers accepted with a Metropolis criterion) is performed for the
   remaining time budget, followed by a cheap greedy pairwise‑swap
   walk that only accepts improving moves.

All operations respect the provided time budget (`budget_s`).  The
final best sequence is returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a))  (a is assumed to sum to 1).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  Σ_k w_k (a∗a)[k]  w is a probability vector.
    Returns 2·a⋆w.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.size - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _gradient_max_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the single maximal convolution entry conv[k].
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i = np.arange(i_min, i_max + 1, dtype=np.int64)
    j = k - i

    eq_mask = (i == j)
    if np.any(eq_mask):
        g[i[eq_mask]] += 2.0 * a[i[eq_mask]]

    if np.any(~eq_mask):
        ii = i[~eq_mask]
        jj = j[~eq_mask]
        g[ii] += a[jj]
        g[jj] += a[ii]
    return g


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * x / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((p - 1) * np.log(np.clip(x, eps, 1.0))
               + (p - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _softmax_descent(a: np.ndarray, deadline: float,
                    best_seq: np.ndarray, best_score: float,
                    lr0: float = 1.0,
                    max_iter: int = 300000,
                    eval_interval: int = 2000,
                    alpha_start: float = 0.1,
                    alpha_end: float = 80.0) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on a soft‑max approximation of max(conv)."""
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        # temperature schedule (linear in iteration count)
        alpha = alpha_start + (alpha_end - alpha_start) * (t / max_iter)
        # stable soft‑max
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()
        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= total

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] soft‑max it={t:,} α={alpha:.2f} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def _hardmax_descent(a: np.ndarray, deadline: float,
                     best_seq: np.ndarray, best_score: float,
                     lr0: float = 0.5,
                     max_iter: int = 300000,
                     eval_interval: int = 2000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true maximal convolution entry."""
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())
        grad = _gradient_max_index(a, k)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= total

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hard‑max it={t:,} idx={k} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, deadline: float,
                         cur_best_score: float) -> tuple[np.ndarray, float]:
    """
    Metropolis walk that moves a random fraction of mass from one index
    to another.  The temperature decays linearly from start_temp to
    end_temp.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_score = cur_best_score
    cur_seq = a.copy()
    cur_score = cur_best_score

    start_time = time.time()
    total_time = deadline - start_time
    if total_time <= 0:
        return best_seq, best_score

    start_temp = max(1e-3, cur_score)   # a scale comparable to the objective
    end_temp = 1e-7

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / total_time
        temp = start_temp * (1 - frac) + end_temp * frac

        # source must have some mass
        nonzero = np.where(cur_seq > 1e-12)[0]
        if nonzero.size == 0:
            break
        i = rng.choice(nonzero)
        j = rng.integers(n)
        if j == i:
            continue

        # move a random fraction (up to 30 %) of the source mass
        delta = cur_seq[i] * rng.random() * 0.30
        if delta < 1e-15:
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # sum unchanged, no need to renorm

        new_score = _objective(new_seq)

        if new_score < cur_score:
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"[{time.time():.1f}] SA new best {best_score:.6f}")
        else:
            # accept uphill moves with probability exp(-(Δ)/temp)
            if temp > 0:
                prob = math.exp(-(new_score - cur_score) / temp)
                if rng.random() < prob:
                    cur_seq = new_seq
                    cur_score = new_score
    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Cheap random mass‑transfer walk that only keeps improving moves."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue

        delta = a[i] * rng.random() * 0.30
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score

            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] !!! new global best len={n} score={sc:.6f}")
            elite.append((sc, cand.copy()))

        # dense symmetric Beta sweep (p∈[0.2,0.9])
        for p in np.linspace(0.20, 0.90, 35):
            cand = _distribution_beta_sym(n, p)
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] !!! new global best (Beta p={p:.3f}) len={n} score={sc:.6f}")
            elite.append((sc, cand.copy()))

        # random Dirichlet draws (different concentrations)
        for _ in range(6):
            alpha = 10 ** rng.uniform(-2, 1)   # from 0.01 to 10
            cand = _distribution_dirichlet(n, rng, alpha=alpha)
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] !!! new global best (Dirichlet α={alpha:.3g}) len={n} score={sc:.6f}")
            elite.append((sc, cand.copy()))

        # sparse random seeds
        for dens in (0.005, 0.01, 0.02):
            cand = _distribution_random_sparse(n, rng, density=dens)
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] !!! new global best (sparse {dens:.3f}) len={n} score={sc:.6f}")
            elite.append((sc, cand.copy()))

        # keep only the very best candidates seen so far
        elite.sort(key=lambda x: x[0])
        elite = elite[:20]

    # --------------------------------------------------------------
    # 3️⃣ If a previous solution is available, inject it into the elite list
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best improved: {sc:.6f}")
            elite.append((sc, prev.copy()))
            elite.sort(key=lambda x: x[0])
            elite = elite[:20]
    except Exception:
        pass

    # --------------------------------------------------------------
    # 4️⃣ Safety fallback (should never happen)
    if overall_best_seq is None:
        n = 8000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 5️⃣ Intensive refinement of the current global best
    if time.time() < deadline:
        # allocate roughly 70% of the remaining budget to deep refinement
        remaining = deadline - time.time()
        refine_budget = remaining * 0.70
        # soft‑max phase
        soft_deadline = time.time() + refine_budget * 0.45
        seq_a, score_a = _softmax_descent(
            overall_best_seq.copy(),
            soft_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=1.0,
            max_iter=400000,
            eval_interval=3000,
            alpha_start=0.1,
            alpha_end=80.0,
        )
        if score_a < overall_best_score:
            overall_best_score = score_a
            overall_best_seq = seq_a.copy()
            print(f"[{time.time():.1f}] after soft‑max new best {overall_best_score:.6f}")

        # hard‑max phase
        if time.time() < deadline:
            hard_deadline = time.time() + refine_budget * 0.45
            seq_b, score_b = _hardmax_descent(
                overall_best_seq.copy(),
                hard_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.5,
                max_iter=400000,
                eval_interval=3000,
            )
            if score_b < overall_best_score:
                overall_best_score = score_b
                overall_best_seq = seq_b.copy()
                print(f"[{time.time():.1f}] after hard‑max new best {overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 6️⃣ Simulated‑annealing polishing (uses the rest of the budget)
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _simulated_annealing(
            overall_best_seq.copy(),
            deadline,
            cur_best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] after SA best {overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 7️⃣ Final cheap greedy pairwise swaps
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # --------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
