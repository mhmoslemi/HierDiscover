"""
Hybrid optimiser for the max‑convolution bound.

Key ideas
---------
*   Generate a very rich seed pool (analytic families, random Dirichlet,
    random sparse, mixture of Beta families and an up‑sampled version of
    the best solution found so far).
*   Run a smooth‑max Adam optimiser (exponentiated‑gradient) with a
    slowly increasing temperature α.  This explores a smooth landscape
    before focussing on the true ℓ∞‑objective.
*   Refine the Adam result with a classic multimax sub‑gradient descent
    (exact max rather than a soft‑max).
*   Apply a cheap greedy pair‑wise mass‑transfer polishing step.
*   Finish with a short simulated‑annealing walk that can escape small
    local minima.
*   All steps respect the global deadline (default 1000 s) and keep the
    vector on the probability simplex (non‑negative, sum = 1).

The routine returns the best coefficient list found within the time
budget.  The sequence can be fed directly to ``evaluate_sequence``.

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate.
    Returns 2·a⋆w where w (length 2·n‑1) sums to 1.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a))/sum(a)^2.
    The vector is assumed to be on the simplex (sum≈1), therefore it
    reduces to 2·n·max(conv).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discretised unit interval."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(alpha,beta) on the discretised unit interval."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_general(n: int, p: float) -> np.ndarray:
    """Power‑law i^{p} (p may be negative)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Few non‑zero coordinates, random values."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    """Two symmetric spikes of equal mass."""
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_edge_power(n: int, a: float, b: float) -> np.ndarray:
    """
    General edge‑biased power shape:
        w_i ∝ (i+1)^{a}·(n‑i)^{b},   i = 0 … n‑1
    Allows simultaneous emphasis of the left and right edges.
    """
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    with np.errstate(divide='ignore'):
        w = np.exp(a * np.log(i + 1 + eps) + b * np.log(n - i + eps))
    if w.sum() == 0.0 or np.isnan(w).any():
        w = np.full(n, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _upsample(seq: np.ndarray, n_new: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = seq.shape[0]
    old_pos = (np.arange(old_n) + 0.5) / old_n
    new_pos = (np.arange(n_new) + 0.5) / n_new
    up = np.interp(new_pos, old_pos, seq, left=0.0, right=0.0)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full(n_new, 1.0 / n_new)
    return up

def _initial_candidates(n: int, rng: np.random.Generator,
                       prev_seq: np.ndarray | None) -> List[np.ndarray]:
    """
    Produce a diverse collection of seed vectors for a given length.
    """
    cand: List[np.ndarray] = []

    # Classical families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # Symmetric Beta – log‑spaced exponents
    for p in np.geomspace(0.01, 5.0, num=12):
        cand.append(_distribution_beta_sym(n, float(p)))

    # Random asymmetric Beta draws
    for _ in range(6):
        alpha = 10 ** rng.uniform(-3, 1)
        beta = 10 ** rng.uniform(-3, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # Power‑law (both increasing and decreasing)
    for p in (-2.5, -1.5, -0.5, 0.5, 1.5, 2.5):
        cand.append(_distribution_power_general(n, p))

    # Edge‑biased power shapes
    for _ in range(4):
        a = rng.uniform(-2.0, 2.0)
        b = rng.uniform(-2.0, 2.0)
        cand.append(_distribution_edge_power(n, a, b))

    # Exponential families
    for lam in (0.001, 0.003, 0.005, 0.01):
        cand.append(_distribution_exponential_increasing(n, lam))

    # Two‑block spikes
    for bf in (0.03, 0.05, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac=bf))

    # Random sparse patterns
    cand.append(_distribution_random_sparse(n, rng, density=0.01))
    cand.append(_distribution_random_sparse(n, rng, density=0.03))

    # Dirichlet draws
    for a in (0.1, 0.3, 0.7, 1.0):
        cand.append(_distribution_dirichlet(n, rng, alpha=a))

    # Completely random probability vectors
    for _ in range(5):
        v = rng.random(n)
        v /= v.sum()
        cand.append(v)

    # Upsampled previous best (if available)
    if prev_seq is not None:
        up = _upsample(prev_seq, n)
        cand.append(up)

    return cand

def _smooth_max_adam_descent(a: np.ndarray,
                             deadline: float,
                             best_seq: np.ndarray,
                             best_score: float,
                             eta0: float = 0.7,
                             beta1: float = 0.9,
                             beta2: float = 0.999,
                             eps: float = 1e-8,
                             alpha0: float = 0.3,
                             alpha_mul: float = 1.2,
                             alpha_max: float = 120.0,
                             max_iter: int = 60000,
                             eval_interval: int = 400) -> Tuple[np.ndarray, float]:
    """
    Adam‑style exponentiated‑gradient descent on the smooth‑max surrogate
    f_α(a) = (1/α)·log Σ exp(α·conv_i(a)).
    Temperature α starts at ``alpha0`` and is increased geometrically.
    """
    n = a.shape[0]
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)

        # soft‑max weights (stable)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        grad = _gradient_from_weights(a, w)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # multiplicative update
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(-step)

        # projection onto simplex (simple normalisation is enough)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # temperature schedule
        if t % 300 == 0 and alpha < alpha_max:
            alpha = min(alpha * alpha_mul, alpha_max)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score

def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.6,
                                 max_iter: int = 35000,
                                 eval_interval: int = 400) -> Tuple[np.ndarray, float]:
    """
    Classical exponentiated‑gradient descent where the gradient is the
    average of the sub‑gradients corresponding to indices attaining the
    maximal convolution value.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()

        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Multimax it={t} new best {best_score:.6f}")

    return best_seq, best_score

def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Greedy pairwise mass‑transfer: move a small amount from i to j if it improves."""
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        # distinct destination
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

                eta0=0.7,
                beta1=0.9,
                beta2=0.999,
                eps=1e-8,
                alpha0=0.3,
                alpha_mul=1.2,
                alpha_max=120.0,
                max_iter=60000,
                eval_interval=300,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after Adam descent new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Second refinement: classic multimax sub‑gradient descent.
    if time.time() < deadline:
        seeds = [overall_best_seq.copy()] + [
            seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:2]
            if not np.may_share_memory(seq, overall_best_seq)
        ]
        for seed_seq in seeds:
            if time.time() >= deadline:
                break
            seq_out, score_out = _subgradient_multimax_descent(
                seed_seq.copy(),
                deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.6,
                max_iter=35000,
                eval_interval=300,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after multimax new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final polishing with cheap greedy pairwise swaps.
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    # 7. Optional simulated‑annealing walk for the last few seconds.
    if time.time() < deadline:
        cur_seq = overall_best_seq.copy()
        cur_score = overall_best_score
        temp_start = 1e-2
        while time.time() < deadline:
            elapsed = time.time() - start_time
            remaining = deadline - time.time()
            # linear cooling
            temp = temp_start * max(0.0, 1.0 - elapsed / (deadline - start_time))
            i = rng.integers(len(cur_seq))
            if cur_seq[i] == 0.0:
                continue
            j = rng.integers(len(cur_seq) - 1)
            if j >= i:
                j += 1
            delta = cur_seq[i] * rng.random() * 0.2
            new_seq = cur_seq.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
            new_seq /= new_seq.sum()
            new_score = _objective(new_seq)
            accept = False
            if new_score < cur_score:
                accept = True
            elif temp > 0.0:
                prob = math.exp(-(new_score - cur_score) / temp)
                if rng.random() < prob:
                    accept = True
            if accept:
                cur_seq = new_seq
                cur_score = new_score
                if cur_score < overall_best_score:
                    overall_best_score = cur_score
                    overall_best_seq = cur_seq.copy()
                    print(f"[{time.time():.1f}] SA new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
