def _fft_self_convolve(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:2*n-1]
```

def _interpolate_distribution(old, new_n):
    old_n = len(old)
    x_old = np.linspace(0,1,old_n)
    x_new = np.linspace(0,1,new_n)
    new = np.interp(x_new, x_old, old)
    new = np.maximum(new,0)
    if new.sum()==0:
        new = np.full(new_n, 1.0/new_n)
    else:
        new = new / new.sum()
    return new
```

#!/usr/bin/env python
"""
Hybrid optimiser for the max‑convolution step‑function problem.

*  Generate a diverse set of seed distributions (uniform, arcsine, beta,
   Dirichlet, sparse, and the previous best solution if available).
*  Run a smooth‑max Adam‑style mirror descent on each seed with a very
   slow temperature schedule – this directly minimises a smooth approximation
   of the maximum convolution.
*  Refine the best candidates with a classic multimax (sub‑gradient) descent.
*  Finish with a cheap greedy pairwise mass‑transfer polishing step.
*  All updates keep the vector on the probability simplex (non‑negative,
   sum = 1).  The routine respects the time budget (default 1000 s) and
   returns the best coefficient list found.
"""

import time
import math
from typing import List, Tuple
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the sub‑gradient 2·a⋆w where w (length 2·n‑1) is a probability
    vector over the convolution indices.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (sum=1 ⇒ 2·n·max)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _dist_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _dist_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _dist_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discretised unit interval."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _dist_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(alpha,beta) on the discretised unit interval."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _dist_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _dist_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _dist_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    """Two symmetric spikes of equal total mass."""
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _dist_dirichlet(n: int, rng: np.random.Generator, alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _dist_random_sparse(n: int, rng: np.random.Generator,
                       density: float = 0.01) -> np.ndarray:
    """Few non‑zero coordinates, random values."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _dist_random_dense(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    v = np.maximum(v, 0.0)
    v /= v.sum()
    return v


# ----------------------------------------------------------------------
def _interpolate_distribution(old: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = len(old)
    if old_n == new_n:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    new = np.maximum(new, 0.0)
    s = new.sum()
    if s == 0.0:
        new = np.full(new_n, 1.0 / new_n)
    else:
        new = new / s
    return new


# ----------------------------------------------------------------------
def _smooth_max_adam_descent(a: np.ndarray,
                             deadline: float,
                             best_seq: np.ndarray,
                             best_score: float,
                             eta0: float = 0.5,
                             beta1: float = 0.9,
                             beta2: float = 0.999,
                             eps: float = 1e-8,
                             alpha0: float = 0.05,
                             alpha_mul: float = 1.1,
                             alpha_max: float = 300.0,
                             max_iter: int = 200_000,
                             eval_interval: int = 500) -> Tuple[np.ndarray, float]:
    """
    Adam‑style exponentiated‑gradient descent on the smooth‑max surrogate
    f_α(a) = (1/α)·log Σ_i exp(α·conv_i(a)).
    The temperature α starts at ``alpha0`` and is increased slowly (by
    ``alpha_mul``) until ``alpha_max``.
    """
    n = a.shape[0]
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        grad = _gradient_from_weights(a, w)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(-step)

        # project back onto simplex (simple normalisation keeps us on it)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % 300 == 0 and alpha < alpha_max:
            alpha = min(alpha * alpha_mul, alpha_max)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.8,
                                 max_iter: int = 100_000,
                                 eval_interval: int = 500) -> Tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent for the exact max‑convolution objective.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)

        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Multimax it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Greedy pairwise mass‑transfer: move a little mass from i to j if it improves."""
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        # distinct destination
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # random transfer amount (up to 30 % of the source)
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects ``budget_s`` seconds and returns the best coefficient
    list found (as a plain list of floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: List[Tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Generate seeds for a couple of lengths (cheap initial scan)
    lengths = [4000, 8000]
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Generating seeds for length {n}")

        seeds: List[np.ndarray] = []

        # basic analytic families
        seeds.append(_dist_uniform(n))
        seeds.append(_dist_arcsine(n))

        # symmetric beta exponents (log‑spaced)
        for p in np.geomspace(0.01, 5.0, num=12):
            seeds.append(_dist_beta_sym(n, float(p)))

        # random asymmetric Beta draws
        for _ in range(5):
            alpha = 10 ** rng.uniform(-3, 1)
            beta = 10 ** rng.uniform(-3, 1)
            seeds.append(_dist_beta_asym(n, alpha, beta))

        # Dirichlet draws with varied concentration
        for a in (0.2, 0.5, 0.8, 1.2):
            seeds.append(_dist_dirichlet(n, rng, alpha=a))

        # sparse patterns
        seeds.append(_dist_random_sparse(n, rng, density=0.01))
        seeds.append(_dist_random_sparse(n, rng, density=0.03))

        # a few dense random vectors
        for _ in range(3):
            seeds.append(_dist_random_dense(n, rng))

        # incorporate previous solution if present
        try:
            prev = globals().get("height_sequence_1")
            if isinstance(prev, (list, np.ndarray)):
                prev_arr = np.asarray(prev, dtype=np.float64)
                if prev_arr.ndim == 1 and prev_arr.size > 0:
                    prev_arr = np.maximum(prev_arr, 0.0)
                    if prev_arr.sum() == 0.0:
                        prev_arr = _dist_uniform(prev_arr.shape[0])
                    prev_arr = prev_arr / prev_arr.sum()
                    # resize if necessary
                    if prev_arr.shape[0] != n:
                        prev_arr = _interpolate_distribution(prev_arr, n)
                    seeds.append(prev_arr)
        except Exception:
            pass

        # evaluate each seed
        for cand in seeds:
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                continue
            cand = cand / s
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] **** new global best len={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))

        # keep only a handful of promising seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:15]

    # ------------------------------------------------------------------
    # 2. Fallback in the unlikely event that we have no candidate yet.
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _dist_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam‑style smooth‑max refinement on the best few seeds.
    remaining = deadline - time.time()
    if remaining > 30:
        # pick the overall best + a few top‑ranked distinct seeds
        seeds_for_adam = [overall_best_seq]
        for sc, seq in top_candidates[:4]:
            if not np.may_share_memory(seq, overall_best_seq):
                seeds_for_adam.append(seq)
        seeds_for_adam = seeds_for_adam[:3]   # limit to keep schedule simple

        for idx, seed_seq in enumerate(seeds_for_adam):
            if time.time() >= deadline:
                break
            # time slice for this seed
            sub_deadline = time.time() + (deadline - time.time()) / max(1, len(seeds_for_adam))
            print(f"[{time.time():.1f}] Adam refinement {idx+1}/{len(seeds_for_adam)} "
                  f"len={len(seed_seq)}")
            seq_out, score_out = _smooth_max_adam_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.5,
                beta1=0.9,
                beta2=0.999,
                eps=1e-8,
                alpha0=0.05,
                alpha_mul=1.1,
                alpha_max=300.0,
                max_iter=200_000,
                eval_interval=500,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after Adam new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Classic multimax sub‑gradient descent on the current best.
    remaining = deadline - time.time()
    if remaining > 15:
        print(f"[{time.time():.1f}] Starting multimax refinement")
        seq_out, score_out = _subgradient_multimax_descent(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.8,
            max_iter=100_000,
            eval_interval=500,
        )
        if score_out < overall_best_score:
            overall_best_score = score_out
            overall_best_seq = seq_out.copy()
            print(f"[{time.time():.1f}] after multimax new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final greedy pair‑wise polishing.
    if time.time() < deadline:
        print(f"[{time.time():.1f}] Final pairwise polishing")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
