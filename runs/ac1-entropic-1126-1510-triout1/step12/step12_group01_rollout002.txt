def _subgradient_max_convolution(a):
    conv = _fft_self_convolve(a)
    max_c = conv.max()
    # active indices
    eps = max_c*1e-12 + 1e-14
    active_idxs = np.where(conv >= max_c - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active_idxs:
        start = max(0, k - (n-1))
        end = min(n-1, k)
        i_vals = np.arange(start, end+1)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    # average across active set to keep scale consistent
    grad /= len(active_idxs)
    return grad

"""
Improved search for a non‑negative step‑function minimizing
``evaluate_sequence``.

The algorithm consists of three phases

1. **Diverse seeding** – generate many candidate vectors from several
   analytic families (uniform, arcsine, beta, edge‑weighted, sine‑weighted,
   random Dirichlet, sparse).  Keep the best ~30 seeds.

2. **Exact active‑set exponential descent** – perform mirror descent on the
   exact objective using the true sub‑gradient of the maximum convolution
   entry (instead of a soft‑max).  The sub‑gradient is obtained by summing
   the shifts of the current vector over the (few) convolution indices that
   are within a tiny tolerance of the maximum.  After each multiplicative
   update the vector is projected back onto the probability simplex.
   The step size η = η₀ / √t and a jitter are used to escape plateaus.

3. **Refinement** – a cheap pairwise‑swap local search and random convex
   mixing of the best solutions are applied.

The whole routine respects the given time budget (default 1000 s) and
returns a plain Python list.  All numbers are non‑negative, and the caller
``evaluate_sequence`` will perform the final normalisation and clipping.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# 1. Utility functions
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# 2. Analytic families (all return a probability vector, i.e. Σ=1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base,   U = uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# 3. Exact sub‑gradient of max(conv) and exponential descent
# ----------------------------------------------------------------------
def _max_convolution_gradient(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Returns (grad, max_c) where `grad` is a sub‑gradient of
    max(conv(a,a)) at the current a, and `max_c` = max(conv(a,a)).
    The sub‑gradient is the average of the gradients of all convolution
    entries that are within a tiny tolerance of the maximum.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_c = conv.max()
    # tolerance – absolute 1e-12 or relative 1e-12, whichever is larger
    eps = max(1e-12, max_c * 1e-12)
    active = np.where(conv >= max_c - eps)[0]

    if active.size == 0:
        # fallback to the single maximal index
        active = np.array([conv.argmax()], dtype=int)

    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=int)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]

    # averaging keeps the step size comparable to the single‑maximum case
    grad /= max(1, active.size)
    return grad, max_c


def _refine_sequence_exact(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.55,
    eval_interval: int = 150,
    patience: int = 2500,
) -> tuple[np.ndarray, float]:
    """
    Exponential‑gradient (mirror) descent on the exact objective.
    Uses the true active‑set gradient.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    t = 0
    no_improve = 0

    while time.time() < deadline and t < 500_000:
        t += 1
        grad, max_c = _max_convolution_gradient(a)

        # multiplicative update
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # projection onto the simplex (keeps Σa = 1)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_interval == 0:
            cur = 2.0 * n * max_c
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] descent it={t} new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            # add a tiny stochastic perturbation to escape a flat region
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    # final evaluation (in case loop ends before an eval)
    conv = _fft_self_convolve(a)
    final_score = 2.0 * n * conv.max()
    if final_score < best_score:
        best_score, best_seq = final_score, a.copy()
        print(f"[{time.time():.1f}] descent final improvement to {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4. Deterministic local refinement (pairwise swaps)
# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly transfer a small amount of mass; accept if the objective improves."""
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score

    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a plain Python list of heights (the sum does not need to be 1 –
    ``evaluate_sequence`` will normalise it).  The routine respects the
    provided time budget (default 1000 s).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.5  # safety margin for the final return

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []  # (score, vector)

    # ------------------------------------------------------------------
    # 1. Build a diverse pool of seeds (different lengths & families)
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 16000, 32000]
    p_grid = np.concatenate(
        [
            np.logspace(-2, -1, 9),   # 0.01 … 0.1
            np.logspace(-1, 0, 9),    # 0.1 … 1
            np.logspace(0, 0.5, 7),   # 1 … 3.16
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 7)  # mixing with uniform

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # parametrised families + mixing
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            sine = _distribution_sine_weighted(n, p)
            for base in (beta, edge, sine):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # asymmetric beta (random a,b)
        for _ in range(3):
            if time.time() >= deadline:
                break
            a_param = 10 ** rng.uniform(-2, 0)   # 0.01 … 1
            b_param = 10 ** rng.uniform(-2, 0)
            cand = _distribution_beta_asym(n, a_param, b_param)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best asym‑beta len={n} a={a_param:.3g} b={b_param:.3g} score={sc:.6f}"
                )

        # random Dirichlet seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.02, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(
                    f"[{time.time():.1f}] new global best Dirichlet len={n} score={sc:.6f}"
                )

        # sparse random seeds
        for _ in range(3):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(
                    f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}"
                )

        # keep only the very promising seeds overall (size limited)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:40]

    # ------------------------------------------------------------------
    # 2. Include previously discovered construction if it exists
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() > 0.0:
                prev = prev / prev.sum()
                sc = _objective(prev)
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, prev.copy()
                    print(f"[{time.time():.1f}] previous best score={sc:.6f}")
                top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Guard – if we somehow have no seed, fall back to uniform
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Exact exponential descent on the most promising seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] start refined descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            a_opt, sc_opt = _refine_sequence_exact(
                seed_seq.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.55,
                eval_interval=120,
                patience=2200,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after descent seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5. Pairwise swap refinement (deterministic)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] pairwise swap refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 6. Convex mixing of top candidates (cheap exhaustive search)
    # ------------------------------------------------------------------
    if time.time() < deadline and len(top_candidates) >= 2:
        print(f"[{time.time():.1f}] convex mixing of top solutions")
        base = overall_best_seq.copy()
        for _, other in top_candidates[:5]:
            if time.time() >= deadline:
                break
            for lam in np.linspace(0.0, 1.0, 11):
                mixed = lam * base + (1 - lam) * other
                mixed = np.maximum(mixed, 0.0)
                mixed /= mixed.sum()
                sc = _objective(mixed)
                if sc < overall_best_score - 1e-14:
                    overall_best_score, overall_best_seq = sc, mixed.copy()
                    print(
                        f"[{time.time():.1f}] mixing improved to {overall_best_score:.6f} (λ={lam:.2f})"
                    )

    # ------------------------------------------------------------------
    # 7. Final report
    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
