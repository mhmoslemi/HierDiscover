"""
Hybrid optimiser for the step‑function max‑convolution problem.

Main ideas
~~~~~~~~~~
*   Generate a very rich set of seed vectors (uniform, arcsine,
    many Beta families, power‑law, exponential, sine, two‑block
    spikes, Dirichlet draws, sparse patterns and the previous best).

*   Refine the best candidates with classic sub‑gradient (multimax)

*   Finish with a greedy pair‑wise mass‑transfer polishing step.
*   All updates stay on the probability simplex (non‑negative,
    sum = 1) via Euclidean projection.

The routine respects a 1000 s budget, prints intermediate scores,
and returns the best coefficient list found.
"""

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT helpers (real)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of   Σ_k w_k·conv_k(a)   w.r.t. a (length n)."""
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Exact objective (identical to evaluate_sequence for sum = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a))  (a is assumed to sum to 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Analytic families (many new shapes added)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discretised interval."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(alpha,beta)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp(
        (alpha - 1) * np.log(np.clip(x, eps, 1.0))
        + (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0))
    )
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_power_decreasing(n: int, p: float) -> np.ndarray:
    """Decreasing power law a_i ∝ (i+1)^{‑p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_exponential_decreasing(n: int, lam: float = 0.005) -> np.ndarray:
    """U‑shaped exponential: high at both ends, low in the centre."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    vals = np.exp(-lam * d)
    vals /= vals.sum()
    return vals


def _distribution_half_sine(n: int) -> np.ndarray:
    """Half‑sine shape (zero at the ends, max in the centre)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * i / (n + 1.0))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    """Two symmetric spikes of equal mass."""
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Few random non‑zero coordinates."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(
    n: int, rng: np.random.Generator, alpha: float = 0.5
) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Candidate generation
# ----------------------------------------------------------------------
def _initial_candidates_extended(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    """Produce a diverse collection of seed vectors for a given length."""
    cand = []

    # Core families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # Symmetric Beta – log‑spaced exponents
    for p in np.geomspace(0.01, 5.0, num=12):
        cand.append(_distribution_beta_sym(n, float(p)))

    # Random asymmetric Beta draws
    for _ in range(6):
        alpha = 10 ** rng.uniform(-3, 1)   # 0.001 … 10
        beta = 10 ** rng.uniform(-3, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # Power‑law families
    for p in (0.5, 1.0, 2.0, 3.0):
        cand.append(_distribution_power_increasing(n, p))
        cand.append(_distribution_power_decreasing(n, p))

    # Exponential families (both increasing and decreasing)
    for lam in (0.001, 0.003, 0.005, 0.008, 0.01):
        cand.append(_distribution_exponential_increasing(n, lam))
        cand.append(_distribution_exponential_decreasing(n, lam))

    # Half‑sine envelope
    cand.append(_distribution_half_sine(n))

    # Two‑block spikes – various widths
    for bf in (0.02, 0.04, 0.07, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac=bf))

    # Random sparse patterns
    cand.append(_distribution_random_sparse(n, rng, density=0.008))
    cand.append(_distribution_random_sparse(n, rng, density=0.025))

    # Dirichlet draws with varied concentration
    for a in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            cand.append(_distribution_dirichlet(n, rng, alpha=a))

    # Completely random probability vectors
    for _ in range(6):
        v = rng.random(n)
        v = np.maximum(v, 0.0)
        if v.sum() == 0.0:
            v[0] = 1.0
        v /= v.sum()
        cand.append(v)

    return cand


# ----------------------------------------------------------------------
# Adam‑style smooth‑max descent
# ----------------------------------------------------------------------
def _smooth_max_adam_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    alpha0: float = 0.3,
    alpha_mul: float = 1.25,
    alpha_max: float = 150.0,
    max_iter: int = 80000,
    eval_interval: int = 500,
) -> Tuple[np.ndarray, float]:
    """Adam on the smooth‑max surrogate f_α(a) = (1/α)·log Σ exp(α·conv_k)."""
    n = a.shape[0]
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)

        # soft‑max weights (stable)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient of the smooth surrogate
        grad = _gradient_from_weights(a, w)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # multiplicative update (exponentiated‑gradient style)
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(-step)

        # simplex projection
        a = _project_onto_simplex(a)

        # temperature schedule
        if t % 300 == 0 and alpha < alpha_max:
            alpha = min(alpha * alpha_mul, alpha_max)

        # occasional exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Classic multimax (sub‑gradient) descent
# ----------------------------------------------------------------------
def _subgradient_multimax_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.6,
    max_iter: int = 50000,
    eval_interval: int = 400,
) -> Tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent using the sub‑gradient of max(conv)."""
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)

        a = a * np.exp(-eta * grad)
        a = _project_onto_simplex(a)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Multimax it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise mass transfer
# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> Tuple[np.ndarray, float]:
    """Randomly move a small fraction of mass from i to j if it improves the score."""
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        # distinct destination
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # random transfer amount (up to 30 % of source)
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_a = _project_onto_simplex(new_a)

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Respects ``budget_s`` seconds and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Choose a set of lengths (larger lengths give more flexibility)
    # ------------------------------------------------------------------
    lengths = [4000, 8000, 12000]
    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: List[Tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 2. Quick scan of analytic families (cheap)
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Scanning length {n}")

        cand_list = _initial_candidates_extended(n, rng)
        for cand in cand_list:
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            cand = _project_onto_simplex(cand)   # enforce sum=1
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] **** new global best len={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))

        # keep only the most promising seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:24]

    # ------------------------------------------------------------------
    # 3. Include the previous solution if available.
    # ------------------------------------------------------------------
    try:
        prev = globals().get("height_sequence_1")
        if isinstance(prev, (list, np.ndarray)):
            prev = np.asarray(prev, dtype=np.float64)
            if prev.ndim == 1 and prev.size > 0:
                prev = np.maximum(prev, 0.0)
                prev = _project_onto_simplex(prev)
                sc = _objective(prev)
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = prev.copy()
                    print(f"[{time.time():.1f}] previous solution improved: {sc:.6f}")
                top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Fallback in the extremely unlikely case that we have nothing.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. First refinement: Adam‑style smooth‑max descent on a few seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 60:
        # select the best seeds (global best + a few others)
        seeds = [overall_best_seq.copy()] + [
            seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]
        ]
        for seed_seq in seeds:
            if time.time() >= deadline:
                break
            # allocate a fair slice of the remaining budget
            sub_deadline = time.time() + (deadline - time.time()) / max(1, len(seeds))
            seq_out, score_out = _smooth_max_adam_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.6,
                beta1=0.9,
                beta2=0.999,
                eps=1e-8,
                alpha0=0.3,
                alpha_mul=1.25,
                alpha_max=150.0,
                max_iter=80000,
                eval_interval=400,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after Adam descent new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Second refinement: classic multimax sub‑gradient descent.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30:
        seeds = [overall_best_seq.copy()] + [
            seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:3]
            if not np.may_share_memory(seq, overall_best_seq)
        ]
        for seed_seq in seeds:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + (deadline - time.time()) / max(1, len(seeds))
            seq_out, score_out = _subgradient_multimax_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.6,
                max_iter=60000,
                eval_interval=400,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after multimax new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final cheap polishing with greedy pairwise transfers.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return as a plain Python list as required by the evaluator.
    return overall_best_seq.tolist()
