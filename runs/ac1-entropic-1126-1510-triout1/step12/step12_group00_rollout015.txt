```
def evaluate_sequence(sequence):
    # checks: list, non-empty, types numbers, non-NaN, no inf
    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))
```

- Let a_i be the input after clipping negative to zero and >1000 to 1000.
- n = len(a)
- b = a * a (full convolution, length 2n-1). max_b = maximum convolution entry.
- sum_a = sum of a_i.
- Objective = 2 * n * max_b / (sum_a^2).

def _adam_optimize(a, deadline, best_seq, best_score,
                   lr0=0.05, beta1=0.9, beta2=0.999, epsilon=1e-8,
                   temp_factor_start=0.5, temp_factor_end=0.02):
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        # smoothed max
        temp_factor = temp_factor_start * (temp_factor_end / temp_factor_start) ** (t / max_iters) # geometric decay
        temp = max_c * max(temp_factor, 1e-12)  # ensure positive

        # compute softmax weights
        # shift for numerical stability
        shifted = conv - max_c
        exps = np.exp(shifted / temp)
        w = exps / exps.sum()

        # gradient
        grad_full = _fft_convolve(w, a)  # length 3n-2
        grad = 2.0 * grad_full[:n]

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr0 * m_hat / (np.sqrt(v_hat) + epsilon)

        # exponentiated gradient step
        a = a * np.exp(-step)

        # renorm
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

"""
Propose a non‑negative step function that minimizes ``evaluate_sequence``.

The algorithm
--------------

1. **Generate a diverse set of seeds** – analytic families
   (uniform, arcsine, symmetric beta, edge‑weighted, edge‑inverse‑power,
   sine‑weighted) together with a few random Dirichlet / sparse vectors.
   For each length we keep the best few seeds according to the exact
   objective ``2·n·max(conv(a,a))`` (sum‑normalised to 1).

2. **Smooth max optimisation** – a custom Adam optimiser.
   We replace the non‑smooth ``max`` by a temperature‑controlled
   soft‑max.  For a current vector `a` we compute
   * the full self‑convolution `c = a*a`,
   * a soft‑max weight `w = softmax(c / T)`,
   * the gradient of the smoothed objective, which can be written
     elegantly as `grad = 2·(w * a)[:n]` (full convolution of `w` with `a`).
   Adam’s adaptive moments are applied to this gradient and an
   *exponentiated‑gradient* step `a ← a·exp(‑step)` is taken.
   The temperature `T` decays geometrically, turning the smoothed
   objective gradually into the true max.  Periodically we evaluate the
   exact objective and keep the best vector seen.

3. **Local refinements** – a greedy pairwise mass‑transfer routine
   followed by a cheap simulated‑annealing style search (identical to the
   one used in the previous baseline).

The whole search respects the supplied time budget (default 1000 s)
and returns a plain Python list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft


# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two 1‑D arrays (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())
    X = rfft(x, n=size)
    Y = rfft(y, n=size)
    conv = irfft(X * Y, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
# Exact objective (sum‑normalised)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a) == 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, sum == 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (-p) + j ** (-p))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Return lam·U + (1‑lam)·base (U is the uniform distribution)."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smoothed max and its gradient
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, temp_factor: float) -> tuple[np.ndarray, float, np.ndarray, np.ndarray]:
    """
    Compute the soft‑max weights of the convolution and the exact gradient
    of the smoothed objective.

    Returns
    -------
    conv : ndarray
        Full self‑convolution a*a (length 2·n‑1).
    max_c : float
        Maximum entry of conv (used for the exact objective).
    w : ndarray
        Soft‑max weight vector, sum(w)=1, same length as conv.
    grad : ndarray
        Gradient of the smoothed objective w.r.t. a (length n).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)                # length 2n‑1
    max_c = conv.max()
    # temperature is a *fraction* of the current maximum
    temp = max(max_c * temp_factor, 1e-14)

    # stable soft‑max
    shifted = conv - max_c                     # non‑positive numbers
    exps = np.exp(shifted / temp)              # max entry becomes 1.0
    w = exps / exps.sum()                      # shape (2n‑1,)

    # gradient = 2·(w * a) on the first n entries
    grad_full = _fft_convolve(w, a)            # length 3n‑2
    grad = 2.0 * grad_full[:n]
    return conv, max_c, w, grad


# ----------------------------------------------------------------------
# Adam optimiser using exponentiated‑gradient updates
# ----------------------------------------------------------------------
def _adam_optimize(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    epsilon: float = 1e-8,
    temp_start: float = 0.5,
    temp_end: float = 0.02,
    eval_every: int = 120,
    jitter_patience: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max objective.
    ``a`` is assumed to sum to 1 and contain non‑negative entries.
    The function respects the supplied deadline.
    """
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    # an estimate of the total number of iterations we can afford
    max_iters = 200_000

    while time.time() < deadline and t < max_iters:
        t += 1

        # linear schedule for the temperature factor (geometric decay)
        factor = temp_start * ((temp_end / temp_start) ** (t / max_iters))

        conv, max_c, w, grad = _smooth_max_and_grad(a, factor)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + epsilon)

        # exponentiated‑gradient update
        a = a * np.exp(-step)

        # re‑normalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # periodic evaluation of the true (non‑smoothed) objective
        if t % eval_every == 0:
            cur = 2.0 * n * max_c
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] adam it={t:,} new global best {best_score:.6f}")
            else:
                no_improve += 1

        # occasional jitter to escape shallow plateaux
        if no_improve >= jitter_patience:
            jitter = rng.exponential(scale=0.005, size=n)
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Deterministic pairwise swap refinement (as in the baseline)
# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly transfer a small amount of mass; keep it iff the objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score

            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a plain Python list of heights (the routine respects the time‑budget).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. Build a pool of seeds for a handful of lengths
    lengths = [2000, 4000, 8000, 12000, 16000]  # diverse resolutions
    p_grid = np.concatenate(
        [
            np.logspace(-3, -1, 9),   # 0.001 … 0.1
            np.logspace(-1, 0, 9),    # 0.1 … 1
            np.logspace(0, 0.5, 7),   # 1 … 3.16
        ]
    )
    lam_grid = np.linspace(0.0, 0.8, 9)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] gathering seeds for length {n}")

        # a few deterministic families (no mixing)
        families = [
            _distribution_uniform(n),
            _distribution_arcsine(n),
        ]
        # parametrised families
        for p in p_grid:
            if time.time() >= deadline:
                break
            families.append(_distribution_beta_sym(n, p))
            families.append(_distribution_edge_weighted(n, p))
            families.append(_distribution_edge_inversepow(n, p))
            families.append(_distribution_sine_weighted(n, p))

        # mixing with uniform
        mixed = []
        for base in families:
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                mixed.append(_mix_with_uniform(base, lam))

        # random Dirichlet and sparse seeds
        for _ in range(5):
            mixed.append(_distribution_dirichlet(n, rng))
        for _ in range(5):
            mixed.append(_distribution_random_sparse(n, rng, density=0.006))

        # add any previously discovered construction if it exists
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.ndim == 1 and prev.size > 0:
                if prev.shape[0] != n:
                    # resample the old vector to the current length
                    x = np.linspace(0, 1, prev.shape[0])
                    xs = np.linspace(0, 1, n)
                    prev_resampled = np.interp(xs, x, prev)
                    prev_resampled = np.maximum(prev_resampled, 0.0)
                    prev_resampled /= prev_resampled.sum()
                    mixed.append(prev_resampled)
                else:
                    mixed.append(prev / prev.sum())
        except Exception:
            pass

        # ------------------------------------------------------------------
        # Evaluate all candidates (fast exact objective) and keep the best few
        scored = [(float(_objective(vec)), vec) for vec in mixed]
        scored.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in scored[:6]]

        # ------------------------------------------------------------------
        # 2. Run Adam optimiser on the most promising seeds
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        per_seed = remaining / max(1, len(top_seeds))
        print(f"[{time.time():.1f}] optimisation phase ({len(top_seeds)} seeds, {per_seed:.1f}s each)")

        for idx, seed_vec in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _adam_optimize(
                seed_vec.copy(),
                sub_deadline,
                overall_best_seq if overall_best_seq is not None else seed_vec,
                overall_best_score,
                lr0=0.05,
                temp_start=0.5,
                temp_end=0.02,
                eval_every=150,
                jitter_patience=2000,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after Adam seed {idx} new global best {overall_best_score:.6f}")

        # ------------------------------------------------------------------
        # keep track of the very best vector seen so far
        if overall_best_seq is not None:
            # record its exact objective (might be slightly outdated)
            cur_score = _objective(overall_best_seq)
            if cur_score < overall_best_score:
                overall_best_score = cur_score

    # ------------------------------------------------------------------
    # 3. Final deterministic pairwise refinement
    if overall_best_seq is not None and time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            overall_best_seq,
            overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 4. Final cheap simulated‑annealing search
    if overall_best_seq is not None and time.time() < deadline:
        print(f"[{time.time():.1f}] final random‑local search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            overall_best_seq,
            overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    if overall_best_seq is None:
        # Fallback – uniform distribution of a reasonable size
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # return a plain Python list – evaluate_sequence will clip/normalise as needed
    return overall_best_seq.tolist()
