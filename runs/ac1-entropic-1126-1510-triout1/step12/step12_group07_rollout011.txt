    x = np.arange(n)

"""
Hybrid global‑search + local‑refinement optimizer for the step‑function
max‑convolution problem.

The algorithm proceeds in four stages:

1. **Coarse global search** – a low‑dimensional parametric family
   (log‑linear interpolation with a few control points) is explored
   by SciPy's differential evolution.  This discovers promising shapes
   without falling into local minima.

2. **Smooth‑max Adam descent** – the best few candidates are refined
   with the Adam optimiser on the smooth‑max surrogate used in the
   reference solution.

3. **Multimax sub‑gradient polishing** – a short exponentiated‑gradient
   walk on the exact L∞‑objective further reduces the max‑convolution.

4. **Greedy pairwise swap** – cheap local mass‑transfer moves are
   performed until the time budget expires.

All sequences are kept on the probability simplex (non‑negative,
sum = 1) which makes the objective scale‑invariant and avoids the
clipping performed in ``evaluate_sequence``.  The routine respects a
user‑supplied time budget (default 1000 s) and returns the best sequence
found as a plain list of floats.
"""

import time
import math
from typing import List, Tuple, Optional

import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy – used for the global differential‑evolution search.
try:
    from scipy.optimize import differential_evolution  # type: ignore
    _HAS_DE = True
except Exception:  # pragma: no cover
    _HAS_DE = False

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full a⋆a via real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  ∑_k w_k·conv_k(a,a)  w.r.t. a.
    Implements 2·a⋆w via FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    shift = np.max(z)
    exp_z = np.exp(z - shift)
    s = exp_z.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / len(z))
    return exp_z / s


# ----------------------------------------------------------------------
def _adam_optimize(z0: np.ndarray,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   alpha0: float = 1.0,
                   alpha_max: float = 500.0,
                   growth_factor: float = 1.02,
                   lr: float = 0.1,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   max_iter: int = 200_000,
                   eval_interval: int = 200) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    """
    n = z0.shape[0]
    z = z0.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        a = _softmax(z)                     # a ≥ 0, sum = 1

        conv = _fft_self_convolve(a)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        grad_a = _gradient_from_weights(a, w)

        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        z = z - step

        if t % 500 == 0:
            alpha = min(alpha * growth_factor, alpha_max)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.5,
                                 max_iter: int = 20_000,
                                 eval_interval: int = 200) -> Tuple[np.ndarray, float]:
    """
    Classic multimax sub‑gradient descent (exponentiated‑gradient on the simplex).
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)

        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] multimax it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """
    Greedy pairwise redistribution walk – accepts only improving moves.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Parameterised low‑dimensional family (log‑linear interpolation)
# ----------------------------------------------------------------------
def construct_distribution(params: np.ndarray, n: int) -> np.ndarray:
    """
    Build a probability vector of length ``n`` from ``params``.
    ``params`` are log‑amplitudes at ``M`` equidistant control points;
    values are linearly interpolated and then projected onto the simplex.
    """
    M = len(params)
    positions = np.linspace(0, n - 1, M)
    ctrl_vals = np.exp(params)                     # enforce positivity
    x = np.arange(n, dtype=np.float64)
    y = np.interp(x, positions, ctrl_vals)
    if y.sum() == 0.0:
        y = np.full_like(y, 1.0 / n)
    else:
        y = y / y.sum()
    return y


# ----------------------------------------------------------------------
# Global variables used by the DE objective (no closures!)
# ----------------------------------------------------------------------
_DE_n: Optional[int] = None  # length of the current DE run


def _de_objective(params: np.ndarray) -> float:
    """
    DE objective – uses the global ``_DE_n`` to know the length.
    Returns the exact L∞ objective for the constructed distribution.
    """
    global _DE_n
    n = _DE_n if _DE_n is not None else 0
    a = construct_distribution(params, n)
    return _objective(a)

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function minimizing ``evaluate_sequence``.
    The routine respects the given time budget and returns the best sequence
    (as a plain ``list[float]``).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 0.  Bring in a possible previous solution (if the host defines it)
    # ------------------------------------------------------------------
    overall_best_seq: Optional[np.ndarray] = None
    overall_best_score = float("inf")
    try:
        prev = globals().get("height_sequence_1")
        if isinstance(prev, (list, np.ndarray)):
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.ndim == 1 and prev_arr.size > 0:
                prev_arr = np.maximum(prev_arr, 0.0)
                s = prev_arr.sum()
                if s > 0.0:
                    prev_arr = prev_arr / s
                    sc = _objective(prev_arr)
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = prev_arr.copy()
                        print(f"[{time.time():.1f}] previous best score {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose a handful of lengths (moderate n keeps each iteration cheap)
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000]          # a few moderate sizes

    # ------------------------------------------------------------------
    # 2. Global search with differential evolution (if SciPy is available)
    # ------------------------------------------------------------------
    if _HAS_DE:
        for n in lengths:
            if time.time() >= deadline:
                break
            print(f"[{time.time():.1f}] DE start length {n}")

            # number of control points – more points give more flexibility but
            # increase dimensionality; 8 works well for the given n.
            M = 8
            global _DE_n
            _DE_n = n

            bounds = [(-8.0, 8.0) for _ in range(M)]

            # Use a modest population and iteration count – enough to explore,
            # cheap enough to stay within the budget.
            max_iter = 30
            popsize = 12

            try:
                result = differential_evolution(_de_objective,
                                                bounds,
                                                maxiter=max_iter,
                                                popsize=popsize,
                                                polish=False,
                                                seed=int(rng.integers(1 << 31)),
                                                updating='deferred',
                                                disp=False)
            except Exception as e:          # pragma: no cover
                print(f"[{time.time():.1f}] DE failed for n={n}: {e}")
                continue

            if result.success:
                cand = construct_distribution(result.x, n)
                cand_score = _objective(cand)
                print(f"[{time.time():.1f}] DE length={n} best score {cand_score:.6f}")
                if cand_score < overall_best_score:
                    overall_best_score = cand_score
                    overall_best_seq = cand.copy()

                # ------------------------------------------------------------------
                # 3. Local Adam refinement of the DE best for this length
                # ------------------------------------------------------------------
                remaining = deadline - time.time()
                if remaining > 20.0:
                    # allocate a thin slice (≈10 s) for Adam
                    sub_deadline = time.time() + min(12.0, remaining * 0.6)
                    z0 = np.log(np.maximum(cand, 1e-300))
                    cand_ref, cand_ref_score = _adam_optimize(
                        z0,
                        sub_deadline,
                        best_seq=overall_best_seq,
                        best_score=overall_best_score,
                        alpha0=1.0,
                        alpha_max=2000.0,
                        growth_factor=1.03,
                        lr=0.14,
                        max_iter=200_000,
                        eval_interval=250,
                    )
                    if cand_ref_score < overall_best_score:
                        overall_best_score = cand_ref_score
                        overall_best_seq = cand_ref.copy()
                        print(f"[{time.time():.1f}] after Adam refinement new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. If DE was unavailable or produced nothing, fall back to a few
    #    deterministic seeds (uniform, arcsine, beta‑sym, etc.).
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        print(f"[{time.time():.1f}] Falling back to deterministic seeds.")
        n = lengths[0]
        # simple uniform seed
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = _objective(overall_best_seq)

    # ------------------------------------------------------------------
    # 5. Multimax sub‑gradient polishing (short but effective)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 25.0:
        sub_deadline = time.time() + remaining * 0.6
        overall_best_seq, overall_best_score = _subgradient_multimax_descent(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.6,
            max_iter=30_000,
            eval_interval=300,
        )
        print(f"[{time.time():.1f}] after multimax best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final greedy pairwise‑swap local walk (very cheap)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] after final swaps best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Return the best sequence as a plain Python list
    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
