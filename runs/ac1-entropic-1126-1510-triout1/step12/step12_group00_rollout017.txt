    v = np.zeros_like(a)

            # compute softmax weights
            weighted = alpha * conv
            maxw = weighted.max()
            expw = np.exp(weighted - maxw)  # avoid overflow

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------

1. **Seed generation** – a diverse pool of candidate distributions.
   Analytic families (uniform, arcsine, symmetric beta,
   edge‑weighted, edge‑inverse‑power, sine‑weighted, exponential decay,
   power‑law) are sampled for several lengths, together with random
   Dirichlet and sparse vectors.  The 15 best seeds (lowest exact
   objective) are retained.

2. **Two‑stage optimisation**
   * **Stage A – smooth log‑sum‑exp descent**  
     The non‑smooth objective 2·n·max(conv) is approximated by a
     log‑sum‑exp with a temperature parameter α.
     For a fixed α we minimise
         fα(a)= (1/α)·log∑exp(α·conv(a))
     using Adam with projection onto the simplex.
     α is increased geometrically (0.5 → 80) so that the optimisation
     starts from a smooth landscape and ends at the true max‑objective.
   * **Stage B – active‑set mirror descent**  
     A refined exponentiated‑gradient descent (same idea as the previous
     solution) but with a tighter active set (fraction 0.02, at most 40
     active indices) and slightly larger learning‑rate decay.

3. **Local polishing**
   Greedy pairwise mass transfers followed by a cheap simulated‑annealing
   walk clean up any remaining irregularities.

The whole routine respects the 1000 s time budget, prints progress
messages and finally returns a plain Python list of heights.
"""

import time
import math

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a⋆a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (-p) + j ** (-p))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential_decay(n: int, p: float) -> np.ndarray:
    """Monotone decreasing exponential: w_i ∝ exp(-p·i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-p * i)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    """Power‑law: w_i ∝ (i+1)^{-p}."""
    i = np.arange(n, dtype=np.float64) + 1.0
    w = i ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Stage A – smooth log‑sum‑exp descent (Adam) -------------------------
def _smooth_descent_adam(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    alphas: list[float],
    lr: float = 0.12,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    iters_per_alpha: int = 1200,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the log‑sum‑exp surrogate."""
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_seq, best_score
            # current convolution
            conv = _fft_self_convolve(a)

            # softmax weights (stable)
            weighted = alpha * conv
            maxw = weighted.max()
            expw = np.exp(weighted - maxw)
            p = expw / expw.sum()               # shape (2n‑1,)

            # gradient = 2·corr(p, a)   (valid part only)
            size = 1 << ((len(p) + n - 1).bit_length())
            Fp = np.fft.rfft(p, n=size)
            Fa = np.fft.rfft(a, n=size)
            corr = np.fft.irfft(Fp * np.conj(Fa), n=size)
            start = n - 1
            grad = 2.0 * corr[start : start + n]

            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr / (np.sqrt(v_hat) + eps)
            a = a - step * m_hat

            # project back onto the simplex
            a = _project_onto_simplex(a, s=1.0)

            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                # tighten learning‑rate a little after success
                lr *= 0.97
    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage B – exponentiated‑gradient mirror descent -----------------------
def _subgradient_descent_momentum(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.7,
    active_frac: float = 0.02,
    max_active: int = 40,
    beta: float = 0.9,
    eval_interval: int = 100,
    patience: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent with momentum.
    Uses the set of convolution indices whose value lies within
    ``active_frac`` of the current maximum.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    ind_grid = np.arange(n, dtype=np.int32)

    while time.time() < deadline and t < 250_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            top_idx = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top_idx]

        grad = np.zeros_like(a)
        for idx in act:
            ind = idx - ind_grid
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * a[ind[mask]]

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * momentum)
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                # print progress
                print(f"[{time.time():.1f}] it={t:,} new global best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            jitter = rng.exponential(scale=0.006, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a, s=1.0)
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Deterministic pairwise swap refinement --------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small mass; keep the move only if the objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style local search --------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers random mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def _optimize_sequence(
    seed_seq: np.ndarray,
    deadline: float,
    global_best_seq: np.ndarray,
    global_best_score: float,
) -> tuple[np.ndarray, float]:
    """Run Stage A + Stage B on a single seed."""
    # Stage A – smooth Adam descent
    alphas = [0.5, 2.0, 5.0, 20.0, 80.0]           # temperature schedule
    best_seq, best_score = _smooth_descent_adam(
        seed_seq,
        deadline,
        global_best_seq,
        global_best_score,
        alphas=alphas,
        lr=0.12,
        beta1=0.9,
        beta2=0.999,
        eps=1e-8,
        iters_per_alpha=1200,
    )
    # Stage B – refined mirror descent
    best_seq, best_score = _subgradient_descent_momentum(
        best_seq,
        deadline,
        best_seq,
        best_score,
        lr0=0.6,
        active_frac=0.02,
        max_active=40,
        beta=0.9,
        eval_interval=100,
        patience=1800,
    )
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function respects the supplied time budget (default 1000 s) and returns
    a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0           # safety margin

    # ------------------------------------------------------------------
    # 1. Build a diverse pool of seeds
    length_grid = [2000, 3000, 4000, 6000, 8000]
    p_grid = np.concatenate(
        [
            np.logspace(-3, -1, 9),     # 0.001 … 0.1
            np.logspace(-1, 0, 9),      # 0.1 … 1
            np.logspace(0, 0.5, 7),     # 1 … 3.16
        ]
    )
    lam_grid = np.linspace(0.0, 0.8, 7)

    all_candidates: list[tuple[float, np.ndarray]] = []

    for n in length_grid:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] sampling length {n}")

        # deterministic families (no mixing)
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            all_candidates.append((sc, cand.copy()))

        # parametrised families + mixing with uniform
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            edge2 = _distribution_edge_inversepow(n, p)
            sine = _distribution_sine_weighted(n, p)
            expd = _distribution_exponential_decay(n, p)
            power = _distribution_power_law(n, p)

            for base in (beta, edge, edge2, sine, expd, power):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    all_candidates.append((sc, cand.copy()))

        # random Dirichlet
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = _distribution_dirichlet(n, rng)
            sc = _objective(a)
            all_candidates.append((sc, a.copy()))

        # sparse random vectors
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(a)
            all_candidates.append((sc, a.copy()))

    # Keep the most promising seeds
    all_candidates.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in all_candidates[:15]]
    overall_best_seq = top_seeds[0].copy()
    overall_best_score = _objective(overall_best_seq)
    print(f"[{time.time():.1f}] initial best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Incorporate a previously discovered construction if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_seq, overall_best_score = prev.copy(), sc
                print(f"[{time.time():.1f}] imported previous best {sc:.6f}")
            top_seeds.append(prev.copy())
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Allocate time for optimisation of seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_seeds:
        per_seed = remaining / max(1, len(top_seeds))
        print(
            f"[{time.time():.1f}] running optimisation on {len(top_seeds)} seeds, ≈{per_seed:.1f}s each"
        )
        for idx, seed_seq in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _optimize_sequence(
                seed_seq.copy(),
                sub_deadline,
                overall_best_seq,
                overall_best_score,
            )
            if sc_opt < overall_best_score:
                overall_best_seq, overall_best_score = a_opt.copy(), sc_opt
                print(
                    f"[{time.time():.1f}] seed {idx} produced new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 4. Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] pairwise swap refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            overall_best_seq,
            overall_best_score,
            max_trials=350_000,
        )

    # ------------------------------------------------------------------
    # 5. Final stochastic local search (cheap simulated annealing)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            overall_best_seq,
            overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
