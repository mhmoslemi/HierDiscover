def solve_lp_max_sum(f, M):
    n = f.shape[0]
    # Build A
    A = np.empty((2*n-1, n), dtype=np.float64)
    # fill with zeros + fill each column
    for j in range(n):
        A[j:j+n, j] = f
    # constraints: A @ g <= M
    b = np.full(2*n-1, M, dtype=np.float64)

    c = -np.ones(n)  # minimize -sum g

    # set bounds
    bounds = [(0, None)] * n

    # solve
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        print("LP failed", res.message)
        return None
    g = res.x
    return g

def refine_lp(f, deadline, max_iters=30, t_values=[0.2,0.1,0.05,0.02,0.01,0.005]):

def deterministic_mass_transfer(a, deadline, max_iters=5000):
    n = len(a)
    best = a.copy()
    best_obj = _objective(best)

    while time.time() < deadline and max_iters>0:
        max_iters -= 1
        conv = _fft_self_convolve(best)
        M = conv.max()
        # active set within relative tolerance
        eps = 1e-7 * M  # absolute tolerance small
        active = np.where(conv >= M - eps)[0]
        if active.size == 0:
            active = np.array([np.argmax(conv)])

        # compute grad
        grad = np.zeros(n, dtype=np.float64)
        ind_grid = np.arange(n, dtype=np.int32)
        for k in active:
            ind = k - ind_grid
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * best[ind[mask]]
        # Determine indices
        i_max = np.argmax(grad)
        i_min = np.argmin(grad)
        if grad[i_max] - grad[i_min] < 1e-9:
            # gradient flatten, break
            break
        # move mass
        delta = best[i_max] * 0.3  # move 30% of mass
        if delta < 1e-12:
            break
        new = best.copy()
        new[i_max] -= delta
        new[i_min] += delta
        # ensure nonnegative
        if new[i_max] < -1e-15:
            # cannot move more than mass
            delta = best[i_max]
            new[i_max] = 0
            new[i_min] += delta
        # compute obj
        obj = _objective(new)
        if obj < best_obj:
            best = new
            best_obj = obj

            # maybe reduce delta factor
            # try smaller delta
            for factor in [0.1, 0.05, 0.02]:

                    break

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Seed generation** – a handful of analytic families (uniform,
   arcsine, symmetric beta, edge‑weighted, edge‑inverse‑power, sine‑weighted)
   and a few random Dirichlet / sparse vectors are generated for a
   moderate length `n` (≈2000).  The best seed (lowest exact objective)
   is kept.

2. **LP‑based refinement** – for the current distribution `f`
   we compute `M = max(conv(f,f))`.  A linear program
   `max sum(g)  s.t.  conv(f,g) ≤ M,  g ≥ 0`
   is solved (SciPy + HiGHS).  The optimal `g` is normalised to a
   probability vector and mixed back into `f` via a small step size
   `t`.  Repeating this 10–15 times rapidly drives the peak of the
   self‑convolution down.

3. **Deterministic mass‑transfer** – using the sub‑gradient of the
   max‑convolution (sum of shifted copies of `f` over the active
   indices), probability mass is moved from the index with the largest
   sub‑gradient to the index with the smallest one.  Only moves that
   improve the exact objective are kept.

4. **Final stochastic polishing** – a cheap simulated‑annealing
   routine (the same as in the previous state‑of‑the‑art code) performs
   random mass transfers while the temperature slowly decays.

All steps respect the overall 1000 s time budget, use only NumPy /
SciPy and return a plain Python list of heights.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft
from scipy.optimize import linprog


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    """Second edge family: w_i ∝ (i+0.5)^(−p) + (n−i−0.5)^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (-p) + j ** (-p))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight proportional to sin(π (i+½)/n)^p, vanishing at the ends."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet sample with a random concentration."""
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Linear‑programming step ------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max sum(g)  s.t.  conv(f,g) ≤ M , g ≥ 0 .
    Returns the optimal g (not normalised).  If the LP fails None is
    returned.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    # Build the convolution matrix A (shape rows × n)
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    b = np.full(rows, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # minimise -sum(g)  <=> maximise sum(g)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x


def _lp_refine(f: np.ndarray, deadline: float,
               max_iters: int = 12,
               t_vals: tuple[float, ...] = (0.2, 0.1, 0.05, 0.02, 0.01, 0.005)) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP described above and mix the obtained g back
    into f.  Returns the improved distribution and its objective.
    """
    best = f.copy()
    best_val = _objective(best)
    while time.time() < deadline and max_iters > 0:
        max_iters -= 1
        conv = _fft_self_convolve(best)
        M = conv.max()
        g = _solve_lp_max_sum(best, M)
        if g is None:
            break
        s = g.sum()
        if s <= 0.0:
            break
        g_norm = g / s                      # make it a probability vector
        improved = False
        for t in t_vals:
            new = (1.0 - t) * best + t * g_norm
            val = _objective(new)
            if val < best_val - 1e-14:
                best, best_val = new, val
                improved = True
                break
        if not improved:
            # small random jitter to escape a possible stall
            rng = np.random.default_rng()
            jitter = rng.exponential(scale=1e-4, size=best.shape)
            cand = best + jitter
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            cand_val = _objective(cand)
            if cand_val < best_val:
                best, best_val = cand, cand_val
    return best, best_val

# Deterministic gradient‑based mass transfer ----------------------------
def _mass_transfer_refine(a: np.ndarray, deadline: float,
                         max_iters: int = 5000,
                         active_rel: float = 1e-6,
                         move_frac: float = 0.3) -> tuple[np.ndarray, float]:
    """
    While time permits, move a fraction of the probability mass from the
    index with the largest sub‑gradient to the one with the smallest,
    accepting only improving moves.
    """
    n = a.shape[0]
    cur = a.copy()
    cur_val = _objective(cur)
    ind_grid = np.arange(n, dtype=np.int32)

    while time.time() < deadline and max_iters > 0:
        max_iters -= 1

        conv = _fft_self_convolve(cur)
        M = conv.max()
        eps = max(active_rel * M, 1e-12)
        active = np.where(conv >= M - eps)[0]

        # limit size of active set (keep the highest values only)
        if active.size > 20:
            top = np.argpartition(conv[active], -20)[-20:]
            active = active[top]

        # sub‑gradient for each coordinate
        grad = np.zeros_like(cur)
        for k in active:
            ind = k - ind_grid
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * cur[ind[mask]]

        i_max = int(np.argmax(grad))
        i_min = int(np.argmin(grad))
        if grad[i_max] - grad[i_min] < 1e-12:
            break

        delta = cur[i_max] * move_frac
        if delta < 1e-14:
            break

        new = cur.copy()
        new[i_max] -= delta
        new[i_min] += delta

        # sanity – enforce non‑negativity (numerical errors)
        if new[i_max] < -1e-15:
            continue

        new_val = _objective(new)
        if new_val < cur_val - 1e-14:
            cur, cur_val = new, new_val
        else:
            # try smaller moves before giving up
            for factor in (0.1, 0.05, 0.02):
                d2 = cur[i_max] * factor
                if d2 < 1e-14:
                    continue
                cand = cur.copy()
                cand[i_max] -= d2
                cand[i_min] += d2
                cand_val = _objective(cand)
                if cand_val < cur_val - 1e-14:
                    cur, cur_val = cand, cand_val
                    break
            else:
                # no improvement – stop the deterministic phase
                break
    return cur, cur_val

# Cheap simulated‑annealing local search (re‑used from the previous code)
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            # print progress (optional)
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function respects the supplied time budget (default 1000 s) and
    returns a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 4.0  # safety margin

    # ------------------------------------------------------------------
    # 1. generate a modest set of diverse seeds (fixed length to keep LP cheap)
    n = 2000
    seeds: list[np.ndarray] = [
        _distribution_uniform(n),
        _distribution_arcsine(n),
        _distribution_beta_sym(n, 0.5),
        _distribution_edge_weighted(n, 0.5),
        _distribution_edge_inversepow(n, 0.5),
        _distribution_sine_weighted(n, 2.0),
    ]

    for _ in range(3):
        seeds.append(_distribution_dirichlet(n, rng))
    for _ in range(2):
        seeds.append(_distribution_random_sparse(n, rng, density=0.005))

    # keep the best seed
    best_seq = None
    best_score = float("inf")
    for s in seeds:
        sc = _objective(s)
        if sc < best_score:
            best_score = sc
            best_seq = s.copy()
    if best_seq is None:
        best_seq = _distribution_uniform(n)
        best_score = _objective(best_seq)
    print(f"[{time.time():.1f}] initial best objective {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. LP‑based refinement (takes roughly half the remaining budget)
    now = time.time()
    if now < deadline:
        lp_deadline = now + (deadline - now) * 0.45
        best_seq, best_score = _lp_refine(best_seq, lp_deadline,
                                          max_iters=12,
                                          t_vals=(0.2, 0.1, 0.05, 0.02, 0.01, 0.005))
        print(f"[{time.time():.1f}] after LP refinement best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Deterministic gradient‑based mass transfer
    now = time.time()
    if now < deadline:
        grad_deadline = now + (deadline - now) * 0.35
        best_seq, best_score = _mass_transfer_refine(best_seq, grad_deadline,
                                                    max_iters=5000,
                                                    active_rel=1e-6,
                                                    move_frac=0.3)
        print(f"[{time.time():.1f}] after deterministic transfer best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final cheap simulated‑annealing polishing
    now = time.time()
    if now < deadline:
        best_seq, best_score = _random_local_search(
            best_seq.copy(),
            deadline,
            best_seq=best_seq,
            best_score=best_score,
            init_temp=0.01,
            decay=0.9995,
        )
        print(f"[{time.time():.1f}] final SA best {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best objective: {best_score:.6f} (length={len(best_seq)})")
    # ``evaluate_sequence`` will normalise / clamp internally,
    # so we can return the raw probability vector.
    return best_seq.tolist()
