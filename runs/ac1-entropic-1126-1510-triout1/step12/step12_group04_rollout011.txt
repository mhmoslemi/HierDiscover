"""
Hybrid softmax‑Adam optimizer for the non‑negative step‑function problem.

The core idea:
  • Work on the simplex by representing a = softmax(z) (z∈Rⁿ). This
    guarantees a_i ≥ 0 and Σa_i = 1 automatically.
  • The true objective is 2·n·max(conv(a,a)).  A smooth surrogate
    Sα = (1/α)·log Σ exp(α·conv) is used; its gradient is the
    convex combination of the individual convolution gradients,
    i.e.   ∇ₐS = 2·corr(a, w)  with w_k ∝ exp(α·conv_k).
  • Adam (first‑order adaptive method) is applied to the logit
    variables z.  The learning rate is constant, α is increased
    slowly from 5 to 30 so the optimisation starts smooth and ends
    sharp.
  • A small pool of analytic seeds (uniform, arcsine, power‑inverse‑√,
    two‑block, sparse Dirichlet…) is generated for several lengths.
    The best few seeds are refined with the Adam optimiser.
  • The final best sequence is polished by a cheap greedy mass‑transfer
    walk (identical to the original code).

The routine respects the global time budget (default 1000 s) and
returns the best sequence found.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)² (sum(a)=1 ⇒ denominator=1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine_offset(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = i + c
    right = (n - 1 - i) + c
    vals = 1.0 / np.sqrt(left * right)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _distribution_power_inv_sqrt(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = i + 0.5
    right = (n - 1 - i) + 0.5
    vals = (left * right) ** (-p / 2.0)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Small pool of analytic seeds for a fixed length."""
    cand = []

    cand.append(_distribution_uniform(n))

    # arcsine with a few offsets
    for c in (0.2, 0.5, 1.0, 2.0):
        cand.append(_distribution_arcsine_offset(n, c))

    # power‑inverse‑sqrt family
    for p in (0.8, 1.0, 1.2):
        cand.append(_distribution_power_inv_sqrt(n, p))

    cand.append(_distribution_two_blocks(n, block_frac=0.05))
    cand.append(_distribution_two_blocks(n, block_frac=0.10))

    cand.append(_distribution_random_sparse(n, rng, density=0.01))

    # a couple of Dirichlet draws
    for _ in range(2):
        cand.append(_distribution_dirichlet(n, rng,
                                            alpha=0.3 + 0.7 * rng.random()))
    return cand


# ----------------------------------------------------------------------
def _gradient_cross_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed max w.r.t. a:
        ∇ₐ S = 2 * correlation(a, w)   (w is a probability vector of length 2n‑1)
    The correlation is computed via FFT for O(n log n).
    """
    n = a.shape[0]
    m = w.shape[0]            # = 2·n‑1
    size = _next_pow2(n + m - 1)
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w, n=size)
    corr_full = np.fft.irfft(A * np.conj(W), n=size)  # real‑valued
    start = m - 1          # lag = 0 corresponds to index m‑1
    grad = 2.0 * corr_full[start:start + n]
    return grad

def _optimize_sequence_adam(seed_seq: np.ndarray,
                            deadline: float,
                            lr: float = 0.05,
                            alpha_start: float = 5.0,
                            alpha_end: float = 30.0) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the softmax parametrisation.
    Returns the best vector (as a np.ndarray) and its objective value.
    """
    n = seed_seq.shape[0]
    eps = 1e-12
    # initialise logits
    z = np.log(seed_seq + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    best_seq = seed_seq.copy()
    best_score = _objective(best_seq)
    t = 0
    alpha = alpha_start

    while time.time() < deadline:
        t += 1

        # ---- softmax ----
        a = np.exp(z - np.max(z))
        a /= a.sum()

        # ---- convolution & softmax weights ----
        conv = _fft_self_convolve(a)                     # length 2n‑1
        max_conv = conv.max()
        # linearly increase α, capped by α_end
        if alpha < alpha_end:
            alpha = min(alpha_end, alpha + 0.003)  # ~10 s per unit increase
        w_raw = np.exp(alpha * (conv - max_conv))      # max → 1
        w = w_raw / w_raw.sum()                         # probability vector

        # ---- gradient wrt a ----
        grad_a = _gradient_cross_corr(a, w)

        # ---- gradient wrt logits z (softmax chain rule) ----
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- keep best ----
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Adam iter={t} new best {best_score:.6f}")

        # occasional jitter to escape plateaus
        if t % 2000 == 0:
            z += 0.01 * np.random.randn(*z.shape)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Cheap random mass‑transfer polishing (greedy acceptance)."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue
        delta = a[i] * rng.random() * 0.30  # move up to 30 % of source
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur_score = _objective(new_a)
        if cur_score < best_score - 1e-13:
            a = new_a
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score

def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Main entry point.
    Generates several analytic seeds, runs a softmax‑Adam optimiser on the
    most promising ones and finishes with a greedy polishing step.
    Returns the best sequence (list of floats) found within the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # 1.  Generate a modest pool of analytic seeds for a few lengths.
    lengths = [1800, 2500, 3200, 4000, 5000]   # 5000 is the longest we try
    seed_pool: list[tuple[float, np.ndarray]] = []   # (score, vector)

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] === scanning length {n} ===")
        # analytic families
        for cand in _initial_candidates(n, rng):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            cand_sum = cand.sum()
            if cand_sum == 0.0:
                cand = _distribution_uniform(n)
                cand_sum = cand.sum()
            cand = cand / cand_sum
            sc = _objective(cand)
            seed_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] !!! new global best len={n} score={sc:.6f}")

        # a few extra Dirichlet draws for diversity
        for _ in range(2):
            if time.time() >= deadline:
                break
            cand = _distribution_dirichlet(n, rng, alpha=0.2 + 0.8 * rng.random())
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = _objective(cand)
            seed_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] !!! new global best (extra) len={n} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  If a historic solution exists, incorporate it.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            prev /= prev.sum()
            sc = _objective(prev)
            seed_pool.append((sc, prev.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best improved: {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Keep only the elite seeds (few best ones)
    seed_pool.sort(key=lambda x: x[0])
    elite = seed_pool[:8]   # at most eight seeds

    # ------------------------------------------------------------------
    # 3.  Main optimisation – Adam on the simplex (softmax).
    remaining = deadline - time.time()
    if remaining > 20.0 and elite:
        per_seed = max(5.0, remaining / len(elite))
        for sc, seq in elite:
            if time.time() >= deadline:
                break
            sub_deadline = min(time.time() + per_seed, deadline)
            seq_opt, sc_opt = _optimize_sequence_adam(seq.copy(),
                                                     sub_deadline,
                                                     lr=0.05,
                                                     alpha_start=5.0,
                                                     alpha_end=30.0)
            if sc_opt < overall_best_score:
                overall_best_score = sc_opt
                overall_best_seq = seq_opt.copy()
                print(f"[{time.time():.1f}] after Adam new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Final greedy polishing (mass‑transfer moves).
    if overall_best_seq is not None and time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    if overall_best_seq is None:
        # fallback: uniform distribution of a modest size
        n_fallback = 1000
        overall_best_seq = _distribution_uniform(n_fallback)
        overall_best_score = _objective(overall_best_seq)

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
