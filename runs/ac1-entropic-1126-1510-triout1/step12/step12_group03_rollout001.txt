"""
step_function_optimizer.py
==========================

A fast, self‑contained optimizer for the ``evaluate_sequence`` challenge.
The algorithm consists of three stages:

1. **Rich parametric family scan** – a dense sweep over several analytic
   families (uniform, arcsine, sine, cosine, quadratic, Beta(p,p),
   edge‑weighted, Gaussian‑edge) on a set of lengths.  For the
   edge‑weighted and Beta families we optimise the shape exponent ``p`` and
   a mixing weight ``lam`` with the uniform distribution using a two‑level
   grid (coarse → fine).

2. **Mirror‑descent (exponentiated‑gradient) refinement** – the best few
   seeds are polished with an EG update that uses the true sub‑gradient
   formed from the *k* largest convolution entries (``k`` is larger than
   in the reference solution).  The step‑size decays as ``1/√t`` and the
   iterate is always projected onto the probability simplex.

3. **Greedy pairwise mass transfer** – a hill‑climbing phase that repeatedly
   moves a random fraction of mass from one coordinate to another and keeps
   the move only if the exact objective improves.  It runs until the time
   budget expires.

All helpers are top‑level functions – no closures or lambda use.
The returned list is a plain ``list[float]``; ``evaluate_sequence`` will
perform its own clipping and normalisation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    idx = np.arange(1, len(v) + 1)
    cond = u - cssv / idx > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (fast for large n)."""
    n = a.shape[0]
    # next power of two ≥ 2*n‑1
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1.
    ``a`` must be non‑negative; no safety clipping is performed.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the grid {0,…,n‑1} (p>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use log to avoid overflow/underflow
    logw = (p - 1) * (np.log(x) + np.log1p(-x))
    w = np.exp(logw)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """
    Edge‑weighted family: w_i ∝ ((i+½)*(n‑i‑½))^{‑p}.
    For p>0 the distribution is heavy at the two ends.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_gaussian_edge(n: int, sigma: float) -> np.ndarray:
    """
    Sum of two Gaussian “bumps’’ centred at the two ends.
    ``sigma`` is an absolute standard deviation (in index units).
    """
    i = np.arange(n, dtype=np.float64)
    left = np.exp(- (i ** 2) / (2.0 * sigma ** 2))
    right = np.exp(- ((n - 1 - i) ** 2) / (2.0 * sigma ** 2))
    w = left + right
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base  (lam ∈ [0,1])."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated gradient) -------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    top_k: int = 10,
    eval_every: int = 120,
    max_iter: int = 400_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror descent on the exact max‑convolution objective.
    The gradient uses the ``top_k`` largest convolution entries.
    Returns the best sequence ever observed.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit and t < max_iter:
        t += 1
        conv = _fft_self_convolve(a)
        # indices of the largest entries (including the centre)
        idxs = np.argpartition(-conv, range(top_k))[:top_k]

        grad = np.zeros_like(a)
        for idx in idxs:
            # contribution of a shifted version of itself
            ind = idx - np.arange(n)
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * a[ind[mask]]

        eta = lr0 / math.sqrt(t)          # decaying step size
        a = a * np.exp(-eta * grad)       # EG update (keeps positivity)
        # simplex projection
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / a_sum

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad it={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise polishing --------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from one index to another.
    The move is kept only if it strictly improves the exact objective.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score

        overall_best_score = _objective(overall_best_seq)
        print(
            f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 4. Mirror‑descent refinement on the most promising seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # keep the 5 best seeds for deeper refinement
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] start subgradient descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sg_deadline = time.time() + per_seed
            a_opt, sc_opt = _subgradient_descent(
                seed_seq,
                sg_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                top_k=12,
                eval_every=120,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] after subgrad seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5. Final greedy pairwise polishing (pure hill‑climbing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy swap polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (2·n·max(conv)): {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list – the outer harness will clip/normalise as needed.
    return overall_best_seq.tolist()
