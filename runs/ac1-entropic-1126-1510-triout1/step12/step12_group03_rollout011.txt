```python
def _smooth_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    conv = _fft_self_convolve(a)  # length 2n-1
    # compute weights w = exp(alpha*(conv - conv.max()))  # max weight = 1
    w = np.exp(alpha * (conv - conv.max()))
    # compute gradient via cross-correlation
    # Equivalent to gradient_i = 2 * sum_k w_k a_{k-i}
    # Use convolution of a with reversed w.
    w_rev = w[::-1]
    # Use real FFT of appropriate size.
    size = 1 << ((n + len(w) - 1) - 1).bit_length()
    # but we have shape n and 2n-1 length.
    size = 1 << ((n + (2*n-1) - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w_rev, n=size)
    conv_full = np.fft.irfft(A * W, n=size)
    # conv_full length = size, but we need slice.
    start = len(w) - 1
    grad = 2 * conv_full[start:start + n]
    return grad
```

```python
def _smooth_exponential_gradient_descent(a0, time_limit, best_seq, best_score,
                                         alpha0=0.5, lr0=0.7, max_iter=200000):
    a = a0.copy()
    n = a.shape[0]
    t = 0
    alpha = alpha0
    while time.time() < time_limit and t < max_iter:
        t += 1
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # smoothing weights
        w = np.exp(alpha * (conv - maxc))   # max weight = 1
        # gradient compute
        grad = _gradient_from_weights(a, w)  # define function
        # step size: decaying
        eta = lr0 / math.sqrt(t)
        # EG update via log
        log_a = np.log(np.maximum(a, 1e-30))
        log_a = log_a - eta * grad
        a = np.exp(log_a)
        # project onto simplex
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum
        # possibly increase alpha slowly
        if t % 200 == 0:
            alpha = min(alpha * 1.05, 20.0)  # gradually increase up to 20
        if t % 150 == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smoothgrad it={t} alpha={alpha:.2f} new best {best_score:.6f}")
    return best_seq, best_score
```

```
def _gradient_from_weights(a, w):
    n = a.shape[0]
    w_rev = w[::-1]
    size = 1 << ((n + len(w) - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w_rev, n=size)
    conv_full = np.fft.irfft(A * W, n=size)
    start = len(w) - 1
    grad = 2.0 * conv_full[start:start + n]
    return grad
```

def _targeted_swap(a0, time_limit, best_seq, best_score, max_trials=None):
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    # precompute conv once, but each iteration recalc after accepted move.
    conv = _fft_self_convolve(a)
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        k = conv.argmax()
        # valid source indices
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # compute contributions a[i] * a[k-i] for valid i
        idxs = np.arange(i_min, i_max + 1)
        contrib = a[idxs] * a[k - idxs]
        # source i probabilistically weighted by contrib (higher source)
        if contrib.sum() == 0:
            src = np.random.choice(idxs)
        else:
            src = np.random.choice(idxs, p=contrib / contrib.sum())
        # destination j random (maybe from low probability indices)
        # choose j uniformly from all indices; ensure j != src
        j = np.random.randint(n)
        while j == src:
            j = np.random.randint(n)
        # propose moving a fraction of src to dest
        delta = a[src] * np.random.rand() * 0.4  # up to 40%
        if delta < 1e-15:
            continue
        new_a = a.copy()
        new_a[src] -= delta
        new_a[j] += delta
        # evaluate new objective
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_seq = a.copy()
            best_score = cur
            conv = _fft_self_convolve(a)
            print(f"[{time.time():.1f}] targeted swap src={src}→dst={j} δ={delta:.6g} new best {best_score:.6f}")
    return best_seq, best_score

"""
Propose a non‑negative step‑function (probability vector) that minimizes
evaluate_sequence().  The algorithm works in three phases:

1. **Rich family scan** – deterministic and parametric families (uniform,
   arcsine, sine, cosine, quadratic, Beta, edge‑weighted, sparse random,
   Dirichlet…) are evaluated on several lengths.  The best ~30 seeds are kept.

2. **Smoothed exponential‑gradient descent** – a log‑sum‑exp smoothing of the
   max‑convolution objective yields a dense gradient that focuses on the
   largest convolution entries while remaining differentiable.  Each seed is
   refined with a mirror‑descent step (exponentiated‑gradient) and a
   Euclidean projection onto the simplex.  The smoothing parameter α is
   increased slowly, turning the problem from a smooth approximation to the
   true max‑convolution.

3. **Targeted pairwise polishing** – the current worst convolution index k
   is identified.  Mass is moved from indices that contribute heavily to that
   entry towards random destinations; moves are kept only if they strictly
   improve the exact objective.  This hill‑climbing phase finalises the
   construction.

The routine respects a 1000 s budget, prints progress, and returns a plain
Python list; evaluate_sequence will perform its own normalisation.
"""

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()          # next pow‑2 ≥ 2n‑1
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1.
    The input must be non‑negative; no safety clipping is performed.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on {0,…,n‑1} (p>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """
    Edge‑weighted family: w_i ∝ ((i+½)*(n‑i‑½))^{‑p}.
    For p>0 the distribution is heavy at the two ends.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed objective for a given weight vector w
    (same length as the full convolution a*a).
    grad_i = 2 * Σ_k w_k * a_{k-i}
    computed via a single FFT‑based convolution.
    """
    n = a.shape[0]
    # reverse w for cross‑correlation
    w_rev = w[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w_rev, n=size)
    conv_full = np.fft.irfft(A * W, n=size)
    start = w.shape[0] - 1                      # index where a aligns
    grad = 2.0 * conv_full[start:start + n]
    return grad


def _smooth_exponential_gradient_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    alpha0: float = 0.5,
    lr0: float = 0.7,
    max_iter: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror descent on a log‑sum‑exp smoothing of max(conv).
    A gradually increasing smoothing parameter α moves the problem from a
    smooth approximation to the true max‑convolution.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    alpha = alpha0
    while time.time() < time_limit and t < max_iter:
        t += 1
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # exponential weights – largest entry gets weight 1, others <1
        w = np.exp(alpha * (conv - maxc))
        grad = _gradient_from_weights(a, w)

        # learning rate decays like 1/sqrt(t)
        eta = lr0 / math.sqrt(t)

        # exponentiated‑gradient update (log‑space for stability)
        log_a = np.log(np.maximum(a, 1e-30))
        log_a -= eta * grad
        a = np.exp(log_a)

        # simplex projection (renormalise to sum=1)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= a_sum

        # slowly increase smoothing
        if t % 250 == 0:
            alpha = min(alpha * 1.07, 20.0)

        # occasional reporting
        if t % 120 == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth‑grad it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


def _targeted_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climbing: move mass from indices that heavily contribute
    to the current worst convolution entry to a random destination.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        k = int(conv.argmax())                     # index of the current max
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)

        # contributions a[i] * a[k-i] for all admissible i
        idxs = np.arange(i_min, i_max + 1)
        contrib = a[idxs] * a[k - idxs]

        # source index: weighted by contribution (larger => more harmful)
        if contrib.sum() > 0:
            src = np.random.choice(idxs, p=contrib / contrib.sum())
        else:                                      # degenerate case
            src = np.random.choice(idxs)

        # destination index – uniform random, different from src
        dst = np.random.randint(n)
        while dst == src:
            dst = np.random.randint(n)

        # move a modest fraction of source mass
        delta = a[src] * np.random.rand() * 0.35   # up to 35 %
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_seq = a.copy()
            best_score = cur
            conv = _fft_self_convolve(a)
            print(
                f"[{time.time():.1f}] swap src={src}→dst={dst} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Main entry point.  Executes the three‑phase search described in the
    module docstring and returns the best sequence found as a plain list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # leave a safety margin for the caller that will evaluate our answer
    deadline = start + budget_s - 3.0

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, seq)

    # ------------------------------------------------------------------
    # 1.  Family scan
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 12000, 20000, 30000, 50000]

    p_grid = np.concatenate(
        [
            np.logspace(-2, -0.3, 12),      # 0.01 … 0.5  (edge‑heavy)
            np.linspace(0.5, 2.0, 10),      # moderate
            np.linspace(2.0, 5.0, 6),       # centre‑heavy
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 6)        # mixing with uniform

    atten_grid = np.linspace(30, 150, 7)      # Chebyshev attenuation (dB)
    beta_grid = np.linspace(0.0, 14.0, 8)     # Kaiser beta

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
        ):
            cand = base_fn(n)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # parametric families (Beta & edge‑weighted) with mixture
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta_seq = _distribution_beta_sym(n, p)
            edge_seq = _distribution_edge_weighted(n, p)
            for base in (beta_seq, edge_seq):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # optional SciPy windows
        if _HAS_SCI:
            for atten in atten_grid:
                if time.time() >= deadline:
                    break
                cand = _distribution_chebyshev(n, atten)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(f"[{time.time():.1f}] Cheb len={n} att={atten:.0f} score={sc:.6f}")

            for bet in beta_grid:
                if time.time() >= deadline:
                    break
                cand = _distribution_kaiser(n, bet)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(f"[{time.time():.1f}] Kaiser len={n} beta={bet:.2f} score={sc:.6f}")

        # random Dirichlet seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.01)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

        # keep only the best ~30 candidates overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2.  Include a previously discovered construction if it exists.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback – should never happen.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 5000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(
            f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 4.  Smoothed exponential‑gradient refinement on the top seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # keep the 5 best seeds for deeper work
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] start smoothed gradient descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            gd_deadline = time.time() + per_seed
            seq_opt, sc_opt = _smooth_exponential_gradient_descent(
                seed_seq,
                gd_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                alpha0=0.5,
                lr0=0.7,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = seq_opt.copy()
                print(
                    f"[{time.time():.1f}] after smooth‑grad seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Final targeted pairwise polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final targeted swap polishing")
        seq_opt, sc_opt = _targeted_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = seq_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (2·n·max(conv)): {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list – evaluate_sequence will normalise it.
    return overall_best_seq.tolist()
