    # rev slice conv[ start : start+n ]

    return subset

"""
Propose a non‑negative step‑function (a list of heights) that minimizes
evaluate_sequence().  The search proceeds in three phases:

1.  Scan a diverse collection of analytic families (uniform, arcsine,
    beta‑type, edge‑weighted, sine/cosine windows, Chebyshev/Kaiser, etc.)
    on several lengths.  The best ~30 candidates are retained.

2.  Refine the retained seeds with a *smooth‑max* (entropy‑smoothed) descent.
    We minimise the soft‑maximum of the auto‑convolution
    f*f by an exponentiated‑gradient / projected‑gradient scheme,
    gradually increasing the smoothing parameter β.  This often finds
    lower peaks than the plain sub‑gradient method.

3.  Finish with a cheap greedy pairwise mass‑transfer hill‑climber that
    accepts any move strictly decreasing the exact objective
    2·n·max(conv(f,f)).  All operations respect the simplex
    (non‑negative, sum = 1) constraints, so the final list can be passed
    directly to evaluate_sequence().
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _cross_corr_weighted(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the cross‑correlation Σ_i w_i·shift(a,i) for all i∈[0,n‑1].
    w must have length 2·n‑1 and be already normalised (Σw=1).
    Returns a length‑n vector equal to:
        g[l] = Σ_i w_i·a[i‑l]   (with zero outside the support)
    which is exactly the sub‑gradient of the soft‑max of the auto‑convolution.
    """
    n = a.shape[0]
    L = w.shape[0]               # = 2·n‑1
    size = 1 << ((n + L - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    w_rev = w[::-1]
    W = np.fft.rfft(w_rev, n=size)
    conv = np.fft.irfft(A * W, n=size)      # full convolution a * w_rev
    # slice contains entries L‑n … L‑1 (inclusive) in forward order;
    # the required gradient is the *reversed* slice.
    start = L - n
    slice_vals = conv[start : start + n]
    return slice_vals[::-1]      # reverse to obtain correct ordering


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a (Σa = 1, a ≥ 0)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the grid {0,…,n‑1} (p>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ ((i+½)*(n‑i‑½))^{‑p}  (edge‑heavy for p>0)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_powerlaw_edge(n: int, p: float) -> np.ndarray:
    """Additive power‑law weighting of both ends: w_i ∝ i^{-p}+ (n‑i)^{-p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = i ** (-p) + (n - np.arange(n, dtype=np.float64) - 0.5) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from one index to another.
    A move is kept only if it strictly improves the exact objective.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score

def _smooth_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    beta_initial: float = 0.5,
    beta_max: float = 64.0,
    beta_factor: float = 2.0,
    iter_per_beta: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Entropy‑smoothed max‑descent.
    Uses the soft‑max of the auto‑convolution with a temperature β.
    The gradient is the weighted cross‑correlation of a with the softmax weights.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        beta = min(beta_initial * (beta_factor ** ((t - 1) // iter_per_beta)), beta_max)

        # self‑convolution and exact objective
        conv = _fft_self_convolve(a)
        M = conv.max()
        cur_score = 2.0 * n * M
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] smooth it={t} β={beta:.2f} new best {best_score:.6f}")

        # soft‑max weights (already normalised)
        w = np.exp(beta * (conv - M))
        w /= w.sum()

        # gradient of the smoothed objective
        grad = 2.0 * _cross_corr_weighted(a, w)

        # decaying step size (mirrors EG schedule)
        eta = lr0 / math.sqrt(t)
        a = a - eta * grad

        # Euclidean projection back onto the simplex Σa=1, a≥0
        a = _project_onto_simplex(a)

    return best_seq, best_score

            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(
                    f"[{time.time():.1f}] new global best len={n} score={sc:.6f}"
                )

        # ---- parametric families (beta / edge‑weighted / power‑law) -----
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta_seq = _distribution_beta_sym(n, p)
            edge_seq = _distribution_edge_weighted(n, p)
            power_seq = _distribution_powerlaw_edge(n, p)

            for base in (beta_seq, edge_seq, power_seq):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # ---- optional SciPy windows ------------------------------------
        if _HAS_SCI:
            for atten in atten_grid:
                if time.time() >= deadline:
                    break
                cand = _distribution_chebyshev(n, atten)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(f"[{time.time():.1f}] Cheb len={n} att={atten:.0f} score={sc:.6f}")

            for bet in beta_grid:
                if time.time() >= deadline:
                    break
                cand = _distribution_kaiser(n, bet)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(f"[{time.time():.1f}] Kaiser len={n} beta={bet:.2f} score={sc:.6f}")

        # ---- random Dirichlet seeds ------------------------------------
        for _ in range(4):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} score={sc:.6f}")

        # ---- sparse random seeds ---------------------------------------
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

        # keep only the best ~30 seeds (bounds memory & time)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2.  Incorporate a previously discovered construction, if present.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback – should never happen
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 5000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(
            f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 4.  Smooth‑max descent refinement on the top seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] start smooth descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sg_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_descent(
                seed_seq,
                sg_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.8,
                beta_initial=0.5,
                beta_max=64.0,
                beta_factor=2.0,
                iter_per_beta=200,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] after smooth seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Final greedy pairwise polishing (pure hill‑climbing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy swap polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (2·n·max(conv)): {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list – evaluate_sequence will clip/normalise as needed.
    return overall_best_seq.tolist()
