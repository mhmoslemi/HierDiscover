def _random_walk_local(a0, deadline, best_seq, best_score,
                       init_temp=1e-3, decay=0.9995, min_temp=1e-8,
                       move_frac=0.3,
                       max_iter=200000):
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        src_idx = rng.choice(np.where(a > 1e-12)[0])
        dst_idx = rng.integers(n)
        if dst_idx == src_idx:
            continue
        delta = a[src_idx] * rng.random() * move_frac
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[src_idx] -= delta
        new_a[dst_idx] += delta
        # evaluate
        val = _objective(new_a)
        delta_obj = val - best_score
        if delta_obj < 0 or rng.random() < math.exp(-delta_obj / T):
            a = new_a
            if val < best_score:
                best_score = val
                best_seq = a.copy()
                print(f"[{time.time():.1f}] SA iter={it} new best {best_score:.6f}")
        # schedule temperature
        T = max(T * decay, min_temp)

"""
Hybrid search for a step‑function that minimises the max‑convolution objective.

The routine works in four phases:

1. **Broad seeding** – a large collection of analytic families (uniform,
   arcsine, beta, power‑law, exponential, sinusoidal, triangular, Gaussian,
   random Dirichlet, sparse patterns) is generated for several lengths.
   All candidates are normalised to sum = 1 and the best few (by the exact
   objective) are kept.

2. **Adam smooth‑max optimisation** – each seed is refined using Adam on the
   smooth‑max surrogate  fₐ(a) = (1/α)·log ∑ₖexp(α·convₖ(a)).  The temperature
   α is increased up to a large value (α_max = 2000) so the surrogate converges
   to the true max.  The best result from this stage becomes the current global
   best.

3. **Projected gradient descent on the true max‑convolution** – a classic
   sub‑gradient step is performed on the exact max‐convolution (active‑set
   gradient) with Euclidean projection onto the simplex.  This often lowers
   the bound further than the exponentiated‑gradient variant.

4. **Stochastic local walk (simulated‑annealing style)** – random mass transfers
   between positions are accepted if they improve the objective or with a
   probability depending on a decaying temperature.  This greedy walk can
   escape shallow local minima.

All steps respect the global time budget (default = 1000 s) and constantly
track the best sequence found.  The final sequence (as a plain list of floats)
is returned.
"""

import time
import math
from typing import List, Tuple, Optional

import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the weighted sum Σₖ wₖ·(a*a)[k] w.r.t a.
    The weight vector w must sum to 1.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)²)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Analytic families for seeding
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp(
        (alpha - 1) * np.log(np.clip(x, eps, 1.0))
        + (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0))
    )
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(
    n: int, rng: np.random.Generator, alpha: float = 0.5
) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.minimum(i + 1, n - i)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    vals /= vals.sum()
    return vals

def _generate_candidates_for_length(
    n: int, rng: np.random.Generator
) -> List[np.ndarray]:
    """Return a large diverse list of probability vectors of length n."""
    cand: List[np.ndarray] = []

    # Basic analytic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_sine(n))
    cand.append(_distribution_triangular(n))
    cand.append(_distribution_gaussian(n))

    # Symmetric beta with a fine logarithmic grid
    for p in np.geomspace(0.01, 5.0, num=15):
        cand.append(_distribution_beta_sym(n, p))

    # Random asymmetric beta draws
    for _ in range(10):
        alpha = 10 ** rng.uniform(-3, 1)   # 0.001 … 10
        beta = 10 ** rng.uniform(-3, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # Power‑law family – both increasing and decreasing
    for p in np.linspace(-2.0, 5.0, 18):
        if p >= 0:
            cand.append(_distribution_power_increasing(n, p))
        else:
            inc = _distribution_power_increasing(n, -p)
            cand.append(inc[::-1])

    # Exponential families (log‑space)
    for lam in np.geomspace(5e-4, 2e-2, num=12):
        cand.append(_distribution_exponential_increasing(n, lam))

    # Two‑block spikes with several block fractions
    for bf in (0.02, 0.05, 0.10):
        cand.append(_distribution_two_blocks(n, bf))

    # Random sparse patterns
    for dens in (0.01, 0.02, 0.05):
        cand.append(_distribution_random_sparse(n, rng, density=dens))

    # Dirichlet draws with varied concentration
    for _ in range(12):
        alpha = rng.uniform(0.05, 1.0)
        cand.append(_distribution_dirichlet(n, rng, alpha=alpha))

    # Completely random probability vectors
    for _ in range(8):
        v = rng.random(n)
        v /= v.sum()
        cand.append(v)

    return cand


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _softmax(z: np.ndarray) -> np.ndarray:
    shift = np.max(z)
    exp_z = np.exp(z - shift)
    s = exp_z.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / len(z))
    return exp_z / s


def _adam_optimize(
    z0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    alpha0: float = 1.0,
    alpha_max: float = 2000.0,
    growth_factor: float = 1.05,
    lr: float = 0.15,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    max_iter: int = 400_000,
    eval_interval: int = 250,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    The surrogate is f_α(a) = (1/α)·log Σₖ exp(α·convₖ(a)).
    """
    n = z0.shape[0]
    z = z0.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        a = _softmax(z)                     # a ∈ simplex

        conv = _fft_self_convolve(a)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        grad_a = _gradient_from_weights(a, w)          # ∇_a smooth‑max
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)                    # chain rule through softmax

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        z = z - step

        if t % 500 == 0:
            alpha = min(alpha * growth_factor, alpha_max)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}"
                )

    return best_seq, best_score


def _projected_gradient_descent_maxconv(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.5,
    decay: float = 0.95,
    min_eta: float = 1e-4,
    eps_active: float = 1e-12,
    max_iter: int = 20_000,
    eval_interval: int = 200,
) -> Tuple[np.ndarray, float]:
    """
    Euclidean projected gradient descent on the true objective
    2·n·max(conv(a,a)).  The sub‑gradient is taken as the average of the
    gradients of all active (max‑attaining) convolution indices.
    """
    a = a0.copy()
    n = a.shape[0]
    eta = eta0
    it = 0
    while it < max_iter and time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        M = conv.max()
        mask = conv >= M - eps_active
        cnt = mask.sum()
        w = mask.astype(np.float64) / max(1, cnt)
        grad = _gradient_from_weights(a, w)   # 2·a⋆w
        a_new = a - eta * grad
        a_new = _project_onto_simplex(a_new, s=1.0)

        if it % eval_interval == 0:
            cur = _objective(a_new)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a_new.copy()
                print(
                    f"[{time.time():.1f}] ProjGD it={it} η={eta:.5f} new best {best_score:.6f}"
                )
            # simple adaptive step size
            if cur < best_score:
                pass
            else:
                eta = max(eta * decay, min_eta)

        a = a_new

    return best_seq, best_score


def _random_walk_local(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.01,
    decay: float = 0.999,
    min_temp: float = 1e-9,
    move_frac: float = 0.3,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing style walk.
    Randomly moves a fraction of mass from a source index to a destination.
    Accepts moves that improve the objective or with probability exp(-Δ/T).
    """
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    cur_obj = _objective(a)
    T = init_temp
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        # choose source with non‑zero mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * move_frac
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        new_obj = _objective(new_a)
        d_obj = new_obj - cur_obj

        accept = False
        if d_obj < 0:
            accept = True
        else:
            prob = math.exp(-d_obj / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            cur_obj = new_obj
            if new_obj < best_score - 1e-13:
                best_score = new_obj
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] SA it={it} Δ={d_obj:.3e} T={T:.3e} new best {best_score:.6f}"
                )

        T = max(T * decay, min_temp)

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function minimizing ``evaluate_sequence``.
    The routine respects the supplied time budget (default 1000 s) and
    returns the best sequence found as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # 1.  Choose a handful of lengths (moderate n – cheap per‑iteration cost)
    lengths = [2000, 4000, 8000, 12000]

    overall_best_seq: Optional[np.ndarray] = None
    overall_best_score = float("inf")
    all_candidates: List[Tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 2.  Generate and evaluate a broad set of seeds
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Generating seeds for length {n}")

        cand_list = _generate_candidates_for_length(n, rng)
        for cand in cand_list:
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                continue
            cand = cand / s
            sc = _objective(cand)
            all_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(
                    f"[{time.time():.1f}] **** new global best len={n} score={sc:.6f}"
                )

    # ------------------------------------------------------------------
    # 3.  Include a previous solution if available (height_sequence_1)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s == 0.0:
                prev = _distribution_uniform(prev.shape[0])
                s = prev.sum()
            prev = prev / s
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best improved: {sc:.6f}")
            all_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4.  Fallback – should never happen
    if overall_best_seq is None:
        n = lengths[0]
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(
            f"[{time.time():.1f}] fallback uniform seq (len={n}) score={overall_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 5.  Keep only the most promising seeds (across all lengths)
    all_candidates.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in all_candidates[:12]]

    # ------------------------------------------------------------------
    # 6.  Adam smooth‑max optimisation on the top seeds
    remaining = deadline - time.time()
    if remaining > 30.0:
        per_seed = max(10.0, remaining / max(1, len(top_seeds)))
        for seed_seq in top_seeds:
            if time.time() >= deadline:
                break
            slice_deadline = time.time() + per_seed
            z0 = np.log(np.maximum(seed_seq, 1e-300))
            seq_out, score_out = _adam_optimize(
                z0,
                slice_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                alpha0=1.0,
                alpha_max=2000.0,
                growth_factor=1.05,
                lr=0.15,
                max_iter=300_000,
                eval_interval=250,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(
                    f"[{time.time():.1f}] after Adam new best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 7.  Projected gradient descent on the exact max‑convolution
    remaining = deadline - time.time()
    if remaining > 20.0:
        proj_deadline = time.time() + remaining * 0.6
        overall_best_seq, overall_best_score = _projected_gradient_descent_maxconv(
            overall_best_seq.copy(),
            proj_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.5,
            decay=0.95,
            min_eta=1e-4,
            max_iter=30_000,
            eval_interval=250,
        )
        print(
            f"[{time.time():.1f}] after ProjGD best {overall_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 8.  Final stochastic local walk / simulated‑annealing
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _random_walk_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.02,
            decay=0.998,
            move_frac=0.35,
            max_iter=200_000,
        )
        print(
            f"[{time.time():.1f}] after final SA best {overall_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
