"""
Fast hill‑climbing optimiser for the self‑convolution max‑ratio problem.

*   Starts from a few analytically good seeds (the previous best
    `height_sequence_1`, arcsine law, uniform distribution).
*   The sum of the vector is kept fixed to 1, therefore the objective is
    `score = 2·n·max(conv(a,a))`.
*   The main loop performs millions of cheap “mass‑transfer” proposals.
    For a move   i → j   (transfer δ from a[i] to a[j]) only three
    convolution entries (2·i, i+j, 2·j) are affected.
    Using the exact quadratic update formulas the new maximal entry can be
    evaluated in O(1) time, so we can accept/reject a proposal without a full
    FFT.  When a move improves the objective we recompute the full convolution
    (cost O(n log n)) to keep the state consistent.
*   The source index i is drawn from the set of positions that actually
    contribute to the current maximal convolution entry – this greatly
    increases the acceptance rate.
*   The algorithm runs until the time budget is exhausted and returns the
    best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

# ----------------------------------------------------------------------
def _conv_stats(conv: np.ndarray):
    """Return (max_val, max_count, second_max)."""
    max_val = conv.max()
    # tolerance for equality (double precision)
    eps = 1e-15
    max_mask = np.abs(conv - max_val) <= eps
    max_count = int(max_mask.sum())
    if max_count == conv.size:
        second_max = 0.0
    else:
        # mask out the maximal entries and take the max of the rest
        tmp = np.where(max_mask, -np.inf, conv)
        second_max = float(tmp.max())
    return max_val, max_count, second_max

# ----------------------------------------------------------------------
def _initial_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Generate a small diversified set of starting vectors (sum = 1)."""
    seeds = []

    # 1. previous best, if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size == n:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev[:] = 1.0 / n
            else:
                prev /= prev.sum()
            seeds.append(prev.copy())
    except Exception:
        pass

    # 2. arcsine distribution (known good shape)
    i = np.arange(n, dtype=np.float64)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    arcs /= arcs.sum()
    seeds.append(arcs)

    # 3. uniform distribution (baseline)
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 4. a random Dirichlet draw (adds diversity)
    dirich = rng.gamma(0.5, size=n)
    dirich /= dirich.sum()
    seeds.append(dirich)

    # keep only unique ones (by sum)
    uniq = []
    for s in seeds:
        if not any(np.allclose(s, u) for u in uniq):
            uniq.append(s)
    return uniq

# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray,
                rng: np.random.Generator,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform the cheap O(1) mass‑transfer hill‑climbing on vector a.
    Returns the best vector and its score (2·n·max(conv)).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_val, max_count, second_max = _conv_stats(conv)

    best_seq = a.copy()
    best_score = 2.0 * n * max_val

    # tolerance used when comparing floating numbers
    eq_eps = 1e-15
    # minimal delta to avoid degenerate moves
    min_delta = 1e-12

    # schedule for the maximal transferred fraction (starts large, decays)
    start_frac = 0.30
    end_frac = 0.03

    total_iters = 0
    accepted = 0

    while time.time() < deadline:
        total_iters += 1

        # --- pick a source index that actually participates in the max entry ---
        # the maximal convolution index k_max:
        k_max = int(conv.argmax())
        # all i such that 0≤i<n and 0≤k_max-i<n
        i_low = max(0, k_max - (n - 1))
        i_high = min(k_max, n - 1)
        if i_low > i_high:
            # should not happen, but fall back to a uniform pick
            i = rng.integers(n)
        else:
            # weighted pick proportional to current mass (helps focus on heavy contributors)
            sub_range = np.arange(i_low, i_high + 1, dtype=np.int64)
            weights = a[sub_range]
            wsum = weights.sum()
            if wsum < 1e-15:
                i = rng.integers(n)
            else:
                probs = weights / wsum
                i = rng.choice(sub_range, p=probs)

        if a[i] < min_delta:
            continue

        # pick a destination distinct from i
        j = rng.integers(n)
        if j == i:
            continue

        # linearly decreasing maximal transfer fraction
        t = total_iters
        frac = start_frac + (end_frac - start_frac) * min(1.0, t / 200_000.0)

        delta = a[i] * rng.random() * frac
        if delta < min_delta:
            continue

        # indices affected in the convolution
        k1 = 2 * i
        k3 = 2 * j
        k2 = i + j

        ai = a[i]
        aj = a[j]

        # new values of the three affected entries
        new_k1 = conv[k1] - 2.0 * ai * delta + delta * delta
        new_k3 = conv[k3] + 2.0 * aj * delta + delta * delta
        if i != j:
            new_k2 = conv[k2] + 2.0 * ai * delta - 2.0 * aj * delta - 2.0 * delta * delta
        else:          # i == j cannot happen because we skip j==i, but keep for safety
            new_k2 = conv[k2]

        # compute the best possible overall maximum after the move
        new_local_max = max(new_k1, new_k2, new_k3)

        # How many of the current maximal entries would disappear?
        # Count how many max entries lie inside the changed set.
        changed = {k1, k2, k3}
        removed = sum(1 for idx in changed if abs(conv[idx] - max_val) <= eq_eps)

        # If at least one maximal entry stays untouched, the fallback stays max_val.
        if max_count > removed:
            fallback = max_val
        else:
            fallback = second_max

        new_overall_max = max(new_local_max, fallback)

        # accept only strictly improving moves
        if new_overall_max < max_val - 1e-16:
            # --- accept ---
            a[i] -= delta
            a[j] += delta
            # renormalise occasionally to kill accumulated floating errors
            if accepted % 500 == 0:
                a /= a.sum()

            # recompute full convolution (costly, but happens rarely)
            conv = _fft_self_convolve(a)
            max_val, max_count, second_max = _conv_stats(conv)

            accepted += 1
            cur_score = 2.0 * n * max_val
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] it={total_iters} accepted={accepted} "
                      f"new best {best_score:.6f}")

        # otherwise simply reject – state unchanged

    return best_seq, best_score

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Entry point for the checker.
    Generates a few high‑quality seeds, then runs a fast hill‑climbing
    optimiser which evaluates millions of cheap mass‑transfer moves.
    Returns the best sequence found (list of non‑negative floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Determine a reasonable length.
    # Prefer the length of the previously submitted sequence, otherwise 4000.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        n = int(prev.size) if prev.ndim == 1 and prev.size > 0 else 4000
    except Exception:
        n = 4000

    # ------------------------------------------------------------------
    # Build a small pool of seeds (all normalised to sum = 1).
    seeds = _initial_seeds(n, rng)

    overall_best_seq = None
    overall_best_score = float("inf")

    # allocate roughly equal time to each seed
    per_seed_deadline = deadline
    if len(seeds) > 1:
        per_seed_deadline = deadline - time.time() - 0.5
        per_seed_deadline = time.time() + per_seed_deadline / len(seeds)

    for seed_vec in seeds:
        if time.time() >= deadline:
            break
        seq, sc = _hill_climb(seed_vec.copy(), rng, per_seed_deadline)
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = seq.copy()
            print(f"[{time.time():.1f}] seed finished – new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never happen
    if overall_best_seq is None:
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
