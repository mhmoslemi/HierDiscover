"""
Propose a non‑negative step function that (almost) minimizes
``evaluate_sequence``.

The search consists of five stages:

1.  Build a diverse pool of “seed” probability vectors from many analytic
    families (uniform, arcsine, beta, edge‑weighted, Hann, generalized‑normal,
    power‑law, random Dirichlet, sparse).  The best 30 seeds are retained.

2.  Refine the top seeds with exact exponentiated‑gradient descent on the true
    objective (the gradient is the sub‑gradient of the active convolution entry).

3.  Apply a Frank‑Wolfe style refinement that moves mass from the coordinate
    with the *largest* gradient to the coordinate with the *smallest* gradient.
    This monotone update often squeezes out the remaining slack in the max‑convolution.

4.  If time permits, up‑sample the current best distribution (linear
    interpolation while staying on the simplex) and run a short burst of gradient
    descent on the larger vector.

5.  Finish with a long‑run pairwise‑mass‑transfer annealing that accepts
    improving moves always and occasionally accepts worsening moves early on.

All work respects a 1000 s total budget (2 s safety margin).  The routine
returns a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

# ----------------------------------------------------------------------
# Simplex projection (used only for safety)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, linear convolution)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective used by ``evaluate_sequence`` (scale‑invariant).
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return the exact value of ``evaluate_sequence`` for a (unnormalised)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic families – always return a probability vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{-p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(-|x-1/2|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    """Monotone decreasing power‑law (p>0)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.01) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Gradient of the *active* convolution entry (sub‑gradient of max‑conv)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Compute the sub‑gradient of the currently maximal convolution entry.
    Returns (grad, active_index).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))

    grad = np.zeros_like(a)
    if k < n:
        # overlap length = k+1
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        # k >= n : overlap length = 2·n-1-k
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent (multiplicative mirror descent)
# ----------------------------------------------------------------------
def _exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """Perform exponentiated‑gradient descent on the exact objective."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Frank‑Wolfe style refinement (pairwise mass transfer guided by the gradient)
# ----------------------------------------------------------------------
def _frank_wolfe_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """
    Simple Frank‑Wolfe iteration: move a fraction of the mass towards the
    coordinate with the smallest gradient (i.e. the least harmful direction).
    This respects the simplex constraints by construction.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        i_min = int(np.argmin(grad))          # best vertex of the simplex
        # FW direction: e_i_min - a
        gamma = 2.0 / (t + 2.0)                # classic diminishing step size
        a = a + gamma * ((np.eye(1, n, i_min)[0]) - a)  # stay on simplex

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] FW it={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style pairwise mass transfers (final polishing)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins; accept if improves
    or with a temperature‑dependent probability."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling while preserving the simplex (used after a good candidate)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the total 1000 s budget and returns a plain list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Populate a diverse pool of seeds.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000]   # moderate sizes – longer later
    p_grid = np.concatenate(
        [
            np.logspace(-2.0, -0.5, 12, base=10),
            np.logspace(-0.5, 0.0, 12, base=10),
            np.logspace(0.0, 0.7, 10, base=10),
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 5)
    q_grid = np.linspace(0.5, 3.0, 6)
    power_grid = np.linspace(0.2, 2.5, 7)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families (uniform, arcsine, Hann)
        for cand in (_distribution_uniform(n), _distribution_arcsine(n), _distribution_hann(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # beta and edge‑weighted families
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            for base in (beta, edge):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(f"[{time.time():.1f}] new best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # generalized normal shapes
        for q in q_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_generalized_normal(n, q)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best gn q={q:.2f} len={n} score={sc:.6f}")

        # power‑law families (monotone decreasing)
        for pwr in power_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_power_law(n, pwr)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best powerlaw p={pwr:.2f} len={n} score={sc:.6f}")

        # random Dirichlet seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.01)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best sparse len={n} score={sc:.6f}")

        # keep only the best few (30)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2.  Include a previously discovered construction if available.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Guarantee a non‑empty result.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Exact exponentiated‑gradient descent on a few top seeds.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seed_seqs = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining * 0.35 / max(1, len(seed_seqs))   # ~35 % of remaining time
        print(f"[{time.time():.1f}] EG descent on {len(seed_seqs)} seeds")
        for idx, seed_seq in enumerate(seed_seqs):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _exp_gradient_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.9,
                eval_every=80,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] after EG seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Frank‑Wolfe refinement on the current best (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fw_time = (deadline - time.time()) * 0.20   # ~20 % of remaining time
        print(f"[{time.time():.1f}] Frank‑Wolfe refinement (budget {fw_time:.1f}s)")
        sub_deadline = time.time() + fw_time
        overall_best_seq, overall_best_score = _frank_wolfe_descent(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eval_every=100,
        )
        print(f"[{time.time():.1f}] after FW best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Upsample to a larger length and short gradient burst.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n <= 64000:
            print(f"[{time.time():.1f}] upsampling from {cur_n} → {target_n}")
            up_seq = _upsample(overall_best_seq, target_n)
            up_score = _objective(up_seq)
            if up_score < overall_best_score:
                overall_best_score = up_score
                overall_best_seq = up_seq
                print(f"[{time.time():.1f}] upsampled new best {overall_best_score:.6f}")

            # brief EG burst on upsampled vector
            sub_deadline = time.time() + (deadline - time.time()) * 0.15
            a_opt, sc_opt = _exp_gradient_descent(
                overall_best_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.7,
                eval_every=60,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] GD after upsample new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Final long‑run annealing / polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing polish")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
            max_trials=None,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after annealing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)/sum²): {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
