"""
Hybrid grid‑search + Adam sub‑gradient optimizer for the
step‑function “low‑autocorrelation” problem.

1.  A dense family of analytic shapes a_i ∝ ((i+c)*(n-1-i+c))^{-p/2}
    (arcsine with offset and a tunable exponent) is scanned for several
    lengths.  The exact objective 2·n·max(conv(a,a)) (sum(a)=1) is computed
    via a real FFT.  The best few seeds are kept.

2.  Each seed is refined by a projected Adam optimizer that uses a
    multi‑peak sub‑gradient: all convolution entries whose value is within
    a small relative tolerance of the current maximum are aggregated.
    After the additive Adam step the vector is projected onto the simplex
    (non‑negative, unit sum) to guarantee validity.

3.  A cheap greedy “mass‑transfer” polish (random δ‑moves) is applied as
    a final local search.

The function `propose_candidate` respects the overall time budget
(`budget_s`, default 1000 s) and returns the best sequence it found as a
plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (assume sum(a)=1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    # sum is forced to 1 in the calling code – guard against degenerate case
    if max_conv < 0:
        return float("inf")
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
def _distribution_power_offset(n: int, c: float, p: float) -> np.ndarray:
    """Family a_i ∝ ((i+c)*(n‑1‑i+c))^{-p/2}."""
    i = np.arange(n, dtype=np.float64)
    left = i + c
    right = (n - 1 - i) + c
    vals = (left * right) ** (-p / 2.0)
    # protect against overflow / nan
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return vals / total


# ----------------------------------------------------------------------
def _multi_peak_gradient(a: np.ndarray, eps_rel: float = 1e-6) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) obtained by averaging the
    gradients of all convolution entries whose value is within
    eps_rel·max of the maximum.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_c = conv.max()
    # indices that are “almost maximal”
    mask = conv >= max_c - eps_rel * max_c
    ks = np.nonzero(mask)[0]
    if ks.size == 0:                # should never happen
        return np.zeros_like(a)
    grad = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1)
        grad[idx_i] += 2.0 * a[k - idx_i]      # g_i = 2·a_{k‑i}
    grad /= ks.size
    return grad


# ----------------------------------------------------------------------
def _adam_subgrad_descent(a0: np.ndarray,
                          deadline: float,
                          best_seq: np.ndarray,
                          best_score: float,
                          lr: float = 0.10,
                          beta1: float = 0.90,
                          beta2: float = 0.999,
                          eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser that minimises max(conv(a,a)).
    Uses the multi‑peak sub‑gradient defined above.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    while time.time() < deadline:
        t += 1
        grad = _multi_peak_gradient(a, eps_rel=1e-6)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)          # enforce non‑negativity
        a = _project_onto_simplex(a)    # enforce sum=1

        cur_score = _objective(a)
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Adam iter={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Greedy random mass‑transfer polish (accept‑only‑if‑improve)."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue
        # move a random fraction (up to 30 %) of the source mass
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur_score = _objective(new_a)
        if cur_score < best_score - 1e-13:
            a = new_a
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}→{j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score

    # ------------------------------------------------------------------
    # 3.  Refine the elite seeds with Adam sub‑gradient descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0 and top_candidates:
        # allocate roughly equal time per seed
        top_candidates.sort(key=lambda x: x[0])
        seeds_to_use = [seq for _, seq in top_candidates[:8]]
        per_seed = max(0.5, remaining / len(seeds_to_use))
        for seed_seq in seeds_to_use:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            seq_out, score_out = _adam_subgrad_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr=0.10,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after Adam new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Final greedy polishing (random mass transfers)
    # ------------------------------------------------------------------
    if overall_best_seq is not None and time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    # 5.  Return the result
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        # fallback – uniform of moderate size
        n_fallback = 2000
        overall_best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
