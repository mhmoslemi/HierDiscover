"""
Propose a non‑negative step function (list of heights) that minimises
`evaluate_sequence`.

The algorithm
--------------

1. **Build a small but diverse seed pool**.
   - Lengths 6000, 12000 and 18000.
   - Analytic families (uniform, arcsine, symmetric beta,

     logarithmically spaced shape parameters p∈[1e‑4,10] and a few
     convex mixtures with the uniform distribution.
   - Random Dirichlet and sparse Dirichlet samples.

2. **Refine each promising seed** with a *projected gradient descent*.
   - The objective for a normalised vector (∑a=1) is
     `F(a)=2·n·max(conv(a,a))`.
   - In every iteration only the indices whose convolution value is
     within a fraction `active_frac` of the current maximum are kept.
     The gradient for those “active” indices is computed exactly
     (`grad_i = 2·∑_{k∈active} a_{k‑i}`) and a momentum term is added.
   - An additive update `a ← a – η·momentum` is performed followed by
     Euclidean projection onto the probability simplex.
   - `active_frac` is annealed from 0.08 to 0.02, the learning rate
     decays as `η₀/√t`, and a small exponential jitter is injected
     after a long stagnation period.

3. **Heavy‑weight local search**.
   - Pairwise mass‑transfer swaps.
   - A cheap simulated‑annealing style walk that moves random
     fractions of mass between two positions, accepting uphill moves
     with a slowly decaying temperature.

The whole routine respects the supplied time budget (default 1000 s)
and returns a plain Python list; `evaluate_sequence` will take care of
the final normalisation and clipping.

The implementation is self‑contained, uses only NumPy and the
standard library, and makes no file or network accesses.

import time
import math
import numpy as np
from numpy.fft import rfft, irfft


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1.
    The implementation works for any non‑negative a; it normalises to
    sum 1 internally for consistency.
    """
    s = a.sum()
    if s == 0.0:
        return float("inf")
    a = a / s
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (-p) + j ** (-p))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Projected gradient descent (additive, momentum, active set) -------
def _refine_with_active_gradient(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    beta: float = 0.9,
    max_active: int = 50,
    patience: int = 1200,
) -> tuple[np.ndarray, float]:
    """
    Refine vector a (already normalised to sum 1) by projected gradient
    descent using only the active convolution entries.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    ind_grid = np.arange(n, dtype=np.int32)

    # dynamic schedule for the active fraction
    def active_fraction(it):
        if it < 4000:
            return 0.08 - 0.06 * (it / 4000.0)   # 0.08 → 0.02
        return 0.02

    while time.time() < deadline and t < 200_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()
        act_frac = active_fraction(t)
        thresh = max_c * (1.0 - act_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            # keep the top‑max_active entries
            top_idx = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top_idx]

        # gradient for the active set
        grad = np.zeros_like(a)
        for k in act:
            shift = k - ind_grid
            mask = (shift >= 0) & (shift < n)
            grad[mask] += 2.0 * a[shift[mask]]

        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size
        eta = eta0 / math.sqrt(t)

        # additive update + projection onto simplex
        a_new = a - eta * momentum
        a = _project_onto_simplex(a_new, s=1.0)

        # occasional evaluation
        if t % 150 == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] iter={t:,} improved to {best_score:.6f}")
            else:
                no_improve += 1

        # jitter to escape a plateau
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.006, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a, s=1.0)
            no_improve = 0

    return best_seq, best_score

# Pairwise mass‑transfer greedy refinement -----------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Move a random fraction of mass from one index to another if it improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score

# Simulated‑annealing style local search -------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers random mass between two indices."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            # metropolis acceptance
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 5. Final cheap simulated‑annealing style local search
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final random‑local search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list; evaluate_sequence will handle any final clipping.
    return overall_best_seq.tolist()
