"""
Hybrid LP‑gradient optimiser for the self‑convolution max‑ratio problem.

The algorithm works in three stages:

1️⃣  Seed generation – low‑dimensional analytic families (uniform, arcsine,
    sine, two‑block, Dirichlet, sparse, beta, power‑law…) are created for a
    few modest lengths (2000‑4000).  All seeds are normalised to sum = 1 and
    their exact objective `2·n·max(conv(a,a))` is evaluated.  The best few
    seeds are kept as *elite* candidates.

2️⃣  Refinement – each elite seed is first polished by a short hard‑max
    (exponentiated‑gradient) descent.  Afterwards a linear‑programming step
    is applied: with the current sequence `f` and its maximal convolution
    value `M = max(f*f)` we maximise `∑g_j` under the linear constraints
    `f * g ≤ M`.  Only the `top_k` most critical convolution constraints
    (those where `f*f` is close to `M`) are kept, which yields a tiny sparse
    LP (≈ 200 × n).  The LP solution `g` is normalised to a probability vector,
    mixed with the current sequence (`mix = (1‑t)·f + t·g`) and the mixture is
    briefly re‑optimised by a few more hard‑max steps.  If the objective
    improves we accept the new sequence and repeat the LP‑mix cycle.

3️⃣  Final polishing – a cheap random‑mass‑transfer walk (pairwise swaps)
    runs until the time budget expires.

All operations respect the global time budget (default 1000 s).  The routine
returns the best non‑negative step‑function found as a plain list of floats.
"""

import time, math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Basic utilities -------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (sum assumed >0)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    m = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * m / (s * s)


def _gradient_max_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the single maximal convolution entry `conv[k]`.
    Returns a non‑negative vector of length len(a).
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i = np.arange(i_min, i_max + 1, dtype=np.int64)
    j = k - i

    eq = (i == j)
    if np.any(eq):
        g[i[eq]] += 2.0 * a[i[eq]]

    ne = ~eq
    if np.any(ne):
        ii = i[ne]
        jj = j[ne]
        g[ii] += a[jj]
        g[jj] += a[ii]

    return g


# ----------------------------------------------------------------------
# Hard‑max (exponentiated‑gradient) refinement ---------------------------
def _max_grad_descent(a: np.ndarray,
                      deadline: float,
                      best_seq: np.ndarray,
                      best_score: float,
                      lr0: float = 0.5,
                      max_iter: int = 60000,
                      eval_interval: int = 500) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true maximal convolution entry.
    Uses an exponentiated‑gradient update with a decaying step size.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())
        grad = _gradient_max_index(a, k)
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hard‑max it={t} idx={k} new best {best_score:.6f}")

    return best_seq, best_score


def _hardmax_refine(a: np.ndarray,
                    steps: int = 2000,
                    lr0: float = 0.4) -> np.ndarray:
    """
    A lightweight hard‑max refinement (no logging, no best‑track).
    Returns the refined vector, still normalised to sum = 1.
    """
    n = a.shape[0]
    a = a.copy()
    for t in range(1, steps + 1):
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())
        grad = _gradient_max_index(a, k)
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s
    return a


# ----------------------------------------------------------------------
# Random local polishing ------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Cheap random mass‑transfer walk (pairwise swaps)."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue

        delta = a[i] * rng.random() * 0.30          # move up to 30 % of source mass
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Analytic families ------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * x / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int,
                            rng: np.random.Generator,
                            alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int,
                                rng: np.random.Generator,
                                density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_beta_asym(n: int,
                           alpha: float,
                           beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0))
               + (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Linear‑programming step (active‑set) -----------------------------------
def _lp_step_active(a: np.ndarray,
                    M: float,
                    top_k: int = 200) -> np.ndarray | None:
    """
    Solve  max ∑ g_j   subject to   a * g ≤ M   and g ≥ 0,
    but keep only the `top_k` most critical convolution constraints.
    Returns the raw LP solution (not normalised) or None on failure.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)

    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(conv, -top_k)[-top_k:]

    rows = []
    cols = []
    data = []

    for row_i, k in enumerate(idxs):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        js = np.arange(i_min, i_max + 1, dtype=np.int64)
        coeffs = a[k - js]                     # a_{k‑j}
        rows.extend([row_i] * len(js))
        cols.extend(js.tolist())
        data.extend(coeffs.tolist())

    A = csr_matrix((data, (rows, cols)), shape=(len(idxs), n))
    b = np.full(len(idxs), M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise sum(g) → minimise -sum(g)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None
    return res.x

# Main entry point -------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Hybrid LP‑gradient optimiser.  Returns a list of non‑negative heights.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Generate a broad pool of analytic seeds for modest lengths.
    lengths = [2000, 3000, 4000]
    all_candidates = []

    for n in lengths:
        if time.time() >= deadline:
            break

        # deterministic families
        fams = [
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_sine(n),
            _distribution_two_blocks(n, block_frac=0.07),
        ]
        # random Dirichlet draws
        for a in (0.1, 0.4, 0.8):
            fams.append(_distribution_dirichlet(n, rng, alpha=a))
        # sparse random draws
        for dens in (0.005, 0.01, 0.02):
            fams.append(_distribution_random_sparse(n, rng, density=dens))
        # asymmetric beta draws
        for _ in range(4):
            alpha = 10 ** rng.uniform(-2, 1)
            beta = 10 ** rng.uniform(-2, 1)
            fams.append(_distribution_beta_asym(n, alpha, beta))

        for cand in fams:
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            if cand.sum() == 0.0:
                cand = np.full_like(cand, 1.0 / n)
            cand /= cand.sum()
            sc = _objective(cand)
            all_candidates.append((sc, cand.copy()))
            # keep a very short log
            print(f"[{time.time():.1f}] seed len={n} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣b  Incorporate a previously found construction if available.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            prev /= prev.sum()
            sc = _objective(prev)
            all_candidates.append((sc, prev.copy()))
            print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass  # no previous sequence present

    # ------------------------------------------------------------------
    # Keep only the elite candidates (lowest objective values).
    all_candidates.sort(key=lambda x: x[0])
    elite = all_candidates[:8]          # best few across all lengths

    best_seq = elite[0][1].copy()
    best_score = elite[0][0]
    print(f"[{time.time():.1f}] initial global best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Refinement of each elite seed.
    for init_score, init_seq in elite:
        if time.time() >= deadline:
            break

        # a) short hard‑max descent
        seq, sc = _max_grad_descent(
            init_seq.copy(),
            deadline,
            best_seq,
            best_score,
            lr0=0.6,
            max_iter=15000,
            eval_interval=300,
        )
        if sc < best_score:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time():.1f}] after hard‑max new best {best_score:.6f}")

        # b) LP‑mix loop – at most a few successful iterations per seed
        while time.time() < deadline:
            conv = _fft_self_convolve(best_seq)
            M = conv.max()

            # solve the active‑set LP
            g_raw = _lp_step_active(best_seq, M, top_k=200)
            if g_raw is None:
                break
            if g_raw.sum() == 0.0:
                break
            g = g_raw / g_raw.sum()   # normalise to a probability vector

            improved = False
            for t in (0.25, 0.5, 0.75):
                cand = (1.0 - t) * best_seq + t * g
                cand /= cand.sum()
                # a couple of hard‑max "polish" steps
                cand = _hardmax_refine(cand, steps=2000, lr0=0.4)
                cur = _objective(cand)
                if cur < best_score - 1e-12:
                    best_seq, best_score = cand.copy(), cur
                    print(f"[{time.time():.1f}] LP‑mix t={t:.2f} new best {best_score:.6f}")
                    improved = True
                    break
            if not improved:
                break      # exit the LP‑mix loop for this seed

    # ------------------------------------------------------------------
    # 3️⃣  Final greedy polishing (random swaps) until time runs out.
    if time.time() < deadline:
        best_seq, best_score = _pairwise_swap_local(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
        )

    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
