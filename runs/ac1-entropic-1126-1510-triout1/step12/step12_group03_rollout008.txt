def _gradient_softmax_max(a: np.ndarray, beta: float) -> np.ndarray:
    n = a.shape[0]
    conv = _fft_self_convolve(a)  # length 2n-1
    # compute softmax weights
    max_c = conv.max()
    exp_vals = np.exp(beta * (conv - max_c))
    probs = exp_vals / exp_vals.sum()
    # Reverse probs
    rev = probs[::-1]
    # convolve rev with a (full) to get gradient component
    # using FFT for speed
    size = 1 << ((len(rev) + n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    R = np.fft.rfft(rev, n=size)
    conv_rev = np.fft.irfft(A * R, n=size)
    # conv_rev length = size, but we only need first len(rev)+n-1 elements
    conv_full = conv_rev[: len(rev) + n - 1]
    grad = 2.0 * conv_full[n-1:n-1+n]
    return grad

    a = a0.copy()
    n = a.shape[0]

    total_iters = 0
    lr = 0.5

    stage_iters = [2000, 2000, 2000, 2000, 2000]   # can be adjusted based on time.

    best_score = overall_best_score

    for beta, max_iter in zip(beta_schedule, stage_iters):

        # init Adam moments
        m = np.zeros_like(a)
        v = np.zeros_like(a)

        beta2 = 0.999

"""
Search for a non‑negative step‑function that minimizes
evaluate_sequence().  The method consists of three phases:

1.  Sweep a rich set of analytic families (uniform, arcsine, beta,
    edge‑weighted, quadratic, sine, cosine, Chebyshev/Kaiser windows,
    random Dirichlet or sparse vectors) on several lengths and keep the
    best few seeds.

2.  Refine each seed with an annealed, soft‑max approximation of the
    max‑convolution objective.  The smooth surrogate
        L_β(a)= (1/β)·log Σ_k exp(β·(a*a)_k)
    is differentiable; its gradient is obtained by a single FFT‑based
    convolution.  We run Adam (exponentiated‑gradient) updates, gradually
    raising β (β∈{1,5,15,50,200}) while projecting onto the probability
    simplex Σa_i=1.

3.  A short hill‑climbing phase that moves a random fraction of mass
    between two indices if it strictly improves the *exact* objective
    2·n·max(a*a).  The best sequence found during the whole run is
    returned as a plain list of floats.

The routine respects the supplied time budget (default 1000 s) and
periodically prints progress.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution via FFT (real inputs)."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) for a vector with Σa = 1.
    The input must be non‑negative; no safety clipping.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    np.clip(x, 1e-15, 1 - 1e-15, out=x)
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    np.clip(x, 1e-15, 1 - 1e-15, out=x)
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ ((i+½)*(n‑i‑½))^{‑p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Gradient of the smooth max ------------------------------------------------
def _gradient_from_probs(a: np.ndarray, probs: np.ndarray) -> np.ndarray:
    """Gradient of Σ prob_k·(a*a)_k w.r.t. a (probabilities sum to 1)."""
    rev = probs[::-1]
    conv_rev = _convolve_fft(rev, a)          # length 3n‑2
    n = a.shape[0]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad


def _softmax_gradient(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Returns (grad, max_conv) where grad is the smooth‑max gradient
    and max_conv is the true max of the convolution (used for the exact objective).
    """
    conv = _fft_self_convolve(a)               # length 2n‑1
    max_c = conv.max()
    # soft‑max probabilities
    exp_vals = np.exp(beta * (conv - max_c))
    probs = exp_vals / exp_vals.sum()
    grad = _gradient_from_probs(a, probs)
    return grad, max_c

# Annealed Adam descent ---------------------------------------------------
def _annealed_adam(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    betas: list[float] | None = None,
    it_per_stage: int = 3000,
    lr0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style descent on the smooth max‑convolution surrogate.
    A schedule of increasing β (inverse temperature) is used; after each
    stage the learning rate is halved.
    """
    if betas is None:
        betas = [1.0, 5.0, 15.0, 50.0, 200.0]

    a = a0.copy()
    n = a.shape[0]
    best_local_seq = best_seq
    best_local_score = best_score

    for stage, beta in enumerate(betas):
        if time.time() >= deadline:
            break
        lr = lr0 / (2 ** stage)                # slowly shrink step size
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        t = 0
        while t < it_per_stage and time.time() < deadline:
            t += 1
            grad, max_c = _softmax_gradient(a, beta)
            # Adam moments
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a = a - step
            # projection onto simplex
            a = _project_onto_simplex(a, s=1.0)

            if t % 150 == 0:
                cur_score = 2.0 * n * max_c           # exact objective for current a
                if cur_score < best_local_score - 1e-14:
                    best_local_score = cur_score
                    best_local_seq = a.copy()
                    print(
                        f"[{time.time():.1f}] annealed β={beta:.1f} it={t} new global best {best_local_score:.6f}"
                    )
        # end while per stage
    return best_local_seq, best_local_score

# Greedy pairwise polishing ---------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly transfer a fraction of mass; keep only improving moves."""
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score
