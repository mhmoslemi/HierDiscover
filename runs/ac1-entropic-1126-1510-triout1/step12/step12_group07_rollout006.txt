"""
Propose a step‑function (non‑negative vector) that minimizes

    2 * n * max(conv(a, a)) / sum(a)^2

where `conv` is the full (linear) self‑convolution.
The code builds many analytic and random seeds, evaluates them
exactly, keeps the most promising candidates and refines each one
with three optimisation phases:

1. **Adam on a smooth‑max surrogate** – the convolution is turned
   into a soft‑max using a temperature α that slowly grows.  The
   probabilities are parametrised by logits and updated with Adam.

2. **Exponential‑gradient multimax descent** – a pure sub‑gradient
   method that directly minimises the exact max‑convolution objective.

3. **Aggressive pairwise mass‑transfer local search** – moves a
   random fraction of mass from one index to another and accepts
   only improving steps.  The transfer size is allowed up to the
   whole source entry, which helps escape shallow local minima.

All phases respect a global time budget (default 1000 s).  The best
vector found (scaled to have sum = 1) is returned as a plain
``list[float]``.  The implementation is self‑contained, uses only
numpy and the provided ``evaluate_sequence`` for final validation,
and runs on a single process (the time budget leaves room for a
second core if the host spawns additional threads internally).
"""

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a * a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_i w_i (a*a)_i  with respect to a.
    Uses cross‑correlation via FFT, returns vector of length len(a).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


def _objective(a: np.ndarray) -> float:
    """Exact objective used by the challenge (sum(a) is assumed > 0)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    shift = np.max(z)
    exp_z = np.exp(z - shift)
    s = exp_z.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / len(z))
    return exp_z / s

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.minimum(i + 1, n - i)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    vals /= vals.sum()
    return vals

def _generate_seeds_for_length(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    seeds: List[np.ndarray] = []

    # Analytic families
    seeds.append(_distribution_uniform(n))
    seeds.append(_distribution_arcsine(n))
    seeds.append(_distribution_sine(n))
    seeds.append(_distribution_triangular(n))
    seeds.append(_distribution_gaussian(n))

    for p in (0.05, 0.1, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0, 1.5, 2.0):
        seeds.append(_distribution_beta_sym(n, p))

    # Random asymmetric Beta
    for _ in range(3):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        seeds.append(_distribution_beta_asym(n, alpha, beta))

    # Power‑law families
    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        seeds.append(_distribution_power_increasing(n, p))

    # Exponential families
    for lam in (0.001, 0.003, 0.005, 0.01, 0.02):
        seeds.append(_distribution_exponential_increasing(n, lam))

    # Two‑block spikes
    for block_frac in (0.05, 0.10, 0.15):
        seeds.append(_distribution_two_blocks(n, block_frac))

    # Random Dirichlet draws (various concentrations)
    for _ in range(8):
        alpha = 10 ** rng.uniform(-2, 0)   # 0.01 … 1
        seeds.append(_distribution_dirichlet(n, rng, alpha=alpha))

    # Random sparse patterns
    seeds.append(_distribution_random_sparse(n, rng, density=0.01))

    # Purely random vectors (no special structure)
    for _ in range(8):
        v = rng.random(n)
        v = np.maximum(v, 0.0)
        if v.sum() == 0.0:
            v[0] = 1.0
        v /= v.sum()
        seeds.append(v)

    return seeds

def _adam_optimize(z0: np.ndarray,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   alpha0: float = 0.5,
                   alpha_max: float = 2000.0,
                   growth_factor: float = 1.01,
                   lr: float = 0.15,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   max_iter: int = 400000,
                   eval_interval: int = 300) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns the best vector seen during the run.
    """
    n = z0.shape[0]
    z = z0.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        # probability vector on simplex
        a = _softmax(z)

        # smooth‑max over the convolution entries
        conv = _fft_self_convolve(a)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # exact gradient of the surrogate (2 * a ⋆ w)
        grad_a = _gradient_from_weights(a, w)

        # chain‑rule through softmax
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        z -= step

        # temperature schedule
        if t % 500 == 0:
            alpha = min(alpha * growth_factor, alpha_max)

        # occasional exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


def _multimax_exponential_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.6,
                                 max_iter: int = 200000,
                                 eval_interval: int = 250) -> Tuple[np.ndarray, float]:
    """
    Exponential‑gradient (mirror descent) on the exact max‑convolution.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)          # diminishing step size
        a = a * np.exp(-eta * grad)
        a = _project_onto_simplex(a)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] multimax it={t} new best {best_score:.6f}")

    return best_seq, best_score


def _greedy_pairwise_swap(a: np.ndarray,
                          deadline: float,
                          best_seq: np.ndarray,
                          best_score: float) -> Tuple[np.ndarray, float]:
    """
    Aggressive greedy walk that transfers a random fraction (up to 100%)
    of the source mass to a random destination, accepting only improvements.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        # pick a source that still carries some mass
        src_idx = np.where(a > 1e-15)[0]
        if src_idx.size == 0:
            break
        i = rng.choice(src_idx)

        # pick a distinct destination uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # random fraction of the source mass (0 … 1)
        delta = a[i] * rng.random()
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score


def _refine_candidate(seed: np.ndarray,
                     deadline: float,
                     best_seq: np.ndarray,
                     best_score: float) -> Tuple[np.ndarray, float]:
    """
    Run the full refinement pipeline on a single seed vector.
    """
    # ---- Phase 1: Adam smooth‑max descent ---------------------------------
    z0 = np.log(np.maximum(seed, 1e-300))
    seq, score = _adam_optimize(
        z0,
        deadline,
        best_seq=best_seq,
        best_score=best_score,
        alpha0=0.5,
        alpha_max=2000.0,
        growth_factor=1.01,
        lr=0.15,
        max_iter=400000,
        eval_interval=300,
    )
    if score < best_score:
        best_seq, best_score = seq, score

    # ---- Phase 2: Exponential‑gradient multimax descent --------------------
    if time.time() < deadline:
        seq, score = _multimax_exponential_descent(
            seq,
            deadline,
            best_seq=best_seq,
            best_score=best_score,
            lr0=0.6,
            max_iter=150000,
            eval_interval=250,
        )
        if score < best_score:
            best_seq, best_score = seq, score

    # ---- Phase 3: Greedy pairwise swaps ------------------------------------
    if time.time() < deadline:
        seq, score = _greedy_pairwise_swap(
            seq,
            deadline,
            best_seq=best_seq,
            best_score=best_score,
        )
        if score < best_score:
            best_seq, best_score = seq, score

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step function (list of floats) that minimises
    the challenge objective.  The routine respects the given time budget
    (default 1000 s) and returns the best vector found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a handful of lengths (moderate to large, but still cheap)
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000]

    overall_best_seq: Optional[np.ndarray] = None
    overall_best_score = float("inf")
    candidate_pool: List[Tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 2. Generate a *large* variety of seeds and keep the most promising.
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating seeds for length {n}")

        raw_candidates = _generate_seeds_for_length(n, rng)

        # Evaluate each quickly (exact objective)
        for cand in raw_candidates:
            if time.time() >= deadline:
                break
            # Guard against numeric issues (should already be normalized)
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                continue
            cand = cand / s
            sc = _objective(cand)
            candidate_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] **** new global best (seed) len={n} score={sc:.6f}")

        # keep only the top few across *all* lengths so far
        candidate_pool.sort(key=lambda x: x[0])
        candidate_pool = candidate_pool[:30]   # keep a modest size for speed

    # If we have a previous solution (available as `height_sequence_1`) we
    # incorporate it as an additional seed.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s == 0.0:
                prev = _distribution_uniform(prev.shape[0])
                s = prev.sum()
            prev = prev / s
            sc = _objective(prev)
            candidate_pool.append((sc, prev.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous solution improves best to {sc:.6f}")
    except Exception:
        pass

    # If, by any chance, we have found nothing, fall back to uniform.
    if overall_best_seq is None:
        n = lengths[0]
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Pick the most promising seeds and spend remaining time refining.
    # ------------------------------------------------------------------
    # Sort candidates globally and keep the very best ones for intensive work.
    candidate_pool.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in candidate_pool[:8]]

    while top_seeds and time.time() < deadline:
        # allocate a time slice for the current seed (at least 15 s)
        remaining = deadline - time.time()
        slice_time = max(15.0, remaining / len(top_seeds))
        sub_deadline = time.time() + slice_time

        seed_seq = top_seeds.pop(0)
        print(f"[{time.time():.1f}] refining seed (len={len(seed_seq)}) with slice {slice_time:.1f}s")
        overall_best_seq, overall_best_score = _refine_candidate(
            seed_seq,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] after refinement best={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final cheap polishing (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _greedy_pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] final greedy swaps best={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Return result
    # ------------------------------------------------------------------
    print(f"Finished. Best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # Convert to plain Python list of floats as required by the framework.
    return overall_best_seq.tolist()
