import numpy as np

def _decreasing_power_distribution(n: int, p: float) -> np.ndarray:
    """Return a non‑negative decreasing distribution a_i ∝ (n‑i)^p,
    normalized to sum = 1."""

        return np.full(n, 1.0 / n, dtype=np.float64)
