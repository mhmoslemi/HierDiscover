    n = len(a)
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    while time.time() < deadline and t < max_iter:
        t+=1
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # adaptive active fraction
        active_frac = max(active_frac0 * (active_decay**t), 0.001)  # decays
        thresh = maxc * (1 - active_frac)
        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            # select top max_active conv values
            top = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top]

        grad = np.zeros_like(a)

"""
Propose a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.

Algorithm Overview
------------------

1. **Broad seed generation** – a variety of analytic families
   (uniform, arcsine, symmetric beta, edge‑weighted, edge‑inverse‑power,
   sine‑weighted, exponential, pure power‑law), their mixtures with the
   uniform distribution, and random Dirichlet / sparse vectors are
   sampled for several lengths.  The 20‑30 best seeds (lowest exact
   objective) are retained.

2. **Additive projected gradient descent with momentum** – for each
   retained seed we run a simplex‑projected gradient descent on the
   exact objective.  The sub‑gradient is summed over the *active*
   convolution indices whose value lies within a small fraction of the
   current maximum.  A momentum term stabilises the updates,
   the learning rate decays like 1/√t and an adaptive active fraction
   focuses the descent as optimisation proceeds.  Jitter is injected
   after long stagnation.

3. **Deterministic pairwise mass transfer** – after the descent a cheap
   greedy swap phase repeatedly moves a random fraction of mass from one
   coordinate to another whenever the objective improves.

4. **Cheap simulated‑annealing refinement** – a final stochastic local
   search (geometric temperature decay) explores the neighbourhood of the
   best solution found.

All operations keep the vector on the simplex (∑aᵢ = 1, aᵢ ≥ 0), thus
the evaluation ``2·n·max(conv)`` is exactly the value returned by
``evaluate_sequence`` for a normalised input.  The routine respects a
user‑supplied time budget (default 1000 s) and prints progress
information.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective for a simplex‑normalised vector (∑a = 1):
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: wᵢ ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    """Second edge family: wᵢ ∝ (i+0.5)^(−p) + (n−i−0.5)^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (-p) + j ** (-p))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight proportional to sin(π (i+½)/n)^p, vanishing at the ends."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential(n: int, decay: float) -> np.ndarray:
    """Monotonically decreasing exponential wᵢ ∝ exp(−decay·i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-decay * i)
    w /= w.sum()
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    """Pure power‑law decreasing wᵢ ∝ (i+½)^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet sample with a random concentration."""
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, U = uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Additive projected gradient descent (with momentum) -------------------
def _additive_descent_momentum(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.2,
    active_frac0: float = 0.04,
    active_decay: float = 0.999995,
    max_active: int = 30,
    beta: float = 0.9,
    eval_interval: int = 200,
    patience: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the simplex.
    The sub‑gradient is the sum of gradients of the *active* convolution
    entries (those within ``active_frac`` of the current maximum).
    Momentum and a decaying learning rate are used.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()

    while time.time() < deadline and t < 500_000:
        t += 1
        conv = _fft_self_convolve(a)
        max_c = conv.max()

        # gradually tighten the active set
        active_frac = max(active_frac0 * (active_decay ** t), 0.001)
        thresh = max_c * (1.0 - active_frac)
        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            # keep the largest contributions only
            top_idx = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top_idx]

        # compute sub‑gradient (additive)
        grad = np.zeros_like(a)
        for k in act:
            # i runs over indices where both i and k-i are valid
            i_start = k - (n - 1)
            if i_start < 0:
                i_start = 0
            i_end = k
            if i_end > n - 1:
                i_end = n - 1
            i_range = np.arange(i_start, i_end + 1, dtype=np.int32)
            j_range = k - i_range
            grad[i_range] += 2.0 * a[j_range]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # decaying step size
        eta = lr0 / math.sqrt(t)

        # additive gradient step
        a = a - eta * momentum

        # enforce non‑negativity and project onto the simplex
        a = np.maximum(a, 0.0)
        a = _project_onto_simplex(a, s=1.0)

        # periodic evaluation on the exact objective
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(
                    f"[{time.time():.1f}] it={t:,} new global best {best_score:.6f}"
                )
            else:
                no_improve += 1

        # jitter after long stagnation
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a, s=1.0)
            no_improve = 0

    return best_seq, best_score

# Deterministic pairwise swap refinement --------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small amount of mass; accept if the objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score

# Simulated‑annealing style local search --------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score
