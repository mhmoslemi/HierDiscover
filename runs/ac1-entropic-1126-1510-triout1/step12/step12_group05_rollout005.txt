def _adam_optimize(a_init, deadline, best_seq, best_score,
                   lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8, max_iter=None):
    n = a_init.shape[0]
    a = a_init.copy()
    # Initialize logits l
    eps_log = 1e-12
    l = np.log(np.clip(a, eps_log, None))
    m = np.zeros_like(l)
    v = np.zeros_like(l)
    t = 0
    while time.time() < deadline:
        t += 1
        # compute a from logits
        # for numerical stability: subtract max
        l_shift = l - np.max(l)
        a = np.exp(l_shift)
        sum_a = a.sum()
        if sum_a == 0.0:
            a = np.full(n, 1.0 / n)
            sum_a = 1.0
        a /= sum_a
        # Compute convolution and max index
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())
        max_conv = conv[k]
        # compute gradient of max conv w.r.t a
        grad_a = _gradient_max_index(a, k)
        # compute inner product a·grad_a
        inner = np.dot(a, grad_a)
        # gradient w.r.t logits
        grad_l = 2.0 * n * a * (grad_a - inner)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_l
        v = beta2 * v + (1.0 - beta2) * (grad_l * grad_l)
        m_hat = m / (1.0 - beta1**t)
        v_hat = v / (1.0 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        l = l - step
        # Clip l to avoid overflow
        l = np.clip(l, -20.0, 20.0)
        # evaluation
        cur_score = 2.0 * n * max_conv  # sum a=1
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Adam iter={t} new best {best_score:.6f}")
        if max_iter is not None and t >= max_iter:
            break
    return best_seq, best_score

"""
Hybrid Adam‑based optimiser for the self‑convolution max‑ratio problem.

The algorithm proceeds in three stages:

1. **Seed generation** – a diverse pool of analytic and random
   distributions (uniform, arcsine, sine, two‑block, Dirichlet,
   random sparse, symmetric Beta) is created for a few problem sizes.
   All seeds are normalised to L1‑norm = 1.

2. **Adam mirror descent** – each promising seed is refined by a
   mirror‑descent on the probability simplex (log‑softmax
   parametrisation).  The true sub‑gradient of the maximal convolution
   entry is used and adaptively updated with the Adam optimiser.
   This typically yields a rapid decrease of the objective.

3. **Greedy polishing** – a cheap random mass‑transfer walk (pairwise
   swaps) is performed until the time budget expires.  Only improving
   moves are kept, which often yields a final tiny improvement.

The routine respects the global time budget and returns the best
sequence found as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2  (a is non‑negative)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
def _gradient_max_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution entry conv[k]  with respect to a.
    Returns a non‑negative vector of length n.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i = np.arange(i_min, i_max + 1, dtype=np.int64)
    j = k - i

    eq_mask = (i == j)
    if np.any(eq_mask):
        g[i[eq_mask]] += 2.0 * a[i[eq_mask]]

    if np.any(~eq_mask):
        ii = i[~eq_mask]
        jj = j[~eq_mask]
        g[ii] += a[jj]
        g[jj] += a[ii]

    return g


# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * x / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(
    n: int,
    rng: np.random.Generator,
    alpha: float = 0.5,
) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(
    n: int,
    rng: np.random.Generator,
    density: float = 0.01,
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0)
    eps = 1e-15
    w = np.exp((p - 1) * np.log(np.clip(x, eps, 1.0)) +
               (p - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _initial_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a broad family of analytic and random seeds for length n."""
    seeds = []

    seeds.append(_distribution_uniform(n))
    seeds.append(_distribution_arcsine(n))
    seeds.append(_distribution_sine(n))
    seeds.append(_distribution_two_blocks(n, block_frac=0.05))

    # symmetric beta on a coarse grid
    for p in np.linspace(0.1, 1.0, 6):
        seeds.append(_distribution_beta_sym(n, p))

    # Dirichlet draws with various concentrations
    for a in (0.1, 0.3, 0.5, 1.0):
        seeds.append(_distribution_dirichlet(n, rng, alpha=a))

    # Sparse draws
    for d in (0.005, 0.01, 0.02):
        seeds.append(_distribution_random_sparse(n, rng, density=d))

    # A few extra random Dirichlet samples
    for _ in range(3):
        seeds.append(_distribution_dirichlet(
            n, rng, alpha=rng.uniform(0.05, 1.0)))

    return seeds


# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
def _adam_optimize(
    a_init: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    max_iter: int = None,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the probability simplex using Adam on the logits.
    The true sub‑gradient of the maximal convolution entry is used.
    """
    n = a_init.shape[0]
    # Initialise logits from the seed (logits of a softmax)
    eps_log = 1e-12
    l = np.log(np.clip(a_init, eps_log, None))
    m = np.zeros_like(l)
    v = np.zeros_like(l)
    t = 0
    while time.time() < deadline:
        t += 1
        # Recover a from logits (softmax)
        l_shift = l - np.max(l)
        a = np.exp(l_shift)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
            a_sum = 1.0
        a /= a_sum

        # Convolution and index of the maximal entry
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())
        max_conv = conv[k]

        # Sub‑gradient of conv[k] w.r.t a
        grad_a = _gradient_max_index(a, k)
        inner = np.dot(a, grad_a)

        # Gradient w.r.t logits (mirror‑descent on simplex)
        grad_l = 2.0 * n * a * (grad_a - inner)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_l
        v = beta2 * v + (1.0 - beta2) * (grad_l * grad_l)
        m_hat = m / (1.0 - beta1**t)
        v_hat = v / (1.0 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        l -= step

        # Prevent overflow/underflow of logits
        l = np.clip(l, -20.0, 20.0)

        # Current objective (sum(a)=1)
        cur_score = 2.0 * n * max_conv

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Adam iter={t} new best {best_score:.6f}")

        if max_iter is not None and t >= max_iter:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
def _hill_climb(
    a_init: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Simple random mass‑transfer walk (pairwise swaps).  Only improving
    moves are kept.
    """
    a = a_init.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        # choose a non‑zero source
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if i == j:
            continue
        # move up to 30 % of the source mass
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        # evaluate
        conv = _fft_self_convolve(new_a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-14:
            a = new_a
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] hill‑climb i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """Entry point expected by the checker."""
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    elite = []   # list of (score, seq) across all lengths

    # ------------------------------------------------------------------
    # 1. Choose problem sizes (larger sizes give tighter theoretical bound)
    lengths = [2000, 4000, 8000, 16000]

    # ------------------------------------------------------------------
    # 2. Scan seeds for each length
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] --- scanning seeds for length {n} ---")
        seeds = _initial_seeds(n, rng)

        # incorporate previous best if available
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.ndim == 1 and prev.size > 0:
                if prev.shape[0] != n:
                    prev = _upsample(prev, n)
                seeds.append(prev.copy())
        except Exception:
            pass

        # evaluate each seed
        for s in seeds:
            if time.time() >= deadline:
                break
            s = np.maximum(s, 0.0)
            if s.sum() == 0.0:
                continue
            s = s / s.sum()
            sc = _objective(s)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = s.copy()
                print(f"[{time.time():.1f}] !!! new global best (len={n}) score={sc:.6f}")
            elite.append((sc, s.copy()))

        # keep only the current elite seeds
        elite.sort(key=lambda x: x[0])
        elite = elite[:12]

    # ------------------------------------------------------------------
    # 3. Optimise the elite seeds with Adam + hill‑climb
    remaining = deadline - time.time()
    if remaining > 30.0 and elite:
        per_seed_time = max(2.0, remaining / (len(elite) * 2.0))
        for seed_score, seed_seq in elite:
            if time.time() >= deadline:
                break

            # Phase A: Adam optimisation
            sub_dead_a = time.time() + per_seed_time
            seq_a, score_a = _adam_optimize(
                seed_seq.copy(),
                sub_dead_a,
                overall_best_seq,
                overall_best_score,
                lr=0.05,
                max_iter=200000,
            )
            if score_a < overall_best_score:
                overall_best_score = score_a
                overall_best_seq = seq_a.copy()
                print(f"[{time.time():.1f}] after Adam new best {overall_best_score:.6f}")

            if time.time() >= deadline:
                break

            # Phase B: greedy hill climb
            sub_dead_b = time.time() + per_seed_time
            seq_b, score_b = _hill_climb(
                seq_a.copy(),
                sub_dead_b,
                overall_best_seq,
                overall_best_score,
            )
            if score_b < overall_best_score:
                overall_best_score = score_b
                overall_best_seq = seq_b.copy()
                print(f"[{time.time():.1f}] after hill‑climb new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final polishing on the current best sequence
    if overall_best_seq is not None and time.time() < deadline:
        overall_best_seq, overall_best_score = _hill_climb(
            overall_best_seq.copy(),
            deadline,
            overall_best_seq,
            overall_best_score,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
