"""
Hybrid parametric + simplex‑projected subgradient search.

1.  Build a diverse pool of seeds:
     - uniform, arcsine, edge‑weighted and symmetric‑Beta families.
     - For the edge‑weighted and Beta families the shape
       parameter is *optimised* with a bounded scalar minimisation.
     - a few random Dirichlet vectors.

2.  Keep the ~best 5 seeds and run a projected sub‑gradient descent on
    the exact objective  O(a)=2·n·max(conv(a))   (a lives on the probability
    simplex Σa=1, a≥0).  The sub‑gradient of max(conv) at the currently
    maximal shift k is   g_i = 2·a_{k‑i} (zero when out of range).
    An exponentiated‑gradient update a←a·exp(−η·g) is followed by a
    Euclidean projection onto the simplex.

    Step‑size η decays like η₀/√t, occasional jitter is added after a
    long streak without improvement.

3.  The best sequence encountered is returned as a plain list.  The
    function respects the supplied time budget (default 1000 s).
"""

import time
import math
import numpy as np

# optional scipy for scalar optimisation (falls back to a grid search)
try:
    from scipy.optimize import minimize_scalar
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = v.astype(np.float64, copy=True)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())       # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1.
    The function is homogeneous of degree 0, therefore we keep the sum
    normalised to 1 during optimisation.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+½)*(n‑i‑½))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0) by clipping
    x = np.clip(x, 1e-15, 1 - 1e-15)
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    w /= w.sum()
    return w

def _optimize_edge_weighted(n: int):

    if _SCIPY_AVAILABLE:
        def f(p):

        # Search in a reasonable interval (0.001 … 10)
        res = minimize_scalar(f, bounds=(1e-3, 10.0), method='bounded',

# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exponentiated‑gradient) -------------
def _subgradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    eval_every: int = 200,
    stagnation_patience: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact objective.
    Updates follow a↦a·exp(−η·g) with g the sub‑gradient of max(conv),
    then projected onto the simplex.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        # pick one of the (possibly multiple) maximal indices at random
        max_idx = np.where(conv >= max_val - 1e-12)[0]
        k = int(rng.choice(max_idx))

        # sub‑gradient g_i = 2·a_{k‑i} (zero when out of range)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # exponentiated‑gradient step
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # projection onto the simplex (sum = 1)
        a = _project_onto_simplex(a)

        # occasional evaluation
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] subgrad it={t} new best {best_score:.6f}")
            else:
                no_improve += 1

        # jitter after long stagnation
        if no_improve >= stagnation_patience:
            jitter = rng.exponential(scale=4e-3, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a)
            no_improve = 0

    return best_seq, best_score

            candidate_pool.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"  new global best (Dirichlet) {sc:.6f}")

        # keep only the most promising seeds so far
        candidate_pool.sort(key=lambda x: x[0])
        candidate_pool = candidate_pool[:30]

    # ------------------------------------------------------------------
    # 2. seed from previously discovered construction if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"  previous best incorporated with score {sc:.6f}")
            candidate_pool.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # safety guard (should never trigger)
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Sub‑gradient descent on the current best seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and candidate_pool:
        # select a few top seeds (different lengths)
        candidate_pool.sort(key=lambda x: x[0])
        seeds = [seq for _, seq in candidate_pool[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] launching sub‑gradient descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            best_seq, best_score = _subgradient_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq.copy(),
                best_score=overall_best_score,
                eta0=0.55,
                eval_every=150,
                stagnation_patience=4000,
            )
            if best_score < overall_best_score:
                overall_best_score, overall_best_seq = best_score, best_seq.copy()
                print(
                    f"[{time.time():.1f}] after descent #{idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 4. Very cheap random pairwise swaps (final polishing)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final pairwise polishing")
        # reuse the same function from the original script (light version)
        a = overall_best_seq.copy()
        n = a.shape[0]
        rng = np.random.default_rng()
        best_score = overall_best_score
        best_seq = a.copy()
        trials = 0
        max_trials = 200_000
        while time.time() < deadline and trials < max_trials:
            trials += 1
            src = np.where(a > 1e-12)[0]
            if src.size == 0:
                break
            i = rng.choice(src)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            delta = a[i] * rng.random() * 0.33
            if delta < 1e-14:
                continue
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta
            cur = _objective(new_a)
            if cur < best_score - 1e-14:
                a = new_a
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}"
                )
        overall_best_seq, overall_best_score = best_seq, best_score

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list; evaluate_sequence will perform final checks.
    return overall_best_seq.tolist()
