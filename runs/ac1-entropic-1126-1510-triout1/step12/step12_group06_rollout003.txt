    t=0

        t+=1
        # choose move type

            # transfer
            src = np.random.choice(np.where(a>1e-12)[0])
            dst = np.random.randint(n-1)

            delta = a[src] * np.random.rand() * 0.3

            new_a[src] -= delta
            new_a[dst] += delta
        else:

            best_seq = new_a.copy()

#!/usr/bin/env python
"""
Hybrid optimiser specialised for the max‑convolution step‑function problem.

Main ideas
~~~~~~~~~~
*   For each chosen length we exhaustively scan a dense grid of symmetric
    Beta‑shaped distributions (parameter p) – this dramatically improves the
    initial seed compared with the coarse grid used previously.
*   A handful of random asymmetric Beta draws and sparse/Dirichlet seeds are
    added to increase diversity.
*   The best few seeds undergo a long Adam‑style smooth‑max descent (multiplicative
    updates on the simplex) with a slowly increasing temperature.
*   Afterwards we perform an additive sub‑gradient descent (projected onto the
    simplex) with a simple back‑tracking line search – this often escapes the
    plateau left by the multiplicative method.
*   Finally a cheap random local search (mass transfers and Gaussian jitter)
    polish the result.
*   All stages respect the global 1000 s budget and print intermediate scores.

The routine returns a list of non‑negative floats (the step‑function heights)
that minimises ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient 2·a⋆w where w (length 2·n‑1) sums to 1.
    Implemented via FFT cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a))/sum(a)^2.
    The routine assumes sum(a)=1, therefore it reduces to 2·n·max(conv).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5,0.5))."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discretised unit interval."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(alpha,beta) on the discretised unit interval."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Few non‑zero coordinates, random values."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    """Draw a Dirichlet vector with concentration α."""
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _smooth_max_adam_descent(a: np.ndarray,
                             deadline: float,
                             best_seq: np.ndarray,
                             best_score: float,
                             eta0: float = 0.5,
                             beta1: float = 0.9,
                             beta2: float = 0.999,
                             eps: float = 1e-8,
                             alpha0: float = 0.2,
                             alpha_mul: float = 1.15,
                             alpha_max: float = 200.0,
                             max_iter: int = 200000,
                             eval_interval: int = 1000) -> Tuple[np.ndarray, float]:
    """
    Adam‑style exponentiated‑gradient descent on the smooth‑max surrogate
    f_α(a) = (1/α)·log Σ_i exp(α·conv_i(a)).
    The temperature α is increased geometrically.
    """
    n = a.shape[0]
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        # full self‑convolution
        conv = _fft_self_convolve(a)

        # soft‑max weights (numerically stable)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient of the smooth surrogate
        grad = _gradient_from_weights(a, w)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # multiplicative update
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(-step)

        # simplex projection
        a = _project_onto_simplex(a)

        # temperature schedule
        if t % 300 == 0 and alpha < alpha_max:
            alpha = min(alpha * alpha_mul, alpha_max)

        # periodic exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _additive_subgradient_descent(a: np.ndarray,
                                  deadline: float,
                                  best_seq: np.ndarray,
                                  best_score: float,
                                  init_step: float = 0.2,
                                  eval_interval: int = 2000) -> Tuple[np.ndarray, float]:
    """
    Projected additive sub‑gradient descent.
    Sub‑gradient is obtained from the uniformly weighted active set
    (indices attaining the maximal convolution value).
    A simple back‑tracking line‑search is applied.
    """
    n = a.shape[0]
    step = init_step
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt
        grad = _gradient_from_weights(a, w)

        improved = False
        # try a few decreasing step‑sizes
        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
            a_new = a - step * factor * grad
            a_new = _project_onto_simplex(a_new)
            cur = _objective(a_new)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a_new.copy()
                a = a_new
                step = step * 1.1          # slight increase on success
                improved = True
                break
        if not improved:
            step = step * 0.6                # shrink step on failure

        if it % eval_interval == 0:
            print(f"[{time.time():.1f}] subgrad it={it} step={step:.5g} best={best_score:.6f}")

        if step < 1e-12:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_local_search(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         max_iters: int = 200000) -> Tuple[np.ndarray, float]:
    """
    Simple stochastic polishing:
    * random mass‑transfer between two coordinates,
    * occasional Gaussian jitter on the whole vector.
    Only improvements are kept.
    """
    n = a.shape[0]
    it = 0
    rng = np.random.default_rng(12345)
    while time.time() < deadline and it < max_iters:
        it += 1
        if rng.random() < 0.7:
            # mass transfer
            src_candidates = np.where(a > 1e-12)[0]
            if src_candidates.size == 0:
                continue
            i = rng.choice(src_candidates)
            j = rng.integers(0, n - 1)
            if j == i:
                continue
            delta = a[i] * rng.random() * 0.30
            if delta < 1e-12:
                continue
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        else:
            # Gaussian jitter
            noise = rng.normal(scale=0.01, size=n)
            new_a = a + noise
            new_a = np.maximum(new_a, 0.0)
            new_a = _project_onto_simplex(new_a)

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            best_score = cur
            best_seq = new_a.copy()
            a = new_a
            print(f"[{time.time():.1f}] random move it={it} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects a 1000 s budget and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: List[Tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Initial scan over several lengths
    lengths = [2000, 4000, 8000, 12000]
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Scanning length {n}")

        # ----- basic families ------------------------------------------------
        families = [
            _distribution_uniform(n),
            _distribution_arcsine(n)
        ]

        # ----- dense grid for symmetric Beta(p,p) ---------------------------
        ps = np.geomspace(0.02, 8.0, num=180)   # fine log‑grid
        best_beta = None
        best_beta_score = float("inf")
        for p in ps:
            cand = _distribution_beta_sym(n, float(p))
            sc = _objective(cand)
            if sc < best_beta_score:
                best_beta_score = sc
                best_beta = cand
        if best_beta is not None:
            families.append(best_beta)

        # ----- random asymmetric Beta draws ---------------------------------
        for _ in range(12):
            alpha = 10 ** rng.uniform(-3, 1)   # 0.001 … 10
            beta = 10 ** rng.uniform(-3, 1)
            families.append(_distribution_beta_asym(n, alpha, beta))

        # ----- sparse / Dirichlet seeds ------------------------------------
        families.append(_distribution_random_sparse(n, rng, density=0.01))
        families.append(_distribution_random_sparse(n, rng, density=0.03))
        families.append(_distribution_dirichlet(n, rng, alpha=0.3))
        families.append(_distribution_dirichlet(n, rng, alpha=0.7))

        # ----- evaluate all candidates ---------------------------------------
        for cand in families:
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                continue
            cand /= s
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] **** new global best len={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))

        # keep only the most promising seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:12]

    # ------------------------------------------------------------------
    # 2. Incorporate previous solution if it exists
    try:
        prev = globals().get("height_sequence_1")
        if isinstance(prev, (list, np.ndarray)):
            prev = np.asarray(prev, dtype=np.float64)
            if prev.ndim == 1 and prev.size > 0:
                prev = np.maximum(prev, 0.0)
                s = prev.sum()
                if s == 0.0:
                    prev = _distribution_uniform(prev.shape[0])
                    s = prev.sum()
                prev /= s
                sc = _objective(prev)
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = prev.copy()
                    print(f"[{time.time():.1f}] previous solution improved: {sc:.6f}")
                top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never be required)
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Adam‑style smooth‑max descent on a few top seeds
    remaining = deadline - time.time()
    if remaining > 60:
        seeds = [overall_best_seq.copy()] + [
            seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:4]
        ]
        for seed_seq in seeds:
            if time.time() >= deadline:
                break
            # give each seed a roughly equal slice of the remaining budget
            sub_deadline = time.time() + (deadline - time.time()) / max(1, len(seeds))
            seq_out, score_out = _smooth_max_adam_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.5,
                beta1=0.9,
                beta2=0.999,
                eps=1e-8,
                alpha0=0.2,
                alpha_mul=1.15,
                alpha_max=200.0,
                max_iter=300000,
                eval_interval=800,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after Adam descent new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Additive projected sub‑gradient descent (fine tuning)
    remaining = deadline - time.time()
    if remaining > 20:
        sub_deadline = time.time() + remaining
        seq_out, score_out = _additive_subgradient_descent(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_step=0.2,
            eval_interval=1500,
        )
        if score_out < overall_best_score:
            overall_best_score = score_out
            overall_best_seq = seq_out.copy()
            print(f"[{time.time():.1f}] after additive sub‑grad new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Cheap random local polishing (mass transfers + jitter)
    remaining = deadline - time.time()
    if remaining > 5:
        sub_deadline = time.time() + remaining
        seq_out, score_out = _random_local_search(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if score_out < overall_best_score:
            overall_best_score = score_out
            overall_best_seq = seq_out.copy()
            print(f"[{time.time():.1f}] after random polishing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
