"""
Hybrid search for a low‑max‑convolution step function.

The algorithm scans a rich palette of analytic families (uniform,
arcsine, symmetric Beta with a fine grid, mixtures, Dirichlet draws,
sparse patterns, etc.) for several lengths, keeps the most promising
candidates and refines them with two complementary mirror‑descent
procedures:

* A smooth‑max Adam optimiser (exponentiated‑gradient on the
  log‑sum‑exp surrogate of the max‑convolution).  The temperature
  α starts small (smooth landscape) and is increased slowly.
* A classic sub‑gradient multimax descent that works directly on the
  true max‑convolution.

Finally a cheap greedy pairwise mass‑transfer polishing step is run
until the overall time budget expires.

All updates stay on the probability simplex (sum = 1, non‑negative)
by normalising after each multiplicative step.  The routine respects
the supplied budget (default 1000 s) and returns the best sequence
found as a plain ``list[float]``.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·(a⋆a)_k  with respect to a (full convolution).
    Implemented using FFT cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact value of  2·n·max(conv(a,a))/sum(a)^2 .
    The routine assumes sum(a)=1, so it reduces to 2·n·max(conv).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discretised unit interval."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # w = (x*(1-x))**(p-1) works for any p>0
    w = (x * (1.0 - x)) ** (p - 1.0)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator, alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    s = vals.sum()
    if s == 0.0:
        vals = np.ones(n, dtype=np.float64)
    else:
        vals /= s
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Few non‑zero coordinates, random values."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
    else:
        vals /= s
    return vals


def _distribution_mixture_uniform_arcsine(n: int, alpha: float) -> np.ndarray:
    """Convex combination alpha·arcsine + (1‑alpha)·uniform."""
    a = _distribution_arcsine(n)
    u = _distribution_uniform(n)
    mix = alpha * a + (1.0 - alpha) * u
    # both components already sum to 1, so mix sums to 1
    return mix


# ----------------------------------------------------------------------
def _smooth_max_adam_descent(a: np.ndarray,
                             deadline: float,
                             best_seq: np.ndarray,
                             best_score: float,
                             eta0: float = 0.7,
                             beta1: float = 0.9,
                             beta2: float = 0.999,
                             eps: float = 1e-8,
                             alpha0: float = 0.1,
                             alpha_mul: float = 1.15,
                             alpha_max: float = 200.0,
                             max_iter: int = 200000,
                             eval_interval: int = 400) -> Tuple[np.ndarray, float]:
    """Adam‑style exponentiated‑gradient descent on the soft‑max surrogate."""
    n = a.shape[0]
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)

        # stable soft‑max weights
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad = _gradient_from_weights(a, w)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # multiplicative update (mirror‑descent on simplex)
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(-step)

        # re‑normalise to the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # temperature schedule
        if t % 300 == 0 and alpha < alpha_max:
            alpha = min(alpha * alpha_mul, alpha_max)

        # periodic exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.6,
                                 max_iter: int = 150000,
                                 eval_interval: int = 500) -> Tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent using a true sub‑gradient of max‑conv."""
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)

        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Multimax it={t} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Greedy random mass transfers – cheap final polishing."""
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = int(np.random.choice(src))
        # choose a distinct destination
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3g} best={best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best coefficient list found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # 1. generate a broad pool of seed vectors for a few lengths
    lengths = [2000, 4000, 8000]          # keep lengths modest for speed
    candidate_pool: List[Tuple[float, np.ndarray]] = []

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        seeds: List[np.ndarray] = []

        # basic families
        seeds.append(_distribution_uniform(n))
        seeds.append(_distribution_arcsine(n))

        # fine grid of symmetric Beta(p,p)
        for p in np.linspace(0.05, 1.0, 20):
            seeds.append(_distribution_beta_sym(n, float(p)))
        # very spiky edge‑heavy Beta
        for p in np.linspace(0.01, 0.04, 4):
            seeds.append(_distribution_beta_sym(n, float(p)))

        # power‑law increasing
        for p in (0.5, 1.0, 2.0):
            seeds.append(_distribution_power_increasing(n, float(p)))

        # exponential families
        for lam in (0.001, 0.003, 0.005):
            seeds.append(_distribution_exponential_increasing(n, float(lam)))

        # two‑block spikes
        for bf in (0.03, 0.07, 0.10):
            seeds.append(_distribution_two_blocks(n, block_frac=bf))

        # Dirichlet draws
        for a in (0.2, 0.5, 0.8):
            seeds.append(_distribution_dirichlet(n, rng, alpha=a))

        # sparse patterns
        seeds.append(_distribution_random_sparse(n, rng, density=0.01))
        seeds.append(_distribution_random_sparse(n, rng, density=0.03))

        # mixtures of arcsine and uniform
        for alpha in np.linspace(0.0, 1.0, 11):
            seeds.append(_distribution_mixture_uniform_arcsine(n, alpha))

        # evaluate all seeds
        for vec in seeds:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, 0.0)
            s = vec.sum()
            if s == 0.0:
                continue
            vec = vec / s
            score = _objective(vec)
            candidate_pool.append((score, vec.copy()))
            if score < 1.6:            # a cheap early‑win filter
                print(f"[{time.time():.1f}] len={n} seed score={score:.6f}")

        # keep only the most promising across lengths
        candidate_pool.sort(key=lambda x: x[0])
        candidate_pool = candidate_pool[:30]

    # ------------------------------------------------------------------
    # 2. possibly reuse the previous solution
    try:
        prev = globals().get("height_sequence_1")
        if isinstance(prev, (list, np.ndarray)):
            prev_arr = np.asarray(prev, dtype=np.float64)
            prev_arr = np.maximum(prev_arr, 0.0)
            s = prev_arr.sum()
            if s > 0.0:
                prev_arr = prev_arr / s
                score = _objective(prev_arr)
                candidate_pool.append((score, prev_arr.copy()))
                candidate_pool.sort(key=lambda x: x[0])
                candidate_pool = candidate_pool[:30]
                print(f"[{time.time():.1f}] previous solution score={score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. initialise the global best
    if not candidate_pool:
        # fallback to uniform of moderate size
        n = 4000
        best_seq = _distribution_uniform(n)
        best_score = _objective(best_seq)
    else:
        best_score, best_seq = candidate_pool[0]
        best_seq = best_seq.copy()
        print(f"[{time.time():.1f}] initial global best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. refinement on a few top seeds
    top_seeds = [seq for _, seq in candidate_pool[:5]]

    for seed_seq in top_seeds:
        if time.time() >= deadline:
            break

        # --- Adam smooth‑max descent ---------------------------------
        sub_dead = time.time() + (deadline - time.time()) * 0.45
        seq, sc = _smooth_max_adam_descent(
            seed_seq.copy(),
            sub_dead,
            best_seq,
            best_score,
            eta0=0.7,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            alpha0=0.1,
            alpha_mul=1.15,
            alpha_max=200.0,
            max_iter=200000,
            eval_interval=400,
        )
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] after Adam new best {best_score:.6f}")

        # --- Classic multimax descent --------------------------------
        sub_dead2 = time.time() + (deadline - time.time()) * 0.40
        seq2, sc2 = _subgradient_multimax_descent(
            seq.copy(),
            sub_dead2,
            best_seq,
            best_score,
            lr0=0.6,
            max_iter=150000,
            eval_interval=500,
        )
        if sc2 < best_score:
            best_score, best_seq = sc2, seq2.copy()
            print(f"[{time.time():.1f}] after multimax new best {best_score:.6f}")

        # --- Final cheap polishing ------------------------------------
        if time.time() < deadline:
            best_seq, best_score = _pairwise_swap_local(
                best_seq.copy(),
                deadline,
                best_seq,
                best_score,
            )
            print(f"[{time.time():.1f}] after swap polishing best={best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
