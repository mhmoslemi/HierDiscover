def _smooth_descent(a, deadline, best_seq, best_score,
                    lr0=0.5, beta0=5.0, beta_max=200.0, beta_inc=1.5):
    n = a.shape[0]
    t = 0
    beta = beta0
    size_fft = 1 << ((3 * n - 2 - 1).bit_length())
    while time.time() < deadline:
        t += 1
        # compute convolution for current a
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)   # a sums to 1
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] iter={t} new best {best_score:.6f}")

        # compute softmax weights
        # prevent overflow: subtract max to get numerically stable.
        exp_vals = np.exp(beta * (conv - max_conv))
        w = exp_vals / exp_vals.sum()

        # compute gradient via weighted cross-correlation as described
        # Compute convolution a * reverse(w)
        w_rev = w[::-1]
        A = np.fft.rfft(a, n=size_fft)
        W = np.fft.rfft(w_rev, n=size_fft)
        conv_full = np.fft.irfft(A * W, n=size_fft)[:3*n-2]
        idx = (2*n - 2) - np.arange(n)
        grad = 2.0 * conv_full[idx]

        # update a via exponentiated gradient
        eta = lr0 / np.sqrt(t)
        a = a * np.exp(-eta * grad)

        # project onto simplex
        a = _project_onto_simplex(a, s=1.0)

        # update beta gradually
        if t % 2000 == 0 and beta < beta_max:
            beta = min(beta * beta_inc, beta_max)

    return best_seq, best_score

"""
Hybrid optimisation for low‑autocorrelation step functions.

Strategy
--------
1.  Analytic seeding – generate a diverse set of probability vectors
    (uniform, arcsine, inverse‑sqrt, two‑block, sparse, Dirichlet) for a few
    moderate lengths (≈ 2500‑4000).  Keep the best few after a cheap exact
    evaluation of   2·n·max(conv(a,a))/sum(a)².

2.  Smoothed max‑convolution descent – replace the hard max‑subgradient
    (which only looks at a single shift) by a *soft* max obtained from a
    soft‑max over all convolution entries.  The resulting gradient is a
    weighted cross‑correlation and is computed with two FFTs per iteration.
    The update follows an exponentiated‑gradient step and a projection onto
    the probability simplex.  The soft‑max temperature β is increased
    gradually (continuation) so that the optimisation first moves in a
    smooth landscape and later targets the true max.

3.  Greedy polishing – cheap random mass‑transfer moves that are kept only
    when they improve the exact objective.

The whole search respects the supplied time budget (default 1000 s) and
returns the best sequence (list of non‑negative floats) found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)² (a is assumed non‑negative)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine_offset(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = i + c
    right = (n - 1 - i) + c
    vals = 1.0 / np.sqrt(left * right)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _distribution_power_inv_sqrt(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = i + 0.5
    right = (n - 1 - i) + 0.5
    vals = (left * right) ** (-p / 2.0)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Small pool of analytic seeds for a fixed length."""
    cand = [_distribution_uniform(n)]
    for c in (0.2, 0.5, 1.0, 2.0):
        cand.append(_distribution_arcsine_offset(n, c))
    for p in (0.8, 1.0, 1.2):
        cand.append(_distribution_power_inv_sqrt(n, p))
    cand.append(_distribution_two_blocks(n, block_frac=0.05))
    cand.append(_distribution_two_blocks(n, block_frac=0.10))
    cand.append(_distribution_random_sparse(n, rng, density=0.01))
    for _ in range(2):
        cand.append(_distribution_dirichlet(n, rng,
                                            alpha=0.3 + 0.7 * rng.random()))
    return cand


# ----------------------------------------------------------------------
def _smooth_descent(a: np.ndarray, deadline: float,
                    best_seq: np.ndarray, best_score: float,
                    lr0: float = 0.5,
                    beta0: float = 5.0,
                    beta_max: float = 200.0,
                    beta_inc: float = 1.5) -> tuple[np.ndarray, float]:
    """
    Smoothed max‑convolution descent.
    Uses a soft‑max over the convolution entries (temperature beta) to obtain
    a smooth gradient, which is then applied with an exponentiated‑gradient
    step and projected onto the simplex.
    """
    n = a.shape[0]
    # FFT size large enough for a (len n) convolved with w_rev (len 2n‑1)
    size_fft = 1 << ((3 * n - 2 - 1).bit_length())
    t = 0
    beta = beta0

    while time.time() < deadline:
        t += 1

        # current objective (sum(a) is 1 because we stay on the simplex)
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv  # sum(a)=1
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] iter={t} new best {best_score:.6f}")

        # ---- soft‑max weights over the convolution ----
        # subtract max to keep numbers in a safe range
        exp_vals = np.exp(beta * (conv - max_conv))
        w = exp_vals / exp_vals.sum()   # length 2n‑1, sums to 1

        # ---- gradient = 2 * cross‑correlation(a, w) ----
        # cross‑correlation can be obtained via convolution with w reversed
        w_rev = w[::-1]
        A = np.fft.rfft(a, n=size_fft)
        W = np.fft.rfft(w_rev, n=size_fft)
        conv_full = np.fft.irfft(A * W, n=size_fft)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)          # pick the right slice
        grad = 2.0 * conv_full[idx]

        # ---- exponentiated‑gradient update ----
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # ---- projection onto the probability simplex (sum = 1) ----
        a = _project_onto_simplex(a, s=1.0)

        # increase the temperature from time to time (continuation)
        if t % 2000 == 0 and beta < beta_max:
            beta = min(beta * beta_inc, beta_max)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Cheap random mass‑transfer walk (greedy acceptance)."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur_score = _objective(new_a)
        if cur_score < best_score - 1e-13:
            a = new_a
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}→{j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score
