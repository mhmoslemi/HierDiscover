"""
Hybrid search for the minimal max‑convolution step function.

Strategy
--------
1. **Broad parametric seeding** – a large collection of analytic families
   (uniform, arcsine, symmetric/asymmetric Beta, power‑law, exponential,
   sine, triangular, Gaussian, sparse, Dirichlet) is instantiated for a
   handful of lengths (2000 … 20000).  For each candidate we evaluate the
   exact objective
        J(a) = 2·n·max(conv(a,a))          (sum(a)=1)
   using an FFT based convolution.  The best ~30 candidates across all
   lengths are retained.

2. **Adam on a smooth‑max surrogate** – each retained seed is optimised
   with Adam on the log‑sum‑exp (smooth‑max) surrogate of the convolution.
   The temperature α is increased geometrically up to a large value,
   guaranteeing convergence to the true max.  The Adam phase runs as long
   as the global time budget allows (≈10 s per seed).

3. **Exponential‑gradient polishing** – a classic multimax sub‑gradient
   (exponentiated‑gradient) descent is applied to the current best
   solution for a longer period.

4. **Greedy pairwise‑swap walk** – a cheap local search that transfers a
   random amount of mass from one index to another and accepts only
   improving moves.

All iterates are kept on the probability simplex (non‑negative,
sum = 1).  The routine respects a user‑supplied time budget (default
1000 s) and returns the best sequence found as a plain list of floats.
"""

import time
import math
import random
from typing import List, Tuple, Optional

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Exact gradient of the smooth‑max surrogate:
        grad_i = 2 * Σ_j a_j * w_{i+j}
    Implemented via FFT cross‑correlation (a reversed).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    return 2.0 * conv[start:start + n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))  (assumes sum(a)=1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# ----- Analytic families -------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Beta(p,p) density on {1,…,n}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((p - 1) * np.log(np.clip(x, eps, 1.0)) +
               (p - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.minimum(i + 1, n - i)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    shift = np.max(z)
    exp_z = np.exp(z - shift)
    s = exp_z.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / len(z))
    return exp_z / s

def _adam_optimize(z0: np.ndarray,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   alpha0: float = 1.0,
                   alpha_max: float = 800.0,
                   growth_factor: float = 1.02,
                   lr: float = 0.12,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   max_iter: int = 300000,
                   eval_interval: int = 250) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns (best_seq, best_score) observed during the run.
    """
    n = z0.shape[0]
    z = z0.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        a = _softmax(z)                     # probability vector on simplex

        conv = _fft_self_convolve(a)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        grad_a = _gradient_from_weights(a, w)

        # back‑prop through softmax
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        z = z - step

        # increase temperature geometrically
        if t % 500 == 0:
            alpha = min(alpha * growth_factor, alpha_max)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam iter={t} α={alpha:.2f}  new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.6,
                                 max_iter: int = 50000,
                                 eval_interval: int = 300) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent for the true max‑convolution objective.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        w = mask.astype(np.float64) / max(cnt, 1)

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] multimax it={t}  new best {best_score:.6f}")

    return best_seq, best_score

def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Simple greedy mass‑transfer walk."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        # pick a source index with non‑trivial mass
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        # pick a distinct target
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction of a[i]
        delta = a[i] * rng.random() * 0.25
        if delta < 1e-12:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        cur = _objective(a_new)
        if cur < best_score - 1e-13:
            a = a_new
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}->j={j} δ={delta:.3e}  new best {best_score:.6f}")

    return best_seq, best_score

def _add_candidate(candidates: List[Tuple[float, np.ndarray]],
                   seq: np.ndarray,
                   max_keep: int = 30) -> None:
    """Insert a candidate keeping only the best ``max_keep``."""
    sc = _objective(seq)
    if len(candidates) < max_keep:
        candidates.append((sc, seq.copy()))
    else:
        # find worst currently stored
        worst_idx = max(range(len(candidates)), key=lambda i: candidates[i][0])
        if sc < candidates[worst_idx][0]:
            candidates[worst_idx] = (sc, seq.copy())

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:

    The routine respects a time budget (default 1000 s) and returns the

    # 1.  Choose a set of lengths – a trade‑off between expressivity
    #     and per‑iteration cost.

    # ------------------------------------------------------------------

    cand_pool: List[Tuple[float, np.ndarray]] = []

            break
        print(f"[{time.time():.1f}] generating seeds for length {n}")

        # deterministic families
        _add_candidate(cand_pool, _distribution_uniform(n))
        _add_candidate(cand_pool, _distribution_arcsine(n))
        _add_candidate(cand_pool, _distribution_sine(n))
        _add_candidate(cand_pool, _distribution_triangular(n))
        _add_candidate(cand_pool, _distribution_gaussian(n))

        # symmetric Beta with a grid of exponents
        for p in np.logspace(-2, 1, 12):      # 0.01 … 10
            _add_candidate(cand_pool, _distribution_beta_sym(n, float(p)))

        # asymmetric Beta (random draws)
        for _ in range(6):
            alpha = 10 ** rng.uniform(-2, 1)
            beta = 10 ** rng.uniform(-2, 1)
            _add_candidate(cand_pool, _distribution_beta_asym(n, alpha, beta))

        # power‑law family
        for p in (0.5, 1.0, 2.0, 4.0, 6.0):
            _add_candidate(cand_pool, _distribution_power_increasing(n, p))

        # exponential families
        for lam in (0.0005, 0.001, 0.003, 0.006, 0.012):
            _add_candidate(cand_pool, _distribution_exponential_increasing(n, lam))

        # two‑block spikes
        for block_frac in (0.04, 0.07, 0.10):
            _add_candidate(cand_pool, _distribution_two_blocks(n, block_frac))

        # random Dirichlet draws
        for _ in range(6):
            alpha = 0.2 + 1.6 * rng.random()  # roughly (0.2, 1.8)
            _add_candidate(cand_pool, _distribution_dirichlet(n, rng, alpha))

        # random sparse patterns
        for dens in (0.008, 0.015, 0.03):
            _add_candidate(cand_pool, _distribution_random_sparse(n, rng, density=dens))

        # completely random probability vectors
        for _ in range(4):
            v = rng.random(n)
            v = np.maximum(v, 0.0)
            v /= v.sum()
            _add_candidate(cand_pool, v)

        # ------------------------------------------------------------------
        # Mixed candidates – convex combination of two previously generated families
        # (the combination of two probability vectors stays a probability vector)
        base_idxs = rng.choice(len(cand_pool), size=2, replace=False)
        w1, w2 = rng.random(2)
        lam = w1 / (w1 + w2)
        mixed = lam * cand_pool[base_idxs[0]][1] + (1 - lam) * cand_pool[base_idxs[1]][1]
        _add_candidate(cand_pool, mixed)

    # Keep only the elite candidates globally
    cand_pool.sort(key=lambda x: x[0])
    cand_pool = cand_pool[:30]
    overall_best_seq = cand_pool[0][1].copy()
    overall_best_score = cand_pool[0][0]
    print(f"[{time.time():.1f}] initial global best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Incorporate a previous solution if available (height_sequence_1)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous solution improved to {sc:.6f}")
            _add_candidate(cand_pool, prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4.  Adam optimisation on the top few seeds
    remaining = deadline - time.time()
    top_seeds = [seq for _, seq in sorted(cand_pool, key=lambda x: x[0])[:8]]
    for seed_idx, seed_seq in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        # give each seed a slice of the remaining budget (at least 8 s)
        slice_time = min(40.0, max(8.0, (deadline - time.time()) / max(1, len(top_seeds) - seed_idx)))
        sub_deadline = time.time() + slice_time

        z0 = np.log(np.maximum(seed_seq, 1e-300))
        seq_out, score_out = _adam_optimize(
            z0,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            alpha0=1.0,
            alpha_max=800.0,
            growth_factor=1.02,
            lr=0.14,
            max_iter=300000,
            eval_interval=250,
        )
        if score_out < overall_best_score:
            overall_best_score = score_out
            overall_best_seq = seq_out.copy()
            print(f"[{time.time():.1f}] after Adam new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Exponentiated‑gradient polishing (multimax sub‑gradient)
    remaining = deadline - time.time()
    if remaining > 12.0:
        sub_deadline = time.time() + 0.8 * remaining
        overall_best_seq, overall_best_score = _subgradient_multimax_descent(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.65,
            max_iter=80000,
            eval_interval=400,
        )
        print(f"[{time.time():.1f}] after multimax best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Final local greedy walk (pairwise swaps)
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] after final swaps best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"FINAL best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # Convert to plain Python list before returning
    return overall_best_seq.tolist()
