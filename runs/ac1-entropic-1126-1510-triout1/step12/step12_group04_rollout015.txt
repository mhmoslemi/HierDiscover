"""
Propose a low‑autocorrelation step function.

The search consists of three phases:

1️⃣  Analytic seeding – a wide variety of families (uniform,
    arcsine with several offsets, power‑inverse‑sqrt with a fine
    grid on the exponent, symmetric Gaussian and exponential shapes,
    two‑block, sparse and Dirichlet draws) is generated for several
    lengths.  Each candidate is normalised (sum = 1) and evaluated.
    The best few are kept as an elite pool.

2️⃣  Soft‑max gradient descent – the true objective
    2·n·max(conv(a,a))/sum(a)² is minimised by a smooth
    approximation of the max: a weighted average of all convolution
    values with a temperature β.  At iteration *t* the weight vector
    is wₖ ∝ exp(β·(convₖ–max(conv))) and the gradient is
          g = 2·(a ⋆ w)      (cross‑correlation)
    where “⋆’’ denotes correlation.  An exponentiated‑gradient step
    followed by Euclidean projection onto the probability simplex
    keeps the iterate non‑negative with unit sum.  β is increased
    slowly (continuation) so the optimisation first smooths the
    landscape and then focuses on the true maximum.  After each
    update the vector is symmetrised (a←(a+a[::-1])/2) – symmetry
    never harms the objective and often reduces the peak.

3️⃣  Final polishing – a short sub‑gradient descent (as in the
    reference code) and greedy random mass‑transfer moves are applied
    to squeeze out any remaining spikes.

The best sequence found before the time budget expires is returned.
"""

import time

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)²."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine_offset(n: int, c: float) -> np.ndarray:
    """Arcsine pdf with tunable offset c>0."""
    i = np.arange(n, dtype=np.float64)
    left = i + c
    right = (n - 1 - i) + c
    vals = 1.0 / np.sqrt(left * right)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _distribution_power_inv_sqrt(n: int, p: float) -> np.ndarray:
    """Generalised arcsine family.
       w_i ∝ ((i+0.5)*(n‑i‑0.5))^{‑p/2}."""
    i = np.arange(n, dtype=np.float64)
    left = i + 0.5
    right = (n - 1 - i) + 0.5
    vals = (left * right) ** (-p / 2.0)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Symmetric Gaussian centred in the middle, σ = sigma_frac·n."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_frac * n
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _distribution_exponential(n: int, tau_frac: float) -> np.ndarray:
    """Symmetric exponential decay from both ends,
       τ = tau_frac·n."""
    i = np.arange(n, dtype=np.float64)
    tau = tau_frac * n
    vals = np.exp(-i / tau) + np.exp(-(n - 1 - i) / tau)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Small pool of analytic seeds for a fixed length."""
    cand = []

    cand.append(_distribution_uniform(n))

    # arcsine with a few offsets
    for c in (0.2, 0.5, 1.0, 2.0):
        cand.append(_distribution_arcsine_offset(n, c))

    # power‑inverse‑sqrt family (grid will be added later)
    for p in (0.8, 1.0, 1.2):
        cand.append(_distribution_power_inv_sqrt(n, p))

    cand.append(_distribution_two_blocks(n, block_frac=0.05))
    cand.append(_distribution_two_blocks(n, block_frac=0.10))

    cand.append(_distribution_random_sparse(n, rng, density=0.01))

    # a couple of Dirichlet draws
    for _ in range(2):
        cand.append(_distribution_dirichlet(n, rng,
                                            alpha=0.3 + 0.7 * rng.random()))
    return cand


# ----------------------------------------------------------------------
def _grad_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute 2·(a ⋆ w) where ⋆ is the (linear) cross‑correlation:
        (a ⋆ w)[i] = Σ_j a[j]·w[i+j].
    Returned gradient has length len(a).
    """
    n = a.shape[0]
    m = w.shape[0]               # = 2·n‑1
    size = 1 << ((n + m - 1 - 1).bit_length())

    FA = np.fft.rfft(a, n=size)
    FW = np.fft.rfft(w, n=size)
    corr = np.fft.irfft(FA * np.conj(FW), n=size)
    grad = 2.0 * corr[:n]        # only non‑negative lags are needed
    return grad


# ----------------------------------------------------------------------
def _softmax_descent(a: np.ndarray,
                     deadline: float,
                     best_seq: np.ndarray,
                     best_score: float,
                     beta0: float = 0.2,
                     beta_max: float = 15.0,
                     beta_factor: float = 1.03,
                     lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Smooth (soft‑max) gradient descent.
    Parameters:
        beta0        – initial temperature (smoothness)
        beta_max    – final temperature
        beta_factor – multiplicative increase of beta every few hundred steps
        lr0          – initial step‑size for the exponentiated‑gradient update
    """
    n = a.shape[0]
    t = 0
    beta = beta0

    while time.time() < deadline:
        t += 1

        # --- evaluate current iterate ---
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] iter={t} new best {best_score:.6f}")

        # --- soft‑max weights over the whole convolution ---
        weights = np.exp(beta * (conv - max_conv))          # ≤ 1, stable
        w_sum = weights.sum()
        if w_sum == 0.0:                                   # extreme case
            weights = np.full_like(conv, 1.0 / conv.size)
        else:
            weights /= w_sum

        # --- gradient (correlation a ⋆ w) ---
        grad = _grad_softmax(a, weights)

        # --- exponentiated‑gradient update + simplex projection ---
        eta = lr0 / math.sqrt(t)
        # clamp gradient to avoid overflow inside exp()
        grad_clipped = np.clip(grad, -50, 50)
        a = a * np.exp(-eta * grad_clipped)
        a = _project_onto_simplex(a)

        # --- enforce symmetry (never hurts the max‑convolution) ---
        a = (a + a[::-1]) * 0.5

        # --- increase β gradually (continuation) ---
        if t % 400 == 0:
            beta = min(beta * beta_factor, beta_max)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _max_subgrad_descent(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent that moves mass away from the
    *currently* maximal convolution index.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        k = int(np.argmax(conv))
        max_conv = conv[k]

        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad iter={t} new best {best_score:.6f}")

        # sub‑gradient of the active peak
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        idx_j = k - idx_i
        grad = np.zeros_like(a)
        grad[idx_i] = 2.0 * a[idx_j]

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = _project_onto_simplex(a)

        # (optional) enforce symmetry – helps a little
        a = (a + a[::-1]) * 0.5

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Greedy random mass‑transfer moves (only accept improvements)."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue
        # move up to 30 % of the source mass
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur_score = _objective(new_a)
        if cur_score < best_score - 1e-13:
            a = new_a
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}→{j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score

        print(f"[{time.time():.1f}] >>> scanning length {n}")

        cand_list = _initial_candidates(n, rng)

        # Extra parameter sweeps
        for c in np.linspace(0.1, 2.0, 6):
            cand_list.append(_distribution_arcsine_offset(n, float(c)))
        for p in np.linspace(0.4, 1.6, 7):
            cand_list.append(_distribution_power_inv_sqrt(n, float(p)))
        for sigma in (0.15, 0.20, 0.25):
            cand_list.append(_distribution_gaussian(n, sigma))
        for tau in (0.20, 0.35, 0.50):
            cand_list.append(_distribution_exponential(n, tau))

        # Evaluate all candidates for this length
        for cand in cand_list:
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                continue
            cand /= s
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] NEW GLOBAL BEST len={n} score={sc:.6f}")
            top_seeds.append((sc, cand.copy()))

        # Keep only the elite few across all processed lengths
        top_seeds.sort(key=lambda x: x[0])
        top_seeds = top_seeds[:12]

    # ------------------------------------------------------------------
    # 2.  Include any previously known solution (height_sequence_1)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s == 0.0:
                prev = _distribution_uniform(prev.shape[0])
                s = prev.sum()
            prev /= s
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous solution improved score={sc:.6f}")
            top_seeds.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Main optimisation – soft‑max gradient descent on the elite pool
    remaining = deadline - time.time()
    if remaining > 30.0 and top_seeds:
        top_seeds.sort(key=lambda x: x[0])
        elite = [seq for _, seq in top_seeds[:5]]
        per_seed = max(1.0, remaining / len(elite))
        for init_seq in elite:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            seq_out, score_out = _softmax_descent(
                init_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                beta0=0.2,
                beta_max=12.0,
                lr0=0.5,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after soft‑max new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Final polishing – classic sub‑gradient + greedy swaps
    if overall_best_seq is not None and time.time() < deadline:
        # short sub‑gradient phase (focuses on the strict max)
        overall_best_seq, overall_best_score = _max_subgrad_descent(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.25,
        )
        # cheap greedy polishing
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    if overall_best_seq is None:
        print("[WARN] No feasible sequence found – returning empty list.")
        return []

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # Convert to plain Python list as required by the grader
    return overall_best_seq.tolist()
