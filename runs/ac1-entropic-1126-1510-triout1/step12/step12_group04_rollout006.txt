```
        elapsed = time.time() - start
```

assistantfinal**Solution Overview**

The optimizer builds a large pool of candidate step‑functions, quickly evaluates their
objective, keeps the most promising ones and then refines them with a
stochastic optimisation phase.  
The key ideas are:

1. **Rich seeding** – besides the classic families (uniform, arcsine, power‑inverse‑√, two‑block, sparse, Dirichlet) we draw many random‑parameter versions of each family.  
   This yields a diverse set of high‑quality starting points for a wide
   range of lengths (2000 – 8000).

2. **Fast objective** – the evaluation uses an FFT‑based linear convolution,
   exactly reproducing `evaluate_sequence` (the clamping in `evaluate_sequence`
   never activates for the values we generate).

3. **Simulated annealing** – each seed is improved by many small mass‑transfer
   moves.  Moves that increase the objective are sometimes accepted according
   to a temperature schedule, allowing the search to escape local minima.

4. **Final polishing** – a short exponentiated‑gradient (max‑subgradient) run
   cleans up any remaining spikes.

The whole procedure respects the 1000 s budget, prints progress,
and returns the best sequence found as a plain list of `float`s.

```python
"""
Stochastic optimisation of a non‑negative step function.

The algorithm works in three phases:

1.  **Seeding** – a large, diverse set of analytic families (uniform,
    arcsine with random offset, power‑inverse‑sqrt with random exponent,
    two‑block, sparse, Dirichlet) is instantiated for several lengths.
    All candidates are normalised to sum = 1 and evaluated with a fast FFT
    convolution.  The best few (≤30) are retained.

2.  **Simulated annealing** – each retained seed undergoes a long
    temperature‑controlled random walk.  In one iteration a random
    fraction of mass is moved from index *i* to *j*; the move is always
    accepted if the objective improves, otherwise with probability
    `exp(-(Δ)/T)`.  The temperature falls linearly to zero, allowing the
    search to explore widely at the beginning and to fine‑tune later.

3.  **Polishing** – a short exponentiated‑gradient (max‑subgradient) descent
    (the same update used in the reference code) finishes the optimisation.

All operations stay on the probability simplex (non‑negative, sum = 1),
so the scaling invariance of the objective is respected.
The produced list of floats is returned as the final candidate.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  Utility: projection onto the simplex (used only for safety)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of *v* onto {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Convolution / objective (identical to evaluate_sequence but faster)
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    # next power of two ≥ 2·n‑1
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)²."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
#  Analytic families (all return a *probability* vector, i.e. sum = 1)

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine_offset(n: int, c: float) -> np.ndarray:
    """Arcsine density with offset *c* (>0)."""
    i = np.arange(n, dtype=np.float64)
    left = i + c
    right = (n - 1 - i) + c
    vals = 1.0 / np.sqrt(left * right)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        return _distribution_uniform(n)
    return vals / total


def _distribution_power_inv_sqrt(n: int, p: float) -> np.ndarray:
    """Generalised arcsine: w_i ∝ ((i+0.5)*(n‑i‑0.5))^{‑p/2}."""
    i = np.arange(n, dtype=np.float64)
    left = i + 0.5
    right = (n - 1 - i) + 0.5
    vals = (left * right) ** (-p / 2.0)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        return _distribution_uniform(n)
    return vals / total


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    """Two equal blocks at the ends, total mass 1."""
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Random support of size ~density·n with random positive weights."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(
    n: int, rng: np.random.Generator, alpha: float = 0.5
) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    return vals / vals.sum()

def _seed_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a large, diverse pool of candidates for a fixed length."""
    cand = []

    # 1. Uniform
    cand.append(_distribution_uniform(n))

    # 2. Randomised arcsine offsets
    for c in rng.uniform(0.05, 5.0, size=5):
        cand.append(_distribution_arcsine_offset(n, float(c)))

    # 3. Randomised power‑inverse‑sqrt exponents
    for p in rng.uniform(0.5, 2.0, size=5):
        cand.append(_distribution_power_inv_sqrt(n, float(p)))

    # 4. Two‑block with random fractions
    for bf in rng.uniform(0.01, 0.2, size=3):
        cand.append(_distribution_two_blocks(n, block_frac=float(bf)))

    # 5. Sparse random vectors
    for dens in rng.uniform(0.001, 0.02, size=3):
        cand.append(_distribution_random_sparse(n, rng, density=float(dens)))

    # 6. Dirichlet draws with random α
    for a in rng.uniform(0.05, 2.0, size=3):

    # 7. Purely Gaussian (abs) and re‑normalised

# ----------------------------------------------------------------------
#  Simulated annealing (temperature‑controlled random walk)

def _simulated_annealing(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """Mass‑transfer SA that respects the simplex; updates *best_seq*."""
    rng = np.random.default_rng()
    n = a0.shape[0]
    cur = a0.copy()
    cur_score = _objective(cur)

    start = time.time()
    # safety: keep best locally as well
    local_best_seq = cur.copy()
    local_best_score = cur_score

    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        # Linear cooling, scale set to a fraction of the current global best
        T = max(1e-12, (0.4 * best_score) * (1.0 - elapsed / total))

        # Choose a source index with non‑zero mass
        i = rng.integers(n)
        if cur[i] < 1e-12:
            continue
        j = rng.integers(n)
        if j == i:
            continue

        # Fraction of the source mass that will be moved – decays over time
        max_frac = 0.5 * (1.0 - elapsed / total) + 0.01
        delta = cur[i] * rng.random() * max_frac
        if delta < 1e-12:
            continue

        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        # (Sum is numerically unchanged, but renormalise for safety)
        new = new / new.sum()

        new_score = _objective(new)

        # Accept if better, otherwise with Metropolis probability
        if new_score < cur_score - 1e-15:
            accept = True
        else:
            accept = rng.random() < math.exp(-(new_score - cur_score) / T)

        if accept:
            cur = new
            cur_score = new_score

            # Update *global* best if we improved
            if cur_score < best_score - 1e-13:
                best_score = cur_score
                best_seq = cur.copy()
                print(
                    f"[{time.time():.1f}] SA new global best {best_score:.6f}"
                )
            # Update *local* best (used for final polishing)
            if cur_score < local_best_score - 1e-13:
                local_best_score = cur_score
                local_best_seq = cur.copy()

    # Return the best sequence discovered during this SA run
    if local_best_score < best_score:
        return local_best_seq, local_best_score
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Max‑subgradient descent (exponentiated gradient) – unchanged from reference

def _max_subgrad_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true (non‑smoothed) objective."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        k = int(np.argmax(conv))
        max_conv = conv[k]
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] subgrad iter={t} new best {best_score:.6f}"
            )

        # sub‑gradient of the active (max) shift
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1)
        idx_j = k - idx_i
        grad = np.zeros_like(a)
        grad[idx_i] = 2.0 * a[idx_j]

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # simplex projection
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

    return best_seq, best_score

    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Generate a rich pool of seeds (different lengths)
    lengths = [2000, 3000, 4000, 6000, 8000]
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, vector)
    overall_best_seq = None
    overall_best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] --- Generating seeds for length {n} ---")
        cands = _seed_candidates(n, rng)
        for cand in cands:
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            total = cand.sum()
            if total == 0.0:
                continue
            cand = cand / total
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(
                    f"[{time.time():.1f}] !!! Global best (len={n}) score={sc:.6f}"
                )
            top_candidates.append((sc, cand.copy()))

        # keep only the elite few across all lengths processed so far
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2️⃣  Incorporate a previously‑known solution (if available)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s == 0.0:
                prev = _distribution_uniform(prev.shape[0])
                s = prev.sum()
            prev = prev / s
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous solution improved: {sc:.6f}")
            top_candidates.append((sc, prev.copy()))
            top_candidates.sort(key=lambda x: x[0])
            top_candidates = top_candidates[:30]
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3️⃣  Simulated‑annealing polishing on the elite seeds
    remaining = deadline - time.time()
    if remaining > 10.0 and top_candidates:
        # use the very best (≤10) seeds for a longer SA run each
        elite = [seq for _, seq in top_candidates[:10]]
        per_seed = max(1.0, remaining / len(elite))
        for seed_seq in elite:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            overall_best_seq, overall_best_score = _simulated_annealing(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
            )

    # ------------------------------------------------------------------
    # 4️⃣  Final short max‑subgradient descent (fine tuning)
    remaining = deadline - time.time()
    if remaining > 5.0 and overall_best_seq is not None:
        sub_deadline = time.time() + remaining
        overall_best_seq, overall_best_score = _max_subgrad_descent(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.3,
        )

    # ------------------------------------------------------------------
    # 5️⃣  Return the best vector found (as a plain Python list)
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
```
