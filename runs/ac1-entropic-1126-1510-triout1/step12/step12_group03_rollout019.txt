def _project_onto_simplex(v, s=1.0):
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
```

def _fft_self_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _distribution_beta_sym(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # compute log density to avoid underflow
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))  # subtract max for stability
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _distribution_gaussian(n, sigma_frac):

def _subgradient_descent_improved(a0, time_limit, best_seq, best_score,
                                 lr0=0.6, top_k=10, eval_every=100, max_iter=200000):
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit and t < max_iter:
        t += 1
        conv = _fft_self_convolve(a)
        # indices of the largest entries (including centre)
        if top_k >= conv.shape[0]:
            idxs = np.arange(conv.shape[0])
        else:
            idxs = np.argpartition(-conv, range(top_k))[:top_k]
        grad = np.zeros_like(a)
        for idx in idxs:
            ind = idx - np.arange(n)
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * a[ind[mask]]
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        # simplex projection
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad it={t} new global best {best_score:.6f}")
    return best_seq, best_score

def _pairwise_swap(a0, time_limit, best_seq, best_score, max_trials=None):
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * np.random.rand() * 0.45
        if delta < 1e-15:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")
    return best_seq, best_score

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use log‑space to avoid underflow for p<1
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Edge‑weighted family w_i ∝ ((i+½)*(n‑i‑½))^{‑p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle.  sigma_frac ∈ (0,0.5)"""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = max(sigma_frac * n, 1e-12)
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w

def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:

    k = max(2, int(n * density))

    vals = np.zeros(n, dtype=np.float64)

    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Sub‑gradient / mirror descent on the exact max‑convolution objective
def _subgradient_descent_improved(a0: np.ndarray,
                                 time_limit: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.6,
                                 top_k: int = 10,
                                 eval_every: int = 100,
                                 max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent using the sub‑gradient formed by the
    top_k largest convolution entries.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit and t < max_iter:
        t += 1
        conv = _fft_self_convolve(a)

        if top_k >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(-conv, range(top_k))[:top_k]

        grad = np.zeros_like(a)
        for idx in idxs:
            ind = idx - np.arange(n)
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * a[ind[mask]]

        eta = lr0 / math.sqrt(t)          # decaying step size
        a = a * np.exp(-eta * grad)       # multiplicative EG update

        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad it={t} new global best {best_score:.6f}")

    return best_seq, best_score

# Greedy pairwise “swap’’ polishing
def _pairwise_swap(a0: np.ndarray,
                   time_limit: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   max_trials: int | None = None) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")

    return best_seq, best_score

                cand = _distribution_kaiser(n, beta)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(f"[{time.time():.1f}] new global best len={n} Kaiser beta={beta:.2f} score={sc:.6f}")

            # Classic windows
            for win_fn, win_name in (
                (_distribution_hamming, "hamming"),
                (_distribution_hann, "hann"),
                (_distribution_blackman, "blackman"),
                (_distribution_bartlett, "bartlett"),
            ):
                cand = win_fn(n)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(f"[{time.time():.1f}] new global best len={n} {win_name} score={sc:.6f}")

        # random Dirichlet seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] new global best len={n} Dirichlet score={sc:.6f}")

        # sparse random seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] new global best len={n} sparse score={sc:.6f}")

        # keep only the best ~40 candidates to bound memory
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:40]

    # ------------------------------------------------------------------
    # 1b. Very large lengths (tiny grid, cheap evaluation)
    large_lengths = [40000, 80000]
    for n in large_lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning very large length {n}")
        for p in (0.2, 0.5, 1.0, 2.0):
            if time.time() >= deadline:
                break
            for base_fn, name in ((_distribution_edge_weighted, "edge"),
                                  (_distribution_beta_sym, "beta")):
                base_seq = base_fn(n, p)
                for lam in (0.0, 0.3, 0.5, 0.7, 1.0):
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base_seq, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(f"[{time.time():.1f}] new global best len={n} {name} p={p} lam={lam} score={sc:.6f}")

        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:40]

    # ------------------------------------------------------------------
    # 2.   Incorporate any previously discovered construction (if present)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # noqa: F821
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.   Fallback – should never happen
    if overall_best_seq is None:
        n = 5000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.   Local continuous optimisation on the best few seeds
    param_opt_time = deadline - time.time()
    if param_opt_time > 30.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        print(f"[{time.time():.1f}] starting local param optimisation on {len(seeds)} seeds")
        per_seed = param_opt_time / max(1, len(seeds))
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            n = seed_seq.shape[0]

            # try Beta family optimisation
            cand_beta, sc_beta = _optimize_two_params(_distribution_beta_sym, n,
                                                     p0=1.0, lam0=0.5)
            if sc_beta < overall_best_score - 1e-12:
                overall_best_score = sc_beta
                overall_best_seq = cand_beta.copy()
                print(f"[{time.time():.1f}] beta opt seed {idx} new best {overall_best_score:.6f}")

            # try edge‑weighted family optimisation
            cand_edge, sc_edge = _optimize_two_params(_distribution_edge_weighted, n,
                                                     p0=1.0, lam0=0.5)
            if sc_edge < overall_best_score - 1e-12:
                overall_best_score = sc_edge
                overall_best_seq = cand_edge.copy()
                print(f"[{time.time():.1f}] edge opt seed {idx} new best {overall_best_score:.6f}")

            # try Gaussian family optimisation
            cand_gauss, sc_gauss = _optimize_sigma_param(n, sigma0=0.2, lam0=0.5)
            if sc_gauss < overall_best_score - 1e-12:
                overall_best_score = sc_gauss
                overall_best_seq = cand_gauss.copy()
                print(f"[{time.time():.1f}] gaussian opt seed {idx} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.   Mirror‑descent refinement on the top seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] start subgradient descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sg_deadline = time.time() + per_seed
            a_opt, sc_opt = _subgradient_descent_improved(
                seed_seq,
                sg_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                top_k=8,
                eval_every=150,
                max_iter=300_000,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] after subgrad seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.   Final greedy pairwise polishing (pure hill‑climbing)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy swap polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best score (2·n·max(conv)): {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
