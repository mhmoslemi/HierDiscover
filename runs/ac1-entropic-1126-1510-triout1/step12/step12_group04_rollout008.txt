"""
Hybrid parametric search for a low‑autocorrelation step‑function.

The routine explores several analytic families (power‑inverse‑sqrt,
arcsine with offset, two‑block) for a handful of lengths.  For each
family a dense grid of shape parameters is evaluated with an FFT‑based
exact objective

    2·n·max(conv(a,a)) / (sum(a)²)

(the same quantity used by ``evaluate_sequence``).  The best candidate
found on the grid is refined by a simple hill‑climbing (random
perturbations of the parameters) and a short greedy polishing phase
(random mass transfers that are kept only when they improve the
objective).  The whole search respects the time budget passed via the
``budget_s`` argument and returns the best sequence (list of floats) it
has discovered.
"""

from __future__ import annotations
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    # next power of two for speed
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)²."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
def _distribution_power_inv_sqrt(n: int, p: float) -> np.ndarray:
    """w_i ∝ ((i+0.5)*(n‑i‑0.5))^{‑p/2}  (p>0)."""
    i = np.arange(n, dtype=np.float64)
    left = i + 0.5
    right = (n - 1 - i) + 0.5
    vals = (left * right) ** (-p / 2.0)
    # numerical safety – replace NaNs by 0 (only happens at the exact edges)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _distribution_arcsine_offset(n: int, c: float) -> np.ndarray:
    """Arcsine density with a tunable offset c>0 (default c≈0.5)."""
    i = np.arange(n, dtype=np.float64)
    left = i + c
    right = (n - 1 - i) + c
    vals = 1.0 / np.sqrt(left * right)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    """Two equal blocks of mass at the two ends, total fraction `block_frac` on each side."""
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def _greedy_polish(a: np.ndarray,
                  deadline: float,
                  best_score: float) -> tuple[np.ndarray, float]:
    """Cheap random mass‑transfer walk that accepts only improving moves."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        # pick a source with non‑trivial mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        j = rng.integers(n)
        if j == i:
            continue
        # transfer up to 30 % of the source mass
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # renormalise to stay on the simplex
        new_a = np.maximum(new_a, 0.0)
        new_a /= new_a.sum()
        cur_score = _objective(new_a)
        if cur_score < best_score - 1e-13:
            a = new_a
            best_score = cur_score
            print(f"[{time.time():.1f}] polish move {i}->{j} δ={delta:.3g} new best {best_score:.6f}")
    return a, best_score

    best_score = float("inf")
    best_info = None   # for debugging

    # ------------------------------------------------------------------
    # 1) Scan a few lengths with dense parameter grids
    lengths = [2000, 2500, 3000, 3500, 4000, 4500, 5000]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] --- scanning length {n} ---")

        # uniform baseline (mostly for sanity)
        cand = np.full(n, 1.0 / n, dtype=np.float64)
        sc = _objective(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            best_info = ("uniform", n)
            print(f"    new global best (uniform) {best_score:.6f}")

        # ----- power‑inverse‑sqrt family (p roughly 0.5 … 3) -----
        p = 0.5
        while p <= 3.001 and time.time() < deadline:
            cand = _distribution_power_inv_sqrt(n, p)
            sc = _objective(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                best_info = ("power_inv", n, p)
                print(f"    new best p={p:.2f}  score={best_score:.6f}")
            p += 0.08   # step ≈0.08 gives ~30 values per length

        # ----- arcsine with offset c (c≈0 … 3) -----
        c = 0.05
        while c <= 3.001 and time.time() < deadline:
            cand = _distribution_arcsine_offset(n, c)
            sc = _objective(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                best_info = ("arcsine_off", n, c)
                print(f"    new best c={c:.2f}  score={best_score:.6f}")
            c += 0.15

        # ----- two‑block family (block fraction) -----
        bf = 0.01
        while bf <= 0.21 and time.time() < deadline:
            cand = _distribution_two_blocks(n, bf)
            sc = _objective(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                best_info = ("two_blocks", n, bf)
                print(f"    new best block_frac={bf:.3f}  score={best_score:.6f}")
            bf += 0.01

    # ------------------------------------------------------------------
    # 2) If a previous solution is available, seed from it
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = np.full(prev.shape[0], 1.0 / prev.shape[0])
            else:
                prev /= prev.sum()
            sc = _objective(prev)
            if sc < best_score:
                best_score, best_seq = sc, prev.copy()
                best_info = ("prev", prev.shape[0])
                print(f"[{time.time():.1f}] previous solution improved to {best_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3) Local refinement in parameter space (hill‑climbing)
    if best_info is not None and time.time() < deadline:
        family = best_info[0]
        n_best = best_info[1]
        # keep a reference to parameter dictionary for easy mutation
        param = {}
        if family == "power_inv":
            param["p"] = best_info[2]
        elif family == "arcsine_off":
            param["c"] = best_info[2]
        elif family == "two_blocks":
            param["bf"] = best_info[2]
        # hill‑climbing loop
        sigma = {"p": 0.04, "c": 0.07, "bf": 0.006}
        while time.time() < deadline:
            # propose a small random move
            if family == "power_inv":
                p_new = param["p"] + rng.normal(scale=sigma["p"])
                p_new = max(0.1, min(5.0, p_new))
                cand = _distribution_power_inv_sqrt(n_best, p_new)
                candidate_score = _objective(cand)
                if candidate_score < best_score - 1e-13:
                    best_score, best_seq = candidate_score, cand.copy()
                    param["p"] = p_new
                    print(f"[{time.time():.1f}] hill‑climb p={p_new:.4f} → {best_score:.6f}")
                else:
                    # occasional acceptance of uphill move (simulated‑annealing style)
                    if rng.random() < 0.001:
                        param["p"] = p_new
            elif family == "arcsine_off":
                c_new = param["c"] + rng.normal(scale=sigma["c"])
                c_new = max(0.0, c_new)
                cand = _distribution_arcsine_offset(n_best, c_new)
                candidate_score = _objective(cand)
                if candidate_score < best_score - 1e-13:
                    best_score, best_seq = candidate_score, cand.copy()
                    param["c"] = c_new
                    print(f"[{time.time():.1f}] hill‑climb c={c_new:.4f} → {best_score:.6f}")
                else:
                    if rng.random() < 0.001:
                        param["c"] = c_new
            elif family == "two_blocks":
                bf_new = param["bf"] + rng.normal(scale=sigma["bf"])
                bf_new = np.clip(bf_new, 0.001, 0.5)
                cand = _distribution_two_blocks(n_best, bf_new)
                candidate_score = _objective(cand)
                if candidate_score < best_score - 1e-13:
                    best_score, best_seq = candidate_score, cand.copy()
                    param["bf"] = bf_new
                    print(f"[{time.time():.1f}] hill‑climb bf={bf_new:.4f} → {best_score:.6f}")
                else:
                    if rng.random() < 0.001:
                        param["bf"] = bf_new
            else:
                break

    # ------------------------------------------------------------------
    # 4) Final greedy polishing (random mass transfer)
    if best_seq is not None and time.time() < deadline:
        best_seq, best_score = _greedy_polish(best_seq.copy(),
                                              deadline,
                                              best_score)

    # ------------------------------------------------------------------
    # 5) Return the best sequence found
    if best_seq is None:
        # fallback – a tiny uniform distribution (should never happen)
        best_seq = np.full(10, 0.1, dtype=np.float64)
        best_score = _objective(best_seq)

    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
