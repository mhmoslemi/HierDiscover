def _adam_descent(a, deadline, best_seq, best_score, eta=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    # a is numpy array sum=1.
    n = len(a)
    logits = np.log(a + 1e-16)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    while time.time() < deadline and t < max_iter:
        t+=1
        # compute convolution
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        # find active set indices
        # choose tolerance as relative: maybe max_c - conv <= 1e-12?
        active = np.where(conv >= max_c - 1e-12)[0]
        # compute gradient w.r.t a
        grad = np.zeros_like(a)
        # iterate active ks
        for k in active:
            # derivative: grad_i += 2*a_{k-i}
            # Equivalent: shift reversed a.
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]
        # optional: divide by len(active) to average
        if active.size > 1:
            grad /= active.size
        # update moments
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        # update logits
        lr = eta / (np.sqrt(t))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # compute new a via softmax
        a = np.exp(logits - np.max(logits))
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0/n)
        else:
            a = a / a_sum
        # evaluate objective occasionally
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} new best {best_score:.6f}")
    return best_seq, best_score

"""
Search for a non‑negative step function that minimizes ``evaluate_sequence``.

Algorithm
---------
1. **Generate a diverse pool** –  analytic families (uniform, arcsine,
   edge‑weighted, generalized Beta), random Dirichlet, sparse vectors, and
   random convex mixtures of a few bases.  Each candidate is normalised to
   sum = 1 and its exact objective ``2·n·max(conv)`` is computed.  Only the
   best ≈60 seeds are kept.

2. **Adam‑based exponentiated‑gradient descent** –  for every retained seed we
   run a mirror‑descent on the simplex.  The sub‑gradient of
   ``max(a∗a)`` is obtained from the *active set* of indices where the
   convolution attains its maximum (or is within a tiny tolerance).  The
   update is performed on log‑its (a_i ∝ exp(z_i)) using Adam’s first and
   second moment estimates; this yields fast, stable convergence while
   keeping a_i≥0 and ∑a_i=1.  If no improvement is seen for a while a
   small random jitter is added to escape stagnation.

3. **Deterministic pairwise refinement** –  repeatedly move a tiny amount of
   mass from one index to another if this strictly reduces the objective.

4. **Final stochastic local search** –  a cheap simulated‑annealing walk
   that occasionally accepts worsening moves to explore nearby basins.

The whole routine respects the 1000 s time budget (default) and returns a
plain Python list; ``evaluate_sequence`` will apply the final clipping and
normalisation.  The implementation is self‑contained, uses only NumPy and
standard library, and avoids any closures or file I/O.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1.
    The function is homogeneous of degree 0, therefore we keep the sum
    normalised to 1 during optimisation.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w = np.maximum(w, 0.0)
    total = w.sum()
    if total == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= total
    return w


def _distribution_generalized_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete analogue of Beta(α,β) on the grid i+½∈[0,n]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    # work in log‑space to avoid overflow/underflow
    log_w = (alpha - 1.0) * np.log(i) + (beta - 1.0) * np.log(j)
    # shift for numerical stability
    log_w -= np.max(log_w)
    w = np.exp(log_w)
    total = w.sum()
    if total == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= total
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.008
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_vectors(vectors: list[np.ndarray], weights: np.ndarray) -> np.ndarray:
    """Convex combination of the given vectors (weights must sum to 1)."""
    assert len(vectors) == len(weights)
    out = np.zeros_like(vectors[0])
    for w, v in zip(weights, vectors):
        out += w * v
    out = np.maximum(out, 0.0)
    total = out.sum()
    if total == 0.0:
        out = np.full_like(out, 1.0 / out.size)
    else:
        out /= total
    return out


# ----------------------------------------------------------------------
# Adam‑based exponentiated‑gradient descent ------------------------------
def _adam_exponential_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-12,
    eval_interval: int = 150,
    patience: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Mirror descent on the simplex using Adam on the log‑its.
    The sub‑gradient is taken from the active set of maximum convolution entries.
    """
    n = a.shape[0]
    # initialise log‑its (z) from the current a
    z = np.log(a + 1e-16)
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    while time.time() < deadline and t < 350_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()

        # active set – all indices whose value is within a tiny tolerance of the max
        tol = max_c * 1e-12 + 1e-16
        active_idx = np.where(conv >= max_c - tol)[0]

        # compute sub‑gradient of max(conv) w.r.t a
        grad = np.zeros_like(a)
        # each active convolution contributes 2*a_{k-i}
        for k in active_idx:
            ind = k - np.arange(n)
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * a[ind[mask]]
        if active_idx.size > 1:
            grad /= float(active_idx.size)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning rate that decays like 1/√t (similar to the original code)
        eta = eta0 / math.sqrt(t)

        # update log‑its
        z -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # obtain a from logits (stable softmax)
        a = np.exp(z - np.max(z))
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # occasional evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(
                    f"[{time.time():.1f}] adam iter={t} new best {best_score:.6f}"
                )
            else:
                no_improve += 1

        # jitter on stagnation
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            # reset moments to avoid mistaking past gradients
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise swap (deterministic local refinement) ------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small mass; keep move only if objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score

# Simulated‑annealing style local search ---------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            # metropolis acceptance
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the given time budget (default 1000 s) and returns
    a plain Python list.  It combines a large initial pool of analytic
    constructions with an Adam‑based exponential‑gradient refinement,
    followed by pairwise swaps and a short simulated‑annealing walk.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 3.0           # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Build a diverse pool of seeds
    lengths = [2500, 3500, 5000, 8000, 12000, 20000, 40000]
    # p values for edge‑weighted / beta families (log‑spaced and linear)
    p_grid = np.concatenate(
        [
            np.logspace(-2, -1, 8),      # 0.01 … 0.1
            np.linspace(0.12, 1.5, 10),  # 0.12 … 1.5
            np.logspace(0, 0.5, 6)      # 1 … 3.16
        ]
    )
    # mixture weight grid
    lam_grid = np.linspace(0.0, 1.0, 9)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # parametrised families + mix with uniform
        for p in p_grid:
            if time.time() >= deadline:
                break
            edge = _distribution_edge_weighted(n, p)
            beta_sym = _distribution_generalized_beta(n, p, p)
            # base list for mixing
            bases = [edge, beta_sym]
            for base in bases:
                # direct evaluation
                sc = _objective(base)
                top_candidates.append((sc, base.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, base.copy()
                    print(
                        f"[{time.time():.1f}] new global best len={n} p={p:.3g} score={sc:.6f}"
                    )
                # mix with uniform
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = lam * _distribution_uniform(n) + (1.0 - lam) * base
                    cand = np.maximum(cand, 0.0)
                    cand /= cand.sum()
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # asymmetric beta draws
        for _ in range(6):
            if time.time() >= deadline:
                break
            a_param = 10 ** rng.uniform(-2, 0)   # 0.01 … 1
            b_param = 10 ** rng.uniform(-2, 0)
            cand = _distribution_generalized_beta(n, a_param, b_param)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best asym‑beta len={n} a={a_param:.3g} b={b_param:.3g} score={sc:.6f}"
                )

        # random Dirichlet seeds
        for _ in range(14):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.5)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(
                    f"[{time.time():.1f}] new global best Dirichlet len={n} score={sc:.6f}"
                )

        # sparse random seeds
        for _ in range(10):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(
                    f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}"
                )

            if time.time() >= deadline:
                break
            base1 = _distribution_uniform(n)

            base3 = _distribution_generalized_beta(n, rng.uniform(0.1, 2.0), rng.uniform(0.1, 2.0))
            weights = rng.dirichlet([1.0, 1.0, 1.0])
            cand = _mix_vectors([base1, base2, base3], weights)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best mix len={n} score={sc:.6f}"
                )

        # keep only the most promising seeds overall (limit memory)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:80]

    # ------------------------------------------------------------------
    # 2. Incorporate a previously discovered construction if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Guard against empty result
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Adam‑based exponential descent on the most promising seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # pick a handful of the best seeds for intensive refinement
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] starting Adam descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _adam_exponential_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.55,
                beta1=0.9,
                beta2=0.999,
                eval_interval=120,
                patience=2500,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after Adam seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5. Pairwise deterministic refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 6. Final simulated‑annealing style local search
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final random‑local search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.009,
            decay=0.9996,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return as plain list – the external ``evaluate_sequence`` will perform final checks.
    return overall_best_seq.tolist()
