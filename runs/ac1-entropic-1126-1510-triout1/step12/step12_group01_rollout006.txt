"""
Optimisation of a non‑negative step‑function (probability vector) for the
objective ``evaluate_sequence``.

The core of the search is a *soft‑max* (log‑sum‑exp) approximation of the
∞‑norm of the self‑convolution.  The smoothed objective is differentiable,
its gradient can be obtained by a single FFT‑based convolution, and an
exponentiated‑gradient (mirror descent w.r.t. KL) update keeps the vector on
the simplex.  The temperature τ of the soft‑max is raised slowly so that the
algorithm first follows a smooth landscape and then focuses on the true
maximum.  A diverse pool of analytic and random seeds is generated, the
best are refined with the described descent, followed by cheap pairwise
swaps and a brief simulated‑annealing walk.  Finally the best vector is
returned as a plain Python list.

The routine respects a user‑supplied time budget (default 1000 s) and
produces a valid sequence for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility: Euclidean projection onto the probability simplex (unused – kept for reference)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolutions -------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


def _fft_cross_convolve(w: np.ndarray, a_rev: np.ndarray, size: int) -> np.ndarray:
    """
    Linear cross‑convolution of w with a_rev (both real).
    Returns the full length (len(w)+len(a_rev)-1) array.
    """
    W = np.fft.rfft(w, n=size)
    R = np.fft.rfft(a_rev, n=size)
    full = np.fft.irfft(W * R, n=size)
    return full


# ----------------------------------------------------------------------
# Exact objective (sum of a is assumed to be 1)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families ------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Upsample a probability vector (linear interpolation on the unit interval)
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new = np.full(new_n, 1.0 / new_n, dtype=np.float64)
    else:
        a_new /= a_new.sum()
    return a_new

# Full soft‑max exponential descent (uses all entries of the convolution)
def _full_smooth_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.55,
    tau0: float = 0.5,
    tau_max: float = 6.0,
    eval_interval: int = 150,
    patience: int = 2500,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the log‑sum‑exp smoothed max of the self‑convolution.
    Uses exponentiated‑gradient updates (multiplicative) and projects onto
    the probability simplex by normalisation.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    t = 0
    no_improve = 0

    # pre‑compute FFT size sufficient for both convolutions
    size_cross = 1 << ((3 * n - 2 - 1).bit_length())

    while time.time() < deadline and t < 500_000:
        t += 1

        # ---- self‑convolution ----
        conv = _fft_self_convolve(a)
        max_c = conv.max()

        # ---- soft‑max weights (stable) ----
        # τ is increased linearly until τ_max
        tau = min(tau_max, tau0 + (tau_max - tau0) * t / 300_000.0)
        logw = tau * (conv - max_c)          # shift by max_c for stability
        w = np.exp(logw)
        w_sum = w.sum()
        if w_sum == 0.0 or np.isinf(w_sum):
            # fallback to uniform weighting
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # ---- gradient of the smoothed objective (exact) ----
        # grad_k = 2 * Σ_i w_i * a_{i‑k}
        # This is a cross‑correlation of w with a.
        a_rev = a[::-1]
        full_corr = _fft_cross_convolve(w, a_rev, size_cross)
        # slice the relevant n entries (indices n‑1 … n‑1+n‑1)
        grad = 2.0 * full_corr[n - 1 : n - 1 + n]

        # ---- exponentiated‑gradient update ----
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # ---- simplex projection (renormalisation) ----
        a_sum = a.sum()
        if a_sum == 0.0 or np.isnan(a_sum):
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = np.maximum(a, 0.0)
            a /= a_sum

        # ---- occasional evaluation of the true objective ----
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(
                    f"[{time.time():.1f}] descent it={t} new best {best_score:.6f}"
                )
            else:
                no_improve += 1

        # ---- jitter when stuck ----
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise deterministic local refinement (as in the previous code)
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Move a random fraction of mass between two positions if the true objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing style local search
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    """Transfer a random amount of mass; occasionally accept worsening moves."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Main entry point invoked by the judge.  Generates a collection of
    candidate vectors, refines the most promising ones with a full
    soft‑max exponential descent, then applies greedy swaps and a brief
    simulated‑annealing walk.  Returns the best vector (as a Python list).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 3.0           # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Diverse analytic / random seed pool
    lengths = [2500, 4000, 6000, 9000, 13000]   # moderate sizes to keep per‑iteration cost low
    p_grid = np.concatenate(
        [
            np.logspace(-2, -1, 9, base=10),   # 0.01 … 0.1
            np.logspace(-1, 0, 9, base=10),    # 0.1 … 1
            np.logspace(0, 0.5, 7, base=10),   # 1 … 3.16
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 7)        # mixing with uniform

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating seeds for length {n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            score = _objective(cand)
            top_candidates.append((score, cand.copy()))
            if score < overall_best_score:
                overall_best_score, overall_best_seq = score, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={score:.6f}")

        # parametrised families + mix with uniform
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            sine = _distribution_sine_weighted(n, p)
            for base in (beta, edge, sine):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    score = _objective(cand)
                    top_candidates.append((score, cand.copy()))
                    if score < overall_best_score:
                        overall_best_score, overall_best_seq = score, cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={score:.6f}"
                        )

        # asymmetric beta (randomised)
        for _ in range(4):
            if time.time() >= deadline:
                break
            a_param = 10 ** rng.uniform(-2, 0)   # 0.01 … 1
            b_param = 10 ** rng.uniform(-2, 0)
            cand = _distribution_beta_asym(n, a_param, b_param)
            score = _objective(cand)
            top_candidates.append((score, cand.copy()))
            if score < overall_best_score:
                overall_best_score, overall_best_seq = score, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best asym‑beta len={n} a={a_param:.3g} b={b_param:.3g} score={score:.6f}"
                )

        # random Dirichlet seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            score = _objective(cand)
            top_candidates.append((score, cand.copy()))
            if score < overall_best_score:
                overall_best_score, overall_best_seq = score, cand.copy()
                print(f"[{time.time():.1f}] new global best Dirichlet len={n} score={score:.6f}")

        # sparse random seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.008)
            score = _objective(cand)
            top_candidates.append((score, cand.copy()))
            if score < overall_best_score:
                overall_best_score, overall_best_seq = score, cand.copy()
                print(f"[{time.time():.1f}] new global best sparse len={n} score={score:.6f}")

        # keep only the most promising candidates overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:40]

    # ------------------------------------------------------------------
    # 2. Include any previously discovered vector (if the variable exists)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev /= prev.sum()
            score = _objective(prev)
            if score < overall_best_score:
                overall_best_score, overall_best_seq = score, prev.copy()
                print(f"[{time.time():.1f}] previous best score={score:.6f}")
            top_candidates.append((score, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback guard
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Full soft‑max exponential descent on the top seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] launching full smooth descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _full_smooth_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.55,
                tau0=0.5,
                tau_max=6.0,
                eval_interval=120,
                patience=2500,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after descent seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5. Upsample & further refinement (if time permits)
    if time.time() < deadline - 10.0:
        for factor in (1.5, 2.0):
            new_n = int(round(overall_best_seq.shape[0] * factor))
            if new_n == overall_best_seq.shape[0]:
                continue
            print(f"[{time.time():.1f}] up‑sampling to length {new_n}")
            up_seq = _upsample(overall_best_seq, new_n)
            # short descent on the up‑sampled vector
            sub_deadline = time.time() + min(20.0, deadline - time.time() - 5.0)
            up_seq, up_score = _full_smooth_descent(
                up_seq,
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.5,
                tau0=0.5,
                tau_max=6.0,
                eval_interval=80,
                patience=2000,
            )
            if up_score < overall_best_score:
                overall_best_score, overall_best_seq = up_score, up_seq.copy()
                print(
                    f"[{time.time():.1f}] up‑sample improvement to {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 6. Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=200_000,
        )

    # ------------------------------------------------------------------
    # 7. Final cheap simulated‑annealing walk
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final stochastic local search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list – ``evaluate_sequence`` performs the final checks.
    return overall_best_seq.tolist()
