#!/usr/bin/env python
"""
Hybrid optimizer + fine‑parameter search for the max‑convolution step‑function.

Main ideas
~~~~~~~~~~
1. Scan a rich family of analytic seeds for several lengths (2000‑8000).
   – Uniform, arcsine, symmetric Beta(p,p) with a *fine* golden‑section search
   – Random asymmetric Beta, Dirichlet draws, sparse patterns.
2. Keep the few best seeds (global top‑k) and improve each of them with
   *smooth‑max* Adam descent (exponentiated‑gradient) using a slowly
   increasing temperature schedule.
3. Refine further with classic multimax (sub‑gradient) descent.
4. Finish with cheap greedy pairwise mass transfers and a short random
   hill‑climb, always keeping the vector on the probability simplex.
5. The whole routine respects the 1000 s budget and returns the best
   sequence (list of floats) found.

The implementation is self‑contained, uses only NumPy and the helper
functions defined below, and is deliberately written without closures or
lambdas as required by the competition rules.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient 2·a⋆w where w (length 2·n‑1) sums to 1.
    Implemented via FFT cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective used throughout the search.
    a is assumed to sum to 1, therefore the objective equals 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    # Guard against degenerate vectors
    if max_conv <= 0:
        return float("inf")
    return 2.0 * n * max_conv

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discretised unit interval."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use log‑space to avoid overflow for extreme p
    log_w = (p - 1) * (np.log(x) + np.log(1.0 - x))
    w = np.exp(log_w - np.max(log_w))  # shift for stability
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(alpha,beta) on the discretised unit interval."""
    eps = 1e-15
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_w = (alpha - 1) * np.log(np.clip(x, eps, 1.0)) + \
            (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0))
    w = np.exp(log_w - np.max(log_w))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Few non‑zero coordinates, random values."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    """Two symmetric spikes of equal mass."""
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float = 0.2) -> np.ndarray:
    """Centered Gaussian (normalised) on the discrete grid."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - mu) / (sigma * n)) ** 2)
    vals /= vals.sum()
    return vals

def _fine_optimize_beta_sym(n: int) -> Tuple[np.ndarray, float]:
    """
    Locate a good symmetric Beta exponent p using a two‑phase grid search.
    Returns (distribution, score).
    """
    # Phase 1 – coarse log‑spaced scan
    p_grid = np.geomspace(0.01, 5.0, num=30)
    best_score = float("inf")
    best_dist = None
    best_p = None
    for p in p_grid:
        dist = _distribution_beta_sym(n, float(p))
        sc = _objective(dist)
        if sc < best_score:
            best_score = sc
            best_dist = dist
            best_p = p

    # Phase 2 – refined log‑space around the best coarse p
    lo = max(best_p * 0.2, 0.001)
    hi = min(best_p * 5.0, 20.0)
    p_grid = np.geomspace(lo, hi, num=40)
    for p in p_grid:
        dist = _distribution_beta_sym(n, float(p))
        sc = _objective(dist)
        if sc < best_score:
            best_score = sc
            best_dist = dist
            best_p = p
    return best_dist, best_score

def _smooth_max_adam_descent(a: np.ndarray,
                             deadline: float,
                             best_seq: np.ndarray,
                             best_score: float,
                             eta0: float = 0.5,
                             beta1: float = 0.9,
                             beta2: float = 0.999,
                             eps: float = 1e-8,
                             alpha0: float = 0.2,
                             alpha_mul: float = 1.12,
                             alpha_max: float = 400.0,
                             max_iter: int = 80000,
                             eval_interval: int = 500) -> Tuple[np.ndarray, float]:
    """
    Adam‑style exponentiated‑gradient descent on the smooth‑max surrogate
    f_α(a) = (1/α)·log Σ_i exp(α·conv_i(a)).
    The temperature α is increased slowly.
    """
    n = a.shape[0]
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)

        # smooth‑max weights (stable)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad = _gradient_from_weights(a, w)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # multiplicative update
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(-step)

        # project back onto simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # temperature schedule
        if t % 300 == 0 and alpha < alpha_max:
            alpha = min(alpha * alpha_mul, alpha_max)

        # periodic exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-13 < best_score:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score

def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.5,
                                 max_iter: int = 40000,
                                 eval_interval: int = 500) -> Tuple[np.ndarray, float]:
    """
    Classic exponentiated‑gradient descent using the sub‑gradient of the
    max‑convolution (average over all active indices).
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()

        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)

        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-13 < best_score:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Multimax it={t} new best {best_score:.6f}")

    return best_seq, best_score

def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Greedy pairwise mass‑transfer: move a small amount from i to j if it improves."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # up to 30 % of the source mass
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-13 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

def _random_hill_climb(a: np.ndarray,
                       deadline: float,
                       best_seq: np.ndarray,
                       best_score: float,
                       max_moves: int = 3000) -> Tuple[np.ndarray, float]:
    """
    Very cheap random walk on the simplex.
    Occasionally accepts uphill moves with a simple decaying temperature.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    t = 0
    while time.time() < deadline and t < max_moves:
        t += 1
        # pick donor and receiver
        donor = rng.integers(n)
        receiver = rng.integers(n - 1)
        if receiver >= donor:
            receiver += 1
        if a[donor] <= 1e-12:
            continue
        # propose a transfer
        delta = a[donor] * rng.random() * 0.5   # up to 50 % of donor mass
        new_a = a.copy()
        new_a[donor] -= delta
        new_a[receiver] += delta

        cur = _objective(new_a)
        if cur + 1e-13 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] hill‑climb move {donor}->{receiver} δ={delta:.3g} new best {best_score:.6f}")
        else:
            # simple temperature schedule
            T = max(0.001, 0.5 * (1 - t / max_moves))
            if rng.random() < math.exp(-(cur - best_score) / T):
                a = new_a
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects ``budget_s`` seconds and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a modest set of lengths (larger n gives more freedom).
    lengths = [2000, 4000, 8000]
    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: List[Tuple[float, np.ndarray]] = []   # (score, vector)

    # ------------------------------------------------------------------
    # 2. Exhaustive scan of analytic families (cheap).
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Scanning analytic families – length {n}")

        # 2a. Uniform and arcsine
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] **** new global best (uniform/arcsine) len={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))

        # 2b. Fine‑tuned symmetric Beta
        cand, sc = _fine_optimize_beta_sym(n)
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = cand.copy()
            print(f"[{time.time():.1f}] **** new global best (Beta sym) len={n} score={sc:.6f}")
        top_candidates.append((sc, cand.copy()))

        # 2c. Random asymmetric Beta draws
        for _ in range(6):
            alpha = 10 ** rng.uniform(-3, 1)
            beta = 10 ** rng.uniform(-3, 1)
            cand = _distribution_beta_asym(n, alpha, beta)
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] **** new global best (Beta asym) len={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))

        # 2d. Dirichlet draws
        for a in (0.1, 0.3, 0.7, 1.0):
            cand = _distribution_dirichlet(n, rng, alpha=a)
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] **** new global best (Dirichlet) len={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))

        # 2e. Sparse patterns
        for d in (0.01, 0.03):
            cand = _distribution_random_sparse(n, rng, density=d)
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] **** new global best (sparse) len={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))

        # 2f. Gaussian‑shaped seeds
        for sigma in (0.08, 0.12, 0.18):
            cand = _distribution_gaussian(n, sigma=sigma)
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] **** new global best (gaussian) len={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))

        # keep only a handful of promising seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:12]

    # ------------------------------------------------------------------
    # 3. Incorporate any previously stored solution.
    try:
        prev = globals().get("height_sequence_1")
        if isinstance(prev, (list, np.ndarray)):
            prev = np.asarray(prev, dtype=np.float64)
            if prev.ndim == 1 and prev.size > 0:
                prev = np.maximum(prev, 0.0)
                s = prev.sum()
                if s == 0.0:
                    prev = _distribution_uniform(prev.shape[0])
                    s = prev.sum()
                prev /= s
                sc = _objective(prev)
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = prev.copy()
                    print(f"[{time.time():.1f}] previous solution improved: {sc:.6f}")
                top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Fallback – if nothing was generated (should never happen).
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. First refinement: Adam‑style smooth‑max descent on the best seeds.
    remaining = deadline - time.time()
    if remaining > 60:
        # select seeds – always keep the current global best
        seeds = [overall_best_seq.copy()] + [
            seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:4]
            if not np.may_share_memory(seq, overall_best_seq)
        ]
        for seed_seq in seeds:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + (deadline - time.time()) / max(1, len(seeds))
            seq_out, score_out = _smooth_max_adam_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.5,
                beta1=0.9,
                beta2=0.999,
                eps=1e-8,
                alpha0=0.2,
                alpha_mul=1.12,
                alpha_max=400.0,
                max_iter=80000,
                eval_interval=400,
            )
            if score_out < overall_best_score - 1e-13:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after Adam descent new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Second refinement: classic multimax sub‑gradient descent.
    remaining = deadline - time.time()
    if remaining > 30:
        seeds = [overall_best_seq.copy()] + [
            seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:2]
            if not np.may_share_memory(seq, overall_best_seq)
        ]
        for seed_seq in seeds:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + (deadline - time.time()) / max(1, len(seeds))
            seq_out, score_out = _subgradient_multimax_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.5,
                max_iter=40000,
                eval_interval=500,
            )
            if score_out < overall_best_score - 1e-13:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after multimax new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final polishing: cheap greedy swaps + short random hill‑climb.
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _random_hill_climb(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_moves=5000,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
