    m = b.max()

    sum_exp = exp_terms.sum()
    return exp_terms / sum_exp

def _smooth_optimize(a_initial, deadline, best_seq, best_score):
    a = a_initial.copy()
    n = len(a)
    t = 0
    # schedule parameters.
    alpha = 5.0
    # step size schedule per iteration.
    eta = 0.2
    while time.time() < deadline:
        # compute conv and max
        b = _fft_self_convolve(a)
        # Softmax weights
        # Using current alpha
        max_b = b.max()
        # Softmax scaled:
        exp_terms = np.exp(alpha * (b - max_b))
        p = exp_terms / exp_terms.sum()
        # gradient

        # EG update: a = a * exp(-eta * grad)  # ensures positivity
        # but gradient may be large; we can scale we step size adaptively:
        a = a * np.exp(-eta * grad)
        # project onto simplex
        a_sum = a.sum()

            a = a / a_sum

def _smooth_exponential_gradient_descent(a0, time_limit, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    t = 0
    eta = init_lr
    alpha = init_alpha
    while time.time() < time_limit and t < max_iter:
        t += 1
        b = _fft_self_convolve(a)
        # compute softmax
        max_b = b.max()
        # compute softmax weights
        exp_terms = np.exp(alpha * (b - max_b))
        p = exp_terms / exp_terms.sum()
        # gradient
        grad = _grad_softmax(a, p)
        # exponentiated gradient update
        a = a * np.exp(-eta * grad)
        # simp simplex projection
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum
        # Optional schedule
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth EG it={t} alpha={alpha:.2f} new global best {best_score:.6f}")
        # increase alpha slowly
        if t % (eval_every*2) == 0:
            alpha = min(alpha * alpha_multiplier, max_alpha)
        # maybe adjust eta decays: eta = init_lr / sqrt(t) similar.
        eta = init_lr / math.sqrt(t + 1e-12)
    return best_seq, best_score

"""
Optimise a non‑negative step‑function for the evaluation function
evaluate_sequence().

Approach
--------
1. Scan a wide collection of analytic families (uniform, arcsine, window
   functions, Beta, edge‑weighted, two‑Gaussian mixtures, sparse/random
   Dirichlet seeds) over several lengths (up to ~30 k).  The best ~30
   candidates are retained.

2. Refine the retained seeds with a smooth surrogate of the
   max‑convolution:  f(a)=log‑sum‑exp(α·(a∗a)).  Its gradient is obtained
   analytically and evaluated via FFTs, giving a full‑gradient,
   exponentiated‑gradient (mirror) descent that respects the simplex
   constraint (positivity + unit sum).  α is increased slowly (continuation)
   so the algorithm first follows a smooth landscape and then focuses on
   the true maximum.

3. A final greedy pairwise mass‑transfer hill‑climber (accept‑only‑improve)
   polishes the result.

The procedure respects the 1000 s budget, prints progress and returns a
plain Python list of non‑negative heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1.
    The input must be non‑negative; the function does **not** perform the
    security clipping present in evaluate_sequence().
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a grid {0,…,n‑1} (p>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """
    Edge‑weighted family: w_i ∝ ((i+½)*(n‑i‑½))^{‑p}.
    For p>0 the distribution is heavy at the two ends.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_two_gaussians(n: int, sigma: float, offset: float) -> np.ndarray:
    """
    Two symmetric Gaussian bumps centred at offset and 1‑offset (relative
    to the interval [0,1]).  sigma is the standard deviation in the same
    relative units.
    """
    x = np.arange(n, dtype=np.float64) / n
    mu1 = offset
    mu2 = 1.0 - offset
    w = np.exp(-0.5 * ((x - mu1) / sigma) ** 2) + np.exp(-0.5 * ((x - mu2) / sigma) ** 2)
    w = np.clip(w, 0.0, None)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals

# Soft‑max surrogate and its gradient -----------------------------------
def _softmax_weights(b: np.ndarray, alpha: float) -> np.ndarray:
    """Numerically stable soft‑max over α·b."""
    m = b.max()
    exp_terms = np.exp(alpha * (b - m))
    return exp_terms / exp_terms.sum()


def _grad_softmax(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of  log‑sum‑exp(α·(a*a))  w.r.t. a.
    It equals 2·cross‑correlation(p, a).
    """
    n = a.shape[0]
    L = p.shape[0]                # 2·n‑1
    size = 1 << ((L + n - 1) - 1).bit_length()
    P = np.fft.rfft(p, n=size)
    Ar = np.fft.rfft(a[::-1], n=size)   # a reversed
    conv = np.fft.irfft(P * Ar, n=size)[: L + n - 1]
    # r_j = sum_i p_i * a_{i+j}  for j = 0..n‑1  ==> conv[n‑1‑j]
    r = conv[n - 1::-1][:n]
    # gradient of the surrogate (without the outer 2·n factor)
    return 2.0 * r


# ----------------------------------------------------------------------
# Smooth exponential‑gradient descent ----------------------------------
def _smooth_exponential_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_lr: float = 0.7,
    init_alpha: float = 2.0,
    max_alpha: float = 40.0,
    alpha_mul: float = 1.15,
    eval_every: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the soft‑max surrogate of max‑convolution.
    Uses a multiplicative (exponentiated‑gradient) update followed by a
    simplex projection.  Alpha is increased geometrically to sharpen the
    surrogate.
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0
    alpha = init_alpha

    while time.time() < deadline and it < 250_000:
        it += 1

        # 1) convolution and soft‑max weights
        b = _fft_self_convolve(a)
        p = _softmax_weights(b, alpha)

        # 2) gradient of the surrogate
        grad = _grad_softmax(a, p)
        # clip to avoid extreme multiplicative factors
        grad = np.clip(grad, -10.0, 10.0)

        # 3) exponentiated‑gradient step
        eta = init_lr / math.sqrt(it + 1e-12)
        a = a * np.exp(-eta * grad)

        # 4) project back onto the probability simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # occasional evaluation of the true objective
        if it % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth EG it={it} α={alpha:.2f} new best {best_score:.6f}")

        # slowly increase α to sharpen the approximation to max
        if it % (eval_every * 2) == 0:
            alpha = min(alpha * alpha_mul, max_alpha)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise polishing --------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly transfer a fraction of mass from one index to another."""
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45  # move up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises evaluate_sequence().
    The routine:
      1. scans many analytic families on several lengths,
      2. refines the most promising seeds with a smooth soft‑max based
         exponentiated‑gradient descent,
      3. finishes with a greedy pairwise hill‑climber.
    Returns a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, seq)

    # ------------------------------------------------------------------
    # 1.  Rich family scan
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 12000, 20000, 30000]

    # parameter grids
    p_grid = np.concatenate(
        [
            np.logspace(-2, -0.3, 12),      # 0.01 … 0.5  (edge‑heavy)
            np.linspace(0.5, 2.0, 10),      # moderate
            np.linspace(2.0, 5.0, 6),       # centre‑heavy
            np.linspace(5.0, 12.0, 5),      # very centre‑heavy
        ]
    )
    sigma_grid = np.linspace(0.02, 0.12, 5)
    offset_grid = np.linspace(0.2, 0.4, 5)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families -------------------------------------------------
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
        ):
            cand = base_fn(n)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # parametric Beta / edge‑weighted families -------------------------------
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta_seq = _distribution_beta_sym(n, p)
            edge_seq = _distribution_edge_weighted(n, p)

            for base in (beta_seq, edge_seq):
                sc = _objective(base)
                top_candidates.append((sc, base.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = base.copy()
                    print(
                        f"[{time.time():.1f}] new global best len={n} p={p:.3g} score={sc:.6f}"
                    )

        # Two‑Gaussian mixtures -------------------------------------------------
        for sigma in sigma_grid:
            for offset in offset_grid:
                if time.time() >= deadline:
                    break
                cand = _distribution_two_gaussians(n, sigma, offset)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(
                        f"[{time.time():.1f}] two‑gauss len={n} σ={sigma:.3f} off={offset:.2f} score={sc:.6f}"
                    )

        # Random Dirichlet seeds ------------------------------------------------
        for _ in range(6):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} score={sc:.6f}")

        # Sparse random seeds ---------------------------------------------------
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

        # keep only the best ~30 seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2.  Include any previously discovered construction, if present.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback – should never happen
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 5000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(
            f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 4.  Smooth exponential‑gradient refinement on the top seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] start smooth EG on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sg_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_exponential_gradient_descent(
                seed_seq,
                sg_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                init_lr=0.8,
                init_alpha=2.0,
                max_alpha=45.0,
                alpha_mul=1.12,
                eval_every=180,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] after smooth seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Final greedy pairwise polishing (pure hill‑climbing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy swap polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (2·n·max(conv)): {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
