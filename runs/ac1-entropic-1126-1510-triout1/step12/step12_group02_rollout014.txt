def solve_lp_for_g(f, M):
    n = len(f)
    # Build A_ub as sparse lower-triangular Toeplitz.
    offsets = np.arange(n)  # [0,1,...,n-1]
    data = [np.full(n - d, f[d], dtype=np.float64) for d in offsets]
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csc')
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximize sum(g)
    #bounds = [(0, None)] * n
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs', options={'presolve': True})
    if not res.success:
        raise RuntimeError(f"LP solve failed: {res.message}")
    g = res.x
    return g

def _solve_lp_max_sum(f, M):
    n = f.shape[0]
    # Build A as sparse lower triangular Toeplitz (shape (2*n-1, n))
    import scipy.sparse as sp

"""
Hybrid optimizer for the step‑function evaluation problem.

The algorithm combines three ideas:

1. **Linear‑programming mixing** – for the current distribution `f`
   we solve a LP that maximises the ℓ₁‑mass of a new distribution `g`
   under the same convolution bound `M = max(f∗f)`.  After normalising `g`
   we convex‑combine `f` and `g`.  This is the “LP‑mix” step described
   in the statement and is very effective at moving mass toward the
   edges of the window.

2. **Exponentiated‑gradient descent** – a fast mirror‑descent on the
   exact objective `2·n·max(conv)/sum²`.  The sub‑gradient is the
   gradient of the currently maximal convolution entry.  The update
   `a ← a·exp(-η·grad)` keeps the vector on the probability simplex
   (after normalisation) and rapidly reduces the objective.

3. **Pairwise annealing polish** – a classic random mass‑transfer
   move that accepts improving steps always and occasionally accepts
   worsening steps (early on).  This helps to escape tiny local minima.

The whole search respects the 1000 s budget, uses the previous best
construction `height_sequence_1` if it exists, and falls back to an
arcsine shape otherwise.  The returned list can be fed directly to
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports – required only for the LP step.
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
    import scipy.sparse as _sp
    _HAS_SCI = True
except Exception:  # pragma: no cover
    _HAS_SCI = False

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used for the occasional projected step.
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities – FFT based linear convolution.
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective (scale invariant, sum is assumed to be 1).
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Sub‑gradient of the active convolution entry.
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """Return sub‑gradient of `max(a*a)` and the index where the max occurs."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))

    grad = np.zeros_like(a)
    if k < n:
        # overlap length = k+1
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        # k >= n : overlap length = 2·n-1-k
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the exact objective.
# ----------------------------------------------------------------------
def _gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the true objective (multiplicative updates)."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / a_sum

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] GD it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise random mass‑transfer annealing (final polish).
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins, accept if better."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    while time.time() < deadline:
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)

        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear‑programming “mix” step.
# ----------------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   maximise Σ g_j
    subject to  g ≥ 0,
                (f * g)_k ≤ M   for all k.
    Returns a *scaled* vector g with Σ g = 1.
    """
    n = f.shape[0]
    # Build the lower‑triangular Toeplitz matrix A (size (2n‑1) × n)
    # A_{k,j} = f_{k‑j}  if k ≥ j else 0
    offsets = np.arange(n)                     # 0 … n‑1
    data = [np.full(n - d, f[d], dtype=np.float64) for d in offsets]
    A = _sp.diags(data, offsets, shape=(2 * n - 1, n), format='csc')
    b = np.full(2 * n - 1, M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)          # minimise -Σg  ⇔ maximise Σg
    bounds = [(0, None)] * n

    res = _opt.linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method='highs',
        options={"presolve": True},
    )
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")

    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= 0.0:
        raise RuntimeError("LP gave zero mass")
    return g0 / sum_g0  # normalise to unit sum


def _lp_mix_step(a: np.ndarray, mix_factor: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """One LP‑mix iteration; returns possibly improved vector and its score."""
    M = _fft_self_convolve(a).max()
    if not _HAS_SCI:
        return a, _objective(a)

    try:
        g = _solve_lp_max_sum(a, M)
    except Exception:
        # fall back to the current vector unchanged
        return a, _objective(a)

    # convex combination (mix_factor ∈ (0,1))
    new_a = (1.0 - mix_factor) * a + mix_factor * g
    new_a = _project_onto_simplex(new_a)  # safeguard against numerical drift
    new_score = _objective(new_a)
    return new_a, new_score


# ----------------------------------------------------------------------
# Resampling helper – linear interpolation while staying on the simplex.
# ----------------------------------------------------------------------
def _resample(a: np.ndarray, new_n: int) -> np.ndarray:
    old_n = a.shape[0]
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new


# ----------------------------------------------------------------------
# Main entry point required by the evaluator.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The routine respects the 1000 s limit, mixing an LP‑based update,
    exponentiated‑gradient descent and a final annealing polish.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # 0. Obtain an initial distribution (use the previous best if present)
    # --------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        n = prev.shape[0]
        a = prev.copy()
    except Exception:
        # fallback to a smooth arcsine‑type distribution of moderate size
        n = 4096
        i = np.arange(n, dtype=np.float64)
        a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum == 0.0:
            a[:] = 1.0 / n
        else:
            a /= a_sum

    # make sure we have a proper probability vector (no exact zeros)
    eps = 1e-12
    a = np.maximum(a, eps)
    a /= a.sum()

    best_seq = a.copy()
    best_score = _objective(best_seq)
    print(f"[{time.time():.1f}] initial length={n} score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1. A handful of LP‑mix steps (fast, each solves a modest LP)
    # --------------------------------------------------------------
    if _HAS_SCI:
        for step in range(6):
            if time.time() >= deadline:
                break
            # use a relatively aggressive mixing factor at first
            mix = 0.5 if step < 2 else 0.2
            new_seq, new_score = _lp_mix_step(best_seq, mix, rng)
            if new_score < best_score - 1e-12:
                best_seq, best_score = new_seq, new_score
                print(f"[{time.time():.1f}] LP‑mix step {step+1} improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Long‑run exponentiated‑gradient descent on the best seed.
    # --------------------------------------------------------------
    if time.time() < deadline:
        sub_dead = time.time() + (deadline - time.time()) * 0.55
        best_seq, best_score = _gradient_descent(
            best_seq.copy(),
            sub_dead,
            best_seq,
            best_score,
            eta0=0.9,
            eval_every=150,
        )
        print(f"[{time.time():.1f}] after GD best score {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Final annealing polish (uses random pairwise mass transfers).
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing")
        best_seq, best_score = _pairwise_annealing(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            rng,
        )
        print(f"[{time.time():.1f}] after annealing best score {best_score:.6f}")

    print(
        f"Final best score (objective=2·n·max(conv)/sum²): {best_score:.6f} "
        f"(length={len(best_seq)})"
    )
    return best_seq.tolist()
