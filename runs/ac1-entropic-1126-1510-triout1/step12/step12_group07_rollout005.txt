def _resample_distribution(old_seq, new_n):
    old_n = len(old_seq)
    x_old = np.arange(old_n) / old_n
    x_new = np.arange(new_n) / new_n
    # Use linear interpolation
    new_seq = np.interp(x_new, x_old, np.concatenate([old_seq, [0.0]]))
    # Ensure sum to 1
    if new_seq.sum() == 0:
        new_seq = np.full(new_n, 1.0/new_n)
    else:
        new_seq /= new_seq.sum()
    return new_seq
```

"""
Hybrid optimisation for the max‑convolution step‑function problem.

The routine works in four phases within the allotted time budget:

1. **Fast scouting** – generate many analytic families (uniform,
   arcsine, beta, power‑law, exponential, sinusoidal, triangular,
   Gaussian, random Dirichlet, sparse patterns) for modest lengths
   (≈2k–8k) and keep the few best shapes.

2. **Upscaling** – interpolate the best scouting shapes to larger
   lengths (≈12k–32k) and also add a few canonical seeds (arcsine,
   uniform, Dirichlet).  The resulting vectors are normalised to sum
   to one.

3. **Smooth‑max Adam descent** – each seed is optimised on the
   smooth‑max surrogate
        Fα(a) = (1/α)·log Σ exp(α·conv(a,a))
   using the Adam optimiser on the soft‑max parametrisation
   a = softmax(z).  The temperature α is annealed up to a large
   value (α_max≈2000) so that the surrogate converges to the true
   max.  The best vector found across all seeds is retained.

4. **Polishing** – a long exponentiated‑gradient (multimax) descent
   followed by a cheap greedy pair‑wise‑swap walk are applied within
   the remaining time.

All vectors are kept on the probability simplex (non‑negative,
sum = 1); the objective is scale‑invariant, so we can safely normalise.
The function returns the best sequence (as a plain list of floats)
found before the deadline.
"""

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient 2·a⋆w where w (length 2·n‑1) sums to 1.
    Implemented via FFT cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (scale‑invariant)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax, returns a probability vector."""
    shift = np.max(z)
    exp_z = np.exp(z - shift)
    s = exp_z.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / len(z))
    return exp_z / s

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.minimum(i + 1, n - i)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    vals /= vals.sum()
    return vals

def _initial_candidates_extended(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    """Collect a broad variety of seeds for a given length."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_sine(n))
    cand.append(_distribution_triangular(n))
    cand.append(_distribution_gaussian(n))

    for p in (0.05, 0.1, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0, 1.5, 2.0):
        cand.append(_distribution_beta_sym(n, p))

    for _ in range(4):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))

    for lam in (0.001, 0.003, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential_increasing(n, lam))

    for block_frac in (0.05, 0.10, 0.15):
        cand.append(_distribution_two_blocks(n, block_frac))

    for _ in range(4):
        alpha = 0.1 + 0.9 * rng.random()
        cand.append(_distribution_dirichlet(n, rng, alpha=alpha))

    cand.append(_distribution_random_sparse(n, rng, density=0.01))

    for _ in range(3):
        v = rng.random(n)
        v = np.maximum(v, 0.0)
        if v.sum() == 0.0:
            v[0] = 1.0
        v /= v.sum()
        cand.append(v)

    return cand

def _resample_distribution(old: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a new length (preserving mass)."""
    old_n = old.shape[0]
    # centres of old bins
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    # centres of new bins
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    new = np.interp(x_new, x_old, old, left=0.0, right=0.0)
    new = np.maximum(new, 0.0)
    s = new.sum()
    if s == 0.0:
        return _distribution_uniform(new_n)
    return new / s

def _adam_optimize(z0: np.ndarray,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   alpha0: float = 1.0,
                   alpha_max: float = 2000.0,
                   growth_factor: float = 1.03,
                   lr: float = 0.15,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   max_iter: int = 500000,
                   eval_interval: int = 500) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    """
    n = z0.shape[0]
    z = z0.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        a = _softmax(z)                     # stays on simplex
        conv = _fft_self_convolve(a)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        grad_a = _gradient_from_weights(a, w)

        # softmax Jacobian
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        z = z - step

        # temperature schedule
        if t % 500 == 0:
            alpha = min(alpha * growth_factor, alpha_max)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.8,
                                 max_iter: int = 200000,
                                 eval_interval: int = 200) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the true max‑convolution objective.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)

        # exponentiated‑gradient update (mirror descent on simplex)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] multimax it={t} new best {best_score:.6f}")

    return best_seq, best_score

def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Greedy local mass‑transfer walk (accepts only improving moves)."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function minimizing the supplied
    evaluate_sequence objective.  A hybrid strategy (scouting → up‑scaling
    → Adam → multimax → greedy swaps) is employed within the time budget.
    The best sequence found is returned as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # 1. Fast scouting on modest lengths
    scout_lengths = [2000, 4000, 8000]
    small_cands: List[Tuple[float, np.ndarray, int]] = []

    for n in scout_lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scouting length {n}")
        for cand in _initial_candidates_extended(n, rng):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                continue
            cand = cand / s
            sc = _objective(cand)
            small_cands.append((sc, cand.copy(), n))

    # keep the best few scouting shapes (independent of length)
    small_cands.sort(key=lambda x: x[0])
    top_small = small_cands[:5]

    # ------------------------------------------------------------------
    # 2. Assemble seeds for larger lengths
    large_lengths = [12000, 24000, 32000]
    seeds: List[Tuple[float, np.ndarray]] = []

    # include the previous best if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)     # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() != 0.0:
                prev = prev / prev.sum()
                sc = _objective(prev)
                seeds.append((sc, prev.copy()))
    except Exception:
        pass

    # upscale top scouting shapes
    for _, vec, old_n in top_small:
        for n in large_lengths:
            up = _resample_distribution(vec, n)
            sc = _objective(up)
            seeds.append((sc, up.copy()))

    # add a few canonical families at large sizes
    for n in large_lengths:
        for gen in (_distribution_arcsine, _distribution_uniform, _distribution_dirichlet):
            if gen is _distribution_dirichlet:
                cand = gen(n, rng, alpha=0.5 + 0.5 * rng.random())
            else:
                cand = gen(n)
            cand = np.maximum(cand, 0.0)
            cand = cand / cand.sum()
            sc = _objective(cand)
            seeds.append((sc, cand.copy()))

    # also try a handful of random Dirichlet draws at large sizes
    for n in large_lengths:
        for _ in range(2):
            cand = _distribution_dirichlet(n, rng, alpha=0.3 + 0.7 * rng.random())
            sc = _objective(cand)
            seeds.append((sc, cand.copy()))

    # ------------------------------------------------------------------
    # 3. Keep the most promising seeds (across all lengths)
    seeds.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in seeds[:8]]

    # initialise the overall best with the best scouting candidate
    overall_best_seq = top_small[0][1].copy() if top_small else _distribution_uniform(2000)
    overall_best_score = _objective(overall_best_seq)

    # ------------------------------------------------------------------
    # 4. Adam smooth‑max descent on each seed (time‑slice per seed)
    remaining = deadline - time.time()
    if remaining > 30.0 and top_seeds:
        per_seed = max(20.0, min(80.0, remaining / max(1, len(top_seeds))))
        for seed_seq in top_seeds:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            z0 = np.log(np.maximum(seed_seq, 1e-300))
            seq_out, score_out = _adam_optimize(
                z0,
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                alpha0=1.0,
                alpha_max=2000.0,
                growth_factor=1.03,
                lr=0.15,
                max_iter=500000,
                eval_interval=500,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after Adam new best {overall_best_score:.6f} len={len(overall_best_seq)}")

    # ------------------------------------------------------------------
    # 5. Long multimax (exponentiated‑gradient) polishing
    remaining = deadline - time.time()
    if remaining > 20.0:
        sub_deadline = time.time() + remaining * 0.7
        overall_best_seq, overall_best_score = _subgradient_multimax_descent(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.8,
            max_iter=200000,
            eval_interval=200,
        )
        print(f"[{time.time():.1f}] after multimax best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final greedy pairwise‑swap walk
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] final best {overall_best_score:.6f} (len={len(overall_best_seq)})")

    # ------------------------------------------------------------------
    return overall_best_seq.tolist()
