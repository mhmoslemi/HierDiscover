def _adam_expgd(a, deadline, best_seq, best_score, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        # top_k
        top_k = 5
        if n <= top_k:
            idxs = np.arange(len(conv))
        else:
            idxs = np.argpartition(conv, -top_k)[-top_k:]
        # gradient accumulation
        grad = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k-(n-1))
            i_end = min(n-1, k)
            idx_i = np.arange(i_start, i_end+1)
            idx_j = k - idx_i
            grad[idx_i] += 2.0 * a[idx_j]
        # Adam updates on grad
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        # exponentiated update
        a = a * np.exp(-step)
        # projection onto simplex
        a = _project_onto_simplex(a)
        # periodic evaluation
        if t % 10 == 0:
            cur_score = _objective(a)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] iter={t} new best {best_score:.6f}")
    return best_seq, best_score

"""
Hybrid grid‑search + Adam‑style exponentiated‑gradient refinement.

Stage 1 – Analytic seeding:
    For several lengths we enumerate a rich family of analytic
    distributions (uniform, arcsine with a tunable offset, the
    inverse‑sqrt family with a finely‑swept exponent, two‑block
    patterns, sparse Dirichlet draws, …).  All candidates are
    normalised to sum=1 and the exact objective
        2·n·max(conv(a,a))/sum(a)²
    is evaluated (using an FFT‑based convolution).  The best few
    seeds across all lengths are retained.

Stage 2 – Adam‑style exponentiated‑gradient descent:
    Starting from each elite seed we run a multiplicative update
    a_i ← a_i·exp(‑η·g_i) where the gradient g is obtained from the
    *top‑k* autocorrelation entries (rather than a single peak).
    The update uses Adam‑style moment estimates for an adaptive
    step size and a Euclidean projection onto the probability simplex
    after each iteration.

Stage 3 – Greedy polishing:
    A cheap random mass‑transfer walk (move a fraction of mass from
    one index to another) is performed with greedy acceptance to
    clean up tiny spikes.

The function respects a total time budget (default 1000 s) and
returns the best sequence found as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)² (a assumed non‑negative)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine_offset(n: int, c: float) -> np.ndarray:
    """Arcsine‑type pdf with offset c>0."""
    i = np.arange(n, dtype=np.float64)
    left = i + c
    right = (n - 1 - i) + c
    vals = 1.0 / np.sqrt(left * right)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _distribution_power_inv_sqrt(n: int, p: float) -> np.ndarray:
    """Generalised arcsine family w_i ∝ ((i+0.5)*(n‑i‑0.5))^{-p/2}."""
    i = np.arange(n, dtype=np.float64)
    left = i + 0.5
    right = (n - 1 - i) + 0.5
    vals = (left * right) ** (-p / 2.0)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(round(block_frac * n)))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(round(n * density)))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _grid_candidates(n: int, rng: np.random.Generator) -> list[tuple[float, np.ndarray]]:
    """Generate a diverse pool of analytically‑parameterised candidates."""
    cand = []

    # 1. uniform
    cand.append(_distribution_uniform(n))

    # 2. arcsine offset – fine grid
    for c in np.arange(0.1, 2.51, 0.15):
        cand.append(_distribution_arcsine_offset(n, float(c)))

    # 3. power‑inverse‑sqrt – fine grid on exponent p
    for p in np.arange(0.6, 1.41, 0.04):
        cand.append(_distribution_power_inv_sqrt(n, float(p)))

    # 4. two‑block pattern with several fractions
    for bf in np.arange(0.02, 0.21, 0.02):
        cand.append(_distribution_two_blocks(n, float(bf)))

    # 5. random sparse draws
    for _ in range(5):
        cand.append(_distribution_random_sparse(n, rng, density=0.01))

    # 6. Dirichlet draws with varying concentration
    for _ in range(6):
        alpha = 0.3 + 0.7 * rng.random()
        cand.append(_distribution_dirichlet(n, rng, alpha=float(alpha)))

    # normalise & evaluate
    scored = []
    for a in cand:
        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            continue
        a = a / a.sum()
        sc = _objective(a)
        scored.append((sc, a))
    scored.sort(key=lambda x: x[0])
    # keep only the elite few (they will be merged later)
    return scored[:8]

def _adam_expgd(a: np.ndarray, deadline: float,
                best_seq: np.ndarray, best_score: float,
                lr: float = 0.15,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-8,
                top_k: int = 5) -> tuple[np.ndarray, float]:
    """
    Adam‑style multiplicative update.
    Uses the sub‑gradient built from the top‑k autocorrelation entries.
    """
    n = a.shape[0]
    m = np.zeros_like(a)          # first‑moment estimate
    v = np.zeros_like(a)          # second‑moment estimate
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        # pick the indices of the top‑k peaks (including ties)
        if top_k >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(conv, -top_k)[-top_k:]
        # accumulate the sub‑gradient
        grad = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1)
            idx_j = k - idx_i
            grad[idx_i] += 2.0 * a[idx_j]

        # Adam moments on the raw gradient
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size per coordinate (multiplicative)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(-step)

        # keep on the simplex
        a = _project_onto_simplex(a)

        # occasional evaluation
        if t % 15 == 0:
            cur_score = _objective(a)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] iter={t} new best {best_score:.6f}")

    return best_seq, best_score

def _greedy_polish(a: np.ndarray, deadline: float,
                  best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Random mass‑transfer walk with greedy acceptance."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        # choose a source with non‑trivial mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        j = rng.integers(n)
        if j == i:
            continue
        # move a random fraction of the source mass
        delta = a[i] * rng.random() * 0.35
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur_score = _objective(new_a)
        if cur_score < best_score - 1e-12:
            a = new_a
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score

def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Search for a low‑autocorrelation non‑negative step function.
    Returns the best sequence discovered within the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ----- Stage 1: analytic seeding ---------------------------------
    lengths = [1500, 2000, 2500, 3000, 3500, 4000, 4500]
    elite: list[tuple[float, np.ndarray]] = []       # (score, sequence)
    best_seq = None
    best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] ---- length {n} grid search ----")
        candidates = _grid_candidates(n, rng)
        for sc, seq in candidates:
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time():.1f}] NEW GLOBAL BEST len={n} score={sc:.6f}")
            elite.append((sc, seq.copy()))
        # keep a global elite pool (size ≈ 15)
        elite.sort(key=lambda x: x[0])
        elite = elite[:15]

    # ----- incorporate previously known solution -----------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous solution improved: {sc:.6f}")
            elite.append((sc, prev.copy()))
            elite.sort(key=lambda x: x[0])
            elite = elite[:15]
    except Exception:
        pass

    # ----- Stage 2: Adam‑style refinement -----------------------------
    remaining = deadline - time.time()
    if remaining > 20.0 and elite:
        # split remaining time evenly among the elites
        per_seed = max(5.0, remaining / len(elite))
        for sc, seq in elite:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            seq_opt, sc_opt = _adam_expgd(seq.copy(), sub_deadline,
                                          best_seq, best_score,
                                          lr=0.15, top_k=6)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = seq_opt.copy()
                print(f"[{time.time():.1f}] after Adam new best {best_score:.6f}")

    # ----- Stage 3: greedy polishing ---------------------------------
    if best_seq is not None and time.time() < deadline:
        best_seq, best_score = _greedy_polish(best_seq.copy(),
                                              deadline,
                                              best_seq, best_score)

    # ----- final report ------------------------------------------------
    print(f"FINAL BEST score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
