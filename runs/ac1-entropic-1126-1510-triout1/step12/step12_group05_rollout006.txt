"""
Hybrid optimiser for the self‑convolution max‑ratio problem.

The routine works inside a fixed time budget (default 1000 s) and
performs three stages:

1. **Seed generation** – a large collection of analytic and random
   distributions (uniform, arcsine, beta, Dirichlet, exponential,
   power‑law, sparse, two‑block, sine) is built for a few different
   lengths.  All candidates are normalised to sum = 1 and their exact
   objective 2·n·max(conv)/sum² is evaluated.

2. **Two‑phase descent** – the best few seeds are refined.
   *Phase A* runs a soft‑max exponentiated‑gradient descent that
   approximates the max‑convolution entry with a temperature‑controlled
   soft‑max.  *Phase B* switches to a hard‑max sub‑gradient descent
   (gradient of the true maximal entry).  Both phases operate on the
   probability simplex via exponentiated‑gradient updates and a simple
   renormalisation.

3. **Simulated‑annealing polishing** – a cheap random mass‑transfer
   walk (move a random fraction of mass from one index to another)
   is performed while accepting uphill moves with a temperature that
   decays to zero.  The walk keeps the best sequence ever seen.

All helper functions are top‑level (no closures) and the code respects
the provided time budget.  The final best sequence is returned as a
plain list of floats.
"""

import time

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a))/sum(a)^2.
    The vector a is assumed to be non‑negative.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Sub‑gradient 2·a⋆w for a weight vector w (length 2n‑1, Σw = 1)."""
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.size - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _gradient_max_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the single maximal convolution entry conv[k].
    Returns a vector of size len(a) with non‑negative entries.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i = np.arange(i_min, i_max + 1, dtype=np.int64)
    j = k - i

    eq_mask = (i == j)
    if np.any(eq_mask):
        g[i[eq_mask]] += 2.0 * a[i[eq_mask]]

    if np.any(~eq_mask):
        ii = i[~eq_mask]
        jj = j[~eq_mask]
        g[ii] += a[jj]
        g[jj] += a[ii]

    return g

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * x / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Broad pool of analytic seeds for a given length."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_sine(n))
    cand.append(_distribution_two_blocks(n, block_frac=0.07))

    # symmetric beta on a fine grid
    for p in np.linspace(0.05, 1.0, 12):
        cand.append(_distribution_beta_sym(n, p))

    # random asymmetric beta
    for _ in range(8):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # power law
    for p in (0.5, 1.0, 2.0, 3.0):
        cand.append(_distribution_power_increasing(n, p))

    # exponential variants
    for lam in (0.001, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential_increasing(n, lam))

    # sparse random draws
    for dens in (0.005, 0.01, 0.02):
        cand.append(_distribution_random_sparse(n, rng, density=dens))

    # Dirichlet draws with various concentrations
    for a in (0.1, 0.3, 0.5, 1.0):
        cand.append(_distribution_dirichlet(n, rng, alpha=a))

    return cand


# ----------------------------------------------------------------------
def _softmax_grad_descent(a: np.ndarray, deadline: float,
                          best_seq: np.ndarray, best_score: float,
                          lr0: float = 0.9,
                          max_iter: int = 200000,
                          eval_interval: int = 500,
                          alpha_start: float = 0.2,
                          alpha_end: float = 80.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a soft‑max approximation of the
    maximal convolution entry.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_conv = conv.max()

        # linear temperature schedule
        alpha = alpha_start + (alpha_end - alpha_start) * t / max_iter

        # stable soft‑max weights
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # renormalise onto the simplex
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= total

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] soft‑max it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score

def _max_grad_descent(a: np.ndarray, deadline: float,
                      best_seq: np.ndarray, best_score: float,
                      lr0: float = 0.5,
                      max_iter: int = 200000,
                      eval_interval: int = 500) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent using the true maximal entry of the convolution.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())

        grad = _gradient_max_index(a, k)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= total

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hard‑max it={t} idx={k} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer walk with a cooling schedule.
    """
    n = a.shape[0]
    cur_seq = a.copy()
    cur_score = _objective(cur_seq)

    # parameters for the walk
    max_frac = 0.5          # up‑to‑50 % of source mass can be moved
    eps = 1e-15

    # cooling schedule parameters
    T_start = 5e-3
    T_end = 1e-8
    start_time = time.time()
    total_time = max(deadline - start_time, 1e-9)

    while time.time() < deadline:
        # linear progress of the annealing (0 → 1)
        elapsed = time.time() - start_time
        prog = elapsed / total_time
        temp = T_start * (T_end / T_start) ** prog

        # pick a source with non‑trivial mass
        src_candidates = np.where(cur_seq > eps)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        j = rng.integers(n)
        if j == i:
            continue

        # propose a transfer
        delta = cur_seq[i] * rng.random() * max_frac
        if delta < eps:
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_score = _objective(new_seq)

        # accept if improvement, otherwise accept with Metropolis probability
        if new_score < cur_score:
            cur_seq, cur_score = new_seq, new_score
        else:
            accept_prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < accept_prob:
                cur_seq, cur_score = new_seq, new_score

        # keep the best ever seen
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = cur_seq.copy()
            print(f"[{time.time():.1f}] SA improve to {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Main entry point.  Returns a list of non‑negative heights that
    (hopefully) minimise the evaluation function.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    elite = []   # (score, seq) tuples across all lengths

    # ------------------------------------------------------------------
    # 1. Scan a broad pool of analytic and random seeds for several lengths
    lengths = [8000, 12000, 20000]   # longer gives more flexibility
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] --- scanning length = {n} ---")
        # analytic seeds
        candidates = _initial_candidates_extended(n, rng)
        # add a few pure Dirichlet draws for extra diversity
        for _ in range(6):
            candidates.append(_distribution_dirichlet(n, rng, alpha=0.6))

        for cand in candidates:
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] !!! new global best len={n} score={sc:.6f}")
            elite.append((sc, cand.copy()))

        # keep only the most promising across all lengths so far
        elite.sort(key=lambda x: x[0])
        elite = elite[:12]

    # ------------------------------------------------------------------
    # 2. Incorporate previously found sequence if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best improved: {sc:.6f}")
            elite.append((sc, prev.copy()))
            elite.sort(key=lambda x: x[0])
            elite = elite[:12]
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never be needed)
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Two‑phase refinement on the elite seeds
    remaining = deadline - time.time()
    if remaining > 20.0 and elite:
        # take the top few elite seeds for a deep optimisation
        seeds = [seq for _, seq in sorted(elite, key=lambda x: x[0])[:3]]
        per_seed_time = max(2.0, remaining / (len(seeds) * 2.0))

        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            print(f"[{time.time():.1f}] Refining seed {idx+1}/{len(seeds)} (n={len(seed_seq)})")

            # Phase A – soft‑max
            sub_deadline_a = time.time() + per_seed_time
            seq_a, score_a = _softmax_grad_descent(
                seed_seq.copy(),
                sub_deadline_a,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=1.0,
                max_iter=300000,
                eval_interval=400,
                alpha_start=0.2,
                alpha_end=80.0,
            )
            if score_a < overall_best_score:
                overall_best_score = score_a
                overall_best_seq = seq_a.copy()
                print(f"[{time.time():.1f}] after soft‑max new best {overall_best_score:.6f}")

            # Phase B – hard‑max
            if time.time() >= deadline:
                break
            sub_deadline_b = time.time() + per_seed_time
            seq_b, score_b = _max_grad_descent(
                seq_a.copy(),
                sub_deadline_b,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.5,
                max_iter=300000,
                eval_interval=400,
            )
            if score_b < overall_best_score:
                overall_best_score = score_b
                overall_best_seq = seq_b.copy()
                print(f"[{time.time():.1f}] after hard‑max new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final polishing with simulated‑annealing walk
    if time.time() < deadline:
        print(f"[{time.time():.1f}] Starting final SA polishing, time left {deadline - time.time():.1f}s")
        overall_best_seq, overall_best_score = _simulated_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
