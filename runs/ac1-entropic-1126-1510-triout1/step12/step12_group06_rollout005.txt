"""
Hybrid optimizer for the max‑convolution step‑function problem.

Main ideas
~~~~~~~~~~
*  Generate a broad set of seed vectors (uniform, arcsine, many Beta/Dirichlet
   families, sparse patterns, previous best if any).
*  Run a smooth‑max Adam descent on the soft‑max approximation of the
   convolution maximum.  The temperature is increased slowly so that the
   optimiser first sees a smooth landscape and then focuses on the true max.
*  Refine the current best with a *linear‑programming* step:
   maximise the total mass of a new vector `b` under the linear constraints
   `conv(a,b) ≤ M` where `M` is the present maximum convolution value.
   The LP yields an extreme‑point distribution that often reduces the peak
   autocorrelation dramatically.
*  After the LP, run a few iterations of the classic multimax sub‑gradient
   descent and finish with cheap greedy pairwise mass‑transfer swaps.
*  All updates stay on the probability simplex (sum = 1, non‑negative) by a
   Euclidean projection.

The routine respects a time budget (default 1000 s) and prints intermediate
scores.  The best sequence found is returned as a plain list of floats.
"""

import time
import math
import numpy as np
from typing import List, Tuple
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·(a⋆a)[k] w.r.t. a.
    Implements 2·a⋆w using FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a))/sum(a)^2.
    The caller normally keeps sum(a)=1, therefore this reduces to 2·n·max(conv).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discretised unit interval."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(alpha,beta) on the discretised unit interval."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Few non‑zero coordinates, random values."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    """Two symmetric spikes of equal mass."""
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals

def _initial_candidates_extended_fine(n: int,
                                      rng: np.random.Generator) -> List[np.ndarray]:
    """Rich family of seed vectors for a given length."""
    cand: List[np.ndarray] = []

    # Basic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # Symmetric Beta – log‑spaced exponents
    for p in np.geomspace(0.01, 5.0, num=15):
        cand.append(_distribution_beta_sym(n, float(p)))

    # Random asymmetric Beta draws
    for _ in range(8):
        alpha = 10 ** rng.uniform(-3, 1)   # 0.001 … 10
        beta = 10 ** rng.uniform(-3, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # Power‑law increasing
    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))

    # Exponential families
    for lam in (0.001, 0.003, 0.005, 0.01):
        cand.append(_distribution_exponential_increasing(n, lam))

    # Two‑block spikes with different fractions
    for bf in (0.03, 0.05, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac=bf))

    # Dirichlet draws with varied concentration
    for a in (0.1, 0.3, 0.7, 1.0):
        cand.append(_distribution_dirichlet(n, rng, alpha=a))

    # Random sparse patterns
    cand.append(_distribution_random_sparse(n, rng, density=0.01))
    cand.append(_distribution_random_sparse(n, rng, density=0.03))

    # A few completely random probability vectors
    for _ in range(6):
        v = rng.random(n)
        v = np.maximum(v, 0.0)
        v /= v.sum()
        cand.append(v)

    return cand

def _smooth_max_adam_descent(a: np.ndarray,
                             deadline: float,
                             best_seq: np.ndarray,
                             best_score: float,
                             eta0: float = 0.6,
                             beta1: float = 0.9,
                             beta2: float = 0.999,
                             eps: float = 1e-8,
                             alpha0: float = 0.3,
                             alpha_mul: float = 1.25,
                             alpha_max: float = 200.0,
                             max_iter: int = 60000,
                             eval_interval: int = 400) -> Tuple[np.ndarray, float]:
    """
    Adam‑style exponentiated‑gradient descent on the smooth‑max surrogate

        f_α(a) = (1/α)·log Σ_i exp(α·conv_i(a))

    The temperature α grows geometrically.  The update is multiplicative and
    projected onto the simplex.
    """
    n = a.shape[0]
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        # self‑convolution
        conv = _fft_self_convolve(a)

        # soft‑max weights (numerically stable)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient of the smooth surrogate
        grad = _gradient_from_weights(a, w)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # multiplicative update
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(-step)

        # project back onto the simplex (sum = 1)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # temperature schedule
        if t % 300 == 0 and alpha < alpha_max:
            alpha = min(alpha * alpha_mul, alpha_max)

        # periodic exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score

                                 lr0: float = 0.6,
                                 max_iter: int = 35000,
                                 eval_interval: int = 500) -> Tuple[np.ndarray, float]:
    """
    Classic exponentiated‑gradient descent where the sub‑gradient is the
    average over all active indices attaining the maximal convolution value.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()

        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)

        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Multimax it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Greedy pairwise mass‑transfer: move a tiny amount from i to j if it helps."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        # distinct destination
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # random transfer amount (up to 30 % of the source mass)
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray,
               deadline: float,
               max_active: int = 150) -> Tuple[np.ndarray, float]:
    """
    Linear‑programming refinement.

    For the current vector `a` we compute M = max(conv(a,a)).
    We then maximise Σ b_i subject to
          conv(a, b) ≤ M    (elementwise)
          b_i ≥ 0.
    The LP solution (scaled back to unit sum) often yields a strictly
    smaller peak autocorrelation.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _objective(a)

    while time.time() < deadline:
        conv = _fft_self_convolve(a)
        M = conv.max()

        # active rows – take the highest entries (up to max_active)
        idx_desc = np.argsort(-conv)                     # descending order
        rows = idx_desc[:max_active]                     # at most max_active rows
        # Build dense constraint matrix A (rows × n)
        m_active = rows.size
        A = np.zeros((m_active, n), dtype=np.float64)
        for r_idx, k in enumerate(rows):
            j_start = max(0, k - (n - 1))
            j_end = min(k, n - 1)
            js = np.arange(j_start, j_end + 1, dtype=np.int64)
            A[r_idx, js] = a[k - js]

        # Linear program: maximise sum(b)  → minimise -sum(b)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n
        b_ub = np.full(m_active, M, dtype=np.float64)

        # Solve with HiGHS (fast, supports dense A)
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs',
                          options={'presolve': True,
                                   'time_limit': max(0.0, deadline - time.time())})
        except Exception as exc:
            print("LP exception:", exc)
            break

        if not res.success:
            # LP didn't solve – break out
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            break

        # Scale to unit sum (the objective is scale‑invariant)
        b_norm = b / sum_b

        new_score = _objective(b_norm)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_seq = b_norm.copy()
            a = b_norm                                   # continue refinement
            print(f"[{time.time():.1f}] LP new best {best_score:.6f}")
        else:
            # No improvement – stop the LP loop
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects ``budget_s`` seconds and
    returns the best sequence found (as a plain list of floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a handful of lengths – longer gives flexibility but costs more.
    lengths = [3000, 4000, 6000]                # a compromise between quality & speed

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: List[Tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 2. Scan cheap analytic families and keep a few promising seeds.
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Scanning length {n}")

        cand_list = _initial_candidates_extended_fine(n, rng)

        for cand in cand_list:
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                continue
            cand /= s                                 # normalise to sum = 1
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] **** new global best len={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))

        # keep only the most promising seeds across all lengths
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:12]

    # ------------------------------------------------------------------
    # 3. Incorporate a previous solution if present.
    try:
        prev = globals().get("height_sequence_1")
        if isinstance(prev, (list, np.ndarray)):
            prev = np.asarray(prev, dtype=np.float64)
            if prev.ndim == 1 and prev.size > 0:
                prev = np.maximum(prev, 0.0)
                s = prev.sum()
                if s == 0.0:
                    prev = _distribution_uniform(prev.shape[0])
                    s = prev.sum()
                prev /= s
                sc = _objective(prev)
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = prev.copy()
                    print(f"[{time.time():.1f}] previous solution improved: {sc:.6f}")
                top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Fallback – should never be needed.
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. First refinement: Adam‑style smooth‑max descent on a handful
    #    of seeds (global best plus a few top candidates).
    remaining = deadline - time.time()
    if remaining > 60:
        seeds = [overall_best_seq.copy()] + [
            seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:4]
        ]
        for seed_seq in seeds:
            if time.time() >= deadline:
                break
            # give each seed a slice of the remaining budget
            sub_deadline = time.time() + (deadline - time.time()) / max(1, len(seeds))
            seq_out, score_out = _smooth_max_adam_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.6,
                beta1=0.9,
                beta2=0.999,
                eps=1e-8,
                alpha0=0.2,
                alpha_mul=1.2,
                alpha_max=200.0,
                max_iter=50000,
                eval_interval=400,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after Adam descent new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Linear‑programming refinement (often gives a big jump).
    remaining = deadline - time.time()
    if remaining > 30:
        sub_deadline = time.time() + remaining
        seq_out, score_out = _lp_refine(overall_best_seq.copy(),
                                        sub_deadline,
                                        max_active=150)
        if score_out < overall_best_score:
            overall_best_score = score_out
            overall_best_seq = seq_out.copy()
            print(f"[{time.time():.1f}] after LP refine new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Second refinement: classic multimax sub‑gradient descent.
    remaining = deadline - time.time()
    if remaining > 20:
        seeds = [overall_best_seq.copy()] + [
            seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:2]
            if not np.may_share_memory(seq, overall_best_seq)
        ]
        for seed_seq in seeds:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + (deadline - time.time()) / max(1, len(seeds))
            seq_out, score_out = _subgradient_multimax_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.6,
                max_iter=30000,
                eval_interval=400,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after multimax new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final cheap polishing with greedy pairwise swaps.
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
```

This implementation expands on the previous hybrid optimizer by adding a **linear‑programming refinement step**, which often yields a noticeable drop in the maximum self‑convolution.  It also tightens the temperature schedule, uses a modest set of lengths, and keeps the overall running time comfortably below the 1000‑second budget while aiming for an objective value well below the earlier 1.509956 target.
