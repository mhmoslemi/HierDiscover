def build_constraint_matrix(f):
    n = len(f)
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n-1)
        # i range inclusive
        # j = k - i; j range
        for i in range(i_start, i_end + 1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n - 1, n))
    return A

def line_search_update(f, g, M_current, t_vals):
    best_f = f
    best_obj = _objective(f)
    for t in t_vals:
        candidate = (1 - t) * f + t * g
        candidate /= candidate.sum()
        obj = _objective(candidate)
        if obj < best_obj:
            best_obj = obj
            best_f = candidate
    return best_f, best_obj

"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.

The approach mixes three ideas:

1. **A diverse analytic pool** – uniform, arcsine, beta‑type, edge‑weighted,
   sine‑weighted and random Dirichlet / sparse vectors for a few lengths.
   The best few seeds are kept.

2. **Linear‑programming refinement** – given a current sequence `f` with
   `M = max(conv(f,f))`, we solve a linear programme

        max   Σ g_i
        s.t.  conv(f,g) ≤ M   (all 2n‑1 shifts)
              g_i ≥ 0

   respecting the current worst‑case bound `M`.  A small line‑search over
   the mixing weight `t` (0 < t ≤ 0.5) is performed analytically using the

3. **Fine‑tuning** – the best result is polished with an exponentiated‑gradient
   descent (using the exact sub‑gradient of the max‑convolution) and finally
   a cheap stochastic local search (pairwise swaps + simulated annealing).

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b (real FFT)."""
    n, m = a.shape[0], b.shape[0]
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families -----------------------------------------------------

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric beta (Beta(a,b))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight proportional to sin(π (i+½)/n)^p."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Linear‑programming refinement ------------------------------------------

def _build_constraint_matrix(f: np.ndarray) -> sp.csc_matrix:
    """
    Build the (2n‑1) × n matrix A such that (A @ g)[k] = conv(f,g)[k].
    Because each diagonal of A is constant, we can construct it cheaply
    with ``scipy.sparse.diags``.
    """
    n = f.shape[0]
    data = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n, dtype=np.int32)          # 0,…,n‑1
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csc')
    return A


def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve  max Σ g_i   s.t.  conv(f,g) ≤ M,   g ≥ 0 .
    Returns the optimal g (may be the zero vector if M is too small).
    """
    n = f.shape[0]
    A = _build_constraint_matrix(f)               # (2n‑1)×n, dense diagonals
    b = np.full(2 * n - 1, M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)              # minimise -Σg ⇒ maximise Σg
    bounds = [(0.0, None)] * n

    # Use HiGHS – works well for large sparse problems.
    res = opt.linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    g = np.maximum(res.x, 0.0)
    return g


def _lp_refine(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_iters: int = 30,
) -> tuple[np.ndarray, float]:
    """
    Iteratively improve ``a`` by solving the LP described above and
    mixing the solution back with a line‑search over the mixing weight.
    """
    n = a.shape[0]
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        conv_ff = _fft_self_convolve(a)
        M = conv_ff.max()
        # solving the LP
        try:
            g = _solve_lp_max_sum(a, M)
        except Exception as e:
            print(f"[{time.time():.1f}] LP error: {e}")
            break

        sg = g.sum()
        if sg < 1e-12:                     # no useful direction
            break

        # pre‑compute convolutions needed for the line‑search
        conv_fg = _fft_convolve(a, g)
        conv_gg = _fft_self_convolve(g)

        # try a few mixing coefficients
        candidate_ts = [0.05, 0.10, 0.15, 0.20, 0.30, 0.40, 0.50]
        local_best = best_score
        best_t = 0.0
        for t in candidate_ts:
            s_new = (1.0 - t) + t * sg                     # new sum before renorm
            combined = (
                (1.0 - t) ** 2 * conv_ff
                + 2.0 * t * (1.0 - t) * conv_fg
                + t ** 2 * conv_gg
            )
            max_c = combined.max() / (s_new ** 2)
            obj = 2.0 * n * max_c
            if obj < local_best - 1e-14:
                local_best = obj
                best_t = t

        if best_t > 0.0:
            # perform the chosen mixing and renormalise to sum = 1
            a = (1.0 - best_t) * a + best_t * g
            a = a / a.sum()
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] LP‑iter it={it} t={best_t:.3f} new best {best_score:.6f}")
        else:
            # no improvement possible – early exit
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent (exact sub‑gradient) --------------------

def _exact_subgradient(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at the current maximising indices.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_val = conv.max()
    active = np.where(np.isclose(conv, max_val, atol=1e-15))[0]

    # accumulate gradient contributions from each active shift
    grad = np.zeros_like(a)
    ind_grid = np.arange(n, dtype=np.int32)

    for k in active:
        # indices where a_{k-j} exists
        mask = (k - ind_grid >= 0) & (k - ind_grid < n)
        shifted = np.zeros_like(a)
        shifted[mask] = a[k - ind_grid[mask]]
        # each contribution appears twice except the centre point when k is even
        gk = 2.0 * shifted
        if k % 2 == 0:
            j = k // 2
            if j < n:
                gk[j] -= a[j]      # correct double‑count at the diagonal
        grad += gk
    # average over active set
    grad /= max(len(active), 1)
    return grad


def _smooth_exponential_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.7,
    eval_interval: int = 80,
    patience: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) using the *exact* sub‑gradient
    of the max‑convolution.  No soft‑max is used – the active set is
    computed exactly, which often yields a sharper descent.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    while time.time() < deadline and t < 300_000:
        t += 1
        grad = _exact_subgradient(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # renormalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # occasional evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] expo‑grad it={t} new best {best_score:.6f}")
            else:
                no_improve += 1

        # jitter if stuck
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.005, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0
    return best_seq, best_score


# ----------------------------------------------------------------------
# Local refinements -----------------------------------------------------

def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small chunk of mass between two positions."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Very cheap simulated‑annealing style local walk."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

# Main entry point -------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns a plain Python list (the caller will handle the final sanity checks).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 3.0   # slight safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Diverse analytic / random seeds
    lengths = [2000, 3000, 4000, 5000]
    p_grid = np.concatenate(
        [
            np.logspace(-2, -1, 9),   # 0.01 .. 0.1
            np.logspace(-1, 0, 9),    # 0.1 .. 1
            np.logspace(0, 0.5, 7),   # 1 .. 3.16
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 6)  # 0,0.2,…,1

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # parametrised families + mixing w/ uniform
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            sine = _distribution_sine_weighted(n, p)
            for base in (beta, edge, sine):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # asymmetric beta random draws
        for _ in range(4):
            if time.time() >= deadline:
                break
            a_param = 10 ** rng.uniform(-2, 0)   # 0.01 – 1
            b_param = 10 ** rng.uniform(-2, 0)
            cand = _distribution_beta_asym(n, a_param, b_param)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best asym‑beta len={n} a={a_param:.3g} b={b_param:.3g} score={sc:.6f}"
                )

        # random Dirichlet seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best Dirichlet len={n} score={sc:.6f}"
                )

        # sparse random seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}"
                )

        # keep a manageable shortlist
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:40]

    # ------------------------------------------------------------------
    # 2. Include the previously discovered construction if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never be needed)
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Linear‑programming refinement on the most promising seeds
    if time.time() < deadline:
        top_candidates.sort(key=lambda x: x[0])
        seeds = [seq for _, seq in top_candidates[:5]]
        per_seed = (deadline - time.time()) / max(1, len(seeds))
        print(f"[{time.time():.1f}] LP‑refinement on {len(seeds)} seeds, {per_seed:.1f}s each")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            a_opt, sc_opt = _lp_refine(
                seed_seq.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                max_iters=25,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after LP seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5. Exponentiated‑gradient polishing (exact sub‑gradient)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] polishing with exponentiated‑gradient")
        overall_best_seq, overall_best_score = _smooth_exponential_descent(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.7,
            eval_interval=80,
            patience=2000,
        )

    # ------------------------------------------------------------------
    # 6. Pairwise greedy refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=200_000,
        )

    # ------------------------------------------------------------------
    # 7. Final cheap simulated‑annealing walk
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final random‑local search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
