def _cross_correlation(a, p):
    # compute cross-correlation (a ⋆ p)[j] = Σ_i a_i * p_{i + j}
    # Equivalent to convolution of a with reversed p.
    n = a.shape[0]
    len_p = p.shape[0]  # = 2n-1
    need = n + len_p - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    P_rev = rfft(p[::-1], n=size)
    conv = irfft(A * P_rev, n=size)
    # conv length = size, but we need slice start = len_p-1 (=2n-2) to start+ n
    start = len_p - 1
    return conv[start:start + n]

"""
Propose a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.

Algorithm
---------

1. **Generate a diverse pool of seeds** (analytic families, random Dirichlet,
   sparse vectors, non‑symmetric Beta, mixtures with the uniform distribution).
   All seeds are normalised to sum = 1 and their exact objective
   ``2·n·max(conv(a,a))`` is evaluated.  The ~5 best seeds are kept.

2. **Adam optimiser on the soft‑max approximation of the max‑convolution**.
   The distribution is parametrised as a soft‑max of free parameters θ
   (θ_i = log w_i,  a_i = exp(θ_i)/∑ exp(θ)).  The “max’’ is replaced by a
   temperature‑controlled soft‑max:
       softmax_α(c) = (1/α)·log∑_k exp(α·c_k) .
   The gradient of the smoothed objective is obtained analytically and the
   cross‑correlation required for the gradient is computed with a real FFT.
   The temperature α is increased gradually (annealing) so that the optimiser
   first sees a smooth landscape and finally focuses on the true maximum.

3. **Fine‑grained local polishing** – a deterministic pairwise‑mass‑transfer
   and a cheap simulated‑annealing phase (identical to the previous
   implementation) are applied to the best sequence found.

All steps respect the supplied time budget (default 1000 s) and the
returned list is a plain Python list of non‑negative floats.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Utility FFT helpers ---------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _cross_correlation(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation (a ⋆ p)[j] = Σ_i a_i·p_{i+j},
    where len(p)=2·n‑1, len(a)=n.
    Implemented by FFT of a and reversed p.
    Returns an array of length n.
    """
    n = a.shape[0]
    m = p.shape[0]            # = 2·n‑1
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    Pref = rfft(p[::-1], n=size)   # reversed p
    conv = irfft(A * Pref, n=size)
    start = m - 1                   # = 2·n‑2
    return conv[start:start + n]


# ----------------------------------------------------------------------
# Exact objective (sum of sequence = 1) -------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric beta (Beta(a,b))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0) when a or b < 1
    with np.errstate(divide="ignore"):
        w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x))
    w[np.isnan(w)] = 0.0
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped: w_i ∝ ((i+0.5)(n‑i‑0.5))^{−p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    """U‑shaped: w_i ∝ (i+0.5)^{−p} + (n−i−0.5)^{−p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (-p) + j ** (-p))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ sin(π·(i+½)/n)^p."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet sample with a random concentration."""
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    """Sparse random vector (few non‑zeros)."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base (U is uniform)."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max approximation -------------------------
def _adam_optimize(
    seq_init: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    start_alpha: float = 2.0,
    final_alpha: float = 200.0,
    alpha_mul: float = 1.003,
) -> tuple[np.ndarray, float]:
    """
    Optimises a normalised sequence using Adam on a soft‑max approximation
    of the max‑convolution.
    """
    n = seq_init.shape[0]
    # keep a tiny epsilon to avoid log(0)
    eps_small = 1e-12
    a = np.maximum(seq_init, eps_small)
    a /= a.sum()
    theta = np.log(a)                     # unrestricted parameters
    m = np.zeros_like(theta)
    v = np.zeros_like(theta)

    t = 0
    alpha = start_alpha

    while time.time() < deadline:
        t += 1

        # ----- from parameters to probability vector -----
        max_theta = theta.max()
        w = np.exp(theta - max_theta)     # stabilised soft‑max
        a = w / w.sum()

        # ----- exact objective for bookkeeping -----
        conv = _fft_self_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score - 1e-14:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Adam improvement {cur_obj:.6f}")

        # ----- temperature annealing -----
        if alpha < final_alpha:
            alpha = min(final_alpha, alpha * alpha_mul)

        # ----- soft‑max probabilities over convolution entries -----
        scaled = conv * alpha
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)
        p = exp_scaled / exp_scaled.sum()          # length 2n‑1, sums to 1

        # ----- gradient wrt a (smooth version) -----
        # grad_a = 4·n·(a ⋆ p)   (cross‑correlation)
        grad_base = _cross_correlation(a, p)       # length n
        grad_a = 4.0 * n * grad_base

        # ----- gradient wrt theta (soft‑max Jacobian) -----
        dot = np.dot(a, grad_a)
        grad_theta = a * (grad_a - dot)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_theta
        v = beta2 * v + (1.0 - beta2) * (grad_theta ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        theta -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional re‑centering to avoid overflow
        if t % 5000 == 0:
            theta -= theta.mean()

    return best_seq, best_score

# Deterministic pairwise refinement ------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small mass; keep the move only if it improves the exact objective."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing local search --------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    """Cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            # metropolis step
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score
