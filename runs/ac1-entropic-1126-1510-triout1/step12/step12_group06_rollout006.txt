#!/usr/bin/env python
"""
Hybrid optimiser for the step‑function max‑convolution problem.

Key ideas
~~~~~~~~~
*   Initialise a diverse set of probability vectors on several
    lengths (powers of two are preferred for FFT speed).
*   Keep only the most promising seeds (lowest objective) and devote
    the bulk of the budget to a refined Adam‑style optimisation on a
    smooth‑max surrogate of the max‑convolution.
    – the temperature α is increased aggressively up to a very large
      value (≈400) so that the surrogate eventually coincides with the
      true max.
    – after each exponentiated‑gradient step the vector is projected onto
      the probability simplex *and* forced to be symmetric (a_i = a_{n‑1‑i}),
      a property that empirical optimums seem to respect.
*   Perform a cheap combinatorial mixing of the best few candidates:
    convex combinations of two already good vectors often smooth out the
    remaining peaks.
*   Finish with a fast greedy pair‑wise mass‑transfer polishing step.
*   All stages respect the global time‑budget (default 1000 s); the
    routine prints intermediate scores and returns the best sequence
    found.

The returned list can be fed directly to ``evaluate_sequence``.
"""

import time
import math
import itertools
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the (soft‑)max‑convolution objective.
    w has length 2·n‑1 and sums to 1.
    The result is 2·a⋆w (cross‑correlation) of the same length as a.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a))/sum(a)^2.
    The vector is assumed to be non‑negative; we normalise it to sum‑1
    (the scale does not affect the value).
    """
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    a = a / s
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    return 2.0 * n * max_conv

# Analytic families – they return *probability* vectors (sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discretised unit interval."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(alpha,beta) on the discretised unit interval."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Few non‑zero coordinates, random values."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    """Two symmetric spikes of equal mass."""
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals

def _initial_candidates_extended_fine(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    """Generate a rich pool of seed vectors for a given length."""
    cand = []

    # Basic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # Symmetric Beta – log‑spaced exponents
    for p in np.geomspace(0.01, 5.0, num=15):
        cand.append(_distribution_beta_sym(n, float(p)))

    # Random asymmetric Beta draws
    for _ in range(8):
        alpha = 10 ** rng.uniform(-3, 1)   # 0.001 … 10
        beta = 10 ** rng.uniform(-3, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # Power‑law increasing
    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))

    # Exponential families
    for lam in (0.001, 0.003, 0.005, 0.01):
        cand.append(_distribution_exponential_increasing(n, lam))

    # Two‑block spikes with different fractions
    for bf in (0.03, 0.05, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac=bf))

    # Dirichlet draws with varied concentration
    for a in (0.1, 0.3, 0.7, 1.0):
        cand.append(_distribution_dirichlet(n, rng, alpha=a))

    # Random sparse patterns
    cand.append(_distribution_random_sparse(n, rng, density=0.01))
    cand.append(_distribution_random_sparse(n, rng, density=0.03))

    # Purely random vectors (few of them)
    for _ in range(6):
        v = rng.random(n)
        v = np.maximum(v, 0.0)
        v /= v.sum()
        cand.append(v)

    return cand

                             alpha_mul: float = 1.25,
                             alpha_max: float = 150.0,
                             max_iter: int = 60000,
                             eval_interval: int = 500) -> Tuple[np.ndarray, float]:
    """
    Adam‑style exponentiated‑gradient on the smooth‑max surrogate

        f_α(a) = (1/α)·log Σ_i exp(α·conv_i(a))

    The temperature α starts at ``alpha0`` and is increased geometrically.
    After each update the vector is projected onto the probability simplex
    and forced to be symmetric (a_i = a_{n‑1‑i}) – a heuristic that tends
    to lower the maximum convolution value.
    """
    n = a.shape[0]
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        # full self‑convolution
        conv = _fft_self_convolve(a)

        # soft‑max weights (stable)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient of the smooth surrogate
        grad = _gradient_from_weights(a, w)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # multiplicative update
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(-step)

        # projection onto simplex
        a = _project_onto_simplex(a, s=1.0)

        # enforce symmetry (helps flatten the convolution)
        a = 0.5 * (a + a[::-1])
        a = a / a.sum()                # renormalise after symmetrisation

        # temperature schedule
        if t % 300 == 0 and alpha < alpha_max:
            alpha = min(alpha * alpha_mul, alpha_max)

        # periodic exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.6,
                                 max_iter: int = 35000,
                                 eval_interval: int = 400) -> Tuple[np.ndarray, float]:
    """
    Classic exponentiated‑gradient descent using the true sub‑gradient:
    the average over all indices attaining the maximal convolution value.
    Symmetrisation is performed after each update.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()

        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)

        a = a * np.exp(-eta * grad)

        a = _project_onto_simplex(a, s=1.0)

        # symmetrisation
        a = 0.5 * (a + a[::-1])
        a = a / a.sum()

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Multimax it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Greedy pairwise mass‑transfer: move a small amount from i to j if it improves the objective."""
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        # distinct destination
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # random transfer amount (up to 30 % of the source mass)
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects ``budget_s`` seconds and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a modest set of lengths (powers of two help the FFT).
    lengths = [2048, 4096, 8192, 16384]
    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: List[Tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 2. Scan analytic families (cheap) and keep a short list of promising seeds.
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Scanning length {n}")

        cand_list = _initial_candidates_extended_fine(n, rng)

        # add a few extra Dirichlet draws with random concentration
        for _ in range(5):
            cand_list.append(_distribution_dirichlet(n, rng,
                                                     alpha=10 ** rng.uniform(-2, 1)))

        for cand in cand_list:
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                continue
            cand = cand / s
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] **** new global best len={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))

        # keep only the most promising seeds overall (independent of length)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:12]

    # ------------------------------------------------------------------
    # 3. Fallback – should never be needed.
    if overall_best_seq is None:
        n = 2048
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. First refinement: Adam‑style smooth‑max descent on the best seeds.
    # Only operate on the *best* length to keep the budget under control.
    best_len = len(overall_best_seq)
    same_len_candidates = [(sc, seq) for sc, seq in top_candidates if len(seq) == best_len]
    seeds = [overall_best_seq.copy()] + [seq for _, seq in sorted(same_len_candidates, key=lambda x: x[0])[:4]]

    remaining = deadline - time.time()
    if remaining > 60 and seeds:
        per_seed = max(1, int(remaining / (len(seeds) + 1)))
        for seed_seq in seeds:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            seq_out, score_out = _smooth_max_adam_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.8,
                beta1=0.9,
                beta2=0.999,
                eps=1e-8,
                alpha0=0.5,
                alpha_mul=1.2,
                alpha_max=400.0,
                max_iter=200000,
                eval_interval=400,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after Adam descent new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Second refinement: classic multimax sub‑gradient descent (shorter).
    remaining = deadline - time.time()
    if remaining > 30:
        # pick a couple of distinct seeds of the same length
        other_seeds = [seq for _, seq in sorted(same_len_candidates, key=lambda x: x[0])[:2]
                       if not np.may_share_memory(seq, overall_best_seq)]
        seeds2 = [overall_best_seq.copy()] + other_seeds
        per_seed = max(1, int(remaining / (len(seeds2) + 1)))
        for seed_seq in seeds2:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            seq_out, score_out = _subgradient_multimax_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.6,
                max_iter=100000,
                eval_interval=500,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after multimax new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Mixing of top candidates (convex combinations) – cheap but often useful.
    if time.time() < deadline:
        # take the best few candidates of the same length
        mix_pool = [(sc, seq) for sc, seq in top_candidates if len(seq) == best_len]
        mix_pool = sorted(mix_pool, key=lambda x: x[0])[:6]

        for (sc_i, seq_i), (sc_j, seq_j) in itertools.combinations(mix_pool, 2):
            if time.time() >= deadline:
                break
            for w in np.linspace(0.0, 1.0, 11):
                combo = w * seq_i + (1.0 - w) * seq_j
                combo = np.maximum(combo, 0.0)
                s = combo.sum()
                if s == 0.0:
                    continue
                combo /= s
                cur = _objective(combo)
                if cur < overall_best_score:
                    overall_best_score = cur
                    overall_best_seq = combo.copy()
                    print(f"[{time.time():.1f}] mix ({sc_i:.5f},{sc_j:.5f}) w={w:.2f} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final polishing with cheap greedy pairwise swaps.
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
