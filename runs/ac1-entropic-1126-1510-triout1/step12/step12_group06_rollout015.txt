#!/usr/bin/env python
"""
Hybrid optimiser for the step‑function max‑convolution problem (improved).

Key ideas
~~~~~~~~~
* Generate a rich pool of seed distributions (uniform, arcsine,
  symmetric & asymmetric Beta families, Dirichlet draws, sparse patterns,
  and several *mixture* families – e.g. convex combinations of two Beta
  laws). These are inexpensive to evaluate.
* Run a three‑phase local optimiser on each promising seed:
    1. Adam‑style mirror descent on a smooth‑max (log‑sum‑exp) surrogate.
    2. Classic multi‑max sub‑gradient descent (exact max of the convolution).
    3. Cheap greedy pairwise mass‑transfer polishing.
  All updates stay on the probability simplex.
* Use a two‑stage strategy:
    - First explore short vectors (n≈2000) where many seeds can be tested
      quickly and the landscape is cheap to navigate.
    - Upsample the best short solution to a long vector (n≈20000) and
      continue the three‑phase descent; larger n yields a finer
      approximation of the continuous optimum.
* When time is left, perturb the current best solution with small
  random noise and restart the three‑phase descent (simulated‑annealing
  style) to escape local minima.

The routine respects the 1000 s budget, prints intermediate scores,
and finally returns the best coefficient list found (as plain ``list``).
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the (smooth) max‑convolution surrogate.
    Returns 2·a ⋆ w where w (length 2·n‑1) sums to 1.
    Implemented via FFT cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)^2).
    The vector a is assumed to be non‑negative; sum(a) may differ from 1.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discretised unit interval."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(alpha,beta) on the discretised unit interval."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Few non‑zero coordinates, random values."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    """Two symmetric spikes of equal mass."""
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals

# Mixture families ------------------------------------------------------
def _distribution_mixture_beta(n: int,
                              p1: float, p2: float,
                              w: float,
                              cache: dict) -> np.ndarray:
    """Convex combination w·Beta(p1,p1) + (1‑w)·Beta(p2,p2)."""
    key1 = ('beta', p1)
    key2 = ('beta', p2)
    if key1 in cache:
        d1 = cache[key1]
    else:
        d1 = _distribution_beta_sym(n, p1)
        cache[key1] = d1
    if key2 in cache:
        d2 = cache[key2]
    else:
        d2 = _distribution_beta_sym(n, p2)
        cache[key2] = d2
    a = w * d1 + (1.0 - w) * d2
    a /= a.sum()
    return a


def _distribution_mixture_uniform_arcsine(n: int, w: float,
                                         cache: dict) -> np.ndarray:
    """w·uniform + (1‑w)·arcsine."""
    u = cache.get('uniform')
    if u is None:
        u = _distribution_uniform(n)
        cache['uniform'] = u
    a = cache.get('arcsine')
    if a is None:
        a = _distribution_arcsine(n)
        cache['arcsine'] = a
    mix = w * u + (1.0 - w) * a
    mix /= mix.sum()
    return mix

def _initial_candidates_extended_fine(n: int,
                                      rng: np.random.Generator) -> List[np.ndarray]:
    """
    Produce a broad collection of seed vectors for a given length.
    In addition to the families used previously, we also generate
    mixture families (beta‑beta, uniform‑arcsine).
    """
    cand = []
    cache = {}

    # Basic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # Symmetric Beta – dense log‑spaced grid
    for p in np.geomspace(0.001, 5.0, num=30):
        cand.append(_distribution_beta_sym(n, float(p)))

    # Extreme small p (U‑shaped)
    for p in (0.0005, 0.001, 0.002):
        cand.append(_distribution_beta_sym(n, float(p)))

    # Random asymmetric Beta draws
    for _ in range(10):
        alpha = 10 ** rng.uniform(-3, 1)   # 0.001 … 10
        beta = 10 ** rng.uniform(-3, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # Power‑law increasing
    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))

    # Exponential families
    for lam in (0.001, 0.003, 0.005, 0.01):
        cand.append(_distribution_exponential_increasing(n, lam))

    # Two‑block spikes
    for bf in (0.02, 0.04, 0.07, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac=bf))

    # Dirichlet draws with varied concentration
    for a in (0.1, 0.3, 0.7, 1.0):
        cand.append(_distribution_dirichlet(n, rng, alpha=a))

    # Random sparse patterns
    cand.append(_distribution_random_sparse(n, rng, density=0.01))
    cand.append(_distribution_random_sparse(n, rng, density=0.03))

    # Completely random probability vectors
    for _ in range(8):
        v = rng.random(n)
        v = np.maximum(v, 0.0)
        v /= v.sum()
        cand.append(v)

    # Mixture families -------------------------------------------------
    # uniform–arcsine
    for w in (0.2, 0.5, 0.8):
        cand.append(_distribution_mixture_uniform_arcsine(n, w, cache))

    # beta‑beta mixtures
    beta_ps = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for p1 in beta_ps:
        for p2 in beta_ps:
            if p1 == p2:
                continue
            for w in (0.3, 0.5, 0.7):
                cand.append(_distribution_mixture_beta(n, p1, p2, w, cache))

    return cand

def _smooth_max_adam_descent(a: np.ndarray,
                             deadline: float,
                             best_seq: np.ndarray | None,
                             best_score: float,
                             eta0: float = 0.6,
                             beta1: float = 0.9,
                             beta2: float = 0.999,
                             eps: float = 1e-8,
                             alpha0: float = 0.2,
                             alpha_mul: float = 1.3,
                             alpha_max: float = 300.0,
                             max_iter: int = 80000,
                             eval_interval: int = 400) -> Tuple[np.ndarray, float]:
    """
    Adam‑style exponentiated‑gradient descent on the smooth‑max
    surrogate    f_α(a) = (1/α)·log Σ_i exp(α·conv_i(a)).

    The temperature α is increased geometrically.  Updates are
    multiplicative and then projected onto the simplex.
    """
    n = a.shape[0]
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        # full self‑convolution
        conv = _fft_self_convolve(a)

        # soft‑max weights (stable)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient of the smooth surrogate
        grad = _gradient_from_weights(a, w)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # multiplicative update
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(-step)

        # project onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # temperature schedule
        if t % 250 == 0 and alpha < alpha_max:
            alpha = min(alpha * alpha_mul, alpha_max)

        # periodic exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq if best_seq is not None else a, best_score


# ----------------------------------------------------------------------
def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray | None,
                                 best_score: float,
                                 lr0: float = 0.6,
                                 max_iter: int = 50000,
                                 eval_interval: int = 500) -> Tuple[np.ndarray, float]:
    """
    Classic exponentiated‑gradient descent using the exact sub‑gradient
    defined by the active (maximal) convolution indices.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)

        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Multimax it={t} new best {best_score:.6f}")

    return best_seq if best_seq is not None else a, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray | None,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Greedy pairwise mass‑transfer: move a small amount from i to j if it improves the objective."""
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        # pick a distinct destination
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # random transfer amount (up to 25 % of the source mass)
        delta = a[i] * np.random.rand() * 0.25
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq if best_seq is not None else a, best_score


# ----------------------------------------------------------------------
def _upsample_distribution(a_small: np.ndarray, n_target: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger length."""
    n_small = a_small.shape[0]
    xs_small = (np.arange(n_small, dtype=np.float64) + 0.5) / n_small
    xs_target = (np.arange(n_target, dtype=np.float64) + 0.5) / n_target
    a_target = np.interp(xs_target, xs_small, a_small, left=0.0, right=0.0)
    total = a_target.sum()
    if total == 0.0:
        a_target = np.full(n_target, 1.0 / n_target)
    else:
        a_target = a_target / total
    return a_target


# ----------------------------------------------------------------------
def _run_full_descent(a: np.ndarray,
                      deadline: float,
                      global_best_seq: np.ndarray | None,
                      global_best_score: float) -> Tuple[np.ndarray, float]:
    """
    Three‑phase local optimisation (Adam → multimax → greedy swaps).
    Returns the final vector and its objective value.
    """
    # Phase 1 – smooth‑max Adam
    seq1, sc1 = _smooth_max_adam_descent(
        a, deadline, global_best_seq, global_best_score,
        eta0=0.5, beta1=0.9, beta2=0.999,
        alpha0=0.2, alpha_mul=1.3, alpha_max=300.0,
        max_iter=80000, eval_interval=400)

    if time.time() >= deadline:
        final_score = _objective(seq1)
        return seq1, final_score

    # Phase 2 – exact multimax sub‑gradient
    seq2, sc2 = _subgradient_multimax_descent(
        seq1, deadline, global_best_seq, sc1,
        lr0=0.6, max_iter=50000, eval_interval=500)

    if time.time() >= deadline:
        final_score = _objective(seq2)
        return seq2, final_score

    # Phase 3 – greedy pairwise swaps
    seq3, sc3 = _pairwise_swap_local(
        seq2, deadline, global_best_seq, sc2)

    final_score = _objective(seq3)
    return seq3, final_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the ``budget_s`` seconds limit and returns the
    best sequence found (as a plain list of floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # 1. Short‑length exploration (cheap, many seeds)
    short_n = 2000
    print(f"[{time.time():.1f}] Generating seeds for length {short_n}")
    short_seeds = _initial_candidates_extended_fine(short_n, rng)

    # Evaluate all short seeds and keep the best few
    scored_short = []
    for cand in short_seeds:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand = cand / s
        sc = _objective(cand)
        scored_short.append((sc, cand))
    scored_short.sort(key=lambda x: x[0])
    top_short = [seq for _, seq in scored_short[:6]]

    # ------------------------------------------------------------------
    # 2. Run full three‑phase descent on the short top seeds
    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    for idx, seed_vec in enumerate(top_short):
        if time.time() >= deadline:
            break
        sub_deadline = time.time() + (deadline - time.time()) / max(1, len(top_short) - idx)
        seq_out, score_out = _run_full_descent(
            seed_vec.copy(),
            sub_deadline,
            overall_best_seq,
            overall_best_score,
        )
        if score_out < overall_best_score:
            overall_best_score = score_out
            overall_best_seq = seq_out.copy()
            print(f"[{time.time():.1f}] After short descent new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Upsample the current best to a longer length and refine there
    long_n = 20000
    if overall_best_seq is not None:
        upsampled = _upsample_distribution(overall_best_seq, long_n)
    else:
        upsampled = _distribution_uniform(long_n)

    # Collect a few additional long‑length seeds
    long_seeds: List[np.ndarray] = [upsampled]
    # Add a few random Dirichlet draws (different concentrations)
    for a in (0.1, 0.3, 0.7, 1.0):
        long_seeds.append(_distribution_dirichlet(long_n, rng, alpha=a))
    # Add a random sparse pattern
    long_seeds.append(_distribution_random_sparse(long_n, rng, density=0.01))
    # Add a couple of uniform/arcsine mixtures
    for w in (0.3, 0.6):
        mix = _distribution_mixture_uniform_arcsine(long_n, w, {})
        long_seeds.append(mix)

    # Evaluate long seeds quickly and retain the most promising few
    scored_long = []
    for cand in long_seeds:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand = cand / s
        sc = _objective(cand)
        scored_long.append((sc, cand))
    scored_long.sort(key=lambda x: x[0])
    top_long = [seq for _, seq in scored_long[:4]]

    # ------------------------------------------------------------------
    # 4. Run full descent on the long candidates
    for idx, seed_vec in enumerate(top_long):
        if time.time() >= deadline:
            break
        sub_deadline = time.time() + (deadline - time.time()) / max(1, len(top_long) - idx)
        seq_out, score_out = _run_full_descent(
            seed_vec.copy(),
            sub_deadline,
            overall_best_seq,
            overall_best_score,
        )
        if score_out < overall_best_score:
            overall_best_score = score_out
            overall_best_seq = seq_out.copy()
            print(f"[{time.time():.1f}] After long descent new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Perturbation / simulated‑annealing style restarts while time permits
    while time.time() < deadline - 5.0:
        if overall_best_seq is None:
            n = long_n
            a0 = _distribution_random_sparse(n, rng, density=0.02)
        else:
            n = overall_best_seq.shape[0]
            a0 = overall_best_seq.copy()
            # Gaussian perturbation proportional to current mass
            noise = rng.normal(scale=0.02, size=n)
            a0 = a0 + noise
            a0 = np.maximum(a0, 0.0)
        a0 = a0 / a0.sum()
        sub_deadline = time.time() + min(60.0, deadline - time.time())
        seq_out, score_out = _run_full_descent(
            a0,
            sub_deadline,
            overall_best_seq,
            overall_best_score,
        )
        if score_out < overall_best_score:
            overall_best_score = score_out
            overall_best_seq = seq_out.copy()
            print(f"[{time.time():.1f}] Perturbation improved best to {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # Final report
    if overall_best_seq is None:
        # fallback to a simple uniform vector
        overall_best_seq = _distribution_uniform(short_n)
        overall_best_score = _objective(overall_best_seq)

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # Convert to plain Python list as required by the external evaluator
    return overall_best_seq.tolist()
