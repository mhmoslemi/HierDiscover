# -*- coding: utf-8 -*-
"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.

Algorithm
---------

1. **Coarse analytic sweep** – generate a wide range of candidates from
   families with a single shape parameter (symmetric beta, edge‑weighted,
   sine‑weighted) for a relatively large length (≈20 k).  For each family
   we sweep the shape parameter densely (≤  ~ 100 points) and also
   combine the shape with the uniform distribution using a few mixing
   ratios.  The best few candidates are kept.

2. **Exponential‑gradient fine‑tuning** – run the soft‑max exponential‑
   descent from the top seeds (the same method that was used in the
   previous state‑of‑the‑art code, but with a larger active set and a
   slower temperature schedule).

3. **Exact sub‑gradient descent** – after the smooth phase we switch to
   a true sub‑gradient of the max‑convolution (the active constraints
   are the indices where the convolution attains its maximum).  The
   iterate is projected onto the probability simplex after each step
   and forced to stay symmetric (the optimal shape is conjectured to be
   symmetric).

4. **Local greedy refinement** – pairwise mass swaps and a cheap
   simulated‑annealing walk are applied as final polishing.

The routine respects the supplied time budget (default 1000 s) and
returns a plain Python list; ``evaluate_sequence`` will perform the final
checks and normalisation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1.
    The function is homogeneous of degree 0, therefore we keep the sum
    normalised to 1 during optimisation.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑density to avoid overflow
    log_w = (p - 1) * (np.log(x) + np.log(1.0 - x))
    w = np.exp(log_w)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight proportional to sin(π (i+½)/n)^p, vanishing at the ends."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smooth exponential descent -------------------------------------------
def _smooth_exponential_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    tau0: float = 3.0,
    tau_mult: float = 1.001,
    eval_interval: int = 100,
    active_k: int = 20,
    patience: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the true objective.
    Uses a soft‑max weighting of the convolution entries:
        w_i ∝ exp(τ·conv_i)   (τ grows during optimisation)
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    tau = tau0

    while time.time() < deadline and t < 500_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()

        # ---- soft‑max weighting of the top active_k entries ----
        if active_k >= len(conv):
            top_idx = np.arange(len(conv))
        else:
            # use argpartition for speed
            top_idx = np.argpartition(conv, -active_k)[-active_k:]

        # scale to avoid overflow
        w_raw = np.exp(tau * (conv[top_idx] - max_c))
        w = w_raw / w_raw.sum()

        # ---- compute weighted gradient ----
        grad = np.zeros_like(a)
        ind_grid = np.arange(n, dtype=np.int32)
        for weight, idx in zip(w, top_idx):
            ind = idx - ind_grid
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * weight * a[ind[mask]]

        # ---- exponentiated‑gradient update ----
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex
        a = _project_onto_simplex(a, s=1.0)

        # increase temperature slowly
        tau *= tau_mult

        # ---- periodic evaluation ----
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] smooth descent it={t} new best {best_score:.6f}")
            else:
                no_improve += 1

        # jitter on stagnation
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.004, size=n)
            a = a + jitter
            a = _project_onto_simplex(a, s=1.0)
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent -------------------------------------------
def _subgradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.25,
    max_iter: int = 200_000,
    eval_interval: int = 300,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()
        # tolerance – we treat any entry within 1e‑12 relative of the max as active
        tol = max(1e-12, max_c * 1e-12)
        active = np.where(conv >= max_c - tol)[0]

        # compute sub‑gradient
        grad = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=np.int32)
            grad[idx] += 2.0 * a[k - idx]

        eta = eta0 / math.sqrt(t)
        a = a - eta * grad

        # enforce symmetry (optimal shape is expected to be symmetric)
        a = (a + a[::-1]) * 0.5

        # project back onto the simplex (sum=1, non‑negative)
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise swap (deterministic local refinement) ------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small amount of mass; keep move only if objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style local search ---------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

# -*- coding: utf-8 -*-
"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.

Algorithm
---------

1. **Coarse analytic sweep** – generate a wide range of candidates from
   families with a single shape parameter (symmetric beta, edge‑weighted,
   sine‑weighted) for a relatively large length (≈20 k). For each family we
   sweep the shape parameter densely (≤ ~100 points) and also combine the
   shape with the uniform distribution using a few mixing ratios. The best
   few candidates are kept.

2. **Exponential‑gradient fine‑tuning** – run the soft‑max exponential‑
   descent from the top seeds (the same method that was used in the
   previous state‑of‑the‑art code, but with a larger active set and a slower
   temperature schedule).

3. **Exact sub‑gradient descent** – after the smooth phase we switch to a
   true sub‑gradient of the max‑convolution (the active constraints are the
   indices where the convolution attains its maximum).  The iterate is
   projected onto the probability simplex after each step and forced to stay
   symmetric (the optimal shape is conjectured to be symmetric).

4. **Local greedy refinement** – pairwise mass swaps and a cheap
   simulated‑annealing walk are applied as final polishing.

The routine respects the supplied time budget (default 1000 s) and returns
a plain Python list; ``evaluate_sequence`` will perform the final checks
and normalisation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1.
    The function is homogeneous of degree 0, therefore we keep the sum
    normalised to 1 during optimisation.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_w = (p - 1) * (np.log(x) + np.log(1.0 - x))
    w = np.exp(log_w)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight proportional to sin(π (i+½)/n)^p, vanishing at the ends."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smooth exponential descent -------------------------------------------
def _smooth_exponential_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    tau0: float = 3.0,
    tau_mult: float = 1.001,
    eval_interval: int = 100,
    active_k: int = 20,
    patience: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the true objective.
    Uses a soft‑max weighting of the convolution entries:
        w_i ∝ exp(τ·conv_i)   (τ grows during optimisation)
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    tau = tau0

    while time.time() < deadline and t < 500_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()

        # ---- soft‑max weighting of the top active_k entries ----
        if active_k >= len(conv):
            top_idx = np.arange(len(conv))
        else:
            top_idx = np.argpartition(conv, -active_k)[-active_k:]

        w_raw = np.exp(tau * (conv[top_idx] - max_c))
        w = w_raw / w_raw.sum()

        # ---- compute weighted gradient ----
        grad = np.zeros_like(a)
        ind_grid = np.arange(n, dtype=np.int32)
        for weight, idx in zip(w, top_idx):
            ind = idx - ind_grid
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * weight * a[ind[mask]]

        # ---- exponentiated‑gradient update ----
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex
        a = _project_onto_simplex(a, s=1.0)

        # increase temperature slowly
        tau *= tau_mult

        # ---- periodic evaluation ----
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] smooth descent it={t} new best {best_score:.6f}")
            else:
                no_improve += 1

        # jitter on stagnation
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.004, size=n)
            a = a + jitter
            a = _project_onto_simplex(a, s=1.0)
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent -------------------------------------------
def _subgradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.25,
    max_iter: int = 200_000,
    eval_interval: int = 300,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()
        tol = max(1e-12, max_c * 1e-12)
        active = np.where(conv >= max_c - tol)[0]

        grad = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=np.int32)
            grad[idx] += 2.0 * a[k - idx]

        eta = eta0 / math.sqrt(t)
        a = a - eta * grad

        # enforce symmetry (the optimal shape is expected to be symmetric)
        a = (a + a[::-1]) * 0.5
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise swap (deterministic local refinement) ------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small amount of mass; keep move only if objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style local search ---------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score
