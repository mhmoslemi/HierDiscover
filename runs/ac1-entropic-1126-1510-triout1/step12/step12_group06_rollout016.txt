def solve_lp_for_g(a, M):
    n = len(a)
    from scipy.optimize import linprog
    # Build A_ub as sparse matrix
    import scipy.sparse as sp
    m = 2*n - 1
    # create data for diagonals
    diags_data = []
    offsets = []
    for d in range(n):
        # length of diagonal = m - d
        diags_data.append(np.full(m - d, a[d], dtype=np.float64))
        offsets.append(d)
    A_ub = sp.diags(diags_data, offsets, shape=(m, n), format='csr')
    # Right-hand side: M * np.ones(m)
    b_ub = np.full(m, M, dtype=np.float64)
    # Objective
    c = -np.ones(n, dtype=np.float64)  # minimize -sum(g)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # fallback: return original a?
        return a.copy()
    return res.x

def lp_mixing_step(f, M, t):
    # solve LP.
    g = solve_lp_max_sum(f, M)  # returns raw g.
    # if sum(g) is negligible, return f unchanged.
    s_g = g.sum()
    if s_g == 0:
        return f
    g_norm = g / s_g
    # Mix
    new_f = (1 - t) * f + t * g_norm
    # renormalize sum=1
    new_f = new_f / new_f.sum()
    return new_f

"""
Hybrid optimiser for the max‑convolution step‑function problem.

The optimiser works in several stages while respecting a fixed time budget:

1. **Seed generation** – for a handful of lengths (4000, 8000, 12000) a
   rich collection of analytic families (uniform, arcsine, symmetric / asymmetric
   beta, Dirichlet, sparse, power‑law, exponential, two‑block spikes) and a few
   random piece‑wise linear constructions are created.  Each seed is normalised
   to sum = 1 and its exact objective `2·n·max(conv)/sum^2` is evaluated.

2. **Selection & mixing** – the best ~20 seeds are kept.  Additional
   candidates are generated by convexly mixing pairs of the best seeds,
   producing new shapes that were not explored by the analytic families.

3. **Smooth‑max Adam descent** – every selected candidate is refined with an
   exponentiated‑gradient (Adam) optimiser applied to the smooth‑max surrogate
   `f_α(a) = (1/α)·log Σ exp(α·conv_i(a))`.  The temperature `α` starts small
   (0.3) and is increased geometrically up to a very large value (≈2000) so
   that the surrogate converges to the true maximum.  All updates keep the
   vector on the probability simplex via Euclidean projection.

4. **Multimax sub‑gradient descent** – a classic sub‑gradient method (average
   of the active maximal indices) is run for a short period to further lower
   the peak.

5. **Random local restarts** – the current best solution is perturbed with
   Gaussian noise, renorm­alised and fed to a brief Adam run; any improvement
   is kept.  This explores basins that may have been missed before.

6. **Greedy pairwise mass‑transfer polishing** – finally a cheap greedy loop
   moves a small amount of mass from a source index to a destination index if
   the objective improves.  This often removes the last few hundredths.

The whole pipeline stops before the supplied `budget_s` (default 1000 s)
and returns the best sequence found as a plain list of floats.  The code
has no closures, uses only top‑level functions and relies on NumPy and
SciPy (for the tiny linear‑programming fallback it would use `scipy.optimize`,
but the core algorithm does not depend on it).  All vectors are kept
non‑negative and are projected onto the simplex after every update, so the
returned sequence is always valid for `evaluate_sequence`.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of Σ_i w_i·conv_i(a) w.r.t. a (2·a⋆w)."""
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (sum is 1 after normalisation)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    # sum is guaranteed non‑zero (checked by caller)
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discretised unit interval."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_power_decreasing(n: int, p: float) -> np.ndarray:
    i = np.arange(n, 0, -1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_random_piecewise(n: int, rng: np.random.Generator) -> np.ndarray:
    """A random piecewise‑linear density with a few knots."""
    knots = rng.integers(3, 9)                     # number of knots
    xs = np.linspace(0, n - 1, num=knots, dtype=int)
    heights = rng.random(knots) + 0.1               # avoid exactly zero
    full_x = np.arange(n)
    vals = np.interp(full_x, xs, heights)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_candidates_extended_fine(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    """Rich set of seed vectors for a given length."""
    cand: List[np.ndarray] = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # symmetric Beta – log‑spaced exponents
    for p in np.geomspace(0.01, 5.0, num=15):
        cand.append(_distribution_beta_sym(n, float(p)))

    # random asymmetric Beta draws
    for _ in range(10):
        alpha = 10 ** rng.uniform(-4, 1)   # 1e‑4 … 10
        beta = 10 ** rng.uniform(-4, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # power‑law families
    for p in (0.3, 0.5, 1.0, 2.0, 3.0):
        cand.append(_distribution_power_increasing(n, p))
        cand.append(_distribution_power_decreasing(n, p))

    # exponential families
    for lam in (0.0005, 0.001, 0.003, 0.01):
        cand.append(_distribution_exponential_increasing(n, lam))

    # two‑block spikes
    for bf in (0.02, 0.04, 0.08):
        cand.append(_distribution_two_blocks(n, block_frac=bf))

    # Dirichlet draws with varied concentration
    for a in (0.05, 0.2, 0.6, 1.5):
        cand.append(_distribution_dirichlet(n, rng, alpha=a))

    # Random sparse patterns
    cand.append(_distribution_random_sparse(n, rng, density=0.008))
    cand.append(_distribution_random_sparse(n, rng, density=0.02))

    # Random piecewise‑linear shapes
    for _ in range(4):
        cand.append(_distribution_random_piecewise(n, rng))

    # Purely random probability vectors
    for _ in range(6):
        v = rng.random(n)
        v = np.maximum(v, 0.0)
        v /= v.sum()
        cand.append(v)

    return cand

def _smooth_max_adam_descent(a: np.ndarray,
                             deadline: float,
                             best_seq: np.ndarray,
                             best_score: float,
                             eta0: float = 0.6,
                             beta1: float = 0.9,
                             beta2: float = 0.999,
                             eps: float = 1e-8,
                             alpha0: float = 0.3,
                             alpha_mul: float = 1.2,
                             alpha_max: float = 2000.0,
                             max_iter: int = 150000,
                             eval_interval: int = 400) -> Tuple[np.ndarray, float]:
    """
    Adam‑style exponentiated‑gradient descent on the smooth‑max surrogate
    f_α(a) = (1/α)·log Σ_i exp(α·conv_i(a)).  The temperature α is increased
    geometrically up to ``alpha_max``.  The update is multiplicative and
    projected to the probability simplex.
    """
    n = a.shape[0]
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)

    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)

        # smooth‑max weights (numerically stable)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad = _gradient_from_weights(a, w)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # multiplicative step
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(-step)

        # project onto the simplex (sum = 1)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # temperature schedule
        if t % 250 == 0 and alpha < alpha_max:
            alpha = min(alpha * alpha_mul, alpha_max)

        # periodic exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.6,
                                 max_iter: int = 50000,
                                 eval_interval: int = 400) -> Tuple[np.ndarray, float]:
    """
    Classical exponentiated‑gradient descent where the sub‑gradient is the
    uniform average over all active indices attaining the maximal convolution.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)
        eta = lr0 / math.sqrt(t)

        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Multimax it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Greedy pairwise mass‑transfer: move a small amount from i to j if it improves the objective."""
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        # pick distinct destination
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # random transfer up to 25 % of source mass
        delta = a[i] * np.random.rand() * 0.25
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects ``budget_s`` seconds and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a handful of lengths – longer vectors give more degrees of freedom.
    lengths = [4000, 8000, 12000]
    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: List[Tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 2. Scan analytic families (very cheap) and collect the best few.
    print(f"[{time.time():.1f}] Starting seed generation")
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Generating seeds for length {n}")

        seed_list = _initial_candidates_extended_fine(n, rng)

        for cand in seed_list:
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                continue
            cand /= s
            sc = _objective(cand)

            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] **** new global best len={n} score={sc:.6f}")

            top_candidates.append((sc, cand.copy()))

        # Keep only a few promising seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:20]

    # ------------------------------------------------------------------
    # 3. Add the previous solution if it exists.
    try:
        prev = globals().get("height_sequence_1")
        if isinstance(prev, (list, np.ndarray)):
            prev = np.asarray(prev, dtype=np.float64)
            if prev.ndim == 1 and prev.size > 0:
                prev = np.maximum(prev, 0.0)
                s = prev.sum()
                if s == 0.0:
                    prev = _distribution_uniform(prev.shape[0])
                    s = prev.sum()
                prev /= s
                sc = _objective(prev)
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = prev.copy()
                    print(f"[{time.time():.1f}] previous solution improved: {sc:.6f}")
                top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    if overall_best_seq is None:
        # Fallback – should never be needed.
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Enrich the pool by mixing the current best seeds.
    print(f"[{time.time():.1f}] Creating mixed seeds")
    best_arrays = [seq for _, seq in top_candidates[:8]]
    mixed_seeds: List[np.ndarray] = []
    for _ in range(30):
        i, j = rng.choice(len(best_arrays), size=2, replace=False)
        w = rng.random()
        mixed = w * best_arrays[i] + (1 - w) * best_arrays[j]
        mixed = np.maximum(mixed, 0.0)
        mixed /= mixed.sum()
        mixed_seeds.append(mixed)

    # Combine everything that we will feed to the optimiser.
    seed_pool = [overall_best_seq] + [seq for _, seq in top_candidates[:5]] + mixed_seeds

    # ------------------------------------------------------------------
    # 5. First refinement: Adam‑style smooth‑max descent on the seeds.
    print(f"[{time.time():.1f}] Starting Adam descent on {len(seed_pool)} seeds")
    remaining = deadline - time.time()
    if remaining > 30:
        # allocate an approximately equal share of time to each seed
        per_seed_deadline = time.time() + remaining / max(1, len(seed_pool))
        for seed_seq in seed_pool:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + (deadline - time.time()) / max(1, len(seed_pool))
            seq_out, score_out = _smooth_max_adam_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.6,
                beta1=0.9,
                beta2=0.999,
                eps=1e-8,
                alpha0=0.3,
                alpha_mul=1.2,
                alpha_max=2000.0,
                max_iter=150000,
                eval_interval=400,
            )
            if score_out < overall_best_score:
                overall_best_seq = seq_out.copy()
                overall_best_score = score_out
                print(f"[{time.time():.1f}] after Adam descent new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Second refinement: classic multimax sub‑gradient descent.
    remaining = deadline - time.time()
    if remaining > 20:
        seq_out, score_out = _subgradient_multimax_descent(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.6,
            max_iter=50000,
            eval_interval=500,
        )
        if score_out < overall_best_score:
            overall_best_seq = seq_out.copy()
            overall_best_score = score_out
            print(f"[{time.time():.1f}] after multimax new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Random local restarts with brief Adam runs.
    print(f"[{time.time():.1f}] Random local search")
    while time.time() < deadline - 5.0:
        # Gaussian perturbation of the current best
        noise = rng.normal(scale=0.03, size=overall_best_seq.shape)
        pert = overall_best_seq + noise
        pert = np.maximum(pert, 0.0)
        pert_sum = pert.sum()
        if pert_sum == 0.0:
            continue
        pert /= pert_sum

        sub_deadline = time.time() + 2.0  # short burst
        seq_out, score_out = _smooth_max_adam_descent(
            pert,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.5,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            alpha0=0.5,
            alpha_mul=1.2,
            alpha_max=800.0,
            max_iter=30000,
            eval_interval=200,
        )
        if score_out < overall_best_score:
            overall_best_seq = seq_out.copy()
            overall_best_score = score_out
            print(f"[{time.time():.1f}] perturbation improved to {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final cheap polishing with greedy pairwise swaps.
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
