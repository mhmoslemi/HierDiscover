def _softmax_gradient_descent(a_initial, deadline, best_seq, best_score, start_alpha=0.5, max_alpha=200.0, alpha_multiplier=2.0, eta0=0.5, eval_every=200):
    a = a_initial.copy()
    n = len(a)
    alpha = start_alpha
    t = 0
    while time.time() < deadline:
        t += 1
        # compute convolution
        conv = _fft_self_convolve(a)
        # compute softmax weights
        maxc = conv.max()
        # compute stable softmax: exp(alpha*(conv - maxc))
        exps = np.exp(alpha * (conv - maxc))
        w = exps / exps.sum()
        # compute gradient via convolution
        # convolve a with reversed w
        conv_rev = _fft_convolve(a, w[::-1])
        # extract gradient slice
        idx_start = 2 * n - 2
        grad = 4.0 * n * conv_rev[idx_start - np.arange(n)]
        # step size
        eta = eta0 / math.sqrt(t)
        # update
        a = a * np.exp(-eta * grad)
        # renorm
        s = a.sum()
        if s == 0 or np.isnan(s) or np.isinf(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s
        if t % eval_every == 0:
            # compute objective exactly
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smoothed GD iter={t} alpha={alpha:.2f} best {best_score:.6f}")
        # increase alpha occasionally
        if t % (eval_every * 5) == 0:
            alpha = min(max_alpha, alpha * alpha_multiplier)
    return best_seq, best_score

This program searches for a non‑negative step‑function that minimises  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2},
\]

where `a*a` is the linear self‑convolution of the coefficient vector `a`.  
The objective is invariant under scaling, therefore the optimizer works on
probability vectors (`∑a = 1`).  

The search consists of three phases:

1. **Seed generation** – a compact but diverse pool of analytic families
   (uniform, Hann, arcsine, symmetric‑Beta, edge‑weighted, generalized‑normal,
   random Dirichlet, sparse) for several lengths.  
   The current best construction (`height_sequence_1`, if present) is also added.

2. **Smoothed gradient descent** – instead of using the gradient of the *active*
   convolution entry only, the algorithm minimises a *smooth* surrogate

   \[
   \tilde F_\alpha(a)=\frac{2n}{\alpha}\log\!\Bigl(\sum_k e^{\alpha\,(a*a)_k}\Bigr) .
   \]

   The soft‑max weighting makes the gradient depend on *all* convolution
   entries, which encourages a flatter autocorrelation.  
   The gradient can be written in closed form and is evaluated with two FFT
   convolutions per iteration.  An exponentiated‑gradient update
   `a ← a·exp(-η·∇)` keeps the vector non‑negative; after the update the vector
   is re‑normalised to the simplex.  The smoothing parameter `α` is increased
   geometrically during the run, ending with a value large enough to approximate
   the true max‑operator.

3. **Polishing** – after the descent the best vector is optionally up‑sampled
   (linear interpolation while preserving the simplex) and refined again.
   Finally a long‑run pairwise‑mass‑transfer simulated‑annealing step
   (identical to the one used in the previous state‑of‑the‑art code) removes
   any remaining local irregularities.

The whole procedure respects the 1000 s budget, prints progress information
and returns a plain `list[float]` ready for `evaluate_sequence`.  

```python
"""
Smoothed‑gradient search for a step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a))²

The algorithm builds a small pool of analytic seeds, runs a
soft‑max‑smoothed gradient descent (exponentiated‑gradient on the simplex)
with a geometric increase of the smoothing parameter, optionally upsamples
the current best candidate, and finishes with a long‑run pairwise‑mass‑transfer
annealing.  All operations are O(n log n) via FFTs and the routine finishes
well inside the 1000 s budget, returning a list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# 1. Linear‑algebra utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 2. Convolution helpers (FFT based)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using a real FFT."""
    n = a.shape[0]
    m = b.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: n + m - 1]


# ----------------------------------------------------------------------
# 3. Exact objective used by ``evaluate_sequence``.
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective (scale‑invariant).  Input must be non‑negative and
    summed to 1 (the code guarantees this).  Returns ``2·n·max(conv)``.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# 4. Analytic families (all return a probability vector, sum=1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) density on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{-p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(-|x-0.5|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation while preserving the simplex."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# 5. Smoothed (soft‑max) gradient descent on the simplex
# ----------------------------------------------------------------------
def _softmax_gradient_descent(
    a_init: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    start_alpha: float = 0.5,
    max_alpha: float = 200.0,
    alpha_mult: float = 2.0,
    eta0: float = 0.8,
    eval_every: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the soft‑max surrogate of the objective.
    *a_init* is assumed to be a probability vector (sum = 1).
    The method updates a ← a·exp(-η·∇) and re‑normalises after each step.
    The smoothing parameter α is increased geometrically every ``alpha_step``
    iterations.
    """
    a = a_init.copy()
    n = a.shape[0]
    t = 0
    alpha = start_alpha
    alpha_step = 2000  # increase α after this many iterations
    while time.time() < deadline:
        t += 1
        # 1) self‑convolution
        conv = _fft_self_convolve(a)                # length 2n‑1

        # 2) soft‑max weights w_i = exp(α·conv_i) / Σ exp(α·conv)
        maxc = conv.max()
        exps = np.exp(alpha * (conv - maxc))   # stable
        w = exps / exps.sum()

        # 3) gradient ∇_j = 4·n·Σ_i w_i·a_{i‑j}
        #    = 4·n·cross‑correlation(a, w)
        conv_rev = _fft_convolve(a, w[::-1])      # length 3n‑2
        idx_start = 2 * n - 2
        grad = 4.0 * n * conv_rev[idx_start - np.arange(n)]

        # 4) exponentiated‑gradient step
        eta = eta0 / math.sqrt(t)
        # avoid overflow/underflow in the exponential factor
        factor = np.exp(-eta * grad)
        a = a * factor
        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # 5) occasional evaluation
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] smoothed GD it={t} α={alpha:.2f} "
                    f"new best {best_score:.6f}"
                )

        # 6) increase smoothing parameter
        if t % alpha_step == 0:
            alpha = min(max_alpha, alpha * alpha_mult)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 6. Pairwise mass‑transfer simulated annealing (as in the reference code)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins; accept improvements
    or accept worse moves with a linear cooling schedule."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)

        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
                )
    return best_seq, best_score
```
