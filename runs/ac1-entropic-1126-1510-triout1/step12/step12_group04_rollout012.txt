                             eta0=0.5, alpha0=0.5, alpha_factor=1.01, alpha_max=10.0):

        conv = _fft_self_convolve(a)
        cur_score = 2.0*n*conv.max()
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()

        # compute softmax probs
        max_conv = conv.max()
        # compute shifted conv for stability: conv - max_conv
        # decide smoothing factor
        alpha = min(alpha_max, alpha0 * (alpha_factor ** (t-1)))
        # exponentiate
        w = np.exp(alpha * (conv - max_conv))
        probs = w / w.sum()
        # gradient via convolution of probs with reversed a
        rev_a = a[::-1]
        grad_full = _fft_convolve(probs, rev_a)

        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0/n)
        else:
            a = a / s

"""
Hybrid optimizer for the low‑autocorrelation step‑function problem.

The search proceeds in three stages:

1. Analytic seeding – a diverse pool of candidates is generated from
   simple analytic families (uniform, arcsine, powered inverse‑sqrt,
   two‑block, sparse Dirichlet, and a symmetric beta family).  All
   candidates are normalised to a probability simplex and evaluated
   with the exact objective.

2. Soft‑max sub‑gradient descent – the non‑smooth “max” in the
   objective is replaced by a smooth log‑sum‑exp with a temperature
   parameter α that is slowly increased.  The gradient of the smoothed
   objective can be written as a weighted convolution which is
   evaluated efficiently with FFTs.  An exponentiated‑gradient (mirror
   descent) update with a 1/√t step‑size keeps the iterates on the
   simplex.  The best vector seen during the run is retained.

3. Greedy polishing – cheap random mass‑transfer moves are accepted
   only if they improve the true (unsmoothed) objective.  This removes
   any remaining tiny spikes.

The function `propose_candidate` respects the global time budget,
returns the best sequence found (as a plain list of floats) and prints
progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via real FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n) - 1).bit_length()
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)² (sum is assumed to be 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    return 2.0 * n * max_conv

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine_offset(n: int, c: float) -> np.ndarray:
    """Arcsine‑type pdf with offset c>0."""
    i = np.arange(n, dtype=np.float64)
    left = i + c
    right = (n - 1 - i) + c
    vals = 1.0 / np.sqrt(left * right)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _distribution_power_inv_sqrt(n: int, p: float) -> np.ndarray:
    """Family w_i ∝ ((i+0.5)*(n-i-0.5))^{-p/2}."""
    i = np.arange(n, dtype=np.float64)
    left = i + 0.5
    right = (n - 1 - i) + 0.5
    vals = (left * right) ** (-p / 2.0)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric beta distribution on the discrete grid."""
    # Mid‑point of each bin
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # Beta pdf (ignoring normalising constant)
    vals = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Produce a modest pool of analytic seeds for a given length."""
    cand = []

    # 1. simple families
    cand.append(_distribution_uniform(n))
    for c in (0.2, 0.5, 1.0, 2.0):
        cand.append(_distribution_arcsine_offset(n, c))
    for p in (0.8, 1.0, 1.2):
        cand.append(_distribution_power_inv_sqrt(n, p))
    cand.append(_distribution_two_blocks(n, block_frac=0.05))
    cand.append(_distribution_two_blocks(n, block_frac=0.10))
    cand.append(_distribution_random_sparse(n, rng, density=0.01))
    for _ in range(2):
        cand.append(_distribution_dirichlet(n, rng,
                                            alpha=0.3 + 0.7 * rng.random()))

    # 2. beta shapes (symmetric)
    for a in (0.3, 0.5, 0.8, 1.0, 1.2, 1.5, 2.0):
        cand.append(_distribution_beta(n, a, a))

    return cand


# ----------------------------------------------------------------------
def _softmax_subgrad_descent(a: np.ndarray,
                             deadline: float,
                             best_seq: np.ndarray,
                             best_score: float,
                             eta0: float = 0.5,
                             alpha0: float = 0.5,
                             alpha_factor: float = 1.02,
                             alpha_max: float = 10.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on a smoothed version of the objective.
    The soft‑max temperature α is increased geometrically.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        # exact autocorrelation (needed for score & for the soft‑max)
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] iter={t} new best {best_score:.6f}")

        # ---------------------------------------------------------------
        # smoothed gradient
        max_conv = conv.max()
        alpha = min(alpha_max, alpha0 * (alpha_factor ** (t - 1)))
        # stable exponentiation
        w = np.exp(alpha * (conv - max_conv))
        probs = w / w.sum()               # p_k ≥0, Σ p_k = 1

        # gradient = 2 * (probs * a) correlation
        rev_a = a[::-1]
        grad_full = _fft_convolve(probs, rev_a)   # length = (2n-1)+n-1
        start = len(probs) - 1                     # offset = 2n-2
        grad = 2.0 * grad_full[start:start + n]

        # ---------------------------------------------------------------
        # exponentiated‑gradient update + simplex projection
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)       # mirror direction
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Simple random mass‑transfer walk that keeps only improving moves."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue
        # move a random fraction (up to 30 %) of the source mass
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur_score = _objective(new_a)
        if cur_score < best_score - 1e-13:
            a = new_a
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}→{j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score

def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Main entry point – returns the best non‑negative step‑function
    (as a list of floats) discovered within the allotted time.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # 1 – generate a pool of analytic seeds of various lengths
    lengths = [1500, 2500, 3500, 4500]          # a spread of sizes
    elite: list[tuple[float, np.ndarray]] = []  # (score, vector)

    overall_best_seq = None
    overall_best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] --- scanning length {n} ---")
        for cand in _initial_candidates(n, rng):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] !!! global best len={n} score={sc:.6f}")
            elite.append((sc, cand.copy()))

        # keep only the top few across all lengths explored so far
        elite.sort(key=lambda x: x[0])
        elite = elite[:12]

    # ------------------------------------------------------------------
    # 2 – optionally incorporate the previously submitted solution
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best improved: {sc:.6f}")
            elite.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3 – soft‑max sub‑gradient descent on the elite seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and elite:
        elite.sort(key=lambda x: x[0])
        seeds_to_use = [seq for _, seq in elite[:5]]
        per_seed = max(1.0, remaining / len(seeds_to_use))
        for seed_seq in seeds_to_use:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            seq_out, score_out = _softmax_subgrad_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.6,
                alpha0=0.4,
                alpha_factor=1.015,
                alpha_max=15.0,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after descent new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4 – greedy polishing
    if overall_best_seq is not None and time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
