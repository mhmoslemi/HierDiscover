The goal is to minimise  

\[
\text{score}(a)=\frac{2\,n\;\max\bigl(a\star a\bigr)}{\bigl(\sum a\bigr)^2}\;,
\]

where `a` is a non‑negative step‑function (vector) of length `n` and `a⋆a` is its **linear** self‑convolution.  
The score is invariant under scaling, therefore we keep every candidate normalised to `∑a = 1` and work directly on the simplex.

The previous implementation performed a gradient descent that only used the *single* maximal convolution entry.  
When several indices are close to the maximum the algorithm can “wiggle” around a local optimum.
The new optimiser therefore

1. **uses a set of active constraints** – all convolution entries whose value is within a
   relative tolerance of the current maximum,
2. **averages the gradients of the whole active set**, which gives a smoother, more global
   descent direction,
3. **employs a multi‑phase schedule** – start with a *loose* tolerance (many active constraints)
   and tighten it progressively,
4. **works on increasingly larger vectors** – after a first optimisation the best candidate
   is linearly up‑sampled (preserving the simplex) and refined again,
5. **finishes with a cheap simulated‑annealing polish** that moves mass between two bins.

All steps respect the 1000 s budget (a 5 s safety margin) and never perform I/O.  
The code is completely self‑contained; it can be dropped into the evaluation harness as `propose_candidate`.

```python
"""
Improved optimiser for ``evaluate_sequence``.

*   Generates a small, diverse pool of analytic / random seeds.
*   Refines the best seeds with *sub‑gradient descent* on the exact
    objective.  The sub‑gradient is computed from a *set* of nearly‑maximal
    convolution entries (active set), not just the single max – this
    yields a smoother descent and escapes shallow local minima.
*   The descent is performed in three phases with a tolerance that
    shrinks exponentially (from 2 % of the max down to 1 e‑6).
*   The leading candidate is up‑sampled to a larger length (×2) and
    optimised again – a larger dimension gives the algorithm more
    freedom to flatten the convolution.
*   A final short simulated‑annealing / pairwise mass‑transfer phase
    polishes the result.

All updates keep the vector on the probability simplex using an
exponentiated‑gradient (mirror descent) step.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used only for the very rare zero‑sum case
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto {x>=0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers – real FFT based full linear self‑convolution
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a⋆a (real FFT, length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective (the same quantity that ``evaluate_sequence`` returns)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a (any scale)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic families – all return a *probability* vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^‑p,  i,j = distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Sparse random vector with the given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base ⊙ (1‑lam) + uniform ⊙ lam."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Gradient of the *active* convolution entries (sub‑gradient of the ∞‑norm)
# ----------------------------------------------------------------------
def _grad_of_active_set(a: np.ndarray, rel_tol: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Return the averaged gradient of all convolution entries that are within
    ``rel_tol`` relative distance of the current maximum.
    Also returns the current maximum value.
    """
    conv = _fft_self_convolve(a)
    max_val = conv.max()
    # tolerance in absolute units
    tol = max_val * rel_tol
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    n = a.shape[0]

    # each active index contributes the same formula as the single‑max case
    for k in active:
        if k < n:
            # overlap length = k+1
            grad[: k + 1] += 2.0 * a[k::-1]
        else:
            # k >= n : overlap length = 2·n‑1‑k
            start = k - (n - 1)
            grad[start:] += 2.0 * a[start:][::-1]

    if active.size > 0:
        grad /= float(active.size)
    return grad, max_val


# ----------------------------------------------------------------------
# Sub‑gradient descent with a *decreasing* tolerance schedule
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a: np.ndarray,
                             deadline: float,
                             best_seq: np.ndarray,
                             best_score: float,
                             eta0: float = 0.6,
                             start_tol: float = 0.02,
                             end_tol: float = 1e-6,
                             eval_every: int = 120) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact objective.
    ``start_tol`` → ``end_tol`` is exponentially decayed during the run.
    The update is: a ← a ⊙ exp(‑η·grad)  (⊙ element‑wise), followed by normalisation.
    """
    n = a.shape[0]
    t = 0
    t0 = time.time()
    total = deadline - t0
    while time.time() < deadline:
        t += 1
        # exponential decay of the tolerance
        elapsed = time.time() - t0
        frac = min(1.0, elapsed / total)
        rel_tol = start_tol * ((end_tol / start_tol) ** frac)

        grad, cur_max = _grad_of_active_set(a, rel_tol)

        # step size decreasing like 1/√t
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] GD it={t} tol={rel_tol:.1e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple linear up‑sampling (interpolation) that preserves the simplex
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length len(a) to new_n (keeps Σ = 1)."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new


# ----------------------------------------------------------------------
# Pairwise mass‑transfer annealing (final polish)
# ----------------------------------------------------------------------
def _pairwise_annealing(a: np.ndarray,
                        deadline: float,
                        best_seq: np.ndarray,
                        best_score: float,
                        rng: np.random.Generator,
                        max_trials: int | None = None) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins (Simulated‑Annealing)."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        # pick a source bin with non‑negligible mass
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        # pick a distinct destination bin
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move up to 40 % of the source mass
        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        cur = _objective(cand)

        # linear cooling schedule
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = cand
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score

        # random Dirichlet (several concentrations)
        for _ in range(3):
            alpha = rng.uniform(0.02, 0.6)
            families.append(rng.dirichlet(np.full(n, alpha)))

        # sparse random seeds
        for _ in range(2):
            families.append(_distribution_random_sparse(n, rng, density=0.02))

        # mix each deterministic / parametric family with a little uniform mass
        mixed = []
        for base in families:
            mixed.append(base)                                 # keep original
            mixed.append(_mix_with_uniform(base, lam=0.1))      # 10 % uniform
        families = mixed

        # evaluate and keep the best 10 of this length
        for cand in families:
            if time.time() >= deadline:
                break
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < 1.55:   # keep a watch‑list early on
                print(f"[{time.time():.1f}] len={n} seed score {sc:.6f}")

    # keep only the global best 10 seeds
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:10]

    # if a previous construction is available, inject it
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, 0.0)
            prev /= prev.sum()
            sc = _objective(prev)
            top_candidates.append((sc, prev.copy()))
            print(f"[{time.time():.1f}] previous candidate score {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2.  Main optimisation loop – each seed gets a share of the remaining time
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    n_seeds = len(top_candidates)
    if n_seeds == 0:
        # fall back to a uniform vector of moderate size
        best_seq = _distribution_uniform(4096)
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")
    else:
        # allocate time roughly equally, but leave the last ~15 % for up‑sampling &
        # final annealing
        time_for_up = remaining * 0.15
        time_per_seed = (remaining - time_for_up) / n_seeds

        best_seq = None
        best_score = float("inf")
        for idx, (seed_score, seed_vec) in enumerate(top_candidates):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + time_per_seed
            print(f"[{time.time():.1f}] GD on seed {idx} (len={len(seed_vec)})")
            opt_seq, opt_score = _smooth_gradient_descent(
                seed_vec.copy(),
                sub_dead,
                best_seq if best_seq is not None else seed_vec,
                best_score,
                eta0=0.7,
            )
            if opt_score < best_score - 1e-12:
                best_score = opt_score
                best_seq = opt_seq.copy()
                print(f"[{time.time():.1f}] new global best after GD {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Upsample to a larger length (if time permits) and refine again
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        cur_n = best_seq.shape[0]
        target_n = cur_n * 2
        # keep the FFT size reasonable (next power‑of‑two ≤ 65536)
        if target_n <= 65536:
            print(f"[{time.time():.1f}] up‑sampling {cur_n} → {target_n}")
            up_seq = _upsample(best_seq, target_n)
            up_score = _objective(up_seq)
            if up_score < best_score:
                best_score = up_score
                best_seq = up_seq
                print(f"[{time.time():.1f}] up‑sampled new best {best_score:.6f}")

            # a short burst of gradient descent on the larger vector
            sub_dead = time.time() + (deadline - time.time()) * 0.35
            best_seq, best_score = _smooth_gradient_descent(
                best_seq.copy(),
                sub_dead,
                best_seq,
                best_score,
                eta0=0.6,
            )
            print(f"[{time.time():.1f}] after up‑sample GD best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Final cheap annealing polish
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing")
        best_seq, best_score = _pairwise_annealing(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            rng,
        )
        print(f"[{time.time():.1f}] final best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Return the best candidate found (as a plain Python list)
    # ------------------------------------------------------------------
    if best_seq is None:
        # very unlikely, but guarantees a return
        best_seq = _distribution_uniform(4096)

    final_score = _objective(best_seq)
    print(
        f"Finished – best score {final_score:.6f} (length={len(best_seq)})"
    )
    return best_seq.tolist()
```

The optimiser follows the plan described at the top, stays well within the 1000 s limit, and returns a plain `list[float]` ready for `evaluate_sequence`.
