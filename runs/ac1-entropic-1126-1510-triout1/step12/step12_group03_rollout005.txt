"""
Propose a non‑negative step‑function (list of heights) that minimizes
evaluate_sequence().  The search consists of three phases:

1.  A fast scan of several analytic families (uniform, arcsine,
    beta, edge‑weighted, sine, cosine, quadratic, exponential decay) on

    projected onto the probability simplex.  This improves over the
    earlier “top‑k sub‑gradient” method.

3.  A final greedy pairwise mass‑transfer polishing (hill‑climbing).

The routine respects a global time budget (default 1000 s) and prints
progress.  It returns a plain Python list; evaluate_sequence will apply
its own clipping / normalisation.

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact (unnormalised) objective 2·n·max(conv(a,a)) for a vector with Σa = 1.
    a must be non‑negative.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Edge‑weighted family: w_i ∝ ((i+½)*(n‑i‑½))^{‑p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_exponential_decay(n: int, p: float) -> np.ndarray:
    """Symmetric exponential decay from the centre (p>0)."""
    c = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-p * np.abs(i - c))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse Dirichlet‑like seed (prescribed density of non‑zeros)."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Full‑soft‑max (temperature‑annealed) gradient descent -----------------
def _softmax_weighted_gradient(a: np.ndarray, tau: float):
    """
    Returns (grad, max_conv) where ``grad`` is the gradient of the
    smooth objective 2·n·τ·log∑exp(conv/τ) w.r.t. ``a`` (up to the factor 1).
    The gradient is computed as
        grad_k = 2·∑_i w_i·a_{i‑k}
    where w_i are the soft‑max weights of the convolution entries.
    """
    n = a.shape[0]

    # 1) convolution a*a
    conv = _fft_self_convolve(a)                # length 2n‑1

    # 2) soft‑max weights (stable)
    maxc = conv.max()
    # protect against tau becoming too tiny (which would cause overflow)
    exp_arg = (conv - maxc) / max(tau, 1e-12)
    np.clip(exp_arg, -700, 700, out=exp_arg)   # avoid inf
    exp_vals = np.exp(exp_arg)
    w = exp_vals / exp_vals.sum()

    # 3) weighted gradient via a single FFT
    # correlation of w with a (no reversal) = 2·∑ w_i·a_{i‑k}
    size = 1 << ((3 * n - 2 - 1).bit_length())   # next power of 2 ≥ 3n‑2
    A_rev_fft = np.fft.rfft(a[::-1], n=size)
    W_fft = np.fft.rfft(w, n=size)
    corr = np.fft.irfft(W_fft * A_rev_fft, n=size)
    # extract the n relevant entries (indices n‑1 … n‑1+n‑1)
    grad = 2.0 * corr[n - 1:n - 1 + n]          # factor 2 from the derivative
    return grad, conv.max()


def _gradient_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.5,
    tau_factor_start: float = 0.5,
    tau_decay: float = 0.9,
    decay_every: int = 500,
) -> tuple[np.ndarray, float]:
    """
    Soft‑max gradient descent with simplex projection.
    ``lr0`` – initial learning rate; it decays as 1/√t.
    ``tau_factor_start`` – initial temperature as a fraction of the
    current max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    tau_factor = tau_factor_start

    while time.time() < time_limit:
        t += 1
        # temperature based on current max conv
        cur_max = _fft_self_convolve(a).max()
        tau = cur_max * tau_factor

        grad, cur_max = _softmax_weighted_gradient(a, tau)

        # exact objective (for monitoring & best‑track)
        cur_obj = 2.0 * n * cur_max
        if cur_obj < best_score - 1e-14:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[{time.time():.1f}] GD it={t} new best {best_score:.6f}")

        # learning‑rate schedule (simple 1/√t scaling)
        eta = lr0 / math.sqrt(t)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)

        # temperature decay schedule
        if decay_every and t % decay_every == 0:
            tau_factor = max(tau_factor * tau_decay, 1e-8)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise polishing -------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Randomly transfers a fraction of mass from one index to another,
    accepting only strictly improving moves.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur_obj = _objective(new_a)
        if cur_obj < best_score - 1e-14:
            a = new_a
            best_score = cur_obj
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Main entry point used by the evaluator.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0         # a small safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, seq)

    # -------------------------------------------------------------
    # 1.  Scan a variety of analytic families on several lengths.
    # -------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 12000, 20000, 30000]

    # Parameter grids
    p_grid = np.concatenate(
        [
            np.logspace(-2, -0.3, 12),      # 0.01 … 0.5 (edge‑heavy)
            np.linspace(0.5, 2.0, 10),      # moderate
            np.linspace(2.0, 5.0, 6),       # centre‑heavy
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 6)        # mixing with uniform

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # ----- deterministic families -----
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
            _distribution_exponential_decay,
        ):
            cand = base_fn(n)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # ----- parametric families (beta / edge weighted) -----
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta_seq = _distribution_beta_sym(n, p)
            edge_seq = _distribution_edge_weighted(n, p)

            for base in (beta_seq, edge_seq):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # ----- random Dirichlet seeds -----
        for _ in range(4):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} score={sc:.6f}")

        # ----- sparse random seeds -----
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

        # keep only the most promising ~30 seeds
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2.  If a previous construction is present, inject it.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback – should never happen.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 5000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(
            f"[{time.time():.1f}] fallback uniform n={n} score={overall_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 4.  Soft‑max gradient descent on the best seeds.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] start gradient descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            gd_deadline = time.time() + per_seed
            a_opt, sc_opt = _gradient_descent(
                seed_seq,
                gd_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.7,
                tau_factor_start=0.5,
                tau_decay=0.85,
                decay_every=400,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] after GD seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Optional up‑sampling + extra refinement (if time permits).
    # ------------------------------------------------------------------
    while time.time() < deadline - 15.0:
        n_cur = overall_best_seq.shape[0]
        # we try to double the resolution but stop before 80 k elements
        n_new = min(80000, n_cur * 2)
        if n_new == n_cur:
            break
        # linear interpolation onto a finer grid
        old_pos = (np.arange(n_cur, dtype=np.float64) + 0.5) / n_cur
        new_pos = (np.arange(n_new, dtype=np.float64) + 0.5) / n_new
        up_seq = np.interp(new_pos, old_pos, overall_best_seq)
        up_seq /= up_seq.sum()  # renormalise

        # brief refinement on the up‑sampled vector
        up_deadline = time.time() + min(30.0, deadline - time.time() - 5.0)
        a_opt, sc_opt = _gradient_descent(
            up_seq,
            up_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.5,
            tau_factor_start=0.5,
            tau_decay=0.9,
            decay_every=300,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(
                f"[{time.time():.1f}] up‑sampled to {n_new} improved best to {overall_best_score:.6f}"
            )
        else:
            # no improvement – stop the up‑sampling loop
            break

    # ------------------------------------------------------------------
    # 6.  Final greedy pairwise polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy swap polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (2·n·max(conv)): {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
