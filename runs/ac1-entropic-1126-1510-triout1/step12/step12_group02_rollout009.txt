```
def propose_candidate(seed=42, budget_s=1000):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    # Stage 0: baseline best sequence (maybe from previous variable)
    best_seq = None
    best_score = np.inf

    # Stage 1: generate seeds
    top = []  # list of (score, seq)
    lengths = [2000, 4000, 8000]   # may include 12000 if time.
    for n in lengths:
        if time.time() >= deadline: break
        # families
        families = []
        families.append(_distribution_uniform(n))
        families.append(_distribution_arcsine(n))
        families.append(_distribution_hann(n))
        families.append(_distribution_generalized_normal(n, q=1.5))
        # Beta and edge weighted for a set of p values
        for p in [0.1,0.3,0.5,0.8,1,2]:
            if time.time()>=deadline: break
            beta = _distribution_beta_sym(n,p)
            families.append(beta)
            edge = _distribution_edge_weighted(n,p)
            families.append(edge)
        # random Dirichlet
        for _ in range(3):
            if time.time()>=deadline: break
            alpha = rng.uniform(0.01,2.0)
            families.append(rng.dirichlet(np.full(n, alpha)))
        # sparse random
        for _ in range(2):
            if time.time()>=deadline: break
            families.append(_distribution_random_sparse(n,rng,density=0.02))
        # eval
        for a in families:
            a = np.maximum(a,0.0)
            a = a / a.sum()
            score = _objective(a)
            top.append((score,a.copy()))
            if score < best_score:
                best_score = score
                best_seq = a.copy()
        # keep best 20
        top.sort(key=lambda x: x[0])
        top = top[:20]

    # Stage 2: Gradient descent on top seeds
    # allocate remaining time to these seeds, each gets about remaining / len(top)
    remaining = deadline - time.time()
    if remaining>10 and top:
        per_seed = remaining / len(top)
        for idx, (init_score, a0) in enumerate(top):
            sub_end = time.time() + per_seed
            a_opt, sc_opt = _exp_grad_descent_momentum(a0.copy(), sub_end, best_seq, best_score)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] GD seed {idx} improved to {best_score:.6f}")

    # Stage 3: optional upsample and fine gradient passes
    if time.time() < deadline:
        cur_n = best_seq.shape[0]
        if cur_n*2 <= 64000:
            up = _upsample(best_seq, cur_n*2)
            sc_up = _objective(up)
            if sc_up < best_score:
                best_seq = up
                best_score = sc_up
                print(f"[{time.time():.1f}] Upsample improved to {best_score:.6f}")

            # second GD
            sub_end = time.time() + (deadline - time.time())*0.3
            a_opt, sc_opt = _exp_grad_descent_momentum(best_seq.copy(), sub_end, best_seq, best_score)
            if sc_opt < best_score:
                best_seq = a_opt
                best_score = sc_opt
                print(f"[{time.time():.1f}] GD after upsample improved to {best_score:.6f}")

    # Stage 4: final annealing
    if time.time() < deadline:
        a_opt, sc_opt = _pairwise_annealing(best_seq.copy(), deadline, best_seq, best_score, rng, max_trials=None)
        if sc_opt < best_score:
            best_seq = a_opt
            best_score = sc_opt
            print(f"[{time.time():.1f}] Final annealing improved to {best_score:.6f}")

    print(f"Final best score {best_score:.6f} length {len(best_seq)}")
    return best_seq.tolist()
```

```
def _random_move(a, rng):
    n = len(a)
    # Choose two indices i!=j with a_i > epsilon
    i = rng.choice(np.where(a>1e-8)[0])
    j = rng.integers(n)
    while j == i:
        j = rng.integers(n)
    # Transfer fraction of a_i to a_j
    delta = a[i] * rng.random() * 0.3
    new_a = a.copy()
    new_a[i] -= delta
    new_a[j] += delta
```

assistantfinal**Solution Overview**

We search for a non‑negative step function that minimises  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2},
\]

where `a*a` is the full linear self‑convolution.  
The objective is **scale invariant**, therefore we keep every candidate on the
probability simplex (`∑a = 1`).  
The algorithm has three phases

1. **Fast seed generation** – a small pool of analytically‑defined probability
   vectors (uniform, arcsine, β‑distributions, edge‑weighted, Hann,
   Kaiser, Chebyshev, sparse Dirichlet, …) for a few moderate lengths.
2. **Intensive exponentiated‑gradient descent** – we run a *momentum* version of
   the sub‑gradient method on the exact objective for each of the best seeds.
   The update rule is  

   \[
   a\leftarrow a\;\exp(-\eta_t\,v_t),\qquad
   v_{t}=0.9\,v_{t-1}+0.1\,\nabla F_{\max}(a_t),
   \]

   where `∇Fmax` is the gradient of the currently maximal convolution entry.
   The step size is `η_t = η₀ / √t` (default `η₀=2.0`).  After each update we
   renormalise to the simplex and (optionally) enforce symmetry.
3. **Polishing** – a long‑run simulated‑annealing / pairwise‑mass‑transfer
   phase that accepts improving moves always and occasional worsening moves
   according to a linear cooling schedule.

All phases respect a 1000 s wall‑clock budget (2 s safety margin).  
The routine returns a plain `list[float]` ready for `evaluate_sequence`.

```python
"""
Propose a non‑negative step function that (almost) minimises
evaluate_sequence.  The search consists of:

1.  A tiny, diverse set of analytic seeds (uniform, arcsine, β‑,
    edge‑weighted, Hann, Kaiser, Chebyshev, random Dirichlet,
    sparse) for a few moderate lengths (≈4000–8000).

2.  A long, momentum‑based exponentiated‑gradient descent on the exact
    objective.  The active convolution entry is identified each iteration,
    its sub‑gradient is used, the update is a·exp(‑η·v) with a decaying
    learning‑rate η₀/√t and a simple momentum term (β=0.9).  After each
    step the vector is projected onto the simplex and (optionally)
    symmetrised.

3.  A final simulated‑annealing / pairwise mass‑transfer polishing
    that keeps any improvement and, early on, accepts worsening moves
    with a temperature that decays linearly to zero.

The whole procedure runs for at most 1000 s and returns the best
sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based self‑convolution (full linear)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()          # next power of 2
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective (identical to evaluate_sequence, but without the caps)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a (sum is not forced to 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic families – each returns a probability vector (sum == 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^(-p) where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(-|x‑½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

def _distribution_chebyshev(n: int, atten: float) -> np.ndarray:
    w = chebwin(n, at=atten, sym=True).astype(np.float64)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_kaiser(n: int, beta: float) -> np.ndarray:
    w = kaiser(n, beta, sym=True).astype(np.float64)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Gradient of the *currently maximal* convolution entry
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Compute the sub‑gradient of max_k (a*a)_k.
    Returns (grad, k) where grad_i = 2·a_{k-i} (out‑of‑range ignored).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))

    grad = np.zeros_like(a)
    if k < n:
        # overlap length = k+1
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        # k ≥ n: overlap length = 2·n-1-k
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
# Momentum exponentiated‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _exp_grad_descent_momentum(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 2.0,
    beta: float = 0.9,
    eval_every: int = 200,
    symmetrize: bool = True,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent with a simple momentum term.
    The vector stays in the simplex by construction.
    """
    n = a.shape[0]
    v = np.zeros_like(a)                 # momentum accumulator
    it = 0
    while time.time() < deadline:
        it += 1
        grad, _ = _grad_of_max_conv(a)
        # momentum update (exponential moving average)
        v = beta * v + (1.0 - beta) * grad
        eta = eta0 / math.sqrt(it)       # decaying step size
        a = a * np.exp(-eta * v)          # exponentiated‑gradient step
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if symmetrize:
            # enforce a[i] == a[n-1-i] (helps the landscape)
            a = (a + a[::-1]) * 0.5

        if it % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] GD it={it} new global best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing / pairwise mass transfer polishing
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins; accept with
    Metropolis criterion using a linearly decreasing temperature."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = rng.choice(np.where(a > 1e-12)[0])
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # up to 40 % of source mass
        delta = a[src] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        cur = _objective(new_a)

        # linear cooling schedule
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal src={src}→dst={dst} δ={delta:.3e}"
                    f" new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear interpolation while preserving the simplex
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= a_new.sum()
    return a_new
```
