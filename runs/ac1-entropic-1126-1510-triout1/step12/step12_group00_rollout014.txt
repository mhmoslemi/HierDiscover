from scipy.sparse import diags
from scipy.optimize import linprog

def _lp_maximize_sum(f, M):
    n = len(f)
    # Build convolution matrix A of shape (2n-1, n)
    # Each diagonal offset d (0 <= d < n) has constant value f[d]
    data = [np.full(n, f[d]) for d in range(n)]
    offsets = np.arange(n)  # 0..n-1
    A = diags(data, offsets, shape=(2*n-1, n), format='csr')
    # objective: minimize -sum(g) (i.e., maximize sum(g))
    c = -np.ones(n)
    # inequality constraints: A @ g <= M
    b = np.full(2*n-1, M, dtype=np.float64)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback to zeros
        return np.zeros_like(f)
    return res.x
```

"""
Multi‑phase optimization for the step‑function bound.

1.  Build a diverse pool of analytic and random seeds (uniform,
    arcsine, symmetric beta, edge‑weighted, edge‑inverse‑power,
    sine‑weighted, triangular “tent”, plus Dirichlet and sparse vectors,
    plus random convex mixtures of several families).

2.  For each promising seed run three optimisation phases:
    * Coarse exponentiated‑gradient (large learning‑rate, loose active set).
    * Smooth Adam version that uses a soft‑max weight on all convolution
      entries (λ controls the approximation of the max).
    * Fine exponentiated‑gradient (tight active set).

   All updates are projected onto the probability simplex.

3.  A short linear‑programming refinement: maximise the total mass of a
    vector *g* under the linear constraints  (f * g) ≤ M, where *M* is the
    current maximal convolution value of the best vector *f*.  The improved
    vector is mixed with *f*.

4.  Final greedy pairwise swaps and a cheap simulated‑annealing local
    search clean up the remaining irregularities.

The routine respects the 1000 s budget and returns a plain Python list.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft
from scipy.sparse import diags
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    """Second edge family: w_i ∝ (i+0.5)^(−p) + (n−i−0.5)^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (-p) + j ** (-p))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight proportional to sin(π (i+½)/n)^p, vanishing at the ends."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_tent(n: int) -> np.ndarray:
    """Triangular (tent) shape, peak in the middle, zero at ends."""
    i = np.arange(n, dtype=np.float64)
    w = np.minimum(i + 1, n - i)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet sample with a random concentration."""
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_families(bases: list[np.ndarray], rng: np.random.Generator) -> np.ndarray:
    """Random convex mixture of the supplied bases."""
    w = rng.dirichlet(np.ones(len(bases)))
    mix = np.zeros_like(bases[0])
    for coeff, base in zip(w, bases):
        mix += coeff * base
    mix /= mix.sum()
    return mix

# Exponentiated‑gradient (mirror) descent -------------------------------
def _eg_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.7,
    active_frac: float = 0.04,
    max_active: int = 20,
    beta: float = 0.9,
    eval_interval: int = 150,
    patience: int = 2500,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent with momentum on the exact max‑conv objective."""
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    ind_grid = np.arange(n, dtype=np.int32)

    while time.time() < deadline and t < 300_000:
        t += 1

        # full self‑convolution
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            top_idx = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top_idx]

        grad = np.zeros_like(a)
        for idx in act:
            ind = idx - ind_grid
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * a[ind[mask]]

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = lr0 / math.sqrt(t)

        a = a * np.exp(-eta * momentum)  # multiplicative update
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] EG it={t:,} new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            rng = np.random.default_rng()
            jitter = rng.exponential(scale=0.005, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a, s=1.0)
            no_improve = 0

    return best_seq, best_score

# Adam optimisation with soft‑max weighting ----------------------------
def _adam_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr: float = 0.08,
    lam: float = 30.0,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_interval: int = 200,
) -> tuple[np.ndarray, float]:
    """Adam on a soft‑max approximation of the max‑conv."""
    n = a.shape[0]
    t = 0
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    ind_grid = np.arange(n, dtype=np.int32)

    while time.time() < deadline and t < 200_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()
        # soft‑max weights
        w = np.exp(lam * (conv - max_c))
        Z = w.sum()
        w /= Z  # now Σ w = 1, heavily weighted on near‑max entries

        # smooth gradient = Σ w_k * ∂conv_k/∂a
        grad = np.zeros_like(a)
        # For each k we could add w[k]*2*a shifted, but we can vectorise:
        # grad = 2 * Σ_k w[k] * shift(a, k)
        # Implement by convolution with flipped weight vector.
        # Convolution approach: grad = 2 * (a * w_reversed)  (valid convolution)
        # where w is of length 2n-1.
        # For simplicity we loop over the few large weights (those > 1e-12).
        large = np.where(w > 1e-12)[0]
        for idx in large:
            ind = idx - ind_grid
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * a[ind[mask]] * w[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t:,} new best {best_score:.6f}")

    return best_seq, best_score

# Linear‑programming refinement ----------------------------------------
def _lp_refine(f: np.ndarray, M: float, deadline: float) -> np.ndarray:
    """Solve  max Σ g_i   s.t.   f*g ≤ M,   g ≥ 0 .   Return the (scaled) feasible g."""
    n = f.shape[0]
    # Build convolution matrix A (shape (2n-1, n))
    data = [np.full(n, f[d]) for d in range(n)]
    offsets = np.arange(n)  # 0 .. n-1
    A = diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    c = -np.ones(n)                     # minimise -Σg_i  ↔ maximise Σg_i
    b = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    # Solve with HiGHS; time out early if deadline is near.
    remaining = deadline - time.time()
    if remaining < 0.5:
        return f  # give up quickly

    # SciPy's linprog has no built‑in time limit; we simply abort after solving.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        return f
    g = res.x
    if g.sum() == 0.0:
        return f
    # Scale g back to a probability vector (sum = 1)
    g = g / g.sum()
    return g

# Deterministic pairwise swap refinement --------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small mass; keep the move only if the objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] pairwise i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score

# Simulated‑annealing style local search --------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            # Metropolis criterion
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

            if sc_cur < overall_best_score:
                overall_best_score, overall_best_seq = sc_cur, a_cur.copy()
                print(f"[{time.time():.1f}] after Adam seed {idx} new best {overall_best_score:.6f}")

            # --- fine EG ---
            sub_deadline = time.time() + per_seed * 0.2
            a_cur, sc_cur = _eg_descent(
                a_cur,
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.5,
                active_frac=0.02,
                max_active=25,
                beta=0.9,
                eval_interval=150,
                patience=2000,
            )
            if sc_cur < overall_best_score:
                overall_best_score, overall_best_seq = sc_cur, a_cur.copy()
                print(f"[{time.time():.1f}] after fine EG seed {idx} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Linear‑programming refinement (few iterations)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] LP refinement")
        max_lp_iter = 4
        for it in range(max_lp_iter):
            if time.time() >= deadline:
                break
            M = _fft_self_convolve(overall_best_seq).max()
            g = _lp_refine(overall_best_seq, M, deadline)
            if g is overall_best_seq:
                # LP gave nothing new
                break
            # mix with a modest step size
            t_mix = 0.2
            new_seq = (1 - t_mix) * overall_best_seq + t_mix * g
            new_seq = _project_onto_simplex(new_seq, s=1.0)
            new_score = _objective(new_seq)
            if new_score < overall_best_score - 1e-14:
                overall_best_score, overall_best_seq = new_score, new_seq.copy()
                print(f"[{time.time():.1f}] LP iter {it} improved to {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] Greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 7. Final stochastic local search (simulated annealing style)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] Final random‑local search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
