def _lp_step(f, M, epsilon=1e-12):
    n = f.shape[0]
    # Ensure f>epsilon
    a = np.maximum(f, epsilon)
    # Build A
    A = np.empty((2*n-1, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = a
    # Objective: maximize sum(g) -> minimize -sum
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= 0:
        return None
    g = g0 / sum_g0
    return g

"""
Hybrid search for a near‑optimal step function.

The algorithm proceeds in three stages:

1. **Analytic families** – a variety of parametric families (uniform,
   arcsine, symmetric Beta) are generated.  For the Beta(p,p) family the
   shape parameter p is refined by a bounded scalar optimisation
   (scipy.optimize.minimize_scalar) because the best discrete constant
   differs slightly from the continuous p=1/2 value.

2. **Adam smooth‑max polishing** – the best few seeds (by the exact objective)
   are optimised with a smooth‑max surrogate using the Adam optimiser.
   The probabilities are parametrised by a soft‑max (ensuring non‑negativity
   and unit sum).  The temperature ``α`` is increased geometrically up to a
   large value so the surrogate converges to the true max.  Periodic exact
   evaluations keep the best vector seen.

3. **Greedy pairwise swap** – a cheap random mass‑transfer walk accepts only
   improving moves.  It often yields tiny final gains.

All steps respect the user‑supplied time budget (default 1000 s).  The
routine returns the best sequence as a plain ``list[float]``; scaling of the
vector does not affect the objective because the evaluation function is
scale‑invariant.
"""

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a))/sum(a)^2 (scale‑invariant, assumes sum>0).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) density on the grid i+0.5 / n."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid exact 0/1 which would lead to -inf in log
    eps = 1e-15
    w = np.exp((p - 1) * np.log(np.clip(x, eps, 1.0 - eps)) +
               (p - 1) * np.log(np.clip(1.0 - x, eps, 1.0 - eps)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals

def _softmax(z: np.ndarray) -> np.ndarray:
    shift = np.max(z)
    exp_z = np.exp(z - shift)
    s = exp_z.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / len(z))
    return exp_z / s


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_i w_i·(a⋆a)_i  w.r.t. a .
    Implements 2·a⋆w via FFT cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


def _adam_optimize(z0: np.ndarray,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   alpha0: float = 1.0,
                   alpha_max: float = 2000.0,
                   growth_factor: float = 1.03,
                   lr: float = 0.12,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   max_iter: int = 250000,
                   eval_interval: int = 300) -> (np.ndarray, float):
    """
    Adam optimiser on the smooth‑max surrogate.
    ``alpha`` is the temperature of the log‑sum‑exp approximation.
    ``z`` are the log‑probabilities (soft‑max parametrisation).
    """
    n = z0.shape[0]
    z = z0.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        a = _softmax(z)                     # a is a probability vector
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        # smooth‑max weights
        w = np.exp(alpha * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        grad_a = _gradient_from_weights(a, w)

        # chain rule through softmax
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        z = z - step

        # temperature schedule
        if t % 500 == 0:
            alpha = min(alpha * growth_factor, alpha_max)

        # periodic exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.1f} new best {best_score:.6f}")

    return best_seq, best_score


def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> (np.ndarray, float):
    """Random mass‑transfer walk that only accepts improving moves."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Parameter optimisation for the symmetric Beta family
# ----------------------------------------------------------------------
def _optimise_beta_sym(n: int, deadline: float) -> np.ndarray:
    """
    Finds the p that minimises the objective for Beta(p,p) on length n.
    Uses a bounded scalar optimisation (scipy.optimize.minimize_scalar).
    """
    # cheap inner objective (no logging)
    def f(p):
        a = _distribution_beta_sym(n, p)
        return _objective(a)

    # search in a reasonable interval; p=0.5 is arcsine, we broaden a bit
    bounds = (0.1, 2.0)
    res = minimize_scalar(f, bounds=bounds, method='bounded',
                          options={'xatol': 2e-3, 'maxiter': 30})
    p_opt = res.x if res.success else 0.5
    print(f"[{time.time():.1f}] Beta(p,p) optimisation gave p={p_opt:.4f}")
    return _distribution_beta_sym(n, p_opt)

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function minimizing ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # a small safety margin

    # --------------------------------------------------------------
    # Choose a modest length – the constant factor (2 n) cancels out,
    # so the exact value of n does not affect the minimiseable constant.
    # n = 2000 gives a good trade‑off between resolution and speed.
    # --------------------------------------------------------------
    n = 2000

    # --------------------------------------------------------------
    # 1.  Build a diverse set of seed vectors.
    # --------------------------------------------------------------
    seeds: List[np.ndarray] = []

    seeds.append(_distribution_uniform(n))
    seeds.append(_distribution_arcsine(n))

    # Fine‑tuned symmetric Beta
    seeds.append(_optimise_beta_sym(n, deadline))

    # Random Dirichlet
    for _ in range(3):
        seeds.append(_distribution_dirichlet(n, rng, alpha=0.5 + 0.4 * rng.random()))

    # Random sparse patterns
    seeds.append(_distribution_random_sparse(n, rng, density=0.02))

    # A few completely random probability vectors
    for _ in range(3):
        v = rng.random(n)
        v = np.maximum(v, 0.0)
        v /= v.sum()
        seeds.append(v)

    # --------------------------------------------------------------
    # 2.  Evaluate all seeds and keep the best few.
    # --------------------------------------------------------------
    best_score = float("inf")
    best_seq = None
    evaluated = []

    for vec in seeds:
        if time.time() >= deadline:
            break
        # protect against exact zeros (LP would become unbounded)
        vec = np.maximum(vec, 1e-12)
        vec /= vec.sum()
        sc = _objective(vec)
        evaluated.append((sc, vec.copy()))
        if sc < best_score:
            best_score = sc
            best_seq = vec.copy()
            print(f"[{time.time():.1f}] init seed score {sc:.6f}")

    # keep the top 5 candidates for polishing
    evaluated.sort(key=lambda x: x[0])
    top_candidates = [seq for _, seq in evaluated[:5]]

    # --------------------------------------------------------------
    # 3.  Adam smooth‑max polishing on the best candidates.
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # allocate a fraction of the remaining budget to Adam
        per_candidate = remaining / max(1, len(top_candidates))
        for cand in top_candidates:
            if time.time() >= deadline:
                break
            slice_deadline = time.time() + per_candidate * 0.9

            # initialise Adam on the soft‑max parametrisation
            z0 = np.log(np.maximum(cand, 1e-300))

            best_seq, best_score = _adam_optimize(
                z0,
                slice_deadline,
                best_seq=best_seq,
                best_score=best_score,
                alpha0=0.5,
                alpha_max=2000.0,
                growth_factor=1.03,
                lr=0.12,
                max_iter=200000,
                eval_interval=400,
            )
            print(f"[{time.time():.1f}] after Adam best {best_score:.6f}")

    # --------------------------------------------------------------
    # 4.  Final greedy pairwise‑swap walk (very cheap, often yields tiny gains)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _pairwise_swap_local(
            best_seq.copy(),
            deadline,
            best_seq=best_seq,
            best_score=best_score,
        )
        print(f"[{time.time():.1f}] after final swaps best {best_score:.6f}")

    # --------------------------------------------------------------
    # 5.  Return as plain list.
    # --------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
