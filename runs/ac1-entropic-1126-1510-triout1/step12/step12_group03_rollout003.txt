"""
Algorithm to minimise `evaluate_sequence`.

We search over several analytic families (uniform, sine, power‑sine,
edge‑weighted with both positive and negative exponents) on a handful of
lengths up to 64 000.  For each length we try a small grid of the shape
parameter and a convex mixing with the uniform distribution.  The best
~30 candidates are kept.

Each kept candidate is refined with an exponentiated‑gradient (mirror)
descent that uses the `top_k` largest convolution entries to build a
sub‑gradient.  After each multiplicative update we (i) symmetrise the
vector (the problem is symmetric), (ii) renormalise it onto the simplex.
The step size decays as `η_t = lr0 / sqrt(t)`.

Finally a simple hill‑climbing “pairwise swap” moves a small random
fraction of mass from one index to another whenever it strictly improves
the exact objective.

All operations respect the 1000 s budget and all numbers are kept
non‑negative with unit ℓ₁‑norm, so the returned list can be passed
directly to `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using real FFT (no padding artefacts)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective `2·n·max(conv(a,a))` *provided* Σa = 1.
    The function does **not** perform the clipping that `evaluate_sequence`
    applies – our search maintains positivity and unit sum.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_sine_power(n: int, p: float) -> np.ndarray:
    """sin⁽ᵖ⁾ window – p>0 concentrates mass in the centre, p<0 pushes mass to the edges."""
    i = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(np.pi * i) ** p
    # guard against p very large negative (underflow)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """
    Weight ∝ ((i+½)*(n‑i‑½))^(‑p).
    Positive p → edge‑heavy; negative p → centre‑heavy.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Sub‑gradient / mirror descent ----------------------------------------
def _exp_gradient_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.7,
    top_k: int = 10,
    eval_every: int = 150,
    max_iter: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the exact max‑convolution objective.
    Sub‑gradient is assembled from the `top_k` largest convolution entries.
    After each multiplicative update the vector is (i) symmetrised and
    (ii) normalised to the simplex.
    The best sequence encountered is returned.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit and t < max_iter:
        t += 1
        conv = _fft_self_convolve(a)

        # indices of the largest entries (including centre)
        idxs = np.argpartition(-conv, range(top_k))[:top_k]

        grad = np.zeros_like(a)
        for idx in idxs:
            if idx < n:
                # contribution to positions 0 … idx
                grad[: idx + 1] += 2.0 * a[idx::-1]
            else:
                start_i = idx - n + 1          # first index receiving mass
                # contribution to positions start_i … n‑1
                grad[start_i:] += 2.0 * a[start_i:][::-1]

        # multiplicative (exponentiated) update – keeps positivity
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # symmetrise (the problem is invariant under reversal)
        a = (a + a[::-1]) * 0.5

        # normalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new global best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise polishing --------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Randomly transfer a fraction of mass from one index to another.
    The move is kept only if the exact objective strictly improves.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine:

      1. scans a family of analytic shapes for several lengths,
      2. refines the most promising seeds with an exponentiated‑gradient
         descent that respects the symmetry of the problem,
      3. finishes with a greedy pairwise hill‑climbing.

    The whole process respects the supplied time budget and returns a
    plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, seq)

    # ------------------------------------------------------------------
    # 1. Rich family scan
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 16000, 32000, 64000]

    # parameter grids
    p_grid_pos = np.concatenate(
        [
            np.logspace(-2, -0.3, 8),      # 0.01 … 0.5 (edge‑heavy)
            np.linspace(0.5, 2.0, 6),       # moderate
            np.linspace(2.0, 5.0, 4),       # centre‑heavy
        ]
    )
    p_grid_neg = -np.logspace(-2, 0.7, 8)   # negative exponents → centre‑heavy
    p_grid = np.concatenate((p_grid_neg, p_grid_pos))

    lam_grid = np.linspace(0.0, 1.0, 5)      # mixing weight with uniform
    sin_pow_grid = [0.5, 1.0, 2.0, 3.0, 4.0]  # exponent for sin^p windows

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families ------------------------------------------------
        for fn in (_distribution_uniform, _distribution_arcsine,
                   _distribution_sine):
            cand = fn(n)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # sine‑power family ----------------------------------------------------
        for p in sin_pow_grid:
            cand = _distribution_sine_power(n, p)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(
                    f"[{time.time():.1f}] sin^({p}) len={n} score={sc:.6f}"
                )

        # edge‑weighted family (both signs of p) -------------------------------
        for p in p_grid:
            if time.time() >= deadline:
                break
            base = _distribution_edge_weighted(n, p)
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(
                        f"[{time.time():.1f}] edge p={p:.3g} lam={lam:.2f} len={n} score={sc:.6f}"
                    )

        # random Dirichlet seeds ------------------------------------------------
        for _ in range(3):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds ----------------------------------------------------
        for _ in range(2):
            if time.time() >= deadline:
                break
            a = _distribution_edge_weighted(n, p=0.0)  # start uniform
            # zero out a large fraction
            mask = rng.random(n) < 0.95
            a[mask] = 0.0
            if a.sum() == 0.0:
                a[0] = 1.0
            a /= a.sum()
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

        # keep only the top~30 candidates so far
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2. Include any previously discovered construction, if present.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never trigger)
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 8000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Mirror‑descent refinement on the best seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0 and top_candidates:
        # keep the 5 best seeds for deep refinement
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] start EG refinement on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sg_deadline = time.time() + per_seed
            a_opt, sc_opt = _exp_gradient_descent(
                seed_seq,
                sg_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.7,
                top_k=10,
                eval_every=120,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] after EG seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5. Final greedy pairwise polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (2·n·max(conv)): {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list – `evaluate_sequence` will perform its own
    # normalisation and safety clipping.
    return overall_best_seq.tolist()
