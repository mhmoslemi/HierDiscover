"""
Improved search for a non‑negative step function that minimizes
``evaluate_sequence``.

Algorithm
---------

1. **Seed generation** – a diverse pool of probability vectors (uniform,
   arcsine, Hann, symmetric Beta, edge‑weighted, generalized‑normal,
   random Dirichlet and sparse).  All vectors are normalised to sum to 1.

2. **Exact exponentiated‑gradient descent** – for the most promising
   seeds we minimise the true objective
        2·n·max(conv(a,a))/sum(a)²
   by updates of the form ``a ← a·exp(-η·g)`` where ``g`` is the gradient
   of the currently maximal convolution entry.  The step size η decays as
   1/√t.  The best vector is kept globally.

3. **Upsampling** – the current best vector is linearly interpolated to a
   double length (preserving the simplex) and a short burst of the same
   gradient descent is performed again.

4. **Greedy pairwise mass‑transfer** – while time permits we repeatedly
   move a small fraction of mass from the index with the largest
   gradient (i.e. the strongest contributor to the active convolution
   peak) to an index with the smallest gradient.  The move is kept if
   it improves the objective (or with a small temperature‑dependent
   probability early on).  The step size is exponentially decayed.

5. **Final annealing polish** – a conventional simulated‑annealing
   routine that randomly transfers mass between two bins.  Only moves
   that improve the objective or are accepted by the temperature schedule
   are kept.

All phases respect the global 1000 s budget (the function is called with
``budget_s=1000``).  The returned list is ready for ``evaluate_sequence``.
"""

# Linear algebra utilities

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, linear convolution)
# ----------------------------------------------------------------------
def _next_pow_two(x: int) -> int:
    return 1 << (x - 1).bit_length()


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    n = len(x)
    m = len(y)
    size = _next_pow_two(n + m - 1)
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[: n + m - 1]


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
# Exact objective used by ``evaluate_sequence``.
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a given a (any scale)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_conv / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, sum=1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) density on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{-p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.01) -> np.ndarray:
    """Sparse random vector with given density (at least two non‑zeros)."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation while staying on the probability simplex."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Gradient of the *active* convolution entry (the index attaining the max)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Return the gradient of the currently maximal self‑convolution entry
    together with the index where the maximum occurs.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))

    grad = np.zeros_like(a)
    if k < n:
        # overlap length = k+1
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        # k >= n : overlap length = 2·n-1-k
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k

    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """Gradient descent using the sub‑gradient of the active max convolution entry."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)

        # Normalise gradient to unit max (prevents under‑/overflow)
        max_g = grad.max()
        if max_g > 0:
            grad = grad / max_g

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # keep on the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise mass‑transfer descent (deterministic)
# ----------------------------------------------------------------------
def _greedy_pairwise_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    step_init: float = 0.25,
    step_decay: float = 0.9997,
    min_step: float = 1e-7,
    eval_every: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a fraction of mass from the index with the largest
    gradient (the strongest contributor to the active peak) to an index
    with the smallest gradient.  The step size decays exponentially.
    """
    n = a.shape[0]
    step = step_init
    iter_cnt = 0
    start = time.time()
    while time.time() < deadline:
        iter_cnt += 1

        grad, _ = _grad_of_max_conv(a)

        # pick donor (largest gradient) among entries with appreciable mass
        donor_candidates = np.where((grad > 0) & (a > 1e-12))[0]
        if donor_candidates.size == 0:
            break
        i_src = donor_candidates[grad[donor_candidates].argmax()]

        # pick receiver (smallest gradient) – allow zero‑gradient entries
        receiver_candidates = np.where(a < 0.99)[0]  # almost anything is allowed
        i_dst = rng.choice(receiver_candidates)

        # amount to move
        delta = a[i_src] * step
        if delta < 1e-16:
            delta = step * 1e-12
        if delta > a[i_src]:
            delta = a[i_src]

        new_a = a.copy()
        new_a[i_src] -= delta
        new_a[i_dst] += delta

        cur = _objective(new_a)

        # simple annealing schedule (linear cooling)
        elapsed = time.time() - start
        total = deadline - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.02

        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            if rng.random() < math.exp(-(cur - best_score) / temperature):
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] greedy i={i_src}->{i_dst} δ={delta:.3e} new best {best_score:.6f}")

        # decay step size
        step = max(step * step_decay, min_step)

        if iter_cnt % eval_every == 0:
            # periodic sanity check (in case we got stuck)
            cur_obj = _objective(a)
            if cur_obj < best_score - 1e-14:
                best_score = cur_obj
                best_seq = a.copy()
                print(f"[{time.time():.1f}] greedy periodic best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style pairwise polishing (as in the original code)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Random mass‑transfer moves accepted by a temperature schedule."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)

        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            if rng.random() < math.exp(-(cur - best_score) / temperature):
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that (almost) minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Build a compact yet diverse pool of analytic seeds.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000]            # keep lengths modest for speed
    p_grid = np.concatenate(
        [
            np.logspace(-2.0, -0.5, 9, base=10),
            np.logspace(-0.5, 0.0, 9, base=10),
            np.logspace(0.0, 0.5, 8, base=10),
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 5)      # mixing with the uniform distribution
    q_grid = np.linspace(0.5, 3.0, 6)       # generalized‑normal shapes

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n), _distribution_hann(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # parameterised families
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)

            for base in (beta, edge):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(f"[{time.time():.1f}] new best p={p:.3g} lam={lam:.2f} len={n} score={sc:.6f}")

        # generalized normal shapes
        for q in q_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_generalized_normal(n, q)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best gn q={q:.2f} len={n} score={sc:.6f}")

        # random Dirichlet seeds
        for _ in range(5):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(5):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.015)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best sparse len={n} score={sc:.6f}")

        # keep only the very best 30 seeds
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2. Incorporate any previously discovered construction.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev[:] = 1.0 / prev.size
            else:
                prev /= prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback – guarantee a non‑empty result.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Exact exponentiated‑gradient descent on a few top seeds.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seed_seqs = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seed_seqs))
        print(f"[{time.time():.1f}] gradient descent on {len(seed_seqs)} seeds")
        for idx, seed_seq in enumerate(seed_seqs):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _exponentiated_gradient_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.9,
                eval_every=100,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] after EG seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Upsample the current best (if time permits) and refine again.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n <= 64000:
            print(f"[{time.time():.1f}] upsampling {cur_n} → {target_n}")
            up_seq = _upsample(overall_best_seq, target_n)
            up_score = _objective(up_seq)
            if up_score < overall_best_score:
                overall_best_score = up_score
                overall_best_seq = up_seq
                print(f"[{time.time():.1f}] upsample led to new best {overall_best_score:.6f}")

            # short burst of gradient descent on the upsampled vector
            sub_deadline = time.time() + (deadline - time.time()) * 0.3
            a_opt, sc_opt = _exponentiated_gradient_descent(
                overall_best_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.7,
                eval_every=80,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] EG after upsample new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Deterministic greedy pairwise mass‑transfer descent.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] starting greedy pairwise descent")
        a_opt, sc_opt = _greedy_pairwise_descent(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
            step_init=0.25,
            step_decay=0.9995,
            min_step=1e-7,
            eval_every=200,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after greedy descent new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final long‑run annealing / polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing polish")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
            max_trials=None,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after final annealing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)/sum²): {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    # Return a plain Python list as required by the evaluation harness
    return overall_best_seq.tolist()
