def _corr_fft(a, w):
    n = a.shape[0]
    m = w.shape[0]  # 2n-1
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())

    corr = np.fft.irfft(A * W, n=size)
    return corr[:need]

    tau = 2.0
    tau_mul = 1.0012
    eta0 = 0.4
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    m = np.zeros(n)
    v = np.zeros(n)
    t=0
    no_improve = 0
    rng = np.random.default_rng()
    while time.time() < deadline and t < 300000:

        conv = _fft_self_convolve(a)

"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.

Algorithm
---------
1. **Diversified seed pool** – generate candidates from many analytic families
   (uniform, arcsine, symmetric/asymmetric Beta, edge‑weighted, sine‑weighted,
   geometric) together with random Dirichlet and sparse vectors for several
   lengths.  Keep the ~30 best seeds (by the exact objective).

2. **Adaptive Adam‑based exponential descent** – for each promising seed run a
   mirror‑descent on the simplex using an Adam‑style adaptive step.  The loss
   is the true objective `2·n·max(conv(a,a))`.  A soft‑max (temperature τ)
   over *all* convolution entries is used to obtain a smooth sub‑gradient.
   The temperature slowly grows, turning the soft‑max into the true max.
   Jitter is injected when the objective does not improve for a while.

3. **Deterministic pairwise refinement** – small mass transfers between two
   positions are accepted only if they improve the objective.

4. **Cheap simulated‑annealing walk** – a final stochastic local search that
   occasionally accepts worsening moves, helping to escape tiny local minima.

The whole routine respects a user‑supplied time budget (default 1000 s).  The
returned list is a plain Python list; ``evaluate_sequence`` will perform the
final checks and normalisation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray, size: int) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2·n‑1)."""
    A = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(A * A, n=size)
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


# ----------------------------------------------------------------------
def _corr_fft(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Linear cross‑correlation (a ⋆ w)[i] = Σ_j a[j]·w[i+j] for i = 0..len(a)-1.
    Implemented via FFT; the returned array has length len(a)+len(w)-1.
    """
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w, n=size).conjugate()
    corr_full = np.fft.irfft(A * W, n=size)
    return corr_full[: a.shape[0] + w.shape[0] - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    # size large enough for both conv (2n‑1) and correlation (3n‑2)
    size = 1 << ((3 * n - 2 - 1).bit_length())
    conv = _fft_self_convolve(a, size)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric beta (Beta(a,b)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight proportional to sin(π (i+½)/n)^p, vanishing at the ends."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_geometric(n: int, r: float) -> np.ndarray:
    """Symmetric geometric shape: w_i ∝ r**i + r**(n‑1‑i).  r∈(0,1)."""
    i = np.arange(n, dtype=np.float64)
    w = r ** i + r ** (n - 1 - i)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Adam‑style exponential descent ----------------------------------------
def _adam_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.45,
    tau0: float = 2.5,
    tau_mult: float = 1.0012,
    eval_interval: int = 200,
    patience: int = 2500,
) -> tuple[np.ndarray, float]:
    """
    Adam‑mirrored gradient descent on the exact objective.
    A soft‑max w ∝ exp(τ·conv) furnishes a smooth weighted sub‑gradient.
    """
    n = a.shape[0]
    size = 1 << ((3 * n - 2 - 1).bit_length())  # enough for both FFTs
    rng = np.random.default_rng()
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0
    no_improve = 0
    tau = tau0

    while time.time() < deadline and t < 500_000:
        t += 1

        # ----- self‑convolution (exact) -----
        conv = _fft_self_convolve(a, size)          # length 2n‑1
        max_c = conv.max()

        # ----- soft‑max weighting -----
        w_raw = np.exp(tau * (conv - max_c))        # numerically stable
        w = w_raw / w_raw.sum()

        # ----- correlation = weighted gradient -----
        # grad_i = 2 * Σ_j a_j * w_{i+j}
        corr = _corr_fft(a, w, size)                # length 3n‑2
        grad = 2.0 * corr[:n]

        # ----- Adam update on the simplex (multiplicative) -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = eta0 / math.sqrt(t)                  # decaying base step
        a = a * np.exp(-eta * m_hat / (np.sqrt(v_hat) + eps))

        # re‑project onto probability simplex Σa = 1, a ≥ 0
        a = _project_onto_simplex(a, s=1.0)

        # temperature schedule (focuses more on the true max)
        tau *= tau_mult

        # ----- occasional evaluation -----
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(
                    f"[{time.time():.1f}] adam it={t} new best {best_score:.6f}"
                )
            else:
                no_improve += 1

        # ----- jitter on stagnation -----
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.005, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise swap (deterministic local refinement) ------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small mass; keep move only if objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style local search ---------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}"
            )
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2. Include the previously discovered construction if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback guard (should never be needed)
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Adam‑based descent on the most promising seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] starting Adam descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _adam_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.55,
                tau0=2.5,
                tau_mult=1.0014,
                eval_interval=150,
                patience=2000,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after Adam seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5. Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 6. Final stochastic local search (simulated annealing style)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final random‑local search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list; evaluate_sequence will handle clipping/normalisation.
    return overall_best_seq.tolist()
