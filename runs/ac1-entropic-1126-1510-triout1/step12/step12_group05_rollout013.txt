"""
Hybrid optimizer for the self‑convolution max‑ratio problem.

Approach
--------
1.  Generate a large, diverse pool of seed step‑functions of various lengths.
    Families include uniform, arcsine (with several shifts), symmetric Beta,
    asymmetric Beta, exponential, power‑law, sparse Dirichlet draws,
    two‑block, and sine distributions.

2.  Keep the best seeds (by the exact objective) in a global elite list.

3.  For each elite seed run three refinement phases:
    * Phase‑A – soft‑max (temperature‑controlled) exponentiated‑gradient descent.
    * Phase‑B – hard‑max sub‑gradient exponentiated‑gradient descent.
    * Phase‑C – Adam‑style exponentiated‑gradient descent on the log‑variables.

    The learning‑rate decays as ηₜ ∝ 1/√t and every few hundred
    iterations the exact objective is evaluated.  Improvements are logged
    and the global best is updated.

4.  A cheap pairwise‑swap local walk (biased towards high‑gradient
    indices) polishes the final solution until the time budget expires.

All operations respect the simplex constraint Σaᵢ = 1, keep aᵢ ≥ 0,
and use FFT for O(n log n) self‑convolution.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)²).
    a is assumed to be non‑negative.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient 2·a⋆w for a weight vector w (length 2n‑1, Σw = 1).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.size - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _gradient_max_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the single maximal convolution entry conv[k].
    Returns a vector of size n with non‑negative entries.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i = np.arange(i_min, i_max + 1, dtype=np.int64)
    j = k - i

    eq_mask = (i == j)
    if np.any(eq_mask):
        g[i[eq_mask]] += 2.0 * a[i[eq_mask]]

    if np.any(~eq_mask):
        ii = i[~eq_mask]
        jj = j[~eq_mask]
        g[ii] += a[jj]
        g[jj] += a[ii]

    return g

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_arcsine_variant(n: int, shift: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5 + shift) * (n - i - 0.5 + shift))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * x / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Broad pool of analytic and random seeds for a given length."""
    cand = []

    # deterministic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_sine(n))
    cand.append(_distribution_two_blocks(n, block_frac=0.07))

    # arcsine shifts
    for shift in (-0.4, -0.2, 0.0, 0.2, 0.4, 0.8, 1.2, 2.0):
        cand.append(_distribution_arcsine_variant(n, shift))

    # symmetric beta on a finer grid
    for p in np.linspace(0.05, 2.0, 15):
        cand.append(_distribution_beta_sym(n, p))

    # random asymmetric beta
    for _ in range(8):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # power law
    for p in (0.5, 1.0, 2.0, 3.0):
        cand.append(_distribution_power_increasing(n, p))

    # exponential
    for lam in (0.001, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential_increasing(n, lam))

    # sparse random draws
    for dens in (0.005, 0.01, 0.02):
        cand.append(_distribution_random_sparse(n, rng, density=dens))

    # Dirichlet draws with various concentrations
    for a in (0.1, 0.3, 0.5, 1.0, 2.0):
        cand.append(_distribution_dirichlet(n, rng, alpha=a))

    # a few random Dirichlet with random concentration
    for _ in range(4):
        a = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_dirichlet(n, rng, alpha=a))

    # additional two‑block variants
    for frac in (0.02, 0.04, 0.12, 0.20):
        cand.append(_distribution_two_blocks(n, block_frac=frac))

    return cand

def _softmax_grad_descent(a: np.ndarray,
                          deadline: float,
                          best_seq: np.ndarray,
                          best_score: float,
                          lr0: float = 0.9,
                          max_iter: int = 60000,
                          eval_interval: int = 500,
                          alpha_start: float = 0.5,
                          alpha_end: float = 80.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a soft‑max approximation of max(conv).
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_conv = conv.max()

        # linear schedule for temperature α
        alpha = alpha_start + (alpha_end - alpha_start) * t / max_iter

        # numerically stable soft‑max weights
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # renormalise onto the simplex
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= total

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] soft‑max it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _max_grad_descent(a: np.ndarray,
                      deadline: float,
                      best_seq: np.ndarray,
                      best_score: float,
                      lr0: float = 0.5,
                      max_iter: int = 80000,
                      eval_interval: int = 500) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent using the true maximal convolution entry.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())

        grad = _gradient_max_index(a, k)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= total

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hard‑max it={t} idx={k} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _adam_exp_grad_descent(a: np.ndarray,
                           deadline: float,
                           best_seq: np.ndarray,
                           best_score: float,
                           lr: float = 0.5,
                           beta1: float = 0.9,
                           beta2: float = 0.999,
                           eps: float = 1e-8,
                           max_iter: int = 40000,
                           eval_interval: int = 400) -> tuple[np.ndarray, float]:
    """
    Adam‑style exponentiated‑gradient descent on log‑variables.
    Operates on the simplex via renormalisation each iteration.
    """
    n = a.shape[0]
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    log_a = np.log(np.maximum(a, 1e-30))
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())
        grad = _gradient_max_index(a, k)       # sub‑gradient w.r.t. a

        # gradient w.r.t. log(a) = a * grad
        g_log = a * grad

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g_log
        v = beta2 * v + (1.0 - beta2) * (g_log ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # update log‑variables
        log_a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        a = np.exp(log_a)
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """
    Cheap random mass‑transfer walk. Randomly moves up to 30 % of a
    source entry to a target entry and accepts only improvements.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue

        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Main entry point used by the checker.

    Generates many analytic and random seeds, refines the most promising
    ones with three gradient‑descent phases (soft‑max, hard‑max, Adam),
    and finally performs a greedy pairwise‑swap polishing.
    The best sequence found (as a plain list) is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.0   # a small safety margin

    # --------------------------------------------------------------
    # 1. Choose a set of lengths (shorter lengths are cheap to optimise,
    #    longer lengths give a tighter bound if they succeed).
    lengths = [2000, 4000, 8000, 12000]

    overall_best_seq = None
    overall_best_score = float("inf")
    elite = []   # (score, seq) tuples across all lengths

    # --------------------------------------------------------------
    # 2. Build a large pool of seeds and keep the best few
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] --- scanning length = {n} ---")
        # deterministic families + random draws
        for cand in _initial_candidates_extended(n, rng):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] !!! new global best len={n} score={sc:.6f}")
            elite.append((sc, cand.copy()))

        # keep only the top 20 across all processed lengths so far
        elite.sort(key=lambda x: x[0])
        elite = elite[:20]

    # --------------------------------------------------------------
    # 3. Include the previously discovered sequence if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best improved: {sc:.6f}")
            elite.append((sc, prev.copy()))
            elite.sort(key=lambda x: x[0])
            elite = elite[:20]
    except Exception:
        pass

    # --------------------------------------------------------------
    # 4. Fallback in the extremely unlikely case that no seed succeeded
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Main optimisation – three‑phase descent on the elite seeds
    #    (focus on the very best elite members)
    remaining = deadline - time.time()
    if remaining > 30.0:
        # take the best few elite seeds (up to 6)
        seeds = [seq for _, seq in sorted(elite, key=lambda x: x[0])[:6]]
        per_seed_budget = max(1.0, remaining / (len(seeds) * 3.0))

        for seed_seq in seeds:
            if time.time() >= deadline:
                break

            # ------ Phase A: soft‑max descent --------------------
            sub_deadline_a = time.time() + per_seed_budget
            seq_a, score_a = _softmax_grad_descent(
                seed_seq.copy(),
                sub_deadline_a,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                max_iter=80000,
                eval_interval=400,
                alpha_start=0.4,
                alpha_end=80.0,
            )
            if score_a < overall_best_score:
                overall_best_score = score_a
                overall_best_seq = seq_a.copy()
                print(f"[{time.time():.1f}] after soft‑max new best {overall_best_score:.6f}")

            # ------ Phase B: hard‑max sub‑gradient descent -------
            if time.time() >= deadline:
                break
            sub_deadline_b = time.time() + per_seed_budget
            seq_b, score_b = _max_grad_descent(
                seq_a.copy(),
                sub_deadline_b,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.5,
                max_iter=80000,
                eval_interval=400,
            )
            if score_b < overall_best_score:
                overall_best_score = score_b
                overall_best_seq = seq_b.copy()
                print(f"[{time.time():.1f}] after hard‑max new best {overall_best_score:.6f}")

            # ------ Phase C: Adam‑style exponentiated descent ----
            if time.time() >= deadline:
                break
            sub_deadline_c = time.time() + per_seed_budget
            seq_c, score_c = _adam_exp_grad_descent(
                seq_b.copy(),
                sub_deadline_c,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr=0.5,
                beta1=0.9,
                beta2=0.999,
                max_iter=50000,
                eval_interval=400,
            )
            if score_c < overall_best_score:
                overall_best_score = score_c
                overall_best_seq = seq_c.copy()
                print(f"[{time.time():.1f}] after Adam new best {overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Final polishing phase – cheap random pairwise swaps
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # --------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # Ensure result is a plain list of Python floats
    return overall_best_seq.tolist()
