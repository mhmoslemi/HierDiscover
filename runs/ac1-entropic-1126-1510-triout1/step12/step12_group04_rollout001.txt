"""
Hybrid search with soft‑max gradient descent.

The algorithm consists of three stages:

1.  **Broad seeding** – generate many analytic families
    (uniform, arcsine with several offsets, a dense sweep of the
    “inverse‑sqrt” exponent p, two‑block, sparse, Dirichlet).  All
    candidates are normalised to sum = 1 and evaluated with the exact
    objective.  The best few (≤ 12) are kept.

2.  **Soft‑max projected exponentiated‑gradient descent** – for each
    elite seed we minimise the exact objective
        2·n·max(conv(a,a))/sum(a)²   (with sum(a)=1)
    by iterating
        * compute the full self‑convolution,
        * build a soft‑max weight w_i ∝ exp(β·(conv_i−max)),
        * gradient g = 2·conv(w, a[::-1])  (length n),
        * update a ← a·exp(−η·g) and renormalise to the simplex.
    β starts small (≈0.5) and is slowly increased (≈*1.001 per iteration)
    so that the optimisation first sees a smooth landscape and then
    focuses on the true max.  η follows η₀/√t.  The best vector seen
    during the run is retained.

3.  **Greedy polishing** – cheap random mass‑transfer moves are
    accepted only when they improve the exact objective.

The whole routine respects the global time budget (default 1000 s)
and returns the best sequence (list of floats) found.  All numbers
are non‑negative and normalised to unit sum, satisfying the
`evaluate_sequence` requirements.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear (non‑circular) self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum(a)² (sum(a) should be 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine_offset(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = i + c
    right = (n - 1 - i) + c
    vals = 1.0 / np.sqrt(left * right)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _distribution_power_inv_sqrt(n: int, p: float) -> np.ndarray:
    """w_i ∝ ((i+0.5)*(n-i-0.5))^{-p/2}."""
    i = np.arange(n, dtype=np.float64)
    left = i + 0.5
    right = (n - 1 - i) + 0.5
    vals = (left * right) ** (-p / 2.0)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Generate a diverse set of seeds for a given length."""
    cand = []

    # 1. Uniform
    cand.append(_distribution_uniform(n))

    # 2. Arcsine with several offsets
    for c in (0.1, 0.3, 0.5, 1.0, 2.0):
        cand.append(_distribution_arcsine_offset(n, c))

    # 3. Power‑inverse‑sqrt sweep (dense grid)
    for p in np.linspace(0.4, 1.8, 8):
        cand.append(_distribution_power_inv_sqrt(n, float(p)))

    # 4. Two‑block patterns
    for bf in (0.04, 0.07, 0.10, 0.15):
        cand.append(_distribution_two_blocks(n, block_frac=bf))

    # 5. Sparse random
    cand.append(_distribution_random_sparse(n, rng, density=0.008))
    cand.append(_distribution_random_sparse(n, rng, density=0.02))

    # 6. Dirichlet draws
    for _ in range(4):
        alpha = 0.2 + 0.8 * rng.random()
        cand.append(_distribution_dirichlet(n, rng, alpha=alpha))

    return cand


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Returns (gradient, max_conv) where gradient = 2·conv(w, a_rev)
    with soft‑max weights w_i ∝ exp(beta·(conv_i−max_conv)).
    """
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    # soft‑max weights – numerically stable by subtracting max
    w = np.exp(beta * (conv - max_conv))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size, dtype=np.float64)
    else:
        w /= w_sum

    # gradient: 2 * (w * a_rev) convolution, extract the central n entries
    a_rev = a[::-1]
    grad_full = np.convolve(w, a_rev, mode="full")
    n = a.shape[0]
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad, max_conv


# ----------------------------------------------------------------------
def _smooth_expgd(a: np.ndarray,
                 deadline: float,
                 best_seq: np.ndarray,
                 best_score: float,
                 eta0: float = 1.0,
                 beta0: float = 0.5,
                 beta_max: float = 30.0,
                 beta_factor: float = 1.001) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the *soft‑max* approximation of max().
    Keeps the best vector seen.  Returns (best_sequence, best_score).
    """
    n = a.shape[0]
    t = 0
    beta = beta0
    eps = 1e-14

    while time.time() < deadline:
        t += 1

        # exact objective for monitoring
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] it={t} new best {best_score:.6f}")

        # soft‑max gradient
        grad, _ = _softmax_gradient(a, beta)

        # step size schedule
        eta = eta0 / math.sqrt(t)

        # exponentiated update (mirror descent on simplex)
        a = a * np.exp(-eta * grad)

        # avoid numerical collapse
        a = np.maximum(a, eps)

        # renormalise to the simplex
        total = a.sum()
        if total <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / total

        # slowly raise beta to sharpen the approximation
        beta = min(beta_max, beta * beta_factor)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Simple random mass‑transfer hill‑climb."""
    n = a.shape[0]
    rng = np.random.default_rng()
    eps = 1e-12

    while time.time() < deadline:
        # choose a non‑zero source
        src_idx = np.where(a > eps)[0]
        if src_idx.size == 0:
            break
        i = rng.choice(src_idx)
        j = rng.integers(n)
        if j == i:
            continue
        # move a random fraction (up to 30 %) of the source mass
        delta = a[i] * rng.random() * 0.30
        if delta < eps:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur_score = _objective(new_a)
        if cur_score < best_score - 1e-12:
            a = new_a
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score
