def _smooth_max_gradient(a, alpha):
    n = len(a)
    conv = _fft_self_convolve(a)
    # compute weights w_k = exp(alpha * (conv_k - max_conv)) for numerical stability
    max_c = conv.max()
    exps = np.exp(alpha * (conv - max_c))
    w = exps / exps.sum()
    # compute gradient = 2 * correlation(w, a)
    # Use FFT to compute correlation: correlation_i = Σ_j w_{i+j} * a_j
    # size L = next power of two >= len(w) + n - 1
    L = 1 << ((len(w) + n - 1 - 1).bit_length())
    w_fft = np.fft.rfft(w, n=L)
    a_fft = np.fft.rfft(a, n=L)
    # correlation via inverse FFT of w_fft * conj(a_fft)
    corr = np.fft.irfft(w_fft * np.conj(a_fft), n=L)
    # correlation length L; linear correlation part occupies first len(w)+n-1 entries (0..len(w)+n-2)
    # we need first n entries as gradient_i = 2 * corr[i] (since i from 0 to n-1)
    grad = 2.0 * corr[:n]
    return grad

def _smooth_max_gradient(a, alpha):

    conv = _fft_self_convolve(a)
    max_c = conv.max()
    # weights
    w = np.exp(alpha * (conv - max_c))

    w_fft = np.fft.rfft(w, n=L)
    a_fft = np.fft.rfft(a, n=L)

    corr = np.fft.irfft(w_fft * np.conj(a_fft), n=L)

"""
Propose a non‑negative step function (list of floats) that (almost) minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The optimiser works in three phases:

1.  **Seed generation** – a diverse collection of analytic families
    (uniform, arcsine, Hann, symmetric β, edge‑weighted, generalized‑Normal,
    Dirichlet, sparse) for several lengths.  The best few seeds are kept.

2.  **Smoothed exponential‑gradient descent** – instead of following only the
    active convolution entry, a *log‑sum‑exp* (smooth‑max) is used:
    wₖ ∝ exp(α·cₖ) where cₖ are the convolution values.  The gradient of this
    smooth upper‑bound is computed cheaply via one FFT‑based correlation,
    and the update   a ← a·exp(−η·∇)  is performed with projection onto the
    probability simplex.  The smoothing parameter α is increased
    gradually (α_start → α_end) so that early steps see a smooth landscape
    and later steps focus on the true maximum.

3.  **Upsampling + polishing** – the current best vector is linearly
    interpolated to a larger length, refined again by the smoothed descent,
    and finally polished by a long‑run pair‑wise mass‑transfer simulated
    annealing.

All steps respect the overall 1000 s budget; the best sequence found is
returned as a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# FFT‑based linear self‑convolution (full length 2·n‑1)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective used by ``evaluate_sequence`` (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic families (return probability vectors, Σ = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(−|x−½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Linear interpolation while staying on the simplex
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Smoothed max‑gradient (log‑sum‑exp) of the convolution vector
# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of   f_α(a) = (1/α)·log Σₖ exp(α·cₖ)   where c = a*a.
    The gradient is   ∇f_α = Σₖ wₖ ∂cₖ/∂a,
    with wₖ ∝ exp(α·cₖ).  It is computed via one FFT‑based correlation.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)          # length 2·n‑1
    max_c = conv.max()
    w = np.exp(alpha * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.shape[0])
    else:
        w /= w_sum                         # now Σ w = 1

    # correlation i ↦ Σⱼ w_{i+j} aⱼ   (linear, not circular)
    L = 1 << ((len(w) + n - 1 - 1).bit_length())
    w_fft = np.fft.rfft(w, n=L)
    a_fft = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(w_fft * np.conj(a_fft), n=L)
    grad = 2.0 * corr[:n]                 # gradient of smoothed max
    return grad


# ----------------------------------------------------------------------
# Smoothed exponential‑gradient descent on the *exact* objective
# ----------------------------------------------------------------------
def _smooth_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    alpha_start: float = 5.0,
    alpha_end: float = 200.0,
    eta0: float = 0.6,
    eval_every: int = 30,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smoothed max.  α grows linearly from
    alpha_start to alpha_end; step size η = η₀/√t.
    """
    n = a.shape[0]
    start = time.time()
    t = 0
    while time.time() < deadline:
        t += 1
        # schedule α
        elapsed = time.time() - start
        total = deadline - start
        if total <= 0:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (elapsed / total)

        grad = _smooth_max_gradient(a, alpha)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)          # stays non‑negative
        a = _project_onto_simplex(a)         # enforce Σa = 1

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth‑GD it={t} α={alpha:.1f} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple pairwise mass‑transfer simulated annealing (as in the reference)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = rng.choice(np.where(a > 1e-12)[0])
        # destination ≠ source
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        delta = a[src] * rng.random() * 0.4
        if delta < 1e-16:
            continue
        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01
        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True
        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal {src}->{dst} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_global = time.time()
    deadline = start_global + budget_s - 2.0               # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 1)  Rich pool of analytic seeds for a few moderate lengths.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]                     # increase later by up‑sampling
    top_per_len = 6                                        # keep a few promising seeds

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating seeds of length {n}")

        cand_pool = []

        # deterministic families
        cand_pool.append(_distribution_uniform(n))
        cand_pool.append(_distribution_arcsine(n))
        cand_pool.append(_distribution_hann(n))

        # parametric families (beta & edge‑weighted)
        for p in (0.1, 0.3, 0.5, 0.8, 1.0, 2.0):
            cand_pool.append(_distribution_beta_sym(n, p))
            cand_pool.append(_distribution_edge_weighted(n, p))

        # generalized normal shapes
        for q in (0.5, 0.8, 1.0, 1.2, 2.0, 3.0):
            cand_pool.append(_distribution_generalized_normal(n, q))

        # random Dirichlet seeds
        for _ in range(5):
            alpha = rng.uniform(0.01, 2.0)
            cand_pool.append(rng.dirichlet(np.full(n, alpha)))

        # sparse random seeds
        for _ in range(5):
            cand_pool.append(_distribution_random_sparse(n, rng, density=0.01))

        # evaluate and keep the best few
        scored = []
        for cand in cand_pool:
            sc = _objective(cand)
            scored.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        scored.sort(key=lambda x: x[0])
        top_seeds = [seq for _, seq in scored[:top_per_len]]

        # ------------------------------------------------------------------
        # 2)  Smoothed gradient descent on the top seeds.
        # ------------------------------------------------------------------
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        per_seed_deadline = time_left / max(1, len(top_seeds))

        for idx, seed_seq in enumerate(top_seeds):
            sub_deadline = time.time() + per_seed_deadline
            a_opt, sc_opt = _smooth_gradient_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                alpha_start=5.0,
                alpha_end=150.0,
                eta0=0.8,
                eval_every=40,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] after smooth‑GD seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3)  Incorporate a previously known construction if present.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous construction improved best to {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4)  Upsample & refine (if time permits).
    # ------------------------------------------------------------------
    while time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        if cur_n >= 64000:
            break
        new_n = cur_n * 2
        if new_n > 64000:
            break
        print(f"[{time.time():.1f}] upsampling {cur_n} → {new_n}")
        up_seq = _upsample(overall_best_seq, new_n)
        up_score = _objective(up_seq)
        if up_score < overall_best_score:
            overall_best_seq = up_seq
            overall_best_score = up_score
            print(f"[{time.time():.1f}] upsample gave new best {overall_best_score:.6f}")

        # short burst of smooth‑GD on the upsampled vector
        sub_deadline = time.time() + (deadline - time.time()) * 0.25
        if sub_deadline - time.time() < 5.0:
            break
        a_opt, sc_opt = _smooth_gradient_descent(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            alpha_start=10.0,
            alpha_end=200.0,
            eta0=0.5,
            eval_every=30,
        )
        if sc_opt < overall_best_score:
            overall_best_seq = a_opt
            overall_best_score = sc_opt
            print(f"[{time.time():.1f}] post‑upsample GD new best {overall_best_score:.6f}")
        else:
            # if no progress, stop further up‑sampling
            break

    # ------------------------------------------------------------------
    # 5)  Final long‑run pairwise annealing polish.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing polish")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
            max_trials=None,
        )
        if sc_opt < overall_best_score:
            overall_best_seq = a_opt
            overall_best_score = sc_opt
            print(f"[{time.time():.1f}] after final annealing new best {overall_best_score:.6f}")

    print(
        f"Final best (objective = 2·n·max(conv)/sum²): {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    # ``evaluate_sequence`` will clamp negatives / huge values; we already stay ⩾0.
    return overall_best_seq.tolist()
