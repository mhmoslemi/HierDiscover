def _multipeak_subgrad_descent(a, deadline, best_seq, best_score):
    t=0
    while time.time() < deadline:
        t+=1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        # indices of conv within epsilon of max
        epsilon = 1e-4  # 0.01%
        S = np.where(conv >= max_val * (1 - epsilon))[0]
        # compute gradient as sum contributions across all k in S
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in S:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end+1)
            idx_j = k - idx_i
            grad[idx_i] += 2.0 * a[idx_j]  # accumulate
        eta = lr0 / np.sqrt(t)
        a = a * np.exp(-eta * grad)
        # normalize
        s = a.sum()
        if s == 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s
        # evaluate and update best
        cur_score = 2.0 * n * max_val / (s*s)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"...")
    return best_seq, best_score

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)²."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
def _distribution_power_offset(n: int, p: float, c: float) -> np.ndarray:
    """
    Family a_i ∝ ((i + c)*(n‑1‑i + c))^{-p/2}.
    Parameters:
        p – power (≈1 for classic arcsine)
        c – offset to avoid singularities (c≥0)
    Returns a normalized probability vector (sum=1).
    """
    i = np.arange(n, dtype=np.float64)
    left = i + c
    right = (n - 1 - i) + c
    vals = (left * right) ** (-p / 2.0)
    # Replace possible infinities (c=0 leads to singular ends)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        # fallback to uniform
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _initial_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """
    Produce a diverse set of seed vectors for length n.
    Includes analytic shapes (grid over p,c), a few Dirichlet draws,
    and a uniform baseline.
    """
    seeds = []

    # uniform baseline
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

    # grid over (p,c)
    p_vals = np.linspace(0.7, 1.3, 13)        # 13 values
    c_vals = np.linspace(0.0, 2.0, 13)        # 13 values (c=0 works for interior points)
    for p in p_vals:
        for c in c_vals:
            vec = _distribution_power_offset(n, p, c)
            seeds.append(vec)

    # a few Dirichlet draws (alpha≈0.5)
    for _ in range(4):
        alpha = 0.3 + 0.7 * rng.random()
        vals = rng.gamma(alpha, size=n)
        if vals.sum() == 0.0:
            vals = np.ones(n, dtype=np.float64)
        vals /= vals.sum()
        seeds.append(vals)

    return seeds


# ----------------------------------------------------------------------
def _multi_peak_subgrad_descent(a: np.ndarray,
                                deadline: float,
                                best_seq: np.ndarray,
                                best_score: float,
                                lr0: float = 0.5,
                                eps_start: float = 1e-3,
                                eps_end: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent using a multi‑peak sub‑gradient.
    The set S of near‑maximal convolution indices is defined by
    conv ≥ max_conv·(1‑ε).  ε is linearly decreased from eps_start to
    eps_end during the run to gradually focus on the true maximiser.
    """
    n = a.shape[0]
    t = 0
    # pre‑compute total number of iterations we may perform (rough)
    max_iters = 10_000_000  # safe upper bound
    while time.time() < deadline and t < max_iters:
        t += 1
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        # linearly shrink ε
        elapsed = (deadline - (deadline - (deadline - time.time())))  # dummy
        # simpler: decay with t
        eps = max(eps_end, eps_start * (0.99 ** (t // 2000)))
        # indices whose value is within (1‑eps) of max
        S = np.where(conv >= max_conv * (1.0 - eps))[0]

        # compute multi‑peak gradient
        grad = np.zeros_like(a)
        i_start_common = 0  # not used
        for k in S:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1)
            idx_j = k - idx_i
            grad[idx_i] += 2.0 * a[idx_j]

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # Project back to the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        cur_score = 2.0 * n * max_conv / (s * s)
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] iter={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_swap_local(a: np.ndarray,
                       deadline: float,
                       best_seq: np.ndarray,
                       best_score: float,
                       max_trials: int = 1_000_000) -> tuple[np.ndarray, float]:
    """
    Greedy random mass‑transfer polishing.
    Proposes moving a random fraction (up to 30 %) of mass from i→j.
    Accepts only improving moves.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0
    while time.time() < deadline and trials < max_trials:
        trials += 1
        # choose a source with non‑negligible mass
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue
        delta = a[i] * rng.random() * 0.30  # up to 30 % of source mass
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur_score = _objective(new_a)
        if cur_score < best_score - 1e-13:
            a = new_a
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}→{j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score

def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) minimizing
    the provided evaluation function.  Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    # 5 seconds safety margin for the caller's wrapper
    deadline = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1.  Generate a pool of seed vectors of various lengths
    lengths = [2600, 3000, 3400, 3800, 4200, 4600]  # modestly spread lengths
    top_seeds: list[tuple[float, np.ndarray]] = []  # (score, vector)

    overall_best_seq = None
    overall_best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] *** scanning length {n} ***")
        seeds = _initial_seeds(n, rng)
        for vec in seeds:
            if time.time() >= deadline:
                break
            # Ensure feasibility (non‑negative, sum=1)
            vec = np.maximum(vec, 0.0)
            if vec.sum() == 0.0:
                vec = np.full(n, 1.0 / n, dtype=np.float64)
            else:
                vec = vec / vec.sum()
            sc = _objective(vec)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = vec.copy()
                print(f"[{time.time():.1f}] !!! new global best len={n} score={sc:.6f}")
            top_seeds.append((sc, vec.copy()))

        # keep only the elite across all lengths (size 12)
        top_seeds.sort(key=lambda x: x[0])
        top_seeds = top_seeds[:12]

    # ------------------------------------------------------------------
    # 2.  Incorporate a previously known solution if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = np.full(prev.shape[0], 1.0 / prev.shape[0], dtype=np.float64)
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best improved: {sc:.6f}")
            top_seeds.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Multi‑peak exponentiated‑gradient descent on elite seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_seeds:
        # allocate roughly equal time to the best few seeds (≤5)
        top_seeds.sort(key=lambda x: x[0])
        seeds_to_use = [seq for _, seq in top_seeds[:5]]
        per_seed = max(1.0, remaining / len(seeds_to_use))
        for seed_seq in seeds_to_use:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            seq_out, score_out = _multi_peak_subgrad_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.5,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after subgrad new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Final greedy polishing
    if overall_best_seq is not None and time.time() < deadline:
        overall_best_seq, overall_best_score = _random_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    # 5.  Safeguard: if everything failed, fall back to a uniform vector
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
