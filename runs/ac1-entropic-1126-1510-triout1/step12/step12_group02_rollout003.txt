    t = 0
    n = len(a)
    best_score = _objective(a)
    best_seq = a.copy()
    while time.time() < deadline:
        t += 1
        # progress schedule
        frac = (t-1) / 5000.0 # after 5000 iterations, maybe
        # gradually increase alpha
        alpha = alpha_start * (alpha_end/alpha_start) ** min(frac, 1.0)
        # compute conv
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        # compute softmax weights
        # to improve numerical stability, subtract max_conv
        w = np.exp(alpha * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / len(w))
        else:
            w = w / w_sum
        # compute gradient: g = 2 * conv(w, a_rev)[n-1:n-1+n]
        a_rev = a[::-1]
        # compute convolution using np.fft (real).

        # Use step size decreasing: eta = initial_eta / sqrt(t)
        eta = initial_eta / math.sqrt(t)
        # exponentiated update
        a = a * np.exp(-eta * grad)
        # project onto simplex:
        a = _project_onto_simplex(a, 1.0)

    return best_seq, best_score

def _fft_convolve(u, v):
    size = 1 << ((len(u) + len(v) - 1) - 1).bit_length()
    U = np.fft.rfft(u, n=size)
    V = np.fft.rfft(v, n=size)
    conv = np.fft.irfft(U * V, n=size)
    return conv[:len(u) + len(v) - 1]

assistantfinal**Solution Overview**

The optimiser searches in the space of probability vectors  
`a = (a₀,…,aₙ₋₁) ,  aᵢ ≥ 0 , Σ aᵢ = 1`.  
For a fixed length `n` the objective used by `evaluate_sequence` is  

```
F(a) = 2·n·max(conv(a,a)) / (Σ a)² = 2·n·max(conv(a,a))      (Σ a = 1)
```

`conv(a,a)` is the linear self‑convolution of `a`.  
The idea is:

1. **Generate a diverse seed pool** – analytic families (uniform, arcsine,
   Beta, edge‑weighted, Hann, Kaiser, Chebyshev,
   generalized‑normal, sparse Dirichlet) for several lengths.
2. **Smooth optimisation** – instead of differentiating the nondifferentiable
   `max`, we use a soft‑max approximation  

   `Mα(a) = Σ w_k · (conv(a,a))_k` with  
   `w_k = exp(α·(c_k–c_max)) / Σ exp(α·(c_j–c_max))`.  

   For `α → ∞` the weight concentrates on the true maximum, but for finite
   `α` the objective becomes smooth, allowing an efficient gradient step.
   The gradient of `Mα` is obtained analytically by a second convolution,
   which is computed with an FFT – `O(n log n)`.
3. **Projected (additive) gradient descent** – a simple diminishing step
   size (`η = η₀/√t`) followed by Euclidean projection onto the simplex.
   The method is run on the best seeds for a time‑controlled number of
   iterations.  The soft‑max temperature `α` is increased gradually, so the
   optimiser first moves quickly in a smooth landscape and later sharpens
   around the true maximum.
4. **Upsampling & polishing** – the current best vector is linearly
   interpolated to a larger length (doubling while keeping the simplex
   property) and refined again.  Finally a cheap pairwise “mass‑transfer”
   simulated‑annealing pass removes any remaining small bumps.
5. **Safety‑net** – if nothing succeeds a uniform distribution of length 4000
   is returned.

All steps respect the global 1000 s budget (a 2 s safety margin).  
The code contains no closures, only top‑level functions, and uses only
`numpy` (optionally `scipy` for a few windows).

"""

evaluate_sequence by:

1. building a rich pool of analytic seeds,
2. performing smooth projected gradient descent on a soft‑max
   approximation of the max‑convolution,
3. up‑sampling the best candidate and refining it again,
4. finishing with a short pairwise‑mass‑transfer annealing.

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolutions (linear, not circular)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = u.shape[0]
    m = v.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()
    U = np.fft.rfft(u, n=size)
    V = np.fft.rfft(v, n=size)
    conv = np.fft.irfft(U * V, n=size)
    return conv[: n + m - 1]


# ----------------------------------------------------------------------
# Exact objective (same as evaluate_sequence but without the safety checks)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a probability vector a."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    if a.sum() < 0.01:
        return float("inf")
    return float(2.0 * n * max_conv / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Analytic families (all return probability vectors, Σ = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_kaiser(n: int, beta: float) -> np.ndarray:
    # Kaiser window – fallback if scipy not present
    try:
        from scipy.signal import kaiser
        w = kaiser(n, beta, sym=True).astype(np.float64)
    except Exception:
        # cheap approximation: cosine‑bell
        i = np.arange(n, dtype=np.float64)
        w = np.i0(beta * np.sqrt(1 - ((2 * i) / (n - 1) - 1) ** 2))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_chebyshev(n: int, atten: float) -> np.ndarray:
    try:
        from scipy.signal import chebwin
        w = chebwin(n, at=atten, sym=True).astype(np.float64)
    except Exception:
        # fall back to Hann
        w = _distribution_hann(n)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(-|x-0.5|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base ⋅ (1‑lam) + uniform ⋅ lam."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Upsampling (linear interpolation while staying on the simplex)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to length new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new


# ----------------------------------------------------------------------
# Smooth projected gradient descent (soft‑max approximation)
# ----------------------------------------------------------------------
def _softmax_gd(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.4,
    eval_every: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the smooth surrogate Mα(a).
    Update is additive (a ← a - η·grad) followed by Euclidean projection
    onto the simplex.  α (soft‑max temperature) is increased slowly.
    """
    n = a.shape[0]
    t = 0
    # pre‑allocate a_rev (used each iteration)
    while time.time() < deadline:
        t += 1
        # ----- current convolution & soft‑max weight -----
        conv = _fft_self_convolve(a)
        c_max = conv.max()
        # temperature schedule (more aggressive early, sharper later)
        if t < 2000:
            alpha = 5.0
        elif t < 15000:
            alpha = 30.0
        elif t < 60000:
            alpha = 80.0
        else:
            alpha = 200.0
        w = np.exp(alpha * (conv - c_max))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w = w / w_sum

        # ----- gradient of the soft‑max surrogate -----
        a_rev = a[::-1]
        grad_full = _fft_convolve(w, a_rev)          # length = (2n‑1)+n‑1 = 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]    # extract n relevant entries

        # ----- additive step with diminishing learning rate -----
        eta = eta0 / math.sqrt(t)
        a = a - eta * grad
        a = _project_onto_simplex(a, 1.0)

        # ----- occasional exact evaluation -----
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] soft‑GD it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise mass‑transfer annealing (short polishing phase)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Randomly transfer a fraction of mass between two bins, accept if better."""
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        # pick source with non‑negligible mass
        src = rng.choice(np.where(a > 1e-12)[0])
        # pick distinct destination
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1
        # transfer up to 40 % of the source mass
        delta = a[src] * rng.random() * 0.4
        if delta < 1e-18:
            continue
        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta
        cur = _objective(new_a)
        # simple linear cooling schedule
        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.01
        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True
        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal {src}->{dst} δ={delta:.2e} new best {best_score:.6f}"
                )
    return best_seq, best_score

            _distribution_hann(n),
        ):
            sc = _objective(vec)
            cand_pool.append((sc, vec))
        
        # beta & edge‑weighted families with many parameters
        p_vals = np.concatenate(
            [
                np.logspace(-2.0, -0.5, 8),
                np.logspace(-0.5, 0.0, 8),
                np.logspace(0.0, 0.7, 6),
            ]
        )
        lam_vals = np.linspace(0.0, 1.0, 5)
        for p in p_vals:
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            for base in (beta, edge):
                for lam in lam_vals:
                    mixed = _mix_with_uniform(base, lam)
                    sc = _objective(mixed)
                    cand_pool.append((sc, mixed))

        # generalized normal
        for q in np.linspace(0.5, 3.0, 6):
            gn = _distribution_generalized_normal(n, q)
            sc = _objective(gn)
            cand_pool.append((sc, gn))

        # Kaiser & Chebyshev (if scipy is present, they fall back to simple windows)
        for beta in np.linspace(0.0, 12.0, 5):
            kaiser = _distribution_kaiser(n, beta)
            sc = _objective(kaiser)
            cand_pool.append((sc, kaiser))
        for att in np.linspace(30, 150, 5):
            cheb = _distribution_chebyshev(n, att)
            sc = _objective(cheb)
            cand_pool.append((sc, cheb))

        # random Dirichlet seeds
        for _ in range(6):
            alpha = rng.uniform(0.01, 2.0)
            dirich = rng.dirichlet(np.full(n, alpha))
            sc = _objective(dirich)
            cand_pool.append((sc, dirich))

        # sparse random seeds
        for _ in range(6):
            sp = _distribution_random_sparse(n, rng, density=0.01)
            sc = _objective(sp)
            cand_pool.append((sc, sp))

        # keep only the best `top_k` candidates seen so far
        cand_pool.sort(key=lambda x: x[0])
        cand_pool = cand_pool[: top_k]

        # quick update of the global best
        cur_best_score, cur_best_seq = cand_pool[0]
        print(
            f"[{time.time():.1f}] current best for n={n}: score={cur_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 2.  Include the previous construction if it exists
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            cand_pool.append((sc, prev))
            cand_pool.sort(key=lambda x: x[0])
            cand_pool = cand_pool[: top_k]
            print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if not cand_pool:
        n = 4000
        fallback = _distribution_uniform(n)
        cand_pool.append((_objective(fallback), fallback))
        print(f"[{time.time():.1f}] fallback uniform created (score={cand_pool[0][0]:.6f})")

    # ------------------------------------------------------------------
    # 4.  Smooth projected gradient descent on the best few seeds
    # ------------------------------------------------------------------
    # allocate roughly half of the remaining time to this phase
    remaining = deadline - time.time()
    if remaining > 20.0:
        per_seed_time = remaining * 0.5 / max(1, len(cand_pool))
        print(f"[{time.time():.1f}] smooth GD on {len(cand_pool)} seeds")
        best_score = float("inf")
        best_seq = None
        for idx, (score, seq) in enumerate(cand_pool):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed_time
            seq_opt, sc_opt = _softmax_gd(
                seq.copy(),
                sub_dead,
                best_seq if best_seq is not None else seq,
                best_score,
                eta0=0.5,
                eval_every=100,
            )
            if sc_opt < best_score - 1e-12:
                best_score = sc_opt
                best_seq = seq_opt.copy()
                print(
                    f"[{time.time():.1f}] after GD seed {idx} new global best {best_score:.6f}"
                )
        # ensure we have something
        if best_seq is None:
            best_seq = cand_pool[0][1].copy()
            best_score = cand_pool[0][0]

    else:
        # not enough time – just pick the current best
        best_score, best_seq = cand_pool[0]

    # ------------------------------------------------------------------
    # 5.  Upsample to a larger length and refine again (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur_n = best_seq.shape[0]
        target_n = cur_n * 2
        if target_n <= 64000:
            print(f"[{time.time():.1f}] upsampling {cur_n} → {target_n}")
            up_seq = _upsample(best_seq, target_n)
            up_score = _objective(up_seq)
            if up_score < best_score:
                best_seq = up_seq
                best_score = up_score
                print(f"[{time.time():.1f}] upsampled new best {best_score:.6f}")

            # a short refinement on the up‑sampled vector
            sub_dead = time.time() + (deadline - time.time()) * 0.3
            best_seq, best_score = _softmax_gd(
                best_seq,
                sub_dead,
                best_seq,
                best_score,
                eta0=0.4,
                eval_every=80,
            )
            print(
                f"[{time.time():.1f}] after upsample GD best = {best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 6.  Final short annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing polish")
        best_seq, best_score = _pairwise_annealing(
            best_seq,
            deadline,
            best_seq,
            best_score,
            rng,
        )
        print(f"[{time.time():.1f}] final best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Return the best vector found
    # ------------------------------------------------------------------
    print(
        f"Finished – best objective {best_score:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
