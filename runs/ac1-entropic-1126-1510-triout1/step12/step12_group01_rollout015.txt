Propose a step‑function that (empirically) improves the upper bound
for the `evaluate_sequence` objective.

Strategy
--------
1. Sample a rich set of candidates (analytic families, Dirichlet,
   sparse vectors) for several lengths (up to 50 k).  Keep the best few.
2. Refine each promising seed with an exponentiated‑gradient descent
   that optimises a smooth approximation of the max‑convolution.
   The temperature τ is increased slowly so that the update finally
   targets the true max.
3. A short Adam optimiser on the soft‑max of the convolution (logit
   parametrisation) provides a slightly different search direction.
4. Fine‑grained deterministic pairwise mass transfers and a cheap
   simulated‑annealing walk finish the search.
The routine respects the supplied time budget (default 1000 s) and
returns a plain Python list – `evaluate_sequence` will take care of
normalisation and clipping.

import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]

# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric beta (Beta(a,b)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight proportional to sin(π (i+½)/n)^p, vanishing at the ends."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

def _smooth_exponential_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    tau0: float = 5.0,
    tau_mult: float = 1.002,
    eval_interval: int = 100,
    active_k: int = 20,
    patience: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the exact objective, using a
    soft‑max weighting of the largest convolution entries.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    tau = tau0

    while time.time() < deadline and t < 500_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()

        # ----- active set of largest entries -----
        if active_k >= len(conv):
            top_idx = np.arange(len(conv))
        else:
            top_idx = np.argpartition(conv, -active_k)[-active_k:]

        # softmax weights (temperature tau)
        w_raw = np.exp(tau * (conv[top_idx] - max_c))
        w = w_raw / w_raw.sum()

        # ----- weighted gradient -----
        grad = np.zeros_like(a)
        ind_grid = np.arange(n, dtype=np.int32)
        for weight, idx in zip(w, top_idx):
            ind = idx - ind_grid  # = idx - j
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * weight * a[ind[mask]]

        # ----- exponentiated‑gradient update -----
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # increase temperature
        tau *= tau_mult

        # ----- occasional exact evaluation -----
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] smooth descent it={t} new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            jitter = rng.exponential(scale=0.004, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

# Adam optimiser on logits (soft‑max) ---------------------------------
def _adam_logit_opt(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    tau0: float = 8.0,
    tau_mult: float = 1.0015,
    eval_interval: int = 150,
) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑it parametrisation a = softmax(w) of a probability
    vector.  The loss is a smooth approximation of max(conv) obtained
    through a soft‑max with temperature τ.
    """
    n = a0.shape[0]
    w = np.log(a0 + 1e-12)  # initialise logits
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    tau = tau0

    while time.time() < deadline and t < 300_000:
        t += 1

        # softmax to obtain probability vector
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a = a / a.sum()

        conv = _fft_self_convolve(a)
        max_c = conv.max()

        # soft‑max over convolution entries
        w_raw = np.exp(tau * (conv - max_c))
        probs = w_raw / w_raw.sum()          # shape (2n‑1)

        # weighted gradient w.r.t a
        grad_a = np.zeros_like(a)
        ind_grid = np.arange(n, dtype=np.int32)
        for weight, idx in zip(probs, np.arange(len(conv))):
            # shift contribution for this offset
            ind = idx - ind_grid
            mask = (ind >= 0) & (ind < n)
            grad_a[mask] += 2.0 * weight * a[ind[mask]]

        # gradient w.r.t logits
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        tau *= tau_mult

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam iter={t} new best {best_score:.6f}")

    return best_seq, best_score

def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Transfer a small amount of mass between two positions if it improves the objective."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)

        if delta < 1e-14:

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}->{j} Δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score

# Simulated annealing style local search ---------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Smooth exponential descent -------------------------------------------
def _smooth_exponential_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    tau0: float = 5.0,
    tau_mult: float = 1.002,
    eval_interval: int = 100,
    active_k: int = 20,
    patience: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a smooth surrogate of the max‑convolution.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    tau = tau0

    while time.time() < deadline and t < 500_000:
        t += 1
        conv = _fft_self_convolve(a)
        max_c = conv.max()

        # active set – the largest entries of the convolution
        if active_k >= len(conv):
            top_idx = np.arange(len(conv))
        else:
            top_idx = np.argpartition(conv, -active_k)[-active_k:]

        # soft‑max weights (temperature τ)
        w_raw = np.exp(tau * (conv[top_idx] - max_c))
        w = w_raw / w_raw.sum()

        # weighted gradient of the surrogate loss
        grad = np.zeros_like(a)
        ind_grid = np.arange(n, dtype=np.int32)
        for weight, idx in zip(w, top_idx):
            ind = idx - ind_grid
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * weight * a[ind[mask]]

        # exponentiated‑gradient update
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex
        s = a.sum()
        a = np.full(n, 1.0 / n, dtype=np.float64) if s == 0.0 else a / s

        # temperature schedule
        tau *= tau_mult

        # exact evaluation every few iterations
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] smooth it={t} new best {best_score:.6f}")
            else:
                no_improve += 1

        # jitter on stagnation
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.004, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

# Adam optimiser on logits ---------------------------------------------
def _adam_logit_opt(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    tau0: float = 8.0,
    tau_mult: float = 1.0015,
    eval_interval: int = 150,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑it parameters w with asoft‑max surrogate
    of the max‑convolution.
    """
    n = a0.shape[0]
    w = np.log(a0 + 1e-12)          # initialise logits
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    tau = tau0

    while time.time() < deadline and t < 300_000:
        t += 1

        # soft‑max to obtain a probability vector
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        conv = _fft_self_convolve(a)
        max_c = conv.max()

        # soft‑max over convolution entries
        w_raw = np.exp(tau * (conv - max_c))
        probs = w_raw / w_raw.sum()          # length 2n‑1

        # weighted gradient w.r.t a
        grad_a = np.zeros_like(a)
        ind_grid = np.arange(n, dtype=np.int32)
        for weight, idx in zip(probs, range(len(conv))):
            ind = idx - ind_grid
            mask = (ind >= 0) & (ind < n)
            grad_a[mask] += 2.0 * weight * a[ind[mask]]

        # gradient w.r.t logits (chain rule through softmax)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        tau *= tau_mult

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise swap (deterministic local refinement) ------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Move a small amount of mass between two positions if it improves the objective."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style local search ---------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget
    (default 1000 s) and returns a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 3.0           # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Build a diverse pool of seeds (analytic families + randomness)
    lengths = [2000, 4000, 8000, 16000, 32000, 50000]
    p_grid = np.concatenate(
        [
            np.logspace(-2, -0.3, 7, base=10),   # 0.01 … 0.5
            np.logspace(-0.3, 0.3, 7, base=10), # 0.5 … 2
            np.logspace(0.3, 0.6, 5, base=10),  # 2 … 4
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 9)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # parametrised families + mixing with uniform
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            sine = _distribution_sine_weighted(n, p)

            for base in (beta, edge, sine):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # asymmetric beta seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            a_param = 10 ** rng.uniform(-2, 0)   # 0.01 … 1
            b_param = 10 ** rng.uniform(-2, 0)
            cand = _distribution_beta_asym(n, a_param, b_param)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best asym‑beta len={n} a={a_param:.3g} b={b_param:.3g} score={sc:.6f}"
                )

        # Dirichlet random seeds
        for _ in range(8):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] new global best Dirichlet len={n} score={sc:.6f}")

        # Sparse random seeds
        for _ in range(8):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}")

        # keep the current best 40 seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:40]

    # ------------------------------------------------------------------
    # 2. Include the previous construction if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Guard against the unlikely case of no candidates
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_arcsine(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback arcsine score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Refine the most promising seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:8]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] starting refinement on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            # (a) smooth exponential descent
            sub_deadline = time.time() + per_seed * 0.55
            a_opt, sc_opt = _smooth_exponential_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.55,
                tau0=4.0,
                tau_mult=1.0018,
                eval_interval=120,
                active_k=22,
                patience=3500,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after smooth seed {idx} new best {overall_best_score:.6f}")

            # (b) short Adam optimisation on logits
            if time.time() < deadline:
                sub_deadline2 = time.time() + per_seed * 0.30
                a_opt2, sc_opt2 = _adam_logit_opt(
                    a_opt.copy(),
                    sub_deadline2,
                    best_seq=overall_best_seq,
                    best_score=overall_best_score,
                    lr=0.04,
                    tau0=8.0,
                    tau_mult=1.0012,
                    eval_interval=80,
                )
                if sc_opt2 < overall_best_score:
                    overall_best_score, overall_best_seq = sc_opt2, a_opt2.copy()
                    print(f"[{time.time():.1f}] after Adam seed {idx} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=350_000,
        )

    # ------------------------------------------------------------------
    # 6. Final stochastic local search (simulated annealing)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final random‑local search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.009,
            decay=0.9996,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
