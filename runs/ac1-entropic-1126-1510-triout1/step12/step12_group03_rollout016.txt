def _subgradient_exact(a0, time_limit, best_seq, best_score, lr0=0.5, eval_every=200, max_iter=200000):
    a = a0.copy()
    t = 0
    while time.time() < time_limit and t < max_iter:
        t += 1
        conv = _fft_self_convolve(a)
        k = int(np.argmax(conv))
        # compute gradient
        grad = np.zeros_like(a)
        # vectorized: create reversed a?
        # For each i, j = k-i
        # Use np.arange to compute j = k - np.arange(n)
        n=a.shape[0]
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        # grad[mask] = 2 * a[j[mask]]
        # For diagonal elements where i==j (i==k/2)
        # But condition i==j => 2*i == k
        diag_mask = mask & (idx == j)
        # For diag, later adjust
        grad[mask] = 2.0 * a[j[mask]]
        # adjust diag
        diag_idx = idx[diag_mask]
        if diag_idx.size > 0:
            # subtract a_j (since we added 2*a_j but diag should be a_j)
            grad[diag_idx] -= a[diag_idx]
        # step size
        eta = lr0 / math.sqrt(t)

        # project onto simplex

Propose a step‑function (non‑negative vector) that minimizes
`evaluate_sequence`.  The algorithm consists of three phases:

1.  A broad scan of many analytic families (uniform, arcsine,
    beta, edge‑weighted, exponential‑ends, power‑decay, …) on a

3.  A final local search that (a) solves a linear program
    `max Σ g_j  s.t.  f * g ≤ M` for the current best `f`
    (this can increase the L₁‑norm while keeping the convolution bound)
    and (b) performs greedy pairwise mass swaps.

All vectors are kept on the simplex (non‑negative, Σ = 1), because
the objective `2·n·max(conv)/Σ²` is invariant under positive scaling.
The routine respects the 1000 s wall‑clock budget and prints intermediate
progress.
"""

import math

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector already normalized so
    that Σa = 1.  a must be non‑negative.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on grid {0,…,n‑1} (p>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # w ∝ x^{p-1} (1‑x)^{p‑1}
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """
    Edge‑weighted family: w_i ∝ ((i+½)*(n‑i‑½))^{‑p}.
    Heavy at both ends for p>0.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential_ends(n: int, alpha: float) -> np.ndarray:
    """Symmetric exponential decay from both ends."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_power_decay(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i+1)^{-p} + (n-i)^{-p}  (p>0)."""
    i = np.arange(n, dtype=np.float64) + 1.0
    w = (i ** (-p)) + ((n - np.arange(n, dtype=np.float64)) ** (-p))
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with prescribed density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Exact mirror‑descent on the max‑convolution -------------------------
def _exact_subgradient_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.6,
    eval_every: int = 150,
    max_iter: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent with the *exact* sub‑gradient of the
    max‑convolution (the index attaining the maximum is used).
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit and t < max_iter:
        t += 1
        conv = _fft_self_convolve(a)
        k = int(np.argmax(conv))

        # exact sub‑gradient of the maximal entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]
        # diagonal correction (i == j)
        diag = (mask) & (idx == j)
        if np.any(diag):
            grad[diag] = a[diag]    # replace 2*a_i by a_i

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)     # keep positivity
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s                         # simplex projection

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] exact subgrad it={t} new global best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise swap -------------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from one index to another.
    Keep the move only if it strictly improves the exact objective.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Optional LP refinement (increase L1 while keeping the convolution bound)
def _lp_refine(f: np.ndarray, M: float, time_limit: float) -> np.ndarray | None:
    """
    Solve  maximize Σ g_j  subject to  f * g ≤ M ,  g ≥ 0.
    Returns the optimal g (unnormalised) or None if the LP fails.
    """
    try:
        from scipy.optimize import linprog
        import scipy.sparse as sp
    except Exception:            # SciPy not available – skip LP
        return None

    n = f.shape[0]
    N = 2 * n - 1

    # Build sparse convolution matrix A such that A @ g = f * g
    rows = []
    cols = []
    data = []
    for i in range(n):
        # contribution of f[i] to rows i … i+n-1
        rows.extend(range(i, i + n))
        cols.extend([i] * n)
        data.extend([f[i]] * n)
    A = sp.csr_matrix((data, (rows, cols)), shape=(N, n))

    b = np.full(N, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise Σ g  → minimise –Σ g
    opts = {"disp": False}
    # some SciPy versions accept a time limit via 'time_limit' in options, others use 'max_iter'
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                      method='highs', options=opts)
    except Exception:
        # fall back to the older interior‑point method
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                      method='interior-point', options=opts)

    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Main entry point: search for a step‑function that minimizes
    `evaluate_sequence`.  Returns a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []  # (score, seq)

    # --------------------------------------------------------------
    # 1.  Scan a diverse set of analytic families
    # --------------------------------------------------------------
    lengths = [1000, 1500, 2000, 3000, 4000, 6000, 8000, 12000]
    p_grid = np.concatenate(
        [
            np.logspace(-2, -0.3, 12),      # 0.01 … 0.5
            np.linspace(0.5, 2.0, 10),      # 0.5 … 2
            np.linspace(2.0, 10.0, 9),      # 2 … 10
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 11)      # step 0.1
    alpha_grid = np.logspace(-4, -0.5, 8)     # for exponential ends
    power_grid = np.logspace(-2, 1, 9)       # for power‑decay

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_power_decay,   # as a simple power law (p=1)
        ):
            cand = base_fn(n)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best (det) len={n} score={sc:.6f}")

        # parametric families
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta_seq = _distribution_beta_sym(n, p)
            edge_seq = _distribution_edge_weighted(n, p)

            for base in (beta_seq, edge_seq):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best p={p:.3g} lam={lam:.2f} len={n} score={sc:.6f}"
                        )

        # exponential‑ends family
        for alpha in alpha_grid:
            if time.time() >= deadline:
                break
            base = _distribution_exponential_ends(n, alpha)
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(
                        f"[{time.time():.1f}] new global best expα={alpha:.2e} lam={lam:.2f} len={n} score={sc:.6f}"
                    )

        # power‑decay family (independent of p_grid)
        for p in power_grid:
            if time.time() >= deadline:
                break
            base = _distribution_power_decay(n, p)
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(
                        f"[{time.time():.1f}] new global best powp={p:.3g} lam={lam:.2f} len={n} score={sc:.6f}"
                    )

        # random Dirichlet seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] new global best Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}")

        # retain only the top ~30 candidates (memory / time bound)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # --------------------------------------------------------------
    # 2.  Incorporate a previously discovered construction if available
    # --------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3.  Fallback in case nothing survived the scan
    # --------------------------------------------------------------
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(
            f"[{time.time():.1f}] fallback uniform len={n} score={overall_best_score:.6f}"
        )

    # --------------------------------------------------------------
    # 4.  Exact mirror‑descent refinement on the best seeds
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0 and top_candidates:
        # keep the 5 best seeds for deeper work
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] start exact subgradient descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sg_deadline = time.time() + per_seed
            a_opt, sc_opt = _exact_subgradient_descent(
                seed_seq,
                sg_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.8,
                eval_every=120,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] after subgrad seed {idx} new global best {overall_best_score:.6f}"
                )

    # --------------------------------------------------------------
    # 5.  Optional LP refinement (only if the length is modest)
    # --------------------------------------------------------------
    if time.time() < deadline and overall_best_seq is not None and len(overall_best_seq) <= 1500:
        n = overall_best_seq.shape[0]
        M = _fft_self_convolve(overall_best_seq).max()
        g = _lp_refine(overall_best_seq, M, deadline)
        if g is not None:
            g = g / g.sum()
            g_score = _objective(g)
            if g_score < overall_best_score:
                overall_best_score = g_score
                overall_best_seq = g.copy()
                print(f"[{time.time():.1f}] LP refinement improved score to {g_score:.6f}")

    # --------------------------------------------------------------
    # 6.  Final greedy pairwise polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy swap polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 7.  Report and return
    # --------------------------------------------------------------
    print(
        f"Final best score (2·n·max(conv)): {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list – the external `evaluate_sequence` will
    # perform its own clipping and normalisation.
    return overall_best_seq.tolist()
