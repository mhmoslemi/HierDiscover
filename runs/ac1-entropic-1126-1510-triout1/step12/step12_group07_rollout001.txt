"""
Hybrid LP‑based refinement for the max‑convolution step‑function problem.

The algorithm works in three phases:

1. **Seeding** – a handful of analytic families (uniform, arcsine,
   beta‑sym, Gaussian, Dirichlet) are generated for a few candidate
   lengths.  The best few seeds (by the exact objective) are kept.

2. **LP‑driven mixing** – for each seed we iteratively solve a linear
   program
        maximize   1ᵀg
        subject to a * g ≤ M        (a is the current vector,
                                    * denotes ordinary convolution)
                g ≥ 0
   where M is the current maximum self‑convolution of *a*.  The solution
   *g* carries as much mass as possible without exceeding the bound *M*.
   A convex combination `a ← (1‑t) a + t·ĝ` (with ĝ rescaled to have the
   same ℓ₁‑norm as a) strictly reduces the maximum convolution for a
   suitable step size *t*.  Repeating this step drives the objective
   down.

3. **Greedy polishing** – a short pairwise‑mass‑transfer walk (the same
   routine used in the previous baseline) cleans up any remaining tiny
   improvements.

All vectors are kept on the probability simplex (non‑negative,
sum = 1), which does not affect the final value of
`evaluate_sequence` because the objective is homogeneous of degree 0.
The routine respects a user supplied time budget (default 1000 s) and
returns the best sequence found as a plain list of floats.
"""

import time
import math
from typing import List, Optional, Tuple

import numpy as np
import scipy.sparse
import scipy.optimize


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)² (assumes sum>0)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Analytic families (probability vectors) --------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _build_conv_matrix(a: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Build the (2n‑1)×n matrix A such that A @ g == a * g (ordinary convolution).
    A has a simple diagonal structure: for each offset d (0 ≤ d < n)
    the d‑th diagonal of A is constant a[d].
    """
    n = a.shape[0]
    diagonals = []
    offsets = []
    for d in range(n):
        length = n - d
        diagonals.append(np.full(length, a[d], dtype=np.float64))
        offsets.append(d)
    A = scipy.sparse.diags(diagonals, offsets, shape=(2 * n - 1, n), format='csr')
    return A


# ----------------------------------------------------------------------
def _lp_step(a: np.ndarray, M: float, max_iter: int,
             deadline: float) -> Tuple[np.ndarray, float]:
    """
    Perform a small number of LP‑based refinement steps starting from a.
    Returns the improved vector and its new maximum convolution.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_M = M

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # Linear constraints a * g ≤ cur_M
        A = _build_conv_matrix(cur_a)
        b = np.full(2 * n - 1, cur_M, dtype=np.float64)

        # Maximise sum(g)  <=> minimise -1ᵀg
        c = -np.ones(n, dtype=np.float64)

        res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=[(0, None)] * n,
            method='highs',
            options={'presolve': True}
        )

        if not res.success:
            # LP failed – abort further refinement
            break

        g = res.x
        s_g = g.sum()
        if s_g < 1e-12:
            break

        # Rescale g to have the same ℓ₁‑norm as cur_a (which is 1)
        g = g / s_g

        # Try a few mixing coefficients and keep the best improvement
        best_local_a = cur_a
        best_local_M = cur_M
        for t in (0.05, 0.1, 0.2, 0.3, 0.5):
            a_new = (1.0 - t) * cur_a + t * g
            M_new = _fft_self_convolve(a_new).max()
            if M_new < best_local_M - 1e-15:
                best_local_M = M_new
                best_local_a = a_new

        # If no improvement was found, stop the LP loop
        if best_local_M >= cur_M - 1e-15:
            break

        cur_a = best_local_a
        cur_M = best_local_M
        # Occasionally log progress
        if it % 5 == 0:
            print(f"[{time.time():.1f}] LP iter={it} M={cur_M:.6e}")

    return cur_a, cur_M


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Greedy mass‑transfer walk, accepting only improving moves."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score
