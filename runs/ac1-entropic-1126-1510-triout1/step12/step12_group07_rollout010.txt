    cur = _objective(a)
    cur_score = cur
    T = (best_score - cur_score) * 0.01 if best_score != float('inf') else 0.01
    if T <= 0:
        T = 0.01
    min_T = 1e-6
    decay = 0.9995
    n = a.shape[0]
    eps = 1e-12
    i_iter = 0
    while time.time() < deadline and i_iter < max_iter:
        i_iter += 1
        
        # Choose source index on probability mass > eps
        src_candidates = np.where(a > eps)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        j = rng.integers(n-1)

            j += 1

        max_delta = a[i] * 0.5
        delta = rng.random() * max_delta

        new_a = a.copy()

"""
Hybrid multi‑start optimizer for the max‑convolution step‑function problem.

The algorithm proceeds in five phases:

1. **Exploratory seeding** – a wide variety of analytic families
   (uniform, arcsine, sine, triangular, Gaussian, beta, power‑law,
   exponential, two‑block, Dirichlet, sparse, random mixtures) are
   instantiated for several lengths.  The exact objective is evaluated
   and the best few candidates are retained.

2. **Adam on a smooth‑max surrogate** – each retained seed is optimised
   on the soft‑max surrogate
        Fα(a)= (1/α)·log∑ exp(α·(a*a)[k])
   via Adam on a softmax (log‑it) parametrisation.  The temperature α is
   increased geometrically up to a large value (≈2000) so that the
   surrogate converges to the true max‑convolution.  The best vector
   found in all runs is kept.

3. **Multimax sub‑gradient polishing** – a classic sub‑gradient method
   (exponentiated‑gradient on the simplex) is applied for a few tens of
   thousands of iterations to refine the solution using the exact
   max‑convolution gradient.

4. **Simulated‑annealing local refinement** – random mass‑transfer
   moves are attempted; they are accepted if they improve the exact
   objective or with a Metropolis probability that slowly cools.  This
   helps to escape shallow local minima.

5. **Greedy pairwise‑swap walk** – a cheap final sweep that moves a
   tiny amount of mass between two coordinates whenever it lowers the
   exact objective.

The routine respects a user‑supplied time budget (default 1000 s) and
returns the best sequence (plain ``list[float]``) discovered.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·(a*a)[k]  w.r.t. a.
    Implements 2·a⋆w where w has length 2·n‑1.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (assumes sum(a) > 0)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    """Positive half‑sine shape, symmetric, zero‑free."""
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    """Peak in the centre, linear rise/fall."""
    i = np.arange(n, dtype=np.float64)
    vals = np.minimum(i + 1, n - i)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    vals /= vals.sum()
    return vals


def _distribution_raised_cosine(n: int) -> np.ndarray:
    """Full cosine bump, zero at the ends, positive inside."""
    i = np.arange(n, dtype=np.float64)
    vals = 0.5 * (1 - np.cos(2 * np.pi * (i + 0.5) / n))
    vals /= vals.sum()
    return vals


def _distribution_monotone_decreasing(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sort a random vector descending – a simple decreasing shape."""
    vals = rng.random(n)
    vals = np.sort(vals)[::-1]
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_candidates_extended(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    """Collect a broad variety of seeds for a given length."""
    cand: List[np.ndarray] = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_sine(n))
    cand.append(_distribution_triangular(n))
    cand.append(_distribution_gaussian(n))
    cand.append(_distribution_raised_cosine(n))
    cand.append(_distribution_monotone_decreasing(n, rng))

    # symmetric Beta family
    for p in (0.05, 0.1, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0, 1.5, 2.0):
        cand.append(_distribution_beta_sym(n, p))

    # random asymmetric Beta draws
    for _ in range(4):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # power‑law families
    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))

    # exponential families
    for lam in (0.001, 0.003, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential_increasing(n, lam))

    # two‑block spikes
    for block_frac in (0.05, 0.10, 0.15):
        cand.append(_distribution_two_blocks(n, block_frac))

    # random Dirichlet draws
    for _ in range(4):
        alpha = 0.1 + 0.9 * rng.random()
        cand.append(_distribution_dirichlet(n, rng, alpha=alpha))

    # random sparse patterns
    cand.append(_distribution_random_sparse(n, rng, density=0.01))

    # a few completely random probability vectors
    for _ in range(3):
        v = rng.random(n)
        v = np.maximum(v, 0.0)
        if v.sum() == 0.0:
            v[0] = 1.0
        v /= v.sum()
        cand.append(v)

    # random mixtures of two families (helps diversify)
    for _ in range(2):
        a = rng.choice(cand, size=2, replace=False)
        weight = rng.random()
        mix = weight * a[0] + (1 - weight) * a[1]
        mix = np.maximum(mix, 0.0)
        if mix.sum() == 0.0:
            mix = _distribution_uniform(n)
        else:
            mix /= mix.sum()
        cand.append(mix)

    return cand


# ----------------------------------------------------------------------
def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    shift = np.max(z)
    exp_z = np.exp(z - shift)
    s = exp_z.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / len(z))
    return exp_z / s


# ----------------------------------------------------------------------
def _adam_optimize(z0: np.ndarray,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   alpha0: float = 0.1,
                   alpha_max: float = 2000.0,
                   growth_factor: float = 1.04,
                   lr: float = 0.15,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   max_iter: int = 500000,
                   eval_interval: int = 500) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    The temperature α grows geometrically until alpha_max.
    """
    n = z0.shape[0]
    z = z0.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    t = 0
    alpha = alpha0

    # maintain the current best inside this routine
    cur_best_seq = best_seq.copy()
    cur_best_score = best_score

    while t < max_iter and time.time() < deadline:
        t += 1

        # probability vector on the simplex
        a = _softmax(z)                     # a_i ≥ 0, Σ a_i = 1

        # smooth‑max weights
        conv = _fft_self_convolve(a)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient of the surrogate wrt a
        grad_a = _gradient_from_weights(a, w)

        # chain rule to obtain gradient wrt z (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        z = z - step

        # temperature schedule
        if t % 500 == 0:
            alpha = min(alpha * growth_factor, alpha_max)

        # periodic exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < cur_best_score - 1e-13:
                cur_best_score = cur
                cur_best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {cur_best_score:.6f}")

    return cur_best_seq, cur_best_score


# ----------------------------------------------------------------------
def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.8,
                                 max_iter: int = 40000,
                                 eval_interval: int = 300) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (multimax) descent on the exact max‑convolution.
    """
    n = a.shape[0]
    t = 0
    cur_best_seq = best_seq.copy()
    cur_best_score = best_score
    cur_score = _objective(a)

    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)

        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur_score = _objective(a)
            if cur_score < cur_best_score - 1e-13:
                cur_best_score = cur_score
                cur_best_seq = a.copy()
                print(f"[{time.time():.1f}] multimax it={t} new best {cur_best_score:.6f}")

    return cur_best_seq, cur_best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Greedy mass‑transfer walk; only accept objective‑improving swaps."""
    n = a.shape[0]
    rng = np.random.default_rng()
    cur_best_seq = best_seq.copy()
    cur_best_score = best_score
    cur_score = _objective(a)

    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move up to 30 % of source mass
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        new_score = _objective(new_a)
        if new_score < cur_score - 1e-13:
            a = new_a
            cur_score = new_score
            if new_score < cur_best_score - 1e-13:
                cur_best_score = new_score
                cur_best_seq = a.copy()
                print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {cur_best_score:.6f}")

    return cur_best_seq, cur_best_score


# ----------------------------------------------------------------------
def _annealing_optimize(a: np.ndarray,
                        deadline: float,
                        best_seq: np.ndarray,
                        best_score: float,
                        rng: np.random.Generator,
                        max_iter: int = 20000) -> Tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing walk on the exact objective.
    Random mass transfers are accepted if they improve the score,
    otherwise with probability exp(-Δ/T).  Temperature decays geometrically.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = _objective(cur_a)

    cur_best_seq = best_seq.copy()
    cur_best_score = best_score

    # initialise temperature – a small fraction of current best value
    T = max(1e-4, 0.02 * cur_score)
    Tmin = 1e-8
    decay = 0.9995

    eps = 1e-12
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        # pick a source with non‑negligible mass
        src_idx = np.where(cur_a > eps)[0]
        if src_idx.size == 0:
            break
        i = rng.choice(src_idx)

        # pick a distinct target
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # propose moving a random fraction (up to 40 %) of the source mass
        max_delta = cur_a[i] * 0.40
        delta = rng.random() * max_delta
        if delta < eps:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        new_score = _objective(new_a)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_a = new_a
            cur_score = new_score
            if new_score < cur_best_score - 1e-13:
                cur_best_score = new_score
                cur_best_seq = cur_a.copy()
                print(f"[{time.time():.1f}] anneal iter={it} new best {cur_best_score:.6f}")

        T = max(Tmin, T * decay)

    return cur_best_seq, cur_best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function minimizing ``evaluate_sequence``.
    The routine respects a time budget (default 1000 s) and returns the
    best sequence found as a plain ``list[float]``.
    """
    rng = np.random.default_rng(seed)
    # safety margin of a couple of seconds for the caller
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # (1) choose a modest set of lengths – longer lengths give more freedom
    #     but also increase per‑iteration cost.
    lengths = [1800, 2600, 3400, 4800, 7200]

    overall_best_seq: Optional[np.ndarray] = None
    overall_best_score = float("inf")
    candidates: List[Tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # (2) Analytic families + random draws for each length
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        for cand in _initial_candidates_extended(n, rng):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                continue
            cand = cand / s
            sc = _objective(cand)
            candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] **** new global best len={n} score={sc:.6f}")

        # keep only the most promising seeds across all lengths
        candidates.sort(key=lambda x: x[0])
        # we keep a slightly larger pool for later phases
        candidates = candidates[:30]

    # ------------------------------------------------------------------
    # (3) Incorporate a previous solution if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            sp = prev.sum()
            if sp == 0.0:
                prev = _distribution_uniform(prev.shape[0])
                sp = prev.sum()
            prev = prev / sp
            sc = _objective(prev)
            print(f"[{time.time():.1f}] previous solution score={sc:.6f}")
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous solution improves global best")
            candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # (4) Fallback if nothing was generated
    if overall_best_seq is None:
        n = lengths[0]
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # (5) Adam smooth‑max descent on the top seeds
    remaining = deadline - time.time()
    if remaining > 30.0:
        top_k = min(10, len(candidates))
        top_seeds = [seq for _, seq in sorted(candidates, key=lambda x: x[0])[:top_k]]
        for idx, seed_seq in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            # allocate a time slice – at least 30 s per seed, adjusted to remaining time
            slice_time = min(80.0, (deadline - time.time()) / max(1, top_k - idx))
            sub_deadline = time.time() + slice_time

            # initialise Adam on the log‑softmax parametrisation
            z0 = np.log(np.maximum(seed_seq, 1e-300))

            seq_out, score_out = _adam_optimize(
                z0,
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                alpha0=0.2,
                alpha_max=2000.0,
                growth_factor=1.04,
                lr=0.15,
                max_iter=500000,
                eval_interval=400,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after Adam seed {idx} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # (6) Multimax sub‑gradient polishing
    remaining = deadline - time.time()
    if remaining > 20.0:
        sub_deadline = time.time() + min(120.0, remaining * 0.55)
        overall_best_seq, overall_best_score = _subgradient_multimax_descent(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.9,
            max_iter=50000,
            eval_interval=300,
        )
        print(f"[{time.time():.1f}] after multimax best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # (7) Simulated‑annealing refinement
    remaining = deadline - time.time()
    if remaining > 15.0:
        sub_deadline = time.time() + remaining
        overall_best_seq, overall_best_score = _annealing_optimize(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
            max_iter=30000,
        )
        print(f"[{time.time():.1f}] after annealing best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # (8) Final greedy pairwise‑swap walk
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] after final swaps best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
